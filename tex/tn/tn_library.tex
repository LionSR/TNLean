% Universal reusable tensor-network constructions for TNLean.
%
% The commands in this file compose tn_core.tex into standard local tensors,
% chains, closures, and finite patches.  A document loads the core before this
% file and then builds all complete diagrams from these common atoms.

\makeatletter

% ---------- Standard tensor-network sites ----------

% Canonical port-bearing atoms for the semantic box and insertion classes.
% Their signatures are fixed here so complete diagrams never reconstruct the
% same boundary interface by hand.
\newcommand{\TN@vcomponent}[3]{%
    \TN@component{#1}{#2}{#3}%
    \TN@horizontalports{#1}%
}

\newcommand{\TN@vstate}[3]{%
    \TN@state{#1}{#2}{#3}%
    \TN@horizontalports{#1}%
}

\newcommand{\TN@vexpression}[3]{%
    \TN@expression{#1}{#2}{#3}%
    \TN@horizontalports{#1}%
}

\newcommand{\TN@vinsertion}[3]{%
    \TN@labeledinsertion{#1}{#2}{#3}%
    \TN@horizontalports{#1}%
}

\newcommand{\TN@pinsertion}[3]{%
    \TN@labeledinsertion{#1}{#2}{#3}%
    \TN@pverticalports{#1}%
}

\newcommand{\TN@vfactor}[3]{%
    \TN@factor{#1}{#2}{#3}%
    \TN@vverticalports{#1}%
}

\newcommand{\TN@factorwithlegs}[3]{%
    \TN@factor[minimum width=0.72cm, minimum height=0.52cm,
        inner sep=1pt]{#1}{#2}{#3}%
    \TN@mpoports{#1}%
    \TN@popenport{#1N}{0,0.26}%
    \TN@popenport{#1S}{0,-0.26}%
}

\newcommand{\TN@vjunctionatom}[2]{%
    \TN@junction{#1}{#2}%
    \TN@cardinalports{#1}%
}

% An operator or density matrix has paired system ports on each side.  The
% state glyph indicates that the internal tensor network has been suppressed.
\newcommand{\TN@operatorstate}[3]{%
    \TN@state{#1}{#2}{#3}%
    \TN@vwestport{#1WUp}{#1}{0.32}%
    \TN@vwestport{#1WDown}{#1}{0.68}%
    \TN@veastport{#1EUp}{#1}{0.32}%
    \TN@veastport{#1EDown}{#1}{0.68}%
}

% An MPS tensor has west and east virtual ports and one north physical port.
% The atom itself carries no open legs, so it may be composed without erasing
% pre-drawn index lines.
\newcommand{\TN@mpssite}[3]{%
    \TN@tensor{#1}{#2}%
    \TN@horizontalports{#1}%
    \TN@pport{#1N}{#1}{north}%
    \if\relax\detokenize{#3}\relax\else
        \node at ($(#1.south)+(0,-0.28)$) {$#3$};%
    \fi
}

\newcommand{\TN@labeledmpssite}[3]{%
    \TN@labeledtensor{#1}{#2}{#3}%
    \TN@horizontalports{#1}%
    \TN@pport{#1N}{#1}{north}%
}

% The corresponding standalone motif has named ports at all three free ends.
\newcommand{\TN@mpssiteWithOpenLegs}[3]{%
    \TN@mpssite{#1}{#2}{#3}%
    \TN@vopenport{#1W}{-0.60,0}%
    \TN@vopenport{#1E}{0.60,0}%
    \TN@popenport{#1N}{0,0.46}%
}

% An MPO or MPDO tensor has west and east virtual ports and north and south
% physical ports.
\newcommand{\TN@mposite}[3]{%
    \TN@tensor{#1}{#2}%
    \TN@mpoports{#1}%
    \if\relax\detokenize{#3}\relax\else
        \node[anchor=west] at ($(#1.south east)+(0.12,-0.18)$) {$#3$};%
    \fi
}

\newcommand{\TN@labeledmposite}[3]{%
    \TN@labeledtensor{#1}{#2}{#3}%
    \TN@mpoports{#1}%
}

% The standalone MPO/MPDO motif has named ports at all four free ends.
\newcommand{\TN@mpositeWithOpenLegs}[3]{%
    \TN@mposite{#1}{#2}{#3}%
    \TN@vopenport{#1W}{-0.60,0}%
    \TN@vopenport{#1E}{0.60,0}%
    \TN@popenport{#1N}{0,0.46}%
    \TN@popenport{#1S}{0,-0.46}%
}

% A PEPS vertex has a named north-east physical port.  Lattice-specific
% virtual ports are declared by the surrounding graph, so the physical port
% remains distinct from the virtual north direction.
\newcommand{\TN@pepsvertex}[3]{%
    \TN@tensor{#1}{#2}%
    \TN@pport{#1P}{#1}{45}%
    \TN@popenport{#1P}{0.38,0.38}%
    \if\relax\detokenize{#3}\relax\else
        \node[anchor=north east] at ($(#1.south west)+(-0.10,-0.10)$) {$#3$};%
    \fi
}

% A square-lattice PEPS tensor has four cardinal virtual indices in addition
% to the north-east physical index.  Its five named boundary ports form the
% compositional interface; the base atom has no fixed open stubs.
\newcommand{\TN@pepssite}[3]{%
    \TN@tensor{#1}{#2}%
    \TN@cardinalports{#1}%
    \TN@pport{#1P}{#1}{45}%
    \if\relax\detokenize{#3}\relax\else
        \node[anchor=north east] at ($(#1.south west)+(-0.10,-0.10)$) {$#3$};%
    \fi
}

% The standalone square-lattice site has named ports at all five free ends.
\newcommand{\TN@pepssiteWithOpenLegs}[3]{%
    \TN@pepssite{#1}{#2}{#3}%
    \TN@vopenport{#1W}{-0.58,0}%
    \TN@vopenport{#1E}{0.58,0}%
    \TN@vopenport{#1N}{0,0.58}%
    \TN@vopenport{#1S}{0,-0.58}%
    \TN@popenport{#1P}{0.38,0.38}%
}

% A doubled local tensor with its physical index contracted between the two
% layers.  The prefix names the upper and lower tensor nodes.
\newcommand{\TN@doublelayer}[3]{%
    \TN@tensor{#1U}{#2}%
    \TN@tensor{#1D}{#3}%
    \TN@mpoports{#1U}%
    \TN@mpoports{#1D}%
    \TN@pconnectports{#1US}{#1DN}%
}

% The standalone doubled layer exposes named ports at its four virtual ends.
\newcommand{\TN@doublelayerWithOpenLegs}[3]{%
    \TN@doublelayer{#1}{#2}{#3}%
    \TN@vopenport{#1UW}{-0.60,0}%
    \TN@vopenport{#1UE}{0.60,0}%
    \TN@vopenport{#1DW}{-0.60,0}%
    \TN@vopenport{#1DE}{0.60,0}%
}

% Two displayed tensor-product factors.  Tensor product is written explicitly;
% it is never represented by an index line.
\newcommand{\TN@factorpair}[5]{%
    \TN@factor{#1L}{#2}{#4}%
    \TN@factor{#1R}{#3}{#5}%
    \node at ($(#1L.center)!0.5!(#1R.center)$) {$\otimes$};%
}

% ---------- Oriented fusion maps ----------

% A left-to-right fusion map with one trunk and two ordered branches.  The
% atom retains the semantic rounded blue map glyph; its interface is named by
% role rather than by absolute coordinates.  Top and Bottom are available for
% additional physical indices when the same map participates in a larger
% contraction.
\newcommand{\TN@splitmap}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@vport{#2Mid}{#2}{west}%
    \TN@vport{#2Up}{#2}{north east}%
    \TN@vport{#2Down}{#2}{south east}%
    \TN@pport{#2Top}{#2}{north}%
    \TN@pport{#2Bottom}{#2}{south}%
}

% The mirror image of \TN@splitmap: two ordered branches merge into one
% trunk.  The port names have the same mathematical roles in both
% orientations, so a split and a merge may be interchanged without exposing
% TikZ anchors to the surrounding construction.
\newcommand{\TN@mergemap}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@vport{#2Mid}{#2}{east}%
    \TN@vport{#2Up}{#2}{north west}%
    \TN@vport{#2Down}{#2}{south west}%
    \TN@pport{#2Top}{#2}{north}%
    \TN@pport{#2Bottom}{#2}{south}%
}

% A vertical fusion map has two ordered incoming sector ports and one outgoing
% sector port.  The inverse orientation uses the same role names, so fusion
% trees may be assembled without referring to TikZ anchors.
\newcommand{\TN@fusionmap}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@vport{#2Left}{#2}{north west}%
    \TN@vport{#2Right}{#2}{north east}%
    \TN@vport{#2Out}{#2}{south}%
}

\newcommand{\TN@cofusionmap}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@vport{#2In}{#2}{north}%
    \TN@vport{#2Left}{#2}{south west}%
    \TN@vport{#2Right}{#2}{south east}%
}

% A Stinespring isometry carries a system index from west to east and one
% environment index.  The two orientations expose the environment above and
% below, respectively.
\newcommand{\TN@systemancillamapup}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@horizontalports{#2}%
    \TN@vport{#2Ancilla}{#2}{north}%
}

\newcommand{\TN@systemancillamapdown}[4][]{%
    \TN@map[#1]{#2}{#3}{#4}%
    \TN@horizontalports{#2}%
    \TN@vport{#2Ancilla}{#2}{south}%
}

% An irregular graph edge obtains a typed port at each chosen boundary anchor
% before composing the two atoms.
\newcommand{\TN@vedge}[5]{%
    \TN@vport{#1A}{#2}{#3}%
    \TN@vport{#1B}{#4}{#5}%
    \TN@vconnectports{#1A}{#1B}%

% A lattice site is a genuine four-port tensor atom.  The generic patch below
% places named atoms, joins adjacent boundary ports, and never draws a line
% through the centre of a tensor glyph.
\newcount\TN@normalgridserial

\newcommand{\TN@latticesite}[2]{%
    \TN@tensor{#1}{#2}%
    \TN@cardinalports{#1}%
}

\newcommand{\TN@plainpepssite}[2]{%
    \TN@pepssite{#1}{#2}{}%
    \TN@popenport{#1P}{0.38,0.38}%
}

\newcommand{\TN@squaregridatoms}[5]{%
    \foreach \i in {0,...,#2} { % chktex 11
        \foreach \j in {0,...,#3} { % chktex 11
            \pgfmathsetmacro{\TN@gridx}{#4*\i}%
            \pgfmathsetmacro{\TN@gridy}{#4*\j}%
            #5{#1\i\j}{(\TN@gridx,\TN@gridy)}%
        }
    }%
}

\newcommand{\TN@squaregridedges}[4]{%
    \pgfmathtruncatemacro{\TN@gridlastx}{#2-1}%
    \pgfmathtruncatemacro{\TN@gridlasty}{#3-1}%
    \foreach \i in {0,...,\TN@gridlastx} { % chktex 11
        \pgfmathtruncatemacro{\TN@gridnext}{\i+1}%
        \foreach \j in {0,...,#3} { % chktex 11
            \TN@vconnectports{#1\i\j E}{#1\TN@gridnext\j W}%
        }
    }%
    \foreach \i in {0,...,#2} { % chktex 11
        \foreach \j in {0,...,\TN@gridlasty} { % chktex 11
            \pgfmathtruncatemacro{\TN@gridnext}{\j+1}%
            \TN@vconnectports{#1\i\j N}{#1\i\TN@gridnext S}%
        }
    }%
}

\newcommand{\TN@squarelatticepatch}[4]{%
    \TN@squaregridatoms{#1}{#2}{#3}{#4}{\TN@latticesite}%
    \TN@squaregridedges{#1}{#2}{#3}{#4}%
}

\newcommand{\TN@squarepepspatch}[4]{%
    \TN@squaregridatoms{#1}{#2}{#3}{#4}{\TN@plainpepssite}%
    \TN@squaregridedges{#1}{#2}{#3}{#4}%
}

\newcommand{\TN@normalPepsGrid}{%
    \global\advance\TN@normalgridserial by 1\relax
    \edef\TN@normalgridprefix{normalGrid\the\TN@normalgridserial}%
    \TN@squarelatticepatch{\TN@normalgridprefix}{6}{6}{0.5}%
    \foreach \j in {0,...,6} { % chktex 11
        \TN@vopenport{\TN@normalgridprefix0\j W}{-1,0}%
        \TN@vopenport{\TN@normalgridprefix6\j E}{1,0}%
    }%
    \foreach \i in {0,...,6} { % chktex 11
        \TN@vopenport{\TN@normalgridprefix\i0S}{0,-1}%
        \TN@vopenport{\TN@normalgridprefix\i6N}{0,1}%
    }%
}
