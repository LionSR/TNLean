% Shared tensor-network constructions for symmetry and parent-Hamiltonian diagrams.
%
% These motifs own the fixed local geometry of transfer cells and string-order
% words.  Catalogue declarations supply only mathematical labels and relative
% placement relations.

\makeatletter

% The existing trace-cell and vertical-word constructions use a centre point.
% These wrappers give catalogue terms a coordinate-free origin convention.
\newcommand{\TNSymmetryTraceCell}[4]{%
    \TNEventInternal{motif|picture=\the\TN@pictureid|name=#1|kind=trace-cell}%
    \TNCompactTraceCell{#1}{(0,0)}{#2}{#3}{#4}%
}

\newcommand{\TNSymmetryVerticalWord}[3]{%
    \TNEventInternal{motif|picture=\the\TN@pictureid|name=#1|kind=vertical-word}%
    \TNVerticalWord{#1}{(0,0)}{#2}{#3}%
}

% One twisted transfer cell: two tensor layers meet at the physical insertion.
% The remaining arguments name the twist, input, output, and layer indices.
\newcommand{\TNSymmetryTransferCell}[7]{%
    \TNEventInternal{motif|picture=\the\TN@pictureid|name=#1|kind=transfer-cell}%
    \TNInsertion{#1Twist}{#2}{}%
    \TNTensor{#1Upper}{above=\TN@layerpitch cm of #1Twist}% chktex 1
    \TNTensor{#1Lower}{below=\TN@layerpitch cm of #1Twist}% chktex 1
    \TNVirtualPort{#1UpperWest}{#1Upper}{west}%
    \TNVirtualPort{#1UpperEast}{#1Upper}{east}%
    \TNVirtualPort{#1LowerWest}{#1Lower}{west}%
    \TNVirtualPort{#1LowerEast}{#1Lower}{east}%
    \TNOpenVirtualWest{#1UpperWest}%
    \TNOpenVirtualWest{#1LowerWest}%
    \TNOpenVirtualEast{#1UpperEast}%
    \TNOpenVirtualEast{#1LowerEast}%
    \TNPhysicalPort{#1UpperPhysical}{#1Upper}{south}%
    \TNPhysicalPort{#1TwistUpper}{#1Twist}{north}%
    \TNPhysicalPort{#1TwistLower}{#1Twist}{south}%
    \TNPhysicalPort{#1LowerPhysical}{#1Lower}{north}%
    \TNConnectPhysical{#1UpperPhysical}{#1TwistUpper}%
    \TNConnectPhysical{#1TwistLower}{#1LowerPhysical}%
    \TNLabelAbove{#1Twist}{#3}%
    \TNLabelLeft{#1Twist}{#4}%
    \TNLabelRight{#1Twist}{#5}%
    \TNLabelAbove{#1Upper}{#6}%
    \TNLabelBelow{#1Lower}{#7}%
}

% A local cell in a string-order word has the same vertical contraction but
% leaves its horizontal ports available to the surrounding word.
\newcommand{\TN@stringcell}[3]{%
    \TNEventInternal{motif|picture=\the\TN@pictureid|name=#1|kind=string-cell}%
    \TNInsertion{#1Twist}{#2}{}%
    \TNTensor{#1Upper}{above=\TN@layerpitch cm of #1Twist}% chktex 1
    \TNTensor{#1Lower}{below=\TN@layerpitch cm of #1Twist}% chktex 1
    \TNVirtualPort{#1UpperWest}{#1Upper}{west}%
    \TNVirtualPort{#1UpperEast}{#1Upper}{east}%
    \TNVirtualPort{#1LowerWest}{#1Lower}{west}%
    \TNVirtualPort{#1LowerEast}{#1Lower}{east}%
    \TNPhysicalPort{#1UpperPhysical}{#1Upper}{south}%
    \TNPhysicalPort{#1TwistUpper}{#1Twist}{north}%
    \TNPhysicalPort{#1TwistLower}{#1Twist}{south}%
    \TNPhysicalPort{#1LowerPhysical}{#1Lower}{north}%
    \TNConnectPhysical{#1UpperPhysical}{#1TwistUpper}%
    \TNConnectPhysical{#1TwistLower}{#1LowerPhysical}%
    \TNLabelRight{#1Twist}{#3}%
}

% The string-order word consists of two displayed cells, an omitted middle
% column, and the final traced cell.  Its topology is fixed independently of L.
\newcommand{\TNSymmetryStringOrderWord}[3]{%
    \TNEventInternal{motif|picture=\the\TN@pictureid|name=#1|kind=string-order-word}%
    \TNTensor{#1Weight}{at=origin}%
    \TNLabelBelow{#1Weight}{\Lambda}%
    \TN@stringcell{#1First}{right=of #1Weight}{#2}%
    \TN@stringcell{#1Second}
        {right=\TN@chainsep cm of #1FirstTwist}{#2}% chktex 1
    \TN@stringcell{#1Last}
        {right=\TN@chainright cm of #1FirstTwist}{#2}% chktex 1
    \TNVirtualPort{#1WeightUpper}{#1Weight}{east}%
    \TNVirtualPort{#1WeightLower}{#1Weight}{east}%
    \TNConnectVirtual{#1WeightUpper}{#1FirstUpperWest}%
    \TNConnectVirtual{#1WeightLower}{#1FirstLowerWest}%
    \TNConnectVirtual{#1FirstUpperEast}{#1SecondUpperWest}%
    \TNConnectVirtual{#1FirstLowerEast}{#1SecondLowerWest}%
    \TNOpenVirtualEast{#1SecondUpperEast}%
    \TNOpenVirtualEast{#1SecondLowerEast}%
    \TNOpenVirtualWest{#1LastUpperWest}%
    \TNOpenVirtualWest{#1LastLowerWest}%
    \TNTraceVirtualRight{#1LastUpperEast}{#1LastLowerEast}%
    \TNOmission{#1UpperOmission}
        {right=\TN@cellpitch cm of #1SecondUpper}% chktex 1
    \TNOmission{#1LowerOmission}
        {right=\TN@cellpitch cm of #1SecondLower}% chktex 1
    \TNLabelBelow{#1LowerOmission}{#3}%
    \TNLabelLeft{#1LastTwist}{#2}%
    \TNLabelRight{#1LastTwist}{\tr}%
}

\makeatother
