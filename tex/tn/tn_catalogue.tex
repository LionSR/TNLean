% Catalogue of chapter-facing tensor-network diagrams.
%
% Each declaration contains the mathematical command, its argument schema,
% display role, layout profile, audit sample, and chapter context.  The
% repository-wide TikZ library loads this file after the tensor-network core
% and motif library.

\ExplSyntaxOn

\cs_if_exist:NT \TNDeclareDiagram
  {
    \PackageError { tn }
      { The~tensor-network~diagram~catalogue~was~loaded~more~than~once }
      { Load~the~TikZ~library~tn~once;~it~loads~the~catalogue~itself. }
    \ExplSyntaxOff
    \tex_endinput:D
  }
\cs_if_exist:NF \TNMPSSite
  {
    \PackageError { tn }
      { The~tensor-network~diagram~catalogue~was~loaded~before~its~library }
      { Load~tn_core.tex~and~tn_library.tex~before~tn_catalogue.tex. }
    \ExplSyntaxOff
    \tex_endinput:D
  }

\seq_new:N \g_tn_diagram_names_seq
\prop_new:N \g_tn_diagram_arguments_prop
\prop_new:N \g_tn_diagram_roles_prop
\prop_new:N \g_tn_diagram_profiles_prop
\prop_new:N \g_tn_diagram_samples_prop
\prop_new:N \g_tn_diagram_contexts_prop

\msg_new:nnn { tn } { invalid-diagram-name }
    { The~diagram~name~'#1'~must~have~the~form~TNName. }
\msg_new:nnn { tn } { duplicate-diagram }
    { The~diagram~'#1'~has~already~been~declared. }
\msg_new:nnn { tn } { invalid-argument-name }
    { The~argument~name~'#1'~must~be~lower-case~snake~case. }
\msg_new:nnn { tn } { duplicate-argument }
    { The~argument~'#1'~occurs~twice~in~one~diagram~schema. }
\msg_new:nnn { tn } { invalid-diagram-role }
    { The~diagram~role~'#1'~must~be~'display'~or~'figure'. }
\msg_new:nnn { tn } { invalid-diagram-profile }
    { The~diagram~profile~'#1'~must~be~'compact'~or~'normal'. }
\msg_new:nnn { tn } { empty-diagram-metadata }
    { The~diagram~'#1'~has~an~empty~#2~field. }

\seq_new:N \l_tn_diagram_arguments_seq
\tl_new:N \l_tn_diagram_signature_tl

\cs_new_protected:Npn \tn_declare_diagram:nnnnnnn #1#2#3#4#5#6#7
  {
    \regex_match:nnF { \A TN [A-Za-z]+ \Z } {#1}
      { \msg_error:nnn { tn } { invalid-diagram-name } {#1} }
    \seq_if_in:NnT \g_tn_diagram_names_seq {#1}
      { \msg_error:nnn { tn } { duplicate-diagram } {#1} }
    \seq_clear:N \l_tn_diagram_arguments_seq
    \clist_map_inline:nn {#2}
      {
        \regex_match:nnF { \A [a-z] [a-z0-9_]* \Z } {##1}
          { \msg_error:nnn { tn } { invalid-argument-name } {##1} }
        \seq_if_in:NnT \l_tn_diagram_arguments_seq {##1}
          { \msg_error:nnn { tn } { duplicate-argument } {##1} }
        \seq_put_right:Nn \l_tn_diagram_arguments_seq {##1}
      }
    \str_case:nnF {#3}
      { {display} {} {figure} {} }
      { \msg_error:nnn { tn } { invalid-diagram-role } {#3} }
    \str_case:nnF {#4}
      { {compact} {} {normal} {} }
      { \msg_error:nnn { tn } { invalid-diagram-profile } {#4} }
    \tl_if_blank:nT {#5}
      { \msg_error:nnnn { tn } { empty-diagram-metadata } {#1} {sample} }
    \tl_if_blank:nT {#6}
      { \msg_error:nnnn { tn } { empty-diagram-metadata } {#1} {context} }
    \tl_set:Nx \l_tn_diagram_signature_tl
      { \prg_replicate:nn { \clist_count:n {#2} } {m} }
    \exp_args:NcV \NewDocumentCommand {#1} \l_tn_diagram_signature_tl
      {
        \TNEventInternal
          {diagram|name=#1|role=#3|profile=#4|context=\tl_to_str:n{#6}}
        #7
      }
    \seq_gput_right:Nn \g_tn_diagram_names_seq {#1}
    \prop_gput:Nnn \g_tn_diagram_arguments_prop {#1} {#2}
    \prop_gput:Nnn \g_tn_diagram_roles_prop {#1} {#3}
    \prop_gput:Nnn \g_tn_diagram_profiles_prop {#1} {#4}
    \prop_gput:Nnn \g_tn_diagram_samples_prop {#1} {#5}
    \prop_gput:Nnn \g_tn_diagram_contexts_prop {#1} {#6}
  }

\NewDocumentCommand \TNDeclareDiagram { m m m m m m +m }
  { \tn_declare_diagram:nnnnnnn {#1} {#2} {#3} {#4} {#5} {#6} {#7} }

\ExplSyntaxOff
