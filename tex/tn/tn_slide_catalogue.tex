% Complete tensor-network diagrams used by TNLean's slide collection.
%
% This catalogue contains mathematical constructions only.  Geometry,
% typography, port meanings, and colour themes belong to the shared `tn' TikZ
% library.  The commands below form the sole interface used by the decks.

\makeatletter

% A common periodic MPS word.  Omitted sites are represented by two typed
% external terminals and an ellipsis; the trace is a port-to-port closure.
\newcommand{\SlideTNPeriodicMPS}[2]{%
  \begin{TNDiagram}[normal]
    \TNCyclicMPSWord{sltnWord}{#1}{#2}
      {\text{trace; bond dimension }D}
  \end{TNDiagram}%
}

% The left-hand tensor is B, matching B^i = X A^i X^{-1}.
\newcommand{\SlideTNGaugeConjugation}[3]{%
  \begin{TNEquationRow}[normal]
    \TNTerm[normal]{%
      \TNMPSSite{sltnB}{at=origin}{#1}
      \TNOpenVirtualWest{sltnBWest}
      \TNOpenVirtualEast{sltnBEast}
      \TNOpenPhysicalNorth{sltnBKet}
      \TNLabelAbove{sltnBKet}{i}%
    }%
    \TNRelation{=}%
    \TNTerm[normal]{%
      \TNGaugeConjugatedMPSSite{sltnGauge}{at=origin}
        {#2}{#3}{#3^{-1}}%
      \TNLabelAbove{sltnGaugeSiteKet}{i}%
    }%
  \end{TNEquationRow}%
}

\newcommand{\SlideTNBlockingIdentity}[3]{%
  \begin{TNEquationRow}[compact]
    \TNTerm{%
      \TNMPSSite{sltnC}{at=origin}{#1}
      \TNOpenVirtualWest{sltnCWest}
      \TNOpenVirtualEast{sltnCEast}
      \TNOpenPhysicalNorth{sltnCKet}
      \TNLabelAbove{sltnCKet}{(i_1,\ldots,i_{#3})}%
    }%
    \TNRelation{=}%
    \TNTerm{%
      \TNMPSSite{sltnBOne}{at=origin}{#2}
      \TNMPSSite{sltnBTwo}{right=of sltnBOne}{#2}
      \TNOmission{sltnOmitted}{right=of sltnBTwo}
      \TNMPSSite{sltnBLast}{right=of sltnOmitted}{#2}
      \foreach \name/\i in {One/i_1,Two/i_2,Last/i_{#3}} {
          \TNOpenPhysicalNorth{sltnB\name Ket}
          \TNLabelAbove{sltnB\name Ket}{\i}
      }
      \TNOpenVirtualWest{sltnBOneWest}
      \TNConnectVirtual{sltnBOneEast}{sltnBTwoWest}
      \TNOpenVirtualEast{sltnBTwoEast}
      \TNOpenVirtualWest{sltnBLastWest}
      \TNOpenVirtualEast{sltnBLastEast}
      \TNLabelBelow{sltnOmitted}{\text{#3 copies}}%
    }%
  \end{TNEquationRow}%
}

\newcommand{\SlideTNMixedTransfer}[4]{%
  \begin{TNDiagram}[compact]
    \TNStackedMPOProduct{sltnOne}{at=origin}{#1}{#2^\dagger}{}
    \TNStackedMPOProduct{sltnTwo}{right=of sltnOneUpper}
      {#1}{#2^\dagger}{}
    \TNStackedOmission{sltnOmitted}{right=of sltnTwoUpper}
    \TNStackedMPOProduct{sltnPen}{right=of sltnOmittedUpper}
      {#1}{#2^\dagger}{}
    \TNStackedMPOProduct{sltnLast}{right=of sltnPenUpper}
      {#1}{#2^\dagger}{}
    \TNOpenVirtualWest{sltnOneUpperWest}
    \TNOpenVirtualWest{sltnOneLowerWest}
    \TNConnectVirtual{sltnOneUpperEast}{sltnTwoUpperWest}
    \TNConnectVirtual{sltnOneLowerEast}{sltnTwoLowerWest}
    \TNOpenVirtualEast{sltnTwoUpperEast}
    \TNOpenVirtualEast{sltnTwoLowerEast}
    \TNOpenVirtualWest{sltnPenUpperWest}
    \TNOpenVirtualWest{sltnPenLowerWest}
    \TNConnectVirtual{sltnPenUpperEast}{sltnLastUpperWest}
    \TNConnectVirtual{sltnPenLowerEast}{sltnLastLowerWest}
    \TNOpenVirtualEast{sltnLastUpperEast}
    \TNOpenVirtualEast{sltnLastLowerEast}
    \TNLabelLeft{sltnOneCentre}{#3}
    \TNLabelRight{sltnLastCentre}{\mathcal F^{#4}_{#1,#2}(#3)}
  \end{TNDiagram}%
}

\newcommand{\SlideTNTransferMap}[2]{%
  \begin{TNDiagram}[normal]
    \TNStackedMPOProduct{sltn}{at=origin}{#1}{\overline{#1}}{}
    \TNOpenVirtualWest{sltnUpperWest}
    \TNOpenVirtualWest{sltnLowerWest}
    \TNOpenVirtualEast{sltnUpperEast}
    \TNOpenVirtualEast{sltnLowerEast}
    \TNLabelLeft{sltnCentre}{#2}
    \TNLabelRight{sltnUpperBra}{\sum_i}
    \TNLabelRight{sltnCentre}{\mathcal E_{#1}(#2)}
  \end{TNDiagram}%
}

\newcommand{\SlideTNBlockingComparison}[2]{%
  \begin{TNDiagram}[compact]
    \TNMPSSite{sltnPOne}{at=origin}{#1}
    \TNMPSSite{sltnPTwo}{right=of sltnPOne}{#1}
    \TNMPSSite{sltnPThree}{right=of sltnPTwo}{#1}
    \TNOmission{sltnOmitted}{right=of sltnPThree}
    \TNMPSSite{sltnPPen}{right=of sltnOmitted}{#1}
    \TNMPSSite{sltnPLast}{right=of sltnPPen}{#1}
    \foreach \name in {One,Two,Three,Pen,Last} {
        \TNOpenPhysicalNorth{sltnP\name Ket}
    }
    \TNConnectVirtual{sltnPOneEast}{sltnPTwoWest}
    \TNConnectVirtual{sltnPTwoEast}{sltnPThreeWest}
    \TNOpenVirtualEast{sltnPThreeEast}
    \TNOpenVirtualWest{sltnPPenWest}
    \TNConnectVirtual{sltnPPenEast}{sltnPLastWest}
    \TNOpenVirtualWest{sltnPOneWest}
    \TNOpenVirtualEast{sltnPLastEast}
    \TNLabelLeft{sltnPOne}{\text{chain of }N\text{:}}
    \TNMPSSite{sltnBlockOne}{below=of sltnPOne}{#1^{(#2)}}
    \TNMPSSite{sltnBlockTwo}{right=of sltnBlockOne}{#1^{(#2)}}
    \TNMPSSite{sltnBlockLast}{right=of sltnBlockTwo}{#1^{(#2)}}
    \TNLabelLeft{sltnBlockOne}{\text{after blocking }#2\text{:}}
    \foreach \name in {One,Two,Last} {
      \TNOpenPhysicalNorth{sltnBlock\name Ket}
    }
    \TNConnectVirtual{sltnBlockOneEast}{sltnBlockTwoWest}
    \TNConnectVirtual{sltnBlockTwoEast}{sltnBlockLastWest}
    \TNOpenVirtualWest{sltnBlockOneWest}
    \TNOpenVirtualEast{sltnBlockLastEast}
  \end{TNDiagram}%
}

\makeatother
