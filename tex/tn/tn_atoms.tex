% Semantic atom catalogue for TNLean tensor-network diagrams.

\ExplSyntaxOn
\seq_new:N \g__tn_atom_names_seq
\prop_new:N \g__tn_atom_ports_prop
\prop_new:N \g__tn_atom_profile_prop
\prop_new:N \g__tn_atom_sample_prop

\cs_new_protected:Npn \tn_declare_atom:nnnn #1#2#3#4
  {
    \seq_if_in:NnTF \g__tn_atom_names_seq {#1}
      {
        \PackageError{tnlean-tikz}{Duplicate semantic atom `#1'}
          {Every atom and oriented trivalent map must be declared once.}
      }
      {
        \seq_gput_right:Nn \g__tn_atom_names_seq {#1}
        \prop_gput:Nnn \g__tn_atom_ports_prop {#1} {#2}
        \prop_gput:Nnn \g__tn_atom_profile_prop {#1} {#3}
        \prop_gput:Nnn \g__tn_atom_sample_prop {#1} {#4}
      }
  }
\cs_new_eq:NN \TNDeclareAtom \tn_declare_atom:nnnn
\ExplSyntaxOff
