% Input: public affine frame specifications from picture, group, and atom keys.
% Output: normalized rotation angles, model frame records, and frame events.
% Owned state: parser scratch only; callers own their scope-local angle state.
% Invariants: labels remain untransformed; every accepted transform is logged.
% Next stage: front ends apply the normalized map to coordinates or glyph ink.

\ExplSyntaxOn

\msg_new:nnnn {tenkz}{frame-spec}
  { [TKZ-FRAME-SPEC]~unknown~affine~frame~'#1' }
  { Use~flat~or~rotate=<degrees>. }

\tl_new:N \l__tenkz_frame_parse_tl
\seq_new:N \l__tenkz_frame_match_seq
\tl_new:N \l__tenkz_frame_picture_angle_tl
\tl_new:N \l__tenkz_frame_group_angle_tl
\tl_new:N \l__tenkz_frame_local_angle_tl
\tl_new:N \l__tenkz_frame_total_angle_tl
\tl_new:N \l__tenkz_frame_glyph_options_tl

\cs_new_protected:Npn \__tenkz_frame_parse_rotation:nN #1#2
  {
    \tl_set:Nn \l__tenkz_frame_parse_tl {#1}
    \tl_trim_spaces:N \l__tenkz_frame_parse_tl
    \str_if_eq:VnTF \l__tenkz_frame_parse_tl {flat}
      { \tl_set:Nn #2 {0} }
      {
        \regex_extract_once:nVNTF
          {
            \A \s* rotate \s* = \s*
            ( [-+]? (?: [0-9]+ (?: \. [0-9]* )? | \. [0-9]+ ) )
            \s* \Z
          }
          \l__tenkz_frame_parse_tl
          \l__tenkz_frame_match_seq
          {
            \tl_set:Ne #2
              {
                \fp_eval:n
                  {
                    \seq_item:Nn \l__tenkz_frame_match_seq {2}
                    - 360 * floor(
                        \seq_item:Nn \l__tenkz_frame_match_seq {2} / 360)
                  }
              }
          }
          {
            \msg_error:nnV {tenkz}{frame-spec}
              \l__tenkz_frame_parse_tl
          }
      }
  }

\cs_new:Npn \__tenkz_frame_map:n #1 { rotate(#1) }

\cs_new_protected:Npn \__tenkz_frame_prepare_atom:n #1
  {
    \tl_set:Ne \l__tenkz_frame_total_angle_tl
      {
        \fp_eval:n
          {
            \l__tenkz_frame_picture_angle_tl
            + \l__tenkz_frame_group_angle_tl
            + (#1)
          }
      }
    \tl_clear:N \l__tenkz_frame_glyph_options_tl
    \fp_compare:nNnF {\l__tenkz_frame_total_angle_tl} = {0}
      {
        \tl_set:Ne \l__tenkz_frame_glyph_options_tl
          {, tenkz~glyph~frame={#1}}
      }
  }

\cs_new_protected:Npn \__tenkz_frame_upright:n #1
  {
    \tl_if_empty:NTF \l__tenkz_frame_glyph_options_tl
      {#1}
      {\rotatebox{\fp_eval:n {-\l__tenkz_frame_total_angle_tl}}{#1}}
  }

\cs_new_protected:Npn \__tenkz_frame_record:nn #1#2
  {
    \use:e
      {
        \exp_not:N \__tenkz_model_record_frame:n
          {map={\__tenkz_frame_map:n {#2}},scope={#1}}
      }
  }

\cs_new_protected:Npn \__tenkz_frame_event:nn #1#2
  {
    \tenkz@event{frame|picture=\the\tenkz@pictureid|scope=#1|
      map=\__tenkz_frame_map:n {#2}|
      a=\fp_eval:n {round(cosd(#2),6)}|
      b=\fp_eval:n {round(-sind(#2),6)}|
      c=\fp_eval:n {round(sind(#2),6)}|
      d=\fp_eval:n {round(cosd(#2),6)}}
  }

\pgfkeys{
  /tenkz/group/.is~family,
  /tenkz/group/frame/.code=
    {\__tenkz_frame_parse_rotation:nN {#1} \l__tenkz_frame_local_angle_tl},
}
\pgfkeysdef{/tenkz/group/.unknown}
  {
    \msg_error:nnn {tenkz}{language-value}
      {group~key~'\pgfkeyscurrentname'}
  }

% A group is a coordinate transform, not a TeX visibility boundary: TikZ node
% names created in the scope remain available to later joins.
\NewDocumentCommand \tngroup { O{} +m }
  {
    \group_begin:
    \tl_set:Nn \l__tenkz_frame_local_angle_tl {0}
    \tl_if_empty:nF {#1} { \pgfqkeys{/tenkz/group}{#1} }
    \tl_set:Ne \l__tenkz_frame_group_angle_tl
      {
        \fp_eval:n
          {
            \l__tenkz_frame_group_angle_tl
            + \l__tenkz_frame_local_angle_tl
          }
      }
    \fp_compare:nNnF {\l__tenkz_frame_local_angle_tl} = {0}
      {
        \__tenkz_frame_record:nn {group} {\l__tenkz_frame_local_angle_tl}
        \__tenkz_frame_event:nn {group} {\l__tenkz_frame_local_angle_tl}
      }
    \begin{scope}[rotate=\l__tenkz_frame_local_angle_tl]
      #2
    \end{scope}
    \group_end:
  }

\ExplSyntaxOff
