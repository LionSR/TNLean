% Commutative diagrams are tikz-cd material, outside the tenkz language.
% The channel style keeps the ink these arrows carried as tenkz typed
% maps: a headless stroke of wire weight in the marked hue, its label
% seated on the stroke over paper.  Both the print preamble and the
% standalone web picture units read this file in the preamble after
% loading tenkz, which supplies the marked hue; tikz-cd itself is loaded
% here, since the tenkz package no longer carries it.
\usepackage{tikz-cd}
\tikzcdset{
    channel/.style={
            every arrow/.append style={draw=tenkzMarked, line width=0.55pt,
                    no head},
            every label/.append style={description}},
}

% Quantum circuits are drawn with quantikz (TikZ library quantikz2), outside
% the tenkz language. Both the print preamble and the standalone web picture
% units read this file, so a quantikz body compiles the same way in each.
\usetikzlibrary{quantikz2}
