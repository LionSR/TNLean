% (lem:columns: "Fixing the input basis index separately in each
% physical tensor selects the same column without altering a virtual
% leg"); no source figure.
% Ink-to-index: each ellipse is a site tensor A_v of the operator
% network (named once, at the front-left vertex); the in-plane lines are the
% virtual bonds e with dimension D_e, identical in both pictures; the upward
% leg at v is the output index s_v, the downward leg the input index r_v
% (tagged once each on the right edge: out at the top of the back-right
% output leg, in at the bottom of the front-right input leg). In the right
% picture each downward leg ends in a dark dot, the basis vector |z_v>, so
% the input index is contracted to the value z_v; the dot on the leg of the
% middle site of the right column carries the generic label |z_v>. The input
% dots are basis states, not a party, so their species is redeclared to
% near-black (shared-spec H2). The names sigma and sigma|z> stand under the
% two pictures. The row skew is the plane frame's own foreshortening; tenkz
% exposes no key for it.
% Boundary signatures: left, nine open output and nine open input
% physical legs (the operator sigma); right, nine open output legs only
% (the column sigma|z>). No virtual leg is open in either picture.
% In the right panel each cell has two members: the operator tensor
% and its input basis vector. Their intra-cell physical wire is closed;
% only the operator member carries the twelve original virtual bonds
% between adjacent sites and the nine open output legs.
% Label macros: none.
\begin{tenkzequation}
    \tenkzkernel
    \begin{tenkz}[rows={op,op,op}, cols=3, frame=plane,
            physical=updown]
        \tnmark[form=label, label pos=w]{(3,1)}{$A_v$}
        \tnmark[form=label, label pos=e]{on leg-n-1-3 1}{$\mathrm{out}$}
        \tnmark[form=label, label pos=e]{on leg-s-3-3 1}{$\mathrm{in}$}
        \tnmark[form=label, label pos=s]{on leg-s-3-2 1}{$\sigma$}
    \end{tenkz}
    $\quad\longmapsto\quad$
    \begin{tenkz}[rows={op,op,op}, cols=3,
            frame={plane, basis={op at (0,0), ket at (1,-2)}}, bonds=none]
        \tndeclare{species}{input}{hue=source:black!80}
        \tn[at=(1,1,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(1,1,1)}{(1,1,2)}
        \tnwire{(1,1,1)}{open up}
        \tnwire{(1,1,1)}{(1,2,1)}
        \tnwire{(1,1,1)}{(2,1,1)}
        \tn[at=(1,2,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(1,2,1)}{(1,2,2)}
        \tnwire{(1,2,1)}{open up}
        \tnwire{(1,2,1)}{(1,3,1)}
        \tnwire{(1,2,1)}{(2,2,1)}
        \tn[at=(1,3,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(1,3,1)}{(1,3,2)}
        \tnwire{(1,3,1)}{open up}
        \tnwire{(1,3,1)}{(2,3,1)}
        \tn[at=(2,1,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(2,1,1)}{(2,1,2)}
        \tnwire{(2,1,1)}{open up}
        \tnwire{(2,1,1)}{(2,2,1)}
        \tnwire{(2,1,1)}{(3,1,1)}
        \tn[at=(2,2,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(2,2,1)}{(2,2,2)}
        \tnwire{(2,2,1)}{open up}
        \tnwire{(2,2,1)}{(2,3,1)}
        \tnwire{(2,2,1)}{(3,2,1)}
        \tn[at=(2,3,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(2,3,1)}{(2,3,2)}
        \tnwire{(2,3,1)}{open up}
        \tnwire{(2,3,1)}{(3,3,1)}
        \tn[at=(3,1,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(3,1,1)}{(3,1,2)}
        \tnwire{(3,1,1)}{open up}
        \tnwire{(3,1,1)}{(3,2,1)}
        \tn[at=(3,2,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(3,2,1)}{(3,2,2)}
        \tnwire{(3,2,1)}{open up}
        \tnwire{(3,2,1)}{(3,3,1)}
        \tn[at=(3,3,2), skin=dot, species=input, label pos=e]{}
        \tnwire{(3,3,1)}{(3,3,2)}
        \tnwire{(3,3,1)}{open up}
        \tnmark[form=label, label pos=e]{(2,3,2)}{$\ket{z_v}$}
        \tnmark[form=label, label pos=s]{(3,2,2)}{$\sigma\ket{z}$}
    \end{tenkz}
\end{tenkzequation}
