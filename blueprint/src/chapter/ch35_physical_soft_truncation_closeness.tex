\section{Compatible soft truncation and actual operator tensors}

\begin{lemma}[Weighted sums of operator chains]
    \label{lem:trace_chain_linear_combination}
    \lean{MPOTensor.exists_traceChainOperator_linearCombination_varying,
        MPOTensor.exists_traceChainOperator_linearCombination,
        MPOTensor.exists_traceChainOperator_add}
    \leanok
    On a nonempty physical chain, a finite weighted sum of
    site-dependent matrix product operators with virtual dimensions
    $D_a$ has actual operator tensors of dimension $\sum_aD_a$.
    The coefficients may be arbitrary complex numbers, and zero
    virtual dimensions are permitted. In particular, $s$ summands
    of common dimension $D$ give dimension $sD$, and adding two
    chains of dimensions $D,E$ gives dimension $D+E$.
\end{lemma}
\begin{proof}\leanok
    Put the site tensors on diagonal blocks. Their ordered product
    has the corresponding products on its diagonal blocks. Put the
    coefficients in the same blocks of the closing matrix. Taking
    the trace gives the required weighted sum. Reindex the virtual
    coordinates by a finite interval and absorb the closing matrix
    into the first site tensor.
\end{proof}

\begin{lemma}[Exact local sum and scalar]
    \label{lem:local_sum_scalar_mpo}
    \lean{TNLean.MPS.AreaLaw.exists_local_sum_add_scalar_mpo}
    \leanok
    \uses{thm:operator_chain_from_all_cut_ranks}
    Let $q\geq1$ and let $h_i$, $0\leq i<N$, act on the designated
    pairs $\{i,i+1\}$ of $N+1$ sites. For every scalar $c$, the
    operator $\sum_i h_i+cI$ has actual site-dependent operator
    tensors of virtual dimension $1+Nq^2$. Hermiticity and
    positivity are unnecessary for this statement.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:operator_chain_from_all_cut_ranks}
    At every cut, each designated pair contributes rank at most
    $q^2$, and the scalar identity contributes at most one.
    Subadditivity gives the bound at all cuts. Apply the constructed
    operator-chain representation.
\end{proof}

\begin{lemma}[Constructed heat approximants after the energy shift]
    \label{thm:shifted_regional_heat_mpo}
    \lean{TNLean.MPS.AreaLaw.exists_shifted_heat_mpo}
    \leanok
    \uses{thm:physical_cluster_label_mpo}
    Let $A=\sum_i h_i$ be an actual nearest-neighbour sum of
    Hermitian terms of norm at most one, and let $\epsilon I\leq A$.
    If $\beta\geq0$, $\ell\geq1$, and $b_\beta=2(e^\beta-1)\leq1$,
    there are actual tensors of dimension $\ell^2q^\ell$ for an
    operator $Q$ satisfying
    \begin{align*}
        \|e^{-\beta(A-\epsilon I)}-Q\|
        \leq e^{N^2b_\beta^\ell}-1.
    \end{align*}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:physical_cluster_label_mpo}
    Apply the normalized cluster theorem at the lower energy
    $\epsilon$ and multiply its actual tensors by
    $e^{\beta\epsilon}$ through the closing matrix. The exact
    identity $e^{-\beta(A-\epsilon I)}=e^{\beta\epsilon}e^{-\beta A}$
    makes this scalar cancel the normalization in the error bound.
    The scalar shift is kept separate from the local interactions.
\end{proof}

\begin{lemma}[Actual soft tensors at a specified cluster order]
    \label{thm:soft_mpo_cluster_order}
    \lean{TNLean.MPS.AreaLaw.exists_softTruncation_mpo_of_cluster_order}
    \leanok
    \uses{thm:shifted_regional_heat_mpo, lem:trace_chain_linear_combination}
    Let $U=A-\epsilon I\geq0$ have the preceding actual local
    decomposition, and let $m\geq0$, $t>0$, $\ell\geq1$, and
    $b=2(e^{m/t}-1)\leq1$. There are actual operator tensors of
    dimension $(m+1)\ell^2q^\ell$ for an operator $B$ such that
    \begin{align*}
        \|h_{m,t}(U)-B\|
        \leq t2^{m+1}(e^{N^2b^\ell}-1).
    \end{align*}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:shifted_regional_heat_mpo, lem:trace_chain_linear_combination}
    Expand $h_{m,t}$ as the finite heat sum with coefficients
    $c_a$. Construct each heat approximant directly at time $a/t$,
    and combine its actual tensors by diagonal blocks. Every heat
    time is at most $m/t$, so the triangle inequality and the
    absolute coefficient budget $\sum_a|c_a|\leq t2^{m+1}$ give the
    bound. No approximate heat operator is raised to the power $m$.
\end{proof}

\begin{lemma}[A complete soft Hamiltonian at prescribed norm accuracy]
    \label{thm:soft_hamiltonian_mpo_accuracy}
    \lean{TNLean.MPS.AreaLaw.exists_softTruncation_mpo_of_accuracy,
        TNLean.MPS.AreaLaw.exists_softTruncatedHamiltonian_mpo_of_accuracy}
    \leanok
    \uses{thm:soft_mpo_cluster_order, lem:local_sum_scalar_mpo,
        lem:trace_chain_linear_combination}
    Let $m\geq1$, $t=3m$, $\xi>0$, and let $I$ be a set of the
    actual bond labels. Write $A=\sum_{i\in I}h_i$, suppose
    $\epsilon I\leq A$, and put
    \begin{align*}
        W    & =3m2^{m+1},    & b & =2(e^{1/3}-1), \\
        \ell & =\max\left\{1,
        \left\lceil\frac{\log\max\{1,N^2\}
            -\log\log(1+\xi/W)}{-\log b}\right\rceil\right\}.
    \end{align*}
    Then $0<b<1$. The actual regional soft term $h_{m,3m}(A-\epsilon I)$
    has a norm approximation of error at most $\xi$ with virtual
    dimension $(m+1)\ell^2q^\ell$. The full replacement
    \begin{align*}
        \widetilde H=H-A+\epsilon I+h_{m,3m}(A-\epsilon I)
    \end{align*}
    has actual approximating tensors of dimension
    \begin{align*}
        D=1+Nq^2+(m+1)\ell^2q^\ell,
        \qquad\|\widetilde H-T\|\leq\xi.
    \end{align*}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:soft_mpo_cluster_order, lem:local_sum_scalar_mpo,
        lem:trace_chain_linear_combination}
    The strict inequality $1/3<\log(3/2)$ gives $b<1$.
    The logarithmic ceiling ensures
    $W(e^{N^2b^\ell}-1)\leq\xi$. Apply the preceding constructed
    approximation to the original terms masked by $I$.
    Represent the complementary local sum and $\epsilon I$ exactly
    by the local-sum lemma, and add its tensors. Thus the
    approximation concerns the full replacement, with no omitted
    exterior or scalar term.
\end{proof}

For fixed local dimension and soft order $m$, the displayed dimension
bound is polynomial in the chain length and the inverse accuracy
$\xi^{-1}$. This statement is an existence result with explicit dimensions;
a computational running-time claim is not included.

\begin{lemma}[Both physical residuals at independent bounded scale]
    \label{thm:interval_physical_scaled_soft_residuals}
    \lean{TNLean.MPS.AreaLaw.exists_interval_physical_scaled_soft_truncation_residual,
        TNLean.MPS.AreaLaw.exists_interval_physical_scaled_soft_reverse_residual,
        TNLean.MPS.AreaLaw.exists_interval_physical_scaled_soft_residuals}
    \leanok
    \uses{thm:interval_regional_exponential_moment,
        thm:nearest_neighbour_weighted_low_restriction}
    Let $H$ be a positive unit nearest-neighbour chain with one
    prescribed interval, and let its attained original regional
    minima be chosen before the approximate interval energy
    $\epsilon_J-10\leq\epsilon\leq\epsilon_J$ and the parameters.
    For $m\geq1$, $0<t\leq4m$, set $\alpha=(t/2)2^{-m}$.
    The actual soft replacement satisfies $\widetilde H\leq H$.
    On the original window $G_\eta$ its residual is at most
    \begin{align*}
        \|(H-\widetilde H)|_{G_\eta}\|
        \leq80e^{(17+\eta)/8-t/16}.
    \end{align*}
    If $t/2\geq10$, the actual centered soft minimum $E_s$ can
    be chosen before the bandwidth with $-\alpha\leq E_s\leq E_0$.
    If $17+\eta+\alpha\leq t/2$, its own window satisfies
    \begin{align*}
        \|(H-\widetilde H)|_{\widetilde G_\eta}\|
        \leq80e^{(17+\eta+\alpha)/8-t/16}.
    \end{align*}
    Both estimates use the same regional minima and the actual
    original and soft ground energies.
\end{lemma}
\begin{proof}\leanok
    The independent-scale weighted remainder uses the same
    exponential moment of $U/8$ as the equal-scale argument.
    General soft/hard coercivity gives the cutoff $t/2-10$ and
    the correction $\alpha$. The original and soft moments give
    the two residuals. Their normalized minimizing witnesses
    identify all attained minima with the canonical minima.
\end{proof}

\begin{lemma}[Windows at a heat-compatible scale]
    \label{thm:interval_physical_scaled_soft_windows}
    \lean{TNLean.MPS.AreaLaw.exists_interval_physical_scaled_soft_low_energy_closeness}
    \leanok
    \uses{thm:interval_physical_scaled_soft_residuals}
    Let $\eta,w,\delta>0$, and set
    \begin{align*}
        \rho & =\min\left\{1,\frac w2,
        \frac{\delta w^2}{2(\eta+2)}\right\}, \\
        m    & =\left\lceil12+\frac23\eta+
        \frac{16}{3}\log(80/\rho)\right\rceil,
        \qquad t=3m.
    \end{align*}
    This order is chosen before every physical chain and interval.
    The actual soft replacement has $0\leq E_0-E_s\leq\rho$.
    Its window below $E_s+\eta+w$ is $\delta$-close to the original
    window below $E_0+\eta$, and the original window below
    $E_0+\eta+w$ is $\delta$-close to its window below $E_s+\eta$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:interval_physical_scaled_soft_residuals}
    The chosen order ensures $\alpha\leq3/4$, the reverse cutoff,
    and both residuals at most $\rho$. The existing finite-window
    perturbation theorem applies in both directions because
    $2(\eta+2)\rho\leq\delta w^2$ and $\rho\leq w/2$.
\end{proof}

\begin{theorem}[The same physical soft Hamiltonian has windows and tensors]
    \label{thm:physical_soft_windows_and_mpo}
    \lean{TNLean.MPS.AreaLaw.exists_interval_physical_soft_closeness_and_mpo}
    \leanok
    \uses{thm:interval_physical_scaled_soft_windows,
        thm:soft_hamiltonian_mpo_accuracy}
    For the preceding uniformly chosen $m$ and $t=3m$, the same
    actual physical $\widetilde H$ has both directed window
    comparisons and, for every $\xi>0$, actual operator tensors
    of the dimension $D$ specified in
    Lemma~\ref{thm:soft_hamiltonian_mpo_accuracy}, with error
    $\|\widetilde H-T\|\leq\xi$.
    The regional minima and original ground data are constructed
    before $\epsilon$, and the soft ground energy is its actual
    minimum. The cluster order depends only on $N,m,\xi$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:interval_physical_scaled_soft_windows,
        thm:soft_hamiltonian_mpo_accuracy}
    Use the constructed window theorem and retain its literal
    soft Hamiltonian. Its regional lower bound and
    $\epsilon\leq\epsilon_J$ supply the lower energy for the heat
    construction. Mask the actual internal bonds of the interval,
    derive the original support and norm bounds, and apply the
    full soft-Hamiltonian tensor construction.
\end{proof}

These are corrected one-interval results, using scale three times order,
the proved exponential-moment residual and the interaction-label cluster
estimate. The false printed uniform energy and physical-support cluster
claims remain unmarked and their counterexamples remain in
\cite{gap:alvv16_soft_truncation_energy_estimate,gap:alvv16_cluster_approximation}.
Simultaneous replacements on several prescribed regions, a complete
improving polynomial family, and the algorithmic running-time conclusion
remain separate mathematical tasks.
