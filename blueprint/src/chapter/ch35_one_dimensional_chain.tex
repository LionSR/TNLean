\chapter{The one-dimensional area law and matrix product state approximation}
\label{ch:one_dimensional_chain}

% The chain specialization of the 2026 area-law and PEPS manuscripts, following the
% TNLean note docs/notes/one_dimensional_note. Every constant is explicit. The model is
% stated natively on the chain of L sites. The chain proof uses only the site-generic
% layer TNLean/Lattice and QICLean; no module of TNLean/MPS/AreaLaw imports TNLean.PEPS.

The chain of $L$ sites carries a local dimension $q$, a finite-range Hamiltonian
with one Hermitian term of norm at most $J$ per support of diameter at most $R$,
and a unit ground vector separated from the rest of the spectrum by a gap
$\Delta$. On a chain the two-dimensional argument of \cite{OpenAI2026AreaLaw}
collapses to three steps: the marginal tail bound at an interval cut, the
buffered discount with a buffer of constant width, and the Araki--Lieb
inequality. The matrix product state approximation of
\cite{OpenAI2026PolynomialPEPS} then follows from the tail bound alone by
truncating the Schmidt decompositions across the cuts in sequence. This chapter
follows \cite{TNLean2026OneDimensional}; all constants are explicit, and the
area-law constant is linear in $1+J/\Delta$. The two finite-dimensional cores,
the tail bound and the buffered discount, are stated in the companion library
for an arbitrary finite set of sites and are instantiated here directly.

\paragraph{Main results.}
Theorem~\ref{thm:domain_area_law} (Theorem~A) bounds the entropy of every
region of a gapped ground state on a finite set of integers by a constant
times the number of boundary edges of the region, with a constant that
depends only on $q$, $R$ and $J/\Delta$; Theorem~\ref{thm:chain_area_law}
is the case of a chain. Theorem~\ref{thm:chain_mps_approximation}
(Theorem~B) gives a normalized open-boundary matrix product state, of bond
dimension polynomial in $L$, within distance $1/L$ of the ground state.
The corollaries that follow them give explicit and sharper constants.
Chapter~\ref{ch:one_dimensional_preliminaries} collects the definitions on
chains, and Chapter~\ref{ch:one_dimensional_open_mps} the facts on matrix
product states used here. Chapter~\ref{ch:one_dimensional_protocol} proves
Theorem~B a second way, by a sequential preparation protocol.

\section{The marginal tail bound at a cut}

\begin{lemma}[Marginal moment and tail bounds at a cut]
    \label{lem:chain_marginal_tail}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.surprisalTail_regionState_le,
        Entropy.log_surprisalMoment_cut_le}
    \leanok
    \uses{def:chain_model}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on the chain, and
    let $A$ be any set of sites. Let $K$ be the surprisal $-\log p$ of the
    eigenvalue $p$ of the reduced state on $A$ drawn with probability $p$, so
    that $\mathbb E K=S(A)$. For every real $u$ with
    $|u|\le1/(32\sqrt{(1+J/\Delta)\,\mathcal B_A})$,
    \begin{align}
        \log\mathbb E e^{uK}
        \le uS(A)+512e\frac J\Delta\mathcal B_Au^2.\notag
    \end{align}
    For every real $w$,
    \begin{align}
        \Pr\{|K-S(A)|>w\}\le
        \min\Bigl\{1,\ 2e^{e/2}\exp\Bigl(-\frac{w}{32\sqrt{(1+J/\Delta)\,
                \mathcal B_A}}\Bigr)\Bigr\}.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    These are the log-moment and tail bounds of Lemma~3.1 of
    \cite{OpenAI2026AreaLaw}, valid on any finite tensor product: each term
    acts as the identity off its support, is
    Hermitian and has norm at most $J$, and the ground vector is a unit
    eigenvector with the projector gap. The chain model supplies exactly these
    hypotheses.
\end{proof}

\begin{definition}[Doubled chain Hamiltonian]
    \label{def:chain_doubled_hamiltonian}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled}
    \leanok
    \uses{def:chain_model,def:qc_site_pair_coordinates}
    Let $L,q,R$ be natural numbers, let $J$ be real, and let
    $H=\sum_X h_X$ be a range-$R$ chain Hamiltonian with local dimension $q$
    and $\|h_X\|\le J$. Pair the two copies at each original site by the
    bijection $p$ of Definition~\ref{def:qc_site_pair_coordinates}, with
    $p(\sigma,\tau)(i)=q\sigma(i)+\tau(i)$. Write $p_*M$ for simultaneous
    relabelling of the rows and columns of a matrix $M$ by $p$. The
    \emph{doubled Hamiltonian} has local dimension $q^2$ and interactions
    \begin{align}
        \widehat h_X=p_*(h_X\otimes\mathbb 1+\mathbb 1\otimes h_X).
    \end{align}
    Each $\widehat h_X$ is Hermitian, is supported on the same set $X$, and
    satisfies $\|\widehat h_X\|\le2J$. Thus the range remains $R$.
    This is the two-copy construction in the paragraph ``Two-sided Gaussian
    filtering'' of \cite{TNLean2026OneDimensional}.
\end{definition}
\begin{proof}
    \leanok
    \uses{def:qc_local_circuit,thm:area_law_supported_identity,thm:qc_site_pair_tensor_support}
    By the matrix-entry characterization, $h_X$ belongs to the algebra
    supported on $X$; the identity belongs to that algebra as well. The tensor-product support identity,
    applied to each summand, gives support on $X\cup X=X$, and the
    matrix-entry characterization gives the required support condition.
    Taking adjoints preserves each summand and commutes with relabelling.
    Relabelling preserves the operator norm, and tensoring with an identity
    does not increase it. Hence
    \begin{align}
        \|\widehat h_X\|
        \le\|h_X\otimes\mathbb 1\|+\|\mathbb 1\otimes h_X\|
        \le\|h_X\|+\|h_X\|\le2J.
    \end{align}
\end{proof}

\begin{lemma}[Operator of the doubled Hamiltonian]
    \label{lem:chain_doubled_operator}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled_operator}
    \leanok
    \uses{def:chain_doubled_hamiltonian}
    Under the hypotheses of Definition~\ref{def:chain_doubled_hamiltonian},
    the sum of the doubled interactions satisfies
    \begin{align}
        \widehat H=\sum_X\widehat h_X
        =p_*(H\otimes\mathbb 1+\mathbb 1\otimes H).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian}
    Sum the defining interaction identity over the original supports $X$.
    Relabelling is linear, and the tensor product is linear in each factor.
\end{proof}

\begin{definition}[Regionally swapped doubled Hamiltonian]
    \label{def:chain_swapped_doubled_hamiltonian}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.swappedDoubled,
        TNLean.MPS.AreaLaw.ChainHamiltonian.swappedDoubled_operator}
    \leanok
    \uses{def:chain_doubled_hamiltonian,def:peps_frame_sheet_swap}
    Let $H=\sum_Xh_X$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$, with local dimension $q$ and
    $\|h_X\|\le J$, where $L,q,R$ are natural numbers and $J$ is real.
    For any $D\subseteq I$, let $F=p_*F_D$ in the paired coordinates of
    Definition~\ref{def:chain_doubled_hamiltonian}. The regionally swapped
    doubled Hamiltonian has interactions and sum
    \begin{align}
        \widehat h'_X & =F\widehat h_XF^\dagger,\notag              \\
        \widehat H'   & =\sum_X\widehat h'_X=F\widehat H F^\dagger.
        \label{eq:chain_swapped_doubled_operator}
    \end{align}
    Each $\widehat h'_X$ is Hermitian, is supported on the same set $X$,
    and has norm at most $2J$. Thus this is a range-$R$ chain Hamiltonian
    with local dimension $q^2$ and interaction bound $2J$, also when
    $q=0$ or $L=0$.
\end{definition}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian,thm:area_law_supported_identity,
        thm:peps_sheet_swap_conjugate_support,thm:peps_frame_sheet_swap_norm}
    The support identity for the conjugated doubled interaction gives
    support on $X$, and taking adjoints preserves Hermiticity.
    Since $\|F\|\le1$, the doubled interaction estimate gives
    \begin{align}
        \|F\widehat h_XF^\dagger\|
        \le\|\widehat h_X\|\le2J.
    \end{align}
    Distributing left and right multiplication over the finite sum proves
    (\ref{eq:chain_swapped_doubled_operator}).
\end{proof}

\begin{lemma}[The ground vectors of the doubled and swapped chains]
    \label{thm:chain_doubled_swapped_ground_gap}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled_swappedDoubled_isGappedGroundState}
    \leanok
    \uses{def:chain_model,def:chain_swapped_doubled_hamiltonian}
    In the setting of Definition~\ref{def:chain_swapped_doubled_hamiltonian},
    let $E_0,\Delta$ be real with $\Delta>0$, and suppose the original
    vector $\Omega$ satisfies
    \begin{align}
        \|\Omega\|     & =1,\notag                                          \\
        H\Omega        & =E_0\Omega,\notag                                  \\
        H-E_0\mathbb 1 & \ge\Delta(\mathbb 1-|\Omega\rangle\langle\Omega|).
    \end{align}
    Define the vectors in the paired physical coordinates by
    $\Psi(p(\sigma,\tau))=\Omega(\sigma)\Omega(\tau)$ and
    $\Phi=F\Psi$. Then
    \begin{align}
        \|\Psi\|        & =\|\Phi\|=1,\notag                                   \\
        \widehat H\Psi  & =2E_0\Psi,\notag                                     \\
        \widehat H'\Phi & =2E_0\Phi,\notag                                     \\
        \widehat H-2E_0\mathbb 1
                        & \ge\Delta(\mathbb 1-|\Psi\rangle\langle\Psi|),\notag \\
        \widehat H'-2E_0\mathbb 1
                        & \ge\Delta(\mathbb 1-|\Phi\rangle\langle\Phi|).
    \end{align}
    In particular, these vectors give the ground lines of the two
    Hamiltonians, with common ground energy $2E_0$ and gap at least
    $\Delta$. This is the contiguous-chain assertion in the paragraph
    ``Two-sided Gaussian filtering'' of~\cite{TNLean2026OneDimensional},
    before the filtering construction; the distinction from arbitrary
    finite induced domains is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:chain_doubled_operator,def:chain_swapped_doubled_hamiltonian,
        thm:permutation_matrix_unitary,thm:mpu_reindex_unitary}
    Put $P=|\Omega\rangle\langle\Omega|$, $Q=\mathbb 1-P$ and
    $G=H-E_0\mathbb 1-\Delta Q\ge0$. Normalization gives $Q\ge0$.
    Before pairing coordinates, the doubled defect is
    \begin{align}
        H\otimes\mathbb 1+\mathbb 1\otimes H-2E_0\mathbb 1
        -\Delta(\mathbb 1-P\otimes P)
         & =G\otimes\mathbb 1+\mathbb 1\otimes G+\Delta Q\otimes Q
        \ge0.
    \end{align}
    The tensor-square vector satisfies
    \begin{align}
        (H\otimes\mathbb 1+\mathbb 1\otimes H)(\Omega\otimes\Omega)
         & =2E_0\,\Omega\otimes\Omega,
         &
        \|\Omega\otimes\Omega\|
         & =\|\Omega\|^2=1.
    \end{align}
    Pairing coordinates is unitary, so the same norm, eigen-equation and
    positive defect hold for $\Psi$ and $\widehat H$.
    The regional swap is a permutation matrix; after pairing it remains
    unitary, $F^\dagger F=\mathbb 1$. Since $\widehat H'=F\widehat HF^\dagger$,
    \begin{align}
        \widehat H'\Phi
         & =F\widehat HF^\dagger F\Psi=F\widehat H\Psi=2E_0\Phi,
         &
        \|\Phi\|
         & =\|F\Psi\|=\|\Psi\|=1,
    \end{align}
    and
    \begin{align}
         & \widehat H'-2E_0\mathbb 1
        -\Delta(\mathbb 1-|\Phi\rangle\langle\Phi|)\notag \\
         & \qquad=F\bigl(\widehat H-2E_0\mathbb 1
        -\Delta(\mathbb 1-|\Psi\rangle\langle\Psi|)\bigr)F^\dagger
        \ge0.
    \end{align}
    Finally, a positive projector gap forces every vector at the ground
    energy to lie in the span of the displayed unit vector, giving the
    asserted ground lines.
\end{proof}

\begin{lemma}[Interactions split by an interval]
    \label{lem:chain_crossing_count}
    \lean{
        TNLean.MPS.AreaLaw.IsChainSupport.subset_window_of_splits,
        TNLean.MPS.AreaLaw.chainCrossConst,
        TNLean.MPS.AreaLaw.card_crossingTerms_interval_le,
        TNLean.MPS.AreaLaw.cutBudgetConst,
        TNLean.MPS.AreaLaw.one_le_cutBudgetConst,
        TNLean.MPS.AreaLaw.chainBudgetConst,
        TNLean.MPS.AreaLaw.one_le_chainBudgetConst,
        TNLean.MPS.AreaLaw.supportDim_le_real,
        TNLean.MPS.AreaLaw.log_sq_supportDim_le,
        TNLean.MPS.AreaLaw.cutLogBudget_interval_le
    }
    \leanok
    \uses{def:chain_model,thm:chain_supports_in_window,thm:chain_single_splitting}
    Let $X=[a,b)$ be an interval of the chain. A support of range $R$ that
    meets both $X$ and its complement lies in the window $[a-R,a+R)$ or in the
    window $[b-R,b+R)$, each of at most $2R$ sites. Hence at most
    \begin{align}
        N_R=2\cdot2^{2R}
    \end{align}
    supports are split by $X$, and the cut budget of $X$ is at most
    \begin{align}
        B_R=1+N_R\log^2\bigl(e\,q^{R+1}\bigr).
    \end{align}
    Both bounds are independent of the interval, and no support of range $R$
    is split by two of the shells $[l-(j+1)R,\,r+(j+1)R)$, $j\ge0$, of an
    interval $[l,r)$. For a natural number $N$ put
    $B(N)=1+N\log^2\bigl(e\,q^{R+1}\bigr)$, so that $B(N)\ge1$ and
    $B_R=B(N_R)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    A split support contains a site $v$ in $X$ and a site $w$ outside $X$
    with $|v-w|\le R$. If $w<a$ then every site $x$ of the support satisfies
    $a-R\le v-R\le x\le w+R<a+R$; if $w\ge b$ then $b-R\le w-R\le x\le v+R<b+R$.
    A window of at most $2R$ sites has at most $2^{2R}$ subsets. The budget
    bound follows from the crossing count and the dimension bound $q^{R+1}$. If
    a support met the shell with index $j$ and left the shell with index
    $j'>j$, two of its sites would be at distance at least $R+1$.
\end{proof}

\begin{lemma}[Consecutive-window interactions split by an interval]
    \label{thm:chain_consecutive_window_count}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.card_crossingTerms_interval_le_of_consecutive_terms}
    \leanok
    \uses{def:chain_model}
    Let $L,q,R$ be natural numbers, let $J$ be real, and let
    $H=\sum_X h_X$ be a finite-range Hamiltonian on the chain of $L$ sites.
    Suppose that for every $X$ with $h_X\ne0$ there is a natural number $a$
    such that $a+R+1\le L$ and $X=[a,a+R+1)$.
    For natural numbers $l,r$, put $A=[l,r)\cap\{0,\ldots,L-1\}$. Then
    \begin{align}
        \bigl|\{X:h_X\ne0,\ X\cap A\ne\varnothing,
        \ X\setminus A\ne\varnothing\}\bigr|\le2R.\notag
    \end{align}
    This is the consecutive-window counting refinement in the remark on
    sharper constants of~\cite[Part~I]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    A full window is determined by its least site $a$. If it crosses the cut
    at position $c$, then $a<c\le a+R$, so its start belongs to $[c-R,c)$,
    which has at most $R$ sites. A window split by $A$ crosses at least one
    of the cuts at $l$ and $r$. Its start therefore lies in
    $[l-R,l)\cup{[r-R,r)}$, a set of at most $2R$ sites. All intervals are
    clipped to the chain, and natural subtraction is truncated at zero.
    When $R=0$, or when $A$ is empty, no window is split.
\end{proof}

\begin{lemma}[The cut budget for consecutive-window interactions]
    \label{thm:chain_window_cut_budget}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.cutLogBudget_interval_le_of_consecutive_terms}
    \leanok
    \uses{thm:chain_consecutive_window_count}
    Let $q\ge1$, and suppose that every nonzero interaction is assigned to
    a full window of $R+1$ consecutive sites contained in the chain.
    For every coordinate interval $A$, put
    \begin{align}
        \mathcal B(A)
         & =1+\sum_{\substack{X:\ h_X\ne0,\ X\cap A\ne\varnothing \\
                X\setminus A\ne\varnothing}}
        \log^2(e q^{|X|}).\notag
    \end{align}
    Then $\mathcal B(A)\le1+2R\log^2(e q^{R+1})$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:chain_consecutive_window_count,lem:chain_crossing_count}
    At most $2R$ nonzero interactions are split by $A$. Each support has
    at most $R+1$ sites. Since $q\ge1$, each summand is at most
    $\log^2(e q^{R+1})$.
\end{proof}

\begin{lemma}[Marginal moments for consecutive-window interactions]
    \label{thm:chain_consecutive_window_moment}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.log_surprisalMoment_interval_le_of_consecutive_terms}
    \leanok
    \uses{def:chain_model}
    Let $L,R$ be natural numbers, let $q\ge1$ be an integer, and let
    $J\ge0$ and $\Delta>0$ be real. Let $H=\sum_X h_X$ be a
    finite-range Hamiltonian on the chain of $L$ sites. Suppose that for
    every $X$ with $h_X\ne0$ there is a natural number $a$ with
    $a+R+1\le L$ and $X=[a,a+R+1)$.
    Let $E_0$ be real and let $\Omega$ be a unit vector satisfying
    $H\Omega=E_0\Omega$ and
    $H-E_0\ge\Delta(1-|\Omega\rangle\langle\Omega|)$.
    For natural numbers $l,r$, let $\rho_A$ be the reduced state on
    $A=[l,r)\cap\{0,\ldots,L-1\}$, and let $K$ take the value $-\log p$
    with probability $p$ over the positive eigenvalues of $\rho_A$.
    Put $B=1+2R\log^2(e\,q^{R+1})$. For every real $u$ with
    $|u|\le1/(32\sqrt{(1+J/\Delta)B})$,
    \begin{align}
        \log\mathbb E e^{uK}
        \le uS(A)+512e\frac J\Delta B u^2.\notag
    \end{align}
    This is the marginal moment estimate with the consecutive-window count
    from the remark on sharper constants of
    \cite[Part~I]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model,thm:chain_consecutive_window_count,lem:chain_crossing_count,lem:chain_marginal_tail}
    Restrict the interaction sum to $h_X\ne0$. The sum remains exactly $H$,
    and its terms retain their supports, Hermiticity and norm bound $J$.
    The same unit eigenvector and projector gap therefore apply.
    The crossing count and $\dim\mathcal H_X\le q^{R+1}$ bound the budget
    for this restricted family by $B$. Apply the marginal moment estimate
    to that family. Since $J/\Delta\ge0$, replacing its budget by $B$
    decreases the allowed radius and increases the quadratic error.
\end{proof}

\begin{lemma}[Single splitting of nested chain intervals]
    \label{thm:chain_single_splitting}
    \lean{
        TNLean.MPS.AreaLaw.single_splitting_shells,
        TNLean.MPS.AreaLaw.not_splits_two_shells
    }
    \leanok
    \uses{def:chain_model}
    Let $L,R,l,r$ be natural numbers. For each natural number $i$, define
    the coordinate interval
    \begin{align}
        X_i=[l-iR,r+iR)\cap\{0,\ldots,L-1\}.\notag
    \end{align}
    Let $Y$ be a set of chain sites with $|x-y|\le R$ for all $x,y\in Y$.
    Suppose $Y$ meets both $X_j$ and its complement for some $j\ge0$.
    For every natural number $i$,
    \begin{align}
        i>j & \ \Longrightarrow\ Y\subseteq X_i,\notag        \\
        i<j & \ \Longrightarrow\ Y\cap X_i=\varnothing,\notag \\
        Y\cap X_i\ne\varnothing\ \text{and}\ Y\setminus X_i\ne\varnothing
            & \ \Longrightarrow\ i=j.\notag
    \end{align}
    In particular, $Y$ is split by at most one of these intervals, including
    $X_0$. For a nonempty initial interval, these coordinate intervals are
    its metric enlargements by $iR$, so this gives the single-splitting
    lemma of \cite[Part~I]{TNLean2026OneDimensional}. The coordinate
    statement also holds for arbitrary endpoints $l,r$ and for $R=0$.
\end{lemma}
\begin{proof}
    \leanok
    Choose $v\in Y\cap X_j$ and $w\in Y\setminus X_j$.
    If $i>j$, then $iR\ge jR+R$. For every $y\in Y$,
    \begin{align}
        l-iR & \le l-jR-R\le v-R\le y,\notag \\
        y    & \le v+R<r+jR+R\le r+iR.\notag
    \end{align}
    Thus $Y\subseteq X_i$. If $i<j$ and $z\in Y\cap X_i$, the same
    comparison with $z$ in place of $v$ and the indices $i,j$ exchanged
    would give $Y\subseteq X_j$, contradicting $w\notin X_j$.
    Hence $Y\cap X_i=\varnothing$. Neither a containing interval nor a
    disjoint interval splits $Y$, which proves uniqueness. Taking the
    positive indices $j+1$ gives the corresponding assertion for the
    family starting with radius $R$.
\end{proof}

\begin{lemma}[Logarithmic dependence of the interval cut budget]
    \label{thm:chain_budget_log_sq}
    \lean{TNLean.MPS.AreaLaw.chainBudgetConst_le_mul_log_sq}
    \leanok
    \uses{lem:chain_crossing_count}
    Let $R$ be a nonnegative integer and put $N_R=2\cdot2^{2R}$ and
    $B_R(q)=1+N_R\log^2(e q^{R+1})$. For every integer $q\ge2$,
    \begin{align}
        B_R(q)\le\frac{B_R(2)}{(\log2)^2}(\log q)^2.
    \end{align}
    Here $N_R$ is the interval crossing bound for the
    family of all admissible supports, as in the cut-budget definition of
    \cite[Part~I]{TNLean2026OneDimensional}.
    The remark on sharper constants in that part describes its application
    to the logarithmic dependence of the area-law constant.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:chain_crossing_count}
    Put $c=\log2>0$ and $t=\log q\ge c$. The identity
    $\log(e q^{R+1})=1+(R+1)t$ gives
    $c(1+(R+1)t)\le t(1+(R+1)c)$.
    Both sides are nonnegative. Squaring, multiplying by $N_R$ and adding
    $c^2\le t^2$ yields
    \begin{align}
        c^2 B_R(q)
         & =c^2+N_R\bigl(c(1+(R+1)t)\bigr)^2 \notag \\
         & \le t^2+N_R\bigl(t(1+(R+1)c)\bigr)^2
        =t^2 B_R(2). \notag
    \end{align}
    Divide by $c^2>0$.
\end{proof}

\begin{lemma}[Rényi entropies near order one]
    \label{thm:chain_renyi_near_one}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.near_one_renyi_entropy_interval_bounds}
    \leanok
    \uses{def:chain_model,lem:chain_crossing_count,lem:chain_marginal_tail}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range chain Hamiltonian with local dimension $q\ge1$ and term
    norms at most $J\ge0$. For an interval $X$, let $\rho_X$ be its reduced
    density matrix and put
    \begin{align}
        \vartheta=J/\Delta,\qquad
        B_R=1+N_R\log^2(e q^{R+1}),\qquad
        u_0=\frac1{32\sqrt{(1+\vartheta)B_R}}.\notag
    \end{align}
    For $0<\alpha<1$, define
    \begin{align}
        S_\alpha(X)=\frac{\log\operatorname{Tr}(\rho_X^\alpha)}{1-\alpha},
        \qquad S_1(X)=S(X).\notag
    \end{align}
    Zero eigenvalues contribute zero to the trace. For every
    $\alpha\in[1-u_0,1]$,
    \begin{align}
        S(X)               & \le S_\alpha(X)
        \le S(X)+512e\vartheta B_R(1-\alpha),\notag               \\
        |S_\alpha(X)-S(X)| & \le16e\sqrt{(1+\vartheta)B_R}.\notag
    \end{align}
    This is the near-one Rényi bound in Part~I of
    \cite{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    Write $p_i$ for the eigenvalues of $\rho_X$ and set
    $M(u)=\sum_i p_i\exp(-u\log p_i)$, with zero terms interpreted as zero.
    Since $p_i\ge0$ and $\sum_i p_i=1$, Jensen's inequality gives
    \begin{align}
        \exp(uS(X))\le M(u).\notag
    \end{align}
    The physical log-moment estimate, with the interval cut budget bounded
    by $B_R$, gives
    \begin{align}
        uS(X)\le\log M(u)
        \le uS(X)+512e\vartheta B_Ru^2
        \qquad(0\le u\le u_0).\notag
    \end{align}
    Put $s=\sqrt{(1+\vartheta)B_R}\ge1$. Then $u_0\le1/32$, so the
    orders in the statement are positive. For $u=1-\alpha>0$,
    $M(u)=\operatorname{Tr}(\rho_X^\alpha)$, including when $\rho_X$
    has a kernel. Dividing by $u$ proves the first pair of inequalities.
    Moreover,
    \begin{align}
        512e\vartheta B_Ru
        \le16e\frac{\vartheta B_R}{s}
        \le16es,\notag
    \end{align}
    which proves the absolute bound. At $u=0$, all conclusions follow
    from $S_1(X)=S(X)$.
\end{proof}

\begin{lemma}[Supports split by an arbitrary region of a chain]
    \label{lem:chain_boundary_crossing_count}
    \lean{TNLean.MPS.AreaLaw.IsChainSupport.exists_boundary_window_of_split,
        TNLean.MPS.AreaLaw.card_crossingTerms_le_mul_card_boundaryEdges}
    \leanok
    \uses{def:chain_model,lem:chain_entropy_inequalities}
    Let $L,R$ be natural numbers, let $I=\{0,\ldots,L-1\}$ be the
    contiguous chain, and let $D\subseteq I$. Write $\partial D$ for the
    positions $p$ with $p+1<L$ such that exactly one of $p,p+1$ belongs
    to $D$. For $p\in\partial D$ put
    \begin{align}
        W_p=I\cap[\max\{0,p+1-R\},\,p+1+R).
    \end{align}
    If a nonempty support $X\subseteq I$ of diameter at most $R$ meets
    both $D$ and $I\setminus D$, then $X\subseteq W_p$ for some boundary
    edge $p\in\partial D$. Consequently,
    \begin{align}
        \#\{X\subseteq I:\ X\ne\varnothing,\ \operatorname{diam}X\le R,
        \ X\cap D\ne\varnothing,\ X\setminus D\ne\varnothing\}
        \le 2^{2R}|\partial D|.
    \end{align}
    No interval assumption on $D$ is needed. The conclusions include the
    empty chain and $R=0$; in either case there is no split support.
    This is the boundary-window count underlying the interval estimate
    in Section~2, equation~\textup{(count)}, and the boundary count in the
    paragraph ``Two-sided Gaussian filtering'' of
    \cite{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model,lem:chain_entropy_inequalities,
        lem:chain_crossing_count}
    Choose two sites of $X$ with different membership in $D$, and order
    their coordinates as $a<b$. Along the integer interval from $a$ to
    $b$, the indicator of $D$ must change at two consecutive sites
    $p,p+1$. Thus $a\le p<p+1\le b<L$ and $p\in\partial D$.

    For every $x\in X$, the diameter bound gives
    $x\le a+R$ and $b\le x+R$. Hence
    \begin{align}
        \max\{0,p+1-R\}\le x<p+1+R,
    \end{align}
    proving $X\subseteq W_p$. Each window has at most $2R$ sites, so
    contains at most $2^{2R}$ subsets. Cover the split supports by the
    families of subsets of $W_p$, one family for each $p\in\partial D$,
    and bound the cardinality of the finite union by the sum of their
    cardinalities. This proves the stated bound, including the case of
    an empty boundary.
\end{proof}

\begin{lemma}[The boundary perturbation of a doubled chain]
    \label{thm:chain_partial_swap_boundary}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_term_difference_spec,
        TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_difference_eq_sum_crossing,
        TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_boundary_spec}
    \leanok
    \uses{def:chain_model,def:chain_doubled_hamiltonian,
        def:qc_site_pair_coordinates,def:peps_frame_sheet_swap,
        lem:chain_boundary_crossing_count}
    Let $L,q,R$ be natural numbers, let $J\in\mathbb R$, and let
    $H=\sum_Xh_X$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$, with local dimension $q$ and
    $\|h_X\|\le J$. Let $D\subseteq I$ be arbitrary. In the paired
    coordinates $p$ of Definition~\ref{def:chain_doubled_hamiltonian}, put
    \begin{align}
        F       & =p_*F_D,\notag                                \\
        T_{X,0} & =p_*(h_X\otimes\mathbb 1),\notag              \\
        T_{X,1} & =p_*(\mathbb 1\otimes h_X),\notag             \\
        K_{X,b} & =FT_{X,b}F^\dagger-T_{X,b}\quad(b\in\{0,1\}).
    \end{align}
    Every $K_{X,b}$ is Hermitian, is supported on the original set $X$,
    and satisfies $\|K_{X,b}\|\le2J$. Let $\mathcal C$ be the set of
    original supports meeting both $D$ and $I\setminus D$, and let
    $N_R=2\cdot2^{2R}$. For the doubled operator $\widehat H$,
    \begin{align}
        F\widehat H F^\dagger-\widehat H
                      & =\sum_{X\in\mathcal C}\sum_{b\in\{0,1\}}K_{X,b},\notag \\
        2|\mathcal C| & \le N_R|\partial D|.
    \end{align}
    Let $Z=\bigcup_{X\in\mathcal C}X$ be the union of crossing
    supports and put $\Delta H=F\widehat H F^\dagger-\widehat H$.
    The operator $\Delta H$ is supported on $Z$. For every $z\in Z$ there
    is a boundary position $a\in\partial D$ with $|z-a|\le R$,
    and
    \begin{align}
        \|\Delta H\|\le 2J N_R|\partial D|.
    \end{align}
    These assertions include $q=0$, the empty chain, and empty or full
    $D$. They give the boundary decomposition on a contiguous chain used
    in the paragraph ``Two-sided Gaussian filtering'' of
    \cite{TNLean2026OneDimensional}; the distinction from arbitrary finite
    induced domains is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian,thm:qc_site_pair_tensor_support,
        thm:area_law_supported_identity,thm:peps_sheet_swap_conjugate_support,
        thm:peps_frame_sheet_swap_norm,
        thm:peps_sheet_swap_doubled_cancellation,
        lem:chain_boundary_crossing_count}
    The support reduction for a conjugated tensor product in paired
    coordinates shows that $FT_{X,b}F^\dagger$ is supported on $X$.
    The same is true of $T_{X,b}$, so subtraction preserves its support.
    Taking adjoints proves Hermiticity. Pairing coordinates preserves
    the operator norm, and the regional swap satisfies $\|F\|\le1$.
    Tensoring with an identity does not increase the operator norm, hence
    \begin{align}
        \|T_{X,b}\|           & \le\|h_X\|\le J,\notag \\
        \|FT_{X,b}F^\dagger\| & \le J,\notag           \\
        \|K_{X,b}\|           & \le J+J=2J.
    \end{align}

    Swapping exchanges the two replica terms of a support inside $D$ and
    fixes those of a support disjoint from $D$:
    \begin{align}
        FT_{X,0}F^\dagger & =T_{X,1},\quad FT_{X,1}F^\dagger=T_{X,0}
        \qquad(X\subseteq D),\notag                                  \\
        FT_{X,b}F^\dagger & =T_{X,b}\qquad(X\cap D=\varnothing).
    \end{align}
    Adding the two replica terms gives, for every support
    outside $\mathcal C$,
    \begin{align}
        F(T_{X,0}+T_{X,1})F^\dagger=T_{X,0}+T_{X,1}.
    \end{align}
    By definition, each doubled interaction is $T_{X,0}+T_{X,1}$.
    Summing these identities expresses $\widehat H$ as the sum of these
    pairs. Distributing conjugation and subtraction over the finite sum
    cancels each pair outside $\mathcal C$, leaving exactly the sum over
    the crossing supports. Finally, Lemma~\ref{lem:chain_boundary_crossing_count}
    gives $|\mathcal C|\le2^{2R}|\partial D|$; counting the two replica
    labels for each support gives the stated bound.

    Each term in the resulting expansion is supported on some
    $X\subseteq Z$, so their finite sum is supported on $Z$. If $z\in Z$,
    choose a crossing support $X$ containing it. The boundary-window
    inclusion gives $a+1-R\le z<a+1+R$ for a boundary position
    $a$, whence $|z-a|\le R$.
    If $L=0$, the boundary and the crossing family are empty and
    $\Delta H=0$. Otherwise, a singleton interaction implies
    $0\le\|h_{\{x\}}\|\le J$ at a site $x$, so $J\ge0$.
    The triangle inequality and the bound on the number of replica labels
    now give
    \begin{align}
        \|\Delta H\|\le 2J(2|\mathcal C|)\le 2J N_R|\partial D|.
    \end{align}
\end{proof}

\begin{lemma}[Interaction norm budget at a site]
    \label{lem:chain_site_norm_budget}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.sum_norm_terms_containing_le,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_bound_nonneg}
    \leanok
    \uses{def:chain_model}
    Let $L,q,R$ be natural numbers and $J\in\mathbb R$. On the chain of $L$
    sites with local dimension $q$, let $H=\sum_X h_X$ be a finite-range
    Hamiltonian with one term of norm at most $J$ for each nonempty support
    $X$ of diameter at most $R$. For every site $x$,
    \begin{align}
        \sum_{X\ni x}\lVert h_X\rVert\le 2^{2R+1}J.
    \end{align}
    The presence of a site also forces $J\ge0$.
    This is an auxiliary consequence of the Hamiltonian setting
    in Section~1 of~\cite{TNLean2026OneDimensional}; it supplies a
    per-site interaction estimate of the form assumed immediately before
    the Lieb--Robinson lemma there. No spectral-gap assumption is needed.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model, lem:chain_crossing_count}
    Fix a site $x$. The singleton $\{x\}$ is an admissible support, so
    $0\le\lVert h_{\{x\}}\rVert\le J$. Every support containing $x$ lies in
    the clipped interval $[x-R,x+R+1)$, which has at most $2R+1$ sites.
    Counting subsets of this interval gives at most $2^{2R+1}$ such supports.
    Summing their norm bounds and using $J\ge0$ proves the claim. If the chain
    has no sites, the assertion is vacuous.
\end{proof}

\begin{lemma}[Localization of the relative-evolution generator]
    \label{thm:chain_relative_generator_localization}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.norm_relativeGenerator_sub_siteExpectation_le}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,thm:chain_partial_swap_boundary,
        def:qc_site_expectation}
    Let $H$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$ with local dimension $q\ge1$ and interaction
    bound $J\in\mathbb R$. Let $D\subseteq I$, and let
    $K=\widehat H$ and $K'=\widehat H'$ be its doubled and regionally
    swapped doubled operators. Write $\mathcal C_D$ for the original
    supports meeting both $D$ and its complement, and put
    $Z=\bigcup_{X\in\mathcal C_D}X$.
    For every real $\ell\ge0$ and $t\in\mathbb R$, define
    \begin{align}
        B    & =N_\ell(Z),\notag         \\
        G(t) & =e^{itK'}(K'-K)e^{-itK'}.
    \end{align}
    Set
    \begin{align}
        b_0 & =2^{2R+1}(2J),\notag                          \\
        a   & =\frac{2(2+4|b_0|e^{2R})e}{1-e^{-1/2}},\notag \\
        v   & =2(R+1)|b_0|e^{2R}+1,\notag                   \\
        N_R & =2\cdot2^{2R}.
    \end{align}
    Then the normalized expectation on the paired physical sites satisfies
    \begin{align}
        \|G(t)-E_B(G(t))\|
        \le4|J|a(R+1)^2N_R L\,e^{v|t|-\ell/2}.
    \end{align}
    No nonemptiness of the chain or of $\mathcal C_D$, sign condition on
    $J$, or spectral-gap hypothesis is required. This is the intermediate
    generator estimate in the localization paragraph of the proof of the
    information-reset lemma in Part~II of~\cite{TNLean2026OneDimensional},
    on a contiguous chain; the domain distinction is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,thm:chain_partial_swap_boundary,
        lem:chain_site_norm_budget,thm:chain_lieb_robinson,
        lem:qc_site_expectation_properties,thm:qc_site_expectation_supported_approximant}
    The swapped doubled interactions have range $R$ and bound $2J$.
    Their per-site norm sum is at most $b_0$. The chain localization
    estimate therefore has constants $a,v$ as displayed.
    By Lemma~\ref{thm:chain_partial_swap_boundary},
    \begin{align}
        K'-K            & =\sum_{X\in\mathcal C_D}\sum_{b\in\{0,1\}}A_{X,b},\notag \\
        \|A_{X,b}\|     & \le2J\le2|J|,\notag                                      \\
        2|\mathcal C_D| & \le N_R|\partial D|\le N_R L.
    \end{align}
    Each $A_{X,b}$ is supported on $X$, with $|X|\le R+1$.
    For one such term put $A=e^{itK'}A_{X,b}e^{-itK'}$ and
    $A_0=E_{N_\ell(X)}(A)$. The chain estimate gives
    \begin{align}
        \|A-A_0\|\le2|J|a(R+1)^2e^{v|t|-\ell/2}.
    \end{align}
    Since $X\subseteq Z$, the operator $A_0$ is supported on $B$ and is
    fixed by $E_B$. Hence
    \begin{align}
        A-E_B(A)     & =(A-A_0)-E_B(A-A_0),\notag \\
        \|A-E_B(A)\| & \le2\|A-A_0\|
        \le4|J|a(R+1)^2e^{v|t|-\ell/2}.
    \end{align}
    Conjugation and the expectation are linear. Summing this estimate
    over the crossing replica labels and using their cardinality
    bound proves the assertion. An empty crossing family gives the zero
    generator; the same nonnegative estimates include $J=0$.
\end{proof}

\begin{lemma}[Separation of the reset support from the cores]
    \label{lem:chain_reset_support_separation}
    \lean{TNLean.MPS.AreaLaw.chain_reset_support_separation}
    \leanok
    \uses{def:chain_model}
    Let $I=\{0,\ldots,L-1\}$ be a contiguous chain, let $R$ be a
    natural number, and let $D,C,R_0,C_0\subseteq I$, with $D\cap C=\varnothing$.
    Suppose that a real number $w$ satisfies
    \begin{align}
        w & <|r-y| &  & (r\in R_0,\ y\in I\setminus D),\notag \\
        w & <|c-x| &  & (c\in C_0,\ x\in I\setminus C).
    \end{align}
    Put $Q=I\setminus(D\cup C)$. Let $\mathcal C_D$ consist of all
    nonempty supports $X\subseteq I$ of diameter at most $R$ that meet
    both $D$ and its complement, and put $Z=\bigcup_{X\in\mathcal C_D}X$.
    Then
    \begin{align}
        w-R & <|a-z| &  & (z\in Q\cup Z,\ a\in R_0\cup C_0).
    \end{align}
    Consequently, for every real $\ell$ with $\ell+R\le w$,
    \begin{align}
        \{y\in I:\ \exists z\in Q\cup Z,\ |z-y|\le\ell\}
        \cap(R_0\cup C_0)=\varnothing.
    \end{align}
    Empty chains and empty sets are allowed. No sign assumption on
    $w$ or $\ell$, or containment of the cores in the outer regions,
    is needed beyond the displayed hypotheses.
    This is the support-separation step in Part~II
    of~\cite{TNLean2026OneDimensional}, on a contiguous
    chain; see~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    If $z\in Q$, the two margin inequalities apply directly; weakening
    the lower bound from $w$ to $w-R$ is permitted since $R\ge0$.
    If $z\in Z$, choose a crossing support $X$ containing $z$,
    together with $y\in X\setminus D$ and $x\in X\cap D$.
    The diameter bound gives $|z-y|,|z-x|\le R$.
    For $r\in R_0$, the first margin and the triangle inequality give
    \begin{align}
        w<|r-y|\le|r-z|+R.
    \end{align}
    Since $D$ and $C$ are disjoint, $x\notin C$; the second margin
    gives the same conclusion for every $c\in C_0$.
    This proves the first assertion with the exact loss $R$.
    If a core site $a$ also belonged to the displayed neighborhood,
    there would be $z\in Q\cup Z$ with $|z-a|\le\ell$.
    Symmetry of distance would then give
    $w-R<|a-z|\le\ell\le w-R$, a contradiction.
\end{proof}

    The elementary exponential estimate, reserving a factor
    $e^{-\ell/4}$ to absorb $P$, gives
    \begin{align}
        \|M_{\mathrm{loc}}-M_T\|
         & \le C L^2(1+T_0B_*)e^{vT_0B_*-\ell/2}
        \le\varepsilon.
    \end{align}
    Indeed $\ell/4\ge P$, $P e^{-P}\le1$, and the remaining quarter-decay
    absorbs $L^4(1+T_0B_*)$, with coefficient $P$; here $L^2\le L^4$.
    The two normalized ground vectors therefore have residuals at most
    $4\varepsilon$ for $M_{\mathrm{loc}}$.
    The same-support polar correction of
    Theorem~\ref{thm:peps_reset_polar_step} constructs a unitary $V$ on $S$
    satisfying
    \begin{align}
        \delta:=\|V\Psi-\Phi\|\le8L^{-r}.
    \end{align}

    The ceiling bound and $B_*\ge1$ give
    $\ell+R<C_pB_*=w$.
    Lemma~\ref{lem:chain_reset_support_separation} makes $S$ disjoint from
    both cores, so $V$ acts trivially on their joint marginal in each
    replica. The entropy comparison of
    Lemma~\ref{thm:regional_reset_cores_entropy}, applied to the unitary
    $V$ supported on $S$, yields
    \begin{align}

         & \le8\sqrt\delta\,\log(e q^{|R_0|})\notag \\
         & \le8\sqrt\delta\,(1+L\log q).
    \end{align}
    Since $L\ge L_0$, Theorem~\ref{thm:peps_reset_entropy_scalar}, with
    $r=2p+10$, bounds the last expression by $L^{-p}$.
    Empty cores cause no exception: their mutual information is zero.
\end{proof}

\begin{lemma}[Araki--Lieb and the area law from interval bounds]
    \label{lem:chain_entropy_inequalities}
    \lean{
        TNLean.Lattice.regionEntropy_union_le,
        TNLean.Lattice.regionEntropy_le_union_add,
        TNLean.Lattice.regionEntropy_empty,
        TNLean.Lattice.regionEntropy_univ,
        TNLean.Lattice.regionEntropy_le_card_mul_log,
        TNLean.MPS.AreaLaw.boundaryEdges,
        TNLean.MPS.AreaLaw.mem_boundaryEdges,
        TNLean.MPS.AreaLaw.boundaryEdges_empty,
        TNLean.MPS.AreaLaw.boundaryEdges_univ,
        TNLean.MPS.AreaLaw.exists_interval_subset_boundaryEdges_sdiff_ssubset,
        TNLean.MPS.AreaLaw.regionEntropy_le_mul_card_boundaryEdges_of_intervals,
        TNLean.MPS.AreaLaw.card_boundaryEdges_prefix_le
    }
    \leanok
    \uses{def:chain_model}
    For a unit vector $\Omega$ on the chain and disjoint sets of sites $X,T$,
    $S(X\cup T)\le S(X)+S(T)$ and $S(X)\le S(X\cup T)+S(T)$; the empty set and
    the whole chain have entropy zero, and $S(A)\le|A|\log q$. The
    \emph{boundary edges} of a set $A$ are the edges $\{p,p+1\}$ of the chain
    with exactly one endpoint in $A$. A prefix $[0,k)$ has at most one boundary edge.
    If every interval has entropy at most
    $C\ge0$, then every subset $A$ has entropy at most $C\,|\partial A|$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    Subadditivity and purity are the finite-product entropy inequalities of the
    companion library; the Araki--Lieb inequality is purity,
    $S(X)=S(\Lambda\setminus X)$, followed by subadditivity for the disjoint
    union $\Lambda\setminus X=(\Lambda\setminus(X\cup T))\cup T$. The dimension
    bound is the entropy bound $\log\dim$ for a trace-one positive state. For
    the last claim, induct on the number of sites of $A$. If $A$ is empty or
    the whole chain its entropy vanishes. Otherwise split off the maximal run
    $I$ of consecutive sites starting at the least position of $A$.
    Subadditivity gives $S(A)\le S(I)+S(A\setminus I)\le C+S(A\setminus I)$,
    every boundary edge of $A\setminus I$ is a boundary edge of $A$, and $A$
    has at least one more boundary edge (the edge at the right end of $I$ when
    $A\setminus I$ is nonempty, and an edge at an end of $I$ otherwise, since
    $I$ is then a proper nonempty subset of the chain).
\end{proof}

        +1024\,e\,B_{\mathrm{win}}(q,R).
        \label{eq:chain_window_area_constant}
    \end{align}
    These are the factor $5+E_R+128B_R$ and the constant $C_0(q,R)$ of
    Theorem~\ref{thm:chain_area_law} with the crossing count $N_R$ replaced by
    $2R$ and the cut budget $B_R$ by $B_{\mathrm{win}}$; they are fixed before
    the chain length, Hamiltonian, ground vector and region. Let $L$ be any
    natural number and let $H=\sum_X h_X$ be a finite-range Hamiltonian on
    its chain, with local dimension $q$ and term norms at most $J$ as in
    Definition~\ref{def:chain_model}. Suppose that every $X$ with
    $h_X\ne0$ has the form $X=[a,a+R+1)$ for some natural number $a$
    satisfying $a+R+1\le L$.
    Let $E_0$ be real and let $\Omega$ be a unit vector satisfying
    $H\Omega=E_0\Omega$ and
    $H-E_0\ge\Delta(1-|\Omega\rangle\langle\Omega|)$.
    For every set $A$ of sites,
    \begin{align}
        S_\Omega(A)
         & \le C_{0,\mathrm{win}}(q,R)(1+\vartheta)|\partial A|.
        \label{eq:chain_window_area_law}
    \end{align}
    This combines the consecutive-window count of
    \cite[Part~I, Remark~6.3]{TNLean2026OneDimensional} with the constants of
    Theorem~\ref{thm:chain_area_law}.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_window_cut_budget,thm:chain_family_budget_discount,
        thm:chain_interval_enlargement_size,lem:chain_entropy_inequalities}
    Restrict the interaction sum to its nonzero terms. The sum is still
    the original $H$, and each term retains its support, Hermiticity
    and norm bound $J$. At most $2R$ nonzero terms cross an interval, and
    Lemma~\ref{thm:chain_window_cut_budget} bounds every interval cut budget
    by $B_{\mathrm{win}}$. Choose
    $m=\lceil\max\{4,\min\{64Rq^{2(R+1)},128\,e\,R\ell_R^2\}\vartheta,
        128\sqrt{(1+\vartheta)B_{\mathrm{win}}}\}\rceil$.
    Lemma~\ref{thm:chain_family_budget_discount}, Araki--Lieb and
    Lemma~\ref{thm:chain_interval_enlargement_size} give
    \begin{align}
        S(X)
         & \le8mR\log q+\frac{4096\,e\,\vartheta B_{\mathrm{win}}}{m}+m\log2\notag \\
         & \le C_{0,\mathrm{win}}(q,R)(1+\vartheta)\notag
    \end{align}
    for every interval $X$, as in the proof of
    Theorem~\ref{thm:chain_area_law}. The interval-to-subset
    inequality of Lemma~\ref{lem:chain_entropy_inequalities} proves
    (\ref{eq:chain_window_area_law}).
\end{proof}

\begin{corollary}[Polynomial dependence on the interaction range]
    \label{thm:chain_window_polynomial_range}
    \lean{TNLean.MPS.AreaLaw.chainAreaLaw_of_consecutive_terms_polynomial_range}
    \leanok
    \uses{thm:chain_window_area_law}
    Put $c=\log2$ and fix the absolute constants
    \begin{align}
        C & =1197+576e+\frac{1024e}{c},\notag \\
        K & =c^{-2}+2(c^{-1}+2)^2.\notag
    \end{align}
    For every natural number $R$, every integer $q\ge1$, every natural
    chain length $L$, and every finite-range Hamiltonian $H=\sum_X h_X$
    with term norms at most $J\ge0$, suppose that each nonzero $h_X$ is
    assigned to a full window $X=[a,a+R+1)$ with $a+R+1\le L$.
    For every unit ground vector $\Omega$ satisfying the original global
    projector gap inequality with gap $\Delta>0$, and every set $A$ of
    sites,
    \begin{align}
        S_\Omega(A)
         & \le CK(1+J/\Delta)R\bigl[\log(q^R)\bigr]^3|\partial A|.
        \label{eq:chain_window_range_entropy}
    \end{align}
    In particular, $C$ and $K$ are chosen before $q,R,L,J,\Delta,H,\Omega,A$.
    This is the polynomial range dependence in
    \cite[Part~I, Remark~6.3]{TNLean2026OneDimensional} for the
    consecutive-window interaction family.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_window_area_law,thm:chain_range_zero_entropy,
        thm:chain_entropy_dim_one}
    If $R=0$, Lemma~\ref{thm:chain_range_zero_entropy} gives
    $S_\Omega(A)=0$.
    If $q=1$, Lemma~\ref{thm:chain_entropy_dim_one} gives the same
    conclusion. The right-hand side of
    (\ref{eq:chain_window_range_entropy}) vanishes in both cases.

    It remains to consider $R\ge1$ and $q\ge2$. Put $x=R\log q\ge c$
    and abbreviate $B=B_{\mathrm{win}}(q,R)\ge1$.
    The energy entry of $C_{1,\mathrm{win}}$ is at most
    $128\,e\,R\ell_R^2\le64eB$, so $C_{1,\mathrm{win}}\le(133+64e)B$; also
    $8x+c\le9x$ and $1024eB\le(1024e/c)Bx$. Formula
    (\ref{eq:chain_window_area_constant}) therefore gives
    $C_{0,\mathrm{win}}\le CBx$. Also $R+1\le2R$ and
    $1\le c^{-2}Rx^2$, so
    \begin{align}
        B
         & =1+2R\bigl[1+(R+1)\log q\bigr]^2\notag       \\
         & \le\bigl[c^{-2}+2(c^{-1}+2)^2\bigr]Rx^2
        =KRx^2,\notag                                   \\
        C_{0,\mathrm{win}}(q,R)
         & \le CKRx^3=CKR\bigl[\log(q^R)\bigr]^3.\notag
    \end{align}
    Substitution in~(\ref{eq:chain_window_area_law}) proves
    (\ref{eq:chain_window_range_entropy}).
\end{proof}

\begin{definition}[Finite domains, supports and admissible Hamiltonians]
    \label{def:domain_model}
    \lean{
        TNLean.MPS.AreaLaw.sitePos,
        TNLean.MPS.AreaLaw.sitePos_mem,
        TNLean.MPS.AreaLaw.exists_sitePos_eq,
        TNLean.MPS.AreaLaw.sub_le_sitePos_sub,
        TNLean.MPS.AreaLaw.val_eq_add_one_of_sitePos_eq_add_one,
        TNLean.MPS.AreaLaw.domainGraph,
        TNLean.MPS.AreaLaw.sitePos_sub_val_of_walk,
        TNLean.MPS.AreaLaw.IsDomainSupport,
        TNLean.MPS.AreaLaw.DomainSupport,
        TNLean.MPS.AreaLaw.IsDomainSupport.exists_walk,
        TNLean.MPS.AreaLaw.IsDomainSupport.isChainSupport,
        TNLean.MPS.AreaLaw.IsDomainSupport.subset_or_disjoint_of_gap,
        TNLean.MPS.AreaLaw.DomainHamiltonian,
        TNLean.MPS.AreaLaw.DomainHamiltonian.operator,
        TNLean.MPS.AreaLaw.DomainHamiltonian.operator_isHermitian,
        TNLean.MPS.AreaLaw.DomainHamiltonian.toChain,
        TNLean.MPS.AreaLaw.DomainHamiltonian.toChain_operator,
        TNLean.MPS.AreaLaw.domainBoundaryEdges,
        TNLean.MPS.AreaLaw.domainRegion,
        TNLean.MPS.AreaLaw.IsDomainEdge,
        TNLean.MPS.AreaLaw.card_domainBoundaryEdges
    }
    \leanok
    \uses{def:chain_model}
    A finite set $\Lambda\subset\mathbb Z$ carries the induced path graph, whose
    edges join consecutive integers that both lie in $\Lambda$, and its graph
    distance $d_\Lambda$. Its sites are labelled $0,\ldots,|\Lambda|-1$ in
    increasing order, each carrying $\mathbb C^q$. A \emph{support of range $R$}
    in $\Lambda$ is a nonempty set of sites of $d_\Lambda$-diameter at most $R$.
    Along a path of the induced graph the position of a site minus its label is
    constant, so a support lies between two consecutive gaps of $\Lambda$ and
    is a support of range $R$ of the chain of $|\Lambda|$ sites. An
    \emph{admissible Hamiltonian} assigns to every support $X$ of range $R$ in
    $\Lambda$ a Hermitian term $h_X$ acting as the identity off $X$ with
    operator norm at most $J$, zero terms allowed; with zero terms on the
    remaining supports of the chain it is a finite-range Hamiltonian on the
    chain with the same Hermitian sum. The \emph{edge boundary} $\partial_\Lambda A$ of a
    set $A\subseteq\mathbb Z$ is the set of edges $\{x,x+1\}\subseteq\Lambda$
    with exactly one endpoint in $A$; these edges correspond to the boundary
    edges, on the chain, of the sites in $A$ that join sites at consecutive
    integers.
\end{definition}

\begin{lemma}[Interaction norm budget on a finite integer domain]
    \label{lem:domain_site_norm_budget}
    \lean{TNLean.MPS.AreaLaw.IsDomainSupport.singleton,
        TNLean.MPS.AreaLaw.DomainSupport.toChainSupport,
        TNLean.MPS.AreaLaw.DomainSupport.toChainSupport_injective,
        TNLean.MPS.AreaLaw.DomainHamiltonian.toChain_term,
        TNLean.MPS.AreaLaw.DomainHamiltonian.norm_bound_nonneg,
        TNLean.MPS.AreaLaw.DomainHamiltonian.sum_norm_terms_containing_le}\leanok
    \uses{def:domain_model}
    In the setting of Definition~\ref{def:domain_model}, every site $x$ satisfies
    $J\ge0$ and
    \begin{align}
        \sum_{X\ni x}\|h_X\| & \le 2^{2R+1}J.\notag
    \end{align}
    The sum ranges over the original admissible domain supports.
    This is a consequence of the interaction assumptions
    in~\cite[Part~I]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_model,lem:chain_site_norm_budget}
    The singleton $\{x\}$ is admissible, so
    $0\le\|h_{\{x\}}\|\le J$.
    Extend the interactions by zero to the chain of increasing labels.
    The inclusion of domain supports into chain supports is injective and
    preserves both the terms and membership of $x$.
    Restricting the nonnegative chain sum of
    Lemma~\ref{lem:chain_site_norm_budget} gives the bound.
\end{proof}

\begin{lemma}[Crossing count on a finite integer domain]
    \label{lem:domain_crossing_count}
    \lean{TNLean.MPS.AreaLaw.card_domain_crossingTerms_le_mul_card_boundaryEdges,
        TNLean.MPS.AreaLaw.card_domain_crossingTerms_mul_two_le}\leanok
    \uses{def:domain_model}
    In the setting of Definition~\ref{def:domain_model}, let $D$ be any set of
    labelled sites and let $\mathcal C$ be the original admissible supports
    meeting both $D$ and its complement. Write $\partial_{\mathrm{ch}}D$
    for the edge boundary of $D$ in the chain of increasing labels and put
    $N_R=2\cdot2^{2R}$. Then
    \begin{align}
        |\mathcal C|  & \le 2^{2R}|\partial_{\mathrm{ch}}D|,\notag \\
        2|\mathcal C| & \le N_R|\Lambda|.\notag
    \end{align}
    The chain boundary is used only for this upper bound; it need not equal
    the physical edge boundary in $\Lambda$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_model,lem:domain_site_norm_budget,lem:chain_boundary_crossing_count}
    As in the proof of Lemma~\ref{lem:domain_site_norm_budget}, the inclusion
    into chain supports is injective and preserves both crossing witnesses. Apply Lemma~\ref{lem:chain_boundary_crossing_count}, then use
    $|\partial_{\mathrm{ch}}D|\le |\Lambda|$ and multiply by two.
\end{proof}

\begin{definition}[Doubled domain interactions and regional swaps]
    \label{def:domain_doubled_swapped_hamiltonian}
    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.doubled,
        TNLean.MPS.AreaLaw.DomainHamiltonian.doubled_operator,
        TNLean.MPS.AreaLaw.DomainHamiltonian.swappedDoubled,
        TNLean.MPS.AreaLaw.DomainHamiltonian.swappedDoubled_operator}\leanok
    \uses{def:domain_model,def:qc_site_pair_coordinates,def:peps_frame_sheet_swap}
    Let $H=\sum_Xh_X$ be an admissible domain Hamiltonian and let
    $D\subseteq V=\{0,\ldots,|\Lambda|-1\}$. Let $p$ pair the two replica
    coordinates at each site and put $F=p_*F_D$. Define
    \begin{align}
        \widehat h_X  & =p_*(h_X\otimes\mathbb 1+\mathbb 1\otimes h_X),\notag                \\
        \widehat h'_X & =F\widehat h_XF^\dagger,\notag                                       \\
        \widehat H    & =\sum_X\widehat h_X=p_*(H\otimes\mathbb 1+\mathbb 1\otimes H),\notag \\
        \widehat H'   & =\sum_X\widehat h'_X=F\widehat H F^\dagger.
        \label{eq:domain_doubled_swapped}
    \end{align}
    Both families are indexed by the original admissible domain supports.
    Their terms are Hermitian, supported on $X$, and have norm at most $2J$.
    Thus they have local dimension $q^2$, range $R$ and interaction bound $2J$.
    Empty domains and $q=0$ are allowed. These are the two-replica operators
    in \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{definition}
\begin{proof}\leanok
    \uses{def:domain_model,def:chain_doubled_hamiltonian,
        def:chain_swapped_doubled_hamiltonian,lem:domain_site_norm_budget,
        thm:peps_sheet_swap_conjugate_support}
    At each original nonempty support $X$, choose a site of $X$ to obtain
    $J\ge0$. Extending the original interactions by zero to the chain of
    increasing labels preserves the term at $X$ and its support. The
    doubled and swapped chain estimates therefore apply at that same $X$.
    Conjugation preserves Hermiticity and the regional swap preserves support.
    Distributing the finite sums proves (\ref{eq:domain_doubled_swapped}).
\end{proof}

\begin{lemma}[Crossing decomposition for a domain regional swap]
    \label{thm:domain_partial_swap_crossing}
    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.partialSwap_term_difference_spec,
        TNLean.MPS.AreaLaw.DomainHamiltonian.partialSwap_difference_eq_sum_crossing}\leanok
    \uses{def:domain_doubled_swapped_hamiltonian,lem:domain_crossing_count}
    With the notation of Definition~\ref{def:domain_doubled_swapped_hamiltonian},
    put $T_{X,0}=p_*(h_X\otimes\mathbb 1)$,
    $T_{X,1}=p_*(\mathbb 1\otimes h_X)$, and
    $K_{X,b}=FT_{X,b}F^\dagger-T_{X,b}$.
    Every $K_{X,b}$ is Hermitian, supported on $X$, and satisfies
    $\|K_{X,b}\|\le2J$. Let $\mathcal C$ be the original admissible supports
    meeting both $D$ and $V\setminus D$, and put $N_R=2\cdot2^{2R}$. Then
    \begin{align}
        \widehat H'-\widehat H
                      & =\sum_{X\in\mathcal C}\sum_{b\in\{0,1\}}K_{X,b},\notag \\
        2|\mathcal C| & \le N_R|\Lambda|.\notag
    \end{align}
    The count here is a bound by the total number of domain sites.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_doubled_swapped_hamiltonian,thm:chain_partial_swap_boundary,
        thm:peps_sheet_swap_doubled_cancellation,lem:domain_crossing_count}
    The per-replica support, Hermiticity and norm estimates follow from
    the corresponding chain estimates at the same original support.
    If $X\subseteq D$ or $X\cap D=\varnothing$, then
    \begin{align*}
        F(T_{X,0}+T_{X,1})F^\dagger=T_{X,0}+T_{X,1}.
    \end{align*}
    Thus $K_{X,0}+K_{X,1}=0$ for every noncrossing support.
    Restrict the finite sum to crossing supports, then distribute conjugation
    over the two replica terms. Lemma~\ref{lem:domain_crossing_count} gives
    the displayed count.
\end{proof}

\begin{lemma}[Relative-generator localization on a finite integer domain]
    \label{thm:domain_relative_generator_localization}
    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.norm_relativeGenerator_sub_siteExpectation_le}
    \leanok
    \uses{def:domain_doubled_swapped_hamiltonian,thm:domain_partial_swap_crossing,
        def:qc_site_expectation,thm:domain_interval_neighborhood}
    Let $H$ be an admissible domain Hamiltonian with local dimension $q\ge1$,
    range $R$ and real interaction bound $J$. For $D\subseteq V$, put
    $K=\widehat H$, $K'=\widehat H'$ and
    $Z=\bigcup_{X\in\mathcal C_D}X$, using the original crossing supports.
    For $m\in\mathbb N$ and $t\in\mathbb R$, define
    $B=N_m(Z)$ using intrinsic graph distance and
    $G(t)=e^{itK'}(K'-K)e^{-itK'}$. Put
    \begin{align}
        b_0 & =2^{2R+1}(2J),\notag                          \\
        a   & =\frac{2(2+4|b_0|e^{2R})e}{1-e^{-1/2}},\notag \\
        v   & =2(R+1)|b_0|e^{2R}+1,\notag                   \\
        N_R & =2\cdot2^{2R}.\notag
    \end{align}
    Then
    \begin{align}
        \|G(t)-E_B(G(t))\|
         & \le4|J|a(R+1)^2N_R|\Lambda|e^{v|t|-m/2}.\notag
    \end{align}
    Empty domains and crossing families are allowed. This is an auxiliary
    generator estimate in the localization argument of
    \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:domain_partial_swap_crossing,lem:domain_site_norm_budget,
        thm:domain_lieb_robinson_localization,
        lem:qc_site_expectation_properties,thm:qc_site_expectation_supported_approximant}
    The swapped domain family has interaction bound $2J$, so its per-site
    norm budget is $b_0$. Domain localization gives an error
    $a|X|^2\|A_{X,b}\|e^{v|t|-m/2}$ for each evolved replica perturbation
    at its own neighborhood $N_m(X)$.
    Since $X\subseteq Z$, that approximant is supported on $B$.
    Theorem~\ref{thm:qc_site_expectation_supported_approximant},
    $|X|\le R+1$ and $\|A_{X,b}\|\le2J\le2|J|$ bound its projection error by
    $4|J|a(R+1)^2e^{v|t|-m/2}$.
    Linearity of evolution and expectation permits summing over the exact
    crossing decomposition. The bound
    $2|\mathcal C_D|\le N_R|\Lambda|$ finishes the estimate.
\end{proof}

\begin{lemma}[Supported Gaussian averaging on a finite integer domain]
    \label{thm:domain_supported_gaussian_integral}
    \lean{TNLean.MPS.AreaLaw.exists_supported_gaussian_integral_domain}\leanok
    \uses{def:domain_doubled_swapped_hamiltonian,def:qc_site_expectation}
    For every range $R$ and real interaction bound $J$ there are constants
    $C,v>0$ with the following property. Let $q\ge1$, let $H$ be an admissible
    Hamiltonian on a finite integer domain $\Lambda$, let $D\subseteq V$, and
    let $m\in\mathbb N$. Put $K=\widehat H$, $K'=\widehat H'$ and
    $Z=\bigcup_{X\in\mathcal C_D}X$. Write $N_m$ for the intrinsic graph
    neighborhood of radius $m$.

    There is a differentiable path $U:\mathbb R\to M_{q^{2|\Lambda|}}(\mathbb C)$
    of unitary matrices, supported on $N_m(Z)$ at every real time, satisfying
    \begin{align}
        U(0)      & =\mathbb 1,\notag                                          \\
        \dot U(t) & =iE_{N_m(Z)}\bigl(e^{itK'}(K'-K)e^{-itK'}\bigr)U(t).\notag
    \end{align}
    The same path works for every set $Q\subseteq V$, operator $W$ supported
    on $Q$, variance $\eta\ge0$ and cutoff $T\ge0$. For the centered Gaussian
    probability measure $\gamma_\eta$, define
    \begin{align}
        M   & =\int_{[-T,T]}E_{N_m(Q)}(e^{itK'}We^{-itK'})U(t)\,d\gamma_\eta(t),\notag \\
        M_0 & =\int_{[-T,T]}e^{itK'}We^{-itK}\,d\gamma_\eta(t).\notag
    \end{align}
    Then $M$ is supported on $N_m(Q\cup Z)$, has $\|M\|\le\|W\|$, and
    \begin{align}
        \|M-M_0\| & \le\|W\|C|\Lambda|^2(1+T)e^{vT-m/2}.\notag
    \end{align}
    This includes empty domains, zero variance and zero cutoff.
    It is the supported averaging step in
    \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_doubled_swapped_hamiltonian,
        thm:qc_supported_gaussian_projection_bounds,
        thm:domain_relative_generator_localization,thm:domain_lieb_robinson_localization,
        lem:domain_site_norm_budget}
    Use the constants $a,v$ from the domain generator estimate and put
    $d=4|J|a(R+1)^2N_R$, $C=a+d$. The actual doubled and swapped operators
    are Hermitian because their terms are Hermitian.
    Theorem~\ref{thm:qc_supported_gaussian_projection_bounds} constructs $U$
    before the observable and averaging data.
    Put
    \begin{align*}
        \varepsilon_g=d|\Lambda|e^{vT-m/2}, \\
        \varepsilon_o=a|\Lambda|^2e^{vT-m/2}.
    \end{align*}
    On $|t|\le T$, the observable projection error is at most
    $\|W\|\varepsilon_o$ and the generator projection error is at most
    $\varepsilon_g$. Both projection regions lie in $N_m(Q\cup Z)$.
    The common averaging theorem gives support, contraction and
    \begin{align*}
        \|M-M_0\|\le\|W\|(\varepsilon_o+T\varepsilon_g).
    \end{align*}
    Since $|\Lambda|\le|\Lambda|^2$ for every finite domain and $a,d,T\ge0$,
    \begin{align*}
        a|\Lambda|^2+d|\Lambda|T\le(a+d)|\Lambda|^2(1+T).
    \end{align*}
    This proves the stated estimate with $C=a+d$.
    The restricted Gaussian measure is not renormalized.
\end{proof}

\begin{lemma}[Entropy comparison after a regional replica swap]
    \label{thm:regional_reset_cores_entropy}
    \lean{TNLean.Lattice.mutualInformation_cores_le_of_reset_unitary}\leanok
    Let $V$ be a finite site set, let $q\ge1$, and let $\Omega$ be a unit
    vector in $(\mathbb C^q)^{\otimes V}$. Suppose $R_0\subseteq D\subseteq V$
    and $C_0\cap D=\varnothing$. On two replicas put
    $\Psi=\Omega\otimes\Omega$ and $\Phi=F_D\Psi$, where $F_D$ exchanges
    the replicas at the sites of $D$.
    Let $U$ be a unitary supported on $S\subseteq V$ in paired-site coordinates,
    with $S\cap(R_0\cup C_0)=\varnothing$.
    If $\|U\Psi-\Phi\|\le\delta$ for a real $\delta$, then
    \begin{align}
        I_\Omega(R_0:C_0) & \le8\sqrt\delta\log(e q^{|R_0|}).\notag
    \end{align}
    This is the entropy comparison in the information-reset argument of
    \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:lattice_supported_unitary_entropy,thm:peps_reset_exchanged_cores,
        thm:peps_reset_paired_support}
    Both replicas are normalized, and $F_D$ is unitary. The first-replica
    marginal of $\Phi$ on $R_0\cup C_0$ is a product, by
    Theorem~\ref{thm:peps_reset_exchanged_cores}.
    Put $E=\{0,\ldots,q-1\}$. Let
    $P:E^V\times E^V\to(E\times E)^V$ be the pairing bijection
    $P(c_1,c_2)(v)=(c_1(v),c_2(v))$. Write $\psi$ and $F$ for the
    paired-coordinate forms of $\Psi$ and $F_D$, respectively. Then
    \begin{align}
        \Psi(c_1,c_2) & =\psi(P(c_1,c_2))=\Omega(c_1)\Omega(c_2),\notag \\
        \Phi(c_1,c_2) & =(F\psi)(P(c_1,c_2)).\notag
    \end{align}
    The support assumption means that $U$ acts trivially on both cores.
    The supported-unitary entropy comparison of
    Lemma~\ref{lem:lattice_supported_unitary_entropy}, and the first core dimension
    $q^{|R_0|}$, give the assertion. Nonnegativity of the norm bound implies
    $\delta\ge0$.
\end{proof}

\begin{lemma}[Information reset on a finite integer domain]
    \label{thm:domain_information_reset}
    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.information_reset}\leanok
    \uses{def:domain_model}
    Fix $q\ge1$, a natural range $R$, a real interaction bound $J$, a gap
    $\Delta>0$ and an exponent $p>0$. There exist $C_p>0$ and a natural
    $L_0\ge1$, depending only on these parameters, such that the following
    holds for every finite integer domain $\Lambda$ with $L=|\Lambda|\ge L_0$.
    Let $H$ be an admissible domain Hamiltonian and $\Omega$ a unit ground
    vector with energy $E_0$ and full-system gap at least $\Delta$.
    Let $D,C\subseteq\Lambda$ be disjoint, $R_0\subseteq D$, $C_0\subseteq C$,
    and $b\in\mathbb R$ with $I_\Omega(D:C)\le b$. Put
    $w=C_p(1+b+\log L)$. Suppose
    \begin{align}
        w & <d_\Lambda(x,y) &  & (x\in R_0,\ y\notin D),\notag \\
        w & <d_\Lambda(x,y) &  & (x\in C_0,\ y\notin C),\notag
    \end{align}
    where $d_\Lambda$ is the intrinsic extended graph distance. Then
    $I_\Omega(R_0:C_0)\le L^{-p}$.
    Infinite distances satisfy either margin automatically; empty cores are
    allowed. This extends the chain information-reset
    lemma in \cite[Part~IIa]{TNLean2026OneDimensional}; see also
    \cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}\leanok

\begin{lemma}[Intrinsic neighborhoods of order intervals]
    \label{thm:domain_interval_neighborhood}
    \lean{TNLean.MPS.AreaLaw.domainNeighborhood_eq_interval_of_ordConnected,
        TNLean.MPS.AreaLaw.domainNeighborhood_interval_eq}\leanok
    \uses{def:domain_model,def:chain_model}
    In the setting of Definition~\ref{def:domain_model}, identify the domain
    with its increasing labels $V=\{0,\ldots,|\Lambda|-1\}$.
    For $m\in\N$ and $I\subseteq V$, put
    \begin{align}
        N_m(I) & =\bigcup_{x\in I}\{y\in V:d_\Lambda(x,y)\le m\}.\notag
    \end{align}
    If $I$ is order-convex in $V$, there exist natural numbers $a,b$ such that
    $N_m(I)=[a,b)$, with the interval taken in $V$.
    In particular, the intrinsic neighborhood of a relative interval is
    again a relative interval. Empty domains and empty intervals are allowed.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_model,def:chain_model}
    Every edge of the induced graph joins consecutive labels. Induction on
    a walk therefore shows that it visits every label between its endpoints.
    If $d_\Lambda(x,y)\le m$ and $z$ lies between $x$ and $y$, the initial
    segment of a shortest walk ending at $z$ gives $d_\Lambda(x,z)\le m$.

    Let $u\le z\le v$ with $u,v\in N_m(I)$, and choose $x,y\in I$ with
    $d_\Lambda(x,u)\le m$ and $d_\Lambda(y,v)\le m$.
    If $z<x$, use the walk from $x$ to $u$.
    Otherwise, if $z\le y$, order convexity gives $z\in I$.
    In the remaining case $y<z$, use the walk from $y$ to $v$.
    Thus $N_m(I)$ is order-convex. If it is nonempty, take $a$ to be its
    least label and $b$ one plus its greatest label; if it is empty, take
    $a=b=0$.
\end{proof}

\begin{lemma}[Entropy of intrinsic neighborhoods of interval unions]
    \label{thm:domain_interval_neighborhood_entropy}
    \lean{TNLean.MPS.AreaLaw.regionEntropy_domainNeighborhood_biUnion_intervals_le,
        TNLean.MPS.AreaLaw.DomainHamiltonian.regionEntropy_neighborhood_biUnion_intervals_le}
    \leanok
    \uses{def:domain_model}
    Let $\Omega$ be a normalized vector on the finite domain of
    Definition~\ref{def:domain_model}. Suppose that every relative interval has
    entropy at most $c$. For any finite index set $s$, relative intervals
    $I_\alpha$ and $m\in\mathbb N$,
    \begin{align}
        S_\Omega\left(N_m\left(\bigcup_{\alpha\in s}I_\alpha\right)\right)
         & \le |s|c.\notag
    \end{align}
    The intervals may overlap or be empty, and $s$ may be empty.
    If $\Omega$ is the original gapped ground vector of an admissible domain
    Hamiltonian, with $q\ge1$, $J\ge0$ and $\Delta>0$, one may take
    $c=C_{\mathrm{int}}(q,R,J,\Delta)$.
    This is the interval-neighborhood entropy estimate used in
    \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:domain_interval_neighborhood,thm:chain_area_law,
        lem:chain_entropy_inequalities,lem:zero_entropy_flip,def:domain_model}
    Submodularity and nonnegativity of reduced-state entropy give
    $S_\Omega(X\cup Y)\le S_\Omega(X)+S_\Omega(Y)$.
    Since $S_\Omega(\varnothing)=0$, finite subadditivity gives
    \begin{align}
        S_\Omega\left(\bigcup_{\alpha\in s}N_m(I_\alpha)\right)
         & \le\sum_{\alpha\in s}S_\Omega(N_m(I_\alpha))\le |s|c.\notag
    \end{align}
    Each $N_m(I_\alpha)$ is itself a relative interval by
    Lemma~\ref{thm:domain_interval_neighborhood}, and
    \begin{align}
        N_m\left(\bigcup_{\alpha\in s}I_\alpha\right)
         & =\bigcup_{\alpha\in s}N_m(I_\alpha).\notag
    \end{align}
    For a domain Hamiltonian, extend its terms by zero to the
    chain of increasing labels. Its operator and original ground vector
    are unchanged, so Theorem~\ref{thm:chain_area_law} supplies the interval
    entropy bound.
\end{proof}

\begin{lemma}[Information at logarithmic physical separation on a finite domain]
    \label{thm:domain_logarithmic_separation}
    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.mutualInformation_le_of_logarithmic_separation}
    \leanok
    \uses{def:domain_model,def:chain_model}
    Fix $q\ge1$, a natural range $R$, a real interaction bound $J$, a gap
    $\Delta>0$, an exponent $p>0$ and a natural $k_0\ge1$.
    There exist $D_p>0$ and a natural $L_0\ge1$, depending only on these
    parameters, with the following property. Let $\Lambda\subset\mathbb Z$
    be finite with $L=|\Lambda|\ge L_0$, and let $H$ be an admissible
    domain Hamiltonian with unit ground vector $\Omega$, energy $E_0$ and
    full-system gap at least $\Delta$.
    Suppose $T$ is a union of at most $k_0$ relative intervals in the
    increasing order on $\Lambda$. If $E\subseteq\Lambda$ satisfies
    $D_p\log L\le|x-y|$ for every $x\in T$ and $y\in E$, using the original
    integer positions, then $I_\Omega(T:E)\le L^{-p}$.
    The interval family may overlap or contain empty members.
    This extends the chain logarithmic-separation lemma in
    \cite[Part~IIa]{TNLean2026OneDimensional}; see also
    \cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:domain_information_reset,thm:domain_interval_neighborhood_entropy,
        lem:domain_site_norm_budget}
    Put $c=\max\{0,C_{\mathrm{int}}\}$ and $b=4k_0c$, where
    $C_{\mathrm{int}}$ is the interval entropy bound.
    Choose $C_p,L_{\mathrm{reset}}$ from
    Lemma~\ref{thm:domain_information_reset}, and take
    \begin{align}
        D_p & =3C_p(2+b)+4,\notag                               \\
        L_0 & =\max\{L_{\mathrm{reset}},\lceil e\rceil\}.\notag
    \end{align}
    Write $w=C_p(1+b+\log L)$ and $n=\lceil w+1\rceil$.
    Then $w+1\le n<w+2$ and $2n+w<3w+4\le D_p\log L$.
    Use intrinsic natural-radius neighborhoods to define
    $D=N_n(T)$ and $C=\Lambda\setminus N_{2n}(T)$.
    We have $D\cap C=\varnothing$ and $T\subseteq D$.

    Whenever $d_\Lambda(x,y)\le m<\infty$, a shortest walk gives
    $|x-y|\le m$ in the original integer coordinates.
    This comparison and physical separation give $E\subseteq C$.
    If $x\in T$, $y\notin D$, and $d_\Lambda(x,y)=d<\infty$, then $d>n>w$.
    If $e\in E$, $y\notin C$, and $d_\Lambda(e,y)=d<\infty$, choose
    $x\in T$ with $d_\Lambda(x,y)\le2n$. The triangle inequality gives
    $|x-e|\le2n+d$. Thus $d\le w$ would contradict
    $2n+w<D_p\log L\le|x-e|$.
    Both strict reset margins follow, including pairs at infinite distance.

    The size threshold supplies a site, whose singleton interaction implies
    $J\ge0$. Lemma~\ref{thm:domain_interval_neighborhood_entropy} gives
    $S_\Omega(D)\le k_0c$, and hence
    $I_\Omega(D:C)\le2S_\Omega(D)\le b$.
    Applying Lemma~\ref{thm:domain_information_reset} proves the claim.
\end{proof}

\begin{lemma}[Connected components of an induced integer graph]
    \label{thm:chain_integer_domain_components}
    \lean{
        SimpleGraph.hasse_int_induce_reachable_iff,
        SimpleGraph.ConnectedComponent.ordConnected_image_hasse_int_induce,
        SimpleGraph.ConnectedComponent.nonempty_pathGraph_iso_hasse_int_induce_finset
    }
    \leanok
    Let $S\subseteq\mathbb Z$, and let $G$ be the graph on $S$ in which two
    vertices are adjacent exactly when their integer coordinates differ by one.
    For any $x,y\in S$,
    \begin{align}
        x\text{ and }y\text{ are joined by a walk in }G
        \quad\Longleftrightarrow\quad
        \lbrack\min\{x,y\},\max\{x,y\}\rbrack\cap\mathbb Z\subseteq S.
    \end{align}
    The coordinate image of each connected component is order-convex:
    if $x,y$ belong to that image, every integer between $x$ and $y$ belongs to
    the same image. No finiteness assumption on $S$ is needed.

    If $S$ is finite, the graph induced on each connected component $C$ is
    isomorphic to the path $P_{|C|}$ on $|C|$ vertices. Here $P_n$ has vertices
    $0,\ldots,n-1$ and edges between consecutive vertices. This includes
    singleton components and requires no further hypothesis on a component.
    This is the induced integer geometry used in the reduction to connected
    domains in \cite[Part~I]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    Fix an integer $z\notin S$. If two vertices are adjacent, their integer
    coordinates are consecutive and neither equals $z$. Thus either both
    coordinates are less than $z$, or neither is. This property is preserved
    along a walk. A walk therefore cannot join vertices on opposite sides of
    a missing integer, which proves the forward implication.

    Conversely, suppose first that $x\le y$ and that every integer in $[x,y]$
    belongs to $S$. Induction by integer successors joins $x$ to each integer
    $n$ with $x\le n\le y$: the first vertex is $x$, and the step uses the
    edge from $n$ to $n+1$ when $n+1\le y$. This reaches $y$. Reversing the
    walk handles $y\le x$.

    If $x,y$ are in one connected component and $z$ lies between them, the
    criterion places $z$ in $S$. Every integer between $x$ and $z$ is also
    between $x$ and $y$, so the reverse implication joins $x$ to $z$. Hence
    $z$ lies in that same component.

    For finite $S$, give a component $C$ the order inherited from the integers.
    Its coordinate map is an order embedding with order-convex image, so it
    preserves and reflects the covering relation. The increasing enumeration
    of the finite linear order $C$ is an order isomorphism from
    $\{0,\ldots,|C|-1\}$ to $C$. In either graph, adjacency is the symmetric
    covering relation. This enumeration is therefore a graph isomorphism
    from $P_{|C|}$ to the graph induced on $C$.
\end{proof}

\begin{lemma}[Distance spheres in a finite integer domain]
    \label{thm:domain_graph_spheres}
    \lean{TNLean.MPS.AreaLaw.domainGraph_card_sphere_le_two,
        TNLean.MPS.AreaLaw.domainGraphIsoHasse}

\end{lemma}

\begin{lemma}[Propagation on a finite integer domain]

    and let $q,R\in\mathbb N$. Let $(h_k)_k$ be a finite labelled family of
    Hermitian operators, each supported on a nonempty set $X_k$ of graph
    diameter at most $R$. Suppose $b_0\in\mathbb R$ satisfies
    $\sum_{k:x\in X_k}\|h_k\|\le b_0$ for every site $x$.

    \begin{align}
        C & =2+4|b_0|e^{2R},\notag      \\
        v & =2(R+1)|b_0|e^{2R}+1.\notag
    \end{align}

    $\inf\varnothing=\infty$ and $e^{-\infty/2}=0$. For every real $t$,
    \begin{align}
        \|[\tau_t(A),B]\|
        \le C|S|\|A\|\|B\|e^{v|t|}e^{-\delta/2}.\notag
    \end{align}
    This extends the commutator estimate of the Lieb--Robinson lemma in
    \cite[Part~II]{TNLean2026OneDimensional} to arbitrary finite integer
    domains. No connectivity, positive local dimension, ground-state or
    spectral-gap hypothesis is required.
\end{lemma}
\begin{proof}\leanok
    \uses{def:domain_model,thm:qc_graph_lieb_robinson_finset,
        thm:qc_graph_lieb_robinson_zero_finset}
    Every range-$R$ support has at most $R+1$ sites. For finite $\delta$,
    the graph propagation estimate with exponential weight parameter one
    gives the stronger decay $e^{-\delta}$, which is at most
    $e^{-\delta/2}$. For infinite $\delta$, every $x\in S$ has
    $d_\Lambda(x,y)=\infty$, so $[\tau_t(A),B]=0$ by
    Theorem~\ref{thm:qc_graph_lieb_robinson_zero_finset} and the left side
    vanishes. All distances are those of the
    induced graph; sites separated by a hole are not identified with
    consecutive sites of a longer chain.
\end{proof}

\begin{lemma}[Localization on a finite integer domain]
    \label{thm:domain_lieb_robinson_localization}
    \lean{TNLean.MPS.AreaLaw.norm_heisenberg_sub_siteExpectation_domainNeighborhood_le}
    \leanok
    \uses{def:domain_model,def:qc_site_expectation}
    Let $\Lambda\subset\mathbb Z$ be finite, let $q,R\in\mathbb N$ with $q\ge1$,
    and let $H=\sum_kh_k$ be a finite labelled sum of Hermitian operators.
    Each $h_k$ is supported on a nonempty set $X_k$ of induced-graph diameter
    at most $R$. Suppose $\sum_{k:x\in X_k}\|h_k\|\le b_0$ for every site $x$.
    Put
    \begin{align}
        C  & =2+4|b_0|e^{2R},\notag         \\
        v  & =2(R+1)|b_0|e^{2R}+1,\notag    \\
        C' & =\frac{2Ce}{1-e^{-1/2}}.\notag
    \end{align}
    For $m\in\mathbb N$ write
    $B_m(x)=\{y\in\Lambda:d_\Lambda(x,y)\le m\}$.

    $N_\ell(S)=\bigcup_{x\in S}B_{\lfloor\ell\rfloor}(x)$.

    operators supported on this set. Every operator $A$ supported on $S$
    satisfies, for every real $t$,
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
        \le C'|S|^2\|A\|e^{v|t|-\ell/2}.\notag
    \end{align}

    hypothesis is imposed. The neighborhood uses the induced graph and
    never joins sites across a missing integer. This is the finite-domain
    extension of the localization conclusion in the Lieb--Robinson lemma
    of \cite[Part~II]{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:domain_graph_spheres,thm:qc_graph_neighborhood_localization,
        thm:area_law_supported_identity}

    localization with this support-size bound, sphere coefficient $2$,
    sphere exponent zero, and natural radius $m=\lfloor\ell\rfloor$.
    Its neighborhood is precisely the displayed union of closed balls.
    Graph distances are integers when finite, and
    $e^{-m/2}\le e^{1/2}e^{-\ell/2}$.
    The natural-radius coefficient is $2Ce^{1/2}/(1-e^{-1/2})$;
    multiplication by this rounding factor gives $C'$.
    The graph localization theorem includes the empty union and its
    vanishing factor $|S|^2$.
\end{proof}

\begin{lemma}[A cut that splits no term]

\begin{lemma}[Flipping a region of zero entropy]
    \label{lem:zero_entropy_flip}
    \lean{
        TNLean.Lattice.regionEntropy_union_add_inter_le,
        TNLean.Lattice.regionEntropy_sdiff_eq_of_eq_zero,
        TNLean.Lattice.regionEntropy_union_eq_of_eq_zero,
        TNLean.Lattice.regionEntropy_symmDiff_eq_of_eq_zero,
        TNLean.MPS.AreaLaw.exists_mem_val_eq_iff,
        TNLean.MPS.AreaLaw.boundaryEdges_symmDiff_interval,
        TNLean.MPS.AreaLaw.exists_regionEntropy_eq_boundaryEdges_eq_symmDiff
    }
    \leanok
    \uses{lem:chain_entropy_inequalities}
    Let $\Omega$ be a unit vector on the chain of $L$ sites and $P$ a set of
    sites with $S(P)=0$. Then $S(A\triangle P)=S(A)$ for every set of sites
    $A$. If $P=[0,p+1)$ with $p+1<L$, the boundary edges of $A\triangle P$ are
    those of $A$ with the edge $\{p,p+1\}$ added or removed. Consequently, if
    $G$ is a set of edges $\{p,p+1\}$ whose prefixes $[0,p+1)$ all have zero
    entropy, every $A$ has a set of sites $A'$ with $S(A')=S(A)$ whose boundary
    edges are those of $A$ toggled exactly at the edges of $G$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:chain_entropy_inequalities}
    For $X\subseteq P$ the Araki--Lieb inequality gives
    $S(X)\le S(P)+S(P\setminus X)=S(P\setminus X)$ and symmetrically, so
    $S(X)=S(P\setminus X)$. For $X\subseteq P$ and $Y$ disjoint from $P$,
    strong subadditivity in the form
    $S(P\cup Y)+S(X)\le S(P)+S(X\cup Y)$ and the Araki--Lieb inequality
    $S(Y)\le S(P\cup Y)+S(P)$ give $S(X)+S(Y)\le S(X\cup Y)$, and
    subadditivity gives the reverse. Writing $A=(A\cap P)\cup(A\setminus P)$
    and $A\triangle P=(P\setminus(A\cap P))\cup(A\setminus P)$, both entropies
    equal $S(A\cap P)+S(A\setminus P)$. Flipping $[0,p+1)$ reverses membership
    at every site up to $p$, so it changes the status of the edge
    $\{p,p+1\}$ only. The last claim follows by induction on $G$.
\end{proof}

    \lean{TNLean.MPS.AreaLaw.DomainHamiltonian.exists_regionEntropy_eq_card_boundaryEdges_eq}

    \end{align}
    Here $\partial A'$ is the edge boundary in the chain of labels.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:zero_entropy_cut,lem:zero_entropy_flip}
    Put $A_0=\{j:x_j\in A\}$. At a gap $x_{j+1}>x_j+1$, every
    support lies on one side of the prefix $[0,j+1)$, since its
    diameter in the induced path graph is finite. Thus this prefix
    has zero entropy by Lemma~\ref{lem:zero_entropy_cut}.
    Apply Lemma~\ref{lem:zero_entropy_flip} to the prefixes at precisely
    those gaps whose label edges lie in $\partial A_0$. The resulting
    set $A'$ has the same entropy as $A_0$, and its boundary consists
    exactly of the edges of $\partial A_0$ joining consecutive integers
    of $\Lambda$. These edges are in bijection with
    $\partial_\Lambda A$.
\end{proof}

\begin{corollary}[Polynomial range dependence on a finite set of integers]
    \label{thm:domain_window_polynomial_range}
    \lean{TNLean.MPS.AreaLaw.domainAreaLaw_of_consecutive_terms_polynomial_range}

    \begin{align}
        C_{\mathrm{abs}}
         & =\left(1197+576e+\frac{1024e}{c}\right)
        \left[c^{-2}+2(c^{-1}+2)^2\right].
        \label{eq:domain_window_constant}
    \end{align}
    For every integer $q\ge1$, $R\in\mathbb N$, $J\ge0$ and $\Delta>0$,
    let $\Lambda\subset\mathbb Z$ be finite and let $H=\sum_Xh_X$ be
    an admissible Hamiltonian of Definition~\ref{def:domain_model},
    with local dimension $q$, range $R$ and term norms at most $J$.
    Write $x_0<\cdots<x_{L-1}$ for the sites of $\Lambda$, where
    $L=|\Lambda|$. Identify each interaction support with its set
    $X\subseteq\{0,\ldots,L-1\}$ of labels. Suppose that whenever
    $h_X\ne0$ there is an integer $a$ such that
    \begin{align}
        \{x_j:j\in X\}
         & =[a,a+R+1)\cap\mathbb Z\subseteq\Lambda.
        \label{eq:domain_full_window}
    \end{align}
    Let $\Omega$ be a unit ground vector of the original $H$, at
    energy $E_0$, satisfying
    $H-E_0\mathbb 1\ge\Delta(\mathbb 1-|\Omega\rangle\langle\Omega|)$.
    Then every finite $A\subseteq\mathbb Z$ satisfies
    \begin{align}
        S_\Omega(A\cap\Lambda)
         & \le C_{\mathrm{abs}}(1+J/\Delta)R
        \bigl[\log(q^R)\bigr]^3|\partial_\Lambda A|.
        \label{eq:domain_window_area_law}
    \end{align}
    The constant $C_{\mathrm{abs}}$ is chosen before all
    $q,R,J,\Delta,\Lambda,H,E_0,\Omega,A$. The full-window condition
    (\ref{eq:domain_full_window}) is the specialization in
    \cite[Part~I, the remark on sharper constants]{TNLean2026OneDimensional}.
\end{corollary}
\begin{proof}
    \leanok
    \uses{def:domain_model,thm:chain_full_support_window,
        thm:domain_entropy_boundary_reduction,thm:chain_window_polynomial_range}
    Order the sites of $\Lambda$ as the chain of $L$ labels. Every
    nonzero interaction has $R+1$ sites and label diameter at most $R$.
    By Lemma~\ref{thm:chain_full_support_window}, its labels form a
    full contained window. Put zero on the remaining chain supports,
    as in Definition~\ref{def:domain_model}. Every original term retains
    its norm bound $J$; the sum, ground vector and global gap are unchanged.

    Choose $A'$ by Lemma~\ref{thm:domain_entropy_boundary_reduction},
    so that
    \begin{align}
        S_\Omega(A')  & =S_\Omega(A\cap\Lambda), &
        |\partial A'| & =|\partial_\Lambda A|.
        \label{eq:domain_window_reduction}
    \end{align}
    Apply Corollary~\ref{thm:chain_window_polynomial_range} at $A'$.
    Its absolute constant is exactly
    (\ref{eq:domain_window_constant}); substituting the two equalities
    above proves (\ref{eq:domain_window_area_law}). The empty domain,
    $R=0$ and $q=1$ are included in that chain estimate.
\end{proof}

\begin{lemma}[Neighborhoods of intervals and their entropy]
    \label{thm:chain_interval_neighborhood_entropy}
    \lean{TNLean.MPS.AreaLaw.chainNeighborhood_interval_eq,
        TNLean.MPS.AreaLaw.regionEntropy_chainNeighborhood_biUnion_intervals_le}
    \leanok
    \uses{def:chain_model}
    Let $I=\{0,\ldots,L-1\}$ be a contiguous chain. Write
    $[a,z)_I=\{x\in I:a\le x<z\}$ for natural endpoints $a,z$, and
    for a real $\rho\ge0$ put
    \begin{align}
        N_\rho(A)=\{y\in I:\ \exists x\in A,\ |x-y|\le\rho\},
        \qquad d=\lfloor\rho\rfloor.
    \end{align}
    Then the exact interval neighborhood is
    \begin{align}
        N_\rho([a,z)_I)=
        \begin{cases}
            \lbrack\max\{0,a-d\},z+d)_I, & \lbrack a,z)_I\ne\varnothing, \\
            \varnothing,                 & [a,z)_I=\varnothing.
        \end{cases}
    \end{align}
    There is no condition $z\le L$ on the original endpoint.

    Let $q$ be any natural number and let $\Omega$ be a normalized
    vector on the $L$ sites with local dimension $q$.
    Suppose a real number $c$ bounds the entropy of every interval:
    \begin{align}
        S_\Omega([a,z)_I)\le c\qquad(a,z\in\mathbb N).
    \end{align}
    For any finite index set $s$ and natural endpoint families $a_i,z_i$,
    put $T=\bigcup_{i\in s}[a_i,z_i)_I$. Then
    \begin{align}
        S_\Omega(N_\rho(T))\le |s|c.
    \end{align}
    Empty intervals and empty families are included, and the intervals
    may overlap or repeat. No nonnegativity assumption on $c$ is added,
    and no Hamiltonian, gap or positive local dimension is assumed.
    These are the interval-neighborhood and entropy steps in Part~II
    of~\cite{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model,lem:chain_entropy_inequalities}
    Coordinate distances are integers, so $|x-y|\le\rho$ is equivalent
    to $|x-y|\le d$. If the interval is nonempty, its first and last
    sites are $a$ and $\min\{z,L\}-1$. A site lies within distance $d$
    of it precisely when it belongs to the displayed clipped interval;
    one may choose the nearer endpoint when the site lies outside, or
    the site itself when it lies inside. If the interval is empty,
    its neighborhood is empty by definition. This explains why an
    empty interval must not be enlarged by its endpoints.

    The entropy of the empty interval is zero, so the interval bound
    at $[0,0)_I$ gives $c\ge0$.
    Each interval neighborhood has entropy at most $c$: it is either
    an interval or the empty set. Moreover,
    \begin{align}
        N_\rho\Bigl(\bigcup_{i\in s}[a_i,z_i)_I\Bigr)
        =\bigcup_{i\in s}N_\rho([a_i,z_i)_I).
    \end{align}
    Entropy submodularity and nonnegativity imply
    $S_\Omega(A\cup B)\le S_\Omega(A)+S_\Omega(B)$ even when $A,B$
    overlap. Induction on the finite index set therefore bounds the
    entropy of this union by the sum of its component entropies,
    which is at most $|s|c$. The empty-family case is $S_\Omega(\varnothing)=0$.
\end{proof}

\begin{lemma}[Information at logarithmic separation on a contiguous chain]
    \label{thm:chain_logarithmic_separation}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.mutualInformation_le_of_logarithmic_separation}
    \leanok
    \uses{def:chain_model}
    Fix a natural local dimension $q\ge1$, a natural range $R$, a real
    interaction bound $J$, a gap $\Delta>0$, an exponent $p>0$, and a
    natural number $k_0\ge1$.
    There are $D_p>0$ and a natural number $L_0\ge1$, depending only on
    these fixed parameters, with the following property.
    For every $L\ge L_0$, let $H$ be a range-$R$ Hamiltonian with
    interaction bound $J$ on the contiguous chain $I=\{0,\ldots,L-1\}$,
    and let $\Omega$ be a normalized ground vector with full-system gap
    at least $\Delta$.
    Let $T\subseteq I$ be a union of at most $k_0$ intervals, and let
    $E\subseteq I$ satisfy
    \begin{align}
        D_p\log L\le|x-y|\qquad(x\in T,\ y\in E).
    \end{align}
    Then
    \begin{align}
        I_\Omega(T:E)\le L^{-p}.
    \end{align}
    Empty intervals, empty $T$ or $E$, and overlapping interval
    representations are allowed. No additional disjointness assumption
    is imposed. The constants precede the length, Hamiltonian, state,
    regions and interval representation.
    This is the contiguous-chain form of the logarithmic-separation
    lemma in Part~II of~\cite{TNLean2026OneDimensional};
    see~\cite{gap:tnlean_one_dimensional_connected_domain_scope} for the
    domain distinction.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:chain_information_reset,thm:chain_area_law,
        thm:chain_interval_neighborhood_entropy}
    Put $\overline C=\max\{0,C_{\mathrm{int}}(q,R,J,\Delta)\}$ and
    $b=4k_0\overline C$. Choose $C_p$ and the length threshold from
    Lemma~\ref{thm:chain_information_reset}, and set
    \begin{align}
        D_p=3C_p(2+b)+3.
    \end{align}
    Enlarge the threshold so that $\log L\ge1$.
    These choices use only the fixed parameters.
    In any such chain there is a site, and its singleton term
    has $0\le\|h_{\{x\}}\|\le J$. Thus $J\ge0$ is derived from the
    model, rather than imposed as a further hypothesis, and
    Theorem~\ref{thm:chain_area_law} applies.
    Put $s=\log L$ and $w=C_p(1+b+s)>0$. Then
    \begin{align}
        D_p s-(3w+3)
        =\bigl(3C_p(1+b)+3\bigr)(s-1)\ge0.
    \end{align}

    For a real $\rho\ge0$, write
    \begin{align}
        N_\rho(T)=\{y\in I:\ \exists x\in T,\ |x-y|\le\rho\}.
    \end{align}
    Choose the regions
    \begin{align}
        D=N_{w+1}(T),\qquad C=I\setminus N_{2w+2}(T),
        \qquad R_0=T,\quad C_0=E.
    \end{align}
    The regions $D,C$ are disjoint, $T\subseteq D$, and $E\subseteq C$.
    If $y\notin D$, then $|x-y|>w+1>w$ for every $x\in T$.
    If $y\notin C$, choose $x\in T$ with $|x-y|\le2w+2$.
    For every $e\in E$, the separation hypothesis and the triangle
    inequality give
    \begin{align}
        |e-y|\ge D_p\log L-(2w+2)\ge w+1>w.
    \end{align}
    These are the two strict margins required by the reset theorem.

    The interval area law bounds every interval entropy by
    $\overline C$. Lemma~\ref{thm:chain_interval_neighborhood_entropy},
    applied at radius $w+1$, therefore gives $S(D)\le k_0\overline C$.
    Its empty-component alternative retains empty interval labels,
    and its union estimate requires no disjointness of the enlarged intervals.
    The Araki--Lieb inequality now gives
    \begin{align}
        I_\Omega(D:C)\le2S(D)\le2k_0\overline C\le b.
    \end{align}
    Applying Lemma~\ref{thm:chain_information_reset} to these regions
    and cores proves the assertion. All pairwise conditions involving
    an empty set are vacuous, so no nonemptiness condition is needed.
\end{proof}

\begin{lemma}[Spectral truncation at a prefix cut]
    \label{thm:chain_prefix_spectral_bounds}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.spectralProjectionGE_leftBlock_bounds}
    \leanok
    \uses{def:chain_model}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on a chain of
    $L\ge0$ sites with local dimension $q\ge1$.
    For any natural number $k$, let $X$ be the sites $i$ with $0\le i<k$,
    let $\rho_X$ be the reduced state, and put
    $s=\sqrt{(1+J/\Delta)B_R}$.
    For $w\ge0$, let $P$ be the spectral projection of $\rho_X$ onto
    eigenvalues at least $e^{-(C_{\mathrm{int}}+w)}$. Then $P$ is an
    orthogonal projection and
    \begin{align}
        \operatorname{rank}P
         & \le e^{C_{\mathrm{int}}+w}, \notag                  \\
        \operatorname{Tr}((1-P)\rho_X)
         & \le 2e^{e/2}\exp\left(-\frac{w}{32s}\right). \notag
    \end{align}
    The constants are independent of $L$ and $k$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:chain_area_law, lem:chain_marginal_tail,
        lem:chain_crossing_count, lem:chain_sequential_truncation}
    The eigenvalues retained by $P$ are at least
    $e^{-(C_{\mathrm{int}}+w)}$, and their sum is at most
    $\operatorname{Tr}\rho_X=1$, giving the rank bound.
    Write $S=S(X)\le C_{\mathrm{int}}$ and
    $v=C_{\mathrm{int}}+w-S\ge w\ge0$.
    The discarded mass is at most the surprisal tail beyond $v$ centred
    at $S$, because $S+v=C_{\mathrm{int}}+w$.
    The marginal tail estimate with the uniform interval budget $B_R$ gives
    $\operatorname{Tr}((1-P)\rho_X)\le2e^{e/2}e^{-v/(32s)}
        \le2e^{e/2}e^{-w/(32s)}$.
    This is the prefix-cut estimate used in the argument for Theorem~B
    of \cite[Part~IIa]{TNLean2026OneDimensional}.
\end{proof}

\begin{corollary}[Matrix product state approximation at a prescribed accuracy]
    \label{thm:chain_mps_approximation_error}
    \lean{TNLean.MPS.AreaLaw.exists_isOpenMPS_norm_sub_le_of_error}
    \leanok
    \uses{def:chain_model, def:chain_open_mps}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on the chain of
    $L\ge1$ sites with local dimension $q\ge1$. Put
    $s=\sqrt{(1+J/\Delta)B_R}$ and $c_0=\sqrt{2e^{e/2}}$.
    For every $0<\varepsilon\le1$, there is a normalized open-boundary
    matrix product state $\Phi$ with $\|\Phi-\Omega\|\le\varepsilon$ and
    bond dimension at most
    \begin{align}
        \left\lfloor\exp\left(C_{\mathrm{int}}
        +64s\log\frac{2Lc_0}{\varepsilon}\right)\right\rfloor. \notag
    \end{align}
    This is a supplementary accuracy form of the argument for Theorem~B
    in \cite[Part~IIa]{TNLean2026OneDimensional}; the stated theorem there
    has error $1/L$.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_area_law, lem:chain_marginal_tail,
        lem:chain_sequential_truncation, thm:chain_mps_of_cut_rank,
        lem:chain_crossing_count}
    The entropy of each left block is at most $C_{\mathrm{int}}$, and its
    surprisal tail beyond $w$ is at most $c_0^2e^{-w/(32s)}$.
    Set $w=64s\log(2Lc_0/\varepsilon)$. Since $s>0$ and
    $2Lc_0/\varepsilon>1$, the sequential truncation produces a vector
    $\widetilde\Phi$ whose cut ranks are at most
    $\lfloor e^{C_{\mathrm{int}}+w}\rfloor$ and for which
    \begin{align}
        \|\widetilde\Phi-\Omega\|
         & \le (L-1)\sqrt{c_0^2e^{-w/(32s)}}
        =(L-1)\frac{\varepsilon}{2L}
        \le\frac{\varepsilon}{2}\le\frac12. \notag
    \end{align}
    Thus $\widetilde\Phi\ne0$. Normalize this same vector by setting
    $\Phi=\widetilde\Phi/\|\widetilde\Phi\|$.
    The reverse triangle inequality gives
    $\|\Phi-\widetilde\Phi\|=|1-\|\widetilde\Phi\||
        \le\|\widetilde\Phi-\Omega\|$, hence
    $\|\Phi-\Omega\|\le\varepsilon$.
    Bounded cut ranks give the matrix product representation, which is
    preserved by this scalar normalization.
\end{proof}

\begin{corollary}[Matrix product approximation for consecutive-window interactions]
    \label{thm:chain_window_mps_error}
    \lean{TNLean.MPS.AreaLaw.exists_isOpenMPS_norm_sub_le_of_error_of_consecutive_terms}
    \leanok
    \uses{def:chain_model,def:chain_open_mps}

    $\Delta>0$. With $B_{\mathrm{win}}$ and $C_{0,\mathrm{win}}$ as in
    Corollary~\ref{thm:chain_window_area_law}, put
    \begin{align}
        C & =C_{0,\mathrm{win}}(q,R)(1+J/\Delta),\notag      \\
        s & =\sqrt{(1+J/\Delta)B_{\mathrm{win}}(q,R)},\qquad
        c_0=\sqrt{2e^{e/2}}.\notag
    \end{align}
    These constants are fixed before the chain length, Hamiltonian,
    ground vector and accuracy. Let $H=\sum_Xh_X$ act on a chain of
    $L\ge1$ sites, with self-adjoint terms $h_X$ of norm at most $J$,
    and suppose that every
    nonzero $h_X$ has support $X=[a,a+R+1)$ with $a+R+1\le L$.
    Let $\Omega$ be a unit vector satisfying $H\Omega=E_0\Omega$ and
    $H-E_0\ge\Delta(1-|\Omega\rangle\langle\Omega|)$.
    For every $0<\varepsilon\le1$, there is a normalized open-boundary
    matrix product state $\Phi$ such that $\|\Phi-\Omega\|\le\varepsilon$
    and its bond dimension is at most
    \begin{align}
        \left\lfloor\exp\left(C+64s\log\frac{2Lc_0}{\varepsilon}\right)
        \right\rfloor.\notag
    \end{align}
    This is the prescribed-accuracy consequence of the argument for
    Theorem~B in \cite[Part~IIa]{TNLean2026OneDimensional}, with the
    consecutive-window constants from Part~I, Remark~6.3.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_window_area_law,thm:chain_window_cut_budget,
        lem:chain_entropy_inequalities,lem:chain_marginal_tail,
        lem:chain_sequential_truncation,thm:chain_mps_of_cut_rank}
    A prefix $[0,k)$ has at most one boundary edge, so
    Corollary~\ref{thm:chain_window_area_law} gives $S_\Omega([0,k))\le C$.
    Restrict the interaction sum to its nonzero terms, preserving its
    sum $H$ and the original projector gap. By
    Lemma~\ref{thm:chain_window_cut_budget}, its actual cut budget
    $B_k$ satisfies $1\le B_k\le B_{\mathrm{win}}$. The two-sided
    marginal tail estimate gives, for $w\ge0$,
    \begin{align}
        \operatorname{tail}_k(w)
         & \le 2e^{e/2}\exp\left(-\frac{w}{32s}\right).\notag
    \end{align}
    Indeed, the concentration bound with $B_k$ is at most its exponential
    part, and replacing $B_k$ by $B_{\mathrm{win}}$ increases that part.
    Set $w=64s\log(2Lc_0/\varepsilon)$. Sequential truncation gives a vector
    $\widetilde\Phi$ with cut ranks at most $\lfloor e^{C+w}\rfloor$ and

         & \le\sum_{k=1}^{L-1}\sqrt{\operatorname{tail}_k(w)}
        \le(L-1)\frac{\varepsilon}{2L}\le\frac\varepsilon2.\notag
    \end{align}
    Thus $\widetilde\Phi\ne0$. Its normalization
    $\Phi=\widetilde\Phi/\|\widetilde\Phi\|$ has the same cut ranks, and
    the reverse triangle inequality gives $\|\Phi-\Omega\|\le\varepsilon$.
    The cut-rank criterion yields the required matrix product state.
\end{proof}

    The cut-rank criterion gives the asserted matrix product state.
\end{proof}

\begin{corollary}[Theorem B with the printed constants]
    \label{thm:chain_polynomial_mps_printed_bound}
    \lean{TNLean.MPS.AreaLaw.exists_isOpenMPS_norm_sub_le_polynomial}
    \leanok
    \uses{thm:chain_polynomial_interval_entropy,def:chain_open_mps}
    Under the Hamiltonian and ground-state hypotheses of
    Lemma~\ref{thm:chain_polynomial_interval_entropy}, suppose $L\ge2$.
    Put $s=\sqrt{(1+J/\Delta)B_R}$, $u_0=1/(32s)$ and
    $c_0=\sqrt{2e^{e/2}}$. Define
    \begin{align}
        D & =\left\lfloor\exp\left(C_{\mathrm{int}}^{\mathrm{pol}}
        +64s\log(2L^2c_0)\right)\right\rfloor.
        \label{eq:chain_polynomial_mps_dimension}
    \end{align}
    There is a unit nonzero open-boundary matrix product state $\Phi$
    of bond dimension at most $D$ such that
    \begin{align}
        \|\Phi-\Omega\| & \le L^{-1},\notag                       \\
        D               & \le2e^{C_{\mathrm{int}}^{\mathrm{pol}}}
        \bigl(32e^{e/2}L^4\bigr)^{1/u_0}.
        \label{eq:chain_polynomial_mps_printed_bound}
    \end{align}
    This gives the normalized phase conclusion of
    \cite[Part~IIa, Theorem~B]{TNLean2026OneDimensional} with phase zero
    and its printed interval constant.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_polynomial_mps_accuracy}
    Apply Corollary~\ref{thm:chain_polynomial_mps_accuracy} with
    $\varepsilon=1/L$. To compare the bond dimensions, put
    $y=2L^2c_0$ and $b=32e^{e/2}L^4$. Then $0<y$ and
    $y^2=8e^{e/2}L^4\le b$, so $2\log y\le\log b$. Since $s\ge0$,
    \begin{align}
        D & \le e^{C_{\mathrm{int}}^{\mathrm{pol}}+64s\log y}
        \le e^{C_{\mathrm{int}}^{\mathrm{pol}}}b^{32s}
        \le2e^{C_{\mathrm{int}}^{\mathrm{pol}}}b^{1/u_0}.\notag
    \end{align}
    Unit norm implies that the chosen approximant is nonzero.
\end{proof}

\begin{corollary}[Prescribed accuracy with a polynomial bond bound]
    \label{thm:chain_polynomial_mps_accuracy_bound}
    \lean{TNLean.MPS.AreaLaw.exists_isOpenMPS_norm_sub_le_of_error_rpow,
        TNLean.MPS.AreaLaw.polynomial_floor_exp_le}
    \leanok
    \uses{thm:chain_polynomial_mps_accuracy}
    Under the hypotheses of Corollary~\ref{thm:chain_polynomial_mps_accuracy},
    for every $0<\varepsilon\le1$ there is a unit open-boundary matrix
    product state $\Phi$ with $\|\Phi-\Omega\|\le\varepsilon$ whose bond
    dimension $D$ satisfies
    \begin{align}
        D & \le e^{C_{\mathrm{int}}^{\mathrm{pol}}}(2c_0)^{64s}
        \left(\frac{L}{\varepsilon}\right)^{64s}.
        \label{eq:chain_polynomial_mps_accuracy_bound}
    \end{align}
    The case $\varepsilon=1/L$ is the bond bound $L^{128s}$ of Theorem~B.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:chain_polynomial_mps_accuracy}
    Take the state of Corollary~\ref{thm:chain_polynomial_mps_accuracy}, with
    bond dimension $D_\varepsilon$ of
    (\ref{eq:chain_polynomial_mps_accuracy}). Put $y=2Lc_0/\varepsilon>0$.
    For $s\ge0$ and every $b\ge y^2$,
    $\lfloor e^{C+64s\log y}\rfloor\le e^{C+32s\log b}=e^Cb^{32s}$.
    With $b=y^2$ this gives
    \begin{align}
        D_\varepsilon & \le e^{C_{\mathrm{int}}^{\mathrm{pol}}}y^{64s}
        =e^{C_{\mathrm{int}}^{\mathrm{pol}}}(2c_0)^{64s}
        \left(\frac{L}{\varepsilon}\right)^{64s}.\notag
    \end{align}
\end{proof}
