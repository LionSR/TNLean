\chapter{The one-dimensional area law and matrix product state approximation}
\label{ch:one_dimensional_chain}

% The chain specialization of the 2026 area-law and PEPS manuscripts, following the
% TNLean note docs/notes/one_dimensional_note. Every constant is explicit.

A chain is a finite induced lattice domain all of whose sites lie on the
horizontal axis. The finite-range Hamiltonians, gapped ground vectors and
regional entropies of Definitions~\ref{def:area_law_domain}
and~\ref{def:area_law_hamiltonian} apply to it without change. On a chain the
two-dimensional argument of \cite{OpenAI2026AreaLaw} collapses to three steps:
the marginal tail bound at an interval cut, the buffered discount with a buffer
of constant width, and the Araki--Lieb inequality. The matrix product state
approximation of \cite{OpenAI2026PolynomialPEPS} then follows from the tail
bound alone by truncating the Schmidt decompositions across the cuts in
sequence. This chapter follows \cite{TNLean2026OneDimensional}; all constants
are explicit, and the area-law constant is linear in $1+J/\Delta$.

\section{Chains and intervals}

\begin{definition}[Chains, intervals and buffers]
    \label{def:chain_domain}
    \lean{
        TNLean.MPS.AreaLaw.IsOneRow,
        TNLean.MPS.AreaLaw.chainDomain,
        TNLean.MPS.AreaLaw.mem_chainDomain,
        TNLean.MPS.AreaLaw.isOneRow_chainDomain,
        TNLean.MPS.AreaLaw.card_chainDomain,
        TNLean.MPS.AreaLaw.interval,
        TNLean.MPS.AreaLaw.mem_interval,
        TNLean.MPS.AreaLaw.card_interval_le,
        TNLean.MPS.AreaLaw.chainSite,
        TNLean.MPS.AreaLaw.chainSiteEquiv,
        TNLean.MPS.AreaLaw.rowRect,
        TNLean.MPS.AreaLaw.rectRegion_rowRect_dilate,
        TNLean.MPS.AreaLaw.edgeBoundary_univ,
        TNLean.MPS.AreaLaw.isSafe_univ
    }
    \leanok
    \uses{def:area_law_domain}
    A domain $\Lambda\subset\mathbb Z^2$ is a \emph{one-row domain} when every
    site has second coordinate zero. The \emph{chain of $L$ sites} is
    $\{(0,0),\ldots,(L-1,0)\}$; its sites are enumerated by position. For
    integers $l\le r$ the \emph{interval} $[l,r]$ of a domain is the set of its
    sites with first coordinate between $l$ and $r$; on a one-row domain it has
    at most $r+1-l$ sites, and the dilation by $d$ of the one-row rectangle
    $[l,r]\times\{0\}$ cuts out exactly the interval $[l-d,r+d]$. The whole
    domain has no boundary edge, so it is a safe cut for every rectangle.
\end{definition}

\begin{lemma}[Interactions split by an interval]
    \label{lem:chain_crossing_count}
    \lean{
        TNLean.MPS.AreaLaw.exists_mem_ends_of_splits_interval,
        TNLean.MPS.AreaLaw.chainCrossConst,
        TNLean.MPS.AreaLaw.card_crossingTerms_interval_le,
        TNLean.MPS.AreaLaw.chainBudgetConst,
        TNLean.MPS.AreaLaw.one_le_chainBudgetConst,
        TNLean.MPS.AreaLaw.cutLogBudget_interval_le,
        TNLean.MPS.AreaLaw.supportDim_le_real
    }
    \leanok
    \uses{def:chain_domain, def:area_law_hamiltonian}
    Let $\Lambda$ be a one-row domain and let $X=[a,b]$ be an interval. Every
    admissible support of range $R$ that meets both $X$ and its complement has a
    site in $[a,a+R]\cup[b-R,b]$. Hence at most
    \begin{equation}
        N_R=2(R+1)\,\mu_R,\qquad \mu_R=2^{v_R-1},\qquad v_R=1+2R(R+1),
    \end{equation}
    supports are split by $X$, and the logarithmic cut budget of the marginal
    tail bound at $X$ is at most
    \begin{equation}
        B_R=1+N_R\log^2\bigl(e\,q^{v_R}\bigr).
    \end{equation}
    Both bounds are independent of the interval.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_domain}
    A split support contains a site $v$ in $X$ and a site $w$ outside $X$,
    joined by a walk of length at most $R$, so $|v_1-w_1|\le R$; as $w_1<a$
    or $w_1>b$, the site $v$ lies within $R$ of an end. The split supports are
    therefore among the supports containing one of the at most $2(R+1)$ sites
    near the ends, and each site lies in at most $\mu_R$ admissible supports.
    The budget bound follows from the crossing count and the dimension bound
    $q^{v_R}$ for an admissible support.
\end{proof}

\section{The buffered discount with a constant-width buffer}

\begin{lemma}[Buffered discount on a chain]
    \label{lem:chain_buffer_discount}
    \lean{
        TNLean.MPS.AreaLaw.chainEnergyThreshold,
        TNLean.MPS.AreaLaw.chainTailThreshold,
        TNLean.MPS.AreaLaw.chainShellCount,
        TNLean.MPS.AreaLaw.four_le_chainShellCount,
        TNLean.MPS.AreaLaw.chainEnergyThreshold_le_chainShellCount,
        TNLean.MPS.AreaLaw.chainTailThreshold_le_chainShellCount,
        TNLean.MPS.AreaLaw.chainDiscountConst,
        TNLean.MPS.AreaLaw.chainBuffer,
        TNLean.MPS.AreaLaw.disjoint_interval_chainBuffer,
        TNLean.MPS.AreaLaw.card_chainBuffer_le,
        TNLean.MPS.AreaLaw.two_mul_condEntropy_interval_le
    }
    \leanok
    \uses{def:chain_domain, def:area_law_hamiltonian, def:area_law_ground_entropy}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on a one-row
    domain with local dimension $q\ge1$. Put $\vartheta=J/\Delta$ and
    \begin{equation}
        m=\Bigl\lceil\max\bigl\{4,\ 32N_Rq^{2v_R}\vartheta,\
        128\sqrt{(1+\vartheta)B_R}\bigr\}\Bigr\rceil .
    \end{equation}
    For an interval $X=[l,r]$ let $T=X^{+m(R+1)}\setminus X$ be the buffer,
    which has at most $2m(R+1)$ sites. Then
    \begin{equation}
        2\bigl(S(X\cup T)-S(T)\bigr)\le S(X)+
        \frac{4096\,e\,\vartheta B_R}{m}+m\log2 .
    \end{equation}
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:chain_crossing_count, def:chain_domain}
    This is the finite-dimensional buffered discount (optimized filters,
    stationarity, the energy estimate, the prefix comparison and Schmidt
    pinning) of the two-dimensional buffer lemma, applied to the nested shells
    $X_j=X^{+(j+1)(R+1)}$, $0\le j<m$, with the uniform weights $a_j=2/m$ of
    total mass two. The shells are intervals, so each is split by at most $N_R$
    supports with cut budget at most $B_R$ by
    Lemma~\ref{lem:chain_crossing_count}, and consecutive shells are $R+1$
    apart, so no support is split by two of them. The choice of $m$ makes the
    energy of the filtered vector at most $\Delta/2$, each weight at most
    $1/2$, and each tilt $a_j/(1-a_j)$ at most the tail radius
    $1/(32\sqrt{(1+\vartheta)B_j})$. The error terms sum to
    $1024e\vartheta\sum_jB_ja_j^2\le4096e\vartheta B_R/m$, and the overlap
    losses to $m\log2$.
\end{proof}

\section{Closing the argument}

\begin{lemma}[Araki--Lieb and the area law from interval bounds]
    \label{lem:chain_entropy_inequalities}
    \lean{
        TNLean.MPS.AreaLaw.regionalEntropy_compl,
        TNLean.MPS.AreaLaw.regionalEntropy_union_le,
        TNLean.MPS.AreaLaw.regionalEntropy_le_union_add,
        TNLean.MPS.AreaLaw.adj_chainDomain_iff,
        TNLean.MPS.AreaLaw.exists_interval_subset_edgeBoundary_sdiff_ssubset,
        TNLean.MPS.AreaLaw.regionalEntropy_le_mul_card_edgeBoundary_of_intervals
    }
    \leanok
    \uses{def:chain_domain, def:area_law_ground_entropy}
    For a unit vector $\Omega$ and disjoint regions $X,T$,
    $S(X)\le S(X\cup T)+S(T)$. On the chain of $L$ sites, if every interval
    has entropy at most $C\ge0$, then every subset $A$ has entropy at most
    $C\,|\partial A|$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_domain}
    The first inequality is purity, $S(X)=S(\Lambda\setminus X)$, followed by
    subadditivity for the disjoint union
    $\Lambda\setminus X=(\Lambda\setminus(X\cup T))\cup T$. For the second,
    induct on the number of sites of $A$. If $A$ is empty or the whole chain
    its entropy vanishes. Otherwise split off the maximal run $I$ of
    consecutive sites starting at the least position of $A$. Subadditivity
    gives $S(A)\le S(I)+S(A\setminus I)\le C+S(A\setminus I)$, every boundary
    edge of $A\setminus I$ is a boundary edge of $A$, and $A$ has at least one
    more boundary edge (the edge at the right end of $I$ when
    $A\setminus I$ is nonempty, and an edge at an end of $I$ otherwise, since
    $I$ is then a proper nonempty subset of the chain).
\end{proof}

\begin{theorem}[Theorem A: the one-dimensional area law]
    \label{thm:chain_area_law}
    \lean{
        TNLean.MPS.AreaLaw.interval_eq_empty,
        TNLean.MPS.AreaLaw.regionalEntropy_empty,
        TNLean.MPS.AreaLaw.chainIntervalEntropyBound,
        TNLean.MPS.AreaLaw.chainDiscountConst_nonneg,
        TNLean.MPS.AreaLaw.chainIntervalEntropyBound_nonneg,
        TNLean.MPS.AreaLaw.regionalEntropy_interval_le,
        TNLean.MPS.AreaLaw.chainShellConst,
        TNLean.MPS.AreaLaw.chainShellCount_le,
        TNLean.MPS.AreaLaw.chainAreaLawConst,
        TNLean.MPS.AreaLaw.chainIntervalEntropyBound_le,
        TNLean.MPS.AreaLaw.regionalEntropy_le_mul_card_edgeBoundary,
        TNLean.MPS.AreaLaw.chainUniformAreaLaw
    }
    \leanok
    \uses{def:chain_domain, def:area_law_hamiltonian, def:area_law_ground_entropy}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on a one-row
    domain with local dimension $q\ge1$. Every interval $X$ satisfies
    \begin{equation}
        S(X)\le C_{\mathrm{int}}(q,R,J,\Delta)
        =8m(R+1)\log q+\frac{4096\,e\,\vartheta B_R}{m}+m\log2
        \le C_0(q,R)\,(1+J/\Delta),
    \end{equation}
    with
    $C_0(q,R)=\bigl(8(R+1)\log q+\log2\bigr)\bigl(5+32N_Rq^{2v_R}+128B_R\bigr)
        +1024\,e\,B_R$. On the chain of $L$ sites every subset $A$ satisfies
    $S(A)\le C_{\mathrm{int}}\,|\partial A|$. In particular the uniform
    area-law statement of Theorem~1.1 of \cite{OpenAI2026AreaLaw}, restricted
    to chains, holds with the constant $C_0(q,R)(1+J/\Delta)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:chain_buffer_discount, lem:chain_entropy_inequalities}
    For an empty interval the entropy vanishes. Otherwise the discount of
    Lemma~\ref{lem:chain_buffer_discount} and the Araki--Lieb inequality of
    Lemma~\ref{lem:chain_entropy_inequalities} give
    $S(X)-2S(T)\le S(X\cup T)-S(T)\le\frac12S(X)+\frac12C_{\mathrm{disc}}$,
    hence $S(X)\le4S(T)+C_{\mathrm{disc}}$, and $S(T)\le|T|\log q\le
        2m(R+1)\log q$. The linear bound uses
    $m\le(5+32N_Rq^{2v_R}+128B_R)(1+\vartheta)$, which follows from
    $\sqrt{(1+\vartheta)B_R}\le(1+\vartheta)B_R$, and $m\ge4$. The subset
    bound is the second part of Lemma~\ref{lem:chain_entropy_inequalities}.
\end{proof}

\section{Matrix product states and sequential truncation}

\begin{definition}[Open-boundary matrix product states and cut matrices]
    \label{def:chain_open_mps}
    \lean{
        TNLean.MPS.AreaLaw.leftBlock,
        TNLean.MPS.AreaLaw.mem_leftBlock,
        TNLean.MPS.AreaLaw.cutMatrix,
        TNLean.MPS.AreaLaw.cutMatrix_apply,
        TNLean.MPS.AreaLaw.IsOpenMPS,
        TNLean.MPS.AreaLaw.IsOpenMPS.smul,
        TNLean.MPS.AreaLaw.IsChainMPS
    }
    \leanok
    \uses{def:chain_domain}
    A vector $\psi$ on $L$ sites of local dimension $q$ is an
    \emph{open-boundary matrix product state of bond dimension at most $D$}
    when there are matrices $A_i(s)\in\mathbb C^{D\times D}$ for each site $i$
    and value $s$ and vectors $v_L,v_R\in\mathbb C^D$ with
    \begin{equation}
        \psi(x_1,\ldots,x_L)=v_L^{\mathsf T}A_1(x_1)\cdots A_L(x_L)\,v_R .
    \end{equation}
    Scaling such a vector scales $v_R$. The \emph{cut matrix} at $k$ has the
    configurations of the first $k$ sites as rows and those of the remaining
    sites as columns; its rank is the Schmidt rank across the cut. A state of
    the chain of $L$ sites is a matrix product state when its coefficients,
    read along the chain, are.
\end{definition}

\begin{lemma}[Bounded cut ranks give a matrix product state]
    \label{lem:chain_mps_of_cut_rank}
    \lean{
        Submodule.exists_fin_span_of_finrank_le,
        TNLean.MPS.AreaLaw.glueLeft,
        TNLean.MPS.AreaLaw.sliceMatrix,
        TNLean.MPS.AreaLaw.sliceMatrix_apply,
        TNLean.MPS.AreaLaw.rank_sliceMatrix_le,
        TNLean.MPS.AreaLaw.sliceSpace,
        TNLean.MPS.AreaLaw.finrank_sliceSpace_le,
        TNLean.MPS.AreaLaw.glueLeft_snoc,
        TNLean.MPS.AreaLaw.slice_mem_sliceSpace,
        TNLean.MPS.AreaLaw.isOpenMPS_of_cutMatrix_rank_le
    }
    \leanok
    \uses{def:chain_open_mps}
    A vector all of whose cut matrices have rank at most $D$ is an
    open-boundary matrix product state of bond dimension at most $D$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_open_mps}
    Let $V_k$ be the span of the columns of the cut matrix at $k$, a space of
    dimension at most $D$, and choose a spanning family $u_k$ of $D$ vectors.
    Slicing a column of the cut matrix at $k+1$ at a fixed value of site $k$
    gives a column of the cut matrix at $k$, so each $u_{k+1}(\alpha)$ sliced
    at the value $s$ is a combination $\sum_\beta A_k(s)_{\beta\alpha}u_k(\beta)$.
    By induction $u_k(\alpha)=v_L^{\mathsf T}A_1\cdots A_k$ with $v_L=u_0$,
    and at $k=L$ the vector itself lies in $V_L$, which supplies $v_R$.
\end{proof}

\begin{lemma}[Sequential truncation]
    \label{lem:chain_sequential_truncation}
    \lean{
        Matrix.schmidtCoeffMatrix_kronecker_one_mulVec,
        Matrix.toEuclideanLin_one_apply,
        Matrix.toEuclideanLin_mul_apply,
        Entropy.localLift_sub,
        Entropy.exists_localLift_eq_localLift_of_subset,
        Entropy.norm_sq_toEuclideanLin_localLift_eq_trace,
        Entropy.norm_toEuclideanLin_localLift_le,
        Entropy.trace_regionState_eq_one,
        Matrix.PosSemidef.trace_complement_spectralProjectionGE_le_surprisalTail,
        Matrix.PosSemidef.rank_spectralProjectionGE_exp_le,
        TNLean.MPS.AreaLaw.cutMatrix_localLift,
        TNLean.MPS.AreaLaw.rank_cutMatrix_localLift_le,
        TNLean.MPS.AreaLaw.rank_cutMatrix_zero_le,
        TNLean.MPS.AreaLaw.rank_cutMatrix_le_one_of_le,
        TNLean.MPS.AreaLaw.LeftBlockMatrix,
        TNLean.MPS.AreaLaw.cutTruncation,
        TNLean.MPS.AreaLaw.sequentialTruncation,
        TNLean.MPS.AreaLaw.leftBlock_mono,
        TNLean.MPS.AreaLaw.norm_sub_sequentialTruncation_le,
        TNLean.MPS.AreaLaw.exists_sequentialTruncation_eq_localLift,
        TNLean.MPS.AreaLaw.rank_cutMatrix_sequentialTruncation_le,
        TNLean.MPS.AreaLaw.exists_sequentialTruncation
    }
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi$ be a unit vector on $L$ sites and $w\ge0$. For each cut $k$ let
    $S_k$ be the entropy of the first $k$ sites and $P_k$ the spectral
    projector of their reduced state onto the eigenvalues at least
    $e^{-(S_k+w)}$, extended by the identity. Then $P_k$ has rank at most
    $e^{S_k+w}$, the mass it discards is at most the surprisal tail of the
    reduced state beyond $w$, and the vector $\Phi=P_1P_2\cdots P_{L-1}\psi$
    has cut rank at most $\max\{1,e^{S_k+w}\}$ at every cut and satisfies
    \begin{equation}
        \|\psi-\Phi\|\le\sum_{k=1}^{L-1}\sqrt{\text{tail}_k(w)} .
    \end{equation}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_open_mps}
    The lift of a matrix on the first $k$ sites acts on the cut matrix at $k$
    by left multiplication, so the cut rank of $M\psi$ is at most the rank of
    $M$; the projector on the smallest block is applied last, and the product
    of the projectors on blocks of size at most $k$ is one lift on the first
    $k$ sites ending in $P_k$. The distance follows by induction from
    $\psi-P_k\Phi'=(1-P_k)\psi+P_k(\psi-\Phi')$ and the contraction property
    of a projector, and $\|(1-P_k)\psi\|^2$ is the trace of the discarded part
    of the reduced state, which is at most the surprisal tail.
\end{proof}

    \leanok
    \uses{def:chain_domain, def:chain_open_mps, def:area_law_ground_entropy}
    Reading a state of the chain of $L$ sites along the chain is a linear
    isometry. The reduced state of the first $k$ positions is the regional
    state of the interval $[0,k-1]$ up to relabeling of configurations, so
    the two have the same entropy and the same surprisal tails.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_domain}
    Both reduced states are partial traces of the same rank-one matrix in two
    coordinate systems. Equal characteristic polynomials of relabeled
    Hermitian matrices give equal eigenvalue multisets, hence equal values of
    every symmetric function of the eigenvalues.
\end{proof}

\begin{theorem}[Theorem B: polynomial matrix product state approximation]
    \label{thm:chain_mps_approximation}
    \lean{
        TNLean.MPS.AreaLaw.tailPrefactorRoot,
        TNLean.MPS.AreaLaw.tailPrefactorRoot_pos,
        TNLean.MPS.AreaLaw.tailPrefactorRoot_sq,
        TNLean.MPS.AreaLaw.one_lt_tailPrefactorRoot,
        TNLean.MPS.AreaLaw.chainTruncationWidth,
        TNLean.MPS.AreaLaw.chainTruncationWidth_nonneg,
        TNLean.MPS.AreaLaw.exp_neg_chainTruncationWidth,
        TNLean.MPS.AreaLaw.chainBondDimension,
        TNLean.MPS.AreaLaw.exists_isChainMPS_norm_sub_le,
        TNLean.MPS.AreaLaw.chainBondDimension_le_mul_rpow
    }
    \leanok
    \uses{def:chain_domain, def:chain_open_mps, def:area_law_hamiltonian,
        def:area_law_ground_entropy}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on the chain of
    $L\ge1$ sites with local dimension $q\ge1$. Put
    $s=\sqrt{(1+J/\Delta)B_R}$, $c_0=\sqrt{2e^{e/2}}$ and
    \begin{equation}
        w(L)=64\,s\log\bigl(2L^2c_0\bigr),\qquad
        D(L)=\bigl\lfloor\exp\bigl(C_{\mathrm{int}}+w(L)\bigr)\bigr\rfloor
        \le e^{C_{\mathrm{int}}}(2c_0)^{64s}\,L^{128s}.
    \end{equation}
    There is a normalized matrix product state $\Phi$ of bond dimension at
    most $D(L)$ with $\|\Phi-\Omega\|\le1/L$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:chain_area_law, lem:chain_sequential_truncation,
        lem:chain_mps_of_cut_rank, lem:chain_transport, lem:chain_crossing_count}
    Read $\Omega$ along the chain. By Lemma~\ref{lem:chain_transport} and
    Theorem~\ref{thm:chain_area_law} the entropy of every initial block is at
    most $C_{\mathrm{int}}$, and by the marginal tail bound at the interval
    $[0,k-1]$ with the budget $B_R$ of Lemma~\ref{lem:chain_crossing_count}
    its surprisal tail beyond $w$ is at most $2e^{e/2}e^{-w/(32s)}$. The
    sequential truncation of Lemma~\ref{lem:chain_sequential_truncation} at
    width $w(L)$ gives cut ranks at most $e^{C_{\mathrm{int}}+w(L)}$ and
    distance at most $(L-1)\cdot\frac1{2L^2}\le\frac1{2L}$. Normalizing moves
    the vector by at most the same amount, so the distance is at most $1/L$,
    and Lemma~\ref{lem:chain_mps_of_cut_rank} exhibits the matrix product
    structure, which survives the normalization. Reading back along the
    chain is an isometry.
\end{proof}
