\chapter{The one-dimensional area law and matrix product state approximation}
\label{ch:one_dimensional_chain}

% The chain specialization of the 2026 area-law and PEPS manuscripts, following the
% TNLean note docs/notes/one_dimensional_note. Every constant is explicit. The model is
% stated natively on the chain of L sites; nothing is derived from the two-dimensional
% lattice development.

The chain of $L$ sites carries a local dimension $q$, a finite-range Hamiltonian
with one Hermitian term of norm at most $J$ per support of diameter at most $R$,
and a unit ground vector separated from the rest of the spectrum by a gap
$\Delta$. On a chain the two-dimensional argument of \cite{OpenAI2026AreaLaw}
collapses to three steps: the marginal tail bound at an interval cut, the
buffered discount with a buffer of constant width, and the Araki--Lieb
inequality. The matrix product state approximation of
\cite{OpenAI2026PolynomialPEPS} then follows from the tail bound alone by
truncating the Schmidt decompositions across the cuts in sequence. This chapter
follows \cite{TNLean2026OneDimensional}; all constants are explicit, and the
area-law constant is linear in $1+J/\Delta$. The two finite-dimensional cores,
the tail bound and the buffered discount, are stated in the companion library
for an arbitrary finite set of sites and are instantiated here directly.

\section{Chains, supports and Hamiltonians}

\begin{definition}[Chains, intervals and finite-range Hamiltonians]
    \label{def:chain_model}
    \lean{
        TNLean.MPS.AreaLaw.interval,
        TNLean.MPS.AreaLaw.mem_interval,
        TNLean.MPS.AreaLaw.interval_mono,
        TNLean.MPS.AreaLaw.interval_eq_empty,
        TNLean.MPS.AreaLaw.interval_zero_eq_univ,
        TNLean.MPS.AreaLaw.card_interval_le,
        TNLean.MPS.AreaLaw.IsChainSupport,
        TNLean.MPS.AreaLaw.ChainSupport,
        TNLean.MPS.AreaLaw.IsChainSupport.card_le,
        TNLean.MPS.AreaLaw.supportDim_le,
        TNLean.MPS.AreaLaw.one_le_supportDim,
        TNLean.MPS.AreaLaw.ChainHamiltonian,
        TNLean.MPS.AreaLaw.ChainHamiltonian.operator,
        TNLean.MPS.AreaLaw.ChainHamiltonian.operator_isHermitian,
        TNLean.MPS.AreaLaw.ChainHamiltonian.zero,
        TNLean.MPS.AreaLaw.cutLogBudget,
        TNLean.MPS.AreaLaw.one_le_cutLogBudget
    }
    \leanok
    The \emph{chain of $L$ sites} has sites $0,\ldots,L-1$, each carrying
    $\mathbb C^q$. For natural numbers $a\le b$ the \emph{interval} $[a,b)$ is
    the set of sites with position at least $a$ and less than $b$; it has at
    most $b-a$ sites, and its dilation $[a-d,b+d)$ is clipped at the ends of the
    chain. A \emph{support of range $R$} is a nonempty set of sites of diameter
    at most $R$; it has at most $R+1$ sites and Hilbert-space dimension at most
    $q^{R+1}$. A \emph{finite-range Hamiltonian} assigns to every support $X$
    of range $R$ a Hermitian term $h_X$ acting as the identity off $X$ with
    operator norm at most $J$, zero terms allowed, and
    $H=\sum_X h_X$. A \emph{gapped ground vector} with gap $\Delta$ is a unit
    vector $\Omega$ with $H\Omega=E_0\Omega$ and
    $H-E_0\ge\Delta(1-|\Omega\rangle\langle\Omega|)$. For a cut $A$ the
    \emph{cut budget} is
    $\mathcal B_A=1+\sum_X\log^2(e\,q^{|X|})$ over the supports $X$ meeting
    both $A$ and its complement; it is at least one.
\end{definition}

\begin{lemma}[Distance and spheres on a finite path]
    \label{lem:chain_path_metric_spheres}
    \lean{SimpleGraph.pathGraph_edist, SimpleGraph.pathGraph_card_sphere_le_two}
    \leanok
    Let $L$ be a natural number, and let $P_L$ be the graph on
    $\{0,\ldots,L-1\}$ joining consecutive coordinates. For vertices $x,y$ and
    every natural number $d$,
    \begin{align}
        d_{P_L}(x,y)                     & =|x-y|, \\
        \bigl|\{y:d_{P_L}(x,y)=d\}\bigr| & \le2.
    \end{align}
    Thus the sphere growth bound holds with coefficient $2$ and exponent $0$,
    independently of $L$. This concerns the contiguous chain setting of
    Section~1 of~\cite{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    Each edge changes the coordinate by one, so the displacement of a walk is
    at most its length. Taking the infimum over walks gives one inequality.
    In the other direction, successive coordinates between $x$ and $y$ stay
    inside the chain and give a walk of length $|x-y|$. A site at distance $d$
    from $x$ has coordinate $x-d$ or $x+d$; discarding coordinates outside the
    chain leaves at most two sites. At $d=0$ the candidates coincide. When
    $L=0$ there are no vertices, so the assertions are vacuous.
\end{proof}

\begin{lemma}[Marginal tail bound at a cut]
    \label{lem:chain_marginal_tail}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.surprisalTail_regionState_le}
    \leanok
    \uses{def:chain_model}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on the chain, and
    let $A$ be any set of sites. Let $K$ be the surprisal $-\log p$ of the
    eigenvalue $p$ of the reduced state on $A$ drawn with probability $p$, so
    that $\mathbb E K=S(A)$. For every real $w$,
    \begin{align}
        \Pr\{|K-S(A)|>w\}\le
        \min\Bigl\{1,\ 2e^{e/2}\exp\Bigl(-\frac{w}{32\sqrt{(1+J/\Delta)\,
                \mathcal B_A}}\Bigr)\Bigr\}.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    This is Lemma~3.1 of \cite{OpenAI2026AreaLaw}, which holds on any finite
    tensor product: each term acts as the identity off its support, is
    Hermitian and has norm at most $J$, and the ground vector is a unit
    eigenvector with the projector gap. The chain model supplies exactly these
    hypotheses.
\end{proof}

\begin{definition}[Doubled chain Hamiltonian]
    \label{def:chain_doubled_hamiltonian}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled}
    \leanok
    \uses{def:chain_model,def:qc_site_pair_coordinates}
    Let $L,q,R$ be natural numbers, let $J$ be real, and let
    $H=\sum_X h_X$ be a range-$R$ chain Hamiltonian with local dimension $q$
    and $\|h_X\|\le J$. Pair the two copies at each original site by the
    bijection $p$ of Definition~\ref{def:qc_site_pair_coordinates}, with
    $p(\sigma,\tau)(i)=q\sigma(i)+\tau(i)$. Write $p_*M$ for simultaneous
    relabelling of the rows and columns of a matrix $M$ by $p$. The
    \emph{doubled Hamiltonian} has local dimension $q^2$ and interactions
    \begin{align}
        \widehat h_X=p_*(h_X\otimes\mathbb 1+\mathbb 1\otimes h_X).
    \end{align}
    Each $\widehat h_X$ is Hermitian, is supported on the same set $X$, and
    satisfies $\|\widehat h_X\|\le2J$. Thus the range remains $R$.
    This is the two-copy construction in the paragraph ``Two-sided Gaussian
    filtering'' of \cite{TNLean2026OneDimensional}.
\end{definition}
\begin{proof}
    \leanok
    \uses{def:qc_local_circuit,thm:area_law_supported_identity,thm:qc_site_pair_tensor_support}
    By the matrix-entry characterization, $h_X$ belongs to the algebra
    supported on $X$; the identity belongs to that algebra as well. The tensor-product support identity,
    applied to each summand, gives support on $X\cup X=X$, and the
    matrix-entry characterization gives the required support condition.
    Taking adjoints preserves each summand and commutes with relabelling.
    Relabelling preserves the operator norm, and tensoring with an identity
    does not increase it. Hence
    \begin{align}
        \|\widehat h_X\|
        \le\|h_X\otimes\mathbb 1\|+\|\mathbb 1\otimes h_X\|
        \le\|h_X\|+\|h_X\|\le2J.
    \end{align}
\end{proof}

\begin{lemma}[Operator of the doubled Hamiltonian]
    \label{lem:chain_doubled_operator}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled_operator}
    \leanok
    \uses{def:chain_doubled_hamiltonian}
    Under the hypotheses of Definition~\ref{def:chain_doubled_hamiltonian},
    the sum of the doubled interactions satisfies
    \begin{align}
        \widehat H=\sum_X\widehat h_X
        =p_*(H\otimes\mathbb 1+\mathbb 1\otimes H).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian}
    Sum the defining interaction identity over the original supports $X$.
    Relabelling is linear, and the tensor product is linear in each factor.
\end{proof}

\begin{definition}[Regionally swapped doubled Hamiltonian]
    \label{def:chain_swapped_doubled_hamiltonian}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.swappedDoubled,
        TNLean.MPS.AreaLaw.ChainHamiltonian.swappedDoubled_operator}
    \leanok
    \uses{def:chain_doubled_hamiltonian,def:peps_frame_sheet_swap}
    Let $H=\sum_Xh_X$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$, with local dimension $q$ and
    $\|h_X\|\le J$, where $L,q,R$ are natural numbers and $J$ is real.
    For any $D\subseteq I$, let $F=p_*F_D$ in the paired coordinates of
    Definition~\ref{def:chain_doubled_hamiltonian}. The regionally swapped
    doubled Hamiltonian has interactions and sum
    \begin{align}
        \widehat h'_X & =F\widehat h_XF^\dagger,\notag              \\
        \widehat H'   & =\sum_X\widehat h'_X=F\widehat H F^\dagger.
        \label{eq:chain_swapped_doubled_operator}
    \end{align}
    Each $\widehat h'_X$ is Hermitian, is supported on the same set $X$,
    and has norm at most $2J$. Thus this is a range-$R$ chain Hamiltonian
    with local dimension $q^2$ and interaction bound $2J$, also when
    $q=0$ or $L=0$.
\end{definition}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian,thm:area_law_supported_identity,
        thm:peps_sheet_swap_conjugate_support,thm:peps_frame_sheet_swap_norm}
    The support identity for the conjugated doubled interaction gives
    support on $X$, and taking adjoints preserves Hermiticity.
    Since $\|F\|\le1$, the doubled interaction estimate gives
    \begin{align}
        \|F\widehat h_XF^\dagger\|
        \le\|\widehat h_X\|\le2J.
    \end{align}
    Distributing left and right multiplication over the finite sum proves
    (\ref{eq:chain_swapped_doubled_operator}).
\end{proof}

\begin{theorem}[The actual ground vectors of the doubled and swapped chains]
    \label{thm:chain_doubled_swapped_ground_gap}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.doubled_swappedDoubled_isGappedGroundState}
    \leanok
    \uses{def:chain_model,def:chain_swapped_doubled_hamiltonian}
    In the setting of Definition~\ref{def:chain_swapped_doubled_hamiltonian},
    let $E_0,\Delta$ be real with $\Delta>0$, and suppose the original
    vector $\Omega$ satisfies
    \begin{align}
        \|\Omega\|     & =1,\notag                                          \\
        H\Omega        & =E_0\Omega,\notag                                  \\
        H-E_0\mathbb 1 & \ge\Delta(\mathbb 1-|\Omega\rangle\langle\Omega|).
    \end{align}
    Define the vectors in the paired physical coordinates by
    $\Psi(p(\sigma,\tau))=\Omega(\sigma)\Omega(\tau)$ and
    $\Phi=F\Psi$. Then
    \begin{align}
        \|\Psi\|        & =\|\Phi\|=1,\notag                                   \\
        \widehat H\Psi  & =2E_0\Psi,\notag                                     \\
        \widehat H'\Phi & =2E_0\Phi,\notag                                     \\
        \widehat H-2E_0\mathbb 1
                        & \ge\Delta(\mathbb 1-|\Psi\rangle\langle\Psi|),\notag \\
        \widehat H'-2E_0\mathbb 1
                        & \ge\Delta(\mathbb 1-|\Phi\rangle\langle\Phi|).
    \end{align}
    In particular, these actual vectors give the ground lines of the two
    Hamiltonians, with common ground energy $2E_0$ and gap at least
    $\Delta$. This is the contiguous-chain assertion in the paragraph
    ``Two-sided Gaussian filtering'' of~\cite{TNLean2026OneDimensional},
    before the filtering construction; the distinction from arbitrary
    finite induced domains is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:chain_doubled_operator,def:chain_swapped_doubled_hamiltonian,
        thm:permutation_matrix_unitary,thm:mpu_reindex_unitary}
    Put $P=|\Omega\rangle\langle\Omega|$, $Q=\mathbb 1-P$ and
    $G=H-E_0\mathbb 1-\Delta Q\ge0$. Normalization gives $Q\ge0$.
    Before pairing coordinates, the doubled defect is
    \begin{align}
        H\otimes\mathbb 1+\mathbb 1\otimes H-2E_0\mathbb 1
        -\Delta(\mathbb 1-P\otimes P)
         & =G\otimes\mathbb 1+\mathbb 1\otimes G+\Delta Q\otimes Q
        \ge0.
    \end{align}
    The tensor-square vector has norm one and eigenvalue $2E_0$.
    Pairing coordinates is unitary, so the same norm, eigen-equation and
    positive defect hold for $\Psi$ and $\widehat H$.
    The regional swap is a permutation matrix; after pairing it remains
    unitary. Therefore $\Phi=F\Psi$ has norm one and eigenvalue $2E_0$
    for $\widehat H'$. Moreover,
    \begin{align}
         & \widehat H'-2E_0\mathbb 1
        -\Delta(\mathbb 1-|\Phi\rangle\langle\Phi|)\notag \\
         & \qquad=F\bigl(\widehat H-2E_0\mathbb 1
        -\Delta(\mathbb 1-|\Psi\rangle\langle\Psi|)\bigr)F^\dagger
        \ge0.
    \end{align}
    Finally, a positive projector gap forces every vector at the ground
    energy to lie in the span of the displayed unit vector, giving the
    asserted ground lines.
\end{proof}

\begin{lemma}[Interactions split by an interval]
    \label{lem:chain_crossing_count}
    \lean{
        TNLean.MPS.AreaLaw.IsChainSupport.subset_window_of_splits,
        TNLean.MPS.AreaLaw.chainCrossConst,
        TNLean.MPS.AreaLaw.card_filter_subset_le,
        TNLean.MPS.AreaLaw.card_crossingTerms_interval_le,
        TNLean.MPS.AreaLaw.chainBudgetConst,
        TNLean.MPS.AreaLaw.one_le_chainBudgetConst,
        TNLean.MPS.AreaLaw.supportDim_le_real,
        TNLean.MPS.AreaLaw.cutLogBudget_interval_le,
        TNLean.MPS.AreaLaw.not_splits_two_shells
    }
    \leanok
    \uses{def:chain_model}
    Let $X=[a,b)$ be an interval of the chain. A support of range $R$ that
    meets both $X$ and its complement lies in the window $[a-R,a+R)$ or in the
    window $[b-R,b+R)$, each of at most $2R$ sites. Hence at most
    \begin{align}
        N_R=2\cdot2^{2R}
    \end{align}
    supports are split by $X$, and the cut budget of $X$ is at most
    \begin{align}
        B_R=1+N_R\log^2\bigl(e\,q^{R+1}\bigr).
    \end{align}
    Both bounds are independent of the interval, and no support of range $R$
    is split by two of the shells $[l-(j+1)R,\,r+(j+1)R)$, $j\ge0$, of an
    interval $[l,r)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    A split support contains a site $v$ in $X$ and a site $w$ outside $X$
    with $|v-w|\le R$. If $w<a$ then every site $x$ of the support satisfies
    $a-R\le v-R\le x\le w+R<a+R$; if $w\ge b$ then $b-R\le w-R\le x\le v+R<b+R$.
    A window of at most $2R$ sites has at most $2^{2R}$ subsets. The budget
    bound follows from the crossing count and the dimension bound $q^{R+1}$. If
    a support met the shell with index $j$ and left the shell with index
    $j'>j$, two of its sites would be at distance at least $R+1$.
\end{proof}

\begin{lemma}[Supports split by an arbitrary region of a chain]
    \label{lem:chain_boundary_crossing_count}
    \lean{TNLean.MPS.AreaLaw.IsChainSupport.exists_boundary_window_of_split,
        TNLean.MPS.AreaLaw.card_crossingTerms_le_mul_card_boundaryEdges}
    \leanok
    \uses{def:chain_model,lem:chain_entropy_inequalities}
    Let $L,R$ be natural numbers, let $I=\{0,\ldots,L-1\}$ be the
    contiguous chain, and let $D\subseteq I$. Write $\partial D$ for the
    positions $p$ with $p+1<L$ such that exactly one of $p,p+1$ belongs
    to $D$. For $p\in\partial D$ put
    \begin{align}
        W_p=I\cap[\max\{0,p+1-R\},\,p+1+R).
    \end{align}
    If a nonempty support $X\subseteq I$ of diameter at most $R$ meets
    both $D$ and $I\setminus D$, then $X\subseteq W_p$ for some actual
    boundary edge $p\in\partial D$. Consequently,
    \begin{align}
        \#\{X\subseteq I:\ X\ne\varnothing,\ \operatorname{diam}X\le R,
        \ X\cap D\ne\varnothing,\ X\setminus D\ne\varnothing\}
        \le 2^{2R}|\partial D|.
    \end{align}
    No interval assumption on $D$ is needed. The conclusions include the
    empty chain and $R=0$; in either case there is no split support.
    This is the boundary-window count underlying the interval estimate
    in Section~2, equation~\textup{(count)}, and the boundary count in the
    paragraph ``Two-sided Gaussian filtering'' of
    \cite{TNLean2026OneDimensional}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model,lem:chain_entropy_inequalities,
        lem:chain_crossing_count}
    Choose two sites of $X$ with different membership in $D$, and order
    their coordinates as $a<b$. Along the integer interval from $a$ to
    $b$, the indicator of $D$ must change at two consecutive sites
    $p,p+1$. Thus $a\le p<p+1\le b<L$ and $p\in\partial D$.

    For every $x\in X$, the diameter bound gives
    $x\le a+R$ and $b\le x+R$. Hence
    \begin{align}
        \max\{0,p+1-R\}\le x<p+1+R,
    \end{align}
    proving $X\subseteq W_p$. Each window has at most $2R$ sites, so
    contains at most $2^{2R}$ subsets. Cover the split supports by the
    families of subsets of $W_p$, one family for each $p\in\partial D$,
    and bound the cardinality of the finite union by the sum of their
    cardinalities. This proves the stated bound, including the case of
    an empty boundary.
\end{proof}

\begin{theorem}[The boundary perturbation of a doubled chain]
    \label{thm:chain_partial_swap_boundary}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_term_difference_spec,
        TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_difference_eq_sum_crossing,
        TNLean.MPS.AreaLaw.ChainHamiltonian.partialSwap_boundary_spec}
    \leanok
    \uses{def:chain_model,def:chain_doubled_hamiltonian,
        def:qc_site_pair_coordinates,def:peps_frame_sheet_swap,
        lem:chain_boundary_crossing_count}
    Let $L,q,R$ be natural numbers, let $J\in\mathbb R$, and let
    $H=\sum_Xh_X$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$, with local dimension $q$ and
    $\|h_X\|\le J$. Let $D\subseteq I$ be arbitrary. In the paired
    coordinates $p$ of Definition~\ref{def:chain_doubled_hamiltonian}, put
    \begin{align}
        F       & =p_*F_D,\notag                                \\
        T_{X,0} & =p_*(h_X\otimes\mathbb 1),\notag              \\
        T_{X,1} & =p_*(\mathbb 1\otimes h_X),\notag             \\
        K_{X,b} & =FT_{X,b}F^\dagger-T_{X,b}\quad(b\in\{0,1\}).
    \end{align}
    Every $K_{X,b}$ is Hermitian, is supported on the original set $X$,
    and satisfies $\|K_{X,b}\|\le2J$. Let $\mathcal C$ be the set of
    original supports meeting both $D$ and $I\setminus D$, and let
    $N_R=2\cdot2^{2R}$. For the actual doubled operator $\widehat H$,
    \begin{align}
        F\widehat H F^\dagger-\widehat H
                      & =\sum_{X\in\mathcal C}\sum_{b\in\{0,1\}}K_{X,b},\notag \\
        2|\mathcal C| & \le N_R|\partial D|.
    \end{align}
    Let $Z=\bigcup_{X\in\mathcal C}X$ be the actual union of crossing
    supports and put $\Delta H=F\widehat H F^\dagger-\widehat H$.
    The operator $\Delta H$ is supported on $Z$. For every $z\in Z$ there
    is an actual boundary position $a\in\partial D$ with $|z-a|\le R$,
    and
    \begin{align}
        \|\Delta H\|\le 2J N_R|\partial D|.
    \end{align}
    These assertions include $q=0$, the empty chain, and empty or full
    $D$. They give the boundary decomposition on a contiguous chain used
    in the paragraph ``Two-sided Gaussian filtering'' of
    \cite{TNLean2026OneDimensional}; the distinction from arbitrary finite
    induced domains is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:chain_doubled_hamiltonian,thm:qc_site_pair_tensor_support,
        thm:area_law_supported_identity,thm:peps_sheet_swap_conjugate_support,
        thm:peps_frame_sheet_swap_norm,
        thm:peps_sheet_swap_doubled_cancellation,
        lem:chain_boundary_crossing_count}
    The support reduction for a conjugated tensor product in paired
    coordinates shows that $FT_{X,b}F^\dagger$ is supported on $X$.
    The same is true of $T_{X,b}$, so subtraction preserves its support.
    Taking adjoints proves Hermiticity. Pairing coordinates preserves
    the operator norm, and the regional swap satisfies $\|F\|\le1$.
    Tensoring with an identity does not increase the operator norm, hence
    \begin{align}
        \|T_{X,b}\|           & \le\|h_X\|\le J,\notag \\
        \|FT_{X,b}F^\dagger\| & \le J,\notag           \\
        \|K_{X,b}\|           & \le J+J=2J.
    \end{align}

    If $X$ lies inside $D$, swapping exchanges the two replica terms;
    if $X$ is disjoint from $D$, each replica term commutes with the swap.
    Thus, for every support outside $\mathcal C$,
    \begin{align}
        F(T_{X,0}+T_{X,1})F^\dagger=T_{X,0}+T_{X,1}.
    \end{align}
    By definition, each doubled interaction is $T_{X,0}+T_{X,1}$.
    Summing these identities expresses $\widehat H$ as the sum of these
    pairs. Distributing conjugation and subtraction over the finite sum
    cancels each pair outside $\mathcal C$, leaving exactly the sum over
    the crossing supports. Finally, Lemma~\ref{lem:chain_boundary_crossing_count}
    gives $|\mathcal C|\le2^{2R}|\partial D|$; counting the two replica
    labels for each support gives the stated bound.

    Each term in the resulting expansion is supported on some
    $X\subseteq Z$, so their finite sum is supported on $Z$. If $z\in Z$,
    choose a crossing support $X$ containing it. The boundary-window
    inclusion gives $a+1-R\le z<a+1+R$ for an actual boundary position
    $a$, whence $|z-a|\le R$.
    If $L=0$, the boundary and the crossing family are empty and
    $\Delta H=0$. Otherwise, a singleton interaction implies
    $0\le\|h_{\{x\}}\|\le J$ at an actual site $x$, so $J\ge0$.
    The triangle inequality and the bound on the number of replica labels
    now give
    \begin{align}
        \|\Delta H\|\le 2J(2|\mathcal C|)\le 2J N_R|\partial D|.
    \end{align}
\end{proof}

\begin{lemma}[Interaction norm budget at a site]
    \label{lem:chain_site_norm_budget}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.sum_norm_terms_containing_le}
    \leanok
    \uses{def:chain_model}
    Let $L,q,R$ be natural numbers and $J\in\mathbb R$. On the chain of $L$
    sites with local dimension $q$, let $H=\sum_X h_X$ be a finite-range
    Hamiltonian with one term of norm at most $J$ for each nonempty support
    $X$ of diameter at most $R$. For every site $x$,
    \begin{align}
        \sum_{X\ni x}\lVert h_X\rVert\le 2^{2R+1}J.
    \end{align}
    This is an auxiliary consequence of the Hamiltonian setting
    in Section~1 of~\cite{TNLean2026OneDimensional}; it supplies a
    per-site interaction estimate of the form assumed immediately before
    the Lieb--Robinson lemma there. No spectral-gap assumption is needed.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model, lem:chain_crossing_count}
    Fix a site $x$. The singleton $\{x\}$ is an admissible support, so
    $0\le\lVert h_{\{x\}}\rVert\le J$. Every support containing $x$ lies in
    the clipped interval $[x-R,x+R+1)$, which has at most $2R+1$ sites.
    Counting subsets of this interval gives at most $2^{2R+1}$ such supports.
    Summing their norm bounds and using $J\ge0$ proves the claim. If the chain
    has no sites, the assertion is vacuous.
\end{proof}

\section{Locality on a contiguous chain}

\begin{theorem}[Lieb--Robinson estimates on a contiguous chain]
    \label{thm:chain_lieb_robinson}
    \lean{TNLean.MPS.AreaLaw.norm_heisenberg_commutator_le_chainDistance,
        TNLean.MPS.AreaLaw.norm_heisenberg_sub_siteExpectation_chainNeighborhood_le}
    \leanok
    \uses{def:chain_model,def:qc_site_expectation}
    Let $L,q,R$ be natural numbers. Let $K$ be a finite set of interaction
    labels, and for each $k\in K$ let $X_k$ be a nonempty support of diameter
    at most $R$ in the contiguous chain $I=\{0,\ldots,L-1\}$.
    Let $h_k$ be Hermitian operators supported on $X_k$, and suppose that
    $b_0\in\mathbb R$ satisfies
    \begin{align}
        \sum_{k:\,x\in X_k}\|h_k\|\le b_0
    \end{align}
    for every site $x$. Put $H=\sum_{k\in K}h_k$ and
    $\tau_t(A)=e^{itH}Ae^{-itH}$, and set
    \begin{align}
        C_0 & =2+4|b_0|e^{2R},\notag      \\
        v   & =2(R+1)|b_0|e^{2R}+1,\notag \\
        \mu & =\tfrac12,\qquad
        C'=\frac{2C_0e}{1-e^{-1/2}}.\notag
    \end{align}
    These constants are independent of $L,q$, the interaction family, and
    the observable support. For $S\subseteq I$ let
    $d(S,y)=\min_{x\in S}|x-y|$, with $d(\varnothing,y)=\infty$ and
    $e^{-\mu\infty}=0$. For every operator $A$ supported on $S$, every site
    $y$, every operator $B_y$ supported on $\{y\}$, and every real $t$,
    \begin{align}
        \|[\tau_t(A),B_y]\|
        \le C_0|S|\|A\|\|B_y\|e^{v|t|}e^{-d(S,y)/2}.
    \end{align}
    This estimate allows every natural local dimension $q$.
    If $q\ge1$, define, for every real $\ell\ge0$,
    $N_\ell(S)=\{y\in I:\exists x\in S,\ |x-y|\le\ell\}$.
    The normalized partial-trace expectation $E_{N_\ell(S)}$ then satisfies
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
        \le C'|S|^2\|A\|e^{v|t|-\ell/2}.
    \end{align}
    Both estimates allow $S=\varnothing$. This is the contiguous-chain case
    of the lemma ``Lieb--Robinson bound on a chain'' in Part~II
    of~\cite{TNLean2026OneDimensional}, as used for its Theorem~B;
    the distinction from the broader finite-domain reading is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:chain_path_metric_spheres,thm:area_law_supported_identity,
        thm:qc_graph_lieb_robinson_finset,
        thm:qc_graph_neighborhood_localization}
    Use the path graph on $I$. Its distance is $|x-y|$, each $X_k$ has
    graph diameter at most $R$ and at most $R+1$ sites, and its nonemptiness
    permits a choice of an anchor in $X_k$. The matrix-entry characterization
    of support supplies the hypotheses on the supported algebras.
    Apply the arbitrary-support graph commutator estimate with decay
    parameter $1$ and support-size bound $R+1$. If $S$ is nonempty,
    its least distance to $y$ is a finite nonnegative integer $d$, and
    $e^{-d}\le e^{-d/2}$. For $S=\varnothing$, the factor $|S|$ makes
    the commutator bound zero without selecting a point of $S$.

    For localization assume $q\ge1$ and put $m=\lfloor\ell\rfloor$.
    Every distance sphere of the path has at most two sites, so the graph
    localization estimate applies with $K_g=2$ and growth exponent zero.
    Since path distances are integers, the union of graph balls of radius
    $m$ about $S$ is exactly $N_\ell(S)$. The tail constant and the rounding
    bound are
    \begin{align}
        T_0      & =\frac{e^{1/2}}{1-e^{-1/2}},\notag \\
        e^{-m/2} & \le e^{1/2}e^{-\ell/2}.\notag
    \end{align}
    Thus the graph coefficient $2C_0T_0$, multiplied by the rounding factor
    $e^{1/2}$, is $C'$. Empty unions and the zero factor $|S|^2$ cover
    $S=\varnothing$ as well.
\end{proof}

\begin{theorem}[Localization of the relative-evolution generator]
    \label{thm:chain_relative_generator_localization}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.norm_relativeGenerator_sub_siteExpectation_le}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,thm:chain_partial_swap_boundary,
        def:qc_site_expectation}
    Let $H$ be a range-$R$ Hamiltonian on the contiguous chain
    $I=\{0,\ldots,L-1\}$ with local dimension $q\ge1$ and interaction
    bound $J\in\mathbb R$. Let $D\subseteq I$, and let
    $K=\widehat H$ and $K'=\widehat H'$ be its doubled and regionally
    swapped doubled operators. Write $\mathcal C_D$ for the original
    supports meeting both $D$ and its complement, and put
    $Z=\bigcup_{X\in\mathcal C_D}X$.
    For every real $\ell\ge0$ and $t\in\mathbb R$, define
    \begin{align}
        B    & =N_\ell(Z),\notag         \\
        G(t) & =e^{itK'}(K'-K)e^{-itK'}.
    \end{align}
    Set
    \begin{align}
        b_0 & =2^{2R+1}(2J),\notag                          \\
        a   & =\frac{2(2+4|b_0|e^{2R})e}{1-e^{-1/2}},\notag \\
        v   & =2(R+1)|b_0|e^{2R}+1,\notag                   \\
        N_R & =2\cdot2^{2R}.
    \end{align}
    Then the normalized expectation on the paired physical sites satisfies
    \begin{align}
        \|G(t)-E_B(G(t))\|
        \le4|J|a(R+1)^2N_R L\,e^{v|t|-\ell/2}.
    \end{align}
    No nonemptiness of the chain or of $\mathcal C_D$, sign condition on
    $J$, or spectral-gap hypothesis is required. This is the intermediate
    generator estimate in the localization paragraph of the proof of the
    information-reset lemma in Part~II of~\cite{TNLean2026OneDimensional},
    on a contiguous chain; the domain distinction is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,thm:chain_partial_swap_boundary,
        lem:chain_site_norm_budget,thm:chain_lieb_robinson,
        lem:qc_site_expectation_properties}
    The swapped doubled interactions have range $R$ and bound $2J$.
    Their per-site norm sum is at most $b_0$. The chain localization
    estimate therefore has constants $a,v$ as displayed.
    By Theorem~\ref{thm:chain_partial_swap_boundary},
    \begin{align}
        K'-K            & =\sum_{X\in\mathcal C_D}\sum_{b\in\{0,1\}}A_{X,b},\notag \\
        \|A_{X,b}\|     & \le2J\le2|J|,\notag                                      \\
        2|\mathcal C_D| & \le N_R|\partial D|\le N_R L.
    \end{align}
    Each $A_{X,b}$ is supported on $X$, with $|X|\le R+1$.
    For one such term put $A=e^{itK'}A_{X,b}e^{-itK'}$ and
    $A_0=E_{N_\ell(X)}(A)$. The chain estimate gives
    \begin{align}
        \|A-A_0\|\le2|J|a(R+1)^2e^{v|t|-\ell/2}.
    \end{align}
    Since $X\subseteq Z$, the operator $A_0$ is supported on $B$ and is
    fixed by $E_B$. Hence
    \begin{align}
        A-E_B(A)     & =(A-A_0)-E_B(A-A_0),\notag \\
        \|A-E_B(A)\| & \le2\|A-A_0\|
        \le4|J|a(R+1)^2e^{v|t|-\ell/2}.
    \end{align}
    Conjugation and the expectation are linear. Summing this estimate
    over the actual crossing replica labels and using their cardinality
    bound proves the assertion. An empty crossing family gives the zero
    generator; the same nonnegative estimates include $J=0$.
\end{proof}

\begin{theorem}[A supported Gaussian integral on a contiguous chain]
    \label{thm:chain_supported_gaussian_integral}
    \lean{TNLean.MPS.AreaLaw.exists_supported_gaussian_integral}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,thm:chain_partial_swap_boundary,
        def:qc_site_expectation}
    For every natural number $R$ and real $J$, there are positive constants
    $C,v$, depending only on $R,J$, with the following property.
    Let $q\ge1$ and $L$ be natural numbers, let $H$ be a range-$R$
    Hamiltonian with interaction bound $J$ on the contiguous chain
    $I=\{0,\ldots,L-1\}$, let $D\subseteq I$, and let $\ell\ge0$ be real.
    Put $K=\widehat H$, $K'=\widehat H'$ and
    $Z=\bigcup_{X\in\mathcal C_D}X$, where $\mathcal C_D$ is the set of
    original supports crossing $D$. Write $N(S)=N_\ell(S)$.
    There is a differentiable unitary path $U$, supported on $N(Z)$ at
    every real time, such that
    \begin{align}
        U(0)  & =\mathbb 1,\notag                                  \\
        U'(t) & =iE_{N(Z)}\bigl(e^{itK'}(K'-K)e^{-itK'}\bigr)U(t).
    \end{align}
    The same path is chosen before all the following data.
    Let $Q\subseteq I$, let $W$ be any operator supported on $Q$,
    and let $\eta\ge0$ and $T\ge0$ be real.
    Let $\gamma_\eta$ be the centred Gaussian measure of variance $\eta$,
    with $\gamma_0$ the point mass at zero. Put $B=N(Q\cup Z)$ and
    define the actual operators
    \begin{align}
        M_{\mathrm{loc}} & =\int_{[-T,T]}
        E_{N(Q)}\bigl(e^{itK'}We^{-itK'}\bigr)U(t)\,d\gamma_\eta(t),\notag   \\
        M_T              & =\int_{[-T,T]}e^{itK'}We^{-itK}\,d\gamma_\eta(t).
    \end{align}
    Then $M_{\mathrm{loc}}$ is supported on $B$ and
    \begin{align}
        \|M_{\mathrm{loc}}\| & \le\|W\|,\notag                  \\
        \|M_{\mathrm{loc}}-M_T\|
                             & \le\|W\|C L^2(1+T)e^{vT-\ell/2}.
    \end{align}
    These assertions include the empty chain and empty supports.
    This is the localized integral in the localization paragraph of the
    information-reset proof in Part~II of~\cite{TNLean2026OneDimensional},
    on a contiguous chain; the domain distinction is recorded
    in~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:chain_swapped_doubled_hamiltonian,
        thm:chain_relative_generator_localization,thm:chain_lieb_robinson,
        lem:chain_site_norm_budget,thm:area_law_supported_identity,
        thm:qc_supported_projected_relative_evolution,
        lem:qc_site_expectation_properties}
    Let $a,v$ be the constants in
    Theorem~\ref{thm:chain_relative_generator_localization}, and put
    $d=4|J|a(R+1)^2N_R$ and $C=a+d>0$.
    Construct the supported projected relative evolution on $N(Z)$.
    Its equation uses only $K,K'$ and $N(Z)$, so this choice precedes
    $Q,W,\eta,T$.
    For $|t|\le T$, the generator estimate and the evolution comparison give
    \begin{align}
        \|e^{itK'}e^{-itK}-U(t)\|
        \le dL|t|e^{vT-\ell/2}.
    \end{align}
    The separately localized observable has error at most
    $a|Q|^2\|W\|e^{vT-\ell/2}$.
    Using the exact factorization
    \begin{align}
        \bigl(e^{itK'}We^{-itK'}\bigr)
        \bigl(e^{itK'}e^{-itK}\bigr)=e^{itK'}We^{-itK},
    \end{align}
    insert $e^{itK'}We^{-itK'}U(t)$ between the two integrands.
    Unitarity of $U(t)$, invariance of the observable norm under evolution,
    and the two estimates bound their difference by
    \begin{align}
        \|W\|\bigl(a|Q|^2+dL|t|\bigr)e^{vT-\ell/2}
         & \le\|W\|(a+d)L^2(1+T)e^{vT-\ell/2}.
    \end{align}
    Here $|Q|\le L$ and $L\le L^2$ for every natural $L$; the latter is
    zero at $L=0$ and follows from $1\le L$ otherwise.

    Both factors of the localized integrand act on $B$, and its norm is
    at most $\|W\|$. The integrands are continuous and bounded, hence
    integrable for every $\gamma_\eta$. Since the expectation $E_B$ is
    continuous and linear, commutes with integration, and fixes the
    localized integrand, it also fixes $M_{\mathrm{loc}}$.
    This proves its support. Finally,
    $\gamma_\eta([-T,T])\le1$ gives both the contraction bound and,
    upon integrating the pointwise difference, the asserted error.
    The Gaussian mass is not renormalized.
\end{proof}

\begin{lemma}[Separation of the reset support from the cores]
    \label{lem:chain_reset_support_separation}
    \lean{TNLean.MPS.AreaLaw.chain_reset_support_separation}
    \leanok
    \uses{def:chain_model}
    Let $I=\{0,\ldots,L-1\}$ be a contiguous chain, let $R$ be a
    natural number, and let $D,C,R_0,C_0\subseteq I$, with $D\cap C=\varnothing$.
    Suppose that a real number $w$ satisfies
    \begin{align}
        w & <|r-y| &  & (r\in R_0,\ y\in I\setminus D),\notag \\
        w & <|c-x| &  & (c\in C_0,\ x\in I\setminus C).
    \end{align}
    Put $Q=I\setminus(D\cup C)$. Let $\mathcal C_D$ consist of all
    nonempty supports $X\subseteq I$ of diameter at most $R$ that meet
    both $D$ and its complement, and put $Z=\bigcup_{X\in\mathcal C_D}X$.
    Then
    \begin{align}
        w-R & <|a-z| &  & (z\in Q\cup Z,\ a\in R_0\cup C_0).
    \end{align}
    Consequently, for every real $\ell$ with $\ell+R\le w$,
    \begin{align}
        \{y\in I:\ \exists z\in Q\cup Z,\ |z-y|\le\ell\}
        \cap(R_0\cup C_0)=\varnothing.
    \end{align}
    Empty chains and empty sets are allowed. No sign assumption on
    $w$ or $\ell$, or containment of the cores in the outer regions,
    is needed beyond the displayed hypotheses.
    This is the support-separation step in Part~II
    of~\cite{TNLean2026OneDimensional}, lines 419--425, on a contiguous
    chain; see~\cite{gap:tnlean_one_dimensional_connected_domain_scope}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    If $z\in Q$, the two margin inequalities apply directly; weakening
    the lower bound from $w$ to $w-R$ is permitted since $R\ge0$.
    If $z\in Z$, choose an actual crossing support $X$ containing $z$,
    together with $y\in X\setminus D$ and $x\in X\cap D$.
    The diameter bound gives $|z-y|,|z-x|\le R$.
    For $r\in R_0$, the first margin and the triangle inequality give
    \begin{align}
        w<|r-y|\le|r-z|+R.
    \end{align}
    Since $D$ and $C$ are disjoint, $x\notin C$; the second margin
    gives the same conclusion for every $c\in C_0$.
    This proves the first assertion with the exact loss $R$.
    If a core site $a$ also belonged to the displayed neighborhood,
    there would be $z\in Q\cup Z$ with $|z-a|\le\ell$.
    Symmetry of distance would then give
    $w-R<|a-z|\le\ell\le w-R$, a contradiction.
\end{proof}

    \end{align}

        2\bigl(S(X\cup T)-S(T)\bigr)\le S(X)+

    \end{align}

\end{lemma}
\begin{proof}

    \uses{lem:chain_crossing_count, def:chain_model}

    stationarity, the energy estimate, the prefix comparison and Schmidt
    pinning) of Lemma~3.2 of \cite{OpenAI2026AreaLaw}, applied to the nested
    shells $X_j=X^{+(j+1)R}$, $0\le j<m$, with the uniform weights $a_j=2/m$ of
    total mass two. The shells are intervals, so each is split by at most $N_R$

    Lemma~\ref{lem:chain_crossing_count}, and no support is split by two of

    $1024e\vartheta\sum_jB_ja_j^2\le4096e\vartheta B_R/m$, and the overlap
    losses to $m\log2$.
\end{proof}

\begin{lemma}[Araki--Lieb and the area law from interval bounds]
    \label{lem:chain_entropy_inequalities}
    \lean{
        TNLean.MPS.AreaLaw.regionEntropy_union_le,
        TNLean.MPS.AreaLaw.regionEntropy_le_union_add,
        TNLean.MPS.AreaLaw.regionEntropy_empty,
        TNLean.MPS.AreaLaw.regionEntropy_univ,
        TNLean.MPS.AreaLaw.regionEntropy_le_card_mul_log,
        TNLean.MPS.AreaLaw.boundaryEdges,
        TNLean.MPS.AreaLaw.mem_boundaryEdges,
        TNLean.MPS.AreaLaw.boundaryEdges_empty,
        TNLean.MPS.AreaLaw.boundaryEdges_univ,
        TNLean.MPS.AreaLaw.exists_interval_subset_boundaryEdges_sdiff_ssubset,
        TNLean.MPS.AreaLaw.regionEntropy_le_mul_card_boundaryEdges_of_intervals
    }
    \leanok
    \uses{def:chain_model}
    For a unit vector $\Omega$ on the chain and disjoint sets of sites $X,T$,
    $S(X\cup T)\le S(X)+S(T)$ and $S(X)\le S(X\cup T)+S(T)$; the empty set and
    the whole chain have entropy zero, and $S(A)\le|A|\log q$. The
    \emph{boundary edges} of a set $A$ are the edges $\{p,p+1\}$ of the chain
    with exactly one endpoint in $A$. If every interval has entropy at most
    $C\ge0$, then every subset $A$ has entropy at most $C\,|\partial A|$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_model}
    Subadditivity and purity are the finite-product entropy inequalities of the
    companion library; the Araki--Lieb inequality is purity,
    $S(X)=S(\Lambda\setminus X)$, followed by subadditivity for the disjoint
    union $\Lambda\setminus X=(\Lambda\setminus(X\cup T))\cup T$. The dimension
    bound is the entropy bound $\log\dim$ for a trace-one positive state. For
    the last claim, induct on the number of sites of $A$. If $A$ is empty or
    the whole chain its entropy vanishes. Otherwise split off the maximal run
    $I$ of consecutive sites starting at the least position of $A$.
    Subadditivity gives $S(A)\le S(I)+S(A\setminus I)\le C+S(A\setminus I)$,
    every boundary edge of $A\setminus I$ is a boundary edge of $A$, and $A$
    has at least one more boundary edge (the edge at the right end of $I$ when
    $A\setminus I$ is nonempty, and an edge at an end of $I$ otherwise, since
    $I$ is then a proper nonempty subset of the chain).
\end{proof}

\begin{definition}[Open-boundary matrix product states and cut matrices]
    \label{def:chain_open_mps}
    \lean{
        TNLean.MPS.AreaLaw.leftBlock,
        TNLean.MPS.AreaLaw.mem_leftBlock,
        TNLean.MPS.AreaLaw.leftBlock_eq_interval,
        TNLean.MPS.AreaLaw.cutMatrix,
        TNLean.MPS.AreaLaw.cutMatrix_apply,
        TNLean.MPS.AreaLaw.IsOpenMPS,
        TNLean.MPS.AreaLaw.IsOpenMPS.smul
    }
    \leanok
    \uses{def:chain_model}
    A vector $\psi$ on $L$ sites of local dimension $q$ is an
    \emph{open-boundary matrix product state of bond dimension at most $D$}
    when there are matrices $A_i(s)\in\mathbb C^{D\times D}$ for each site $i$
    and value $s$ and vectors $v_L,v_R\in\mathbb C^D$ with
    \begin{align}
        \psi(x_0,\ldots,x_{L-1})=v_L^{\mathsf T}A_0(x_0)\cdots A_{L-1}(x_{L-1})\,v_R .
    \end{align}
    Scaling such a vector scales $v_R$. The \emph{left block} of $k$ sites is
    the interval $[0,k)$, and the \emph{cut matrix} at $k$ has the
    configurations of the left block as rows and those of the remaining sites
    as columns; its rank is the Schmidt rank across the cut.
\end{definition}

\begin{lemma}[Bounded cut ranks give a matrix product state]
    \label{lem:chain_mps_of_cut_rank}
    \lean{
        Submodule.exists_fin_span_of_finrank_le,
        TNLean.MPS.AreaLaw.glueLeft,
        TNLean.MPS.AreaLaw.sliceMatrix,
        TNLean.MPS.AreaLaw.sliceMatrix_apply,
        TNLean.MPS.AreaLaw.rank_sliceMatrix_le,
        TNLean.MPS.AreaLaw.sliceSpace,
        TNLean.MPS.AreaLaw.finrank_sliceSpace_le,
        TNLean.MPS.AreaLaw.glueLeft_snoc,
        TNLean.MPS.AreaLaw.slice_mem_sliceSpace,
        TNLean.MPS.AreaLaw.isOpenMPS_of_cutMatrix_rank_le
    }
    \leanok
    \uses{def:chain_open_mps}
    A vector all of whose cut matrices have rank at most $D$ is an
    open-boundary matrix product state of bond dimension at most $D$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_open_mps}
    Let $V_k$ be the span of the columns of the cut matrix at $k$, a space of
    dimension at most $D$, and choose a spanning family $u_k$ of $D$ vectors.
    Slicing a column of the cut matrix at $k+1$ at a fixed value of site $k$
    gives a column of the cut matrix at $k$, so each $u_{k+1}(\alpha)$ sliced
    at the value $s$ is a combination $\sum_\beta A_k(s)_{\beta\alpha}u_k(\beta)$.
    By induction $u_k(\alpha)=v_L^{\mathsf T}A_0\cdots A_{k-1}$ with $v_L=u_0$,
    and at $k=L$ the vector itself lies in $V_L$, which supplies $v_R$.
\end{proof}

\begin{lemma}[Sequential truncation]
    \label{lem:chain_sequential_truncation}
    \lean{
        Matrix.schmidtCoeffMatrix_kronecker_one_mulVec,
        Matrix.toEuclideanLin_one_apply,
        Matrix.toEuclideanLin_mul_apply,
        Entropy.localLift_sub,
        Entropy.exists_localLift_eq_localLift_of_subset,
        Entropy.norm_sq_toEuclideanLin_localLift_eq_trace,
        Entropy.norm_toEuclideanLin_localLift_le,
        Entropy.trace_regionState_eq_one,
        Matrix.PosSemidef.trace_complement_spectralProjectionGE_le_surprisalTail,
        Matrix.PosSemidef.rank_spectralProjectionGE_exp_le,
        TNLean.MPS.AreaLaw.cutMatrix_localLift,
        TNLean.MPS.AreaLaw.rank_cutMatrix_localLift_le,
        TNLean.MPS.AreaLaw.rank_cutMatrix_zero_le,
        TNLean.MPS.AreaLaw.rank_cutMatrix_le_one_of_le,
        TNLean.MPS.AreaLaw.LeftBlockMatrix,
        TNLean.MPS.AreaLaw.cutTruncation,
        TNLean.MPS.AreaLaw.sequentialTruncation,
        TNLean.MPS.AreaLaw.leftBlock_mono,
        TNLean.MPS.AreaLaw.norm_sub_sequentialTruncation_le,
        TNLean.MPS.AreaLaw.exists_sequentialTruncation_eq_localLift,
        TNLean.MPS.AreaLaw.rank_cutMatrix_sequentialTruncation_le,
        TNLean.MPS.AreaLaw.exists_sequentialTruncation
    }
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi$ be a unit vector on $L$ sites and let $w$ be a real number. For
    each cut $k$,
    $1\le k\le L-1$, let $S_k$ be the entropy of the first $k$ sites, let
    $\Pi_k$ be the spectral projector of their reduced state onto the
    eigenvalues at least $e^{-(S_k+w)}$, an operator on the first $k$ sites,
    and let $P_k=\Pi_k\otimes1$ be its extension by the identity on the
    remaining sites. Then the regional projector $\Pi_k$ has rank at most
    $e^{S_k+w}$, the mass $\|(1-P_k)\psi\|^2$ discarded by $P_k$ is at most
    the surprisal tail of the reduced state beyond $w$, and the vector
    $\Phi=P_1P_2\cdots P_{L-1}\psi$ has cut rank at most
    $\max\{1,e^{S_k+w}\}$ across every cut $k$ of the chain, the trivial
    cuts at its ends included; it satisfies
    \begin{align}
        \|\psi-\Phi\|\le\sum_{k=1}^{L-1}\sqrt{\text{tail}_k(w)} .
    \end{align}
    This is the threshold form of the sequential truncation, Proposition~7.1
    of \cite{TNLean2026OneDimensional}: the note projects every cut onto its
    $D$ largest eigenvalues, or onto all of them when the cut has fewer than
    $D$, for an arbitrary bond dimension $D$, with the error
    $\sum_k\sqrt{\epsilon_k(D)}$; that form is not formalized, and
    Theorem~B uses only the threshold form; see
    \cite{gap:tnlean_one_dimensional_threshold_truncation}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:chain_open_mps}
    Write $M_k(\psi)$ for the cut matrix of $\psi$ at $k$, with the
    configurations of the first $k$ sites as rows. For an operator $\Pi$ on
    the first $k$ sites, the extension $\Pi\otimes1$ acts on it by left
    multiplication,
    \begin{align}
        M_k\bigl((\Pi\otimes1)\psi\bigr)=\Pi\,M_k(\psi),
    \end{align}
    so the cut rank of $(\Pi\otimes1)\psi$ is at most the rank of $\Pi$; the
    rank of the extended operator itself, which is the rank of $\Pi$ times the
    dimension of the remaining sites, plays no role. The projector on the
    smallest block is
    applied last, and the product of the projectors on blocks of size at most
    $k$ is one extension of an operator on the first $k$ sites ending in
    $\Pi_k$. The distance follows by induction from
    $\psi-P_k\Phi'=(1-P_k)\psi+P_k(\psi-\Phi')$ and the contraction property
    of a projector, and $\|(1-P_k)\psi\|^2$ is the trace of $1-\Pi_k$ against
    the reduced state, the mass of the eigenvalues below the threshold, which
    is at most the surprisal tail.
\end{proof}

\begin{theorem}[Theorem B: polynomial matrix product state approximation]
    \label{thm:chain_mps_approximation}
    \lean{
        TNLean.MPS.AreaLaw.tailPrefactorRoot,
        TNLean.MPS.AreaLaw.tailPrefactorRoot_pos,
        TNLean.MPS.AreaLaw.tailPrefactorRoot_sq,
        TNLean.MPS.AreaLaw.one_lt_tailPrefactorRoot,
        TNLean.MPS.AreaLaw.chainTruncationWidth,
        TNLean.MPS.AreaLaw.chainTruncationWidth_nonneg,
        TNLean.MPS.AreaLaw.exp_neg_chainTruncationWidth,
        TNLean.MPS.AreaLaw.chainBondDimension,
        TNLean.MPS.AreaLaw.exists_isOpenMPS_norm_sub_le,
        TNLean.MPS.AreaLaw.chainBondDimension_le_mul_rpow
    }
    \leanok
    \uses{def:chain_model, def:chain_open_mps}
    Let $\Omega$ be a gapped ground vector, with gap $\Delta>0$, of a
    finite-range Hamiltonian with term norms at most $J\ge0$ on the chain of
    $L\ge1$ sites with local dimension $q\ge1$. Put
    $s=\sqrt{(1+J/\Delta)B_R}$, $c_0=\sqrt{2e^{e/2}}$ and
    \begin{align}
        w(L)=64\,s\log\bigl(2L^2c_0\bigr),\qquad
        D(L)=\bigl\lfloor\exp\bigl(C_{\mathrm{int}}+w(L)\bigr)\bigr\rfloor
        \le e^{C_{\mathrm{int}}}(2c_0)^{64s}\,L^{128s}.
    \end{align}
    There is a normalized open-boundary matrix product state $\Phi$ of bond
    dimension at most $D(L)$ with $\|\Phi-\Omega\|\le1/L$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:chain_area_law, lem:chain_marginal_tail,
        lem:chain_sequential_truncation, lem:chain_mps_of_cut_rank,
        lem:chain_crossing_count}
    By Theorem~\ref{thm:chain_area_law} the entropy of every left block, an
    interval $[0,k)$, is at most $C_{\mathrm{int}}$, and by the tail bound of
    Lemma~\ref{lem:chain_marginal_tail} with the budget $B_R$ of
    Lemma~\ref{lem:chain_crossing_count} its surprisal tail beyond $w$ is at
    most $2e^{e/2}e^{-w/(32s)}$. The sequential truncation of
    Lemma~\ref{lem:chain_sequential_truncation} at width $w(L)$ gives cut
    ranks at most $e^{C_{\mathrm{int}}+w(L)}$ and distance at most
    $(L-1)\cdot\frac1{2L^2}\le\frac1{2L}$. Normalizing moves the vector by at
    most the same amount, so the distance is at most $1/L$, and
    Lemma~\ref{lem:chain_mps_of_cut_rank} exhibits the matrix product
    structure, which survives the normalization.
\end{proof}
