\begin{definition}[$p$-divisible CP map]
    \label{def:p_divisible_channel}
    \lean{MPSTensor.IsPDivisibleChannel}
    \leanok
    A linear endomorphism $\E$ of $\MD$ is \emph{$p$-divisible}
    if there exists a CPTP map $\E'$ of $\MD$ such that
    $\E = (\E')^p$.
\end{definition}

\begin{definition}[$p$-refinable MPS tensor]
    \label{def:p_refinable}
    \lean{MPSTensor.IsPRefinable}
    \leanok
    \uses{def:mps_tensor, def:blocked_tensor}
    Given a tensor $B$, the family $\mc V(B)$ can be \emph{$p$-refined} if
    there exist another tensor $A$ and an isometry
    $W : \C^d \to (\C^d)^{\otimes p}$ such that, for every $N$,
    \begin{align}
        \ket{V^{(pN)}(A)}
         & = W^{\otimes N}\ket{V^{(N)}(B)}.
        \notag
    \end{align}
    Equivalently, after identifying a $p$-letter with an element of
    $\{0,\ldots,d^p-1\}$, the $p$-blocked tensor $A^{[p]}$ satisfies
    \begin{align}
        V^{(N)}(A^{[p]})_\tau
         & =
        \sum_{\sigma \in \{0,\ldots,d-1\}^N}
        \left(\prod_k W_{\tau(k),\sigma(k)}\right)
        V^{(N)}(B)_\sigma,
        \notag
    \end{align}
    for every $N \in \N$ and every
    $\tau \in \{0,\ldots,d^p-1\}^N$.
\end{definition}

\begin{theorem}[The one-site boundary of refinement and divisibility]
    \label{lem:periodic_refinement_one_site}
    \lean{MPSTensor.isPRefinable_one, MPSTensor.isPDivisibleChannel_one_iff}
    \leanok
    \uses{def:p_refinable, def:p_divisible_channel}
    Every tensor is $1$-refinable. A linear map is $1$-divisible if and only
    if it is a channel. Thus, at $p=1$, the forward implication of the
    printed Theorem~4.1 requires trace preservation of $\E_B$; see
    \cite{gap:dccsp17_thm41_forward_trace_preservation}.
\end{theorem}

\begin{proof}\leanok
    One-site blocking only permutes the physical labels. Use the associated
    permutation matrix for the refinement isometry. The channel assertion
    follows because the first power of a linear map is the map itself.
\end{proof}

\begin{theorem}[The tensor $C$ associated to a refinement witness]
    \label{thm:p_refinement_pullback_irreducible_form}
    \lean{MPSTensor.pRefinementCanonicalization_pullback_of_irreducibleForm}
    \leanok
    \uses{def:irreducible_form, def:p_refinable}
    Let $B$ be an MPS tensor in irreducible form~II. If $\mc V(B)$ can be
    $p$-refined, then there exist a tensor $A$, an isometry
    $W : \C^d \to (\C^d)^{\otimes p}$, and a tensor $C$ defined by
    $C^\alpha := \sum_{i=0}^{d-1} W_{\alpha,i}B^i$ for
    $\alpha \in \{0,\ldots,d^p-1\}$,
    such that $C$ is again in irreducible form~II, $\E_C = \E_B$,
    and $C$ has the same MPV family as $A^{[p]}$.
\end{theorem}

\begin{proof}\leanok
    Choose $(A,W)$ from the refinement data. The definition of $C$ gives
    $\E_C = \E_B$ and equality of MPV families between $C$
    and $A^{[p]}$. To see that $C$ is still in irreducible form~II, transport
    each periodic block $B_j$ of $B$ through the same physical isometry $W$.
    The left-canonical normalization is preserved because
    $W^\dagger W = \Id$,
    while irreducibility and the peripheral spectrum are preserved because the
    transfer map of each transported block agrees with that of $B_j$.
    Combining these transported blocks with the same weights gives an
    irreducible-form decomposition of $C$.
\end{proof}

\begin{theorem}[Normalization from periodic equal-case matching]
    \label{thm:periodic_equalcase_weight_norm}
    \lean{MPSTensor.SectorDecomposition.weight_norm_eq_one_of_periodic_sameMPV₂Pos,
        MPSTensor.SectorDecomposition.leftCanonical_of_periodic_sameMPV₂Pos}
    \leanok
    \uses{thm:periodic_ft}
    Let $P,Q$ be literal sector decompositions with trace-preserving
    periodic basis tensors, pairwise inequivalent up to scalar gauge
    within each basis. Suppose their periodic vectors agree at every
    positive length. If every copy weight of $Q$ has modulus one, then
    every copy weight of $P$ has modulus one and the assembled tensor
    $P$ is trace preserving.
\end{theorem}

\begin{proof}\leanok
    Equal-case matching gives
    $z_{j,q}\xi_j\mu_{j,q}=\nu_{\pi(j),\tau_j(q)}$, where
    $|\xi_j|=1$ and $z_{j,q}^{m_j}=1$ with $m_j>0$.
    Taking absolute values gives $|\mu_{j,q}|=1$.
    Each weighted block is therefore trace preserving, as is their
    direct sum.
\end{proof}

\begin{theorem}[Unitary gauges within a periodic phase class]
    \label{thm:periodic_phase_class_unitary}
    \lean{MPSTensor.MPVBlockPhaseEquiv.exists_unitary_of_isPeriodic,
      MPSTensor.hetRepeatedBlocks_of_mpvBlockPhaseEquiv_of_isPeriodic,
      MPSTensor.dim_eq_of_mpvBlockPhaseEquiv_of_isPeriodic}
    \leanok
    Let $A,B$ be trace-preserving periodic tensors of periods $m,n$.
    Suppose there is a nonzero scalar $\lambda$ such that
    $v_N(B)=\lambda^Nv_N(A)$ for every $N\geq1$.
    Their bond dimensions agree, and there are a unitary $U$ and a
    scalar $\zeta$ of modulus one such that $A^i=\zeta UB^iU^*$.
\end{theorem}

\begin{proof}\leanok
    If the blocks were not repeated, their cross overlap would tend
    to zero. Proportionality gives
    \[
      |\langle v_N(A),v_N(B)\rangle|^2
      =\|v_N(A)\|^2\|v_N(B)\|^2.
    \]
    Along common multiples of $m$ and $n$, the right side tends
    to $mn>0$, a contradiction. The repeated-block relation thus
    gives a similarity and a unit-modulus phase. Canonical
    normalization and irreducibility allow the similarity to be
    chosen unitary.
\end{proof}

\begin{theorem}[Grouping periodic blocks while retaining copy scalars]
    \label{thm:periodic_unitary_phase_class_grouping}
    \lean{MPSTensor.exists_unitary_phaseClass_grouping}
    \leanok
    \uses{thm:periodic_phase_class_unitary, thm:periodic_unitary_block_assembly}
    Let $(\mu_kA_k)_k$ be a finite family of trace-preserving periodic
    blocks with nonzero weights. There is a sector decomposition
    with pairwise non-repeated periodic representatives $B_j$ and
    nonzero copy weights $\nu_s$, together with a bijection $e$
    from its copies to the original blocks. The representative of
    copy $s$ and $A_{e(s)}$ have the same bond dimension and period,
    and $|\nu_s|=|\mu_{e(s)}|$.

    For every family of copy scalars $z_s$, there is a unitary $Y$
    between the assembled bond spaces such that
    \[
      \bigoplus_s z_s\nu_sB_{j(s)}^i
      =Y\left(\bigoplus_k z_{e^{-1}(k)}\mu_kA_k^i\right)Y^*.
    \]
\end{theorem}

\begin{proof}\leanok
    Partition the blocks by proportionality of their periodic vectors
    through powers of a nonzero scalar. Choose one representative per
    class. Theorem~\ref{thm:periodic_phase_class_unitary} gives
    $B_j^i=\zeta_sU_sA_{e(s)}^iU_s^*$, with $|\zeta_s|=1$.
    Set $\nu_s=\mu_{e(s)}/\zeta_s$. These weights have the required
    moduli, and repeated periodic blocks have equal periods.
    Multiplying each weighted identity by $z_s$ and assembling the
    block unitaries proves the formula.
\end{proof}

\begin{theorem}[Unitary periodic matching with unchanged phases]
    \label{thm:periodic_equalcase_unitary_matching}
    \lean{MPSTensor.fundamentalTheorem_periodic_equalCase_unitary_matching}
    \leanok
    \uses{thm:periodic_ft}
    Under the periodic equal-case hypotheses with trace-preserving
    basis tensors, the matched basis gauges may be chosen unitary
    without changing the phases in the copy-weight identity. Thus
    $A_j^i=\xi_jU_jB_{\pi(j)}^iU_j^*$ and
    $z_{j,q}\xi_j\mu_{j,q}=\nu_{\pi(j),\tau_j(q)}$ hold with the
    same $\xi_j$, where $z_{j,q}^{m_j}=1$.
\end{theorem}

\begin{proof}\leanok
    Retain the matching, phases, and copy permutations supplied by the
    equal-case theorem. Canonical normalization and irreducibility make
    each matching similarity a scalar multiple of a unitary, so its
    scalar cancels in conjugation. The factor $\xi_j$ and all copy-weight
    identities remain unchanged.
\end{proof}

\begin{theorem}[Unitary assembly of matched weighted blocks]
    \label{thm:periodic_unitary_block_assembly}
    \lean{MPSTensor.toTensorFromBlocks_unitary_conj_of_matched_blocks}
    \leanok
    Suppose a bijection $\pi$ matches the dimensions of two finite
    families of blocks and unitaries $U_{\pi(k)}$ satisfy
    $\mu_kA_k^i=\nu_{\pi(k)}U_{\pi(k)}B_{\pi(k)}^iU_{\pi(k)}^*$.
    There is a unitary $Y$ between the full bond spaces such that
    \[
        \bigoplus_k\mu_kA_k^i
        =Y\left(\bigoplus_l\nu_lB_l^i\right)Y^*.
    \]
\end{theorem}

\begin{proof}\leanok
    Compose the direct sum of the $U_l$ with the permutation of the
    matched block coordinates. Both maps are unitary, and their
    conjugation acts on each block by the prescribed identity.
\end{proof}

\begin{theorem}[Global unitary in the periodic equal case]
    \label{thm:periodic_equalcase_unitary_global}
    \lean{MPSTensor.fundamentalTheorem_periodic_equalCase_unitary_global}
    \leanok
    \uses{thm:periodic_equalcase_unitary_matching,
        thm:periodic_unitary_block_assembly}
    Let $A,B$ be literal sector decompositions satisfying the periodic
    equal-case hypotheses, with trace-preserving basis tensors.
    Let $m_j$ be the periods of the basis tensors of $A$, and let
    $L$ be a common multiple of the $m_j$. There are scalars
    $z_{j,q}$ with $z_{j,q}^{m_j}=1$ and a unitary $Y$ such that,
    on setting $Z=\bigoplus_{j,q}z_{j,q}\Id_{D_j}$,
    \[
        Z^L=\Id,\qquad ZA^i=A^iZ,\qquad ZA^i=YB^iY^*.
    \]
    The periodic vectors of $A$ and $ZA$ agree at every positive
    length. This is the unitary relation used in
    \cite[Theorem~4.1, lines~752--765]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    Unitary matching gives
    $A_j^i=\xi_jU_jB_{\pi(j)}^iU_j^*$ and
    $z_{j,q}\xi_j\mu_{j,q}=\nu_{\pi(j),\tau_j(q)}$.
    Apply Theorem~\ref{thm:periodic_unitary_block_assembly} to these
    weighted copies. Since $Z$ is scalar on each copy, it commutes
    with $A^i$, and $Z^L=\Id$. At length $N$, a copy contributes
    zero unless $m_j\mid N$; otherwise its additional factor
    $z_{j,q}^N$ is one.
\end{proof}

\begin{theorem}[Distributing a phase across a blocked cyclic word]
    \label{lem:periodic_phase_distribution}
    \lean{MPSTensor.phaseDistribution_of_residue_coordinates}
    \leanok
    Let $m,n\geq 1$ and $p\geq 0$. Suppose each cyclic site
    $u\in\mathbb Z/m\mathbb Z$ has an orbit label $\alpha(u)$ and a
    coordinate $k(u)\in\mathbb Z/n\mathbb Z$ such that
    $\alpha(u+p)=\alpha(u)$ and $k(u+p)=k(u)+1$. Every orbit has a
    label, and two sites have the same label exactly when their
    difference is an integer multiple of $p$ modulo $m$. Thus there
    are exactly $\gcd(m,p)$ orbit labels.
    Given phases $c_a$ with $c_a^n=1$, put
    $b_u=c_{\alpha(u)}^{k(u)}$ and $d_u=b_{u+1}/b_u$, where the exponent
    is the representative in $\{0,\ldots,n-1\}$. Then
    \begin{align}
        \prod_{j=0}^{p-1}d_{u+j}=c_{\alpha(u)}.
    \end{align}
\end{theorem}

\begin{proof}\leanok
    The product telescopes to $b_{u+p}/b_u$. The hypotheses on the orbit
    coordinates give $b_{u+p}=c_{\alpha(u)}b_u$; the closing step uses
    $c_{\alpha(u)}^n=1$.
\end{proof}

\begin{theorem}[Cyclic residue coordinates]
    \label{lem:periodic_refinement_residue_coordinates}
    \lean{MPSTensor.phase_residue_orbit_length, MPSTensor.exists_phase_residue_coordinates}
    \leanok
    Let $m\geq1$ and $p\geq0$. The shift by $p$ on
    $\mathbb Z/m\mathbb Z$ partitions the sites into orbits, each of length
    $n=m/\gcd(m,p)$. There are an orbit label $\alpha(u)$ and a coordinate
    $k(u)\in\mathbb Z/n\mathbb Z$ such that
    $\alpha(u+p)=\alpha(u)$ and $k(u+p)=k(u)+1$.
\end{theorem}

\begin{proof}\leanok
    Quotient the cyclic group by the subgroup generated by $p$ to obtain
    the orbit labels. Choose one representative in each orbit and identify
    the generated subgroup with $\mathbb Z/n\mathbb Z$; translation by $p$
    increases the resulting coordinate by one.
\end{proof}

\begin{theorem}[Existence of the distributed phases]
    \label{lem:periodic_refinement_distributed_phases}

\begin{theorem}[Phase-twisted tensor on blocked words]
    \label{thm:periodic_phase_twisted_block}
    \lean{MPSTensor.blockTensor_phaseTwisted_apply_projection}
    \leanok
    \uses{def:blocked_tensor}
    Let $P_u$ be orthogonal cyclic-sector projections summing to the
    identity, with $P_uA^i=A^iP_{u+1}$. For arbitrary complex scalars
    $d_u$, define $A'^i=\sum_u d_uP_uA^i$. Then the $p$-blocked letter
    indexed by a word $w$ of length $p$ satisfies
    \begin{align}
      A'^{[p]}_w
      &=\sum_u\left(\prod_{j=0}^{p-1}d_{u+j}\right)
        P_u A^w P_{u+p}.
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Expand the twisted tensor one letter at a time. Orthogonality leaves
    only paths through consecutive sectors, and the scalar factors multiply
    along each surviving path.
\end{proof}

\begin{theorem}[Corners of a blocked cyclic tensor]
    \label{lem:periodic_blocked_orbit_corners}
    \lean{MPSTensor.cyclic_projection_blockTensor_shift,
      MPSTensor.cyclic_projection_blockTensor_corner_zero,
      MPSTensor.cyclic_projection_blockTensor_offOrbit_zero}

    $P_u A^{[p]}_w=A^{[p]}_wP_{u+p}$. Consequently the corner
    $P_u A^{[p]}_wP_v$ vanishes unless $v=u+p$; in particular it
    vanishes when $u$ and $v$ belong to distinct orbits of the blocked
    shift.
\end{theorem}

\begin{proof}\leanok
    Apply the one-site relation successively along the word, then use
    orthogonality of distinct sector projections.
\end{proof}

\begin{theorem}[Cyclic projections under a blocked transfer map]

\begin{theorem}[Orbit corners of a blocked periodic tensor]
    \label{lem:periodic_blocked_orbit_decomposition}

\begin{theorem}[Compression to blocked shift orbits]
    \label{lem:periodic_blocked_orbit_compression}
    \lean{MPSTensor.exists_blockTensor_orbit_compression}
    \leanok
    \uses{lem:periodic_blocked_orbit_decomposition}
    For a period-$m$ tensor $A$, the $p$-blocked tensor has a direct-sum
    presentation with one compressed tensor $C_a$ for each of the
    $\gcd(m,p)$ shift orbits. There are mutually orthogonal projections
    $Q_a$ summing to the identity and isometries $V_a$ onto their ranges
    such that every blocked letter obeys
    \begin{align}
      A^{[p]}_w=\sum_a V_a C_a^w V_a^*.
    \end{align}
    Each compressed bond space is nonzero, each $C_a$ is trace
    preserving, and the unit-weight direct sum of
    the $C_a$ generates the same matrix-product vectors as $A^{[p]}$.
\end{theorem}

\begin{proof}\leanok
    Sum the cyclic projections over each shift orbit. The vanishing of
    off-orbit corners makes these orbit projections commute with every
    blocked letter. Compress each orbit to the range of its projection;
    the isometries restore the ambient letters upon summation.
\end{proof}

\begin{theorem}[Fixed space of a positive power]
    \label{lem:periodic_blocked_fixed_spectral_decomposition}

    \leanok
    \uses{lem:periodic_blocked_fixed_peripheral}
    For a positive trace-preserving map $T$ and $p>0$, the fixed space
    of $T^p$ is the sum of the peripheral eigenspaces of $T$ with

\begin{proof}\leanok

\begin{theorem}[Dimension bound for the blocked fixed space]
    \label{lem:periodic_blocked_fixed_dimension}
    \lean{MPSTensor.finrank_eigenspace_pow_one_le_gcd}
    \leanok
    \uses{lem:periodic_blocked_fixed_spectral_decomposition}
    If the peripheral spectrum of a positive trace-preserving map
    consists of all $m$th roots of unity and each peripheral eigenspace
    has dimension at most one, then the fixed space of its $p$th power,
    for $p>0$, has dimension at most $\gcd(m,p)$.
\end{theorem}

\begin{theorem}[Period under arbitrary positive blocking]
    \label{lem:periodic_arbitrary_blocking}
    \lean{MPSTensor.IsPeriodic.exists_blockTensor_periodic_orbit_decomposition,
      MPSTensor.IsPeriodic.exists_blockTensor_periodic_orbit_compression,
      MPSTensor.IsPeriodic.blockTensor_isIrreducibleForm}
    \leanok
    \uses{lem:periodic_blocked_orbit_irreducible,
      lem:periodic_blocked_orbit_root_spectrum,
      lem:periodic_compressed_spectrum_lift,
      lem:periodic_blocked_spectrum_orbit_bound}
    If a tensor has period $m$ and is blocked by $p>0$ sites, the
    blocked tensor decomposes into exactly $\gcd(m,p)$ nonzero
    irreducible orbit tensors. Each has period $m/\gcd(m,p)$, each
    has unit coefficient, and their direct sum has the same
    matrix-product vectors as the blocked tensor.
    More precisely, there are nonzero orthogonal cyclic projections
    $P_u$ summing to the identity, with $P_uA^i=A^iP_{u+1}$,
    and a surjective orbit map $\alpha$ satisfying
    $\alpha(u+p)=\alpha(u)$. The orbit projections and isometries
    satisfy
    \[
        Q_j=\sum_{\alpha(u)=j}P_u=V_jV_j^*,\qquad V_j^*V_j=\Id,
        \qquad [Q_j,A^{[p],I}]=0.
    \]
    They are pairwise orthogonal and sum to the identity. The
    compressed tensors $C_j$ have nonzero bond dimension, the stated
    period, and the literal decomposition
    \[
        V_jC_j^IV_j^*=Q_jA^{[p],I}Q_j,\qquad
        A^{[p],I}=\sum_jV_jC_j^IV_j^*.
    \]
\end{theorem}

\begin{proof}\leanok
    The orbit compression gives the direct sum and trace preservation;
    the fixed-space argument gives irreducibility. Its surviving
    cyclic projections yield every root of unity of order dividing
    $m/\gcd(m,p)$. Conversely, an eigenvalue of a compressed orbit
    lifts to the blocked transfer map, whose peripheral eigenvalues
    have order dividing that same orbit length.
\end{proof}

\begin{theorem}[Coordinate reconstruction of a direct sum]
    \label{thm:periodic_block_coordinate_reconstruction}
    \lean{MPSTensor.reindex_blockDiagonal_eq_sum_blockInclusion,

      \bigoplus_jB_j=\sum_jE_jB_jE_j^*,\qquad
      \sum_jE_jE_j^*=\Id.

\end{theorem}

\begin{proof}\leanok
    Compressing to the $(j,k)$ coordinate corner gives $B_j$
    when $j=k$ and zero otherwise. Taking $B_j=\Id$ gives
    the second identity.
\end{proof}

\begin{theorem}[Unitary coordinates for an isometric block decomposition]
    \label{thm:periodic_isometric_block_unitary}
    \lean{MPSTensor.exists_unitary_of_isometric_block_decomposition}
    \leanok
    Suppose $V_j\colon\mathbb C^{d_j}\to\mathbb C^D$ satisfy
    $V_j^*V_j=\Id$ and $\sum_jV_jV_j^*=\Id$. There is a unitary
    $Y\colon\bigoplus_j\mathbb C^{d_j}\to\mathbb C^D$ such that,
    for every family of matrices $B_j\in M_{d_j}(\mathbb C)$,
    \[
      Y\left(\bigoplus_j B_j\right)Y^*=\sum_jV_jB_jV_j^*.
    \]
\end{theorem}

\begin{proof}\leanok
    Define $Y$ by placing the matrices $V_j$ side by side. Then
    $YY^*=\sum_jV_jV_j^*=\Id$. Taking traces gives
    $D=\sum_jd_j$, so $Y^*Y=\Id$ as well. Expanding a
    block-diagonal matrix as the sum of its coordinate corners
    gives the displayed formula.
\end{proof}

\begin{theorem}[Unitary coordinates for nested isometric decompositions]

\begin{proof}\leanok

\begin{theorem}[Absorbing phases of the compressed orbit tensors]
    \label{thm:periodic_orbit_phase_twist}
    \lean{MPSTensor.IsPeriodic.exists_phaseTwisted_orbit_decomposition}
    \leanok
    \uses{lem:periodic_arbitrary_blocking, lem:periodic_phase_distribution}
    Let $A$ be a trace-preserving periodic tensor of period $m$,
    let $p>0$, and put $r=\gcd(m,p)$. The isometric orbit
    decomposition
    \[
        A^{[p],I}=\sum_{j=0}^{r-1}V_jC_j^IV_j^*
    \]
    can be chosen so that, for every family of phases
    $c_j^{m/r}=1$, there is a trace-preserving one-site tensor
    $A'$ on the same physical and bond spaces satisfying
    \[
        A'^{[p],I}=\sum_{j=0}^{r-1}c_jV_jC_j^IV_j^*.
    \]
    Each $C_j$ has nonzero bond dimension and period $m/r$.
    The orbit tensors and their isometries are independent of
    the subsequent choice of phases.
\end{theorem}

\begin{proof}\leanok
    Give a cyclic sector $u$ the phase $c_{\alpha(u)}$, where
    $\alpha$ is the orbit map. This phase is invariant under
    $u\mapsto u+p$ and has order dividing $m/r$. Distribute it
    among the one-site letters by
    Theorem~\ref{lem:periodic_phase_distribution}. Orthogonality
    removes the terminal cyclic projection from each blocked
    corner. Summing over each fibre of $\alpha$ then gives
    $c_jQ_jA^{[p],I}Q_j=c_jV_jC_j^IV_j^*$. Unit-modulus
    one-site phases preserve trace preservation.
\end{proof}

    \label{lem:periodic_phase_twist_trace_preserving}
    \lean{MPSTensor.phaseTwisted_tracePreserving}
    \leanok
    If $\sum_i (A^i)^*A^i=I$ and each $d_u$ has modulus one, then the
    tensor $A'^i=\sum_u d_uP_uA^i$ also satisfies
    $\sum_i (A'^i)^*A'^i=I$, provided the $P_u$ are complete,
    pairwise orthogonal projections.
\end{theorem}

\begin{proof}\leanok
    The matrix $U=\sum_u d_uP_u$ is unitary and $A'^i=UA^i$.
\end{proof}

\begin{theorem}[Absorbing blocked-sector phases into one-site letters]
    \label{thm:periodic_absorb_block_phases}
    \lean{MPSTensor.exists_phaseTwistedTensor_blockTensor}
    \leanok
    \uses{lem:periodic_refinement_distributed_phases, thm:periodic_phase_twisted_block,
      lem:periodic_phase_twist_trace_preserving}
    Let $P_u$ be orthogonal cyclic-sector projections summing to the
    identity, with $P_uA^i=A^iP_{u+1}$. Suppose $c_u$ is a complex unit
    phase, constant under $u\mapsto u+p$, whose order divides
    $m/\gcd(m,p)$. Then there is a tensor $A'$ on the same physical and
    bond dimensions such that every blocked letter satisfies
    \begin{align}
      A'^{[p]}_w=\sum_u c_u P_u A^w P_{u+p}.
    \end{align}
    If $A$ is trace preserving, then so is $A'$.
\end{theorem}

\begin{proof}\leanok
    Distribute each blocked phase into one-site factors and apply
    Theorem~\ref{thm:periodic_phase_twisted_block}.
\end{proof}

\begin{theorem}[A refinable positive-weight tensor without a channel square root]
    \label{thm:periodic_refinement_normalization_counterexample}
    \lean{MPSTensor.refinementScalarCounterexample,
      MPSTensor.refinement_normalization_counterexample}
    \leanok
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel}
    Let the physical and bond dimensions both be one, and take $B=(4)$.
    The tensor $B$ is literally the period-one normalized block $(1)$
    with positive weight $4$. It is $2$-refinable with root $A=(2)$,
    but its transfer map $\E_B(x)=16x$ is not $2$-divisible as a channel.
    This contradicts the forward implication of the printed
    \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}; see
    \cite{gap:dccsp17_thm41_forward_trace_preservation}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:bond_dimension_one_identity_transfer_normal,
      thm:periodic_refinement_from_blocked_transfer}
    The block $(1)$ has identity transfer map and period one.
    Since $A^{[2]}=B$, equality of the blocked and target transfer maps
    gives a refinement isometry. If $\E_B=\mathcal F^2$ for a channel
    $\mathcal F$, then $\E_B$ preserves trace. This contradicts
    $\tr \E_B(1)=16\ne 1$.
\end{proof}

\begin{remark}[Trace preservation in the forward direction]
    \label{rem:thm_4_1_forward_trace_preservation}
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel,
      thm:periodic_literal_refinement_forward,
      thm:periodic_refinement_normalization_counterexample}
    In \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}, $p$-divisibility presupposes
    trace preservation of $\E_B$, which is explicit in the hypotheses here.
    Without trace preservation, let $p \ge 1$, let $B_0$ be $p$-refinable with witness $(A_0, W)$ and in
    irreducible form~II with every multiplicity equal to one, and let
    $\lambda > 1$. Set
    \begin{align}
        B
         & = \lambda B_0,
        \qquad
        A = \lambda^{1/p} A_0.
        \notag
    \end{align}
    Then $B$ is in irreducible form~II with the blocks of $B_0$ and the
    multiplicities $\lambda$, and it is $p$-refinable with witness $(A, W)$:
    \begin{align}
        \ket{V^{(pN)}(A)}
         & = \lambda^{N}\ket{V^{(pN)}(A_0)}
        = \lambda^{N} W^{\otimes N}\ket{V^{(N)}(B_0)}
        = W^{\otimes N}\ket{V^{(N)}(B)}.
        \notag
    \end{align}
    Its transfer map satisfies
    \begin{align}
        \E_B
         & = \lambda^{2}\,\E_{B_0},
        \qquad
        \tr \E_B(\Id) = \lambda^{2}\tr \Id \neq \tr \Id,
        \notag
    \end{align}
    so $\E_B$ is not trace preserving and is not the $p$-th power of a CPTP
    map. At physical and bond dimension one, with $B_0 = (1)$, the same data
    give $C = A^{[p]} = (\lambda)$ related by the trivial $\Z_1$-gauge, and
    no left-canonical $\widetilde A$ or CPTP root has $p$-th power
    $x \mapsto \lambda^2 x$. Thus a blocked finite-order gauge
    alone does not imply existence of a trace-preserving root.
    The corrected statement takes $B$ literally in the block form of
    irreducible form~II from \cite{DeLasCuevas2017Irreducible}, as the source
    does, and adds trace preservation of $\E_B$; for such a block form this is
    equivalent to every multiplicity being one, since
    $\sum_i (B^i)^\dagger B^i = \bigoplus_j \mu_j^2 \Id$. Under this
    hypothesis the printed argument closes
    \cite{gap:dccsp17_thm41_forward_trace_preservation}. The equivalence does
    not extend to the generated-family reading of
    Definition~\ref{def:irreducible_form}: a non-unitary similarity of a
    left-canonical normal tensor generates the same family with unit weights,
    while its transfer map is not trace preserving.
\end{remark}

\begin{theorem}[Refinement from a blocked transfer identity]
    \label{thm:periodic_refinement_from_blocked_transfer}
    \lean{MPSTensor.isPRefinable_of_transferMap_eq_blockTensor}
    \leanok
    \uses{def:p_refinable}
    Let $A$ and $B$ have the same one-site physical and bond
    dimensions, and let $p>0$. If
    $\mathcal E_B=\mathcal E_{A^{[p]}}$, then $B$ is $p$-refinable.
\end{theorem}

\begin{proof}\leanok
    The two tensors give Kraus representations of the same completely
    positive map, of respective sizes $d^p$ and $d$. Since $d\le d^p$,
    isometric freedom of Kraus representations gives
    $A^{[p],\alpha}=\sum_j W_{\alpha j}B^j$ with $W^*W=I$.
    Expanding products and taking traces gives the refinement identity.
\end{proof}

\begin{theorem}[Irreducibility and period constraints on a refinement root]
    \label{thm:periodic_irreducible_root_constraints}
    \lean{MPSTensor.isIrreducibleFamily_of_blockTensor,
      MPSTensor.isIrreducibleFamily_of_transferMap_eq_blockTensor,
      MPSTensor.IsPeriodic.coprime_of_isIrreducibleFamily_blockTensor}
    \leanok
    \uses{lem:periodic_blocked_orbit_compression}
    Let $A$ be a tensor and let $p\geq0$. If its blocked tensor
    $A^{(p)}$ is irreducible, then $A$ is irreducible. In particular,
    this holds if $\mathcal E_A^p=\mathcal E_B$ for an irreducible
    tensor $B$, even when $A$ and $B$ have different physical dimensions.
    If moreover $A$ is periodic of period $m$, then $\gcd(m,p)=1$.
    These are necessary conditions for the converse construction in
    \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    A subspace invariant under every $A^i$ is invariant under every
    product of $p$ letters. Equality of transfer maps preserves
    irreducibility. For the period assertion, blocking gives
    $\gcd(m,p)$ nonzero orthogonal orbit projections, each commuting
    with every blocked letter. Irreducibility forces each projection
    to be the identity. Orthogonality then permits only one orbit.
\end{proof}

\begin{theorem}[A supplied channel root of larger Kraus rank]
    \label{thm:periodic_supplied_root_rank_obstruction}
    \lean{MPSTensor.CyclicResetCounterexample.root,
      MPSTensor.CyclicResetCounterexample.square,
      MPSTensor.CyclicResetCounterexample.choiRank_root,
      MPSTensor.CyclicResetCounterexample.choiRank_square,
      MPSTensor.CyclicResetCounterexample.root_isTP,
      MPSTensor.CyclicResetCounterexample.square_isTP,
      MPSTensor.CyclicResetCounterexample.root_isChannel,
      MPSTensor.CyclicResetCounterexample.square_isChannel,
      MPSTensor.CyclicResetCounterexample.transferMap_square_eq_pow,
      MPSTensor.CyclicResetCounterexample.square_isIrreducibleFamily,
      MPSTensor.CyclicResetCounterexample.root_isIrreducibleFamily,
      MPSTensor.supplied_root_rank_counterexample}
    \leanok
    \uses{thm:spectral_periodic_period_existence,
      thm:periodic_choi_rank_kraus_span}
    Decompose $\mathbb C^7$ into five sectors of dimensions
    $(1,2,2,1,1)$. Let $\mathcal F$ discard all off-diagonal sector
    blocks and send the diagonal block in sector $i$ to its trace
    times the normalized identity in sector $i+1$, with indices modulo five.
    Both $\mathcal F$ and $\mathcal F^2$ are irreducible channels.
    Their Kraus ranks are ten and nine, respectively. A minimal nine-letter tensor
    $B$ for $\mathcal F^2$ is trace preserving and periodic, and is
    a one-block irreducible form with weight one. Thus $\mathcal E_B$
    is $2$-divisible, but the supplied root $\mathcal F$ has no Kraus
    representation on $B$'s nine-letter physical alphabet.
\end{theorem}

\begin{proof}\leanok
    The Kraus matrices for $\mathcal F$ are the matrix units between
    consecutive sectors, divided by the square root of the output-sector
    dimension. They are linearly independent. The analogous two-step
    family represents $\mathcal F^2$, and its matrices are also linearly
    independent. The two ranks are therefore
    $2+4+2+1+1=10$ and $2+2+2+1+2=9$.
    Summing the squared Kraus matrices gives the identity in each input
    sector. The two-step matrix units propagate a nonzero invariant
    subspace through all five sectors, so $B$ is irreducible.
    Trace preservation gives spectral radius one, and the peripheral
    structure theorem supplies its positive period.
\end{proof}

\begin{remark}
    The theorem bounds no alternative root of $\mathcal E_B$.
    It rules out obtaining the required physical dimension merely by
    changing the Kraus representation of an arbitrary supplied root;
    selecting a different root remains possible. See
    \cite{gap:dccsp17_root_kraus_rank_thm41}.
\end{remark}

\begin{theorem}[Refinement from a root with bounded Kraus rank]
    \label{thm:periodic_refinement_from_bounded_root}
    \lean{MPSTensor.isPRefinable_of_channel_root_hasKrausRankLE}
    \leanok
    \uses{thm:periodic_refinement_from_blocked_transfer}
    Let $B$ have physical dimension $d$, let $p>0$, and suppose
    $\mathcal E_B=\mathcal F^p$, where $\mathcal F$ has a Kraus
    representation with at most $d$ operators. Then $B$ is $p$-refinable.
\end{theorem}

\begin{proof}\leanok
    Pad a Kraus family for $\mathcal F$ with zeros to obtain a
    $d$-letter tensor $A$. Then
    $\mathcal E_{A^{[p]}}=\mathcal F^p=\mathcal E_B$; apply
    Theorem~\ref{thm:periodic_refinement_from_blocked_transfer}.
\end{proof}

\begin{theorem}[Refinement and bounded-rank channel roots]
    \label{thm:periodic_refinement_iff_bounded_root}
    \lean{MPSTensor.isPRefinable_iff_exists_channel_root_hasKrausRankLE}
    \leanok
    \uses{thm:periodic_literal_refinement_forward,
      thm:periodic_refinement_from_bounded_root}
    Let $B=\bigoplus_k\mu_kB_k$ be a literal direct sum of
    trace-preserving periodic blocks with $|\mu_k|=1$, and let $p>0$.
    Then $B$ is $p$-refinable if and only if there is a CPTP map
    $\mathcal F$ with $\mathcal F^p=\mathcal E_B$ and Kraus rank at most
    the physical dimension $d$ of $B$.
\end{theorem}

\begin{proof}\leanok
    The construction in
    Theorem~\ref{thm:periodic_literal_refinement_forward} gives a
    left-canonical $d$-letter tensor $R$ with
    $\mathcal E_R^p=\mathcal E_B$. Its transfer map is a channel of
    Kraus rank at most $d$. Conversely, pad a Kraus representation of
    $\mathcal F$ to $d$ terms and apply
    Theorem~\ref{thm:periodic_refinement_from_bounded_root}.
\end{proof}

\begin{remark}
    Channel divisibility alone does not specify the Kraus rank of the
    selected root. Removing the rank bound in
    Theorem~\ref{thm:periodic_refinement_iff_bounded_root} requires a
    further root-selection argument; see
    \cite{gap:dccsp17_root_kraus_rank_thm41}.
\end{remark}

\begin{theorem}[Choi rank and the Kraus span]
    \label{thm:periodic_choi_rank_kraus_span}
    \lean{Channel.choiRank_mapLM_eq_finrank_span,
      Channel.choiRank_mapLM_eq_finrank_wordSpan_one}
    \leanok
    Let $\mathcal F(X)=\sum_a A^aX(A^a)^*$, and let
    $S=\spn_{\C}\{A^a\}_a$. The Choi rank of $\mathcal F$ is
    $\dim_{\C}S$, including when the bond space is zero dimensional.
\end{theorem}

\begin{proof}\leanok
    In positive bond dimension, let $C$ have columns equal to the
    vectorized Kraus operators, multiplied by the common nonzero
    normalization of the Choi matrix. That matrix is $CC^*$, whose rank
    is the rank of $C$. Vectorization preserves the dimension of the
    Kraus span. In dimension zero, both ranks are zero.
\end{proof}

\begin{theorem}[Rank under an injective positive map]
    \label{thm:periodic_injective_positive_rank}
    \lean{Matrix.rank_le_rank_map_of_injective_positive,
      Matrix.rank_le_rank_map_of_injective_positive_fintype}
    \leanok
    Let $\Phi:M_D(\C)\to M_D(\C)$ be an injective positive linear map.
    If $A\ge0$, then $\operatorname{rank}A\le\operatorname{rank}\Phi(A)$.
\end{theorem}

\begin{proof}\leanok
    The positive matrices bounded above by a positive multiple of $A$
    span the matrix algebra on its support. Positivity carries this span
    into the analogous span for $\Phi(A)$. Their complex dimensions are
    $(\operatorname{rank}A)^2$ and $(\operatorname{rank}\Phi(A))^2$,
    respectively, so injectivity gives the inequality.
\end{proof}

\begin{theorem}[Choi rank under injective composition]
    \label{thm:periodic_injective_composition_choi_rank}
    \lean{Channel.choiRank_le_comp_of_injective,
      Channel.choiRank_le_pow_of_injective}
    \leanok
    \uses{thm:periodic_injective_positive_rank}
    Let $\mathcal F$ and $\mathcal G$ be completely positive maps on
    $M_D(\C)$, and suppose $\mathcal F$ is injective. Then
    $\operatorname{rank}C_{\mathcal G}\le
    \operatorname{rank}C_{\mathcal F\circ\mathcal G}$.
    Consequently $\operatorname{rank}C_{\mathcal F}\le
    \operatorname{rank}C_{\mathcal F^p}$ for every $p>0$.
\end{theorem}

\begin{proof}\leanok
    Apply Theorem~\ref{thm:periodic_injective_positive_rank} to the
    injective positive map $\mathcal F\otimes\mathrm{id}$ and the
    positive Choi matrix $C_{\mathcal G}$. The image is
    $C_{\mathcal F\circ\mathcal G}$. Iteration gives the power inequality.
\end{proof}

\begin{theorem}[Refinement for an invertible transfer map]
    \label{thm:periodic_refinement_injective_transfer}
    \lean{MPSTensor.isPRefinable_of_isPDivisibleChannel_of_injective}
    \leanok
    \uses{thm:periodic_injective_composition_choi_rank,
      thm:periodic_refinement_from_bounded_root}
    Let $B$ have physical dimension $d$ and an invertible transfer map,
    and let $p>0$. If $\mathcal E_B$ is $p$-divisible as a channel,
    then $B$ is $p$-refinable.
\end{theorem}

\begin{proof}\leanok
    Choose a channel root $\mathcal F$ with $\mathcal F^p=\mathcal E_B$.
    It is invertible, so the preceding theorem gives
    $\operatorname{rank}C_{\mathcal F}\le
    \operatorname{rank}C_{\mathcal E_B}\le d$.
    Apply Theorem~\ref{thm:periodic_refinement_from_bounded_root}.
\end{proof}

\begin{remark}
    This proves the converse of
    \cite[Theorem~4.1]{DeLasCuevas2017Irreducible} for invertible transfer
    maps without a canonical-form assumption. The corresponding assertion
    for singular transfer maps remains unresolved; see
    \cite{gap:dccsp17_root_kraus_rank_thm41}.
\end{remark}

\begin{theorem}[Word-span dimension under invertible multiplication]
    \label{thm:periodic_root_word_span_dimension}
    \lean{Kraus.wordSpan_one_finrank_le_of_isUnit_mem_wordSpan_pred,
      Kraus.wordSpan_one_finrank_le_of_isUnit_mem_wordSpan_one}
    \leanok
    Let $S$ be the span of a finite family of square matrices, and let
    $S^q$ denote the span of products of exactly $q$ members, with
    $S^0=\C I$. If $p>0$ and $S^{p-1}$ contains an invertible matrix,
    then $\dim S\le\dim S^p$. In particular, the inequality holds
    whenever $S$ contains an invertible matrix.
\end{theorem}

\begin{proof}\leanok
    If $K\in S^{p-1}$ is invertible, the injective linear map
    $X\mapsto KX$ sends $S$ into $S^p$. For the final assertion,
    take the $(p-1)$st power of an invertible member of $S$.
\end{proof}

\begin{theorem}[Refinement from an invertible root word span]
    \label{thm:periodic_refinement_invertible_root_word_span}
    \lean{MPSTensor.choiRank_transferMap_le_of_isUnit_mem_root_wordSpan_pred,
      MPSTensor.isPRefinable_of_root_isUnit_mem_wordSpan_pred}
    \leanok
    \uses{thm:periodic_choi_rank_kraus_span,
      thm:periodic_root_word_span_dimension,
      thm:periodic_refinement_from_bounded_root}
    Let $B$ have physical dimension $d$, and let $A$ have the same
    bond dimension but any finite physical dimension. Suppose $p>0$
    and $\mathcal E_B=(\mathcal E_A)^p$. If the span of products of
    $p-1$ letters of $A$ contains an invertible matrix, then
    $\operatorname{rank}_{\mathrm{Choi}}\mathcal E_A\le
      \operatorname{rank}_{\mathrm{Choi}}\mathcal E_B\le d$,
    and $B$ is $p$-refinable.
\end{theorem}

\begin{proof}\leanok
    If $S$ is the Kraus span of $A$, the preceding two theorems give
    \begin{align}
      \operatorname{rank}_{\mathrm{Choi}}\mathcal E_A
        &=\dim S\le\dim S^p
         =\operatorname{rank}_{\mathrm{Choi}}\mathcal E_B\le d.
      \notag
    \end{align}
    A completely positive map has a Kraus representation with as many
    operators as its Choi rank. Apply
    Theorem~\ref{thm:periodic_refinement_from_bounded_root}.
\end{proof}

\begin{remark}[Continuum-limit application]\notready
    \label{rem:thm_4_1_continuum_limit}
    Refinability along arbitrarily large $p$ corresponds to infinite
    divisibility of the transfer map, which characterizes continuum limits of
    translationally invariant MPS (\cite{DeLasCuevas2017Irreducible}).
\end{remark}
