\begin{definition}[$p$-divisible CP map]
    \label{def:p_divisible_channel}
    \lean{MPSTensor.IsPDivisibleChannel}
    \leanok
    A linear endomorphism $\E$ of $\MD$ is \emph{$p$-divisible}
    if there exists a CPTP map $\E'$ of $\MD$ such that
    $\E = (\E')^p$.
\end{definition}

\begin{definition}[$p$-refinable MPS tensor]
    \label{def:p_refinable}
    \lean{MPSTensor.IsPRefinable}
    \leanok
    \uses{def:mps_tensor, def:blocked_tensor}
    Given a tensor $B$, the family $\mc V(B)$ can be \emph{$p$-refined} if
    there exist another tensor $A$ and an isometry
    $W : \C^d \to (\C^d)^{\otimes p}$ such that, for every $N$,
    \begin{align}
        \ket{V^{(pN)}(A)}
         & = W^{\otimes N}\ket{V^{(N)}(B)}.
        \notag
    \end{align}
    Equivalently, after identifying a $p$-letter with an element of
    $\{0,\ldots,d^p-1\}$, the $p$-blocked tensor $A^{[p]}$ satisfies
    \begin{align}
        V^{(N)}(A^{[p]})_\tau
         & =
        \sum_{\sigma \in \{0,\ldots,d-1\}^N}
        \left(\prod_k W_{\tau(k),\sigma(k)}\right)
        V^{(N)}(B)_\sigma,
        \notag
    \end{align}
    for every $N \in \N$ and every
    $\tau \in \{0,\ldots,d^p-1\}^N$.
\end{definition}

\begin{theorem}[Coordinates for a cyclic step]
    \label{thm:periodic_step_coordinates}
    \lean{Fin.stepOrbit_bijective, Fin.stepOrbitEquiv, Fin.stepOrbitEquiv_add_one}
    \leanok
    Let $m>0$, $p\geq0$, and $r=\gcd(m,p)$. Every residue $u$ modulo $m$
    has a unique expression $u=\alpha+pk\pmod m$, with
    $0\leq\alpha<r$ and $0\leq k<m/r$.
    Adding $p$ preserves $\alpha$ and adds one to $k$ modulo $m/r$.
    These are the coordinates used in
    \cite[proof of Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    If two such expressions agree modulo $m$, reduction modulo $r$
    identifies their representatives $\alpha$. Cancelling that common
    representative gives $pk=pk'\pmod m$, hence $k=k'\pmod{m/r}$.
    Their bounds imply $k=k'$. There are $r(m/r)=m$ pairs of coordinates,
    so this injective map to the $m$ residues is also surjective.
    Since $m$ divides $p(m/r)$, reducing $k+1$ modulo $m/r$ does not
    change $\alpha+p(k+1)$ modulo $m$.
\end{proof}

\begin{definition}[Projections for a prescribed blocking length]
    \label{def:periodic_step_orbit_projection}
    \lean{MPSTensor.stepOrbitProjection}
    \leanok
    \uses{thm:periodic_step_coordinates}
    Let $m>0$, $p\ge0$, $r=\gcd(m,p)$, and let $P_u$ be projections indexed
    modulo $m$. For $0\le\alpha<r$, set
    \begin{align}
        Q_\alpha & =\sum_{k=0}^{m/r-1}P_{[\alpha+pk]_m}. \notag
    \end{align}
    These are the grouped projections of
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}.
\end{definition}

\begin{theorem}[Compression to the step orbits]
    \label{thm:periodic_step_orbit_compression}
    \lean{MPSTensor.exists_stepOrbit_blockDecomposition,
        MPSTensor.adjoint_blockTensor_projection}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Suppose $P_u$ are nonzero orthogonal projections with $\sum_uP_u=\Id_D$,
    $A$ is left canonical, and $P_uA^i=A^iP_{u+1}$. There are positive integers
    $D_\alpha$ with $\sum_\alpha D_\alpha=D$, isometries
    $V_\alpha:\C^{D_\alpha}\to\C^D$, and left-canonical tensors $C_\alpha$
    such that
    \begin{align}
        V_\alpha^\dagger V_\alpha & =\Id_{D_\alpha}, \notag                      \\
        V_\alpha V_\alpha^\dagger & =Q_\alpha, \notag                            \\
        C_\alpha^I                & =V_\alpha^\dagger(A^{[p]})^IV_\alpha. \notag
    \end{align}
    The tensors $A^{[p]}$ and $\bigoplus_\alpha C_\alpha$ have the same MPV
    at every length, including zero. The adjoint transfer map satisfies
    $\E_{A^{[p]}}^\dagger(P_u)=P_{u+p}$, with indices taken modulo $m$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_step_coordinates}
    The step coordinates partition the original projections, so the $Q_\alpha$
    are mutually orthogonal nonzero projections summing to $\Id_D$.
    Iterating $P_uA^i=A^iP_{u+1}$ over a word of length $p$ shifts the
    projector index by $p$. Multiplying by the adjoint word and summing,
    left canonicality gives $\E_{A^{[p]}}^\dagger(P_u)=P_{u+p}$.
    Iterating the shift relation gives $P_u(A^{[p]})^I=(A^{[p]})^IP_{u+p}$.
    Since $u\mapsto u+p$ permutes the orbit of $\alpha$, summing over that
    orbit gives $Q_\alpha(A^{[p]})^I=(A^{[p]})^IQ_\alpha$.
    Compress to the ranges of these projections. Their dimensions sum to $D$,
    and the commuting projection resolution preserves both left canonicality
    and traces of every word. The empty-word trace gives the same dimension sum.
\end{proof}

\begin{theorem}[Step-orbit compression of a periodic tensor]
    \label{thm:periodic_prescribed_blocking_compression}
    \lean{MPSTensor.IsPeriodic.exists_stepOrbit_blockDecomposition}
    \leanok
    \uses{thm:periodic_step_orbit_compression}
    If $A$ is periodic of period $m$ and $p\ge0$, there is a family of nonzero
    cyclic projections $P_u$ with $\sum_uP_u=\Id_D$ and
    $P_uA^i=A^iP_{u+1}$ for which the dimensions $D_\alpha$, isometries
    $V_\alpha$, and left-canonical tensors $C_\alpha$ of
    Theorem~\ref{thm:periodic_step_orbit_compression} exist, with the same
    identities and the same MPV at every length. If $p>0$, every resulting
    block satisfies $\spec_{\mathrm{per}}(\E_{C_\alpha})=\{z\in\C:z^{m/\gcd(m,p)}=1\}$
    and is periodic of period $m/\gcd(m,p)$.
    This is the prescribed-blocking conclusion of
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_step_orbit_compression,thm:periodic_compressed_exact_spectrum,thm:periodic_compressed_irreducible}
    Take the cyclic spectral projections, with their indices negated to give
    $P_uA^i=A^iP_{u+1}$. No projection vanishes: left canonicality and the
    shift relation give $\E_A^\dagger(P_u)=P_{u+1}$, and $\E_A^\dagger(0)=0$, so
    $P_u=0$ would give $P_{u+1}=0$ and hence, around the cycle,
    $\Id_D=\sum_vP_v=0$. Apply Theorem~\ref{thm:periodic_step_orbit_compression}.
    For $p>0$, Theorem~\ref{thm:periodic_compressed_exact_spectrum} identifies
    each block's peripheral spectrum, and
    Theorem~\ref{thm:periodic_compressed_irreducible} gives irreducibility.
    Together with left canonicality, these prove periodicity.
\end{proof}

\begin{theorem}[Peripheral roots after positive blocking]
    \label{thm:periodic_blocking_peripheral_roots}
    \lean{MPSTensor.peripheralEigenvalues_blockTensor,
        MPSTensor.IsPeriodic.peripheral_blockTensor,
        MPSTensor.IsPeriodic.peripheral_compressed_blockTensor}
    \leanok
    Let $p>0$. For any tensor $A$, the peripheral eigenvalues of
    $\E_{A^{[p]}}$ are the $p$th powers of those of $\E_A$.
    If $A$ is periodic of period $m$, put $q=m/\gcd(m,p)$. Then
    \begin{align}
        \spec_{\mathrm{per}}(\E_{A^{[p]}})
         & =\{z\in\C:z^q=1\}.
        \label{eq:periodic_blocking_roots}
    \end{align}
    Moreover, if $V^\dagger V=\Id_n$, $VV^\dagger$ commutes with every
    letter of $A^{[p]}$, and $C^I=V^\dagger(A^{[p]})^IV$, every peripheral
    eigenvalue of $\E_C$ satisfies $z^q=1$.
    This is the ambient spectral calculation in
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}, with the missing power in
    the printed root set corrected as in~\cite{gap:dccsp17_blocking_peripheral_roots}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:transfer_map_corner_intertwine}
    Spectral mapping for $\E_A^p$ gives the first assertion: a complex number
    has unit modulus if and only if its positive $p$th power does.
    A primitive $m$th root raised to the $p$th power has order $q$, giving
    (\ref{eq:periodic_blocking_roots}). For compression,
    $(A^{[p]})^IV=VC^I$, so
    $\E_{A^{[p]}}(VXV^\dagger)=V\E_C(X)V^\dagger$.
    The isometry makes $X\mapsto VXV^\dagger$ injective; hence every
    eigenvalue of $\E_C$ is an eigenvalue of $\E_{A^{[p]}}$.
\end{proof}

\begin{theorem}[Cyclic motion of compressed projections]
    \label{thm:periodic_compressed_projection_shift}
    \lean{MPSTensor.adjoint_compressed_blockTensor_projection}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Suppose $P_u$ are orthogonal projections summing to $\Id_D$, $A$ is
    left canonical, and $P_uA^i=A^iP_{u+1}$. Let $p\ge0$, fix an orbit
    $\alpha$, and suppose $VV^\dagger=Q_\alpha$ and
    $C^I=V^\dagger(A^{[p]})^IV$. For $u=[\alpha+pk]_m$,
    \begin{align}
        \E_C^\dagger(V^\dagger P_uV) & =V^\dagger P_{u+p}V.
        \label{eq:periodic_compressed_projection_shift}
    \end{align}
    This is the compressed form of the final display in the proof of
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_step_orbit_compression}
    Orthogonality gives $Q_\alpha P_u=P_uQ_\alpha=P_u$. Therefore
    $V(V^\dagger P_uV)V^\dagger=P_u$. Expanding the compressed adjoint map
    yields $\E_C^\dagger(Y)=V^\dagger\E_{A^{[p]}}^\dagger(VYV^\dagger)V$.
    Apply this identity to $Y=V^\dagger P_uV$ and use
    $\E_{A^{[p]}}^\dagger(P_u)=P_{u+p}$.
\end{proof}

\begin{theorem}[Exact peripheral spectrum of each orbit block]
    \label{thm:periodic_compressed_exact_spectrum}
    \lean{MPSTensor.IsPeriodic.peripheral_compressed_stepOrbit_eq}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Let $A$ be periodic of period $m$, let $p>0$, and put
    $r=\gcd(m,p)$ and $q=m/r$. Suppose $P_u$ are nonzero orthogonal
    projections summing to $\Id_D$ with $P_uA^i=A^iP_{u+1}$.
    Fix $\alpha<r$, an isometry $V$ with $VV^\dagger=Q_\alpha$, and set
    $C^I=V^\dagger(A^{[p]})^IV$. Then
    \begin{align}
        \spec_{\mathrm{per}}(\E_C)
         & =\{\lambda\in\C:\lambda^q=1\}.
        \label{eq:periodic_compressed_exact_spectrum}
    \end{align}
    This is the spectral-set assertion in
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}, with the root notation
    corrected as in~\cite{gap:dccsp17_blocking_peripheral_roots}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_blocking_peripheral_roots,thm:periodic_compressed_projection_shift}
    The ambient spectral bound gives one inclusion. For the other, put
    $R_k=V^\dagger P_{[\alpha+pk]_m}V$. The compressed adjoint map sends
    $R_k$ to $R_{k+1}$. If $z^q=1$, the sum
    $X_z=\sum_{k=0}^{q-1}z^kR_k$ therefore satisfies
    $\E_C^\dagger(X_z)=z^{-1}X_z$. It is nonzero: since
    $P_{[\alpha+pk]_m}\le Q_\alpha=VV^\dagger$ and distinct orbit projections
    are orthogonal,
    \begin{align}
        VX_zV^\dagger P_\alpha & =\sum_{k=0}^{q-1}z^kP_{[\alpha+pk]_m}P_\alpha=P_\alpha\ne0. \notag
    \end{align}
    Given $\lambda^q=1$, choose $z=\overline{\lambda}^{-1}$.
    The adjoint has eigenvalue $\overline{\lambda}$, so $\E_C$ has
    eigenvalue $\lambda$. Its modulus is one.
\end{proof}

\begin{theorem}[Fixed spaces of positive adjoint powers]
    \label{thm:periodic_adjoint_fixed_gcd}
    \lean{MPSTensor.IsPeriodic.adjoint_pow_fixed_of_pow_fixed,
        MPSTensor.IsPeriodic.adjoint_pow_fixed_iff_gcd}
    \leanok
    Let $A$ be periodic of period $m$, let $p>0$, and write
    $F=\E_A^\dagger$ and $r=\gcd(m,p)$. If $F^p(X)=X$, then $F^m(X)=X$.
    Moreover, $F^p(X)=X$ if and only if $F^r(X)=X$.
\end{theorem}

\begin{proof}\leanok
    On $S=\ker(F^p-1)$, the restriction of $F$ is annihilated by the
    separable polynomial $t^p-1$, and hence is diagonalizable.
    Each eigenvalue has modulus one and therefore is an $m$th root
    of unity. Thus $F^m$ is the identity on $S$.
    A point of periods $m$ and $p$ also has period $r$; conversely,
    $r\mid p$ gives the other implication.
\end{proof}

\begin{theorem}[Fixed-point rigidity on one orbit of sectors]
    \label{thm:periodic_orbit_fixed_rigidity}
    \lean{MPSTensor.sum_adjoint_pow_stepOrbitProjection,
        MPSTensor.IsPeriodic.adjoint_pow_fixed_eq_smul_stepOrbitProjection}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Let $A$ be periodic of period $m$, let $p>0$, and put $r=\gcd(m,p)$.
    Suppose $P_u$ are orthogonal projections summing to $\Id_D$ with
    $P_uA^i=A^iP_{u+1}$. Write $F=\E_A^\dagger$ and fix $\alpha<r$.
    Then $\sum_{j=0}^{r-1}F^j(Q_\alpha)=\Id_D$.
    If $F^p(X)=X$ and $Q_\alpha X=X$, there is $c\in\C$ such that
    $X=cQ_\alpha$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_adjoint_fixed_gcd,thm:periodic_step_coordinates,
        thm:transfer_map_irreducible_of_tensor}
    The projections $R_j=F^j(Q_\alpha)$, for $0\le j<r$, are pairwise
    orthogonal and sum to $\Id_D$, since their cyclic-sector indices
    partition $\Z_m$. The letter relation gives
    $F(P_uY)=P_{u+1}F(Y)$, so $R_jF^j(X)=F^j(X)$.
    By Theorem~\ref{thm:periodic_adjoint_fixed_gcd}, $F^r(X)=X$.
    Thus $S=\sum_{j=0}^{r-1}F^j(X)$ is fixed by the irreducible unital
    map $F$, and $S=c\Id_D$. Orthogonality gives
    $Q_\alpha S=X=cQ_\alpha$.
\end{proof}

\begin{theorem}[Irreducibility of each compressed orbit block]
    \label{thm:periodic_compressed_irreducible}
    \lean{MPSTensor.IsPeriodic.isIrreducibleFamily_compressed_stepOrbit}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Let $A$ be periodic of period $m$, let $p>0$, and let $P_u$ be
    orthogonal projections summing to $\Id_D$ with $P_uA^i=A^iP_{u+1}$.
    Fix $\alpha<\gcd(m,p)$ and an isometry $V$ with
    $VV^\dagger=Q_\alpha$. The compressed tensor
    $C^I=V^\dagger(A^{[p]})^IV$ is irreducible.
    This supplies the irreducibility assertion in
    \cite[Lemma~6]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_orbit_fixed_rigidity,thm:adjoint_corner_fixed_of_preserved,
        thm:transfer_map_irreducible_of_tensor}
    Let $R$ be an invariant orthogonal projection for $\E_C^\dagger$.
    The projection $L=VRV^\dagger$ has invariant corner under
    $F^p$, where $F=\E_A^\dagger$. The faithful invariant state of $\E_A$
    implies $F^p(L)=L$. Since $Q_\alpha L=L$, fixed-point rigidity
    gives $L=cQ_\alpha$, and compression gives $R=c\Id$.
    Idempotence forces $R=0$ or $R=\Id$. Irreducibility is preserved
    when passing between a Kraus family and its adjoint family.
\end{proof}

\begin{theorem}[Orbit roots as consecutive products]
    \label{thm:periodic_orbit_phase_products}
    \lean{Fin.exists_stepOrbit_phases}
    \leanok
    \uses{thm:periodic_step_coordinates}
    Let $G$ be an abelian group, $m>0$, $p\ge0$, $r=\gcd(m,p)$,
    and $q=m/r$. Suppose $c_\alpha\in G$ satisfies $c_\alpha^q=1$
    for every $0\le\alpha<r$. There are $b_u\in G$, indexed modulo
    $m$, such that, writing $u=\alpha_u+pk_u$ for the coordinates of
    Theorem~\ref{thm:periodic_step_coordinates},
    \begin{align}
        \prod_{j=0}^{p-1}b_{u+j} & = c_{\alpha_u}.
        \label{eq:periodic_orbit_phase_product}
    \end{align}
    For $G=U(1)$, these are the local phases in
    \cite[proof of Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:periodic_step_coordinates}
    Put $f_u=c_{\alpha_u}^{k_u}$ and $b_u=f_{u+1}/f_u$.
    The coordinate shift and $c_\alpha^q=1$ give
    $f_{u+p}=c_{\alpha_u}f_u$, including when $k_u=q-1$.
    The product telescopes to $f_{u+p}/f_u=c_{\alpha_u}$,
    proving~(\ref{eq:periodic_orbit_phase_product}).
\end{proof}

\begin{definition}[Sector rephasing]
    \label{def:periodic_sector_rephasing}
    \lean{MPSTensor.sectorPhaseTensor}
    \leanok
    Let $A$ be an MPS tensor, let $P_0,\ldots,P_{m-1}\in\MD$, and let
    $c_0,\ldots,c_{m-1}\in\C$. Define $A'^i=\sum_u c_u P_u A^i$, as in
    \cite[before eq.~\emph{Aprime-is-cPA}]{DeLasCuevas2017Irreducible}.
\end{definition}

\begin{theorem}[Products of sector-rephased letters]
    \label{thm:periodic_sector_rephasing_words}
    \lean{MPSTensor.mul_sectorPhaseTensor,
        MPSTensor.mul_evalWord_sectorPhaseTensor,
        MPSTensor.evalWord_sectorPhaseTensor}
    \leanok
    \uses{def:periodic_sector_rephasing}
    Suppose $m\geq1$, $P_uP_v=\delta_{uv}P_u$, $\sum_u P_u=\Id$, and
    $P_uA^i=A^iP_{u+1}$, with sector indices read modulo $m$.
    For a word $w=(i_1,\ldots,i_n)$, write $A^w=A^{i_1}\cdots A^{i_n}$.
    Then
    \begin{align}
        P_u A'^w & = \left(\prod_{k=0}^{n-1}c_{u+k}\right)P_u A^w, \notag \\
        A'^w     & = \sum_u\left(\prod_{k=0}^{n-1}c_{u+k}\right)P_u A^w.
        \label{eq:periodic_sector_rephasing_words}
    \end{align}
    The empty word has product $\Id$. The case $n=p$ is the first equality of
    \cite[eq.~\emph{Aprime-is-cPA}]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{def:periodic_sector_rephasing}
    Orthogonality gives $P_u A'^i=c_uP_u A^i=c_uA^iP_{u+1}$.
    Induction on the word length gives the first identity in
    (\ref{eq:periodic_sector_rephasing_words}); summing it over $u$ gives the second.
\end{proof}

\begin{theorem}[Left canonicality under sector rephasing]
    \label{thm:periodic_sector_rephasing_left_canonical}
    \lean{MPSTensor.isLeftCanonical_sectorPhaseTensor}
    \leanok
    \uses{def:periodic_sector_rephasing}
    Suppose $P_u^\dagger=P_u$, $P_uP_v=\delta_{uv}P_u$,
    $\sum_uP_u=\Id$, and $|c_u|=1$. If $A$ is left canonical, then
    $A'^i=\sum_uc_uP_uA^i$ is left canonical as well. Thus the rephasing
    in the proof of \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}
    keeps a left-canonical block trace preserving.
\end{theorem}

\begin{proof}\leanok
    \uses{def:periodic_sector_rephasing}
    Put $U=\sum_uc_uP_u$. Orthogonality gives
    $U^\dagger U=\sum_u|c_u|^2P_u=\Id$. Since $A'^i=UA^i$,
    $\sum_i(A'^i)^\dagger A'^i=\sum_i(A^i)^\dagger U^\dagger UA^i=\Id$.
\end{proof}

\begin{theorem}[A left-canonical tensor with prescribed orbit phases]
    \label{thm:periodic_orbit_phase_tensor}
    \lean{MPSTensor.exists_isLeftCanonical_evalWord_eq_sum_orbit_phases}
    \leanok
    \uses{thm:periodic_step_coordinates, def:periodic_sector_rephasing}
    Let $A$ be left canonical and $m>0$. Suppose the Hermitian projections
    $P_u$ satisfy $P_uP_v=\delta_{uv}P_u$, $\sum_uP_u=\Id$, and
    $P_uA^i=A^iP_{u+1}$. Let $p\ge0$, $r=\gcd(m,p)$, and
    $c_\alpha^{m/r}=1$. There is a left-canonical tensor $A'$ such that,
    for every word $w$ of length $p$,
    \begin{align}
        A'^w & = \sum_u c_{\alpha_u}P_uA^w.
        \label{eq:periodic_orbit_phase_tensor}
    \end{align}
    This is the rephasing identity in
    \cite[proof of Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}\leanok
    \uses{def:periodic_sector_rephasing,thm:periodic_orbit_phase_products,
        thm:periodic_sector_rephasing_words,
        thm:periodic_sector_rephasing_left_canonical}
    Since $m/r>0$, each root $c_\alpha$ has modulus one. Apply
    Theorem~\ref{thm:periodic_orbit_phase_products} in $U(1)$ to obtain
    unit phases $b_u$, and set $A'^i=\sum_ub_uP_uA^i$.
    Theorem~\ref{thm:periodic_sector_rephasing_left_canonical} gives left
    canonicality. Substituting~(\ref{eq:periodic_orbit_phase_product}) into
    (\ref{eq:periodic_sector_rephasing_words}) gives
    (\ref{eq:periodic_orbit_phase_tensor}).
\end{proof}

\begin{theorem}[The tensor $C$ associated to a refinement witness]
    \label{thm:p_refinement_pullback_irreducible_form}
    \lean{MPSTensor.pRefinementCanonicalization_pullback_of_irreducibleForm}
    \leanok
    \uses{def:irreducible_form, def:p_refinable}
    Let $B$ be an MPS tensor in irreducible form~II. If $\mc V(B)$ can be
    $p$-refined, then there exist a tensor $A$, an isometry
    $W : \C^d \to (\C^d)^{\otimes p}$, and a tensor $C$ defined by
    $C^\alpha := \sum_{i=0}^{d-1} W_{\alpha,i}B^i$ for
    $\alpha \in \{0,\ldots,d^p-1\}$,
    such that $C$ is again in irreducible form~II, $\E_C = \E_B$,
    and $C$ has the same MPV family as $A^{[p]}$.
\end{theorem}

\begin{proof}\leanok
    Choose $(A,W)$ from the refinement data. The definition of $C$ gives
    $\E_C = \E_B$ and equality of MPV families between $C$
    and $A^{[p]}$. To see that $C$ is still in irreducible form~II, transport
    each periodic block $B_j$ of $B$ through the same physical isometry $W$.
    The left-canonical normalization is preserved because
    $W^\dagger W = \Id$,
    while irreducibility and the peripheral spectrum are preserved because the
    transfer map of each transported block agrees with that of $B_j$.
    Combining these transported blocks with the same weights gives an
    irreducible-form decomposition of $C$.
\end{proof}

\begin{theorem}[$Z$-gauge equivalence from the periodic theorem]
    \label{thm:zgauge_equiv_periodicFT_irreducible_form}
    \lean{MPSTensor.zGaugeEquiv_of_periodicEqualCaseFT_of_irreducibleForm}
    \leanok
    \uses{thm:periodic_ft, def:irreducible_form, def:blocked_tensor, def:same_mpv, def:z_gauge_equiv}
    Assume the periodic equal-case Fundamental Theorem at the blocked physical
    dimension. Let $C$ and $A^{[p]}$ be MPS tensors of the same bond dimension,

Theorem~\ref{thm:thm_4_1_p_refinement_reverse}.

\begin{theorem}[$p$-refinability iff transfer-map $p$-divisibility, conditional]%
    \label{thm:thm_4_1_p_refinement}
    \lean{MPSTensor.thm_4_1_p_refinement}
    \leanok
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel, thm:periodic_ft, thm:thm_4_1_p_refinement_forward, thm:thm_4_1_p_refinement_reverse}
    Let $B$ be an MPS tensor in irreducible form~II and let $p \ge 1$.
    Assume the forward construction of $\widetilde A$ for
    refinability $\Rightarrow$ divisibility and the reverse transfer-map
    construction for divisibility $\Rightarrow$ refinability. Then $B$ is
    $p$-refinable iff its transfer map $\E_B$ is $p$-divisible.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:thm_4_1_p_refinement_forward, thm:thm_4_1_p_refinement_reverse}
    Combine the forward implication
    (Theorem~\ref{thm:thm_4_1_p_refinement_forward}) with the reverse
    implication (Theorem~\ref{thm:thm_4_1_p_refinement_reverse}) into a
    single biconditional. The forward direction assumes the construction of
    $\widetilde A$; the reverse direction assumes the transfer-map construction.
    Together they give the full equivalence under those hypotheses.
\end{proof}

\begin{theorem}[$p$-refinability implies transfer-map $p$-divisibility]%
    \label{thm:thm_4_1_p_refinement_forward}
    \lean{MPSTensor.thm_4_1_p_refinement_forward}
    \leanok
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel}
    Let $B$ be an MPS tensor in irreducible form~II. Assume that every
    refinement witness gives a tensor $\widetilde A$ with
    $\sum_i (\widetilde A^i)^\dagger \widetilde A^i = \Id$ and
    $\E_B = \E_{\widetilde A^{[p]}}$. Then
    $p$-refinability of $B$ implies $p$-divisibility of $\E_B$.
\end{theorem}
% Formalization status: the assumed construction is refuted, not merely
% unproved (Remark~\ref{rem:thm_4_1_forward_trace_preservation}), so the
% Lean theorem carries an Unfaithful marker and this node stays conditional;
% the elimination plan is in
% docs/paper-gaps/dccsp17_thm41_forward_trace_preservation.tex.

\begin{proof}\leanok
    The construction supplies $\widetilde A$ with
    $\sum_i (\widetilde A^i)^\dagger \widetilde A^i = \Id$ and
    $\E_B = \E_{\widetilde A^{[p]}}$. The left-canonical
    property makes $\E_{\widetilde A}$ a CPTP channel, and the
    blocking identity
    $\E_{\widetilde A^{[p]}} = (\E_{\widetilde A})^p$
    then identifies $\E_B$ with the $p$-th power of the channel
    $\E_{\widetilde A}$.
\end{proof}

\begin{remark}[Trace preservation in the forward direction]
    \label{rem:thm_4_1_forward_trace_preservation}
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel, def:peripheral_equalcase_root_from_zgauge, def:peripheral_equalcase_periodicFT_sameMPV, def:peripheral_equalcase_root_channel_from_zgauge}
    In \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}, $p$-divisibility presupposes
    trace preservation of $\E_B$, which is explicit in the hypotheses here.
    Without trace preservation, let $p \ge 1$, let $B_0$ be $p$-refinable with witness $(A_0, W)$ and in
    irreducible form~II with every multiplicity equal to one, and let
    $\lambda > 1$. Set
    \begin{align}
        B
         & = \lambda B_0,
        \qquad
        A = \lambda^{1/p} A_0.
        \notag
    \end{align}
    Then $B$ is in irreducible form~II with the blocks of $B_0$ and the
    multiplicities $\lambda$, and it is $p$-refinable with witness $(A, W)$:
    \begin{align}
        \ket{V^{(pN)}(A)}
         & = \lambda^{N}\ket{V^{(pN)}(A_0)}
        = \lambda^{N} W^{\otimes N}\ket{V^{(N)}(B_0)}
        = W^{\otimes N}\ket{V^{(N)}(B)}.
        \notag
    \end{align}
    Its transfer map satisfies
    \begin{align}
        \E_B
         & = \lambda^{2}\,\E_{B_0},
        \qquad
        \tr \E_B(\Id) = \lambda^{2}\tr \Id \neq \tr \Id,
        \notag
    \end{align}
    so $\E_B$ is not trace preserving and is not the $p$-th power of a CPTP
    map. At physical and bond dimension one, with $B_0 = (1)$, the same data
    give $C = A^{[p]} = (\lambda)$ related by the trivial $\Z_1$-gauge, and
    no left-canonical $\widetilde A$ or CPTP root has $p$-th power
    $x \mapsto \lambda^2 x$; so the constructions assumed in
    Definitions~\ref{def:peripheral_equalcase_root_from_zgauge},
    \ref{def:peripheral_equalcase_periodicFT_sameMPV},
    and~\ref{def:peripheral_equalcase_root_channel_from_zgauge} fail as well.
    The corrected statement takes $B$ literally in the block form of
    irreducible form~II from \cite{DeLasCuevas2017Irreducible}, as the source
    does, and adds trace preservation of $\E_B$; for such a block form this is
    equivalent to every multiplicity being one, since
    $\sum_i (B^i)^\dagger B^i = \bigoplus_j \mu_j^2 \Id$. Under this
    hypothesis the printed argument closes
    \cite{gap:dccsp17_thm41_forward_trace_preservation}. The equivalence does
    not extend to the generated-family reading of
    Definition~\ref{def:irreducible_form}: a non-unitary similarity of a
    left-canonical normal tensor generates the same family with unit weights,
    while its transfer map is not trace preserving.
\end{remark}

\begin{definition}[Blocked-to-root channel-root hypothesis from $Z$-gauge]
    \label{def:peripheral_equalcase_root_channel_from_zgauge}
    \lean{MPSTensor.PeripheralEqualCaseRootChannelOfZGauge}
    \leanok
    \uses{def:peripheral_equalcase_root_from_zgauge, def:z_gauge_equiv, def:p_divisible_channel}
    Fix $(d,D,p)$. This sharpened reconstruction hypothesis asks for a CPTP map
    $\E'$ such that $\E_C = (\E')^p$ and the Choi/Kraus rank of
    $\E'$ is at most $d$, whenever $C$ is related to $A^{[p]}$ by a
    blocked $\Z_m$-gauge witness.
\end{definition}

\begin{theorem}[Channel-root criterion for blocked-to-root reconstruction]
    \label{thm:peripheral_equalcase_root_from_root_channel}
    \lean{MPSTensor.peripheralEqualCaseRootFromZGauge_of_rootChannel}
    \leanok
    \uses{def:peripheral_equalcase_root_channel_from_zgauge, def:peripheral_equalcase_root_from_zgauge}
    If the blocked-to-root channel-root hypothesis holds, then the tensor-level
    blocked-to-root reconstruction hypothesis holds as well.
\end{theorem}

\begin{proof}\leanok
    Choose a tensor $A'$ whose matrix family realizes the root channel with at
    most $d$ nonzero terms. Since this channel is trace-preserving, the matrices
    of $A'$ satisfy
    $\sum_i (A'^i)^\dagger A'^i = \Id$. The identity
    $\E_C = (\E_{A'})^p$ is then equivalent to
    $\E_C = \E_{A'^{[p]}}$ by the blocking formula.
\end{proof}

\begin{theorem}[Transfer-map $p$-divisibility implies $p$-refinability]%
    \label{thm:thm_4_1_p_refinement_reverse}
    \lean{MPSTensor.thm_4_1_p_refinement_reverse}
    \leanok
    \uses{def:irreducible_form, def:p_refinable, def:p_divisible_channel}
    Let $B$ be an MPS tensor in irreducible form~II and let $p \ge 1$.
    Assume the reverse transfer-map root hypothesis: there is a tensor $A$ with
    $\E_B = \E_{A^{[p]}}$ whenever $\E_B$ is
    $p$-divisible. Then $p$-divisibility of $\E_B$ implies
    $p$-refinability of $B$.
\end{theorem}

\begin{proof}\leanok
    From the reverse transfer-map root hypothesis, extract $A$ with
    $\E_B = \E_{A^{[p]}}$. This matches two finite Kraus
    families of the same completely positive map: the blocked family $A^{[p]}$
    with $d^p$ operators and the original family $B$ with $d$ operators.
    The cardinality comparison $d \le d^p$ holds whenever $p \ge 1$, so
    \cite[Theorem~2.1(4)]{Wolf2012Quantum} supplies an isometry
    $V \in M_{d^p \times d}(\C)$ with $V^\dagger V = \Id$ and
    $(A^{[p]})^\alpha = \sum_j V_{\alpha,j}B^j$. Expanding the
    coefficient of $A^{[p]}$ at the word~$\tau$ by linearity of matrix
    multiplication and of the trace, the contributions reorganise into the
    $V$-weighted sum over length-$N$ words of $B$ required by
    Definition~\ref{def:p_refinable}, exhibiting $V$ as the isometry $W$ in the
    statement of $p$-refinability.
\end{proof}

\begin{remark}[Continuum-limit application]\notready
    \label{rem:thm_4_1_continuum_limit}
    Refinability along arbitrarily large $p$ corresponds to infinite
    divisibility of the transfer map, which characterizes continuum limits of
    translationally invariant MPS (\cite{DeLasCuevas2017Irreducible}).
\end{remark}

\begin{theorem}[Prescribed blocking of a periodic tensor]
    \label{thm:periodic_prescribed_blocking}
    \lean{MPSTensor.IsPeriodic.exists_paper_cyclic_projections,
        MPSTensor.IsPeriodic.exists_periodic_stepOrbit_decomposition}
    \leanok
    \uses{def:periodic_tensor,def:periodic_step_orbit_projection}
    Let $A$ be periodic of period $m>0$, and let $p>0$.
    Set $r=\gcd(m,p)$ and $q=m/r$. There are nonzero cyclic projections
    $P_u$ with $\sum_uP_u=\Id_D$ and $P_uA^i=A^iP_{u+1}$ and isometries $V_\alpha$ onto their
    step-orbit sums $Q_\alpha$, for $0\le\alpha<r$, such that
    $B_\alpha^I=V_\alpha^\dagger(A^{[p]})^IV_\alpha$ is periodic of
    period $q$. All $r$ blocks have positive bond dimension; their dimensions
    sum to the bond dimension of $A$; and their direct sum has the same
    matrix product vectors as $A^{[p]}$ at every length.
    This is \cite[Lemma~6]{DeLasCuevas2017Irreducible}, with the displayed
    peripheral-root correction of \cite{gap:dccsp17_blocking_peripheral_roots}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:periodic_step_orbit_compression,thm:periodic_compressed_irreducible,
        thm:periodic_compressed_exact_spectrum}
    Obtain the cyclic projections from periodicity and reverse their cyclic
    indices to give the stated letter convention. Each cyclic sector has
    positive dimension $D_u$ and an isometry $W_u$ with $W_uW_u^\dagger=P_u$,
    so $P_u=0$ would give
    \begin{align}
        \Id_{D_u} & =W_u^\dagger W_uW_u^\dagger W_u=W_u^\dagger P_uW_u=0. \notag
    \end{align}
    Compress onto each orbit sum. The orbit-compression theorem gives left
    canonicality, the dimension identity, and equality of matrix product
    vectors. Each compressed block is irreducible and has exactly the $q$th
    roots of unity as its peripheral spectrum, hence has period $q$.
\end{proof}

\begin{theorem}[Absorbing the orbitwise gauge into the root]
    \label{thm:periodic_absorb_orbit_gauge}
    \lean{MPSTensor.exists_isLeftCanonical_blockTensor_eq_of_orbit_action}
    \leanok
    \uses{def:periodic_step_orbit_projection}
    Let $A$ be left canonical with orthogonal cyclic projections $P_u$
    summing to the identity and satisfying $P_uA^i=A^iP_{u+1}$.
    Let $m>0$, $p\ge0$, $r=\gcd(m,p)$, and $c_\alpha^{m/r}=1$.
    If a matrix $Z$ satisfies
    \begin{align}
        \label{eq:periodic_orbit_gauge_action}
        ZQ_\alpha(A^{[p]})^I & =c_\alpha Q_\alpha(A^{[p]})^I,
    \end{align}
    then there is a left-canonical tensor $A'$ of the same dimensions with
    $(A'^{[p]})^I=Z(A^{[p]})^I$ for every $I$.
    This is the comparison of the two phase formulas in the proof of
    \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:periodic_orbit_phase_tensor}
    Construct $A'$ using Theorem~\ref{thm:periodic_orbit_phase_tensor}.
    Regroup its cyclic-sector sum by step orbits to obtain
    $\sum_\alpha c_\alpha Q_\alpha(A^{[p]})^I$.
    Substitute (\ref{eq:periodic_orbit_gauge_action}) and use
    $\sum_\alpha Q_\alpha=\Id$.
\end{proof}

\begin{theorem}[Unitary conjugation of a blocked root]
    \label{thm:periodic_unitary_blocked_root}
    \lean{MPSTensor.exists_isLeftCanonical_blockTensor_eq_of_unitary_conj}
    \leanok
    If $A$ is left canonical and $(A^{[p]})^I=UC^IU^\dagger$ for a
    unitary $U$, then $C$ has a left-canonical root $B$ at the original
    physical dimension: $B^{[p]}=C$.
    This is the final conjugation in the proof of
    \cite[Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}
    \leanok
    Set $B^i=U^\dagger A^iU$. Unitarity preserves left canonicality and
    cancels the intermediate factors in every word product, giving
    $B^{[p]}=U^\dagger A^{[p]}U=C$.
\end{proof}

\begin{theorem}[Refining a weighted block sum]
    \label{thm:periodic_weighted_block_refinement}
    \lean{MPSTensor.sameMPV₂_toTensorFromBlocks_refinement}
    \leanok
    If $A_k$ and $\bigoplus_{a=0}^{n_k-1}B_{k,a}$ have equal matrix product
    vectors at every length, then $\bigoplus_k\mu_k A_k$ and
    $\bigoplus_{k,a}\mu_k B_{k,a}$ do as well, for arbitrary complex
    weights $\mu_k$.
\end{theorem}

\begin{proof}
    \leanok
    Expand the trace of each block-diagonal product and substitute the
    refined vectors of each $A_k$. At every length $N$,
    \begin{align}
        V^{(N)}\Bigl(\bigoplus_k\mu_kA_k\Bigr)_\sigma
         & =\sum_k\mu_k^NV^{(N)}(A_k)_\sigma
        =\sum_k\sum_{a=0}^{n_k-1}\mu_k^NV^{(N)}(B_{k,a})_\sigma
        =V^{(N)}\Bigl(\bigoplus_{k,a}\mu_kB_{k,a}\Bigr)_\sigma. \notag
    \end{align}
    The last equality reindexes the double sum by the pairs $(k,a)$.
\end{proof}

\begin{theorem}[Positive blocking of an irreducible-form presentation]
    \label{thm:periodic_irreducible_form_blocking}
    \lean{MPSTensor.IsIrreducibleForm.block,MPSTensor.IsIrreducibleForm.sum_dim_eq}
    \leanok
    \uses{def:periodic_tensor,def:irreducible_form}
    Suppose $A$ has the same matrix product vectors at every length,
    including zero, as $\bigoplus_k\mu_k A_k$, where $\mu_k>0$ and
    $A_k$ is left-canonical periodic of period $m_k$.
    For $p>0$, the blocked tensor $A^{[p]}$ has an irreducible-form
    presentation indexed by
    $0\le a<\gcd(m_k,p)$ for each original block $k$.
    The block indexed by $(k,a)$ has weight $\mu_k^p$ and period
    $m_k/\gcd(m_k,p)$. The total block dimension equals the bond dimension
    of $A$. This is the blockwise application of
    \cite[Lemma~6, proof of Theorem~4.1]{DeLasCuevas2017Irreducible}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:periodic_prescribed_blocking,thm:periodic_weighted_block_refinement}
    Apply the prescribed-blocking decomposition separately to each $A_k$,
    giving blocks $B_{k,a}$ with $0\le a<\gcd(m_k,p)$. Blocking raises
    $\mu_k$ to the power $p$: if $\sigma'$ is the unblocked configuration of
    length $pN$ underlying a blocked configuration $\sigma$ of length $N$, then
    \begin{align}
        V^{(N)}\bigl(A^{[p]}\bigr)_\sigma
         & =V^{(pN)}(A)_{\sigma'}
        =\sum_k\mu_k^{pN}V^{(pN)}(A_k)_{\sigma'}
        =\sum_k\bigl(\mu_k^p\bigr)^NV^{(N)}\bigl(A_k^{[p]}\bigr)_\sigma. \notag
    \end{align}
    Refinement preserves the weight $\mu_k^p$ on every orbit block. The preceding theorem assembles the
    resulting vector identities. All new weights remain positive.
    At length zero every matrix product vector is the trace of the identity,
    so, writing $D_{k,a}$ for the bond dimension of the block indexed by
    $(k,a)$,
    \begin{align}
        D & =V^{(0)}\bigl(A^{[p]}\bigr)
        =V^{(0)}\Bigl(\bigoplus_{k,a}\mu_k^pB_{k,a}\Bigr)
        =\sum_{k,a}D_{k,a}. \notag
    \end{align}
\end{proof}
