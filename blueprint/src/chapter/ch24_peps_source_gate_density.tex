\section{Density expansions of source gates}
\label{sec:peps_source_gate_density}

Fix the ordered pair slots of Definition~\ref{def:peps_source_slots},
with halfspaces $U_e,V_e$ and joint register space $\mathcal S$. All tensor
products use this order and the standard associativity identifications. In
these source-coordinate identities from
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
(\texttt{04-compression.tex}, lines 279--355),
the slot set and all finite index sets may be empty.

\begin{definition}[The joint source vector]
    \label{def:peps_joint_source_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotLayout_cons}
    \leanok
    \uses{def:peps_source_slots}
    The joint source vector is the multilinear map
    $\Phi:\prod_e(U_e\otimes V_e)\to\mathcal S$ defined recursively
    by $\Phi_{\varnothing}=1\in\C$ and
    $\Phi_{e::E}(\eta)=\eta_e\otimes\Phi_E(\eta|_E)$, with $\eta_e$
    separated into its two registers.
\end{definition}

\begin{theorem}[Preparation and finite sums of sources]
    \label{thm:peps_source_preparation_sums}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotVector_eq_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSlots_eq_appendIso_symm}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSlots_sum_smul}
    \leanok
    \uses{def:peps_joint_source_vector}
    For every original memory $\mathcal H$,
    $Q_{\eta,\mathcal H}x=\Phi(\eta)\otimes x$.
    In particular, for finite $J_e$, $d_e:J_e\to\C$, and
    $\theta_{e,j}\in U_e\otimes V_e$,
    \begin{align}
        Q_{(\sum_{j\in J_e}d_e(j)\theta_{e,j})_e,\mathcal H}
         & =\sum_{a\in\prod_e J_e}
        \left(\prod_e d_e(a_e)\right)
        Q_{(\theta_{e,a_e})_e,\mathcal H}.
        \label{eq:peps_source_preparation_sum}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_joint_source_vector,def:peps_source_inventory}
    Induction on the slot list of Definition~\ref{def:peps_source_inventory}
    identifies the prepared joint vector with $\Phi(\eta)$ of
    Definition~\ref{def:peps_joint_source_vector}, and changing to the fixed
    register layout gives $Q_{\eta,\mathcal H}x=\Phi(\eta)\otimes x$.
    Multilinearity of $\Phi$ with
    $\Phi((d_e\theta_e)_e)=(\prod_e d_e)\Phi((\theta_e)_e)$ then yields
    (\ref{eq:peps_source_preparation_sum}).
\end{proof}

\begin{definition}[Matrices of prepared branches]
    \label{def:peps_prepared_matrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_layout}
    Let $T:\mathcal S\otimes\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ be the
    operator of a source-only composition as in
    Definition~\ref{def:peps_party_layout}, without normalization or
    contraction bounds. In fixed finite orthonormal bases $(f_j)$ of
    $\mathcal H_{\rm in}$ and $(g_i)$ of $\mathcal H_{\rm out}$, define the
    prepared branch matrix by
    \begin{align}
        K_T(\eta)_{ij}
         & =\langle g_i,TQ_{\eta,\mathcal H_{\rm in}}f_j\rangle,
        \label{eq:peps_prepared_matrix}
    \end{align}
    with no finite-dimensionality assumption on the private spaces.
\end{definition}

\begin{theorem}[Coordinates of source preparation]
    \label{thm:peps_prepared_matrix_coordinates}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_sum_smul}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_eq_sum_basis}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSlots_eq_sum_basis}
    \leanok
    \uses{def:peps_prepared_matrix}
    For finite families as in Theorem~\ref{thm:peps_source_preparation_sums},
    \begin{align}
        K_T\bigl((\sum_j d_e(j)\theta_{e,j})_e\bigr)
         & =\sum_{a\in\prod_e J_e}
        \left(\prod_e d_e(a_e)\right)K_T((\theta_{e,a_e})_e).
        \label{eq:peps_prepared_matrix_sum}
    \end{align}
    In particular, for finite orthonormal bases $(u_{e,r})_r$ of $U_e$ and
    $(v_{e,s})_s$ of $V_e$, setting
    $\eta_e(r,s)=\langle u_{e,r}\otimes v_{e,s},\eta_e\rangle$ and
    $\theta_{e,(r_e,s_e)}=u_{e,r_e}\otimes v_{e,s_e}$ gives
    (\ref{eq:peps_prepared_matrix_sum}) with $d_e(a_e)=\eta_e(a_e)$ and
    left-hand side $K_T(\eta)$, and the same expansion holds for
    $Q_{\eta,\mathcal H}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation_sums}
    Composing (\ref{eq:peps_source_preparation_sum}) with $T$ and taking
    matrix entries via (\ref{eq:peps_prepared_matrix}) gives
    (\ref{eq:peps_prepared_matrix_sum}), and the basis specialization follows
    from $\eta_e=\sum_{r,s}\eta_e(r,s)u_{e,r}\otimes v_{e,s}$.
\end{proof}

\begin{definition}[Density coefficients of prepared branches]
    \label{def:peps_prepared_density_coefficient}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedDensityCoefficient}
    \leanok
    \uses{def:peps_prepared_matrix}
    For possibly nonorthogonal or nonspanning families $u_{e,r}\in U_e$,
    $v_{e,s}\in V_e$ on a ket branch $T$ and $\widetilde u_{e,r}\in U_e$,
    $\widetilde v_{e,s}\in V_e$ on a bra branch $T'$ (with possibly distinct
    index sets), write $K_T(a)=K_T((u_{e,a_e^{(1)}}\otimes v_{e,a_e^{(2)}})_e)$
    and $K_{T'}(b)=K_{T'}((\widetilde u_{e,b_e^{(1)}}\otimes\widetilde v_{e,b_e^{(2)}})_e)$,
    and for any complex input matrix $\rho$ define
    \begin{align}
        C_{T,T'}(\rho;a,b) & =K_T(a)\rho K_{T'}(b)^\dagger.
        \label{eq:peps_prepared_density_coefficient}
    \end{align}
\end{definition}

\begin{theorem}[Density expansion for finite component families]
    \label{thm:peps_prepared_density_components}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_density_sum_smul}
    \leanok
    \uses{def:peps_prepared_density_coefficient}
    Suppose the families in
    Definition~\ref{def:peps_prepared_density_coefficient} have finite
    index sets, including families in proper subspaces of $U_e,V_e$. For
    $\eta_e=\sum_{r,s}d_e(r,s)u_{e,r}\otimes v_{e,s}$,
    $\zeta_e=\sum_{r,s}\widetilde d_e(r,s)\widetilde u_{e,r}\otimes\widetilde v_{e,s}$,
    and any input matrix $\rho$,
    \begin{align}
        K_T(\eta)\rho K_{T'}(\zeta)^\dagger
         & =\sum_{a,b}\left(\prod_e
        d_e(a_e)\overline{\widetilde d_e(b_e)}\right)
        C_{T,T'}(\rho;a,b).
        \label{eq:peps_prepared_density_components}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_prepared_matrix_coordinates}
    Expanding $K_T(\eta)$ and $K_{T'}(\zeta)$ by
    (\ref{eq:peps_prepared_matrix_sum}) and using
    $(\prod_e d_e(a_e))\overline{\prod_e\widetilde d_e(b_e)}=\prod_e d_e(a_e)\overline{\widetilde d_e(b_e)}$
    gives (\ref{eq:peps_prepared_density_components}).
\end{proof}

\begin{theorem}[Coordinate density expansion of a weighted gate]
    \label{thm:peps_prepared_gate_density}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_density_eq_sum_basis}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_gate_density_eq_sum_basis}
    \leanok
    \uses{def:peps_prepared_density_coefficient}
    Let $\Xi$ be finite. For each $\xi\in\Xi$, choose a branch $T_\xi$, source
    vectors $\eta_{\xi,e}$, and finite orthonormal bases $(u_{\xi,e,r})_r$ of
    $U_e$ and $(v_{\xi,e,s})_s$ of $V_e$, which may depend on $\xi$. Set
    $K_\xi=K_{T_\xi}(\eta_\xi)$ and
    $\kappa_\xi(a)=\prod_e\langle u_{\xi,e,a_e^{(1)}}\otimes v_{\xi,e,a_e^{(2)}},\eta_{\xi,e}\rangle$.
    For any input matrix $\rho$ (not necessarily positive or normalized) and
    any $\xi,\zeta\in\Xi$,
    \begin{align}
        K_\xi\rho K_\zeta^\dagger
         & =\sum_{a,b}\kappa_\xi(a)\overline{\kappa_\zeta(b)}
        C_{T_\xi,T_\zeta}(\rho;a,b),
        \label{eq:peps_prepared_density_basis}
    \end{align}
    and for $G=\sum_\xi c_\xi K_\xi$,
    \begin{align}
        G\rho G^\dagger
         & =\sum_{\xi,\zeta}c_\xi\overline{c_\zeta}
        \sum_{a,b}\kappa_\xi(a)\overline{\kappa_\zeta(b)}
        C_{T_\xi,T_\zeta}(\rho;a,b).
        \label{eq:peps_prepared_gate_density}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_prepared_matrix_coordinates}
    Expanding each $\eta_{\xi,e}$ in its orthonormal basis via
    Theorem~\ref{thm:peps_prepared_matrix_coordinates} gives
    (\ref{eq:peps_prepared_density_basis}), and substituting
    (\ref{eq:peps_prepared_density_basis}) into
    $G\rho G^\dagger=\sum_{\xi,\zeta}c_\xi\overline{c_\zeta}K_\xi\rho K_\zeta^\dagger$
    gives (\ref{eq:peps_prepared_gate_density}).
\end{proof}

\begin{theorem}[Operator norm of a prepared branch]
    \label{thm:peps_prepared_matrix_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_preparedMatrix_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_preparedMatrix_tmul_le_one}
    \leanok
    \uses{def:peps_prepared_matrix}
    If $T$ is an allowed composition and $\|\eta_e\|=1$ for all $e$, then
    $\|K_T(\eta)\|\le1$ in the Euclidean operator norm. In particular, if
    $\|u_{e,r}\|=\|v_{e,s}\|=1$ (without assuming they span $U_e,V_e$, as in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 409--427), then $\|K_T(a)\|\le1$ for
    every $a$.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_source_inventory,def:peps_party_layout}
    Since $\|\eta_e\|=1$, $Q_{\eta,\mathcal H_{\rm in}}$ of
    Definition~\ref{def:peps_source_inventory} is allowed and contractive, so
    Definition~\ref{def:peps_party_layout} gives
    $\|TQ_{\eta,\mathcal H_{\rm in}}\|\le1$. Orthonormal coordinates are
    isometries, so (\ref{eq:peps_prepared_matrix}) has $\|K_T(\eta)\|\le1$,
    and $\|u_{e,r}\otimes v_{e,s}\|=\|u_{e,r}\|\|v_{e,s}\|=1$ gives
    $\|K_T(a)\|\le1$.
\end{proof}

\begin{theorem}[Density coefficients constructed from a gate]
    \label{thm:peps_actual_gate_density}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_finite_density_expansion}
    \leanok
    \uses{def:peps_prepared_density_coefficient,def:peps_party_layout}
    Let $P$ and $\Xi$ be finite. For every $\xi\in\Xi$, let $F_\xi$ be an
    allowed source-only composition on $P$ with fixed input and output
    memories, fix finite orthonormal bases of these memories and coefficients
    $c_\xi\in\C$, and let $M_\xi$ be the matrix of $F_\xi$ in these bases.
    Then there exist one ordered slot $e$ per unordered pair of distinct
    parties, common finite halfspaces $U_e=\C^{d_e}$ and $V_e=\C^{d'_e}$, unit
    vectors $\eta_{\xi,e}\in U_e\otimes V_e$, and allowed source-free
    compositions $T_\xi$ (all independent of $\rho$) such that
    $F_\xi=T_\xi Q_{\eta_\xi,\mathcal H_{\rm in}}$, the matrices $K_\xi(a)$
    defined from the standard bases of $U_e,V_e$ as in
    Definition~\ref{def:peps_prepared_density_coefficient} satisfy
    $\|K_\xi(a)\|\le1$, and with $\kappa_\xi(a)=\prod_e\eta_{\xi,e}(a_e)$ and
    $G=\sum_\xi c_\xi M_\xi$, every complex matrix $\rho$ satisfies
    \begin{align}
        G\rho G^\dagger
         & =\sum_{\xi,\zeta}c_\xi\overline{c_\zeta}
        \sum_{a,b}\kappa_\xi(a)\overline{\kappa_\zeta(b)}
        K_\xi(a)\rho K_\zeta(b)^\dagger,
        \label{eq:peps_actual_gate_density}
    \end{align}
    giving the exact source-coordinate density identity preceding the random
    replacement in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 233--355).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_finite_source_gate,thm:peps_prepared_gate_density,
        thm:peps_prepared_matrix_contraction}
    Theorem~\ref{thm:peps_finite_source_gate} provides $U_e,V_e$, the unit
    vectors $\eta_{\xi,e}$, and the allowed source-free compositions $T_\xi$
    with $F_\xi=T_\xi Q_{\eta_\xi,\mathcal H_{\rm in}}$. Standard basis vectors
    have unit norm, so Theorem~\ref{thm:peps_prepared_matrix_contraction} gives
    $\|K_\xi(a)\|\le1$. Since $M_\xi=K_{T_\xi}(\eta_\xi)$,
    Theorem~\ref{thm:peps_prepared_gate_density} gives
    (\ref{eq:peps_actual_gate_density}).
\end{proof}
