\subsection{Physical parent interactions of regular \texorpdfstring{$G$}{G}-isometric tensors}
\label{sec:peps_isometric_regional_parents}

Continue with a finite simple graph $\Gamma=(V,E)$ and a finite group $G$.
At $v$, let $\mathcal V_v=\C^{G^{\operatorname{Inc}(v)}}$ and let $E_v$
be the regular invariant projector. A regular $G$-isometric site map
$T_v:\mathcal V_v\to\C^d$ vanishes on $\ker E_v$ and is a scaled isometry
on $\operatorname{ran}E_v$: for some $c_v>0$,
$T_v^*T_v=c_vE_v$. This is the regular-representation condition of
\cite[Definition~6.1]{Schuch2010PEPS}, allowing a positive normalization.
Write $T_R=\bigotimes_{v\in R}T_v$ and
$\mathcal S_R$ for the open-boundary PEPS range on $R$.
The physical local interaction is $h_R=I-\Pi_{\mathcal S_R}$.

\begin{definition}[Coordinate orthogonal projectors]%
    \label{def:peps_coordinate_projector}
    \lean{TNLean.PEPS.coordinateSubspaceES,
        TNLean.PEPS.coordinateRangeProjector,
        TNLean.PEPS.coordinateRangeProjector_isStarProjection,
        TNLean.PEPS.coordinateRangeProjector_mulVec_eq_self_iff,
        TNLean.PEPS.range_coordinateRangeProjector}
    \leanok
    Equip $\C^I$ with $\langle x,y\rangle=\sum_i\overline{x_i}y_i$.
    For a subspace $S\subseteq\C^I$, its orthogonal projector $\Pi_S$ in the
    coordinate basis satisfies $\Pi_S^2=\Pi_S=\Pi_S^*$,
    $\operatorname{ran}\Pi_S=S$, and $\Pi_Sx=x\iff x\in S$.
\end{definition}

\begin{theorem}[Range characterization of projections]%
    \label{thm:peps_projection_range_unique}
    \lean{TNLean.PEPS.matrix_mul_eq_right_of_fixes_range,
        TNLean.PEPS.matrix_mul_eq_right_of_range_le,
        TNLean.PEPS.matrix_eq_of_isStarProjection_range_eq,
        TNLean.PEPS.eq_coordinateRangeProjector_of_range}
    \leanok
    \uses{def:peps_coordinate_projector}
    If $Px=x$ for all $x\in\operatorname{ran}Q$ (in particular, if $P^2=P$ and
    $\operatorname{ran}Q\subseteq\operatorname{ran}P$), then $PQ=Q$.
    Hence orthogonal projectors $P,Q$ with $\operatorname{ran}P=\operatorname{ran}Q$
    satisfy $P=Q$, and any orthogonal projector with range $S$ equals $\Pi_S$.
\end{theorem}
\begin{proof}\leanok
    Applying $Px=x$ to each column of $Q$ gives $PQ=Q$. If $P^2=P=P^*$,
    $Q^2=Q=Q^*$, and $\operatorname{ran}P=\operatorname{ran}Q$, then $PQ=Q$ and
    $QP=P$, so $P=QP=(PQ)^*=Q^*=Q$.
\end{proof}

\begin{definition}[Extension of regional operators]%
    \label{def:peps_regional_operator_extension}
    \lean{TNLean.PEPS.dependentRegionOperatorLift,
        TNLean.PEPS.dependentSubregionOperatorLift,
        TNLean.PEPS.dependentRegionOperatorLift_assemble,
        TNLean.PEPS.dependentRegionOperatorLift_apply}
    \leanok
    For finite coordinate sets $X_v$, put $X_R=\prod_{v\in R}X_v$,
    $\mathcal H_R=\C^{X_R}$, and write $(\alpha,\tau)\in X_R\times X_{V\setminus R}$.
    For $K\in\End(\mathcal H_R)$, define $\widehat K_R=K\otimes I_{V\setminus R}$ by
    $\widehat K_R((\alpha,\tau),(\beta,\rho))=K(\alpha,\beta)\delta_{\tau,\rho}$,
    and for $R\subseteq U$, its extension inside $U$ by $K\otimes I_{U\setminus R}$.
\end{definition}

In the following statement and proof, $X$ denotes the region $R$ of
Definition~\ref{def:peps_regional_operator_extension}, matching the patch
notation below.

    $[n]=\{k\in\N:k<n\}$, so $[0]=\varnothing$.
    Write $\mathcal H_X=\C^{\prod_{v\in X}Y_v}$.
    For $X\subseteq V$, let $E_X$ be the induced unitary coordinate
    relabelling, so $E_X\ket{(y_v)_{v\in X}}=
        \ket{(e_v(y_v))_{v\in X}}$.

        E_V\widehat K_XE_V^*

        \label{eq:peps_lift_basis_relabelling} \\
        E_X\operatorname{Tr}_{V\setminus X}
        \ketbra{\xi}{\xi}E_X^*
         & =\operatorname{Tr}_{V\setminus X}
        \ketbra{E_V\xi}{E_V\xi}.
        \label{eq:peps_marginal_basis_relabelling}
    \end{align}
    The extension on the right of~(\ref{eq:peps_lift_basis_relabelling})
    uses the relabelled bases; equivalently,
    $E_V(\widehat K_X\xi)=\widehat{E_XKE_X^*}_XE_V\xi$.

    empty regions, empty alphabets, and empty site sets.

\begin{theorem}[Algebra and slices of regional extensions]%
    \label{thm:peps_regional_extension_algebra}
    \lean{TNLean.PEPS.dependentRegionOperatorLift_one,
        TNLean.PEPS.dependentRegionOperatorLift_mul,
        TNLean.PEPS.dependentRegionOperatorLift_conjTranspose,
        TNLean.PEPS.dependentRegionOperatorLift_isStarProjection,
        TNLean.PEPS.dependentRegionOperatorLift_sub,
        TNLean.PEPS.dependentRegionSlice_mulVec_dependentRegionOperatorLift,
        TNLean.PEPS.dependentRegionOperatorLift_mulVec_eq_self_iff,
        TNLean.PEPS.dependentRegionOperatorLift_subregion,
        TNLean.PEPS.exists_dependentRegionOperatorLift_of_subset,
        TNLean.PEPS.exists_dependentRegionOperatorLift_mul_union,
        TNLean.PEPS.exists_dependentRegionOperatorLift_sub_union,
        TNLean.PEPS.exists_dependentRegionOperatorLift_commutator_union}
    \leanok
    \uses{def:peps_regional_operator_extension}
    The map $K\mapsto\widehat K_R$ is a unital $*$-algebra homomorphism
    preserving orthogonal projections. Writing $x_\tau(\alpha)=x(\alpha,\tau)$,
    one has $(\widehat K_Rx)_\tau=Kx_\tau$, so
    $\widehat K_Rx=x\iff \forall\tau,\ Kx_\tau=x_\tau$.
    For $R\subseteq U$, $\widehat{(K\otimes I_{U\setminus R})}_U=\widehat K_R$;
    hence products, differences, and commutators of $\widehat K_R$ and
    $\widehat L_S$ are extensions from $R\cup S$.
\end{theorem}
\begin{proof}\leanok
    In Definition~\ref{def:peps_regional_operator_extension}, summing
    $\widehat K_R((\alpha,\tau),(\beta,\rho))=K(\alpha,\beta)\delta_{\tau,\rho}$
    over intermediate coordinates leaves a single $\tau$-slice, and for
    $R\subseteq U$ we have $\delta_{\tau|_{U\setminus R},\rho|_{U\setminus R}}
        \delta_{\tau|_{V\setminus U},\rho|_{V\setminus U}}=\delta_{\tau,\rho}$.
    Factoring $\widehat K_R,\widehat L_S$ through $U=R\cup S$ gives the closure
    properties.
\end{proof}

\begin{theorem}[Continuity and norm bound for regional extensions]%
    \label{thm:peps_regional_extension_norm}
    \lean{TNLean.PEPS.norm_dependentRegionOperatorLift_mulVec_le,
        TNLean.PEPS.continuous_dependentRegionOperatorLift}
    \leanok
    \uses{def:peps_regional_operator_extension}
    For every region $R$, $K\mapsto\widehat K_R$ is continuous and
    $\|\widehat K_R\xi\|\le\|K\|\,\|\xi\|$ for all $K\in\End(\mathcal H_R)$ and
    $\xi\in\mathcal H_V$.
\end{theorem}
\begin{proof}\leanok
    \uses{def:peps_regional_operator_extension}
    With $\xi_\tau(\alpha)=\xi(\alpha,\tau)$, we have
    $(\widehat K_R\xi)_\tau=K\xi_\tau$, so
    \begin{align}
        \|\widehat K_R\xi\|^2
         & =\sum_{\tau\in X_{V\setminus R}}\|K\xi_\tau\|^2
        \le\|K\|^2\sum_{\tau\in X_{V\setminus R}}\|\xi_\tau\|^2
        =\|K\|^2\|\xi\|^2.
        \notag
    \end{align}
    Taking square roots gives $\|\widehat K_R\xi\|\le\|K\|\,\|\xi\|$, and
    linearity of $K\mapsto\widehat K_R$ in finite dimensions gives continuity.
\end{proof}

\begin{theorem}[Product maps and complementary cancellation]%
    \label{thm:peps_regional_product_cancellation}
    \lean{TNLean.PEPS.regionPhysicalProductMatrix_univ_eq_reindex_kronecker,
        TNLean.PEPS.dependentRegionOperatorLift_intertwine,
        TNLean.PEPS.dependentRegionOperatorLift_mul_regionPhysicalProductMatrix,
        TNLean.PEPS.dependentRegion_reindex_kronecker_eq_zero_iff,
        TNLean.PEPS.dependentRegionOperatorLift_mul_product_eq_zero_iff,
        TNLean.PEPS.dependentRegionOperatorLift_eq_zero_of_mul_product_eq_zero,
        TNLean.PEPS.regionPhysicalProductMatrix_ne_zero}
    \leanok
    \uses{def:peps_regional_operator_extension}
    For site matrices $F_v$, put $F_R=\bigotimes_{v\in R}F_v$. In
    region/complement coordinates,
    \begin{align}
        F_V             & =F_R\otimes F_{V\setminus R},    &
        \widehat K_RF_V & =(KF_R)\otimes F_{V\setminus R}.
        \label{eq:peps_product_split}
    \end{align}
    Thus $HF_R=F_RK\Longrightarrow\widehat H_RF_V=F_V\widehat K_R$.
    If $L\ne0$, then $M\otimes L=0\iff M=0$; hence if $F_{V\setminus R}\ne0$,
    $\widehat K_RF_V=0\iff KF_R=0$ (so $\widehat K_RF_V=0$ and $KF_R=K$ imply
    $K=0$). Moreover, $\forall v\in R,\ F_v\ne0\Longrightarrow F_R\ne0$.
\end{theorem}
\begin{proof}\leanok
    Splitting the entry product over $R$ and $V\setminus R$ gives
    (\ref{eq:peps_product_split}), and tensoring $HF_R=F_RK$ with
    $F_{V\setminus R}$ gives $\widehat H_RF_V=F_V\widehat K_R$.
    Cancelling a nonzero entry $L_{ij}$ from $M_{ab}L_{ij}=0$ gives $M=0$, and
    multiplying one nonzero entry from each $F_v$ gives $F_R\ne0$.
\end{proof}

\begin{theorem}[Single-coordinate generation and fixed spaces]%
    \label{thm:peps_regular_constraint_fixed_spaces}

    \leanok

    Any submonoid of a finite product $\prod_{i\in I}M_i$ containing all
    single-coordinate elements equals $\prod_{i\in I}M_i$.
    For the half-edge actions of Definition~\ref{def:peps_ambient_regional_constraints},
    $g\mapsto T_{r^{e,g^{-1}}}$ is a permutation representation, and
    \begin{align}
        A_vx=x     & \iff \forall g\in G,\ L_{k^{v,g}}x=x,                   &
        B_{R,e}x=x & \iff \forall g\in G,\ T_{r^{e,g}}x=x, \notag                         \\
        D_Rx=x     & \iff \supp(x)\subseteq\{\alpha:\alpha|_R\text{ flat}\}. &   & \notag
    \end{align}
    Consequently, $(\forall v\in R,\ A_vx=x)\iff(\forall k\text{ supported on }R,\ L_kx=x)$,
    and $(\forall e\in E(R),\ B_{R,e}x=x)\iff(\forall r\text{ supported on }E(R),\ T_rx=x)$.
\end{theorem}
\begin{proof}\leanok
    Induction on $|I|$ proves single-coordinate generation.
    For any finite-group representation $\rho$, $|G|^{-1}\sum_{g\in G}\rho(g)x=x
        \iff\forall g,\ \rho(g)x=x$; applying this to $g\mapsto L_{k^{v,g}}$ and
    $g\mapsto T_{r^{e,g^{-1}}}$ (with $(gh)^{-1}=h^{-1}g^{-1}$) and invoking
    single-coordinate generation gives the $A_v$ and $B_{R,e}$ characterizations,
    while $D_Rx=x\iff\mathbf1_{\mathrm{flat}}(\alpha|_R)x(\alpha)=x(\alpha)$.
\end{proof}

\begin{theorem}[The commuting product is the regional range projector]%

        Q_Rx=x          & \ \Longleftrightarrow\ x\in\widehat{\mathcal S}_R^0,
                        & \operatorname{ran}Q_R                                & =\widehat{\mathcal S}_R^0,\notag \\

    and Theorem~\ref{thm:peps_ambient_regional_constraint_projection} gives
    $\operatorname{ran}Q_R=\widehat{\mathcal S}_R^0=\operatorname{ran}\widehat{P_R^0}$.
    Theorem~\ref{thm:peps_projection_range_unique} then yields $\widehat{P_R^0}=Q_R$,

\begin{theorem}[Projectors conjugated by scaled partial isometries]%
    \label{thm:peps_scaled_partial_isometry_projector_transport}
    \lean{TNLean.PEPS.isStarProjection_smul_mul_mul_conjTranspose_of_gram,
        TNLean.PEPS.smul_mul_mul_conjTranspose_mul_of_gram,
        TNLean.PEPS.range_smul_mul_mul_conjTranspose_of_gram}
    \leanok
    Let $A$ be rectangular, $c\in\R\setminus\{0\}$, and $A^*A=cE$.
    If $Q^2=Q=Q^*$ and $EQ=Q$, then $c^{-1}AQA^*$ is an orthogonal projector.
    If $QE=Q$, then $(c^{-1}AQA^*)A=AQ$, and if also $Q^2=Q=Q^*$, then
    $\operatorname{ran}(c^{-1}AQA^*)=A(\operatorname{ran}Q)$.
\end{theorem}
\begin{proof}\leanok
    For $B=AQA^*$, $EQ=Q$ gives $B^2=cAQEQA^*=cB$ and $B^*=B$, while $QE=Q$
    gives $c^{-1}BA=AQE=AQ$. Then $\operatorname{ran}(c^{-1}B)\subseteq A(\operatorname{ran}Q)$
    and $(c^{-1}B)AQx=AQ^2x=AQx$ gives the reverse inclusion.
\end{proof}

\begin{theorem}[Physical regional range and its projector]%
    \label{thm:peps_physical_regional_projector}
    \lean{TNLean.PEPS.regularProjectorOpenRegionRange_map_physical,

        TNLean.PEPS.regularIsometric_regionRangeProjector_intertwine}

    $E_RP_R^0=P_R^0$, where $E_R=\bigotimes_{v\in R}E_v$.
    For regular $G$-isometric site maps with $c_R=\prod_{v\in R}c_v>0$, the
    orthogonal projector $P_R=\Pi_{\mathcal S_R}$ satisfies
    \begin{align}
        T_R^*T_R & =c_RE_R,                &
        P_R      & =c_R^{-1}T_RP_R^0T_R^*, &
        P_RT_R   & =T_RP_R^0.
        \label{eq:peps_physical_projector_transport}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_scaled_partial_isometry_projector_transport,thm:peps_projection_range_unique,
        thm:peps_regional_range_invariant_flat}
    Factoring the open-region map through $T_R$ gives $\mathcal S_R=T_R(\mathcal S_R^0)$,
    and local gauge invariance (Theorem~\ref{thm:peps_regional_range_invariant_flat})
    gives $E_RP_R^0=P_R^0=(P_R^0E_R)^*$. Tensoring $T_v^*T_v=c_vE_v$ gives
    $T_R^*T_R=c_RE_R$, and Theorems~\ref{thm:peps_scaled_partial_isometry_projector_transport}
    and~\ref{thm:peps_projection_range_unique} yield (\ref{eq:peps_physical_projector_transport}).
\end{proof}

\begin{theorem}[Nonzero physical site images]%

    For regular $G$-isometric site maps, $L_v=c_v^{-1}T_v^*$ and $J_v=T_vL_v$
    satisfy $J_v^2=J_v=J_v^*\ne0$, $L_vT_v=E_v$, and $J_vT_v=T_v$.
    More generally, for regular $G$-injective site maps and any matrices $J_v$
    with $J_vT_v=T_v$, the product $J_R=\bigotimes_{v\in R}J_v$ satisfies
    $J_RP_R=P_R$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_physical_regional_projector,thm:peps_projection_range_unique}
    Since $E_v1=1\ne0$, we have $E_v\ne0$. From $T_v^*T_v=c_vE_v$ and
    $T_vE_v=T_v$ we get $L_vT_v=E_v$, $J_vT_v=T_v$, $J_v^2=J_v=J_v^*$, and
    $J_v\ne0$ (else $E_v=L_vJ_vT_v=0$). Finally, $J_RT_R=T_R$ fixes
    $\mathcal S_R=T_R(\mathcal S_R^0)$, so Theorem~\ref{thm:peps_projection_range_unique}
    gives $J_RP_R=P_R$.
\end{proof}

\begin{definition}[Products of site-image projections]%
    \label{def:peps_site_image_products}
    \lean{TNLean.PEPS.productSiteMask,TNLean.PEPS.globalProductSiteMask,
        TNLean.PEPS.globalProductSiteMask_eq_lift,
        TNLean.PEPS.globalProductSiteMask_mul,
        TNLean.PEPS.productSiteMask_isStarProjection,
        TNLean.PEPS.regionPhysicalProductMatrix_isStarProjection,
        TNLean.PEPS.regionPhysicalProductMatrix_productSiteMask_mul,
        TNLean.PEPS.regionPhysicalProductMatrix_productSiteMask_isStarProjection}
    \leanok
    For orthogonal site projectors $J_v$, define $J_v^{[U]}=J_v$ for $v\in U$
    and $J_v^{[U]}=I$ for $v\notin U$, and set
    $M_U=\bigotimes_{v\in V}J_v^{[U]}=\widehat{J_U}$. Then $M_U^2=M_U=M_U^*$,
    $M_RM_S=M_{R\cup S}$, and $(\bigotimes_{v\in R}J_v^{[U]})J_R=J_R$.
\end{definition}

\begin{theorem}[Support under products of site projections]%
    \label{thm:peps_site_image_support}
    \lean{TNLean.PEPS.dependentRegionOperatorLift_commute_globalProductSiteMask,
        TNLean.PEPS.regionPhysicalProductMatrix_mul_supported_right,
        TNLean.PEPS.dependentRegionOperatorLift_mul_globalProductSiteMask_self,
        TNLean.PEPS.dependentRegionOperatorLift_mul_absorbs_union}
    \leanok
    \uses{def:peps_site_image_products,def:peps_regional_operator_extension}
    If $H_R^2=H_R=H_R^*$ and $J_RH_R=H_R$, then $H_RJ_R=H_R$,
    $\widehat H_RM_R=\widehat H_R$, and $[\widehat H_R,M_U]=0$ for every $U\subseteq V$.
    If also $H_S^2=H_S=H_S^*$ with $J_SH_S=H_S$, then
    $(\widehat H_R\widehat H_S)M_{R\cup S}=\widehat H_R\widehat H_S$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regional_product_cancellation,thm:peps_regional_extension_algebra}
    Adjoints give $H_RJ_R=(J_RH_R)^*=H_R$, so $\bigotimes_{v\in R}J_v^{[U]}$
    absorbs into $J_R$ on both sides of $H_R$, which extends by
    Theorem~\ref{thm:peps_regional_product_cancellation} to $[\widehat H_R,M_U]=0$.
    Then $(\widehat H_R\widehat H_S)M_RM_S=\widehat H_RM_R\widehat H_SM_S=\widehat H_R\widehat H_S$.
\end{proof}

\begin{theorem}[Commutation through rectangular product maps]%
    \label{thm:peps_product_transport_commutation}
    \lean{TNLean.PEPS.commute_dependentRegionOperatorLift_of_productSiteTransport}
    \leanok
    \uses{def:peps_site_image_products,def:peps_regional_operator_extension}
    Let $A_vL_v=J_v$ with $J_v^2=J_v=J_v^*\ne0$. If orthogonal projectors
    $H_R,H_S$ satisfy $J_RH_R=H_R$, $J_SH_S=H_S$, $H_RA_R=A_RK_R$,
    $H_SA_S=A_SK_S$, and $[\widehat K_R,\widehat K_S]=0$, then
    $[\widehat H_R,\widehat H_S]=0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regional_product_cancellation,thm:peps_site_image_support,
        thm:peps_regional_extension_algebra}
    Put $U=R\cup S$ and $D=[\widehat H_R,\widehat H_S]=\widehat F_U$
    (Theorem~\ref{thm:peps_regional_extension_algebra}). Intertwining gives
    $DA_V=A_V[\widehat K_R,\widehat K_S]=0$, so $DJ_V=DA_VL_V=0$, while
    Theorem~\ref{thm:peps_site_image_support} gives $DM_U=D$. By
    (\ref{eq:peps_product_split}),
    \begin{align}
        0=DJ_V & =(FJ_U)\otimes J_{V\setminus U}, &
        DM_U   & =\widehat{FJ_U}_U=D.
        \label{eq:peps_spectator_cancellation}
    \end{align}
    Since $J_{V\setminus U}\ne0$, Theorem~\ref{thm:peps_regional_product_cancellation}
    gives $FJ_U=0$, hence $D=0$ by (\ref{eq:peps_spectator_cancellation}).
\end{proof}

\begin{theorem}[Canonical interactions are complementary projections]%
    \label{thm:peps_canonical_complement_projection}
    \lean{TNLean.PEPS.canonicalRegionParentInteraction_isStarProjection,
        TNLean.PEPS.canonicalRegionParentInteraction_eq_one_sub_rangeProjector,
        TNLean.PEPS.regionLocalTerm_eq_dependentRegionOperatorLift,
        TNLean.PEPS.dependentRegionOperatorLift_one_sub}
    \leanok
    \uses{def:peps_coordinate_projector,def:peps_regional_operator_extension}
    For any graph PEPS tensor, the canonical regional interaction
    $h_R=\Pi_{\mathcal S_R^\perp}$ is an orthogonal projector satisfying
    $h_R=I-\Pi_{\mathcal S_R}$ and $\widehat h_R=I-\widehat{\Pi_{\mathcal S_R}}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_projection_range_unique,thm:peps_regional_extension_algebra}
    Since $h_R=\Pi_{\mathcal S_R^\perp}$, $I-h_R$ is an orthogonal projector
    with range $\ker h_R=\mathcal S_R$, so
    Theorem~\ref{thm:peps_projection_range_unique} gives $I-h_R=\Pi_{\mathcal S_R}$,
    and Theorem~\ref{thm:peps_regional_extension_algebra} gives
    $\widehat h_R=I-\widehat{\Pi_{\mathcal S_R}}$.
\end{proof}

\begin{theorem}[Commuting physical parent interactions]%
    \label{thm:peps_isometric_regional_parents_commute}
    \lean{TNLean.PEPS.regularIsometric_regionRangeProjector_lifts_commute,
        TNLean.PEPS.regularIsometric_canonicalRegionParentInteractions_commute}
    \leanok
    \uses{thm:peps_canonical_complement_projection}
    In the finite-simple-graph form of the commuting-parent conclusion
    of \cite[Theorem~6.12]{Schuch2010PEPS}, locally regular $G$-isometric tensors have
    commuting canonical regional range projectors and parent interactions
    for arbitrary (possibly overlapping) regions $R,S\subseteq V$:
    \begin{align}
        [\widehat P_R,\widehat P_S] & =0, &
        [\widehat h_R,\widehat h_S] & =0.
        \label{eq:peps_physical_parent_commutation}
    \end{align}
    Periodic presentations with self-loops or parallel bonds require their
    own contraction conventions.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_nonzero_site_images,thm:peps_physical_regional_projector,
        thm:peps_regular_actual_projector,thm:peps_product_transport_commutation,
        thm:peps_canonical_complement_projection}
    Combining $T_vL_v=J_v\ne0$ and $J_RP_R=P_R$
    (Theorem~\ref{thm:peps_nonzero_site_images}), $P_RT_R=T_RP_R^0$ from
    (\ref{eq:peps_physical_projector_transport}), and
    $[\widehat{P_R^0},\widehat{P_S^0}]=0$ from
    (\ref{eq:peps_actual_virtual_projection}),
    Theorem~\ref{thm:peps_product_transport_commutation} gives
    $[\widehat P_R,\widehat P_S]=0$, and Theorem~\ref{thm:peps_canonical_complement_projection}
    yields $[\widehat h_R,\widehat h_S]=[I-\widehat P_R,I-\widehat P_S]=0$.
\end{proof}
