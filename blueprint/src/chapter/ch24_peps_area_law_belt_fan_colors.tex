\section{Opposite labels on belt-cell sides}
\label{sec:area_law_belt_fan_colors}

For each belt-cell fan, the opposing region on each elementary perimeter segment
determines a binary label, as in \cite[Section~11]{OpenAI2026AreaLaw}. This
assignment holds for arbitrary pitch residues at each layer and uses no sparsity
estimate.

\begin{definition}[Labels of a belt-cell fan]
    \label{def:area_law_belt_fan_color}
    \lean{TNLean.PEPS.AreaLaw.Geometry.beltCellFanColor}
    \leanok
    \uses{def:area_law_selected_elementary_opponent,def:area_law_belt_cells,
        def:area_law_actual_side_mask}
    Fix an origin $o\in\mathbb R^2$, a finite endpoint set $Z$,
    an integer width $C_0\ge2$, and a lower layer index $k_0\ge50\,000\,000$.
    For every layer $h$, choose residues $0\le a_h,b_h<m_h$, where
    $m_h=2^{p_h-\ell_h}$, $\ell_h$ is the fine exponent, and $p_h$ is the
    pitch exponent. Let $\mathcal B_h$ be the resulting belt-cell
    index set. For $k\ge k_0$ and $z\in\mathcal B_k$, denote the elementary
    perimeter segments of the fan by $e_i=[a_i,b_i]$ and their opposing region
    identifiers by $P_{k,z}(i)$. Order the triples $(k,z_1,z_2)$
    lexicographically, and define $c_{k,z}(i)\in\mathbb Z/2\mathbb Z$ by
    \begin{align}
        c_{k,z}(i)=
        \begin{cases}
            (k_0-1)+1, & P_{k,z}(i)=\star,                                              \\
            h+1,       & P_{k,z}(i)=(h,w),\ w\notin\mathcal B_h,                        \\
            0,         & P_{k,z}(i)=(h,w),\ w\in\mathcal B_h,\ (k,z_1,z_2)<(h,w_1,w_2), \\
            1,         & P_{k,z}(i)=(h,w),\ w\in\mathcal B_h,\ (k,z_1,z_2)>(h,w_1,w_2).
        \end{cases}
    \end{align}
    When $P_{k,z}(i)=(h,w)$ is a fine cell, $(k,z)\ne(h,w)$, so the last two
    cases are exhaustive.
\end{definition}

\begin{theorem}[Label opposite the dummy tile]
    \label{thm:area_law_belt_fan_dummy_color}
    \lean{TNLean.PEPS.AreaLaw.Geometry.beltCellFanColor_dummy}
    \leanok
    \uses{def:area_law_belt_fan_color,def:area_law_dyadic_layers}
    With the notation of the preceding definition, if an elementary segment
    $e_i$ lies in $\overline{N_{k_0}}$, then
    $c_{k,z}(i)=(k_0-1)+1\ne k_0-1$ in $\mathbb Z/2\mathbb Z$, so the incident
    triangle receives the label opposite that of the dummy tile.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_unique_elementary_side_opponent,def:area_law_belt_fan_color}
    Since $e_i\subseteq\overline{N_{k_0}}$, uniqueness of the opposing region
    gives $P_{k,z}(i)=\star$, and hence $c_{k,z}(i)=(k_0-1)+1\ne k_0-1$ in
    $\mathbb Z/2\mathbb Z$.
\end{proof}

\begin{theorem}[Label and primary region opposite a non-belt cell]
    \label{thm:area_law_belt_fan_nonbelt_color}
    \lean{TNLean.PEPS.AreaLaw.Geometry.beltCellFanColor_nonbelt_opponent}
    \leanok
    \uses{def:area_law_belt_fan_color,def:area_law_nonbelt_pitch_index,
        def:area_law_primary_pitch_indices,def:area_law_primary_regions}
    With the notation of the label definition, let $h\ge k_0$ and let
    $w\in F_{h,\ell_h}\setminus\mathcal B_h$ be a non-belt fine-cell index.
    If $e_i\cap\overline{Q_{\ell_h}(w)}$ contains two distinct points,
    let $J=j(w)$ be the shifted pitch index at residues $a_h,b_h$.
    Then $J$ belongs to the primary index set $\mathcal P_h$,
    the segment $e_i$ is contained in the primary region
    $\overline{D_h\cap I_{a_h,b_h}(J)}$, and
    $c_{k,z}(i)=h+1\ne h$ in $\mathbb Z/2\mathbb Z$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_selected_opponent_contact,
        thm:area_law_unique_elementary_side_opponent,
        thm:area_law_nonbelt_primary_containment,
        thm:area_law_primary_pitch_membership,def:area_law_belt_fan_color}
    Since $z\in\mathcal B_k$ and $w\notin\mathcal B_h$, we have $(k,z)\ne(h,w)$.
    Two-point contact gives $P_{k,z}(i)=(h,w)$ and
    $e_i\subseteq\overline{Q_{\ell_h}(w)}\subseteq
        \overline{D_h\cap I_{a_h,b_h}(J)}$.
    Because $e_i$ is nonempty, $D_h\cap I_{a_h,b_h}(J)$ is nonempty, so
    $J\in\mathcal P_h$. The second clause of the label definition gives
    $c_{k,z}(i)=h+1\ne h$ in $\mathbb Z/2\mathbb Z$.
\end{proof}

\begin{theorem}[Opposite labels across contacting belt-cell segments]
    \label{thm:area_law_belt_fan_contact_colors}
    \lean{TNLean.PEPS.AreaLaw.Geometry.beltCellFanColor_belt_contact}
    \leanok
    \uses{def:area_law_belt_fan_color}
    Let $(k,z)\ne(h,w)$ be belt-cell pairs with $k,h\ge k_0$, and let $e_i$ and
    $e_j$ be elementary segments of their respective fans.
    If $e_i\cap e_j$ contains two distinct points, then
    $c_{k,z}(i)\ne c_{h,w}(j)$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_selected_opponent_contact,thm:area_law_cell_fan_cover,
        def:area_law_belt_fan_color}
    Each elementary segment lies in its incident fan triangle and therefore in
    its own closed fine cell. Consequently $e_i\cap e_j$ gives two-point contact
    of $e_i$ with $\overline{Q_{\ell_h}(w)}$ and of $e_j$ with
    $\overline{Q_{\ell_k}(z)}$, so $P_{k,z}(i)=(h,w)$ and $P_{h,w}(j)=(k,z)$.
    Since $(k,z_1,z_2)\ne(h,w_1,w_2)$, one triple strictly precedes the other
    lexicographically, and the two segments receive opposite labels in
    $\{0,1\}$.
\end{proof}
