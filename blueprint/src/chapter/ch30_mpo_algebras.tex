\chapter{Matrix product operator algebras}%
\label{ch:mpo_algebras}

A matrix product operator algebra is a finite family of matrix product
operators satisfying the fusion rule~(\ref{eq:mpo_fusion_algebra}) with
structure constants $N_{ab}^c\in\N$
\cite{Bultinck2017Anyons,GarreRubio2023MPOSym,Molnar2022MPOAlgebrasI}.
Chapter~\ref{ch:mpo_symmetry_basics} introduces these algebras and their
non-invertible examples, defines symmetric matrix product states, and proves
the integrality of their action coefficients.
Transfer of the symmetry to the virtual level rests on the multi-block
asymmetric compression theorem (Theorem~\ref{thm:asym_multiblock}): the stacked
tensor of a product of two matrix product operators, or of an operator acting
on a state, need not be in canonical form and in general admits no sitewise
intertwiner with the tensors of the fusion products
(Remark~\ref{rem:asym_zipper}), so the Fundamental Theorem for tensors in
canonical form does not apply to it in general; the asymmetric theorem, which
requires equality of periodic traces only, does.

Throughout, $T_a$ denotes a matrix product operator tensor of bond dimension
$\chi_a$, $O_a^{(N)}$ its periodic operator on $N$ sites, and
$\ket{V^{(N)}(A)}$ the periodic vector of a matrix product state tensor $A$.
For two operator tensors, $T_aT_b$ denotes the stacked tensor
$(T_aT_b)^{ij}=\sum_mT_a^{im}\otimes T_b^{mj}$ of bond dimension
$\chi_a\chi_b$, and $T_a\cdot A$ denotes the tensor
$(T_a\cdot A)^i=\sum_jT_a^{ij}\otimes A^j$. A reduction of a tensor $B$ onto a
tensor $C$ is a pair of maps $(V,W)$ with $VW=\Id$ and $VB^wW=C^w$ for every
word $w$ (Definition~\ref{def:mps_rectangular_reduction}), with compression $V$
and embedding $W$.

\section{Fusion tensors and action tensors}%
\label{sec:mpoalg_virtual}

Equality of periodic traces in the fusion rule (\ref{eq:mpo_fusion_algebra})
decomposes the stacked tensor $T_aT_b$ into copies of the blocks $T_c$.

\begin{definition}[Fusion tensors]%
    \label{def:mpoalg_fusion_tensors}
    \lean{MPOTensor.FusionTensors,
        MPSTensor.NilpotentBiorthogonalReductions}
    \leanok
    \uses{def:mps_rectangular_reduction,def:mpo_to_mps}
    Let $(T_a)_{a\in I}$ be matrix product operator tensors and let
    $N_{ab}^c$ be nonnegative integers, such as the structure constants of a
    fusion algebra (Definition~\ref{def:mpo_fusion_algebra}). A family of
    \emph{fusion tensors} for the pair $(a,b)$ consists of maps
    \begin{align}
        V_{ab}^{c,\mu} & \colon\C^{\chi_a}\otimes\C^{\chi_b}\to\C^{\chi_c}, &
        W_{ab}^{c,\mu} & \colon\C^{\chi_c}\to\C^{\chi_a}\otimes\C^{\chi_b},
        \label{eq:mpoalg_fusion_maps}
    \end{align}
    for $c\in I$ and $1\leq\mu\leq N_{ab}^c$, such that the pairs are mutually
    biorthogonal, $V_{ab}^{c,\mu}W_{ab}^{d,\nu}=\delta_{cd}\delta_{\mu\nu}
        \Id_{\chi_c}$, each pair is a reduction of $T_aT_b$ onto $T_c$,
    \begin{align}
        V_{ab}^{c,\mu}(T_aT_b)^{\mathbf w}W_{ab}^{c,\mu}
         & =T_c^{\mathbf w}
        \label{eq:mpoalg_fusion_reduction}
    \end{align}
    for every word $\mathbf w$ in the pair alphabet, and the remainder
    $(T_aT_b)^{ij}-\sum_{c,\mu}W_{ab}^{c,\mu}T_c^{ij}V_{ab}^{c,\mu}$ is
    nilpotent: for some $\ell$, every product of at least $\ell$ of its
    matrices vanishes. This is the decomposition of
    \cite[Section~2]{GarreRubio2023MPOSym} and
    \cite{Bultinck2017Anyons}, with the nilpotent remainder of
    \cite[Section~4.4]{GarreRubio2023MPOSym} in place of an exact identity.
    With zero remainder, a family of fusion tensors is an exact biorthogonal
    decomposition of $T_aT_b$ onto the copies of the $T_c$
    (Definition~\ref{def:glm_boundary_decomposition}), the form of
    \cite[Section~2]{GarreRubio2023MPOSym}.
    % Source anchor: arXiv:2203.12563, lines 361--391 (fusion tensors and
    %   orthogonality) and lines 1026--1030 (nilpotent off-diagonal blocks);
    %   arXiv:1511.08090, AnyonsPEPS.tex lines 237--251.
    % Source, lines 1001--1028, verbatim: "fusion tensors that decompose the
    %   product of two tensors locally for any length $n$" (equation
    %   fusiontensorG2); "In particular, for $n=0$, this results in the
    %   orthogonality relations for the fusion tensors"; "These off-diagonal
    %   blocks will disappear after a finite number of concatenation: we
    %   denote by $\ell$ the nilpotent length associated to that".
\end{definition}

    \label{thm:mpoalg_fusion_tensors_exist}
    \lean{MPOTensor.IsMPOFusionAlgebra.nonempty_fusionTensors}
    \leanok
    % Existing Lean special cases:
    %   MPOTensor.exists_multiBlockCompression_mulTensor_of_mpo_eq_sum

    does not assert the exact local reconstruction of

    are proved in Lemma~\ref{thm:glm_boundary_source_decomposition}.
    % Source anchor: arXiv:2203.12563, line 361, Appendix A (lines
    %   2305--2455), and lines 1001--1028, where the group case is stated:
    %   "$U_g U_h = U_{gh}$ with $T_g$ injective implies the existence of

\begin{proof}
    \leanok
    \uses{cor:asym_fusion_action_multiplicity}
    Corollary~\ref{cor:asym_fusion_action_multiplicity} gives, for the
    target family consisting of $N_{ab}^c$ copies of $T_c$ for each $c$,
    compression pairs indexed by $(c,\mu)$ that form mutually biorthogonal
    reductions of $T_aT_b$ with a nilpotent remainder.
\end{proof}

For a group-like algebra, with labels in a group $G$ and
$N_{gh}^k=\delta_{k,gh}$, the fusion tensors reduce to one pair
$(V_{g,h},W_{g,h})$ for each pair of group elements; for representations by
normal tensors this is Corollary~\ref{cor:asym_fusion_tensors_group}.

\section{Gauge freedom of fusion and action tensors}%
\label{sec:mpoalg_gauge}

Two reductions of a tensor onto the same normal block, with nilpotent
residuals, agree up to one nonzero scalar after dressing with long words
(Lemma~\ref{thm:mps_reduction_boundary_dressed_uniqueness}); the scalar is
unique by Lemma~\ref{thm:mpug_dressed_proportionality}, and
Corollary~\ref{cor:asym_fusion_gauge_single} states the case of fusion
tensors. This applies to the fusion tensors of a group-like algebra, where
$T_gT_h$ reduces onto the single block $T_{gh}$, to the composite reductions
of Lemma~\ref{thm:mpoalg_composite_reduction}, and to the action tensors of an
invertible label on a permuted block. It does not apply to a single slot of a
reduction onto multiple blocks, whose residual is not nilpotent.
% Source anchor: arXiv:1706.07329, cornerproblem.tex lines 3156--3162
%   (Theorem 22); arXiv:2203.12563, lines 1028--1060.

Fusion tensors with vanishing remainder split the stacked product over the
blocks, as in (\ref{eq:asym_zipper_decomposition}) with $B=T_aT_b$ and
$C_c=T_c$. For normal blocks of positive bond dimension that are pairwise not
gauge related, two such families differ by a unique element of
$\bigoplus_c\GL_{N_{ab}^c}(\C)$, with no word dressing
(Corollary~\ref{cor:asym_zipper_fusion_uniqueness}); this is the uniqueness
of zipper fusion tensors up to basis change in each multiplicity space
\cite{Bultinck2017Anyons}, \cite[Section~4]{Molnar2022MPOAlgebrasI}. The same
holds for action tensors with the gauge group $\bigoplus_y\GL_{M_{a,x}^y}(\C)$
of \cite[Section~3.1]{GarreRubio2023MPOSym}
(Corollary~\ref{cor:asym_zipper_action_uniqueness}).
% Source anchor: arXiv:1511.08090, AnyonsPEPS.tex lines 181--200;
%   arXiv:2204.05940, mpo.tex line 1725; arXiv:2203.12563, lines 416--424
%   and 595--602.

\begin{remark}[Uniqueness fails with a nilpotent remainder]%
    \label{rem:mpoalg_nonunique}
    With only a nilpotent remainder, reductions are not unique up to scalars.
    Let $C$ be normal of bond dimension $D$, let $R^i\in\C^{D\times1}$ be
    arbitrary, and let
    \begin{align}
        B^i & =\begin{pmatrix}C^i & R^i\\ 0 & 0\end{pmatrix},
            &
        W   & =\begin{pmatrix}\Id_D\\ 0\end{pmatrix},
            &
        V_y & =\begin{pmatrix}\Id_D & y\end{pmatrix}
        \label{eq:mpoalg_nonunique}
    \end{align}
    for $y\in\C^D$. Then $V_yW=\Id_D$, $V_yB^wW=C^w$ for every word $w$, and
    the remainder $B^i-WC^iV_y$ has vanishing products of length two, so
    every $(V_y,W)$ is a reduction of $B$ onto $C$ with nilpotent remainder.
    For $y\neq0$ the compressions $V_y$ and $V_0$ are not proportional,
    although $V_yB^w=V_0B^w$ for every nonempty word; with a single target,
    Lemma~\ref{thm:mps_reduction_boundary_dressed_uniqueness} recovers
    uniqueness after dressing.

    With several targets even the dressed identities fail. Let
    $(V_\mu,W_\mu)$ and $(V_\nu,W_\nu)$ be two slots of a compression of $B$
    onto blocks $C_\mu$ and $C_\nu$ of the same bond dimension $D$, with
    $\mu$ before $\nu$ in the block upper triangular order, and conjugate the
    slot coordinates by $S=\begin{pmatrix}\Id & Z\\ 0 & \Id\end{pmatrix}$
    for an arbitrary $Z\in\MN{D}$. The new family is still block upper
    triangular, hence again a compression with nilpotent remainder, and
    $\widetilde V_\mu=V_\mu-ZV_\nu$. Then
    $\widetilde V_\mu B^{\mathbf w}=V_\mu B^{\mathbf w}-ZV_\nu B^{\mathbf w}$
    on every long word, which is not a combination of the
    $V_\kappa B^{\mathbf w}$ with scalar coefficients unless $Z$ is scalar.
    This holds whether $\nu$ is a duplicate copy of the same block or a
    distinct block. The zipper condition of \cite{Bultinck2017Anyons}, with
    vanishing remainder, rules out both ambiguities and restores uniqueness up
    to the multiplicity gauge
    (Corollary~\ref{cor:asym_zipper_fusion_uniqueness}).
    % Source anchor: arXiv:1511.08090, AnyonsPEPS.tex lines 181--200.
\end{remark}

\section{The associator of a group-like algebra}%
\label{sec:mpoalg_associator}

Let $G$ be a group and $(T_g)_{g\in G}$ a fusion algebra of normal tensors of
positive bond dimension with $N_{gh}^k=\delta_{k,gh}$, so that
$O_g^{(N)}O_h^{(N)}=O_{gh}^{(N)}$ for every $N\geq1$. Choose fusion tensors
$(V_{g,h},W_{g,h})$ for every pair. The triple product $T_gT_hT_k$ has two
composite reductions onto $T_{ghk}$,
\begin{align}
    V^{(gh)k}
     & =V_{gh,k}\,(V_{g,h}\otimes\Id_{\chi_k}),
     &
    V^{g(hk)}
     & =V_{g,hk}\,(\Id_{\chi_g}\otimes V_{h,k}),
    \label{eq:mpoalg_composite_compressions}
\end{align}
with the corresponding composite embeddings.

\begin{lemma}[Composite reductions]%
    \label{thm:mpoalg_composite_reduction}
    \lean{MPSTensor.IsReduction.isReductionResidualNilpotencyBound_trans,

        MPOTensor.isReductionResidualNilpotencyBound_mulTensor_kronId,
        MPOTensor.isReductionResidualNilpotencyBound_mulTensor_idKron,
        MPOTensor.isReductionResidualNilpotencyBound_mulTensor_assoc_left,
        MPOTensor.isReductionResidualNilpotencyBound_mulTensor_assoc_right,
        MPOTensor.GroupFamily.isReductionResidualNilpotencyBound_castIndex,
        MPOTensor.isReductionResidualNilpotencyBound_actTensor_kronId,
        MPOTensor.isReductionResidualNilpotencyBound_actTensor_idKron,
        MPOTensor.GroupFamily.FusionData.isReductionResidualNilpotencyBound_left,
        MPOTensor.GroupFamily.FusionData.isReductionResidualNilpotencyBound_right}
    \leanok
    \uses{def:mps_rectangular_reduction,def:mps_reduction_residual_nilpotency_length,def:mpoalg_fusion_tensors,thm:mpug_reduction_composition,thm:mpug_fusion_trees_reductions}
    Let $(V_1,W_1)$ be a reduction of a tensor $C$ onto $B$ whose residual
    letters $N^i=C^i-W_1B^iV_1$ have vanishing products of length $N_1$, and
    let $(V_2,W_2)$ be a pair of matrices, not necessarily a reduction of $B$
    onto $A$, whose residual letters $M^i=B^i-W_2A^iV_2$ have vanishing
    products of length $N_2$. Then the residual letters
    $R^i=C^i-W_1W_2A^iV_2V_1$ of the composite pair $(V_2V_1,W_1W_2)$ have
    vanishing products of length $2N_1+N_2$. The
    residual bound is unchanged when a reduction is tensored with an identity
    strand, $(V\otimes\Id,W\otimes\Id)$ on $T_aT_d$ or
    $(\Id\otimes V,\Id\otimes W)$ on $T_dT_a$, and likewise for an operator
    tensor acting on a state tensor, and it is unchanged under the bond
    associator. In particular, if $(V,W)$ reduces $T_aT_b$ onto $T_c$ and
    $(V',W')$ reduces $T_cT_d$ onto $T_e$, both with residual bound $K$, then
    $(V'(V\otimes\Id),(W\otimes\Id)W')$ reduces $T_aT_bT_d$ onto $T_e$ with
    residual bound $3K$. Lemma~\ref{thm:mpug_reduction_composition}
    establishes the composite reduction property without the residual bound,
    and Lemma~\ref{thm:mpug_fusion_trees_reductions} specializes it to the
    two fusion trees of a triple product.
    % Source anchor: arXiv:1706.07329, cornerproblem.tex lines 3147--3152
    %   (Definition 8) and 3946--3970 (Lemma VNW=0); arXiv:2203.12563, lines
    %   1026--1060; arXiv:2502.20257, main.tex lines 1506--1535.
\end{lemma}

\begin{proof}
    \leanok
    The letters satisfy $R^i=N^i+W_1M^iV_1$, and $V_1N^{\mathbf q}W_1=0$ for
    every nonempty word $\mathbf q$ because $(V_1,W_1)$ is a reduction. Three
    inductions on words give the following.
    \begin{enumerate}
        \item $V_1R^{\mathbf w}W_1=M^{\mathbf w}$, and
              $V_1R^{\mathbf w}N^{\mathbf q}W_1=0$ for nonempty $\mathbf q$.
              Append a letter $x$ to $\mathbf w$: since
              $R^{\mathbf wx}=R^{\mathbf w}N^x+R^{\mathbf w}W_1M^xV_1$, both
              identities for $\mathbf wx$ follow from those for $\mathbf w$
              and from $V_1W_1=\Id$.
        \item $N^{\mathbf p}R^{\mathbf w}W_1=0$ when
              $|\mathbf p|+|\mathbf w|\geq N_1+N_2$. Prepend a letter $x$:
              $N^{\mathbf p}R^{x\mathbf w}W_1=N^{\mathbf px}R^{\mathbf w}W_1
                  +N^{\mathbf p}W_1M^{x\mathbf w}$ by the first step, and the
              last term vanishes because $|\mathbf p|\geq N_1$ or
              $|x\mathbf w|\geq N_2$.
        \item $R^{\mathbf w}N^{\mathbf q}=0$ when
              $|\mathbf w|+|\mathbf q|\geq 2N_1+N_2$. Append a letter $x$:
              $R^{\mathbf wx}N^{\mathbf q}=R^{\mathbf w}N^{x\mathbf q}
                  +R^{\mathbf w}W_1M^xV_1N^{\mathbf q}$, and the last term
              vanishes because $|\mathbf q|\geq N_1$, or else
              $|\mathbf w|\geq N_1+N_2$ and the second step applies.
    \end{enumerate}
    The third step with $\mathbf q$ empty is the claim. The residual of
    $(V\otimes\Id,W\otimes\Id)$ on $T_aT_d$ has words
    $\sum_{\mathbf m}(T_a-WT_cV)^{(\mathbf i,\mathbf m)}\otimes
        T_d^{(\mathbf m,\mathbf j)}$, which vanish at the same length, and an
    invertible intertwiner of letters conjugates the residual letters.
\end{proof}

\begin{remark}[Sharpness of the hypotheses and of the bound]%
    \label{rem:mpoalg_composite_reduction_sharp}
    \uses{thm:mpoalg_composite_reduction}
    In Lemma~\ref{thm:mpoalg_composite_reduction} the factors must have
    nilpotent residuals: a single slot of a reduction onto several blocks has a residual that is not nilpotent, and composing
    it with the identity reduction of $T_cT_d$ for a trivial $T_d$ returns it
    unchanged. The bound $N_1+N_2$ fails: for one letter acting as the shift
    $f_0\to f_1\to e_1\to e_2\to f_3\to f_4$ on $\C^6$, with $(V_1,W_1)$ the
    compression onto $\operatorname{span}\{e_1,e_2\}$ and $(V_2,W_2)$ the
    compression of the shift $e_1\to e_2$ onto $\operatorname{span}\{e_2\}$
    with $A=0$, one has $N_1=3$, $N_2=2$, and $R^5\neq0$.
    Counting the lengths of the nonvanishing terms in the expansion of the
    proof more carefully gives the bound $\max(N_1,2N_1+N_2-2)$, which this
    example attains.
\end{remark}

\begin{remark}[F-symbols with multiplicity under the zipper condition]%
    \label{rem:mpoalg_f_symbols}

    \cite{Bultinck2017Anyons}, which asks that the fusion tensors intertwine

    Lemma~\ref{thm:mpdo_complete_zipper_fusion_pentagon}; both are stated

    Definition~\ref{def:mpdo_complete_zipper_fusion_family}. For a group-like
    algebra, the scalar playing the role of $F^{ghk}_{ghk}$ in the orientation
    of (\ref{eq:rfp_printed_fmove}) is $\omega(g,h,k)^{-1}$, and the pentagon
    equation is
    (\ref{eq:mpug_anomaly_three_cocycle}).

    Theorem~\ref{thm:mpoalg_fusion_tensors_exist} carry a nilpotent remainder

    gauge, even after dressing with long words

    stated in that setting; for group-like algebras, single-target uniqueness
    yields the associator of Definition~\ref{def:mpug_anomaly_three_cochain}.

\begin{remark}[Fusion tensors from a coproduct]%
    \label{rem:mpoalg_weak_hopf}
    For a matrix product operator representation of a cosemisimple
    pre-bialgebra $\mathcal A$, the operators are indexed by irreducible
    representations $\psi_a$ of the dual algebra, and the fusion tensors
    decompose the coproduct of $\mathcal A^*$,
    $(\psi_a\otimes\psi_b)\circ\Delta(f)=\sum_{c,\mu}W_{ab}^{c\mu}\psi_c(f)
        V_{ab}^{c\mu}$ with $V_{ab}^{c\mu}W_{ab}^{d\nu}=
        \delta_{cd}\delta_{\mu\nu}\Id$ \cite[Section~4]{Molnar2022MPOAlgebrasI}.
    The maps are named as in Definition~\ref{def:mpoalg_fusion_tensors}; the
    source writes the same identity as $\sum_{c,\mu}V_{ab}^{c\mu}\psi_c(f)
        W_{ab}^{c\mu}$ with $W_{ab}^{c\mu}V_{ab}^{d\nu}=\delta_{cd}\delta_{\mu\nu}\Id$,
    so the letters $V$ and $W$ are swapped. Lemma~\ref{thm:mpoalg_fusion_coproduct}
    reads this identity backwards: the fusion tensors of a complete zipper
    fusion family define the coproduct, and
    Lemma~\ref{lem:mpoalg_fusion_coproduct_counit} identifies its counit. No
    part of the correspondence concerning units or antipodes is stated here.
    % Source anchor: arXiv:2204.05940, mpo.tex line 1655
    %   (eq:splitting_algebraic) and 1725--1892 (basis change, F-symbols,
    %   pentagon).
\end{remark}

\begin{lemma}[Coproduct of a complete zipper fusion family]%
    \label{thm:mpoalg_fusion_coproduct}
    \lean{MPOTensor.CompleteZipperFusionFamily.coproduct,
        MPOTensor.CompleteZipperFusionFamily.coproduct_apply_eq_sum,
        MPOTensor.CompleteZipperFusionFamily.span_letter_eq_top,
        MPOTensor.CompleteZipperFusionFamily.coproduct_letter,
        MPOTensor.CompleteZipperFusionFamily.coproduct_unique,
        MPOTensor.CompleteZipperFusionFamily.coproduct_mul,
        MPOTensor.CompleteZipperFusionFamily.coproduct_one}
    \leanok
    \uses{def:mpdo_complete_zipper_fusion_family}
    Let $(B_a)_{a\in\Lambda}$ be a complete zipper fusion family
    (Definition~\ref{def:mpdo_complete_zipper_fusion_family}) with fusion
    tensors $X^c_{ab,\mu}$ and left inverses $X^{c+}_{ab,\mu}$. For
    $X=(X_c)_c\in\bigoplus_c\MN{D_c}$ put
    \begin{align}
        \Delta(X)_{ab}
         & =\sum_{c,\mu}X^c_{ab,\mu}\,X_c\,X^{c+}_{ab,\mu}
        \in\MN{D_a}\otimes\MN{D_b}.
        \label{eq:mpoalg_fusion_coproduct}
    \end{align}
    Then the elements $(B_c^{ik})_c$ span $\bigoplus_c\MN{D_c}$, and
    \begin{align}
        \Delta\big((B_c^{ik})_c\big)_{ab}
         & =\sum_jB_a^{ij}\otimes B_b^{jk}.
        \label{eq:mpoalg_fusion_coproduct_letter}
    \end{align}
    The map $\Delta$ is the only linear map with
    (\ref{eq:mpoalg_fusion_coproduct_letter}), so it depends only on the
    blocks. It is multiplicative, $\Delta(XY)=\Delta(X)\Delta(Y)$, and
    $\Delta(\Id)_{ab}=\sum_{c,\mu}X^c_{ab,\mu}X^{c+}_{ab,\mu}$ is the support
    projector of the fusion tensors. In the notation of
    \cite[Section~4]{Molnar2022MPOAlgebrasI}, $X^c_{ab,\mu}$ and
    $X^{c+}_{ab,\mu}$ are the source's $V_{ab}^{c\mu}$ and $W_{ab}^{c\mu}$,
    (\ref{eq:mpoalg_fusion_coproduct}) is the decomposition of the coproduct of
    the dual algebra, (\ref{eq:mpoalg_fusion_coproduct_letter}) is the
    coproduct of the matrix coefficients of the physical representation, and
    multiplicativity is the pre-bialgebra axiom.
    % Source anchor: arXiv:2204.05940, mpo.tex lines 1409--1414
    %   (pre-bialgebra), 1655 (eq:splitting_algebraic), 1708
    %   (eq:delta_epsilon), 1182 (injectivity), 2003--2006
    %   (eq:MPO_tensor_product).
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:mpdo_complete_zipper_fusion_family}
    The simultaneous left inverse of the block letters writes every matrix
    unit of every block as a combination of the elements $(B_c^{ik})_c$, so
    they span. Equation (\ref{eq:mpoalg_fusion_coproduct_letter}) is the
    reconstruction of the product letters through the fusion support, and
    uniqueness follows from spanning. Multiplicativity follows from
    $X^{d+}_{ab,\nu}X^c_{ab,\mu}=\delta_{cd}\delta_{\mu\nu}\Id$, which gives
    $\Delta(X)\Delta(Y)=\sum_{c,\mu}X^c_{ab,\mu}X_cY_cX^{c+}_{ab,\mu}$.
\end{proof}

\begin{definition}[Pre-bialgebra]
    \label{def:mpoalg_pre_bialgebra}
    \lean{PreBialgebra, Bialgebra.toPreBialgebra}
    \leanok
    A pre-bialgebra over a commutative semiring $R$ is an algebra
    $\mathcal A$ over $R$ with linear maps
    $\Delta\colon\mathcal A\to\mathcal A\otimes\mathcal A$ and
    $\epsilon\colon\mathcal A\to R$ such that $\Delta$ is multiplicative and
    coassociative and $\epsilon$ is a counit:
    \begin{align}
        \Delta(xy)                    & =\Delta(x)\Delta(y),            &
        (\Delta\otimes\Id)\circ\Delta & =(\Id\otimes\Delta)\circ\Delta,
        \label{eq:mpoalg_pre_bialgebra}                                   \\
        (\epsilon\otimes\Id)\circ\Delta
                                      & =\Id,                           &
        (\Id\otimes\epsilon)\circ\Delta
                                      & =\Id,
        \label{eq:mpoalg_pre_bialgebra_counit}
    \end{align}
    after the standard identifications of the two triple tensor products and
    of $R\otimes\mathcal A$ and $\mathcal A\otimes R$ with $\mathcal A$.
    Neither $\Delta(\Id)=\Id\otimes\Id$ nor any relation between $\epsilon$
    and the unit or the product is required. Every bialgebra is a
    pre-bialgebra.

    This is \cite[Definition~4.1]{Molnar2022MPOAlgebrasI}: an algebra that is
    also a coalgebra \cite[Definition~3.1]{Molnar2022MPOAlgebrasI} with
    multiplicative coproduct. The source works with finite-dimensional
    complex vector spaces.
    % Source anchor: arXiv:2204.05940, mpo.tex lines 866--875
    %   (Definition 3.1), 1407 and 1409--1414 (Definition 4.1).
\end{definition}

\begin{lemma}[The fusion coproduct is coassociative]%
    \label{thm:mpoalg_fusion_coproduct_coassoc}
    \lean{MPOTensor.CompleteZipperFusionFamily.tensorPairEquiv,
        MPOTensor.CompleteZipperFusionFamily.comul,
        MPOTensor.CompleteZipperFusionFamily.comul_letter,
        MPOTensor.CompleteZipperFusionFamily.comul_coassoc}
    \leanok
    \uses{thm:mpoalg_fusion_coproduct,def:mpoalg_pre_bialgebra}
    Write $\mathcal B=\bigoplus_c\MN{D_c}$. Distributing the tensor product
    over the two direct sums and identifying $\MN{D_a}\otimes\MN{D_b}$ with
    $\MN{D_aD_b}$ by the Kronecker product gives an algebra isomorphism
    $\mathcal B\otimes\mathcal B\cong\bigoplus_{a,b}\MN{D_a}\otimes\MN{D_b}$.
    Read through it, the coproduct (\ref{eq:mpoalg_fusion_coproduct}) is a
    multiplicative linear map $\Delta\colon\mathcal B\to\mathcal B\otimes\mathcal B$
    with
    \begin{align}
        \Delta\big((B_c^{ik})_c\big)
         & =\sum_j(B_c^{ij})_c\otimes(B_c^{jk})_c,
        \notag
    \end{align}
    and it is coassociative,
    $(\Delta\otimes\Id)\circ\Delta=(\Id\otimes\Delta)\circ\Delta$ after
    the standard identification of the two triple tensor products. Thus
    $\Delta$ satisfies (\ref{eq:mpoalg_pre_bialgebra}), while $\Delta(\Id)$
    is the support projector of Lemma~\ref{thm:mpoalg_fusion_coproduct} and
    need not be the unit. The source assumes
    coassociativity and derives the associativity of fusion trees and the
    $F$-symbols; here it is proved from the fusion tensors.
    % Source anchor: arXiv:2204.05940, mpo.tex lines 866--875
    %   (Definition 3.1), 1409--1414 (Definition 4.1), 1732--1787
    %   (eq:associative), 2003--2006 (eq:MPO_tensor_product).
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:mpoalg_fusion_coproduct}
    Under the isomorphism, an elementary tensor $X\otimes Y$ has blocks
    $X_a\otimes Y_b$, so the letter formula is
    (\ref{eq:mpoalg_fusion_coproduct_letter}). Multiplicativity is that of
    Lemma~\ref{thm:mpoalg_fusion_coproduct}, transported along an algebra
    isomorphism. Both iterated coproducts are linear, and by
    Lemma~\ref{thm:mpoalg_fusion_coproduct} the elements $(B_c^{il})_c$
    span $\mathcal B$. By the letter formula applied twice, both send
    $(B_c^{il})_c$ to
    $\sum_{j,k}(B_c^{ij})_c\otimes(B_c^{jk})_c\otimes(B_c^{kl})_c$.
\end{proof}

\begin{lemma}[Counit of the fusion coproduct]%
    \label{lem:mpoalg_fusion_coproduct_counit}
    \lean{MPOTensor.CompleteZipperFusionFamily.rTensor_counit_comp_comul,
        MPOTensor.CompleteZipperFusionFamily.lTensor_counit_comp_comul,

        MPOTensor.CompleteZipperFusionFamily.identityBlockCounit,
        MPOTensor.CompleteZipperFusionFamily.identityBlockCounit_letter,

    \leanok
    \uses{thm:mpoalg_fusion_coproduct_coassoc,def:mpoalg_pre_bialgebra}
    In the setting of Lemma~\ref{thm:mpoalg_fusion_coproduct_coassoc}, let
    $\epsilon\colon\mathcal B\to\C$ be a linear functional with
    \begin{align}
        \epsilon\big((B_c^{ik})_c\big)
         & =\delta_{ik}.
        \label{eq:mpoalg_fusion_counit}
    \end{align}
    Then $\epsilon$ satisfies (\ref{eq:mpoalg_pre_bialgebra_counit}) for
    the fusion coproduct, so $(\mathcal B,\Delta,\epsilon)$ is a
    pre-bialgebra in the sense of Definition~\ref{def:mpoalg_pre_bialgebra}.
    There is at most one functional with (\ref{eq:mpoalg_fusion_counit}). If
    one block is the identity operator, $B_{c_0}^{ik}=\delta_{ik}\Id$, then
    $X\mapsto D_{c_0}^{-1}\tr X_{c_0}$ is such a functional.

    In \cite[Section~4]{Molnar2022MPOAlgebrasI} the boundary algebra is the
    dual algebra $\mathcal A^*$, whose counit is evaluation at the unit of
    $\mathcal A$, and $(B_c^{ik})_c$ is the image of the matrix coefficient
    $x\mapsto\phi(x)_{ik}$ of the physical representation $\phi$. Since
    $\phi(\Id)=\Id$, the source's counit is the functional
    (\ref{eq:mpoalg_fusion_counit}). The fusion data alone do not force such
    a functional to exist: one block of bond dimension one whose letters form
    a nonzero matrix $(B^{ik})_{ik}$ of square zero, with no fusion channel,
    has $\Delta=0$, which admits no counit. The pre-bialgebra is therefore stated
    for a given functional with (\ref{eq:mpoalg_fusion_counit})
    \cite{gap:mrgscp22_pre_bialgebra_counit_scope}.
    Lemma~\ref{lem:mpoalg_fusion_counit_existence} characterizes when such a
    functional exists, and
    Lemma~\ref{lem:mpoalg_fusion_counit_unital_representation} constructs it
    for a family obtained from a unital representation.
    % Source anchor: arXiv:2204.05940, mpo.tex lines 866--875
    %   (Definition 3.1), 880 and 1419 (counit of the dual algebra), 1421
    %   (representations are unital), 1640 (the dual is a pre-bialgebra).
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:mpoalg_fusion_coproduct,thm:mpoalg_fusion_coproduct_coassoc}
    Both sides of each identity in (\ref{eq:mpoalg_pre_bialgebra_counit})
    are linear, and the elements $(B_c^{ik})_c$ span $\mathcal B$ by
    Lemma~\ref{thm:mpoalg_fusion_coproduct}. By the letter formula and
    (\ref{eq:mpoalg_fusion_counit}),
    $(\epsilon\otimes\Id)\Delta\big((B_c^{ik})_c\big)
        =\sum_j\delta_{ij}(B_c^{jk})_c=(B_c^{ik})_c$, and similarly on the
    other side. Spanning also gives uniqueness. For an identity block,
    $D_{c_0}^{-1}\tr(\delta_{ik}\Id)=\delta_{ik}$.
\end{proof}

\begin{lemma}[Existence of the counit]%

\begin{proof}

    \uses{thm:mpoalg_fusion_coproduct,lem:mpoalg_fusion_coproduct_counit}
    The range of $\psi$ contains the elements $(B_c^{ik})_c$, which span
    $\mathcal B$ by Lemma~\ref{thm:mpoalg_fusion_coproduct}, so $\psi$ is
    onto. On $(B_c^{ik})_c=\psi(\phi_{ik})$ the functional takes the value
    $\phi_{ik}(\Id)=\phi(\Id)_{ik}=\delta_{ik}$. The pre-bialgebra structure
    is that of Lemma~\ref{lem:mpoalg_fusion_coproduct_counit}.
\end{proof}

\begin{remark}[Kramers--Wannier duality is an intertwiner]%
    \label{rem:mpoalg_kramers_wannier}
    The Kramers--Wannier operator of Section~\ref{sec:asymex_kw} does not
    belong to a fusion algebra on one lattice: by
    (\ref{eq:asymex_kw_fusion}) its square is $(\Id+\eta)T$ up to
    normalization, which contains the translation $T$ and is not a
    length-independent combination of the operators of a finite family.
    It is a matrix product operator intertwiner between two symmetric
    Hamiltonians, a bimodule between the symmetry algebras of the primal and
    dual chains, rather than a symmetry of one of them
    \cite{Lootens2023Dualities,Lootens2021MPOStringNet}. Fusion and action
    tensors for such intertwiners come from the same asymmetric compression
    theorem, applied to products of operators between different chains; the
    Kramers--Wannier case is worked out in
    Section~\ref{sec:asymex_kw_bimodule}.
    % Source anchor: arXiv:2112.09091, _Hamiltonian.tex lines 22--28 and
    %   _Examples.tex lines 55--63.
\end{remark}

\begin{remark}[Density operators]%
    \label{rem:mpoalg_mpdo}
    Renormalization fixed points of matrix product density operators are
    described by matrix product operator algebras of $C^*$-weak Hopf algebras
    through the pulling-through identity of the canonical regular element
    \cite{RuizDeAlarcon2024MPOAlgebrasII}; the density-operator setting is
    that of Chapter~\ref{ch:mpdo}. The fusion tensors of
    Section~\ref{sec:mpoalg_virtual} apply there without change, since they
    use only equality of periodic traces.

    %   algebra) and 802--818 (canonical regular element).
\end{remark}
