\section{The rough upper comparison for a given label sequence}

\begin{theorem}[The physical upper comparison with a vanishing remainder]
    \label{thm:al_global_schmidt_bell_rough_upper}
    \lean{TNLean.PEPS.AreaLaw.eventually_suppliedGlobalSchmidtBellPrevector_rough_upper}
    \leanok
    \uses{thm:al_global_schmidt_bell_prevector}
    Let $V$ be a finite set of physical sites of local dimension $q$,
    $\Omega$ a unit vector, $X\subseteq V$, and $E$ a spectral set of the
    reduced density on $X$ with mass $z>0$, cardinality $d=|E|$, and normalized
    selected Schmidt vector $\psi$. For a label sequence $\lambda_k$ satisfying
    \begin{align}
        \|(I\otimes P_{\lambda_k})\psi^{\otimes k}\|^2
         & \geq\frac{1}{2(k+1)^{d^2}}
        \label{eq:al_rough_upper_label_mass}
    \end{align}
    for all sufficiently large $k$, form the physical prevector $\xi_k$ from
    $\Omega$ and $\lambda_k$ as in
    Theorem~\ref{thm:al_global_schmidt_bell_prevector}.

    Fix a finite index set $J$, a finite nested list $\mathcal R$ of regions
    disjoint from $X$, and $K\geq8$. Following
    \cite[Section~8, lines~130--169 and~332--381]{OpenAI2026AreaLaw}, there is
    $c\geq0$ and a threshold in $k$ (both chosen before
    $Z,U,Y,T,B_{\mathrm{sh}},B_{\mathrm{exc}}$) such that for any $Z$ with
    $Z^*Z=I$ and $ZZ^*=P_{\mathrm{sym}}$, regions $U_{jg},Y_{jg}\subseteq V$
    ($j\in J$, $0\leq g<K$), and iterated weighted geometric mean $T$ with
    \begin{align}
        U_{jg}\cap Y_{jg}        & =\varnothing,                       &
        X\cap(U_{jg}\cup Y_{jg}) & =\varnothing,                       &
        U_{jg}\cup Y_{jg}        & \subseteq U_{j'h}\quad(g<h), \notag
    \end{align}
    if $S_\Omega(D)\leq B_{\mathrm{sh}}$ for all $D\in\mathcal R$ and each
    $Q\in\{U_{jg},U_{jg}\cup Y_{jg}\}$ is nested with some $D\in\mathcal R$
    satisfying $1+\log\dim\mathcal H_{D\mathbin{\triangle} Q}\leq B_{\mathrm{exc}}$,
    then at $t=2/K$ in the full-dimension replica metric $W$ with auxiliary
    sites $C,R$ and complements in $V$, the operators and vector
    \begin{align}
        L_{jg} & =X\cup U_{jg}\cup\{C\},                               &
        F_{jg} & =(X\cup U_{jg}\cup Y_{jg})^c\cup\{R\},         \notag   \\
        A_{jg} & =(W_{L_{jg}}^{-1}W_{F_{jg}}^{-1}W_{Y_{jg}})^2,        &
        M_k    & =\prod_{g=0}^{K-1}T(Z^*A_{\cdot g}Z),                 &
        w_k    & =M_k^{-1/4}Z^*\xi_k
        \label{eq:al_rough_upper_vector}
    \end{align}
    satisfy $\|w_k\|^2>0$ eventually in $k$ and
    \begin{align}
        -\frac{\log\|w_k\|^2}{k}
         & \leq2\log d-\log z
        +4\left(\frac{B_{\mathrm{sh}}}{z}+B_{\mathrm{exc}}\right)
        +R_{d,c,K}(k),                                         \notag \\
        R_{d,c,K}(k)
         & =\frac{\log4+(d^2+cK)\log(k+2)}{k}\longrightarrow0.
        \label{eq:al_rough_upper_rate}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Express the auxiliary mass (\ref{eq:al_rough_upper_label_mass}) in
    physical-site coordinates and apply the rough upper estimate for $\xi_k$ at
    $t=2/K$ and exponent $1/4$. Since $Y_{jg}\setminus X=Y_{jg}$, the ordered
    product in (\ref{eq:al_rough_upper_vector}) equals $M_k$ with prefactor
    $8(1/4)(2/K)K=4$, yielding (\ref{eq:al_rough_upper_rate}).
\end{proof}

\section{The sharp upper comparison}

\begin{theorem}[Sharp upper rate for the fill and charge roots]
    \label{thm:al_actual_sharp_upper}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.eventually_actualScan_suppliedPrevector_sharp_upper}
    \leanok
    \uses{thm:al_global_schmidt_bell_prevector,def:al_scan_actual_histories,
        def:al_scan_status_marginals}
    Fix finite $\Lambda,T\subseteq\mathbb Z^2$ ($T\ne\varnothing$), local
    dimension $q>0$, a unit vector $\Omega_t$ on $\Lambda$, and a collar
    scanner with $K$ bands, initial near region $A\subseteq\Lambda$, scales
    $m,M>0$, depth $d_T(x)=\min_{t\in T}\|x-t\|_\infty$, target $X=A\cap T$,
    $8Km\leq L$, and horizon $nm$. Fix a spectral set $E$ of the reduced
    density on $X$ with $d=|E|$, mass $z>0$, and a label sequence $\lambda_k$
    satisfying (\ref{eq:al_rough_upper_label_mass}) eventually. Let $a>0$ with
    $6a\leq\min\{r,1\}$, and suppose every marginal $Q$ of the five physical
    marginals of each completed or pre-charge status up to $nm$ has surprisal
    $Z_Q$ satisfying
    \begin{align}
        \log\mathbb E\exp(u Z_Q)
         & \leq uS_{\Omega_t}(Q)+K_{\mathrm{mom}}u^2,
         & |u|                                        & \leq r.
        \label{eq:al_actual_sharp_moments}
    \end{align}
    For each terminal $j$ and band $g$ with positive-depth near region $U_{jg}$
    and middle region $Y_{jg}$, put
    \begin{align}
        G_{jg}
         & =S_{\Omega_t}(U_{jg})+S_{\Omega_t}(U_{jg}\cup Y_{jg})
        -S_{\Omega_t}(Y_{jg}).
        \label{eq:al_actual_sharp_terminal_entropy}
    \end{align}
    Let $T_p$ be the weighted mean tree and $N_{k,p}$ the squared filtered norm
    of the original prevector (at metric parameter $a/2$, with $p$ projected
    onto $[0,1]$). There is $c\geq0$ and a threshold in $k$ (chosen before the
    scan time, $p$, and $T_p$) such that simultaneously at all scan times and
    $p\in\mathbb R$,
    \begin{align}
        N_{k,p}      & >0,\notag          \\
        -\frac{\log N_{k,p}}{k}
                     & \leq2\log d-\log z
        +\frac a2\sum_g\sum_jT_p(j)G_{jg}
        +6K K_{\mathrm{mom}}a^2-\frac K2\log z
        +R_{d,c,K}(k),
        \label{eq:al_actual_sharp_rate}   \\
        R_{d,c,K}(k) & \longrightarrow0.
        \label{eq:al_actual_sharp_remainder}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_scan_band_nesting,thm:al_scan_receiving_regions,
        thm:al_augmented_root_compression,thm:al_scan_nested_band_metrics,
        thm:al_scan_actual_fill_admission,thm:al_scan_actual_charge_admissible}
    Combine all completed and pre-charge partitions up to $nm$ into one finite
    family; terminal maps preserve the partitions and mean-tree weights, and
    Theorem~\ref{thm:al_scan_band_nesting} provides the common band order.
    Applying the sharp upper comparison of
    \cite[Section~8, lines~383--418]{OpenAI2026AreaLaw} in symmetric
    coordinates with exponent $1/4$ gives the prefactors $2(1/4)a=a/2$,
    $24(1/4)K K_{\mathrm{mom}}a^2=6K K_{\mathrm{mom}}a^2$, and $2(1/4)K=K/2$ in
    (\ref{eq:al_actual_sharp_rate})--(\ref{eq:al_actual_sharp_remainder}).
\end{proof}

\begin{theorem}[One physical label sequence with the sharp upper rate]
    \label{thm:al_actual_sharp_prevector}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_actualScan_schmidtBellPrevector_sharp_upper}
    \leanok
    \uses{thm:al_global_schmidt_bell_prevector,thm:al_actual_sharp_upper}
    In Theorem~\ref{thm:al_actual_sharp_upper}, if $\Omega_t$ is an eigenvector
    of a physical Hamiltonian with eigenvalue $e$, there is a single label
    sequence $\lambda_k$ whose prevector has mean-replica energy $e$, full copy
    symmetry, label $\lambda_k$ on both auxiliary registers, selected mass bound
    (\ref{eq:al_rough_upper_label_mass}), $\log\dim\lambda_k-kS(\psi_C)=o(k)$,
    and sharp upper rate (\ref{eq:al_actual_sharp_rate}) at every fill and
    charge root.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_global_schmidt_bell_prevector,thm:al_actual_sharp_upper}
    Select $\lambda_k$ via Theorem~\ref{thm:al_global_schmidt_bell_prevector}
    and apply Theorem~\ref{thm:al_actual_sharp_upper}.
\end{proof}

\begin{theorem}[The truncated target instance with an explicit power margin]
    \label{thm:al_source_target_actual_sharp_upper}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_truncated_target_actualSharpUpper}
    \leanok
    \uses{thm:al_actual_sharp_upper}
    Fix $q\geq1$, range $R\in\mathbb N$, term bound $J\geq0$, gap $\Delta>0$,
    and $C_T\geq0$; these determine $C_{\mathrm{tr}},C_{\mathrm{target}},C_B>0$.
    Fix $0<\ell<1$, $0<\kappa<\mu<1-\ell$, $\nu>0$, $0<e<1$ with $\kappa\leq e$,
    and $\omega\in\mathbb R$ satisfying
    \begin{align}
        \omega+\frac{1+\kappa}{2} & <1-\ell-\mu.
        \label{eq:al_source_target_sharp_margin}
    \end{align}
    There is $N\geq2$ such that for all $n\geq N$, finite
    $\Lambda,T\subseteq\mathbb Z^2$ ($T\ne\varnothing$), range-$R$ Hamiltonian
    $H=\sum_i h_i$ on $\Lambda$ with $\|h_i\|\leq J$, and unit ground state
    $\Omega$ ($H\Omega=E_0\Omega$,
    $H-E_0I\succeq\Delta(I-|\Omega\rangle\langle\Omega|)$), the collar scanner
    with $M>0$, depth $d_T$, target $X=A\cap T$, and scales
    \begin{align}
        L   & =\lfloor n^{1-\ell}\rfloor,                       &
        m   & =\lfloor n^\mu\rfloor,                            &
        D   & =\lceil n^\kappa\rceil,                    \notag   \\
        K   & =\lfloor L/(8m)\rfloor,                           &
        r_0 & =\lceil C_{\mathrm{tr}}(\log n)^2\rceil \notag
    \end{align}
    (assuming $|T|\leq C_Tn^2$, $|T_d\setminus T_{d-1}|\leq n$ for
    $1\leq d\leq L$, and $d_T(x)>2L+10r_0$ for all cut-edge endpoints $x$ of
    $A$) admits a single truncated unit vector $\Omega_t$ satisfying
    Theorem~\ref{thm:al_scan_target_typical_window}, $m,K\geq1$, and the common
    status and moment parameters
    \begin{align}
        B   & =C_B nD(\log n)^{12}\geq1, &
        g   & =\frac{\Delta}{c_+},       &
        C_0 & =512e\frac2g,              &
        c_0 & =\frac{1}{32\sqrt{1+2/g}}.
        \label{eq:al_source_target_sharp_budget}
    \end{align}
    For $0<W\leq n^\omega$, setting $a=W/K$ gives $6a\leq\min\{c_0/\sqrt B,1\}$,
    and every label sequence satisfying (\ref{eq:al_rough_upper_label_mass}) on
    the target spectral set satisfies (\ref{eq:al_actual_sharp_rate}) with
    $K_{\mathrm{mom}}=C_0B$ (giving the target-instance step under the explicit
    margin (\ref{eq:al_source_target_sharp_margin}) in
    \cite[Proposition~9.6]{OpenAI2026AreaLaw}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_scan_target_typical_window,thm:al_scan_common_status_budget,
        thm:al_scan_status_scale_separation,thm:al_moment_weight_smallness,
        thm:al_scan_common_moment_interval,thm:al_actual_sharp_upper}
    Theorem~\ref{thm:al_scan_target_typical_window} supplies $\Omega_t$, and
    Theorem~\ref{thm:al_scan_common_status_budget} bounds all completed and
    pre-charge statuses on $\Omega_t$ by $B$ in
    (\ref{eq:al_source_target_sharp_budget}). By
    Theorems~\ref{thm:al_scan_status_scale_separation} and
    \ref{thm:al_moment_weight_smallness}, the margin
    (\ref{eq:al_source_target_sharp_margin}) ensures $6a\leq\min\{c_0/\sqrt B,1\}$,
    so Theorem~\ref{thm:al_scan_common_moment_interval} yields
    (\ref{eq:al_actual_sharp_moments}) with $K_{\mathrm{mom}}=C_0B$ and
    Theorem~\ref{thm:al_actual_sharp_upper} gives (\ref{eq:al_actual_sharp_rate}).
\end{proof}
