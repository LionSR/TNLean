\section{SPT Phase Labels and Virtual-Boundary Nondecay}
%% ---------------------------------------------------------

For one-dimensional gapped systems within the injective symmetric MPS and
parent-Hamiltonian framework~\cite{Schuch2011Phases}, symmetry-protected
topological (SPT) phases are classified by the second cohomology group
$H^2(G,\mathrm{U}(1))$ of the symmetry group, appearing as the cohomology class
of the virtual projective representation.

\begin{definition}[Equality of virtual cocycle labels]
    \label{def:is_same_spt_phase}
    \lean{MPSTensor.IsSameSPTPhase}
    \leanok
    \uses{def:cohomologous_to, def:projective_rep}
    For tensors $A,B$ and a fixed physical representation
    $U:G\to\GL_d(\C)$, the predicate called \emph{same SPT phase}
    asserts the existence of factor systems
    $\omega_A,\omega_B:G\times G\to\C^\times$ and projective bond
    representations $\rho_A,\rho_B$ such that
    \begin{align}
        \widetilde A_g^i
         & =\rho_A(g^{-1})A^i\rho_A(g^{-1})^{-1}, \\
        \widetilde B_g^i
         & =\rho_B(g^{-1})B^i\rho_B(g^{-1})^{-1},
        \notag
    \end{align}
    and $\omega_A\sim\omega_B$.
\end{definition}

\begin{remark}[Cohomology with $\C^\times$ and $\mathrm{U}(1)$ coefficients]
    \label{rem:spt_scalar_character_conventions}
    \uses{def:is_same_spt_phase, def:h2_cx,
        thm:mpug_circle_complex_units_h2}
    While factor systems here take values in $\C^\times$, for finite groups $G$
    every class admits a unit-modulus representative, identifying
    $H^2(G,\C^\times)$ with $H^2(G,\mathrm U(1))$
    (Theorem~\ref{thm:mpug_circle_complex_units_h2}).
\end{remark}

\begin{definition}[Polar deformation in the original physical space]
    \label{def:spt_polar_deformation}
    \lean{MPSTensor.polarIsometricTensor, MPSTensor.polarDeformationPos,
        MPSTensor.polarDeformation}
    \leanok
    \uses{def:ldpolar_physical_matrix, thm:ldpolar_decomposition}
    Let $M:\C^{D^2}\to\C^d$ be the physical map of a tensor $A$, and
    write $M=VP$ for its right polar decomposition. Define $W$ by its
    physical map $V$, and define $A_\gamma$ by
    \begin{align}
        M_\gamma & =V\bigl((1-\gamma)I+\gamma P\bigr),
        \qquad 0\leq\gamma\leq1.
        \notag
    \end{align}
    This is the polar interpolation of
    \cite[Section~II.C]{Schuch2011Phases},
    expressed in right polar factors. It stays in the original physical
    space; the map $M$ may be rectangular.
\end{definition}

\begin{theorem}[Injectivity and symmetry along the polar deformation]
    \label{thm:spt_polar_deformation}
    \lean{MPSTensor.isInjective_of_leftInverse_physicalMatrix,
        MPSTensor.posDef_polarDeformationPos,
        MPSTensor.isInjective_polarDeformation,
        MPSTensor.polarDeformation_zero, MPSTensor.polarDeformation_one,
        MPSTensor.continuous_polarDeformation,
        MPSTensor.polarPos_commute_of_unitary_covariance,
        MPSTensor.polarIso_covariance_of_isInjective,
        MPSTensor.polarDeformation_covariance,
        MPSTensor.physicalMatrix_mul_left_right,
        MPSTensor.rotatePhysical_polarDeformation_of_unitary_covariance}
    \leanok
    \uses{def:spt_polar_deformation, def:injective}
    Suppose $A$ is one-site injective. Then $P_\gamma=(1-\gamma)I+\gamma P$
    is positive definite, $A_\gamma$ is injective, and the path is continuous,
    with $A_0=W$ and $A_1=A$. If unitaries $U$ and $K$ satisfy $UM=MK$,
    then $P$ commutes with $K$, $UV=VK$, and $UM_\gamma=M_\gamma K$.
    In particular, if $U$ acts on $A$ by conjugation with a unitary bond
    matrix $X$, it acts on every $A_\gamma$ by the same conjugation
    \cite[Section~II.C, isometric form and symmetries]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Injectivity gives $V^*V=I$ and $P>0$. Every $P_\gamma$ is positive
    definite, so $P_\gamma^{-1}V^*$ is a left inverse of $M_\gamma$.
    Its rows express every matrix unit as a linear combination of the
    tensor letters, proving injectivity. Continuity and the endpoint
    identities follow from the affine formula. Unitary covariance gives
    $[M^*M,K]=0$; functional calculus gives $[P,K]=0$. Cancellation of
    the invertible factor $P$ gives $UV=VK$, and hence the covariance
    of the entire path. For bond conjugation, $K=X^T\otimes X^*$.
\end{proof}

\begin{theorem}[Uniform parent gap along an injective polar deformation]
    \label{thm:spt_polar_deformation_gap}
    \lean{MPSTensor.exists_uniform_parentHamiltonianES_gap_polarDeformation,
        MPSTensor.parentInteractionES_polarDeformation_commute_onSiteTensorPow}
    \leanok
    \uses{thm:spt_polar_deformation, thm:compact_parent_gap,
        thm:spt_parent_hamiltonian_symmetry}
    Let $A$ be one-site injective with $D>0$, and let $H_{N,2}(A_\gamma)$
    be its canonical two-site periodic parent Hamiltonian. There is
    $\delta>0$ such that, for every $\gamma\in[0,1]$, every $N\geq2$, and
    every $v\perp\ker H_{N,2}(A_\gamma)$,
    \begin{align}
        \delta\lVert v\rVert & \leq\lVert H_{N,2}(A_\gamma)v\rVert.
        \notag
    \end{align}
    Under the unitary bond-conjugation covariance of the preceding theorem,
    each canonical range-$L$ parent interaction commutes with $U^{\otimes L}$.
    This is the one-site injective case of the polar-deformation argument
    in \cite[Section~II.C and Appendix~A]{Schuch2011Phases}; the source
    also treats normal forms with several blocks
    \cite{gap:spc11_uniform_gap_injective_scope}.
\end{theorem}

\begin{proof}\leanok
    The preceding theorem gives a continuous injective family on the compact
    interval $[0,1]$. The compact-family parent-gap theorem applies to this
    family. Covariance by bond conjugation gives gauge equivalence under
    physical rotation, so the canonical parent interactions commute with
    the physical tensor powers.
\end{proof}

\begin{theorem}[Canonical interactions along a symmetric injective path]
    \label{thm:spt_canonical_injective_gapped_path}
    \lean{MPSTensor.continuous_parentInteraction_matrix_family,
        MPSTensor.interactionHamiltonian_parent_spectrum_gap,
        MPSTensor.canonicalInjectiveGappedPath}
    \leanok
    \uses{def:spt_common_space_gapped_path}
    Let $\gamma\mapsto A_\gamma$ be a continuous path of one-site injective
    tensors on fixed physical and bond spaces, with $D>0$. Fix a unitary
    on-site representation $U$ such that, for every $\gamma\in[0,1]$ and
    every $g\in G$, physical rotation of $A_\gamma$ by $U_g$ is a virtual
    gauge change. Then the canonical two-site parent interactions give a
    symmetric gapped interaction path between the endpoint interactions.
    Their norms are at most one, their ground energy is zero, and one
    positive spectral gap works for every parameter and every ring length
    $N\geq2$ \cite[Section~II.C and Appendix~A]{Schuch2011Phases}.
    The one-site injective restriction is documented in
    \cite{gap:spc11_uniform_gap_injective_scope}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:compact_parent_gap, thm:spt_parent_hamiltonian_symmetry}
    The injective local ground-space projections vary continuously, so their
    complementary projections do as well. Their norms are at most one.
    Compactness gives a uniform norm gap above the kernel. Positivity of
    the parent Hamiltonian makes all its eigenvalues nonnegative; applying
    the norm bound to a unit eigenvector orthogonal to the kernel shows
    that the same positive constant is a lower bound for every nonzero
    eigenvalue. The
    nonzero periodic MPS vector is a zero eigenvector. Finally, covariance
    makes every local parent projection, and hence its periodic sum,
    commute with the physical symmetry.
\end{proof}

\begin{theorem}[Symmetric gapped path to the isometric parent]
    \label{thm:spt_polar_gapped_interaction_path}
    \lean{MPSTensor.polarGappedInteractionPath}
    \leanok
    \uses{def:spt_common_space_gapped_path}
    Let $A$ be one-site injective with $D>0$, let $U$ be a unitary on-site
    representation, and suppose the tensor covariance is expressed by
    unitary bond conjugation. The canonical two-site parent interactions
    of $A$ and of its isometric form $W$ are joined by a symmetric gapped
    interaction path on the original physical space, with the same
    representation $U$ throughout
    \cite[Section~II.C, isometric form and symmetries]{Schuch2011Phases}.
    No equality between the physical dimension and $D^2$ is required.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:spt_canonical_injective_gapped_path, thm:spt_polar_deformation}
    Apply the preceding construction to the injective polar path. The
    endpoint identities give the canonical parents of $W$ and $A$; unitary
    bond-conjugation covariance is preserved by the entire interpolation.
\end{proof}

\begin{theorem}[Unitary virtual covariance from unital normalization]
    \label{thm:spt_unitary_virtual_covariance_unital}
    \lean{MPSTensor.exists_posDef_adjoint_fixedPoint_of_isInjective_unital,
        MPSTensor.exists_unitary_covariance_of_isInjective_unital,
        MPSTensor.exists_unitary_virtual_covariance_of_isOnSiteSymmetric_unital}
    \leanok
    \uses{def:injective, def:on_site_symmetric}
    Let $A$ be one-site injective with $D>0$ and $\E_A(I)=I$.
    There is a positive definite matrix $\Lambda$ with
    $\E_A^\dagger(\Lambda)=\Lambda$. If a unitary physical rotation of $A$
    is a virtual gauge change, the same rotation is implemented by a
    unitary bond matrix. In particular, for a fixed unitary on-site
    representation preserving every periodic MPS vector, unitary virtual
    covariance can be chosen for every group element
    \cite[Section~II.C, isometric form and symmetries]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:injective_irreducible, lem:trace_mul_transfer_map_adjoint,
        thm:virtual_gauge_fixes_boundary_state, lem:gauge_equiv_twisted}
    Perron--Frobenius theory gives $\Lambda>0$ and $r>0$ with
    $\E_A^\dagger(\Lambda)=r\Lambda$. Trace duality and unitality give
    \begin{align}
        \tr\Lambda & =\tr\bigl(\Lambda\E_A(I)\bigr)
        =\tr\bigl(\E_A^\dagger(\Lambda)\bigr)=r\tr\Lambda.
        \notag
    \end{align}
    Since $\tr\Lambda>0$, $r=1$. The boundary-state theorem makes any
    virtual gauge $X$ a nonzero scalar multiple $cW$ of a unitary $W$.
    The scalar cancels in conjugation. On-site symmetry supplies a gauge
    for each group element by the injective fundamental theorem, and the
    preceding argument supplies the unitary choices.
\end{proof}

\begin{theorem}[Prepared polar path from an on-site symmetric injective tensor]
    \label{thm:spt_prepared_polar_gapped_path}
    \lean{MPSTensor.exists_unital_parent_representative_of_isInjective,
        MPSTensor.exists_prepared_polarGappedInteractionPath_of_isOnSiteSymmetric}
    \leanok
    \uses{def:injective, def:on_site_symmetric,
        def:spt_common_space_gapped_path}
    Let $A$ be one-site injective with $D>0$, and let a fixed unitary
    on-site representation $U$ preserve every periodic MPS vector of $A$.
    There are an injective unital tensor $B$, a nonzero scalar $c$, and
    a virtual gauge relating $B$ to $cA$, such that every canonical
    parent interaction of $B$ equals the corresponding interaction of $A$.
    The canonical two-site parent of the isometric form of $B$ and the
    canonical two-site parent of $A$ are joined by a symmetric gapped
    interaction path on the original physical space, with representation
    $U$ throughout \cite[Section~II.C]{Schuch2011Phases}.
    Neither normalization of $A$ nor unitary virtual matrices are assumed.
    The one-site injective restriction is documented in
    \cite{gap:spc11_uniform_gap_injective_scope}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:injective_irreducible, thm:unital_data_irreducible,
        thm:local_ground_space_rescaling,
        thm:local_ground_space_gauge_invariance,
        lem:on_site_symmetric_smul_gauge,
        thm:spt_unitary_virtual_covariance_unital,
        thm:spt_polar_gapped_interaction_path}
    Perron normalization gives $B$ by nonzero rescaling and virtual gauge.
    Both operations preserve the local MPS spaces, hence all canonical
    parent interactions; they also preserve on-site symmetry and
    injectivity. The preceding theorem supplies unitary virtual covariance
    for $B$. Apply the symmetric polar-path construction and replace its
    final interaction by the equal canonical interaction of $A$.
\end{proof}

\begin{theorem}[Virtual class under gauge preparation]
    \label{thm:spt_virtual_class_gauge_preparation}
    \lean{TNLean.Algebra.ProjectiveRepresentation.conjugate,
        MPSTensor.twistedTensor_covariance_of_smul_gauge,
        MPSTensor.cohomologousTo_of_smul_gaugeEquiv}
    \leanok
    \uses{def:projective_rep, thm:cocycle_gauge_independence}
    A virtual change of basis $Y$ replaces a projective representation
    $\rho(g)$ by $Y\rho(g)Y^{-1}$ and retains its factor system. If
    $B_i=Y(cA_i)Y^{-1}$, this representation implements the symmetry of
    $B$ whenever $\rho$ implements the symmetry of $A$. For $D>0$ and
    injective $B$, every virtual representation chosen for $B$ has factor
    system cohomologous to that of $\rho$
    \cite[Sections~II.C and II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Conjugation preserves the projective multiplication law and commutes
    with physical twisting. Scalar rescaling also commutes with twisting.
    Apply gauge independence on $B$ to compare the transported
    representation with any other choice.
\end{proof}

\begin{theorem}[Virtual class along the prepared polar path]
    \label{thm:spt_prepared_polar_virtual_class}
    \lean{MPSTensor.exists_unitary_virtualRep_polarDeformation,
        MPSTensor.cohomologousTo_polarIsometricTensor_of_isInjective_unital,
        MPSTensor.exists_prepared_polarGappedInteractionPath_with_virtual_class}
    \leanok
    \uses{lem:projective_rep_from_virtual_covariance,
        thm:spt_virtual_class_gauge_preparation,
        thm:spt_prepared_polar_gapped_path, thm:cocycle_gauge_independence}
    Let $A$ be one-site injective with $D>0$, with exact on-site symmetry
    under a fixed unitary representation $U$. Choose any virtual
    projective representation for $A$, with factor system $\omega_A$.
    The unital representative $B$ and symmetric gapped path of the
    preparation theorem can be chosen together with a unitary projective
    representation $\rho$ of factor system $\omega\sim\omega_A$ which
    implements virtual symmetry at every point of the polar deformation.
    Consequently, any virtual representations chosen at the endpoints
    of the polar path have cohomologous factor systems
    \cite[Sections~II.C and II.F.2]{Schuch2011Phases}.
    The one-site injective restriction is documented in
    \cite{gap:spc11_uniform_gap_injective_scope}.
\end{theorem}

\begin{proof}\leanok
    Choose unitary virtual covariance for $B$ and assemble these same
    matrices into a projective representation. Polar covariance preserves
    the matrices throughout the deformation, hence also their factor
    system. Gauge preparation identifies this class with $[\omega_A]$;
    gauge independence compares it with any endpoint choices.
\end{proof}

\begin{theorem}[Classification along symmetric MPS parent-Hamiltonian paths]
    \label{thm:spt_classification}
    \notready
    \uses{def:is_same_spt_phase, def:h2_cx}
    For $p=0,1$, let $A_p$ be an injective MPS tensor whose parent Hamiltonian
    $H_p$ has a unique ground state and is symmetric under a unitary
    representation $U^p$ of $G$ on $\mc H_p$. Choose the endpoint phase
    gauges. After a common finite blocking, the embedded endpoints admit a
    continuous symmetric MPS path with corresponding translationally invariant,
    finite-range parent Hamiltonians that vary smoothly and remain uniformly
    gapped if and only if their virtual projective representations determine
    the same class in $H^2(G,\mathrm U(1))$.

    More precisely, for the forward construction there exist a path-sector
    Hilbert space $\mc H_{\mathrm{path}}$ and a linear representation
    $U^{\mathrm{path}}$ such that the endpoints embed into
    \begin{align}
        \mc H & =\mc H_0\oplus\mc H_1\oplus\mc H_{\mathrm{path}}, &
        U_g   & =U_g^0\oplus U_g^1\oplus U_g^{\mathrm{path}}.
        \notag
    \end{align}
    The converse quantifies over smooth paths of MPS with corresponding
    parent Hamiltonians \cite[Section~II.F, lines~929--953]{Schuch2011Phases}.
\end{theorem}

\begin{proof}
    First deform each blocked injective MPS to its isometric form; the
    deformation preserves the symmetry relation
    \begin{align}
        U_g\mc P & =\mc P(V_g\otimes\overline{V_g}).
        \notag
    \end{align}
    Along any continuous symmetric MPS path quantified in the theorem, the
    tensor and its induced virtual action $V_g$ vary continuously. The
    cohomology class is discrete, so it cannot change along that path. This
    proves necessity within the MPS/parent-Hamiltonian framework.

    Conversely, let the endpoint isometric forms have virtual actions
    $V_g^0$ and $V_g^1$ in the same cohomology class.  Embed their bond spaces
    into the direct sum and interpolate between the maximally entangled bonds
    by
    % Local fix: the second range ends at D_0+D_1, not D_1; see
    % docs/paper-gaps/spc11_spt_interpolation_upper_range.tex.
    \begin{align}
        |\omega(\gamma)\rangle
         & =(1-\gamma)\sum_{i=1}^{D_0}|i,i\rangle
        +\gamma\sum_{i=D_0+1}^{D_0+D_1}|i,i\rangle.
        \notag
    \end{align}
    The commuting parent Hamiltonians along this interpolation remain gapped,
    with the uniform spectral estimate of Lemma~\ref{lem:spt_independent_bond_gap}.
    Their symmetry is
    \begin{align}
        (V_g^0\oplus V_g^1)\otimes
        (\overline{V_g^0}\oplus\overline{V_g^1})
         & =\widehat U_g^0\oplus\widehat U_g^1
        \oplus\widehat U_g^{\mathrm{path}},    \\
        \widehat U_g^{\mathrm{path}}
         & =(V_g^0\otimes\overline{V_g^1})
        \oplus(V_g^1\otimes\overline{V_g^0}).
        \notag
    \end{align}
    Equality of the cohomology classes is exactly what makes the cross-term
    action $\widehat U^{\mathrm{path}}$ a linear representation.  Conjugating
    back by the endpoint isometries gives the common direct-sum physical
    representation and the required symmetric gapped path.
    % General endpoint embeddings and the converse classification remain open.
    % The injective polar deformation is proved separately above.
\end{proof}

\begin{theorem}[Virtual-boundary nondecay for injective MPV-symmetric tensors]%
    \label{thm:has_string_order_of_symmetric_injective}
    \lean{MPSTensor.hasStringOrder_of_symmetric_injective}
    \leanok
    \uses{def:has_string_order, thm:virtual_rep_injective}
    Let $A$ be an injective tensor, symmetric under a unitary
    representation $U$. Assume $\E_A(\Id)=\Id$, $\Lambda$ is positive
    definite, $\tr(\Lambda)=1$, and $\E_A^\dagger(\Lambda)=\Lambda$.
    Then the virtual-boundary nondecay predicate holds for every group
    element $g$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:virtual_rep_injective, thm:twisted_transfer_modulus_one_gauge_phase, thm:virtual_unitary_of_twisted_gauge_phase, thm:boundary_state_invariant_of_virtual_unitary, lem:has_string_order_of_local_symmetry}
    The virtual representation theorem produces
    $\rho(g^{-1}) \in \GL_{D}(\C)$ satisfying
    $\sum_j U(g)_{ij}\,A^j
        = \rho(g^{-1})\,A^i\,\rho(g^{-1})^{-1}$.
    A direct computation using $\E_A(\Id)=\Id$ shows that
    $\E_{U(g)}(\rho(g^{-1}))=\rho(g^{-1})$, so
    $\rho(g^{-1})$ is a nonzero eigenvector with eigenvalue~$1$.
    Theorem~\ref{thm:twisted_transfer_modulus_one_gauge_phase} gives
    gauge-phase equivalence with the companion tensor, and
    Theorem~\ref{thm:virtual_unitary_of_twisted_gauge_phase} extracts a virtual
    unitary and phase.
    By Theorem~\ref{thm:boundary_state_invariant_of_virtual_unitary}, this
    unitary preserves the boundary state, so
    Lemma~\ref{lem:has_string_order_of_local_symmetry} gives
    virtual-boundary nondecay.
\end{proof}

\begin{remark}[Virtual-boundary nondecay does not determine the SPT label]%
    \label{rem:string_order_symmetry_diagnostic}
    \uses{thm:has_string_order_iff_of_symmetric_injective}
    Theorem~\ref{thm:has_string_order_of_symmetric_injective} applies to
    every injective symmetric tensor with canonical normalisations,
    irrespective of its cocycle class, so tensors in different SPT phases
    all satisfy virtual-boundary nondecay for every group element.  This
    nondecay is equivalent to virtual local covariance
    (Theorem~\ref{thm:string_order_iff_local_symmetry}); the invariant
    that separates SPT phases is the cohomology class of the virtual
    representation cocycle (Theorem~\ref{thm:spt_classification}).
\end{remark}

\subsection{Virtual actions along symmetric paths}
\label{sec:spt_virtual_path_algebra}

The interpolation in the proof of Theorem~\ref{thm:spt_classification} first
rephases the two endpoint virtual actions to a common factor system and then forms
their direct sum; necessity rests on the discreteness of the class along a
continuous virtual path \cite[Section~II.F.2]{Schuch2011Phases}.

\begin{theorem}[Unitary rephasing of cohomologous virtual actions]
    \label{thm:spt_unitary_rephasing}
    \lean{TNLean.Algebra.ProjectiveRepresentation.rephase,
        TNLean.Algebra.circleScalar_mem_unitaryGroup,
        TNLean.Algebra.ProjectiveRepresentation.rephase_mem_unitaryGroup,
        TNLean.Algebra.ProjectiveRepresentation.exists_unitary_rephase_of_cohomologous,
        TNLean.Algebra.ProjectiveRepresentation.norm_factorSystem,
        TNLean.Algebra.ProjectiveRepresentation.circlePhaseInclusion_eq,
        TNLean.Algebra.ProjectiveRepresentation.exists_unitary_rephase_with_factorSystem}
    \leanok
    \uses{def:projective_rep, def:cohomologous_to}
    Multiplying each matrix $V_g$ of a projective representation with factor system
    $\omega$ by a scalar $\lambda_g\neq0$ gives a projective representation with
    factor system $\lambda_g\lambda_h\lambda_{gh}^{-1}\omega(g,h)$.  Consequently, let
    $V^0$ be a unitary projective representation whose factor system is the
    pointwise phase $\omega_0/|\omega_0|$.  If $\omega_1$ is cohomologous to
    $\omega_0$, then multiplying each $V_g^0$ by a unit-modulus scalar gives a unitary
    projective representation with factor system $\omega_1/|\omega_1|$.
    If two unitary virtual actions on spaces of positive dimension have factor
    systems $\omega_0$ and $\omega_1$, then $|\omega_0(g,h)|=|\omega_1(g,h)|=1$.
    If these systems are cohomologous, one can rephase the first action
    to have the factor system of the second
    \cite[Section~II.F.2, lines~886--921]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Let $\lambda$ witness $\omega_1\sim\omega_0$, so that
    $\omega_1(g,h)=\lambda_g\lambda_h\lambda_{gh}^{-1}\omega_0(g,h)$, and put
    $\psi_g=\lambda_g/|\lambda_g|$.  The unit-phase map $z\mapsto z/|z|$ is
    multiplicative, so
    \begin{align}
        \frac{\omega_1(g,h)}{|\omega_1(g,h)|}
         & =\psi_g\psi_h\psi_{gh}^{-1}\,\frac{\omega_0(g,h)}{|\omega_0(g,h)|}.
        \notag
    \end{align}
    Hence $\psi_gV_g^0$ has factor system $\omega_1/|\omega_1|$, and multiplication
    by a unit-modulus scalar preserves unitarity.
    For a unitary virtual action,
    $1=\|V_gV_h\|=|\omega(g,h)|\,\|V_{gh}\|=|\omega(g,h)|$.
    Hence its factor system equals its pointwise unit phase.
\end{proof}

\begin{definition}[Direct sum with a common factor system]
    \label{def:spt_projective_direct_sum}
    \lean{TNLean.Algebra.ProjectiveRepresentation.directSum}
    \leanok
    \uses{def:projective_rep}
    Let $V^0$ and $V^1$ be projective representations on
    $\C^{D_0}$ and $\C^{D_1}$ with the same factor system $\omega$.
    Their direct sum is the projective representation on
    $\C^{D_0+D_1}$ given by
    \begin{align}
        V_g    & =V_g^0\oplus V_g^1, &
        V_gV_h & =\omega(g,h)V_{gh}.
        \notag
    \end{align}
    This is the virtual direct sum in the symmetric interpolation
    of~\cite[Section~II.F.2, lines~909--921]{Schuch2011Phases}.
\end{definition}

\begin{theorem}[Unitarity of the virtual direct sum]
    \label{thm:spt_projective_direct_sum_unitary}
    \lean{TNLean.Algebra.ProjectiveRepresentation.directSum_mem_unitaryGroup}
    \leanok
    \uses{def:spt_projective_direct_sum}
    If $V_g^0$ and $V_g^1$ are unitary, then $V_g^0\oplus V_g^1$ is unitary.
\end{theorem}

\begin{proof}\leanok
    The product with its adjoint is
    $(V_g^0(V_g^0)^*)\oplus(V_g^1(V_g^1)^*)=I_{D_0}\oplus I_{D_1}$.
\end{proof}

\begin{theorem}[Cohomology along a fixed-dimension virtual path]
    \label{thm:spt_virtual_path_class_invariance}
    \lean{MPSTensor.continuous_rootOfUnity_path_const,
        MPSTensor.exists_continuous_pow_root_path,
        MPSTensor.continuous_projectiveFactor_of_unitaryMatrixPath,
        MPSTensor.projectiveFactor_endpoints_cohomologous_of_det_path,
        MPSTensor.unitary_projectivePath_factor_endpoints_cohomologous}
    \leanok
    \uses{def:cohomologous_to}
    Let $G$ be any group and $D>0$. Suppose matrices
    $V_\gamma(g)\in\mathrm U(D)$ depend continuously on
    $\gamma\in[0,1]$ for each $g\in G$ and obey
    $V_\gamma(g)V_\gamma(h)=\omega_\gamma(g,h)V_\gamma(gh)$.
    Then $\omega_0$ and $\omega_1$ are cohomologous as
    $\C^\times$-valued factor systems. This theorem assumes the
    virtual path; obtaining it from a smooth symmetric MPS
    parent-Hamiltonian path, including changes of bond dimension,
    is a separate step in the classification theorem
    \cite[Section~II.F.2, lines~930--946]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Each $\det V_\gamma(g)$ is a continuous nonzero complex path, so it
    has a continuous $D$th root $\beta_\gamma(g)$ on the interval.
    Taking determinants in $V_\gamma(g)V_\gamma(h)=\omega_\gamma(g,h)V_\gamma(gh)$
    gives $\beta_\gamma(g)^D\beta_\gamma(h)^D=\omega_\gamma(g,h)^D\beta_\gamma(gh)^D$,
    so the rescaled factor system
    \begin{align}
        \eta_\gamma(g,h)
         & =\omega_\gamma(g,h)\,\frac{\beta_\gamma(gh)}{\beta_\gamma(g)\beta_\gamma(h)}
         &
        \eta_\gamma(g,h)^D
         & =1
        \notag
    \end{align}
    takes values among the $D$th roots of unity.  It depends continuously on
    $\gamma$, hence is constant, and $\eta_0=\eta_1$ reads
    \begin{align}
        \omega_1(g,h)
         & =\psi(g)\psi(h)\psi(gh)^{-1}\omega_0(g,h),
         &
        \psi(g)
         & =\beta_1(g)/\beta_0(g).
        \notag
    \end{align}
\end{proof}

\subsection{Renormalization fixed points of SPT phases}
\label{sec:spt_fixed_points}

Within each SPT phase a representative with zero correlation length can be built
from the cocycle data \cite[Section~III.A]{Cirac2021Matrix}.  The tensor printed in
the source commutes with its own gauges and, for a nontrivial group, is not normal;
the construction below
is the dimer fixed point that the printed claims describe
\cite{gap:rmp_spt_fixed_point_tensor}.  The source starts from a 2-cocycle alone.
Most results below take a projective representation with that factor system as
given; for a finite group, the twisted regular representation on $\C^G$ supplies
one whose factor system lies in the class of the cocycle
(Theorem~\ref{thm:spt_fixed_point_from_cocycle},
\cite{gap:rmp_spt_fixed_point_supplied_representation}).  The source's symmetry
carries the phase of a 1-cocycle; the symmetry and class results below are
stated for the trivial character \cite{gap:rmp_spt_fixed_point_trivial_character}.
The source's symmetry group may also contain time-reversal and reflection
symmetries, with twisted cocycles; the results below treat on-site symmetries
only \cite{gap:rmp_spt_fixed_point_onsite_scope}.

\begin{definition}[SPT fixed-point tensor and its symmetry]
    \label{def:spt_fixed_point_tensor}
    \lean{MPSTensor.sptFixedPointTensor, MPSTensor.sptFixedPointAction, MPSTensor.sptGauge, MPSTensor.sptKron}
    \leanok
    \uses{def:projective_rep}
    Let $\rho$ be a projective representation of $G$ on $\C^D$ with factor system
    $\omega$, let $\phi$ be a character of $G$, and put $W_g=\rho(g^{-1})$.  The
    SPT fixed-point tensor has physical space $\C^D\otimes\C^D$ and letters
    $A^{(a,b)}=D^{-1/2}\ket{a}\bra{b}$, and $G$ acts on the physical space by
    $U(g)=\phi(g)\,W_g^{T}\otimes W_g^{-1}$.  The projective representation
    need not be unitary; when it is, $W_g^{-1}=W_g^{\dagger}$, so with
    $X_g=W_g^{\dagger}$ the conjugation $W_gA^iW_g^{-1}$ is the source's
    $X_g^{\dagger}A^iX_g$.
\end{definition}

\begin{theorem}[The SPT fixed point has zero correlation length]
    \label{thm:spt_fixed_point_transfer}
    \lean{MPSTensor.transferMap_sptFixedPointTensor, MPSTensor.star_sptScale_mul_sptScale, MPSTensor.sptFixedPointTensor_isTransferIdempotent, MPSTensor.finrank_range_transferMap_sptFixedPointTensor, MPSTensor.sptFixedPointTensor_isInjective, MPSTensor.sptFixedPointTensor_isNormal, MPSTensor.sum_smul_sptFixedPointTensor_apply, MPSTensor.eq_of_sum_smul_sptFixedPointTensor_eq}
    \leanok
    \uses{def:spt_fixed_point_tensor, def:transfer_map, def:injective}
    Let $D\geq1$.  The transfer map of the SPT fixed-point tensor is
    $X\mapsto D^{-1}\tr(X)\,\Id$, an idempotent of rank one, and the tensor is
    injective, hence normal, so its letters are linearly independent
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    $\sum_{a,b}\ket{a}\bra{b}X\ket{b}\bra{a}=\tr(X)\,\Id$; the letters are nonzero
    multiples of the matrix units.
\end{proof}

\begin{theorem}[The SPT fixed point realizes the class of $\omega$]
    \label{thm:spt_fixed_point_class}
    \lean{MPSTensor.twistedTensor_sptFixedPointTensor, MPSTensor.sptFixedPointTensor_isOnSiteSymmetric, MPSTensor.cohomologousTo_of_sptFixedPointTensor, MPSTensor.exists_virtualRep_sptFixedPointTensor}
    \leanok
    \uses{def:spt_fixed_point_tensor, def:on_site_symmetric}
    The map $g\mapsto U(g)$ is a representation of $G$.  For every $g$ and $i$,
    $\sum_jU(g)_{ij}A^j=\phi(g)\,W_gA^iW_g^{-1}$.  For
    $\phi=1$ the tensor has MPV-family invariance under $U$, and, when $D\geq1$, this
    symmetry has a virtual projective representation, and every virtual projective
    representation of it has a factor system cohomologous to $\omega$
    \cite[Section~III.A]{Cirac2021Matrix}.  The source states the symmetry and
    the class for every 1-cocycle $\phi$; for $\phi\neq1$ only the identity
    $\sum_jU(g)_{ij}A^j=\phi(g)\,W_gA^iW_g^{-1}$ is formalized
    \cite{gap:rmp_spt_fixed_point_trivial_character}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:cocycle_gauge_independence, thm:virtual_rep_injective, thm:spt_fixed_point_transfer}
    Since $W_{gh}$ is a nonzero multiple of $W_hW_g$ and scalars cancel in
    $W\mapsto W^{T}\otimes W^{-1}$, the factor system cancels and
    $U(g)U(h)=U(gh)$.  The symmetry identity is checked entry by entry:
    $W_g\ket{a}\bra{b}W_g^{-1}=\sum_{a',b'}(W_g)_{a'a}(W_g^{-1})_{bb'}
        \ket{a'}\bra{b'}$.  With $\phi=1$ the gauges $\rho(g^{-1})$ realize the
    symmetry, so Theorem~\ref{thm:cocycle_gauge_independence} makes every other
    virtual representation cohomologous to $\rho$.  The tensor is injective, so
    Theorem~\ref{thm:virtual_rep_injective} also produces a virtual representation
    directly, and its factor system is then cohomologous to $\omega$.
\end{proof}

\begin{theorem}[The SPT fixed point of a cocycle class]
    \label{thm:spt_fixed_point_from_cocycle}
    \lean{TNLean.Algebra.ScalarCocycle.inverse,
        TNLean.Algebra.ScalarCocycle.inverse_isCocycle,
        TNLean.Algebra.ScalarCocycle.inverse_circlePhaseInclusion,
        TNLean.Algebra.ScalarCocycle.twistedLeftRegularGL,
        TNLean.Algebra.ScalarCocycle.twistedLeftRegularGL_mul,
        TNLean.Algebra.ScalarCocycle.twistedLeftRegularProjective,
        TNLean.Algebra.ScalarCocycle.twistedLeftRegularProjective_mem_unitaryGroup,
        TNLean.Algebra.ScalarCocycle.regularProjectiveForClass,
        TNLean.Algebra.ScalarCocycle.regularProjectiveForClass_mem_unitaryGroup,
        MPSTensor.sptFixedPointAction_mem_unitaryGroup,
        MPSTensor.exists_unitary_sptFixedPoint_of_isCocycle}
    \leanok
    \uses{thm:spt_fixed_point_class, thm:mpug_twisted_regular_multiplication,
        thm:mpug_twisted_regular_unitary, thm:mpug_circle_complex_units_h2}
    Let $G$ be a finite group and $\omega$ a 2-cocycle.  There is a unitary
    projective representation $\rho$ of $G$ on $\C^G$ whose factor system is
    cohomologous to $\omega$.  With $\phi=1$ and $D=|G|$, the physical action $U$ of
    Definition~\ref{def:spt_fixed_point_tensor} is unitary, the SPT fixed-point
    tensor has MPV-family invariance under $U$, and every virtual projective
    representation of this symmetry has a factor system cohomologous to $\omega$
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:spt_fixed_point_class, thm:mpug_twisted_regular_multiplication,
        thm:mpug_twisted_regular_unitary, thm:mpug_circle_complex_units_h2}
    Let $\bar\omega=\omega/|\omega|$ be the pointwise phase of $\omega$, which is
    cohomologous to $\omega$ because $G$ is finite.
    The twisted left regular matrices $L_g$ of the inverse cocycle
    $\bar\omega^{-1}$ are unitary and satisfy $L_gL_h=\bar\omega(g,h)L_{gh}$.  When
    $\rho$ is unitary, $W_g^{-1}=W_g^{\dagger}$, so $U(g)=W_g^{T}\otimes W_g^{\dagger}$ is
    a tensor product of unitaries.  Theorem~\ref{thm:spt_fixed_point_class} gives the
    symmetry and shows that every virtual factor system is cohomologous to
    $\bar\omega$, hence to $\omega$.
\end{proof}

\subsection{Symmetric gapped paths and bond regrouping}
\label{sec:spt_symmetric_gapped_paths}

The phase relation of \cite[Sections~II.C.1--2]{Schuch2011Phases} is defined by
paths of local interactions, after the endpoint systems have been blocked and
embedded into a common physical space.  The interpolation in the proof of
Theorem~\ref{thm:spt_classification} uses commuting parent Hamiltonians built
from independent bonds \cite[Sections~II.D.2 and~II.F.2]{Schuch2011Phases}.  This
subsection records the path condition, the bond interpolation and its symmetry,
and the regrouping of physical registers into bonds.

\begin{definition}[Symmetric gapped paths on a common physical space]
    \label{def:spt_common_space_gapped_path}
    \lean{MPSTensor.interactionHamiltonian, MPSTensor.SymmetricGappedInteractionPath}
    \leanok
    Fix a unitary representation $U:G\to\mathrm U(d)$ and two
    two-site interactions $h_0,h_1$.  A symmetric gapped path between them
    on this common physical space consists of Hermitian interactions
    $h_\gamma$, $0\le\gamma\le1$, with the prescribed endpoints, such that:
    \begin{enumerate}
        \item $\|h_\gamma\|_{\mathrm{op}}\le1$;
        \item $\gamma\mapsto h_\gamma$ is continuous;
        \item for every periodic chain of length $N\ge2$, the Hamiltonian
              $H_N(\gamma)=\sum_i h_\gamma(i,i+1)$ commutes with $U_g^{\otimes N}$;
        \item there is a $\delta>0$, independent of $N$ and $\gamma$, such
              that every spectral value of $H_N(\gamma)$ above its least spectral
              value $E_N(\gamma)$ is at least $E_N(\gamma)+\delta$.
    \end{enumerate}
    Ground-state degeneracy is allowed.  This is the common-space path
    condition of~\cite[Sections~II.C.1--2, lines~407--453]{Schuch2011Phases};
    the choice of endpoint blocking and symmetry-compatible embeddings precedes
    this definition.  No condition on virtual cocycles enters it.
\end{definition}

\begin{theorem}[Reversal of a symmetric gapped path]
    \label{thm:spt_common_space_path_reversal}
    \lean{MPSTensor.SymmetricGappedInteractionPath.reverse}
    \leanok
    \uses{def:spt_common_space_gapped_path}
    If $h_\gamma$ is a symmetric gapped path from $h_0$ to $h_1$, then
    $h_{1-\gamma}$ is such a path from $h_1$ to $h_0$, with the same gap bound.
\end{theorem}

\begin{proof}\leanok
    The change of parameter $\gamma\mapsto1-\gamma$ is a continuous
    bijection of the unit interval and exchanges its endpoints.
\end{proof}

\begin{theorem}[Concatenation of symmetric gapped paths]
    \label{thm:spt_common_space_path_concatenation}
    \lean{MPSTensor.SymmetricGappedInteractionPath.toPath,
        MPSTensor.SymmetricGappedInteractionPath.concatenatedInteraction,
        MPSTensor.SymmetricGappedInteractionPath.concatenatedInteraction_mem,
        MPSTensor.SymmetricGappedInteractionPath.trans}
    \leanok
    \uses{def:spt_common_space_gapped_path}
    Suppose symmetric gapped paths connect $h_0$ to $h_1$ and $h_1$
    to $h_2$ on the same physical space with the same on-site action.
    There is then a symmetric gapped path from $h_0$ to $h_2$.
    If the original gap bounds are $\delta_0,\delta_1>0$, the
    concatenated path has gap bound $\min\{\delta_0,\delta_1\}$.
\end{theorem}

\begin{proof}\leanok
    Traverse the first path on the first half of the interval and
    the second on the second half.  Their common endpoint ensures
    continuity.  Every interaction belongs to one of the original
    paths, so the norm bound, symmetry, and stated gap bound follow.
\end{proof}

\begin{theorem}[Order and interpolation of periodic Hamiltonians]
    \label{thm:spt_periodic_hamiltonian_order}
    \lean{MPSTensor.interactionHamiltonian_add_smul_sub,
        MPSTensor.interactionHamiltonian_posSemidef,
        MPSTensor.interactionHamiltonian_mono}
    \leanok
    \uses{def:spt_common_space_gapped_path}
    For a two-site interaction $h$, let $H_N(h)$ be the sum of its
    translates on the periodic chain of length $N\ge2$.
    If $h\ge0$, then $H_N(h)\ge0$; if $h\le k$, then
    $H_N(h)\le H_N(k)$.  For every real $t$,
    $H_N(h+t(k-h))=H_N(h)+t(H_N(k)-H_N(h))$.
\end{theorem}

\begin{proof}\leanok
    Embedding an interaction into a cyclic window is a linear map
    preserving positivity.  Sum these identities and inequalities over
    the cyclic positions.
\end{proof}

\begin{theorem}[Symmetry of canonical parent Hamiltonians]
    \label{thm:spt_parent_hamiltonian_symmetry}
    \lean{MPSTensor.onSiteTensorPow_conjTranspose_mul_self,
        MPSTensor.groundSpaceES_invariant_of_gauge_covariance,
        MPSTensor.parentInteractionES_commute_onSiteTensorPow,
        MPSTensor.onSiteTensorPow_reindex_windowComplement,
        MPSTensor.embedLocalOperator_commute_onSiteTensorPow,
        MPSTensor.parentInteractionES_eq_toEuclideanLin_parentMatrix,
        MPSTensor.parentInteraction_matrix_commute_onSiteTensorPow,
        MPSTensor.localTermES_commute_onSiteTensorPow,
        MPSTensor.parentHamiltonianES_commute_onSiteTensorPow}
    \leanok
    \uses{def:gauge_equiv, def:physical_rotation, def:parent_interaction,
        def:parent_hamiltonian_es}
    Let $U$ be a unitary on the physical space, and suppose the physical
    rotation of $A$ by $U$ is gauge equivalent to $A$.  The canonical
    parent interaction of range $L$ commutes with $U^{\otimes L}$.
    For every $N\geq L$, the periodic parent Hamiltonian commutes
    with $U^{\otimes N}$ \cite[Section~II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Gauge changes preserve the open-boundary MPS space.  Physical rotation
    carries this space to its image under the tensor power of $U$.
    The resulting invariant subspace has an orthogonal projection that
    commutes with that unitary.  Each cyclic translate of its complementary
    projection therefore commutes with the global tensor power, as does
    their sum.
\end{proof}

\begin{theorem}[Ordered interactions with common periodic ground spaces]
    \label{thm:spt_ordered_interaction_path}
    \lean{MPSTensor.orderedGappedInteractionPath}
    \leanok
    \uses{def:spt_common_space_gapped_path, thm:spt_periodic_hamiltonian_order,
        thm:common_kernel_spectral_gap_interpolation}
    Let $a,b$ be positive two-site interactions on one physical space,
    of operator norm at most one, with $a\leq b$. Suppose that for every
    periodic length $N\geq2$ their Hamiltonians $H_N(a)$ and $H_N(b)$
    have the same nonzero ground space at zero energy, and both commute
    with the same on-site symmetry. If $H_N(a)$ has a spectral gap
    at least $\delta>0$ for all $N\geq2$, then
    \begin{align}
        h_\gamma & =a+\gamma(b-a), \qquad 0\leq\gamma\leq1,
        \notag
    \end{align}
    is a symmetric gapped interaction path from $a$ to $b$, with gap
    at least $\delta$. This comparison applies to independent-bond and
    canonical parent interactions in the fixed-point construction
    \cite[Sections~II.D.2 and II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    The periodic Hamiltonian satisfies
    $H_N(h_\gamma)=H_N(a)+\gamma(H_N(b)-H_N(a))$. The added term is positive
    and annihilates the original ground space, so the kernel and the
    lower quadratic-form bound on its orthogonal complement are retained.
    Convexity bounds the local norm by one. Hermiticity, continuity and
    symmetry follow from affine interpolation.
\end{proof}

\begin{theorem}[A continuous unit bond interpolation]
    \label{thm:spt_normalized_bond_interpolation}
    \lean{MPSTensor.bondInterpolationWeight,
        MPSTensor.bondInterpolationMatrix,
        MPSTensor.continuous_bondInterpolationMatrix,
        MPSTensor.bondInterpolationMatrix_ne_zero,
        MPSTensor.bondInterpolationSquaredNorm,
        MPSTensor.bondInterpolationMatrix_sum_normSq,
        MPSTensor.bondInterpolationSquaredNorm_pos,
        MPSTensor.normalizedBondInterpolationMatrix,
        MPSTensor.normalizedBondInterpolationMatrix_ne_zero,
        MPSTensor.normalizedBondInterpolationMatrix_sum_normSq,
        MPSTensor.normalizedBondInterpolationMatrix_zero_apply,
        MPSTensor.normalizedBondInterpolationMatrix_one_apply,
        MPSTensor.continuous_normalizedBondInterpolationMatrix}
    \leanok
    Let $D_0,D_1>0$, and let the two direct-sum bond spaces have these
    dimensions.  Write $W(\gamma)$ for the coefficient matrix of
    \begin{align}
        |\omega(\gamma)\rangle
         & =(1-\gamma)\sum_{i=1}^{D_0}|i,i\rangle
        +\gamma\sum_{i=D_0+1}^{D_0+D_1}|i,i\rangle.
        \notag
    \end{align}
    For every real $\gamma$ this matrix is nonzero, and it depends
    continuously on $\gamma$.  Dividing by
    $\sqrt{D_0(1-\gamma)^2+D_1\gamma^2}$ gives the coefficient matrix
    $\widehat W(\gamma)$ of a continuous unit vector.  At the two endpoints,
    only the first and second diagonal blocks, respectively, remain.  The
    second summation range follows the correction recorded in
    \cite{gap:spc11_spt_interpolation_upper_range}.
\end{theorem}

\begin{proof}\leanok
    The two blocks have disjoint support, so the squared norm of the
    coefficient matrix is $D_0(1-\gamma)^2+D_1\gamma^2$.  This is positive
    for every real $\gamma$ when $D_0,D_1>0$.  The entries and the positive
    normalization factor vary continuously with $\gamma$.
\end{proof}

\begin{theorem}[Symmetry of the interpolating bond]
    \label{thm:spt_bond_interpolation_symmetry}
    \lean{MPSTensor.bondInterpolationMatrix_eq_fromBlocks,
        MPSTensor.bondInterpolationMatrix_commute_directSum,
        MPSTensor.bondInterpolationMatrix_conjugate_directSum,
        MPSTensor.normalizedBondInterpolationMatrix_conjugate_directSum}
    \leanok
    \uses{def:spt_projective_direct_sum, thm:spt_projective_direct_sum_unitary,
        thm:spt_normalized_bond_interpolation}
    Let $V^0,V^1$ be unitary projective representations with a common
    factor system, and put $V_g=V_g^0\oplus V_g^1$.  For the bond coefficient
    matrices $W(\gamma)$ and $\widehat W(\gamma)$ of
    Theorem~\ref{thm:spt_normalized_bond_interpolation},
    \begin{align}
        V_gW(\gamma)V_g^*          & = W(\gamma),          &
        V_g\widehat W(\gamma)V_g^* & = \widehat W(\gamma).
        \notag
    \end{align}
    These are the coefficient-matrix identities expressing invariance of
    the bond vector under $V_g\otimes\overline{V_g}$
    \cite[Section~II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    The matrix $W(\gamma)$ is scalar on each of the two summands, so it
    commutes with $V_g$.  Unitarity gives the first identity, and scalar
    normalization gives the second.
\end{proof}

\begin{definition}[Incoming-bond coordinates]
    \label{def:spt_incoming_bond_coordinates}
    \lean{MPSTensor.incomingBondEquiv, MPSTensor.incomingBondPerm}
    \leanok
    Let each site of a cyclic chain carry two registers of dimension $D$.
    For a configuration $s=((L_i,R_i))_i$, define
    \begin{align}
        e_N(s)_i & = (R_{i-1},L_i).
        \notag
    \end{align}
    The inverse recovers $(L_i,R_i)=((b_i)_2,(b_{i+1})_1)$.
    Pair coordinates may equivalently be enumerated by an alphabet of
    size $D^2$.  These are the independent-bond coordinates
    of~\cite[Section~II.D.2]{Schuch2011Phases}.
\end{definition}

\begin{theorem}[Unitarity of bond regrouping]
    \label{thm:spt_incoming_bond_unitary}
    \lean{MPSTensor.incomingBondPerm_mem_unitaryGroup}
    \leanok
    \uses{def:spt_incoming_bond_coordinates}
    The matrix $P_N$ defined by $(P_N)_{s,b}=\delta_{e_N(s),b}$ is unitary.
    Thus it takes a bond-coordinate wavefunction to the corresponding
    physical-coordinate wavefunction by composition with $e_N$.
\end{theorem}

\begin{proof}\leanok
    The map $e_N$ permutes the configuration basis.
\end{proof}

\begin{theorem}[Locality after bond regrouping]
    \label{thm:spt_incoming_bond_locality}
    \lean{MPSTensor.incomingBondPerm_conj_local_apply}
    \leanok
    \uses{def:spt_incoming_bond_coordinates}
    Let $N>0$ and let $K_i$ act as a matrix $K$ on incoming bond $i$
    and as the identity on every other bond.  Then
    $P_NK_iP_N^\dagger$ acts on the registers $(R_{i-1},L_i)$ and as
    the identity on all other registers.  More precisely, its matrix
    entries between physical configurations $s,t$ are
    \begin{align}
        (P_NK_iP_N^\dagger)_{s,t}
         & =K_{e_N(s)_i,e_N(t)_i}
        \prod_{j\ne i}\delta_{e_N(s)_j,e_N(t)_j}.
        \notag
    \end{align}
    Hence every one-bond interaction becomes a nearest-neighbor physical
    interaction (a one-site interaction when $N=1$).
\end{theorem}

\begin{proof}\leanok
    Conjugation by a permutation matrix replaces the row and column
    configurations by their incoming-bond coordinates.  The tensor-product
    formula for $K_i$ gives the displayed entries.
\end{proof}

\begin{theorem}[Symmetry under bond regrouping]
    \label{thm:spt_bond_regrouping_symmetry}
    \lean{MPSTensor.incomingBondPerm_conj_registerAction,
        MPSTensor.incomingBondPerm_conj_sptFixedPointAction}
    \leanok
    \uses{def:spt_incoming_bond_coordinates,
        def:spt_fixed_point_tensor}
    For matrices $L,R$ on a register of dimension $D$,
    \begin{align}
        P_N(R\otimes L)^{\otimes N}P_N^\dagger
         & =(L\otimes R)^{\otimes N}.
        \notag
    \end{align}
    In particular, for the physical action
    $U(g)=W_g^{\mathsf T}\otimes W_g^{-1}$ of the SPT fixed-point tensor with
    trivial character (Definition~\ref{def:spt_fixed_point_tensor}), the
    incoming-bond action is $W_g^{-1}\otimes W_g^{\mathsf T}$
    \cite[Section~II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    The matrix entries are products of left-register and right-register
    entries.  Regrouping shifts the latter factors by one cyclic site;
    this permutation does not change their product.
\end{proof}

\begin{theorem}[Uniform gap for independent interpolating bonds]
    \label{thm:spt_bond_product_parent_gap}
    \lean{MPSTensor.bondPenaltyAt_isStarProjection,
        MPSTensor.bondPenaltyAt_commute,
        MPSTensor.bondProductParentHamiltonian_gap_one,
        MPSTensor.bondProductParentHamiltonian_groundSpace_eq_span,
        MPSTensor.normalizedBondInterpolation_parent_gap_one,
        MPSTensor.normalizedBondInterpolation_parent_groundSpace_eq_span}
    \leanok
    \uses{thm:spt_normalized_bond_interpolation,
        def:spt_incoming_bond_coordinates}
    Let $D_0,D_1>0$, $N>0$, and $\widehat\omega(\gamma)$ be the
    normalized bond vector of
    Theorem~\ref{thm:spt_normalized_bond_interpolation}. In the Hilbert
    space of $N$ independent two-leg bonds, let
    \begin{align}
        h_i(\gamma)                       & = 1-
        |\widehat\omega(\gamma)\rangle_i
        \langle\widehat\omega(\gamma)|_i, &
        H_N(\gamma)                       & = \sum_{i=1}^{N}h_i(\gamma).
        \notag
    \end{align}
    The terms $h_i(\gamma)$ are commuting orthogonal projections. Their
    common kernel is precisely the line spanned by the product vector
    $\widehat\omega(\gamma)^{\otimes N}$, and for every vector $v$,
    \begin{align}
        \langle v,H_N(\gamma)^2v\rangle
         & \geq \langle v,H_N(\gamma)v\rangle.
        \notag
    \end{align}
    Thus every nonzero eigenvalue is at least one, uniformly in $N$ and
    $\gamma$ \cite[Sections~II.D.2 and~II.F.2]{Schuch2011Phases}. The
    same holds for every unit bond vector in place of
    $\widehat\omega(\gamma)$. This statement uses independent-bond
    coordinates; the passage to the original site coordinates uses the
    unitary of Theorem~\ref{thm:spt_incoming_bond_unitary}.
\end{theorem}

\begin{proof}\leanok
    Unit norm makes each bond-line operator a rank-one orthogonal
    projection. Operators on distinct bond factors commute. In the square
    of their sum, each diagonal term is the original projection and each
    off-diagonal quadratic form is nonnegative. The product of the
    complementary projections is the rank-one projection onto
    $\widehat\omega(\gamma)^{\otimes N}$, which identifies the kernel.
\end{proof}

\begin{theorem}[Continuity of the bond parent Hamiltonian]
    \label{thm:spt_bond_parent_continuity}
    \lean{MPSTensor.continuous_bondProductParentHamiltonian,
        MPSTensor.continuous_normalizedBondInterpolation_parent}
    \leanok
    \uses{thm:spt_normalized_bond_interpolation,
        thm:spt_bond_product_parent_gap}
    For $D_0,D_1>0$ and each $N>0$, the independent-bond parent
    Hamiltonian $H_N(\gamma)$ is continuous in $\gamma\in\mathbb R$.
\end{theorem}

\begin{proof}\leanok
    Each matrix entry of a bond-line projection is a product of a bond
    coefficient and a conjugate coefficient. Both depend continuously
    on $\gamma$. Embedding in the finite chain and summing preserve
    continuity.
\end{proof}

\begin{lemma}[Spectral gap in independent-bond coordinates]
    \label{lem:spt_independent_bond_gap}
    \lean{MPSTensor.bondPenalty_norm_le_one,
        MPSTensor.bondProductParentHamiltonian_posSemidef,
        MPSTensor.bondProductParentHamiltonian_spectrum_gap_one,
        MPSTensor.normalizedBondInterpolation_parent_uniform_gap}
    \leanok
    \uses{thm:spt_bond_product_parent_gap, thm:spt_bond_parent_continuity}
    Let $\eta$ be a unit bond vector and $N\geq1$. The local penalty
    $1-|\eta\rangle\langle\eta|$ has operator norm at most one, and the
    independent-bond Hamiltonian $H_N(\eta)$ is positive semidefinite with
    \begin{align}
        0                              & \in\operatorname{spec}(H_N(\eta)), &
        \operatorname{spec}(H_N(\eta)) & \subseteq\{0\}\cup[1,\infty).
        \notag
    \end{align}
    For $D_0,D_1>0$ and the normalized interpolating bond
    $\widehat\omega(\gamma)$, the penalty depends continuously on
    $\gamma$, and these statements hold for every real $\gamma$ and every
    $N\geq1$, including both endpoints
    \cite[Section~II.F.2, lines~899--906]{Schuch2011Phases}.
\end{lemma}

\begin{proof}\leanok
    The penalty is an orthogonal projection, so its norm is at most one,
    and $H_N(\eta)$ is a sum of orthogonal projections. Its kernel is the
    nonzero line spanned by $\eta^{\otimes N}$, which gives the spectral
    value $0$. For a unit eigenvector with eigenvalue $\lambda$, the
    inequality $H_N(\eta)^2\geq H_N(\eta)$ of
    Theorem~\ref{thm:spt_bond_product_parent_gap} gives
    $\lambda\leq\lambda^2$; with $\lambda\geq0$ this forces $\lambda=0$ or
    $\lambda\geq1$. Continuity along the interpolation is
    Theorem~\ref{thm:spt_bond_parent_continuity}.
\end{proof}

\begin{theorem}[Physical parent Hamiltonian of the interpolating bonds]
    \label{thm:spt_physical_bond_parent_gap}
    \lean{MPSTensor.normalizedPhysicalBondInterpolation_parent_gap_one,
        MPSTensor.normalizedPhysicalBondInterpolation_parent_groundSpace,
        MPSTensor.continuous_normalizedPhysicalBondInterpolation_parent}
    \leanok
    \uses{thm:spt_bond_product_parent_gap,
        thm:spt_bond_parent_continuity,
        thm:spt_incoming_bond_unitary}
    Conjugate $H_N(\gamma)$ by the incoming-bond unitary $P_N$. In the
    original physical-site coordinates, the resulting Hamiltonian
    $P_NH_N(\gamma)P_N^*$ varies continuously with $\gamma$, has the
    transported product vector $P_N\widehat\omega(\gamma)^{\otimes N}$
    as its unique ground line, and satisfies
    \begin{align}
        \langle v,(P_NH_N(\gamma)P_N^*)^2v\rangle
         & \geq \langle v,P_NH_N(\gamma)P_N^*v\rangle
        \notag
    \end{align}
    for every $N>0$ and every physical-chain vector $v$.
\end{theorem}

\begin{proof}\leanok
    The incoming-bond change of coordinates is unitary. It carries the
    product ground line and the quadratic-form inequality to physical
    coordinates. Continuity follows from the continuity of the
    finite-chain bond Hamiltonian, since the unitary conjugation is fixed.
\end{proof}

\begin{theorem}[On-site symmetry of the interpolating-bond parent Hamiltonian]
    \label{thm:spt_physical_bond_parent_symmetry}
    \lean{MPSTensor.normalizedBondInterpolationVector_fixed_directSum,
        MPSTensor.bondProductParentHamiltonian_commute_sitewise_of_unitary_fixed,
        MPSTensor.normalizedPhysicalBondInterpolation_parent_commute_onSite}
    \leanok
    \uses{thm:spt_bond_regrouping_symmetry,
        thm:spt_physical_bond_parent_gap,
        thm:spt_bond_interpolation_symmetry}
    Let $V^{(0)}$ and $V^{(1)}$ be finite-dimensional unitary projective
    representations with the same factor system, and let $V$ be their
    direct sum. Let $U$ be the physical action
    $U(g)=W_g^{\mathsf T}\otimes W_g^{-1}$, $W_g=V_{g^{-1}}$, of the SPT
    fixed-point tensor of $V$ with trivial character
    (Definition~\ref{def:spt_fixed_point_tensor}); it does not depend on
    $\gamma$. For every group element $g$ and every $N>0$,
    \begin{align}
        \left[(U_g)^{\otimes N},
        P_N H_N(\gamma)P_N^*\right] & =0
        \notag
    \end{align}
    \cite[Section~II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    The block-scalar coefficient matrix of the bond is invariant under
    conjugation by $V_g$. In bond coordinates this fixes the bond vector,
    so each rank-one bond projection commutes with the bond action.
    Regrouping carries that action to $(U_g)^{\otimes N}$ and therefore
    carries the commutation relation to physical coordinates.
\end{proof}

\begin{theorem}[Spectral gap of the physical bond parent Hamiltonian]
    \label{thm:spt_physical_bond_parent_spectrum}
    \lean{MPSTensor.physicalBondProductParentHamiltonian_posSemidef,
        MPSTensor.physicalBondProductParentHamiltonian_spectrum_gap_one,
        MPSTensor.normalizedPhysicalBondInterpolation_parent_spectrum_gap_one}
    \leanok
    \uses{thm:spt_physical_bond_parent_gap}
    For every unit bond vector, and in particular for
    $\widehat\omega(\gamma)$ with $D_0,D_1>0$, the physical parent
    Hamiltonian $P_NH_N(\gamma)P_N^*$ is positive semidefinite for every
    $N>0$. Its spectrum contains $0$, and every spectral value $z$
    satisfies $\operatorname{Re}z=0$ or $\operatorname{Re}z\geq1$.
\end{theorem}

\begin{proof}\leanok
    The Hamiltonian is a unitary conjugate of a sum of orthogonal
    projections, hence positive semidefinite. Its nonzero kernel gives the
    spectral value $0$. For a unit eigenvector with eigenvalue $\lambda$,
    the quadratic-form inequality of
    Theorem~\ref{thm:spt_physical_bond_parent_gap} reads
    $\lambda\leq\lambda^2$; with $\lambda\geq0$ this forces $\lambda=0$ or
    $\lambda\geq1$.
\end{proof}

\begin{theorem}[Two-site interaction for the bond parent Hamiltonian]
    \label{thm:spt_two_site_bond_interaction}
    \lean{MPSTensor.twoSiteBondInteraction_isStarProjection,
        MPSTensor.twoSiteBondInteraction_norm_le_one,
        MPSTensor.embed_twoSiteBondInteraction_eq_conj,
        MPSTensor.interactionHamiltonian_twoSiteBondPenalty}
    \leanok
    \uses{thm:spt_incoming_bond_locality, thm:spt_physical_bond_parent_gap}
    For a bond matrix $K$, let $h(K)$ act as $K$ on the middle
    registers $(R_0,L_1)$ of two sites, with identity on the exterior
    registers. If $K$ is an orthogonal projection, so is $h(K)$, and
    its Hilbert-space operator norm is at most one. For every $N\geq2$,
    \begin{align}
        \sum_{i=0}^{N-1}h(I-|\eta\rangle\langle\eta|)_{i,i+1}
         & =P_N H_N(\eta)P_N^*.
        \notag
    \end{align}
    Here the indices are cyclic and $H_N(\eta)$ is the sum of the
    independent-bond penalties
    \cite[Section~II.D.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    The two-site operator is a reordering of
    $I\otimes K\otimes I$. Its translate at $i$ is the conjugate
    under $P_N$ of the independent-bond term at $i+1$. Summing and
    reindexing by the cyclic successor gives the identity.
\end{proof}

\begin{theorem}[Symmetric gapped path between the direct-sum fixed points]
    \label{thm:spt_direct_sum_fixed_point_path}
    \lean{MPSTensor.sptFixedPointUnitaryAction,
        MPSTensor.normalizedBondFixedPointGappedPath}
    \leanok
    \uses{def:spt_common_space_gapped_path,
        thm:spt_two_site_bond_interaction,
        thm:spt_physical_bond_parent_spectrum,
        thm:spt_physical_bond_parent_symmetry}
    Let $D_0,D_1>0$, and let $V^{(0)}$ and $V^{(1)}$ be unitary
    projective representations with the same factor system. Set
    $D=D_0+D_1$ and let $U$ be the unitary physical action of the SPT
    fixed-point tensor of $V^{(0)}\oplus V^{(1)}$ with trivial
    character. If $\widehat\omega(\gamma)$ is the normalized
    interpolating bond, define the two-site interaction to be the
    identity on the outer registers and
    $1-|\widehat\omega(\gamma)\rangle
        \langle\widehat\omega(\gamma)|$
    on the bond joining the sites. Then these interactions form a
    symmetric gapped path on the common physical space of dimension
    $D^2$. Their operator norms are at most one, and the gap above the
    unique ground line is at least one for every $N\ge2$ and every
    $\gamma\in[0,1]$ \cite[Section~II.F.2]{Schuch2011Phases}.
\end{theorem}

\begin{proof}\leanok
    Each local interaction is an orthogonal projection and depends
    continuously on the normalized bond vector. The sum of its
    translates is the physical-coordinate independent-bond parent
    Hamiltonian. Its spectrum has zero as the ground value and no
    values strictly between zero and one. The direct-sum action fixes
    the bond vector, so the full Hamiltonian commutes with
    $U_g^{\otimes N}$ for every $g$.
\end{proof}

\begin{theorem}[Fixed-point path for cohomologous virtual actions]
    \label{thm:spt_cohomologous_fixed_point_path}
    \lean{MPSTensor.sptKron_eq_of_eq_smul,

    Set $D=D_0+D_1$ and $W_g=V^{0\prime}_{g^{-1}}\oplus V^1_{g^{-1}}$.

\end{theorem}

    $\omega_1$. Scalars cancel in $W^T\otimes W^{-1}$, so the first

    Theorem~\ref{thm:spt_direct_sum_fixed_point_path} now applies to

\end{proof}

\begin{theorem}[Weighted matrix-unit MPS as a product of bonds]
    \label{thm:spt_weighted_matrix_unit_bond_product}
    \lean{MPSTensor.mpv_weightedMatrixUnitInterpolation_eq_incomingBondProduct,
        MPSTensor.mpv_weightedMatrixUnitInterpolation_zero,
        MPSTensor.mpv_weightedMatrixUnitInterpolation_one,

        MPSTensor.weightedMatrixUnitInterpolation_isOnSiteSymmetric}
    \leanok
    \uses{thm:spt_normalized_bond_interpolation,
        def:spt_incoming_bond_coordinates,
        thm:spt_bond_interpolation_symmetry}
    Let $N>0$ and $D=D_0+D_1$. Define an MPS tensor with physical
    letters $(a,b)$ by
    $A_\gamma^{(a,b)}=\widehat W(\gamma)_{aa}|a\rangle\langle b|$.
    For a cyclic physical configuration $s=((L_i,R_i))_i$, its matrix
    product coefficient equals the product of the incoming-bond
    amplitudes:
    \begin{align}
        \operatorname{tr}\!\left(\prod_{i=1}^N
        A_\gamma^{(L_i,R_i)}\right)
         & =\prod_{i=1}^N
        \widehat W(\gamma)_{R_{i-1},L_i}.
        \notag
    \end{align}
    At $\gamma=0,1$ this identifies the positive-length physical states
    with the embedded endpoint bond-product states, with every nonzero
    incoming bond coefficient equal to $D_p^{-1/2}$ in endpoint block $p$.
    The path of tensors is continuous and symmetric under the physical
    action of the SPT fixed-point tensor of $V^{(0)}\oplus V^{(1)}$ with
    trivial character, which does not depend on $\gamma$
    \cite[Section~II.F.2]{Schuch2011Phases}. The tensor itself
    may have letters with cross-block indices at an endpoint; the cyclic
    coefficient identity is the relevant endpoint statement.
\end{theorem}

\begin{proof}\leanok
    The trace of a cyclic product of matrix units vanishes unless each
    column index equals the next row index. On that support, it is the
    product of the diagonal weights. Regrouping the cyclic indices gives
    the displayed product of bond coefficients. The diagonal weight
    matrix commutes with the direct-sum virtual action, so the symmetry
    relation of the matrix units passes to $A_\gamma$.
\end{proof}

\begin{theorem}[Canonical parent interaction of the SPT fixed point]
    \label{thm:spt_fixed_point_two_site_bond_kernel}

    fixed-point tensor, and set
    $\eta=D^{-1/2}\sum_{a=1}^D|a,a\rangle$. Then $\|\eta\|=1$.
    For every boundary matrix $X$, the two-site open-boundary vector
    $\Psi_X$ has coefficients
    $\Psi_X(a,b,c,d)=D^{-1}\delta_{bc}X_{da}$ and is annihilated by
    the interaction $h=I\otimes(I-|\eta\rangle\langle\eta|)\otimes I$

    $h$ equals the canonical two-site parent interaction of $A$

\end{theorem}

\begin{proof}\leanok
    Multiplication of two letters gives
    $A^{(a,b)}A^{(c,d)}=D^{-1}\delta_{bc}|a\rangle\langle d|$.

    The middle-register vector is $\sum_b|b,b\rangle=\sqrt D\,\eta$,
    which lies in the kernel of $I-|\eta\rangle\langle\eta|$.
    Conversely, for an arbitrary two-site vector $f$, set
    $X_{da}=\sum_x f(a,x,x,d)$. Direct contraction gives
    $(I-h)f=\Psi_X$. Thus $\ker h=G_2(A)$.
    Both interactions are orthogonal projections with this kernel,
    so they coincide.
\end{proof}

\begin{theorem}[Contraction of a middle-bond operator]
    \label{thm:spt_middle_bond_contraction}

\begin{proof}\leanok
    The identity factors constrain the exterior summation indices to
    $a,d$, leaving the middle-index sum. For separable coefficients,
    factor out $F_{ad}$. Finally,
    $(I-|\eta\rangle\langle\eta|)\eta=(1-\|\eta\|^2)\eta=0$.

\begin{definition}[Symmetric paths between weighted canonical parents]

\begin{proof}\leanok

    comparison retains this gap. The tensor covariance gives symmetry of
    the canonical parent, and both local terms are orthogonal projections
    of norm at most one. Concatenation preserves a positive uniform gap.
    This uses the canonical projections only at fixed values of $t$;
    their continuity as the bond coefficients vanish is unnecessary.
\end{proof}

\subsection{Degeneracy of the entanglement spectrum}
\label{sec:spt_entanglement_spectrum}

For an injective tensor in canonical form, the positive fixed point of the
transfer map is the half-chain reduced density matrix, whose spectrum is the
entanglement spectrum.  A symmetry forces this fixed point to commute with the
virtual gauges, and a projective representation whose factor system has a
non-trivial class, and hence no one-dimensional invariant subspace, then forces
every eigenvalue to be degenerate \cite[Section~III.A]{Cirac2021Matrix}.  Here the
canonical form is $\E_A(\Id)=\Id$ and the fixed point is that of the adjoint
map $\E_A^\dagger$; the reflected normalization, in which the fixed point of
$\E_A$ itself is used, describes the same spectrum.

\begin{lemma}[Anticommuting invertible operators force even dimension]
    \label{lem:even_finrank_of_anticommute}
    \lean{Module.End.even_finrank_of_anticommute, Matrix.even_finrank_eigenspace_of_anticommute}
    \leanok
    Let $f,g$ be invertible operators on a finite-dimensional complex vector
    space $V$ with $fg=-gf$.  Then $\dim V$ is even.  Consequently, if a matrix
    $\Lambda$ commutes with two anticommuting invertible matrices $X$ and $Y$,
    every eigenspace of $\Lambda$ has even dimension.
\end{lemma}

\begin{proof}\leanok
    Taking determinants in $fg=-gf$ gives
    $\det f\det g=(-1)^{\dim V}\det g\det f$ with $\det f\det g\neq0$, so
    $(-1)^{\dim V}=1$.  For the second statement, $X$ and $Y$ map each eigenspace
    of $\Lambda$ into itself, and their restrictions are invertible and
    anticommute.
\end{proof}

\begin{theorem}[Virtual gauges fix the boundary state]
    \label{thm:virtual_gauge_fixes_boundary_state}
    \lean{MPSTensor.exists_unitary_gauge_of_symmetry, MPSTensor.commute_boundaryState_of_symmetry}
    \leanok
    \uses{def:injective}
    Let $A$ be injective with $\E_A(\Id)=\Id$, let $\Lambda$ be a positive
    definite matrix with $\E_A^\dagger(\Lambda)=\Lambda$, and let $u$ be a
    unitary with
    $\sum_j u_{ij}A^j=\zeta\,XA^iX^{-1}$ for all $i$, where $X\in\GL_{D}(\C)$
    and $\zeta\neq0$.  Then $X=c\,W$ for a nonzero scalar $c$ and a unitary $W$
    with $W\Lambda W^\dagger=\Lambda$.  In particular $X\Lambda=\Lambda X$.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:gauge_phase_unique, thm:virtual_unitary_of_twisted_gauge_phase, thm:boundary_state_invariant_of_virtual_unitary}
    Theorem~\ref{thm:virtual_unitary_of_twisted_gauge_phase}, applied to the
    physical unitary $u^\dagger$, gives a unitary $V$ and a phase $\mu$ with
    $\sum_j (u^\dagger)_{ij}A^j=\mu\,VA^iV^\dagger$, and
    Theorem~\ref{thm:boundary_state_invariant_of_virtual_unitary} gives
    $V^\dagger\Lambda V=\Lambda$ after normalizing the trace of $\Lambda$.
    Undoing the physical mixing yields $A^i=\mu\zeta\,(VX)A^i(VX)^{-1}$, so
    Lemma~\ref{lem:gauge_phase_unique} makes $VX$ a nonzero scalar.  Hence $X$
    is a multiple of $W=V^\dagger$.
\end{proof}

\begin{lemma}[No one-dimensional eigenspace for a commutant of a non-trivial projective representation]
    \label{lem:projective_commutant_eigenspace}
    \lean{TNLean.Algebra.ProjectiveRepresentation.finrank_eigenspace_ne_one_of_commute}
    \leanok
    \uses{def:projective_rep, def:scalar_cocycle, def:nontrivial_class}
    Let $\rho$ be a projective representation of $G$ on $\C^D$ whose factor system
    $\omega$ is not cohomologous to the trivial one, and let $\Lambda$ commute with
    every $\rho(g)$.  Then no eigenspace of $\Lambda$ is one-dimensional.
\end{lemma}

\begin{proof}\leanok
    Each $\rho(g)$ maps every eigenspace of $\Lambda$ into itself.  If an
    eigenspace were spanned by one vector $v$, then $\rho(g)v=c_gv$ with
    $c_g\neq0$, and $\rho(g)\rho(h)=\omega(g,h)\rho(gh)$ applied to $v$ gives
    $\omega(g,h)=c_gc_hc_{gh}^{-1}$, a coboundary.
\end{proof}

\begin{theorem}[Degeneracy of the entanglement spectrum]
    \label{thm:spt_entanglement_spectrum_degenerate}
    \lean{MPSTensor.finrank_eigenspace_ne_one_of_isNontrivialClass}
    \leanok
    \uses{def:injective, def:projective_rep, def:nontrivial_class}
    Let $A$ be injective with $\E_A(\Id)=\Id$, let $\Lambda$ be a positive
    definite matrix with $\E_A^\dagger(\Lambda)=\Lambda$, and let $U$ be a unitary
    representation of $G$ under which $A$ is symmetric with virtual projective
    representation $\rho$ up to nonzero scalars $\zeta(g)$, that is,
    $\sum_jU(g)_{ij}A^j=\zeta(g)\,\rho(g^{-1})A^i\rho(g^{-1})^{-1}$.  If the factor system of
    $\rho$ has a non-trivial class, no eigenspace of $\Lambda$ is one-dimensional:
    every eigenvalue of the entanglement spectrum is degenerate
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:virtual_gauge_fixes_boundary_state, lem:projective_commutant_eigenspace}
    Every $\rho(g)$ is the virtual gauge of $U(g^{-1})$ and commutes with $\Lambda$
    by Theorem~\ref{thm:virtual_gauge_fixes_boundary_state}; apply
    Lemma~\ref{lem:projective_commutant_eigenspace}.
\end{proof}

\begin{theorem}[Even degeneracy for anticommuting gauges]
    \label{thm:spt_entanglement_spectrum_even}
    \lean{MPSTensor.even_finrank_eigenspace_of_anticommuting_gauges}
    \leanok
    In the setting of Theorem~\ref{thm:virtual_gauge_fixes_boundary_state}, let
    $u,v$ be unitaries with virtual gauges $X,Y$, that is,
    $\sum_j u_{ij}A^j=\zeta\,XA^iX^{-1}$ and
    $\sum_j v_{ij}A^j=\eta\,YA^iY^{-1}$ with $\zeta,\eta\neq0$.  If $XY=-YX$,
    then every eigenspace of $\Lambda$ has even dimension.  This sharpens
    Theorem~\ref{thm:spt_entanglement_spectrum_degenerate} when the projective
    representation contains an anticommuting pair.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:virtual_gauge_fixes_boundary_state, lem:even_finrank_of_anticommute}
    Both gauges commute with $\Lambda$ by
    Theorem~\ref{thm:virtual_gauge_fixes_boundary_state}; apply
    Lemma~\ref{lem:even_finrank_of_anticommute}.
\end{proof}

\subsection{Time reversal, reflection, and symmetry obstructions}
\label{sec:spt_time_reversal_reflection}

Time reversal acts on an MPS tensor by $A^i\mapsto\sum_j P_{ij}\bar A^j$, with
$\bar A$ the entrywise conjugate and $P$ its on-site part, and reflection acts by
transposition $A^i\mapsto(A^i)^{T}$.  Applying either symmetry twice returns the
tensor up to the gauge $X\bar X$, which Lemma~\ref{lem:gauge_phase_unique_normal}
forces to be a scalar \cite[Section~III.A]{Cirac2021Matrix}.

\begin{lemma}[Involution index from a doubled symmetry]
    \label{lem:involution_index_of_twice}
    \lean{MPSTensor.mul_map_star_eq_one_or_neg_one_of_twice, MPSTensor.mul_map_star_eq_of_eq_smul, MPSTensor.mul_map_star_eq_of_gauges}
    \leanok
    Let $A$ be normal with $D\ge1$ and let $X$ be unitary.  If
    $A^i=c\,(X\bar X)^\dagger A^i(X\bar X)$ for all $i$, then $X\bar X=\pm\Id$.
    If two unitaries $X,X'$ satisfy
    $\zeta X^\dagger A^iX=\zeta'X'^\dagger A^iX'$ for all $i$ with $\zeta\neq0$, then
    $X\bar X=X'\bar X'$.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:gauge_phase_unique_normal}
    By Lemma~\ref{lem:gauge_phase_unique_normal} the unitary $X\bar X$ is a scalar
    $\sigma$, and $\sigma=\pm1$ because $X\bar X=\sigma$ implies
    $\bar X X=\sigma$ and $\sigma=\bar\sigma$ with $|\sigma|=1$.  In the second
    statement $X=u\,X'$ with $|u|=1$, so $X\bar X=|u|^2X'\bar X'$.
\end{proof}

\begin{theorem}[Time-reversal index]
    \label{thm:time_reversal_index}
    \lean{MPSTensor.mul_map_star_eq_one_or_neg_one_of_timeReversal_gauge, MPSTensor.mul_map_star_eq_of_timeReversal_gauges}
    \leanok
    Let $A$ be normal with $D\ge1$, let $P$ satisfy $P\bar P=\Id$, and let $X$ be
    unitary with $\sum_jP_{ij}\bar A^j=\zeta\,X^\dagger A^iX$ for all $i$.  Then
    $X\bar X=\pm\Id$, and the sign is the same for every unitary $X$ with this
    property \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:involution_index_of_twice}
    Conjugating the relation and substituting it into itself gives
    $A^i=|\zeta|^2(X\bar X)^\dagger A^i(X\bar X)$, because the doubled on-site
    action is $P\bar P=\Id$.  Apply Lemma~\ref{lem:involution_index_of_twice}; the
    phase $\zeta$ is nonzero because $A$ has a nonzero letter.
\end{proof}

\begin{theorem}[Reflection index]
    \label{thm:reflection_index}
    \lean{MPSTensor.mul_map_star_eq_one_or_neg_one_of_reflection_gauge, MPSTensor.mul_map_star_eq_of_reflection_gauges}
    \leanok
    Let $A$ be normal with $D\ge1$ and let $X$ be unitary with
    $(A^i)^{T}=\zeta\,X^\dagger A^iX$ for all $i$.  Then $X\bar X=\pm\Id$, and the
    sign is the same for every unitary $X$ with this property
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:involution_index_of_twice}
    Transposing the relation and substituting it into itself gives
    $A^i=\zeta^2(X\bar X)^\dagger A^i(X\bar X)$, since
    $X^{T}X^\dagger=(X\bar X)^\dagger$ and $(X^\dagger)^{T}=\bar X$.  Apply
    Lemma~\ref{lem:involution_index_of_twice}.
\end{proof}

\begin{theorem}[Kramers obstruction]
    \label{thm:kramers_obstruction}
    \lean{MPSTensor.not_timeReversal_gauge_of_mul_map_star_eq_neg_one, MPSTensor.pauliY_mul_map_star, MPSTensor.not_wignerTimeReversal_gauge}
    \leanok
    Let $P$ satisfy $P\bar P=-\Id$, as $\sigma_y$ does.  No normal tensor with
    $D\ge1$ admits $X\in\GL_{D}(\C)$ and $\zeta\in\C$ with
    $\sum_jP_{ij}\bar A^j=\zeta\,X^{-1}A^iX$ for all $i$.  In particular no normal
    spin-$\tfrac12$ tensor carries the Wigner time reversal
    $A^i\mapsto\sum_j(\sigma_y)_{ij}\bar A^j$ in this form
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:gauge_phase_unique_normal}
    Applying the relation twice gives
    $-A^i=|\zeta|^2(X\bar X)^{-1}A^i(X\bar X)$.  By
    Lemma~\ref{lem:gauge_phase_unique_normal}, $-|\zeta|^2=1$, which is impossible.
\end{proof}

\begin{theorem}[Kramers obstruction for injective tensors]
    \label{thm:kramers_obstruction_phase}
    \lean{MPSTensor.not_forall_mpv_wignerTimeReversal_eq_smul_of_isInjective}
    \leanok
    \uses{thm:kramers_obstruction, thm:ft_single_phase}
    No injective spin-$\tfrac12$ tensor with $D\ge1$ is invariant up to a phase
    under the Wigner time reversal: it is not the case that for every $N\ge1$ the
    matrix product vector of $\tilde A^i=\sum_j(\sigma_y)_{ij}\bar A^j$ is
    $c_N\ket{\psi_N(A)}$ with $|c_N|=1$ \cite[Section~III.A]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:ft_single_phase}, $\tilde A^i=\lambda\,XA^iX^{-1}$ for an
    invertible $X$ and a phase $\lambda$, which Theorem~\ref{thm:kramers_obstruction}
    excludes.
\end{proof}

\begin{theorem}[Anticommuting physical symmetries are obstructed]
    \label{thm:anticommuting_physical_obstruction}
    \lean{MPSTensor.not_anticommuting_gauges_of_isNormal}
    \leanok
    Let $P,Q$ be physical matrices with $PQ=-QP$.  No normal tensor with $D\ge1$
    satisfies $\sum_jP_{ij}A^j=\zeta\,XA^iX^{-1}$ and
    $\sum_jQ_{ij}A^j=\eta\,YA^iY^{-1}$ for all $i$, with $X,Y\in\GL_{D}(\C)$ and
    $\zeta,\eta\neq0$.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:gauge_phase_unique_normal}
    Composing the two twists in both orders gives
    $\eta\zeta\,(YX)A^i(YX)^{-1}=-\zeta\eta\,(XY)A^i(XY)^{-1}$, so
    Lemma~\ref{lem:gauge_phase_unique_normal} would give $-1=1$.
\end{proof}

\begin{theorem}[No normal spin-$\tfrac12$ tensor is $\mathrm{SU}(2)$ symmetric]
    \label{thm:spin_half_su2_obstruction}
    \lean{MPSTensor.not_forall_mpv_twistedTensor_eq_smul_specialUnitaryGroup_of_isNormal, MPSTensor.not_isOnSiteSymmetric_specialUnitaryGroup_of_isNormal, MPSTensor.mpv_twistedTensor, MPSTensor.mpv_twistedTensor_eq_of_mul_eq, MPSTensor.blockKron_smul}
    \leanok
    \uses{def:on_site_symmetric}
    No normal tensor $A$ on a spin-$\tfrac12$ site with $D\ge1$ is invariant up to
    phases under the defining representation of $\mathrm{SU}(2)$: it is not the
    case that for every $g$ and every $N$ the vector $U(g)^{\otimes N}\ket{\psi_N(A)}$
    is a multiple of $\ket{\psi_N(A)}$ \cite[Section~III.A]{Cirac2021Matrix}.  In
    particular $A$ has no MPV-family invariance under $\mathrm{SU}(2)$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:on_site_symmetric_block_tensor, lem:gauge_equiv_twisted, thm:anticommuting_physical_obstruction}
    Fix a length $N$.  If $\ket{\psi_N(A)}=0$, every twist leaves it
    unchanged.  Otherwise write
    $U(g)^{\otimes N}\ket{\psi_N(A)}=c_N(g)\ket{\psi_N(A)}$; then
    $c_N(gh)=c_N(g)c_N(h)$, so $c_N$ is trivial on commutators.  Thus at every
    length $N$ each commutator leaves $\ket{\psi_N(A)}$ unchanged.  Both $i\sigma_x$ and
    $i\sigma_z$ are commutators in $\mathrm{SU}(2)$:
    $i\sigma_x=(i\sigma_z)b_1(i\sigma_z)^{-1}b_1^{-1}$ and
    $i\sigma_z=(i\sigma_x)b_2(i\sigma_x)^{-1}b_2^{-1}$ with
    $b_1=\tfrac12\begin{pmatrix}1-i&1-i\\-1-i&1+i\end{pmatrix}$ and
    $b_2=\tfrac12\begin{pmatrix}1-i&1+i\\-1+i&1+i\end{pmatrix}$.  So the
    tensor has MPV-family invariance under these two elements.
    Block to an odd length $L$ at which the tensor is injective.  The blocked tensor
    is symmetric under the $L$-fold Kronecker powers, and the powers of
    $i\sigma_x$ and $i\sigma_z$ anticommute because $L$ is odd.  The fundamental
    theorem supplies virtual gauges for both, which
    Theorem~\ref{thm:anticommuting_physical_obstruction} rules out.  The source
    argues instead through the alternation of integer and half-integer virtual
    spins.
\end{proof}
