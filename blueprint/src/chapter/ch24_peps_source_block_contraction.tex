\section{Joint contractions of the free source inputs}
\label{sec:peps_source_block_contraction}

Fix ordered source slots with halfspaces $U_e,V_e$ and joint register
space $\mathcal S$, as in Definition~\ref{def:peps_source_slots}.
All source and physical input coordinates in this section remain free.
This gives the contractions used in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 383--468.

% The operators below come from the specified allowed compositions.
% Constructing the affected and exterior compositions from circuit chronology,
% and deriving the corrected-subset error estimate, remain separate obligations.
% Individual prepared-matrix contraction bounds are not used to infer a
% contraction bound on their combined matrix.

\begin{definition}[Bases of source and input registers]
    \label{def:peps_source_slot_basis}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotBasis}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotBasis_apply}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.preparedInputBasis}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.preparedInputBasis_apply}
    \leanok
    \uses{def:peps_joint_source_vector}
    Choose finite orthonormal bases $(u_{e,r})_{r\in I_e}$ of $U_e$
    and $(v_{e,s})_{s\in J_e}$ of $V_e$. Let
    $\mathcal A=\prod_e(I_e\times J_e)$.
    The orthonormal basis of $\mathcal S$ indexed by $a\in\mathcal A$
    consists of
    \begin{align}
        s_a & =\Phi\bigl((u_{e,a_e^{(1)}}\otimes
        v_{e,a_e^{(2)}})_e\bigr).
        \label{eq:peps_source_slot_basis}
    \end{align}
    The empty source list has its single basis vector $1\in\C$.
    If $(f_j)_{j\in N}$ is an orthonormal basis of the original
    input memory $\mathcal H$, the resulting orthonormal basis of
    $\mathcal S\otimes\mathcal H$ is indexed by $(a,j)$ and satisfies
    \begin{align}
        s_a\otimes f_j
         & =Q_{(u_{e,a_e^{(1)}}\otimes v_{e,a_e^{(2)}})_e,\mathcal H}f_j.
        \label{eq:peps_source_input_basis}
    \end{align}
    Both formulas use the canonical associativity identifications.
\end{definition}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation_sums}
    Induct on the ordered slots. Tensor the bases of the first two
    halfspaces with the basis already constructed for the remaining
    slots, and identify a full assignment with its first component
    and its restriction to the remaining slots. Tensor products and
    isometric identifications preserve orthonormal bases. This gives
    (\ref{eq:peps_source_slot_basis}); tensoring with the input basis
    and applying the preparation identity gives
    (\ref{eq:peps_source_input_basis}).
\end{proof}

\begin{definition}[The complete free-input matrix]
    \label{def:peps_free_source_matrix}
    \lean{TNLean.PEPS.PairEffect.Word.freeSourceMatrix}
    \lean{TNLean.PEPS.PairEffect.Word.freeSourceMatrix_apply}
    \leanok
    \uses{def:peps_source_slot_basis,def:peps_prepared_matrix,def:peps_prepared_density_coefficient}
    Let $T:\mathcal S\otimes\mathcal H\to\mathcal K$ be the
    operator of a composition, and choose an orthonormal output
    basis $(g_k)_{k\in M}$. Its complete matrix $A_T$ has row index
    $k\in M$ and column index $(a,j)\in\mathcal A\times N$:
    \begin{align}
        A_T(k,(a,j)) & =\langle g_k,T(s_a\otimes f_j)\rangle
        =K_T(a)_{kj}.
        \label{eq:peps_free_source_matrix}
    \end{align}
    Here $K_T(a)$ is the actual prepared matrix of
    Definition~\ref{def:peps_prepared_density_coefficient}.
\end{definition}

\begin{definition}[Source columns on a prescribed pure input]
    \label{def:peps_pure_source_matrix}
    \lean{TNLean.PEPS.PairEffect.Word.pureSourceMatrix}
    \leanok
    \uses{def:peps_source_slot_basis,def:peps_free_source_matrix}
    For a prescribed vector $\psi\in\mathcal H$, let $A_{T,\psi}$
    have row index $k\in M$ and column index $a\in\mathcal A$, with
    \begin{align}
        A_{T,\psi}(k,a)
         & =\langle g_k,T(s_a\otimes\psi)\rangle.
        \label{eq:peps_pure_source_matrix}
    \end{align}
    No normalization of $\psi$ is required.
\end{definition}

\begin{theorem}[Pure-input prepared density coefficients]
    \label{thm:peps_pure_source_density_coefficient}
    \lean{TNLean.PEPS.PairEffect.Word.preparedDensityCoefficient_pure_eq_vecMulVec}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient}
    Let $T,T':\mathcal S\otimes\mathcal H\to\mathcal K$ be two
    compositions, expressed in the same output basis. For independently
    chosen $a,b\in\mathcal A$, their prepared matrices satisfy
    \begin{align}
        K_T(a)|\psi\rangle\langle\psi|K_{T'}(b)^\dagger
         & =A_{T,\psi}(:,a)A_{T',\psi}(:,b)^\dagger.
        \label{eq:peps_pure_source_density_coefficient}
    \end{align}
    Neither normalization nor allowedness is required. The identity
    includes the zero input vector and the empty source list.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_prepared_vector_density,def:peps_source_slot_basis,
        thm:peps_source_preparation_sums}
    The preparation identity gives
    $K_T(a)[\psi]=[T(s_a\otimes\psi)]=A_{T,\psi}(:,a)$.
    Apply Theorem~\ref{thm:peps_prepared_vector_density} to the ket
    and bra preparations to obtain
    (\ref{eq:peps_pure_source_density_coefficient}).
\end{proof}

\begin{definition}[The product source matrix]
    \label{def:peps_product_source_matrix}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.productSourceMatrix}
    \leanok
    \uses{def:peps_source_inventory}
    Let $K_e$ be sets indexed by the ordered source positions, and put
    $\mathcal A_K=\prod_e K_e$. For matrices
    $X_e\in\C^{K_e\times K_e}$, define
    $\mathcal X\in\C^{\mathcal A_K\times\mathcal A_K}$ by
    \begin{align}
        \mathcal X(\alpha,\beta)
         & =\prod_e X_e(\alpha_e,\beta_e).
        \label{eq:peps_product_source_matrix}
    \end{align}
    The product over an empty source list is one.
\end{definition}

\begin{theorem}[Finite contraction of source columns]
    \label{thm:peps_source_column_contraction}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.sourceContraction_columns_eq_mul}
    \leanok
    \uses{def:peps_product_source_matrix}
    Suppose each $K_e$ is finite. Let $\mathcal R,\mathcal T$ be
    arbitrary sets, and let
    $L\in\C^{\mathcal R\times\mathcal A_K}$ and
    $M\in\C^{\mathcal T\times\mathcal A_K}$.
    For $H(\alpha,\beta)=L(:,\alpha)M(:,\beta)^\dagger$, the
    contraction of (\ref{eq:peps_source_contraction}) satisfies
    \begin{align}
        \mathcal C_H(X) & =L\mathcal X M^\dagger.
        \label{eq:peps_source_column_contraction}
    \end{align}
    The matrices $X_e$ are arbitrary complex matrices; no positivity
    or nonvanishing condition is required.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_product_source_matrix}
    Pairing the ket and bra assignments at each source position
    identifies them with $(\alpha,\beta)\in\mathcal A_K^2$.
    The $(r,s)$ entry of the contraction is
    \begin{align*}
        \sum_{\alpha,\beta}
        L(r,\alpha)\overline{M(s,\beta)}
        \prod_e X_e(\alpha_e,\beta_e)
         & =\sum_{\beta,\alpha}
        L(r,\alpha)\mathcal X(\alpha,\beta)
        \overline{M(s,\beta)}
        =(L\mathcal X M^\dagger)(r,s).
    \end{align*}
    The sums are finite, so their order may be interchanged.
\end{proof}

\begin{theorem}[Contraction on a prescribed pure input]
    \label{thm:peps_pure_source_contraction}
    \lean{TNLean.PEPS.PairEffect.Word.sourceContraction_preparedDensityCoefficient_pure_eq_mul}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient,
        def:peps_product_source_matrix}
    In the source bases of Definition~\ref{def:peps_source_slot_basis},
    take $K_e=I_e\times J_e$ and let $X_e$ be arbitrary complex
    matrices on $K_e$. For the actual prepared coefficients
    $H(\alpha,\beta)=K_T(\alpha)|\psi\rangle\langle\psi|
        K_{T'}(\beta)^\dagger$, one has
    \begin{align}
        \mathcal C_H(X)
         & =A_{T,\psi}\mathcal X A_{T',\psi}^\dagger.
        \label{eq:peps_pure_source_contraction}
    \end{align}
    The two compositions are independent, and $\psi$ need not be
    normalized. Empty source lists and zero input vectors are included.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_pure_source_density_coefficient,
        thm:peps_source_column_contraction}
    Substitute (\ref{eq:peps_pure_source_density_coefficient}) into
    each prepared coefficient, then apply
    (\ref{eq:peps_source_column_contraction}).
\end{proof}

\begin{theorem}[Physical entries of a pure prepared coefficient]
    \label{thm:peps_pure_source_physical}
    \lean{TNLean.PEPS.PairEffect.Word.preparedDensityCoefficient_pure_partialTrace_apply}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient}
    Suppose the output consists of a physical memory followed by a
    discarded memory. Choose finite orthonormal bases $(h_r)_{r\in X}$ and
    $(t_z)_{z\in D}$ of these memories, and use their tensor product
    as the output basis in (\ref{eq:peps_pure_source_matrix}).
    For independent source assignments $\alpha,\beta\in\mathcal A$,
    put $H_{\alpha,\beta}=K_T(\alpha)|\psi\rangle\langle\psi|
        K_{T'}(\beta)^\dagger$. Then
    \begin{align}
        \bigl(\tr_D H_{\alpha,\beta}\bigr)(r,s)
         & =\sum_{z\in D}
        A_{T,\psi}((r,z),\alpha)
        \overline{A_{T',\psi}((s,z),\beta)}.
        \label{eq:peps_pure_source_physical}
    \end{align}
    No normalization or positivity hypothesis is imposed.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_pure_source_density_coefficient}
    Take the $((r,z),(s,z))$ entry of
    (\ref{eq:peps_pure_source_density_coefficient}) and sum over
    the common discarded index $z$.
\end{proof}

\begin{theorem}[Mixed products of party matrices]
    \label{thm:peps_party_product_conjugation}
    \lean{TNLean.PEPS.PairEffect.partyProductMatrix_conjTranspose,
        TNLean.PEPS.PairEffect.partyProductMatrix_conjugate}
    \leanok
    \uses{def:peps_dependent_physical_product}
    Let $P$ be finite, let $I_p$ be finite input alphabets, and let
    $O_p$ be output alphabets. For matrices
    $A_p,B_p\in\C^{O_p\times I_p}$ and
    $X_p\in\C^{I_p\times I_p}$, one has
    \begin{align}
        \left(\bigotimes_p B_p\right)^\dagger
         & =\bigotimes_p B_p^\dagger,         \\
        \left(\bigotimes_p A_p\right)
        \left(\bigotimes_p X_p\right)
        \left(\bigotimes_p B_p\right)^\dagger
         & =\bigotimes_p (A_pX_pB_p^\dagger).
        \label{eq:peps_party_product_conjugation}
    \end{align}
    The local matrices are arbitrary; the middle global matrix is
    their prescribed product $\bigotimes_p X_p$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dependent_product_calculus}
    Complex conjugation preserves finite products, giving the adjoint
    identity entrywise. Coordinatewise matrix multiplication of
    Theorem~\ref{thm:peps_dependent_product_calculus}, applied twice,
    gives
    \begin{align}
        \left(\bigotimes_p A_p\right)\left(\bigotimes_p X_p\right)
         & =\bigotimes_p(A_pX_p),            \\
        \left(\bigotimes_p(A_pX_p)\right)\left(\bigotimes_p B_p\right)^\dagger
         & =\bigotimes_p(A_pX_pB_p^\dagger),
    \end{align}
    where the second line also uses the adjoint identity.
\end{proof}

\begin{theorem}[Tracing the private coordinates at each party]
    \label{thm:peps_party_product_partial_trace}
    \lean{TNLean.PEPS.PairEffect.partialTraceRight_partyProductMatrix,
        TNLean.PEPS.PairEffect.partialTraceRight_partyProduct_conjugate}
    \leanok
    \uses{def:peps_dependent_physical_product,
        thm:peps_party_product_conjugation}
    Let $P$ be finite, with physical alphabets $H_p$ and finite private
    alphabets $D_p$. For arbitrary matrices
    $M_p\in\C^{(H_p\times D_p)\times(H_p\times D_p)}$, identify
    $\prod_p(H_p\times D_p)$ with
    $(\prod_pH_p)\times(\prod_pD_p)$. Then
    \begin{align}
        \tr_{\prod_pD_p}\left(\bigotimes_p M_p\right)
         & =\bigotimes_p\tr_{D_p}(M_p).
        \label{eq:peps_party_product_partial_trace}
    \end{align}
    In particular, if $O_p=H_p\times D_p$ in
    (\ref{eq:peps_party_product_conjugation}), tracing the mixed
    product gives $\bigotimes_p\tr_{D_p}(A_pX_pB_p^\dagger)$.
    These identities include empty index sets.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_product_conjugation}
    The $(a,b)$ entry of the partial trace is
    \begin{align*}
        \sum_{(z_p)_p}\prod_p M_p((a_p,z_p),(b_p,z_p))
         & =\prod_p\sum_{z_p\in D_p}
        M_p((a_p,z_p),(b_p,z_p)),
    \end{align*}
    by finite distributivity. For the mixed product, first apply
    (\ref{eq:peps_party_product_conjugation}) and then this identity.
\end{proof}

\begin{theorem}[Absorbing a common scalar at one party]
    \label{thm:peps_party_product_scalar}
    \lean{TNLean.PEPS.PairEffect.partyProductMatrix_smul_at_party}
    \leanok
    \uses{def:peps_dependent_physical_product}
    Let $P$ be finite and choose $p_0\in P$. For any family of
    rectangular matrices $M_p$ and any $\mu\in\C$, put
    $c_p=\mu$ when $p=p_0$ and $c_p=1$ otherwise. Then
    \begin{align}
        \bigotimes_p(c_pM_p)
         & =\mu\bigotimes_pM_p.
        \label{eq:peps_party_product_scalar}
    \end{align}
    The scalar is absorbed once, including when it vanishes.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_dependent_physical_product}
    At each matrix entry,
    $\prod_pc_pM_p(x_p,y_p)=\bigl(\prod_pc_p\bigr)\prod_pM_p(x_p,y_p)
        =\mu\prod_pM_p(x_p,y_p)$, since $c_{p_0}=\mu$ and $c_p=1$ for
    $p\ne p_0$.
\end{proof}

\begin{theorem}[Joint contraction of all source inputs]
    \label{thm:peps_free_source_contraction}
    \lean{TNLean.PEPS.PairEffect.Word.norm_freeSourceMatrix_le_one}
    \leanok
    \uses{def:peps_free_source_matrix}
    If $T$ is an allowed composition, then $\|A_T\|\le1$ in the
    Euclidean operator norm. This bounds its action on arbitrary
    superpositions of the source and physical input coordinates,
    independently of the number of slots and their dimensions.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_party_layout,def:peps_source_slot_basis}
    The complete matrix is the matrix of the same operator $T$ in
    orthonormal input and output bases. Coordinate maps preserve
    norms, so $\|A_T\|=\|T\|\le1$.
\end{proof}

\begin{theorem}[The bra-adjoint--ket contraction]
    \label{thm:peps_free_source_overlap}
    \lean{TNLean.PEPS.PairEffect.Word.freeSourceMatrix_conjTranspose_mul_apply}
    \lean{TNLean.PEPS.PairEffect.Word.norm_freeSourceMatrix_conjTranspose_mul_le_one}
    \leanok
    \uses{def:peps_free_source_matrix}
    Let $T$ and $T'$ be compositions with a common output
    memory and output basis. Their source lists, source spaces,
    source bases, and physical input memories and bases may differ.
    Write $A=A_T$ and $B=A_{T'}$, and let $a,b$ denote the respective
    source assignments. Then the rectangular matrix $O=B^\dagger A$
    satisfies
    \begin{align}
        O_{(b,j),(a,i)} & =(K_{T'}(b)^\dagger K_T(a))_{ji}.
        \label{eq:peps_free_source_overlap}
    \end{align}
    Thus the bra source assignment and its physical input index form
    the row index of $O$. If both compositions are allowed, then
    $\|O\|\le1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_free_source_contraction}
    Multiplying the matrices in (\ref{eq:peps_free_source_matrix})
    gives (\ref{eq:peps_free_source_overlap}). The norm bound follows
    from $\|B^\dagger A\|\le\|B^\dagger\|\|A\|=\|B\|\|A\|\le1$.
\end{proof}

\begin{theorem}[Tracing a prepared density coefficient]
    \label{thm:peps_free_source_trace}
    \lean{TNLean.PEPS.PairEffect.Word.trace_preparedMatrix_mul_conjTranspose_eq}
    \lean{TNLean.PEPS.PairEffect.Word.trace_preparedDensityCoefficient_eq_freeSourceMatrix}
    \leanok
    \uses{def:peps_free_source_matrix,def:peps_prepared_density_coefficient}
    With the ket and bra input coordinates as above, let
    $\rho\in\C^{N\times N'}$ be any rectangular matrix. No
    contraction assumption on $T$ or $T'$ is needed for the identity
    \begin{align}
        \tr\bigl(K_T(a)\rho K_{T'}(b)^\dagger\bigr)
         & =\sum_{i\in N}\sum_{j\in N'}
        \rho_{ij}\,O_{(b,j),(a,i)}.
        \label{eq:peps_free_source_trace}
    \end{align}
    When the two branches have common source spaces and physical
    input coordinates, this is the trace of
    $C_{T,T'}(\rho;a,b)$ from
    Definition~\ref{def:peps_prepared_density_coefficient}, with
    independent endpoint bases on the two sides.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_free_source_overlap}
    Cyclicity of the rectangular trace gives
    $\tr(K_T(a)\rho K_{T'}(b)^\dagger)
        =\tr(\rho K_{T'}(b)^\dagger K_T(a))$.
    Expand the trace and substitute the block identity
    (\ref{eq:peps_free_source_overlap}).
\end{proof}

\begin{definition}[Pair labels of the actual slots]
    \label{def:peps_source_pair_index}

    $\{\{p,q\}:p,q\in P,\ p\ne q\}$.

\begin{theorem}[Number of pair-source slots]
    \label{thm:peps_source_pair_count}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.length_eq_choose}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.length_le_card_sq}
    \leanok
    \uses{def:peps_source_pair_index}
    If $P$ is finite and $R$ has the pair coverage and distinctness
    in Definition~\ref{def:peps_source_pair_index}, then
    $|R|=\binom{|P|}{2}\le|P|^2$.
    This includes empty and singleton party sets.
    These are the pair slots used in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 565--585.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_source_pair_index}
    Count the unordered pairs of distinct elements under the stated
    bijection. Their number is $\binom{|P|}{2}$, which is at most
    $|P|^2$.
\end{proof}
\begin{definition}[Local maps on source slots]
    \label{def:peps_source_slot_maps}
    \lean{TNLean.PEPS.PairEffect.Word.mapSourceSlots}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_layout}
    For each slot $e$, let $L_e:U_e\to U'_e$ and
    $M_e:V_e\to V'_e$ be continuous linear maps. Apply these maps
    to the corresponding source registers and leave the original
    input memory $\mathcal H$ unchanged. Denote the resulting
    composition by $F_{L,M}$. This composition depends on the maps
    and the ordered slots, but not on the prepared source vectors.
\end{definition}

\begin{theorem}[Action of the local source maps]
    \label{thm:peps_source_slot_maps}
    \lean{TNLean.PEPS.PairEffect.Word.mapSourceSlots_spec}
    \lean{TNLean.PEPS.PairEffect.Word.eval_mapSourceSlots_prepareSlots}
    \leanok
    \uses{def:peps_source_slot_maps,def:peps_source_slots}
    For every family $\eta_e\in U_e\otimes V_e$,
    \begin{align}
        F_{L,M}Q_{\eta,\mathcal H}
         & =Q_{((L_e\otimes M_e)\eta_e)_e,\mathcal H}.
        \label{eq:peps_source_slot_maps}
    \end{align}
    If $\|L_e\|,\|M_e\|\le1$ for every slot, the composition
    $F_{L,M}$ is allowed and contains no source preparations.
    The identity requires neither these norm bounds nor normalization
    of the vectors $\eta_e$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_map_source_pair,thm:peps_source_preparation}
    Induct on the ordered source slots. Apply the two endpoint maps
    of the first source, and then apply the construction to the
    remaining sources while leaving those first two registers fixed.
    Moving the remaining preparation past these fixed registers gives
    (\ref{eq:peps_source_slot_maps}). Each constituent is a local map,
    and its allowedness follows from the corresponding endpoint norm
    bound.
\end{proof}

\begin{definition}[The complete matrix in source frames]
    \label{def:peps_source_frame_matrix}
    \lean{TNLean.PEPS.PairEffect.Word.sourceFrameMatrix}
    \lean{TNLean.PEPS.PairEffect.Word.sourceFrameMatrix_eq_freeSourceMatrix}
    \leanok
    \uses{def:peps_source_slot_maps,def:peps_free_source_matrix}
    Let $T:\mathcal S'\otimes\mathcal H\to\mathcal K$ be a
    composition on the target halfspaces $U'_e,V'_e$. Use the finite
    orthonormal bases of $U_e,V_e$ and the physical input and output
    bases already specified. Define
    \begin{align}
        A_T^{L,M}(k,(a,j))
         & =\left\langle g_k,
        TQ_{(L_eu_{e,a_e^{(1)}}\otimes
                M_ev_{e,a_e^{(2)}})_e,\mathcal H}f_j\right\rangle.
        \label{eq:peps_source_frame_matrix}
    \end{align}
    This is the complete free-input matrix of $T F_{L,M}$:
    \begin{align}
        A_T^{L,M} & =A_{TF_{L,M}}.
        \label{eq:peps_source_frame_matrix_composition}
    \end{align}
    The images of the endpoint maps need not span the target
    halfspaces.
\end{definition}

\begin{proof}\leanok
    \uses{thm:peps_source_slot_maps}
    Evaluate (\ref{eq:peps_source_slot_maps}) on each elementary
    endpoint-basis preparation and each physical input basis vector.
    The resulting columns agree with
    (\ref{eq:peps_source_frame_matrix}).
\end{proof}

\begin{theorem}[Joint contraction in source frames]
    \label{thm:peps_source_frame_contraction}
    \lean{TNLean.PEPS.PairEffect.Word.norm_sourceFrameMatrix_le_one}
    \leanok
    \uses{def:peps_source_frame_matrix}
    Assume that $T$ is allowed and that each $L_e$ and $M_e$ is a
    contraction. Then $\|A_T^{L,M}\|\le1$ in the Euclidean operator norm.
    In particular, this applies to isometric inclusions of proper
    Schmidt subspaces. No orthonormal basis of the target halfspaces
    is required.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_slot_maps,thm:peps_free_source_contraction}
    Each $L_e$ and $M_e$ is a contraction, so the composition
    $TF_{L,M}$ is allowed. By (\ref{eq:peps_source_frame_matrix_composition})
    and Theorem~\ref{thm:peps_free_source_contraction},
    \begin{align*}
        \|A_T^{L,M}\| & =\|A_{TF_{L,M}}\|\le1.
    \end{align*}
\end{proof}

\begin{theorem}[Contraction through the actual source columns]
    \label{thm:peps_actual_pure_source_contraction}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sourceContraction_preparedDensityCoefficient_pure_eq_mul}
    \uses{def:peps_actual_source_column_matrix,def:peps_product_source_matrix,
        thm:peps_pure_source_density_coefficient}
    Let $\psi$ be the recorded private product input, let $\xi,\zeta$
    be independent choices of gate monomials, and choose finite orthonormal
    bases of the private input and complete output memories. For source
    labels $x,y\in I_W$, let $C_{\xi,\zeta}(x,y)$ be the prepared output
    coefficient of $\ketbra{\psi}{\psi}$ for the canonical remaining
    operations $V_\xi,V_\zeta$ and elementary source vectors $b_x,b_y$.
    For arbitrary complex matrices $X_e$ on the endpoint-coordinate pairs,
    put $\mathcal X_{x,y}=\prod_e(X_e)_{x_e,y_e}$.
    The literal column matrices satisfy
    \begin{align}
        \sum_{x,y\in I_W}\mathcal X_{x,y}C_{\xi,\zeta}(x,y)
         & =M_\xi\mathcal X M_\zeta^\dagger.
        \label{eq:peps_actual_pure_source_contraction}
    \end{align}
    This is the actual-source specialization of the coordinate contraction
    in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 137--139, 279--355 and 409--434.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_actual_private_preparation,
        thm:peps_pure_source_density_coefficient,thm:peps_source_column_contraction}
    The action of the recorded preparation identifies the column indexed by
    $x$ with the output coordinates of $V_\xi(b_x\otimes\psi)$.
    The pure-input coefficient is therefore the outer product of column
    $x$ of $M_\xi$ and column $y$ of $M_\zeta$.
    The finite column-contraction identity gives
    (\ref{eq:peps_actual_pure_source_contraction}).
\end{proof}

\begin{definition}[Source slots as participant pairs]
    \label{def:peps_original_source_pair_indices}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sourcePairEquiv}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sourcePairEquiv_apply}
    \uses{def:peps_source_pair_index,def:compression_operator_samples}
    For each gate occurrence, its recorded source slots are in bijection with
    the unordered pairs of distinct participants of that occurrence. The slot
    corresponding to a pair has the original two endpoint owners. The order of
    these endpoints, their two Euclidean spaces, and the source vector assigned
    to the slot are retained independently of this unordered pair index.
    This is an auxiliary identification in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 233--245, 342--364 and 568--588.
\end{definition}
\begin{proof}
    The recorded slots contain each unordered pair of distinct local participants
    exactly once. The embedding of the participants into the original party set
    is injective, and therefore gives a bijection onto the pairs in its image.
    Composing these two bijections gives the assertion. Under chronological
    composition, the occurrence tags distinguish the two constituent circuits;
    framing leaves the occurrence and its slot unchanged.
\end{proof}

\begin{definition}[Sample positions of the original circuit]
    \label{def:peps_original_source_sample_indices}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sourceSampleEquiv}
    \uses{def:peps_original_source_pair_indices,def:compression_operator_samples}
    Identify an original source occurrence with the sample position consisting
    of its nonempty-participant gate occurrence and its unordered endpoint pair.
    A gate with no participants or only one participant contributes no sample
    position. The inverse identification recovers the original recorded slot,
    with its original endpoint order and dimensions.
    This is an auxiliary identification in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 568--588.
\end{definition}
