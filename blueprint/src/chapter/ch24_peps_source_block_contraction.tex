\section{Joint contractions of the free source inputs}
\label{sec:peps_source_block_contraction}

Fix ordered source slots with halfspaces $U_e,V_e$ and joint register
space $\mathcal S$, as in Definition~\ref{def:peps_source_slots}, with free
source and physical input coordinates, following
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 383--468.

% The operators below come from the specified allowed compositions.
% Constructing the affected and exterior compositions from circuit chronology,
% and deriving the corrected-subset error estimate, remain separate obligations.
% Individual prepared-matrix contraction bounds are not used to infer a
% contraction bound on their combined matrix.

\begin{definition}[Bases of source and input registers]
    \label{def:peps_source_slot_basis}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotBasis}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotBasis_apply}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.preparedInputBasis}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.preparedInputBasis_apply}
    \leanok
    \uses{def:peps_joint_source_vector}
    For finite orthonormal bases $(u_{e,r})_{r\in I_e}$ of $U_e$ and
    $(v_{e,s})_{s\in J_e}$ of $V_e$, let $\mathcal A=\prod_e(I_e\times J_e)$.
    The orthonormal basis $(s_a)_{a\in\mathcal A}$ of $\mathcal S$ is given via
    the associativity identification $\Phi$ of
    Definition~\ref{def:peps_joint_source_vector} by
    \begin{align}
        s_a & =\Phi\bigl((u_{e,a_e^{(1)}}\otimes
        v_{e,a_e^{(2)}})_e\bigr),
        \label{eq:peps_source_slot_basis}
    \end{align}
    with $s_\varnothing=1\in\C$ for the empty slot list. For an orthonormal
    basis $(f_j)_{j\in N}$ of $\mathcal H$, the product orthonormal basis of
    $\mathcal S\otimes\mathcal H$ indexed by $(a,j)\in\mathcal A\times N$
    satisfies
    \begin{align}
        s_a\otimes f_j
         & =Q_{(u_{e,a_e^{(1)}}\otimes v_{e,a_e^{(2)}})_e,\mathcal H}f_j.
        \label{eq:peps_source_input_basis}
    \end{align}
\end{definition}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation_sums}
    Induction on the ordered slots shows that tensoring the endpoint bases of
    the first slot with the inductive basis on the remaining slots preserves
    orthonormality under $\Phi$, giving~(\ref{eq:peps_source_slot_basis});
    tensoring with $(f_j)_{j\in N}$ and applying
    Lemma~\ref{thm:peps_source_preparation_sums} yields~(\ref{eq:peps_source_input_basis}).
\end{proof}

\begin{definition}[The complete free-input matrix]
    \label{def:peps_free_source_matrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.freeSourceMatrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.freeSourceMatrix_apply}
    \leanok
    \uses{def:peps_source_slot_basis,def:peps_prepared_matrix,def:peps_prepared_density_coefficient}
    For a composition operator $T:\mathcal S\otimes\mathcal H\to\mathcal K$ and
    an orthonormal output basis $(g_k)_{k\in M}$, the complete matrix
    $A_T\in\C^{M\times(\mathcal A\times N)}$ has entries
    \begin{align}
        A_T(k,(a,j)) & =\langle g_k,T(s_a\otimes f_j)\rangle
        =K_T(a)_{kj},
        \label{eq:peps_free_source_matrix}
    \end{align}
    where $K_T(a)$ is the prepared matrix of
    Definition~\ref{def:peps_prepared_density_coefficient}.
\end{definition}

\begin{definition}[Source columns on a prescribed pure input]
    \label{def:peps_pure_source_matrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.pureSourceMatrix}
    \leanok
    \uses{def:peps_source_slot_basis}
    For $\psi\in\mathcal H$, define $A_{T,\psi}\in\C^{M\times\mathcal A}$ by
    \begin{align}
        A_{T,\psi}(k,a)
         & =\langle g_k,T(s_a\otimes\psi)\rangle.
        \label{eq:peps_pure_source_matrix}
    \end{align}
\end{definition}

\begin{lemma}[Pure-input prepared density coefficients]
    \label{thm:peps_pure_source_density_coefficient}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedDensityCoefficient_pure_eq_vecMulVec}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient}
    For any compositions $T,T':\mathcal S\otimes\mathcal H\to\mathcal K$ in the
    same output basis, any $\psi\in\mathcal H$, and any $a,b\in\mathcal A$,
    \begin{align}
        K_T(a)|\psi\rangle\langle\psi|K_{T'}(b)^\dagger
         & =A_{T,\psi}(:,a)A_{T',\psi}(:,b)^\dagger.
        \label{eq:peps_pure_source_density_coefficient}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_prepared_vector_density,def:peps_source_slot_basis,
        thm:peps_source_preparation_sums}
    By Lemma~\ref{thm:peps_source_preparation_sums} and
    Definition~\ref{def:peps_source_slot_basis},
    $K_T(a)[\psi]=[T(s_a\otimes\psi)]=A_{T,\psi}(:,a)$, so
    Lemma~\ref{thm:peps_prepared_vector_density} yields~(\ref{eq:peps_pure_source_density_coefficient}).
\end{proof}

\begin{definition}[The product source matrix]
    \label{def:peps_product_source_matrix}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.productSourceMatrix}
    \leanok
    \uses{def:peps_source_inventory}
    For index sets $K_e$ at the ordered source positions,
    $\mathcal A_K=\prod_e K_e$, and matrices $X_e\in\C^{K_e\times K_e}$, define
    $\mathcal X\in\C^{\mathcal A_K\times\mathcal A_K}$ by
    \begin{align}
        \mathcal X(\alpha,\beta)
         & =\prod_e X_e(\alpha_e,\beta_e).
        \label{eq:peps_product_source_matrix}
    \end{align}
\end{definition}

\begin{lemma}[Finite contraction of source columns]
    \label{thm:peps_source_column_contraction}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.sourceContraction_columns_eq_mul}
    \leanok
    \uses{def:peps_product_source_matrix}
    For finite $K_e$, arbitrary sets $\mathcal R,\mathcal T$, matrices
    $L\in\C^{\mathcal R\times\mathcal A_K}$,
    $M\in\C^{\mathcal T\times\mathcal A_K}$, $X_e\in\C^{K_e\times K_e}$, and
    $H(\alpha,\beta)=L(:,\alpha)M(:,\beta)^\dagger$, the contraction
    (\ref{eq:peps_source_contraction}) satisfies
    \begin{align}
        \mathcal C_H(X) & =L\mathcal X M^\dagger.
        \label{eq:peps_source_column_contraction}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_product_source_matrix}
    By (\ref{eq:peps_product_source_matrix}), the $(r,s)$ entry of
    $\mathcal C_H(X)$ is
    \begin{align}
        \sum_{\alpha,\beta}
        L(r,\alpha)\overline{M(s,\beta)}
        \prod_e X_e(\alpha_e,\beta_e)
         & =\sum_{\beta,\alpha}
        L(r,\alpha)\mathcal X(\alpha,\beta)
        \overline{M(s,\beta)}
        =(L\mathcal X M^\dagger)(r,s).\notag
    \end{align}
\end{proof}

\begin{lemma}[Contraction on a prescribed pure input]
    \label{thm:peps_pure_source_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.sourceContraction_preparedDensityCoefficient_pure_eq_mul}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient,
        def:peps_product_source_matrix}
    In the bases of Definition~\ref{def:peps_source_slot_basis} with
    $K_e=I_e\times J_e$, for any $X_e\in\C^{K_e\times K_e}$, $\psi\in\mathcal H$,
    and $H(\alpha,\beta)=K_T(\alpha)|\psi\rangle\langle\psi|K_{T'}(\beta)^\dagger$,
    \begin{align}
        \mathcal C_H(X)
         & =A_{T,\psi}\mathcal X A_{T',\psi}^\dagger.
        \label{eq:peps_pure_source_contraction}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_pure_source_density_coefficient,
        thm:peps_source_column_contraction}
    Substitute (\ref{eq:peps_pure_source_density_coefficient}) into $H(\alpha,\beta)$
    and apply (\ref{eq:peps_source_column_contraction}).
\end{proof}

\begin{lemma}[Physical entries of a pure prepared coefficient]
    \label{thm:peps_pure_source_physical}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedDensityCoefficient_pure_partialTrace_apply}
    \leanok
    \uses{def:peps_pure_source_matrix,def:peps_prepared_density_coefficient}
    In a product orthonormal basis $(h_r\otimes t_z)_{(r,z)\in X\times D}$ of a
    physical and a discarded output memory, for $\alpha,\beta\in\mathcal A$ and
    $H_{\alpha,\beta}=K_T(\alpha)|\psi\rangle\langle\psi|K_{T'}(\beta)^\dagger$,
    \begin{align}
        \bigl(\tr_D H_{\alpha,\beta}\bigr)(r,s)
         & =\sum_{z\in D}
        A_{T,\psi}((r,z),\alpha)
        \overline{A_{T',\psi}((s,z),\beta)}.
        \label{eq:peps_pure_source_physical}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_pure_source_density_coefficient}
    Sum the $((r,z),(s,z))$ entry of (\ref{eq:peps_pure_source_density_coefficient})
    over $z\in D$.
\end{proof}

\begin{lemma}[Mixed products of party matrices]
    \label{thm:peps_party_product_conjugation}
    \lean{TNLean.Lattice.PartyCircuit.partyProductMatrix_conjTranspose,
        TNLean.Lattice.PartyCircuit.partyProductMatrix_conjugate}
    \leanok
    \uses{def:peps_dependent_physical_product}
    For finite $P$, finite input alphabets $I_p$, output alphabets $O_p$, and
    matrices $A_p,B_p\in\C^{O_p\times I_p}$, $X_p\in\C^{I_p\times I_p}$,
    \begin{align}
        \left(\bigotimes_p B_p\right)^\dagger
         & =\bigotimes_p B_p^\dagger,         \notag \\
        \left(\bigotimes_p A_p\right)
        \left(\bigotimes_p X_p\right)
        \left(\bigotimes_p B_p\right)^\dagger
         & =\bigotimes_p (A_pX_pB_p^\dagger).
        \label{eq:peps_party_product_conjugation}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_dependent_product_calculus}
    Multiplicativity of complex conjugation gives
    $(\bigotimes_p B_p)^\dagger=\bigotimes_p B_p^\dagger$, and applying
    Lemma~\ref{thm:peps_dependent_product_calculus} twice gives
    \begin{align}
        \left(\bigotimes_p A_p\right)\left(\bigotimes_p X_p\right)\left(\bigotimes_p B_p\right)^\dagger
         & =\left(\bigotimes_p(A_pX_p)\right)\left(\bigotimes_p B_p^\dagger\right)
        =\bigotimes_p(A_pX_pB_p^\dagger).\notag
    \end{align}
\end{proof}

\begin{lemma}[Tracing the private coordinates at each party]
    \label{thm:peps_party_product_partial_trace}
    \lean{TNLean.Lattice.PartyCircuit.partialTraceRight_partyProductMatrix,
        TNLean.Lattice.PartyCircuit.partialTraceRight_partyProduct_conjugate}
    \leanok
    \uses{def:peps_dependent_physical_product,
        thm:peps_party_product_conjugation}
    For finite $P$, physical alphabets $H_p$, finite private alphabets $D_p$,
    and $M_p\in\C^{(H_p\times D_p)\times(H_p\times D_p)}$, under
    $\prod_p(H_p\times D_p)\simeq(\prod_pH_p)\times(\prod_pD_p)$,
    \begin{align}
        \tr_{\prod_pD_p}\left(\bigotimes_p M_p\right)
         & =\bigotimes_p\tr_{D_p}(M_p).
        \label{eq:peps_party_product_partial_trace}
    \end{align}
    For $O_p=H_p\times D_p$ in (\ref{eq:peps_party_product_conjugation}),
    $\tr_{\prod_pD_p}\bigl((\bigotimes_p A_p)(\bigotimes_p X_p)(\bigotimes_p B_p)^\dagger\bigr)
        =\bigotimes_p\tr_{D_p}(A_pX_pB_p^\dagger)$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_party_product_conjugation}
    By finite distributivity, the $(a,b)$ entry of the left-hand side of
    (\ref{eq:peps_party_product_partial_trace}) is
    \begin{align}
        \sum_{(z_p)_p}\prod_p M_p((a_p,z_p),(b_p,z_p))
         & =\prod_p\sum_{z_p\in D_p}
        M_p((a_p,z_p),(b_p,z_p)),\notag
    \end{align}
    and combining (\ref{eq:peps_party_product_conjugation}) with
    (\ref{eq:peps_party_product_partial_trace}) gives the mixed-product trace.
\end{proof}

\begin{lemma}[Absorbing a common scalar at one party]
    \label{thm:peps_party_product_scalar}
    \lean{TNLean.Lattice.PartyCircuit.partyProductMatrix_smul_at_party}
    \leanok
    \uses{def:peps_dependent_physical_product}
    For finite $P$, $p_0\in P$, rectangular matrices $M_p$, and $\mu\in\C$,
    setting $c_{p_0}=\mu$ and $c_p=1$ for $p\neq p_0$ gives
    \begin{align}
        \bigotimes_p(c_pM_p)
         & =\mu\bigotimes_pM_p.
        \label{eq:peps_party_product_scalar}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_dependent_physical_product}
    Entrywise, $\prod_p c_pM_p(x_p,y_p)=(\prod_p c_p)\prod_p M_p(x_p,y_p)=\mu\prod_p M_p(x_p,y_p)$.
\end{proof}

\begin{lemma}[Joint contraction of all source inputs]
    \label{thm:peps_free_source_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_freeSourceMatrix_le_one}
    \leanok
    \uses{def:peps_free_source_matrix}
    If $T$ is an allowed composition, then $\|A_T\|\le1$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_party_layout,def:peps_source_slot_basis}
    Since $A_T$ represents $T$ in orthonormal bases
    (Definition~\ref{def:peps_source_slot_basis}), $\|A_T\|=\|T\|\le 1$.
\end{proof}

\begin{lemma}[The bra-adjoint--ket contraction]
    \label{thm:peps_free_source_overlap}
    \lean{TNLean.Lattice.PartyCircuit.Word.freeSourceMatrix_conjTranspose_mul_apply}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_freeSourceMatrix_conjTranspose_mul_le_one}
    \leanok
    \uses{def:peps_free_source_matrix}
    For compositions $T,T'$ sharing an output memory and orthonormal output
    basis, let $A=A_T$, $B=A_{T'}$, and $O=B^\dagger A$. Then
    \begin{align}
        O_{(b,j),(a,i)} & =(K_{T'}(b)^\dagger K_T(a))_{ji},
        \label{eq:peps_free_source_overlap}
    \end{align}
    and if $T,T'$ are allowed, then $\|O\|\le1$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_free_source_contraction}
    Multiplying (\ref{eq:peps_free_source_matrix}) gives
    (\ref{eq:peps_free_source_overlap}), and Lemma~\ref{thm:peps_free_source_contraction}
    gives $\|O\|\le\|B^\dagger\|\|A\|=\|B\|\|A\|\le1$.
\end{proof}

\begin{lemma}[Tracing a prepared density coefficient]
    \label{thm:peps_free_source_trace}
    \lean{TNLean.Lattice.PartyCircuit.Word.trace_preparedMatrix_mul_conjTranspose_eq}
    \lean{TNLean.Lattice.PartyCircuit.Word.trace_preparedDensityCoefficient_eq_freeSourceMatrix}
    \leanok
    \uses{def:peps_free_source_matrix,def:peps_prepared_density_coefficient}
    In the setting of Lemma~\ref{thm:peps_free_source_overlap}, for any
    $\rho\in\C^{N\times N'}$ (where $C_{T,T'}(\rho;a,b)=K_T(a)\rho K_{T'}(b)^\dagger$
    is the coefficient of Definition~\ref{def:peps_prepared_density_coefficient}),
    \begin{align}
        \tr\bigl(K_T(a)\rho K_{T'}(b)^\dagger\bigr)
         & =\sum_{i\in N}\sum_{j\in N'}
        \rho_{ij}\,O_{(b,j),(a,i)}.
        \label{eq:peps_free_source_trace}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_free_source_overlap}
    By trace cyclicity and (\ref{eq:peps_free_source_overlap}),
    $\tr(K_T(a)\rho K_{T'}(b)^\dagger)=\tr(\rho K_{T'}(b)^\dagger K_T(a))
        =\sum_{i,j}\rho_{ij}O_{(b,j),(a,i)}$.
\end{proof}

\begin{lemma}[Contraction through the source columns]
    \label{thm:peps_actual_pure_source_contraction}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceContraction_preparedDensityCoefficient_pure_eq_mul}
    \uses{def:peps_actual_source_column_matrix,def:peps_product_source_matrix,
        thm:peps_pure_source_density_coefficient}
    For private product input $\psi$, monomial choices $\xi,\zeta$, prepared
    output coefficients $C_{\xi,\zeta}(x,y)$ of $\ketbra{\psi}{\psi}$ for
    residuals $V_\xi,V_\zeta$ at $b_x,b_y$ ($x,y\in I_W$), and matrices $X_e$
    with $\mathcal X_{x,y}=\prod_e(X_e)_{x_e,y_e}$, the column matrices $M_\xi$
    of Definition~\ref{def:peps_actual_source_column_matrix} satisfy the
    contraction identity of \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 137--139, 279--355 and 409--434:
    \begin{align}
        \sum_{x,y\in I_W}\mathcal X_{x,y}C_{\xi,\zeta}(x,y)
         & =M_\xi\mathcal X M_\zeta^\dagger.
        \label{eq:peps_actual_pure_source_contraction}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_actual_private_preparation,
        thm:peps_pure_source_density_coefficient,thm:peps_source_column_contraction}
    By Lemmas~\ref{thm:peps_actual_private_preparation}
