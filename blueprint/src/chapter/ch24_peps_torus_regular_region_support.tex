\begin{lemma}[Single-site geometry of the native torus]%
    \label{lem:peps_torus_singleton_geometry}
    \lean{TNLean.PEPS.torusIncidentLeg,
        TNLean.PEPS.torusContiguousRectangle_one_eq_singleton,
        TNLean.PEPS.torusGraph_compl_singleton_connected,
        TNLean.PEPS.torusGraph_singleton_connected,
        TNLean.PEPS.torusIncidentLeg_surjective,
        TNLean.PEPS.torusIncidentLeg_injective,
        TNLean.PEPS.torusIncidentLegEquiv,
        TNLean.PEPS.card_regionBoundaryEdge_torus_singleton,
        TNLean.PEPS.torusGraph_connected}
    \leanok
    \uses{def:peps_torus_geometry_translation,
        thm:peps_torus_rectangle_cut_connectivity}
    On the native square torus $\Gamma=\Z_{L_x}\times\Z_{L_y}$ with
    $L_x,L_y\ge 3$ (Definition~\ref{def:peps_torus_geometry_translation}),
    $\Gamma$ is connected. For every vertex $v$, the induced subgraphs on
    $\{v\}$ and $V\setminus\{v\}$ are connected, $|\partial E_{\{v\}}|=4$,
    and the native top, right, down, and left legs bijectively enumerate
    $\operatorname{Inc}(v)=\partial E_{\{v\}}$.
    The regular virtual-space and period restrictions are recorded in
    \cite{gap:rmp_peps_quantum_double_g_isometry}.
\end{lemma}

\begin{proof}\leanok
    Since $\{v\}$ is a $1\times 1$ coordinate rectangle and $L_x,L_y\ge 3$,
    Theorem~\ref{thm:peps_torus_rectangle_cut_connectivity} gives connectivity
    of $\{v\}$, $V\setminus\{v\}$, and $\Gamma$. The four native legs exhaust
    $\operatorname{Inc}(v)$ and are distinct because horizontal and vertical
    bonds are disjoint while opposite bonds have distinct origins when
    $L_x,L_y>2$; each incident bond has one endpoint in $\{v\}$, so
    $\operatorname{Inc}(v)=\partial E_{\{v\}}$.
\end{proof}

\begin{theorem}[Single-site physical support of a regular torus PEPS]%
    \label{thm:peps_regular_torus_singleton_support}
    \lean{TNLean.PEPS.torusSingletonBoundaryEquiv,
        TNLean.PEPS.range_regionReducedDensity_torus_singleton_of_regular,
        TNLean.PEPS.rank_regionReducedDensity_torus_singleton_of_regular,
        TNLean.PEPS.stateCoeff_groupBondTensor_torus_ne_zero_of_regular,
        TNLean.PEPS.card_group_cube_le_physicalDimension_regular_torus,
        TNLean.PEPS.entropy_normalizedRegionReducedDensity_torus_singleton_of_regular}
    \leanok
    \uses{lem:peps_torus_singleton_geometry, def:peps_g_injective,
        def:peps_normalized_region_reduced_density}
    Place site-dependent regular $G$-injective tensors $A_v$ of physical
    dimension $d$ on $\Z_{L_x}\times\Z_{L_y}$ with $L_x,L_y\ge 3$. Then
    $\Psi(A)\ne 0$, and for every vertex $v$ the reduced and normalized
    densities $\rho_{\{v\}}$ and $\widehat\rho_{\{v\}}$
    (Definition~\ref{def:peps_normalized_region_reduced_density}) satisfy
    \begin{align}
        \operatorname{ran}\rho_{\{v\}} & = \mathcal G_{\{v\}},
                                       & \operatorname{rank}\rho_{\{v\}} & = |G|^3\leq d,
        \label{eq:peps_torus_singleton_support_rank}                                      \\
        0\leq S(\widehat\rho_{\{v\}})  & \leq 3\log|G|,
                                       & S_0(\widehat\rho_{\{v\}})       & = 3\log|G|.
        \label{eq:peps_torus_singleton_support_entropy}
    \end{align}
    No isometry or flat spectrum is assumed.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_regular_region_support,
        thm:peps_regular_region_dimension,
        thm:peps_regular_region_entropy_bound,
        lem:peps_torus_singleton_geometry}
    By Lemma~\ref{lem:peps_torus_singleton_geometry}, $\{v\}$ and
    $V\setminus\{v\}$ are connected with $|\partial E_{\{v\}}|-1=3$. Applying
    Theorems~\ref{thm:peps_regular_region_support},~\ref{thm:peps_regular_region_dimension},
    and~\ref{thm:peps_regular_region_entropy_bound} yields~(\ref{eq:peps_torus_singleton_support_rank})--(\ref{eq:peps_torus_singleton_support_entropy}),
    with $|G|^3=\operatorname{rank}\rho_{\{v\}}\leq d$ since $\rho_{\{v\}}$ acts on $\C^d$.
\end{proof}

\begin{theorem}[Local rigidity for equal regular torus states]%
    \label{thm:peps_regular_torus_local_rigidity}
    \lean{TNLean.PEPS.regionGroundSpace_torus_singleton_eq_of_sameState_regular,
        TNLean.PEPS.card_group_eq_of_sameState_regular_torus}
    \leanok
    \uses{lem:peps_torus_singleton_geometry, def:peps_g_injective,
        def:peps_region_ground_space}
    On $\Z_{L_x}\times\Z_{L_y}$ with $L_x,L_y\ge 3$, if regular $G$-injective
    and $H$-injective tensors $A$ and $B$ satisfy $\Psi(A)=\Psi(B)$, then
    $\mathcal G_{\{v\}}(A)=\mathcal G_{\{v\}}(B)$ for every vertex $v$ and
    $|G|=|H|$. No assertion about the multiplication laws or individual bond
    gauges follows from this rank argument.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_same_state_regular_region_spaces,
        thm:peps_regular_group_order_rigidity,
        lem:peps_torus_singleton_geometry}
    By Lemma~\ref{lem:peps_torus_singleton_geometry} and
    Theorem~\ref{thm:peps_same_state_regular_region_spaces},
    $\Psi(A)=\Psi(B)$ implies $\rho_{\{v\}}(A)=\rho_{\{v\}}(B)$ and hence
    $\mathcal G_{\{v\}}(A)=\operatorname{ran}\rho_{\{v\}}(A)=\mathcal G_{\{v\}}(B)$.
    Theorem~\ref{thm:peps_regular_group_order_rigidity} then gives
    $|G|^3=\operatorname{rank}\rho_{\{v\}}(A)=\operatorname{rank}\rho_{\{v\}}(B)=|H|^3$,
    so $|G|=|H|$.
\end{proof}

\begin{theorem}[Physical support of a bounded regular torus rectangle]%
    \label{thm:peps_regular_torus_rectangle_support}
    \lean{TNLean.PEPS.range_regionReducedDensity_torus_rectangle_of_regular}
    \leanok
    \uses{def:peps_g_injective, def:peps_region_ground_space,
        def:peps_region_reduced_density}
    For regular $G$-injective tensors on $\Z_{L_x}\times\Z_{L_y}$ with
    $L_x,L_y\ge 3$ and any coordinate rectangle $R$ of side lengths
    $1\le\ell_x<L_x$ and $1\le\ell_y<L_y$, the reduced density
    $\rho_R$ (Definition~\ref{def:peps_region_reduced_density}) satisfies
    $\operatorname{ran}\rho_R=\mathcal G_R$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_regular_region_support}
    Since $1\le\ell_x<L_x$ and $1\le\ell_y<L_y$, both $R$ and $V\setminus R$
    (via an unused row and column) are connected, so
    Theorem~\ref{thm:peps_regular_region_support} yields
    $\operatorname{ran}\rho_R=\mathcal G_R$.
\end{proof}
