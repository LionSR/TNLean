\section{The splitting criterion}%
\label{sec:asym_splitting}

Periodic vector equality permits upper-triangular nilpotent extensions between
composition factors, preventing sitewise intertwining
(Remark~\ref{rem:asym_zipper}). Theorem~\ref{thm:asym_splitting} provides
an algebraic condition ensuring that the compressions of
Theorem~\ref{thm:asym_multiblock} split into sitewise intertwiners.

\begin{theorem}[Splitting criterion]
    \label{thm:asym_splitting}
    \lean{MPSTensor.exists_multiBlockCompression_remainder_eq_zero_of_star,
        MPSTensor.exists_multiBlockCompression_remainder_eq_zero_of_isSemisimpleModule,
        MPSTensor.isSemisimpleModule_wordModule_of_conjTranspose_mem_adjoin}
    \leanok
    \uses{thm:asym_multiblock, def:asym_word_module}
    In the setting of Theorem~\ref{thm:asym_multiblock}, suppose that the
    unital algebra generated by the matrices $B^i$ is closed under conjugate
    transposition, that is, $(B^i)^\dagger$ lies in the unital algebra generated
    by the $B^j$ for every $i$. Then the word-algebra module of $B$ is
    semisimple, and $B$ admits a block triangular form as in
    Theorem~\ref{thm:asym_multiblock} whose remainder vanishes: the block
    triangular form is block diagonal.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asym_multiblock, thm:asym_flag_theorem}
    Since the algebra is closed under conjugate transposition, the
    orthogonal complement of an invariant subspace is invariant. Thus
    every invariant subspace has an invariant complement and the module
    is semisimple. Every step of the flag of
    Theorem~\ref{thm:asym_flag_theorem} then has an invariant complement,
    the module is the direct sum of these complements, and a basis
    adapted to the direct sum makes every $B^i$ block diagonal with the
    blocks prescribed by the flag.
\end{proof}

\begin{corollary}[Sitewise fusion tensors]
    \label{cor:asym_fusion_tensors_star}
    \lean{MPSTensor.exists_fusionTensors_of_star}
    \leanok
    \uses{thm:asym_splitting}
    Under the hypotheses of Theorem~\ref{thm:asym_splitting} there are
    maps $V_s$ and $W_s$ with $W_sV_t=\delta_{st}\id$, $W_sB^wV_s=C_s^w$
    for every word $w$, and the sitewise relations $B^iV_s=V_sC_s^i$ and
    $W_sB^i=C_s^iW_s$ for every physical index $i$ and block occurrence $s$.
\end{corollary}

    With vanishing remainder, $B^i=\sum_sV_sC_s^iW_s$, and the sitewise

\end{proof}

\begin{corollary}[Obstruction to closure under conjugate transposition]
    \label{cor:asym_anomaly_not_star}
    \lean{MPSTensor.not_forall_conjTranspose_mem_adjoin_of_forall_right_intertwiner_eq_zero,
        CZXCompression.czxSquare_not_starClosed}
    \leanok
    \uses{thm:asym_splitting}
    If, in the setting of Theorem~\ref{thm:asym_multiblock}, some block
    occurrence $s$ admits no nonzero sitewise right intertwiner $X$ with
    $B^iX=XC_s^i$ for all $i$, then the unital algebra generated by the
    $B^i$ is not closed under conjugate transposition. In particular, the
    stacked product of the anomalous CZX symmetry with itself, for which
    both sitewise intertwiner spaces vanish, generates a unital algebra
    that is not closed under conjugate transposition.
\end{corollary}

\begin{proof}
    \leanok
    \uses{thm:asym_splitting}
    By Theorem~\ref{thm:asym_splitting}, closure under conjugate
    transposition would give a compression pair whose right map is a
    nonzero sitewise intertwiner.
\end{proof}

\begin{remark}
    \label{rem:asym_anomaly_unitary}
    The physical operator of the CZX symmetry, a product of controlled-$Z$
    gates and single-site spin flips, is unitary at every length. The
    failure of closure under conjugate transposition in
    Corollary~\ref{cor:asym_anomaly_not_star} therefore concerns the
    virtual algebra of the stacked tensor, rather than the physical operator.
\end{remark}

\subsection{Uniqueness of split compressions}%
\label{sec:asym_zipper_uniqueness}

When the remainder vanishes, the compression maps are determined by the blocks
up to a change of basis of each multiplicity space. Throughout, the blocks are
indexed by labels $c$ with multiplicities $N_c$, and a split compression of $B$
consists of maps $V_{c,\mu}$ and $W_{c,\mu}$, for $1\leq\mu\leq N_c$, with
\begin{align}
    W_{c,\mu}V_{c',\nu} & =\delta_{cc'}\delta_{\mu\nu}\id, &
    B^i                 & =\sum_c\sum_{\mu=1}^{N_c}V_{c,\mu}C_c^iW_{c,\mu}
    \label{eq:asym_zipper_decomposition}
\end{align}
for every physical index $i$. This is the zipper condition of
\cite{Bultinck2017Anyons}: the block triangular form has no nonzero blocks
above the diagonal. The maps $V_{c,\mu}$ need not span the bond space of $B$,
so zero diagonal blocks are allowed.

\begin{lemma}[Intertwiners of normal tensors]
    \label{lem:asym_normal_intertwiner}
    \lean{MPSTensor.eq_zero_or_isGaugeRelated_of_intertwines}
    \leanok
    \uses{def:normal}
    Let $A$ and $A'$ be normal tensors of bond dimensions $D$ and $D'$, and
    let $X\in\C^{D\times D'}$ satisfy $A^iX=XA'^i$ for every physical index
    $i$. Then $X=0$, or $D=D'$ and $X$ is invertible, so that
    $A'^i=X^{-1}A^iX$ for every $i$.
\end{lemma}

\begin{proof}
    \leanok
    The kernel of $X$ is invariant under every word of $A'$. At a length where
    the words of $A'$ span the full matrix algebra, a nonzero vector in the
    kernel is carried to every vector, so $X=0$. Symmetrically, the left kernel
    of $X$ is invariant under right multiplication by the words of $A$. If
    $X\neq0$, both kernels vanish, so $D'\leq D$ and $D\leq D'$, and $X$ is
    invertible.
\end{proof}

\begin{theorem}[Uniqueness of split compressions up to multiplicity gauge]
    \label{thm:asym_zipper_uniqueness}
    \lean{MPSTensor.ZipperDecomposition.V_mul_W_eq_multiplicityGauge_smul,
        MPSTensor.ZipperDecomposition.V_mul_W_eq_zero_of_not_isGaugeRelated,
        MPSTensor.ZipperDecomposition.multiplicityGauge_mul_multiplicityGauge,
        MPSTensor.ZipperDecomposition.exists_unique_multiplicityGauge}
    \leanok
    \uses{lem:asym_normal_intertwiner}
    Let the blocks $C_c$ be normal of positive bond dimension and pairwise
    not related by a gauge transformation, and let $(V,W)$ and
    $(\widetilde V,\widetilde W)$ be two split compressions of $B$ as in
    (\ref{eq:asym_zipper_decomposition}) with the same blocks and
    multiplicities. Then $\widetilde W_{c,\mu}V_{e,\nu}=0$ for $e\neq c$, and
    $\widetilde W_{c,\mu}V_{c,\nu}=(Y_c)_{\mu\nu}\id$ for a matrix
    $Y_c\in\GL_{N_c}(\C)$ whose inverse is given by
    $W_{c,\mu}\widetilde V_{c,\nu}=(Y_c^{-1})_{\mu\nu}\id$. The matrices $Y_c$
    are the unique ones with
    \begin{align}
        \widetilde W_{c,\mu}B^i & =\sum_\nu(Y_c)_{\mu\nu}W_{c,\nu}B^i, &
        B^i\widetilde V_{c,\mu} & =\sum_\nu B^iV_{c,\nu}(Y_c^{-1})_{\nu\mu}
        \label{eq:asym_zipper_gauge}
    \end{align}
    for every physical index $i$, so the two compressions differ by
    $\bigoplus_c\GL_{N_c}(\C)$ on the support of $B$, with no dressing by
    words. This is the uniqueness of fusion tensors under the zipper
    condition in \cite{Bultinck2017Anyons}.
    % Source anchor: arXiv:1511.08090, AnyonsPEPS.tex lines 164--166 (the
    %   gauge Y) and 181--200 (uniqueness under the zipper condition);
    %   arXiv:2203.12563, lines 415--424.
    % PR #7969 plans the MPO-algebra reading as
    %   thm:mpoalg_zipper_fusion_uniqueness; after it merges, that entry should
    %   \uses this theorem and cor:asym_zipper_fusion_uniqueness.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:asym_normal_intertwiner}
    Multiplying (\ref{eq:asym_zipper_decomposition}) on the left by
    $W_{c,\mu}$ and on the right by $V_{c,\mu}$ gives the sitewise relations
    $W_{c,\mu}B^i=C_c^iW_{c,\mu}$ and $B^iV_{c,\mu}=V_{c,\mu}C_c^i$, and
    likewise for the second compression. Hence
    $C_c^i\,\widetilde W_{c,\mu}V_{e,\nu}=\widetilde W_{c,\mu}B^iV_{e,\nu}
        =\widetilde W_{c,\mu}V_{e,\nu}\,C_e^i$, and
    Lemma~\ref{lem:asym_normal_intertwiner} makes this intertwiner zero for
    $e\neq c$. For $e=c$ it commutes with every $C_c^i$, and since the words
    of $C_c$ of a suitable length span the full matrix algebra, it is a scalar
    $(Y_c)_{\mu\nu}$. Compressing $B^i$ with the second compression on both
    sides and expanding $B^i$ by the first gives
    $\delta_{\mu\kappa}C_c^i=\sum_\nu(Y_c)_{\mu\nu}(Y'_c)_{\nu\kappa}C_c^i$,
    where $W_{c,\nu}\widetilde V_{c,\kappa}=(Y'_c)_{\nu\kappa}\id$; a nonzero
    $C_c^i$ gives $Y_cY'_c=\id$. Expanding $\widetilde W_{c,\mu}B^i$ by the
    first compression gives (\ref{eq:asym_zipper_gauge}), and multiplying it
    on the right by $V_{c,\kappa}$ recovers $(Y_c)_{\mu\kappa}$, which gives
    uniqueness.
\end{proof}

\begin{corollary}[Uniqueness of zipper fusion tensors]
    \label{cor:asym_zipper_fusion_uniqueness}
    \lean{MPOTensor.exists_unique_zipperFusionGauge}

    tensors that split the stacked product $T_aT_b$ over the blocks $T_c$

    (\ref{eq:asym_zipper_decomposition}), differ by a unique
    $Y_{ab}^c\in\GL_{N_{ab}^c}(\C)$ for each $c$ in the sense of
    (\ref{eq:asym_zipper_gauge}).

\section{The single-block reduction of Moln\'ar--Ge--Schuch--Cirac}%
\label{sec:asym_single_block}

\begin{corollary}[Single-block case: the normal-target reduction]
    \label{cor:asym_single_block}
    \lean{MPSTensor.exists_isReduction_and_nilpotent_of_isNormal}
    \leanok
    \uses{def:same_mpv2_pos, def:normal, def:mps_rectangular_reduction}
    Let $A$ be a normal tensor (NT) of bond dimension $D_A\geq1$, let $B$
    be an arbitrary tensor of bond dimension $D_B$, and suppose that
    $\ket{V^{(N)}(B)}=\ket{V^{(N)}(A)}$ for every $N\geq1$.
    Then there are maps $V:\C^{D_A}\to\C^{D_B}$ and
    $W:\C^{D_B}\to\C^{D_A}$ with $WV=\id$ such that, for every $N\geq0$
    and every word of length $N$,
    \begin{align}
        WB^{i_1}\cdots B^{i_N}V & =A^{i_1}\cdots A^{i_N},
        \label{eq:asym_single_block_compression}
    \end{align}
    and the remainder $R^i=B^i-VA^iW$ generates a nilpotent non-unital
    algebra. Explicitly, for every word of length $r=D_B-D_A+1$,
    \begin{align}
        R^{i_1}R^{i_2}\cdots R^{i_r} & =0.
        \label{eq:asym_single_block_nilpotency}
    \end{align}
    In particular $D_B\geq D_A$.
\end{corollary}

    (\ref{eq:asym_single_block_compression}), clause (vii) gives
    $D_B=D_A+z\geq D_A$, and clause (vi) gives
    (\ref{eq:asym_single_block_nilpotency}) with $r=|S|+z=1+(D_B-D_A)$.
\end{proof}

The proportional case, in which $\ket{V^{(N)}(B)}=\lambda^N\ket{V^{(N)}(A)}$
for a scalar $\lambda\in\C^\times$, reduces to
Corollary~\ref{cor:asym_single_block} by replacing $B$ with
$\lambda^{-1}B$, and then $WB^{i_1}\cdots B^{i_N}V=\lambda^NA^{i_1}\cdots A^{i_N}$
with remainder $R^i=B^i-\lambda\,VA^iW$; the compression pair of that case
is recorded in Theorem~\ref{thm:mps_proportional_reduction_exists}.

Corollary~\ref{cor:asym_single_block} together with the proportional case
corresponds to Propositions~20 and~21 of~\cite{Molnar2018SemiInjective}
(where $V,W$ are swapped). The hypothesis $\lambda\neq0$ is required because
the zero tensor $B=0$ admits no compression onto a nonzero normal target.
The non-unital remainder algebra generated by the $R^i$ satisfies the
nilpotency length bound $r=D_B-D_A+1$ from Theorem~\ref{thm:asym_multiblock}.
The paper-gap note~\cite{gap:mgsc18_reduction_proportionality_scalar}
discusses the scalar normalization, and~\cite{gap:mgsc18_nilpotency_length_one_terminology}
discusses the nilpotency length terminology.
