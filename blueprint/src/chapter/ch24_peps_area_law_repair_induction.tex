\section{The uniform repair induction}
\label{sec:area_law_repair_induction}

We keep the recursive family $(\mathcal D_j)_{j\ge0}$ of
Definition~\ref{def:area_law_repair_stage}, with stopping scale $S_*$.
Distances are maximum-norm distances. The square
$v+[-r,r]^2$ is the closed ball of radius $r$ about $v$.

\begin{definition}[Ancestors and descendants]

    \lean{TNLean.PEPS.AreaLaw.Geometry.repairMarkParent,

    \leanok

    A mark occurrence $q$ of $\mathcal D_{j+1}$ is a pair of a nonterminal
    parent occurrence $p$ of $\mathcal D_j$ and an offset; write
    $\pi_j(q)=p$. For $n\ge0$ the ancestor at generation $i$ of a mark of
    $\mathcal D_{i+n}$ is
    \begin{align}
        \alpha_{i,0} & =\mathrm{id},
                     & \alpha_{i,n+1} & =\alpha_{i,n}\circ\pi_{i+n}.
        \notag
    \end{align}
    A fresh identifier of $\mathcal D_{j+1}$ is born at one parent
    occurrence of $\mathcal D_j$; retained old identifiers have no parent.
    The descendants of a mark are the marks and fresh identifiers whose
    ancestor it is. Finally, let $\mathcal E_j$ be the union of the
    closed terminal squares of the generations $i<j$.
\end{definition}

\begin{definition}[Conclusion of the uniform repair induction]
    \label{def:area_law_uniform_repair_conclusion}
    \lean{TNLean.PEPS.AreaLaw.Geometry.IsUniformRepairInduction,
        TNLean.PEPS.AreaLaw.Geometry.UniformRepairInductionStatement}
    \leanok
    \uses{def:area_law_repair_ancestors,def:area_law_repair_square,
        def:area_law_isolated_star}
    Let $a_0>0$. The recursive family satisfies the uniform repair
    induction with constant $a_0$ when, for all generations $i,j$ and
    $n\ge0$:
    \begin{enumerate}
        \item every mark $p$ of $\mathcal D_j$, of location $v_p$ and scale
              $S_p$, has an isolated star of radius $a_0S_p$ in the current
              regions of $\mathcal D_j$: inside $v_p+[-a_0S_p,a_0S_p]^2$
              the frontiers are the rays from $v_p$ in the eight axis and
              diagonal directions at which the identifier of the eight
              elementary sectors changes, each sector lies in one
              identifier, and consecutive sectors have equal labels exactly
              when they have equal identifiers;
        \item distinct marks $p\ne q$ of $\mathcal D_j$ satisfy
              $\lVert v_p-v_q\rVert\ge a_0\max(S_p,S_q)$;
        \item the star square of every mark of $\mathcal D_j$ is disjoint
              from $\mathcal E_j$;
        \item distinct marks of $\mathcal D_j$ have disjoint removal
              squares;
        \item if $x$ is a fresh identifier of $\mathcal D_{j+1}$ and
              $y\ne x$ has the same label, then the closures of their
              current regions meet only at mark locations of
              $\mathcal D_{j+1}$;
        \item every descendant mark of a mark $(v,S)$ of $\mathcal D_i$
              lies within distance $2d(S)$ of $v$, and so does every birth
              region of a descendant fresh identifier;
        \item for every mark $p$ of $\mathcal D_i$ there is $n$ such that
              no mark of $\mathcal D_{i+n}$ has ancestor $p$.
    \end{enumerate}
    The statement of the uniform repair induction asserts that there are
    $a_0>0$ and $T$ such that for every $S_*\ge T$ and every width $C_0$
    the family satisfies the uniform repair induction with constant $a_0$
    whenever the lower layer index $k_0$ is sufficiently large.
    Since there are finitely many marks, the fifth item implies that every
    positive-length interface of a replacement with a distinct identifier
    joins opposite labels.
\end{definition}

\begin{lemma}[Metric invariants of the repair]
    \label{lem:area_law_repair_metric_invariants}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.le_scale,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.separated,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.disjoint_repairSquare,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.dist_descendant_le,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.birth_descendant_subset,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.disjoint_earlierTerminalSquares,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.terminates}
    \leanok
    \uses{def:area_law_repair_ancestors,thm:area_law_repair_margins}
    Let $0<a_0\le1$ and $S_*>0$, suppose $d(S)\le(a_0/1000)S$ for every
    $S\ge S_*$, that every initial mark has scale at least $S_*$, and that
    distinct initial marks satisfy the separation inequality with constant
    $a_0$. Then every mark of every generation has scale at least $S_*$,
    and items two, three, four, six and seven of
    Definition~\ref{def:area_law_uniform_repair_conclusion} hold.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_repair_margins}
    Children of a parent of scale $S$ have scale $d(S)\ge S_*$ and lie
    within distance $d(S)$ of it; distinct offsets are at distance at
    least one. Two children of one parent are therefore at least
    $d(S)\ge a_0d(S)$ apart. For children of distinct parents of scales
    $S,T$ and $M=\max(S,T)$,
    \begin{align}
        a_0M-d(S)-d(T) & \ge\Bigl(1-\frac{2}{1000}\Bigr)a_0M
        \ge a_0\max\bigl(d(S),d(T)\bigr),
        \notag
    \end{align}
    which proves separation by induction on the generation. Disjointness
    of the removal squares follows from $d(S)+d(T)<a_0M$. Since
    $2d(S)\le S$, induction along a branch gives
    $\lVert v_q-v\rVert+S_q\le2d(S)$ for every descendant $q$ of positive
    generation of $(v,S)$; a fresh birth lies in the removal square of its
    parent. Along a branch the scales at least halve, while they stay at
    least $S_*$, so every branch terminates. Finally, let $p$ be a mark of
    generation $j$ and $t$ a terminal mark of generation $i<j$. The
    ancestor $a$ of $p$ at generation $i$ is nonterminal, so $a\ne t$, and
    the star square of $p$ lies within distance $2d(S_a)$ of $v_a$. A
    common point with the terminal square of $t$ would give
    $a_0\max(S_a,S_t)\le2d(S_a)+d(S_t)\le(3a_0/1000)\max(S_a,S_t)$.
\end{proof}

\subsection{The induction step}
\label{sec:area_law_repair_induction_step}

Index the eight elementary triangles of $[-1,1]^2$ cyclically, and let the
triangle $s$ have initial direction $a_s$ and final direction $b_s$, so
that $b_s=a_{s+1}$. With $a\times b=a_1b_2-a_2b_1$ we have
$a_s\times b_s=1$. The closed cone of $s$ is
$K_s=\{y:a_s\times y\ge0,\ y\times b_s\ge0\}$, and its open cone is given
by the strict inequalities.

\begin{lemma}[Cones of the normalized fan]
    \label{lem:area_law_repair_fan_cones}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairFanCone,
        TNLean.PEPS.AreaLaw.Geometry.repairFanOpenCone,
        TNLean.PEPS.AreaLaw.Geometry.exists_mem_repairFanCone,
        TNLean.PEPS.AreaLaw.Geometry.eq_zero_of_mem_repairFanCone,
        TNLean.PEPS.AreaLaw.Geometry.mem_repairFanCone_inter_next_iff,
        TNLean.PEPS.AreaLaw.Geometry.norm_eq_of_mem_repairFanCone,
        TNLean.PEPS.AreaLaw.Geometry.mem_repairCurrentFanTriangle_iff,
        TNLean.PEPS.AreaLaw.Geometry.interior_repairCurrentFanTriangle,
        TNLean.PEPS.AreaLaw.Geometry.mem_closure_openCone_inter}
    \leanok
    \uses{def:area_law_repair_fan}
    Every vector lies in some closed cone. Two cones $K_s,K_u$ with
    $s\ne u$ and $u\ne s\pm1$ meet only at the origin, and
    $K_s\cap K_{s+1}$ is the ray of $b_s$. On $K_s$ the sup norm equals
    $y\times b_s+a_s\times y$. Hence the closed triangle of half-side
    $r>0$ about $c$ is $\{x:x-c\in K_s,\ \lVert x-c\rVert\le r\}$, its
    interior is given by the open cone and $\lVert x-c\rVert<r$, and every
    point of a translated closed cone in an open set $N$ is a limit of
    points of the translated open cone in $N$.
\end{lemma}

\begin{proof}
    \leanok
    The cone conditions are two linear inequalities with integer
    coefficients, and the assertions reduce to finitely many linear
    comparisons. A point $y$ with $y\times b=0$ equals $(a\times y)\,b$
    when $a\times b=1$. The triangle is the convex hull of $0,a_s,b_s$, whose
    barycentric coordinates are $y\times b_s$ and $a_s\times y$. For the
    approximation move from $x$ in the direction $a_s+b_s$.
\end{proof}

\begin{lemma}[Isolated stars and local radial models]
    \label{lem:area_law_star_local_model}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairSectorSet,
        TNLean.PEPS.AreaLaw.Geometry.repairActiveRaySet,
        TNLean.PEPS.AreaLaw.Geometry.isOpen_repairSectorSet_diff,
        TNLean.PEPS.AreaLaw.Geometry.mem_repairActiveRaySet_of_mem_two,
        TNLean.PEPS.AreaLaw.Geometry.isIsolatedStar_of_model,
        TNLean.PEPS.AreaLaw.Geometry.IsIsolatedStar.exists_model}
    \leanok
    \uses{lem:area_law_repair_fan_cones,def:area_law_isolated_star}
    Let $\tau$ assign identifiers to the eight wedges about $c$. The closed
    sector of $i$ is the union of the translated cones $c+K_s$ with
    $\tau(s)=i$, and the active rays are the rays from $c$ in the
    directions $b_s$ with $\tau(s)\ne\tau(s+1)$. Suppose that in the open
    square of radius $\rho'>\rho>0$ about $c$ the closure of every region
    is its closed sector, every region contains its closed sector with the
    active rays removed, and consecutive wedges have equal labels exactly
    when they have equal identifiers. Then the closed square of radius
    $\rho$ about $c$ is an isolated star. Conversely, an isolated star of
    radius $\rho$ has this description in the open square of radius
    $\rho$, with the region interiors in place of the regions.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:area_law_repair_fan_cones}
    A closed sector minus the active rays is open: near such a point only
    cones containing it occur, and these carry one identifier, since a
    common point of two cones with different identifiers lies on an active
    ray. A point on an active ray between $s$ and $s+1$ lies in the closure
    of the region of $\tau(s)$; were it interior to that region, nearby
    points of the open cone of $s+1$ would lie in that region, against the
    closure description. The remaining assertions follow from the cone
    descriptions of the triangles.
\end{proof}

\begin{theorem}[Isolated stars at the children of one repair]
    \label{thm:area_law_repair_child_stars}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairChildAssignment,
        TNLean.PEPS.AreaLaw.Geometry.isIsolatedStar_repairChild}
    \leanok
    \uses{lem:area_law_star_local_model,lem:area_law_repair_child_cones,
        def:area_law_repair_birth}
    Let a parent $(v,S)$ with $d=d(S)$ have a local radial model with
    wedge assignment $\sigma$ in the open square of radius $r\ge3d/2$
    about $v$. Let a labelled family agree, in the open square of
    half-side $d/2$ about a child $c=v+de$, with the old regions minus
    $v+[-d,d]^2$, keeping their labels, and with the interiors of the
    transported replacement runs of $\sigma$, with the opposite labels.

\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:area_law_star_local_model,lem:area_law_repair_child_cones}
    Give each outer child wedge the old identifier of its parent wedge and
    each inner one the replacement run of its parent wedge. A point of an
    open outer child cone lies in the open parent cone, off the parent rays
    and outside the removal square, so in the remnant; a point of an open
    inner child cone lies in the interior of the replacement triangle.
    Conversely, approximating by points of open parent cones places every
    point of a remnant or replacement in a child cone of the right kind.
    A parent ray between wedges of different identifiers, and a side of the
    removal square, lie on child rays between child wedges of different
    identifiers. Labels: across a parent ray they are the parent's labels or
    their reversals, and across a side they are opposite by construction.
    Lemma~\ref{lem:area_law_star_local_model} concludes.
\end{proof}

\begin{theorem}[One generation of the repair]
    \label{thm:area_law_repair_generation_step}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairCurrentSectorAssignment_eq,

\begin{theorem}[The number of descendant marks]
    \label{thm:area_law_repair_descendant_marks}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairMarkAncestor_parent_offset_injective,
        TNLean.PEPS.AreaLaw.Geometry.card_repairMarkAncestor_fiber_succ_le,
        TNLean.PEPS.AreaLaw.Geometry.card_repairMarkAncestor_fiber_le}
    \leanok
    \uses{def:area_law_repair_ancestors,thm:area_law_repair_parent_offset,
        thm:area_law_repair_finite_marks}
    Fix a mark $p\in K_i$. For $n\ge0$, let
    \begin{align}
        F_n(p)=\{q\in K_{i+n}:\alpha_{i,n}(q)=p\}.
        \label{eq:area_law_repair_descendant_marks}
    \end{align}
    These sets are finite. The parent and offset map restricts to an
    injection $F_{n+1}(p)\longrightarrow F_n(p)\times\{0,1,2\}^2$.
    Consequently,
    \begin{align}
        |F_{n+1}(p)|\le9|F_n(p)|,
        \qquad |F_n(p)|\le9^n.
        \label{eq:area_law_repair_descendant_count}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_repair_parent_offset,thm:area_law_repair_finite_marks}
    Each $F_n(p)$ is a subset of the finite mark family $K_{i+n}$.
    The parent of a descendant in $F_{n+1}(p)$ belongs to $F_n(p)$ by the
    defining recursion for the ancestor map. Its parent and offset
    determine the child, which proves the stated injection and the
    first inequality. Since $F_0(p)=\{p\}$, induction gives the second.
\end{proof}

\begin{theorem}[A common empty generation]
    \label{thm:area_law_repair_common_empty_generation}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.exists_isEmpty_markIndex}

        thm:area_law_repair_finite_marks}

\begin{proof}\leanok
    \uses{lem:area_law_repair_metric_invariants,
        thm:area_law_repair_finite_marks}

\end{proof}
