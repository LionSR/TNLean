\begin{definition}[Ancestors and descendants]
    \label{def:area_law_repair_ancestors}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairMarkParent,
        TNLean.PEPS.AreaLaw.Geometry.repairMarkAncestor,
        TNLean.PEPS.AreaLaw.Geometry.repairRegionParent,
        TNLean.PEPS.AreaLaw.Geometry.repairEarlierTerminalSquares}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_repair_children,
        def:area_law_repair_successor_regions}
    A mark occurrence $q\in K_{j+1}$ of $\mathcal D_{j+1}$ (Definition~\ref{def:area_law_repair_stage})
    consists of a nonterminal parent $p=\pi_j(q)\in K_j$ and an offset; for $n\ge0$, its ancestor at
    generation $i$ is given by
    \begin{align}
        \alpha_{i,0} & =\mathrm{id},
                     & \alpha_{i,n+1} & =\alpha_{i,n}\circ\pi_{i+n}.
        \label{eq:area_law_repair_ancestor_map}
    \end{align}
    Each fresh identifier of $\mathcal D_{j+1}$ has a unique parent in $K_j$ (retained identifiers
    have none); the descendants of $p\in K_i$ are the marks and fresh identifiers with ancestor $p$.
    Set $\mathcal E_j=\bigcup_{i<j}\bigcup_{p\in K_i,\,d(S_p)<S_*}Q(v_p,S_p)$ for the union of earlier
    closed terminal squares.
\end{definition}

\begin{theorem}[Scales of descendants]
    \label{thm:area_law_repair_descendant_scales}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_scale_eq_iterate}
    \leanok
    \uses{def:area_law_repair_ancestors,def:area_law_repair_terminal_children}
    With $S_j$ the scale map on $K_j$ and $d^{[n]}$ the $n$-fold iterate of $d$, every
    $q\in K_{i+n}$ ($i,n\ge0$) satisfies
    \begin{align}
        S_{i+n}(q) & =d^{[n]}\bigl(S_i(\alpha_{i,n}(q))\bigr).
        \label{eq:area_law_repair_descendant_scales}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{def:area_law_repair_ancestors,def:area_law_repair_terminal_children}
    By induction on $n$: $n=0$ is immediate, and for $q\in K_{i+n+1}$ with $p=\pi_{i+n}(q)$,
    Definition~\ref{def:area_law_repair_terminal_children} and (\ref{eq:area_law_repair_ancestor_map})
    give $S_{i+n+1}(q)=d(S_{i+n}(p))=d^{[n+1]}(S_i(\alpha_{i,n+1}(q)))$.
\end{proof}

\begin{definition}[Conclusion of the uniform repair induction]
    \label{def:area_law_uniform_repair_conclusion}
    \lean{TNLean.PEPS.AreaLaw.Geometry.IsUniformRepairInduction,
        TNLean.PEPS.AreaLaw.Geometry.UniformRepairInductionStatement}
    \leanok
    \uses{def:area_law_repair_ancestors,def:area_law_repair_square,
        def:area_law_isolated_star}
    For $a_0>0$, the family $(\mathcal D_j)_{j\ge0}$ satisfies the \emph{uniform repair induction}
    with constant $a_0$ if for all $i,j,n\ge0$:
    \begin{enumerate}
        \item every $p\in K_j$ with $(v_p,S_p)$ has an isolated star
              $v_p+[-a_0S_p,a_0S_p]^2$ (Definition~\ref{def:area_law_isolated_star}) in $\mathcal D_j$;
        \item distinct $p,q\in K_j$ satisfy $\lVert v_p-v_q\rVert\ge a_0\max(S_p,S_q)$;
        \item for every $p\in K_j$, $(v_p+[-a_0S_p,a_0S_p]^2)\cap\mathcal E_j=\varnothing$;
        \item distinct $p,q\in K_j$ have $Q(v_p,S_p)\cap Q(v_q,S_q)=\varnothing$;
        \item if $x\in I_{j+1}$ is fresh and $y\in I_{j+1}\setminus\{x\}$ has $\xi_y^{(j+1)}=\xi_x^{(j+1)}$,
              then $\overline{R_x^{(j+1)}}\cap\overline{R_y^{(j+1)}}\subseteq\{v_q^{(j+1)}:q\in K_{j+1}\}$;
        \item every descendant mark and descendant birth region of $(v,S)\in K_i$ lies in $v+[-2d(S),2d(S)]^2$;
        \item for every $p\in K_i$ there exists $n\ge0$ such that $\alpha_{i,n}^{-1}(\{p\})=\varnothing$.
    \end{enumerate}
    The \emph{uniform repair induction statement} asserts the existence of $a_0>0$ and $T$ such that
    for every $S_*\ge T$, width $C_0$, and sufficiently large $k_0$, $(\mathcal D_j)_{j\ge0}$
    satisfies (1)--(7) with constant $a_0$.
\end{definition}

\begin{lemma}[Metric invariants of the repair]
    \label{lem:area_law_repair_metric_invariants}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.le_scale,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.separated,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.disjoint_repairSquare,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.dist_descendant_le,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.birth_descendant_subset,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.disjoint_earlierTerminalSquares,
        TNLean.PEPS.AreaLaw.Geometry.RepairMetricHypotheses.terminates}
    \leanok
    \uses{def:area_law_repair_ancestors,thm:area_law_repair_margins}
    If $0<a_0\le1$, $S_*>0$, $d(S)\le(a_0/1000)S$ for all $S\ge S_*$, $S_p^{(0)}\ge S_*$ for all
    $p\in K_0$, and distinct $p,q\in K_0$ satisfy $\lVert v_p^{(0)}-v_q^{(0)}\rVert\ge a_0\max(S_p^{(0)},S_q^{(0)})$,
    then every mark at every generation has scale $\ge S_*$, and items (2), (3), (4), (6), and (7)
    of Definition~\ref{def:area_law_uniform_repair_conclusion} hold.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_repair_margins}
    Children of a nonterminal parent of scale $S$ have scale $d(S)\ge S_*$, lie within $d(S)$ of $v$,
    and are $\ge d(S)\ge a_0d(S)$ apart for distinct offsets; for children of distinct parents of
    scales $S,T$ with $M=\max(S,T)$,
    \begin{align}
        a_0M-d(S)-d(T) & \ge\Bigl(1-\frac{2}{1000}\Bigr)a_0M
        \ge a_0\max\bigl(d(S),d(T)\bigr),
        \notag
    \end{align}
    giving (2) by induction, and (4) since $d(S)+d(T)<a_0M$. Since $2d(S)\le S$, induction along a
    branch gives $\lVert v_q-v\rVert+S_q\le2d(S)$ for every positive-generation descendant $q$ of
    $(v,S)$, giving (6). Since $d(S)\le S/2$ and scales stay $\ge S_*$, every branch terminates (7).
    Finally, for $p\in K_j$ with generation-$i$ ($i<j$) nonterminal ancestor $a\ne t$ ($t\in K_i$ terminal),
    $(v_p+[-a_0S_p,a_0S_p]^2)\subseteq v_a+[-2d(S_a),2d(S_a)]^2$ cannot meet $Q(v_t,S_t)$ since
    $2d(S_a)+d(S_t)\le\frac{3a_0}{1000}\max(S_a,S_t)<a_0\max(S_a,S_t)$, giving (3).
\end{proof}
