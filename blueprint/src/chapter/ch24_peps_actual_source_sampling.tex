\section{Sampling the source circuit}
\label{sec:peps_actual_source_sampling}

This section completes the Gaussian sampling estimate for a source-only circuit in
\cite[proof of Theorem 5.2]{OpenAI2026PolynomialPEPS}.
% Source: 04-compression.tex, lines 342--558.
The circuit retains its source occurrences, local branch labels, physical registers,
and discarded registers; its tensor-network representation is proved separately.

\begin{theorem}[Finite-dimensional spaces of discarded registers]
    \label{thm:peps_finite_register_memories}
    \lean{TNLean.Lattice.PartyCircuit.Layout.finiteDimensional_mem_of_registers,TNLean.Lattice.PartyCircuit.Layout.finiteDimensional_mem_restrict_mapOwner_mapOwner}
    \leanok
    For any finite ordered list $R$ of registers with finite-dimensional complex
    Hilbert spaces, $\mathcal H_R$ is finite-dimensional, and remains so after
    relabelling register parties twice and restricting to either side of any
    party partition.
\end{theorem}
\begin{proof}
    \leanok
    Induction on $R$ shows $\dim\mathcal H_R<\infty$; party relabelling preserves
    each register's Hilbert space, and restriction selects a sublist of $R$.
\end{proof}

\begin{definition}[Orthonormal coordinates of selected discarded registers]
    \label{def:peps_restricted_discard_basis}
    \uses{thm:peps_finite_register_memories}
    \lean{TNLean.Lattice.PartyCircuit.Layout.restrictedDiscardBasis}
    \leanok
    Under Theorem~\ref{thm:peps_finite_register_memories}, choose a finite
    orthonormal basis of the restricted discarded space indexed by
    $\{0,\ldots,d-1\}$, where $d$ is its dimension.
\end{definition}

\begin{definition}[Exterior physical coordinates]
    \label{def:peps_exterior_family_physical_basis}
    \uses{def:peps_family_physical_output,thm:peps_affected_original_dimensions}
    \lean{TNLean.Lattice.PartyCircuit.exteriorFamilyPhysicalBasis}
    \leanok
    For a finite party set $P$ listed without repetition with physical
    dimensions $d_p$ and $A\subseteq P$, mapping the product physical basis
    outside $A$ (indexed by $\prod_{p\notin A}\{0,\ldots,d_p-1\}$) through the
    party identifications of Definition~\ref{def:peps_family_physical_output}
    and Theorem~\ref{thm:peps_affected_original_dimensions} gives an orthonormal
    basis of the exterior physical space.
\end{definition}

\begin{theorem}[Exterior physical dimension]
    \label{thm:peps_exterior_family_physical_dimension}
    \lean{TNLean.Lattice.PartyCircuit.finrank_exteriorFamilyPhysicalLayout}
    \leanok
    The dimension of the exterior physical space after the two party
    identifications is $\prod_{p\notin A}d_p$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_exterior_family_physical_basis}
    The orthonormal basis of Definition~\ref{def:peps_exterior_family_physical_basis}
    has cardinality $|\prod_{p\notin A}\{0,\ldots,d_p-1\}|=\prod_{p\notin A}d_p$.
\end{proof}

\begin{theorem}[Expected norm of a partial branch]
    \label{thm:peps_actual_gaussian_branch}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integrable_actualGaussianBranch_and_integral_le}
    \leanok
    Let $W$ be an allowed circuit with physical output registers followed by
    discarded registers, $S$ a set of source occurrences with endpoint parties
    $A(S)$, $p_0$ an allowed source-free scalar-input preparation, and
    $\xi,\zeta$ partial branches for $A(S)$. Suppose every source occurrence and
    branch has isometric endpoint frames and probability Schmidt weights, and
    choose orthonormal bases of the affected and exterior physical spaces
    (indexed by $X$ and $Y$) and discarded spaces (indexed by $D$ and $E$). For
    $k>0$, the Gaussian partial-branch operator $M_{\xi,\zeta}(\omega)$ on input
    $p_0(1)$, traced over the discarded space, satisfies that
    $\|M_{\xi,\zeta}\|_1$ is integrable and
    \begin{align}
        \mathbb E\,\|M_{\xi,\zeta}\|_1
         & \leq |X|^2 k^{-|S|/2}.
        \label{eq:peps_actual_branch_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_corrected_schmidt_output,thm:peps_gaussian_separated_physical_bound,thm:peps_local_schmidt_output_coordinates,def:peps_separated_physical_density,thm:peps_gaussian_branch_density_bound}
    By Theorems~\ref{thm:peps_actual_corrected_schmidt_output}
    and~\ref{thm:peps_local_schmidt_output_coordinates}, $\xi$ and $\zeta$
    determine separated ket and bra outputs $\Psi_u$ and $\Phi_v$ as in
    Definition~\ref{def:peps_separated_physical_density}. With centered Schmidt
    corrections $\delta_s^{\xi_s,\zeta_s}(\omega)$ at $s\in S$ and
    \begin{align}
        Z^{\xi,\zeta}_{u,v}(\omega)
         & =\prod_{s\in S}
        \bigl(\delta_s^{\xi_s,\zeta_s}(\omega)\bigr)_{u_s,v_s},
        \label{eq:peps_actual_branch_coefficients}
    \end{align}
    reindexing over $S$ and tracing out the discarded registers via
    Theorem~\ref{thm:peps_gaussian_branch_density_bound} gives
    \begin{align}
        M_{\xi,\zeta}(\omega)
         & =\operatorname{tr}_{\mathrm{disc}}
        \sum_{u,v}Z^{\xi,\zeta}_{u,v}(\omega)
        |\Psi_u\rangle\langle\Phi_v|
        =M\bigl(Z^{\xi,\zeta}(\omega)\bigr).
        \label{eq:peps_actual_branch_identification}
    \end{align}
    Applying Theorem~\ref{thm:peps_gaussian_separated_physical_bound} to
    (\ref{eq:peps_actual_branch_identification}) with
    (\ref{eq:peps_actual_branch_coefficients}) gives integrability and
    (\ref{eq:peps_actual_branch_bound}).
\end{proof}

\begin{theorem}[Expected norm of the corrected term]
    \label{thm:peps_actual_corrected_source_bound}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integral_actualCorrectedSourceTerm_le}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:peps_actual_gaussian_branch} on input
    state $|p_0(1)\rangle\langle p_0(1)|$, for $k>0$ the physical corrected term
    $F_S$ in any orthonormal physical and discarded output bases satisfies
    \begin{align}
        \mathbb E\,\|F_S\|_1
         & \leq
        \left(\sum_{\xi,\zeta}
        |c_{\xi}\overline{c_{\zeta}}|\right)
        |X|^2 k^{-|S|/2},
        \label{eq:peps_actual_corrected_bound}
    \end{align}
    summed over partial branch choices $\xi,\zeta$ at gate occurrences meeting
    $A(S)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_gaussian_branch,thm:peps_physical_corrected_basis_invariance,thm:peps_gaussian_branch_density_bound}
    By Theorem~\ref{thm:peps_physical_corrected_basis_invariance}, $\|F_S\|_1$ is
    unchanged in the bases adapted to $A(S)$, and combining
    Theorem~\ref{thm:peps_gaussian_branch_density_bound} with
    (\ref{eq:peps_actual_branch_bound}) gives (\ref{eq:peps_actual_corrected_bound}).
\end{proof}

\begin{theorem}[Corrected-term bound from circuit participation]
    \label{thm:peps_actual_source_lifetime_bound}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integral_actualCorrectedSourceTerm_le_lifetime}
    \leanok
    Let $P$ be a finite party set listed without repetition with physical
    dimensions $d_p\le D$ ($D\ge 1$) and finite-dimensional discarded registers.
    Let $W$ be an allowed circuit with party participation at most $b$ and gate
    coefficient sums at most $B\ge 1$, and let $p_0$ be an allowed source-free
    scalar-input preparation. With normalized Schmidt decompositions of all
    source vectors, for every $S\subseteq\mathcal E$ and $k>0$, the physical
    corrected term $F_S$ on input $|p_0(1)\rangle\langle p_0(1)|$ satisfies
    \begin{align}
        \mathbb E\,\|F_S\|_1
         & \leq
        \bigl(B^{4b}D^4\bigr)^{|S|} k^{-|S|/2}.
        \label{eq:peps_actual_lifetime_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_corrected_source_bound,thm:peps_actual_source_lifetime_cost,thm:peps_affected_original_dimensions,def:peps_exterior_family_physical_basis,def:peps_restricted_discard_basis}
    Using the bases of Definitions~\ref{def:peps_restricted_discard_basis}
    and~\ref{def:peps_exterior_family_physical_basis},
    Theorem~\ref{thm:peps_affected_original_dimensions} gives
    $|X|^2=\prod_{p\in A(S)}d_p^2$, and combining
    (\ref{eq:peps_actual_corrected_bound}) with
    Theorem~\ref{thm:peps_actual_source_lifetime_cost} yields
    (\ref{eq:peps_actual_lifetime_bound}).
\end{proof}

\begin{theorem}[Choosing one sample from corrected-term bounds]
    \label{thm:peps_source_sample_choice_conditional}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_sampledSourceMatrix_error_le_of_integral_bound}
    \leanok
    For a source-only circuit with $N=|\mathcal E|$, finite orthonormal input
    and physical/discard output bases, an input matrix, representing source
    weights and frames, $Q\geq0$, and $0<\varepsilon\leq1$, set
    \begin{align}
        k=\left\lceil\left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2\right\rceil+1.
        \label{eq:peps_conditional_sample_count}
    \end{align}
    If $\mathbb E\,\|F_S\|_1\leq Q^{|S|}k^{-|S|/2}$ for every nonempty
    $S\subseteq\mathcal E$, then there exists a Gaussian sample $\omega$ with
    $\|\widehat\rho_{\mathrm{phys}}(\omega)-\rho_{\mathrm{phys}}\|_1\le\varepsilon/4$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_sample_count_bound,thm:peps_sampled_error_integral_bound,def:peps_source_sample_count}
    By Theorems~\ref{thm:peps_sampled_error_integral_bound}
    and~\ref{thm:peps_source_sample_count_bound} with $k$ from
    (\ref{eq:peps_conditional_sample_count})
    (Definition~\ref{def:peps_source_sample_count}), some realization $\omega$
    satisfies
    $\|\widehat\rho_{\mathrm{phys}}(\omega)-\rho_{\mathrm{phys}}\|_1
        \le\sum_{\varnothing\neq S\subseteq\mathcal E}Q^{|S|}k^{-|S|/2}\le\varepsilon/4$.
\end{proof}

\begin{theorem}[A single sample for the source circuit]
    \label{thm:peps_actual_source_sampling}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_sourceSample_lifetime_error_le}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:peps_actual_source_lifetime_bound},
    let $N=|\mathcal E|$ and $0<\varepsilon\leq1$, and set
    \begin{align}
        Q & = B^{4b}D^4,
          & k            & = \left\lceil
        \left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2
        \right\rceil+1.
        \label{eq:peps_actual_sample_count}
    \end{align}
    Then there exist normalized Schmidt decompositions for every source
    occurrence and branch label and a Gaussian sample $\omega$ of size $k$ such
    that the sampled physical operator $\widehat\rho(\omega)$ and exact physical
    output $\rho$ satisfy
    \begin{align}
        \|\widehat\rho(\omega)-\rho\|_1 & \leq \varepsilon/4.
        \label{eq:peps_actual_sample_error}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_source_lifetime_bound,thm:peps_local_schmidt_sources,thm:peps_source_sample_choice_conditional}
    Choose normalized Schmidt decompositions by
    Theorem~\ref{thm:peps_local_schmidt_sources}; then
    Theorem~\ref{thm:peps_actual_source_lifetime_bound} gives
    $\mathbb E\,\|F_S\|_1\leq Q^{|S|}k^{-|S|/2}$ for all $S\neq\varnothing$, and
    Theorem~\ref{thm:peps_source_sample_choice_conditional} yields $\omega$
    satisfying (\ref{eq:peps_actual_sample_error}).
\end{proof}

\begin{theorem}[Power bounds for the sample count]
    \label{thm:peps_source_sampling_polynomial}
    \lean{TNLean.Lattice.Compression.sourceSamplingCount_lt_of_power_bounds,TNLean.Lattice.Compression.sourceSamplingCount_lt_inv_pow,TNLean.Lattice.Compression.sourceSamplingCount_lt_of_gate_power_bounds}
    \leanok
    If $L\geq1$, $Q\geq0$, $\varepsilon>0$, $n,q,p\in\N$, $C_N\geq1$, and
    $C_Q\geq0$ satisfy $N\leq C_NL^n$, $Q\leq C_QL^q$, and
    $\varepsilon^{-1}\leq C_\varepsilon L^p$, then $k=k(N,Q,\varepsilon)$
    satisfies
    \begin{align}
        k<\bigl(64C_N^2C_Q^2C_\varepsilon^2+2\bigr)
        L^{2(n+q+p)}
        \label{eq:peps_source_sampling_polynomial}
    \end{align}
    (with $C_\varepsilon=1$ when $\varepsilon=L^{-p}$). In particular, if
    $0\leq B\leq C_BL^m$ and $0\leq D\leq C_DL^r$ ($C_B\ge 0$, $m,r\in\N$), then
    for $Q=B^{4b}D^4$ and $\varepsilon=L^{-p}$,
    \begin{align}
        k<\bigl[64C_N^2(C_B^{4b}C_D^4)^2+2\bigr]
        L^{2(n+4bm+4r+p)}.\notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_source_sample_count}
    Since $\max(1,N)\le C_NL^n$ and $L^{2(n+q+p)}\ge 1$, Definition~\ref{def:peps_source_sample_count}
    gives $k<(8\max(1,N)Q/\varepsilon)^2+2\le (64C_N^2C_Q^2C_\varepsilon^2+2)L^{2(n+q+p)}$;
    substituting $Q=B^{4b}D^4\le C_B^{4b}C_D^4L^{4bm+4r}$ gives the second bound.
\end{proof}

\begin{theorem}[Polynomial sampling for real accuracy exponents]
    \label{thm:peps_actual_source_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_sourceSample_polynomial_error_le}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:peps_actual_source_sampling}, if
    $L\geq1$, $n,m,r\in\N$, $C_N\geq1$, $C_B\geq0$, and
    \begin{align}
        N & \leq C_N L^n,
          & B             & \leq C_B L^m,
          & D             & \leq C_D L^r,
        \label{eq:peps_actual_resource_powers}
    \end{align}
    then for any $p\in\R$ and $q=\max(0,\lceil p\rceil)$, setting $k$ by
    (\ref{eq:peps_actual_sample_count}) at $\varepsilon=L^{-q}$ gives
    \begin{align}
        k & < \left[64C_N^2\bigl(C_B^{4b}C_D^4\bigr)^2+2\right]
        L^{2(n+4bm+4r+q)},
        \label{eq:peps_actual_polynomial_count}
    \end{align}
    and there exist Schmidt decompositions and a size-$k$ sample with
    $\|\widehat\rho-\rho\|_1\leq L^{-p}/4$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_source_sampling,thm:peps_source_sampling_polynomial,def:peps_source_sample_count}
    Since $L\ge 1$ and $q\ge\max(0,p)$, $\varepsilon=L^{-q}\in(0,1]$ and
    $L^{-q}\le L^{-p}$; applying Theorems~\ref{thm:peps_actual_source_sampling}
    and~\ref{thm:peps_source_sampling_polynomial} with
    (\ref{eq:peps_actual_resource_powers}) yields
    (\ref{eq:peps_actual_polynomial_count}) and $\|\widehat\rho-\rho\|_1\le L^{-q}/4\le L^{-p}/4$.
\end{proof}

\begin{theorem}[Polynomial stack length at the per-gate error]
    \label{thm:peps_effect_stack_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.stackLength_sourceGateBudget_eq}
    \lean{TNLean.Lattice.PartyCircuit.stackLength_sourceGateBudget_lt_of_power_bounds}
    \leanok
    For $r,N\in\N$, $S\in\mathbb R$, $\varepsilon>0$, and
    $\delta=\varepsilon/(8\max(1,N))$, the stack length is
    \begin{align}
        m & = \left\lceil(rS/\delta)^2\right\rceil+1
        = \left\lceil
        \bigl(8\max(1,N)rS/\varepsilon\bigr)^2
        \right\rceil+1.
        \label{eq:peps_effect_stack_count}
    \end{align}
    If $L\geq1$, $N\leq C_NL^n$, and $0\leq S\leq C_SL^s$ ($C_N\geq1$,
    $C_S\geq0$, $n,s,p\in\N$), then at $\varepsilon=L^{-p}$,
    $m<H L^{2(n+s+p)}$ with $H=64C_N^2(rC_S)^2+2$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_sampling_polynomial,def:peps_source_sample_count}
    By (\ref{eq:peps_effect_stack_count}), $m=k(N,rS,\varepsilon)$ in
    Definition~\ref{def:peps_source_sample_count}, so
    Theorem~\ref{thm:peps_source_sampling_polynomial} with $Q=rS\le rC_SL^s$
    gives $m<HL^{2(n+s+p)}$.
\end{proof}

\begin{theorem}[Polynomial number of replacement monomials]
    \label{thm:peps_effect_replacement_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.replacementMonomialBound_le_of_power_bounds}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.card_branchLabels_replacement_le_power}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:peps_effect_stack_polynomial}, if
    $K\in\N$ satisfies $K\leq C_KL^a$ ($C_K\geq0$, $a\in\N$), then
    \begin{align}
        K m^r & \leq C_K H^r L^{a+2r(n+s+p)}.
        \label{eq:peps_effect_monomial_count}
    \end{align}
    In particular, if each nonprivate gate has at most $K$ monomials, each with
    at most $r$ pair effects, and coefficient sum at most $S$, then every gate
    occurrence of the source-only replacement has at most
    $C_K H^r L^{a+2r(n+s+p)}$ branch labels.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_effect_stack_polynomial,thm:peps_source_only_resources}
    Combining $m\le HL^{2(n+s+p)}$ from
    Theorem~\ref{thm:peps_effect_stack_polynomial} with $K\le C_KL^a$ gives
    (\ref{eq:peps_effect_monomial_count}), and
    Theorem~\ref{thm:peps_source_only_resources} bounds each replacement gate's
    branch count by $Km^r$.
\end{proof}

\begin{theorem}[Sampled physical operator as a sum of party tensor products]
    \label{thm:peps_actual_pure_sampled_physical_sum}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalSourceReplacedDensity_pure_sampled_eq_scalar_smul_physicalSum}
    \leanok
    \uses{thm:peps_actual_pure_physical_source_replacement,
        thm:peps_actual_source_column_physical_trace,thm:peps_actual_sampled_physical_scalar,
        def:peps_selected_physical_private_basis}
    For product input $\psi$, product physical basis of dimensions $d_p$,
    grouped local physical-and-private bases
    (Definition~\ref{def:peps_selected_physical_private_basis}), branch
    coefficients $c_\xi$, zero-party scalar $\lambda_W$, and $n$ source
    occurrences, the sampled physical operator
    $\widehat\rho_{\mathrm{phys},\psi}(\widehat X)$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS} satisfies
    \begin{align}
        \widehat\rho_{\mathrm{phys},\psi}(\widehat X)
         & =k^{-n}\lambda_W\overline{\lambda_W}
        \sum_{\xi,\zeta}c_\xi\overline{c_\zeta}
        \sum_{j\in[k]^n}\bigotimes_{p\in\mathcal P}M_{\xi,\zeta,j,p},
        \label{eq:peps_actual_pure_sampled_physical_sum}
    \end{align}
    where $M_{\xi,\zeta,j,p}$ is the local mixed physical trace of
    Theorem~\ref{thm:peps_actual_sampled_physical_scalar} at party $p$ (with
    $0^{-1}=0$ for $k=0$).
    % Source: 04-compression.tex, lines 279--355, 409--434 and 565--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_actual_pure_physical_source_replacement,
        thm:peps_actual_source_column_physical_trace,thm:peps_actual_sampled_physical_scalar}
    Expand $\widehat\rho_{\mathrm{phys},\psi}(\widehat X)$ over $\xi,\zeta$ by
    Theorem~\ref{thm:peps_actual_pure_physical_source_replacement}, pass to the
    party-grouped private basis via
    Theorem~\ref{thm:peps_actual_source_column_physical_trace}, and factor each
    $(\xi,\zeta)$ term by Theorem~\ref{thm:peps_actual_sampled_physical_scalar}
    to obtain (\ref{eq:peps_actual_pure_sampled_physical_sum}).
\end{proof}
