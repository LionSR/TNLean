\section{Sampling the source circuit}
\label{sec:peps_actual_source_sampling}

This section completes the Gaussian sampling estimate for a source-only circuit in
\cite[proof of Theorem 5.2]{OpenAI2026PolynomialPEPS}.
% Source: 04-compression.tex, lines 342--558.
The circuit retains its source occurrences, local branch labels, physical registers,
and discarded registers. The sampled physical output is an operator that need not be
positive; its tensor-network representation is proved separately.

\begin{theorem}[Finite-dimensional spaces of discarded registers]
    \label{thm:peps_finite_register_memories}
    \lean{TNLean.PEPS.PairEffect.Layout.finiteDimensional_mem_of_registers,TNLean.PEPS.PairEffect.Layout.finiteDimensional_mem_restrict_mapOwner_mapOwner}
    \leanok
    Let $R$ be a finite ordered list of registers, each with a finite-dimensional
    complex Hilbert space. Their tensor product is finite-dimensional, and
    remains so after relabelling register parties twice and restricting to either
    side of any partition of the resulting parties.
\end{theorem}
\begin{proof}
    \leanok
    Induct on the register list, using finite-dimensionality of tensor products.
    Relabelling parties does not change the underlying Hilbert spaces, and
    restricting to one side of a partition selects a sublist of registers.
\end{proof}

\begin{definition}[Orthonormal coordinates of selected discarded registers]
    \label{def:peps_restricted_discard_basis}
    \uses{thm:peps_finite_register_memories}
    \lean{TNLean.PEPS.PairEffect.Layout.restrictedDiscardBasis}
    \leanok
    Under the assumptions of Theorem~\ref{thm:peps_finite_register_memories},
    choose an orthonormal basis of the restricted register space indexed by
    $\{0,\ldots,d-1\}$, where $d$ is its dimension. Finite-dimensionality of
    each register gives this basis without any uniform dimension bound.
\end{definition}

\begin{definition}[Exterior physical coordinates]
    \label{def:peps_exterior_family_physical_basis}
    \uses{def:peps_family_physical_output,thm:peps_affected_original_dimensions}
    \lean{TNLean.PEPS.PairEffect.exteriorFamilyPhysicalBasis}
    \leanok
    Let $P$ be the finite set of parties, listed without repetition, and
    let party $p$ have physical dimension $d_p$. For $A\subseteq P$, the
    tensor product of the physical coordinate bases outside $A$ is indexed
    by $\prod_{p\notin A}\{0,\ldots,d_p-1\}$. Transporting this basis through
    the party identifications gives an orthonormal basis of the exterior physical
    space.
\end{definition}

\begin{theorem}[Exterior physical dimension]
    \label{thm:peps_exterior_family_physical_dimension}
    \lean{TNLean.PEPS.PairEffect.finrank_exteriorFamilyPhysicalLayout}
    \leanok
    The dimension of the exterior physical space after the two
    party identifications is $\prod_{p\notin A}d_p$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_exterior_family_physical_basis}
    Count the indices of the orthonormal basis from the preceding definition.
\end{proof}

\begin{theorem}[Expected norm of a partial branch]
    \label{thm:peps_actual_gaussian_branch}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.integrable_actualGaussianBranch_and_integral_le}
    \leanok
    Let $W$ be an allowed circuit whose output registers consist of physical
    registers followed by discarded registers. Fix a set $S$ of
    source occurrences and let $A(S)$ be their endpoint parties. Let $p_0$
    be an allowed scalar-input preparation containing no sources, and fix
    two partial branches $\xi,\zeta$ for $A(S)$.

    At every source occurrence and local branch label, suppose the two
    endpoint matrices have orthonormal columns and the source weights
    are nonnegative with sum one. Choose orthonormal bases for the affected
    and exterior physical spaces, indexed by $X$ and $Y$, and for the affected
    and exterior discarded spaces, indexed by $D$ and $E$. Under the party
    relabellings and register partitions, these four bases determine bases of
    the full physical and discarded spaces.

    For $k>0$, let $M_{\xi,\zeta}(\omega)$ be the Gaussian partial-branch
    operator on the input $p_0(1)$, traced over the discarded space
    in these bases. Then $\|M_{\xi,\zeta}\|_1$ is integrable with respect to
    the source Gaussian law and
    \begin{align}
        \mathbb E\,\|M_{\xi,\zeta}\|_1
         & \leq |X|^2 k^{-|S|/2}.
        \label{eq:peps_actual_branch_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_corrected_schmidt_output,thm:peps_gaussian_separated_physical_bound,thm:peps_local_schmidt_output_coordinates,def:peps_separated_physical_density,thm:peps_gaussian_branch_density_bound}
    Construct the local Schmidt output representation
    independently for $\xi$ and $\zeta$, and denote their separated ket and
    bra outputs by $\Psi_u$ and $\Phi_v$, as in
    Definition~\ref{def:peps_separated_physical_density}.
    At each occurrence $s\in S$, let $\xi_s,\zeta_s$ be the
    selected local labels and let $\delta_s^{\xi_s,\zeta_s}(\omega)$ be
    the centered correction in Schmidt coordinates. For assignments
    $u,v$ of pairs of Schmidt indices on $S$, set
    \begin{align}
        Z^{\xi,\zeta}_{u,v}(\omega)
         & =\prod_{s\in S}
        \bigl(\delta_s^{\xi_s,\zeta_s}(\omega)\bigr)_{u_s,v_s}.
        \label{eq:peps_actual_branch_coefficients}
    \end{align}
    The output-coordinate identities match each prepared output vector
    with the separated coordinates of the corresponding local words. Reindexing
    both corrected-coordinate sums over $S$ and tracing out the discarded
    registers gives
    \begin{align}
        M_{\xi,\zeta}(\omega)
         & =\operatorname{tr}_{\mathrm{disc}}
        \sum_{u,v}Z^{\xi,\zeta}_{u,v}(\omega)
        |\Psi_u\rangle\langle\Phi_v|
        =M\bigl(Z^{\xi,\zeta}(\omega)\bigr),
        \label{eq:peps_actual_branch_identification}
    \end{align}
    where $M(z)$ is the operator in
    Definition~\ref{def:peps_separated_physical_density} and
    $\operatorname{tr}_{\mathrm{disc}}$ traces out both discarded spaces.
    Applying Theorem~\ref{thm:peps_gaussian_separated_physical_bound}
    to~(\ref{eq:peps_actual_branch_identification}) with the coefficients
    in~(\ref{eq:peps_actual_branch_coefficients}) gives integrability
    and~(\ref{eq:peps_actual_branch_bound}).
\end{proof}

\begin{theorem}[Expected norm of the corrected term]
    \label{thm:peps_actual_corrected_source_bound}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.integral_actualCorrectedSourceTerm_le}
    \leanok
    Let $W$ be an allowed circuit whose output registers consist of
    physical registers followed by discarded registers. Let $p_0$ be an
    allowed source-free scalar-input preparation, with input state
    $|p_0(1)\rangle\langle p_0(1)|$. Fix a set $S$ of
    source occurrences and Schmidt decompositions of every source
    vector, with nonnegative weights summing to one and orthonormal endpoint
    columns.

    Let $A(S)$ be the endpoint parties of $S$. Choose orthonormal bases of
    the affected and exterior physical spaces and of the affected and
    exterior discarded spaces under the party identifications, and write $X$
    for the index set of the affected physical basis. For $k>0$, the
    physical corrected term $F_S$, in any orthonormal bases of the physical and
    discarded spaces, satisfies
    \begin{align}
        \mathbb E\,\|F_S\|_1
         & \leq
        \left(\sum_{\xi,\zeta}
        |c_{\xi}\overline{c_{\zeta}}|\right)
        |X|^2 k^{-|S|/2},
        \label{eq:peps_actual_corrected_bound}
    \end{align}
    where $\xi,\zeta$ range over the partial branch choices at gate occurrences
    meeting $A(S)$ and $c_\xi,c_\zeta$ are their branch coefficients.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_gaussian_branch,thm:peps_physical_corrected_basis_invariance,thm:peps_gaussian_branch_density_bound}
    Write $F_S$ in the bases adapted to $A(S)$, which leaves the trace norm
    unchanged. Expand $F_S$ into partial branches, apply the triangle inequality
    inside the expectation with weights $|c_\xi\overline{c_\zeta}|$, and bound
    each pair by Theorem~\ref{thm:peps_actual_gaussian_branch}. Finite sums of
    products of Gaussian entries are integrable, so expectation commutes with
    the sum.
\end{proof}

\begin{theorem}[Corrected-term bound from circuit participation]
    \label{thm:peps_actual_source_lifetime_bound}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.integral_actualCorrectedSourceTerm_le_lifetime}
    \leanok
    Let the finite set of parties be listed without repetition, with one
    physical register of dimension $d_p$ at each party $p$, and assume every
    discarded register is finite-dimensional. Let $W$ be an allowed circuit and
    $p_0$ an allowed source-free scalar-input preparation.

    Suppose each party participates in at most $b$ gate occurrences over
    the circuit, the sum of absolute branch coefficients at each gate occurrence
    is at most $B\geq1$, and $d_p\leq D$ for all $p$ with $D\geq1$. Fix normalized
    Schmidt decompositions of all source vectors. For every set $S$
    of source occurrences and every $k>0$, in any orthonormal bases of the
    input, physical output, and discarded spaces,
    \begin{align}
        \mathbb E\,\|F_S\|_1
         & \leq
        \bigl(B^{4b}D^4\bigr)^{|S|} k^{-|S|/2},
        \label{eq:peps_actual_lifetime_bound}
    \end{align}
    for the input $|p_0(1)\rangle\langle p_0(1)|$ under the source Gaussian law.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_corrected_source_bound,thm:peps_actual_source_lifetime_cost,thm:peps_affected_original_dimensions,def:peps_exterior_family_physical_basis,def:peps_restricted_discard_basis}
    Take the product physical bases on $A(S)$ and its complement, together with
    orthonormal bases on the two parts of the finite-dimensional discarded space.
    The discarded dimensions do not enter~(\ref{eq:peps_actual_corrected_bound}),
    and the squared affected physical dimension is $|X|^2=\prod_{p\in A(S)}d_p^2$.
    Since each of the at most $2|S|$ parties in $A(S)$ participates in at most
    $b$ gates, the product of $|X|^2$ with the ket--bra coefficient sum is at most
    $\bigl(B^{4b}D^4\bigr)^{|S|}$. Multiplying by $k^{-|S|/2}$ gives the result.
\end{proof}

\begin{theorem}[Choosing one sample from corrected-term bounds]
    \label{thm:peps_source_sample_choice_conditional}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.exists_sampledSourceMatrix_error_le_of_integral_bound}
    \leanok
    Fix a source-only circuit with finite orthonormal bases on the input,
    physical output, and discarded output spaces, and an input matrix.
    Fix endpoint matrices and nonnegative source weights reproducing every source
    vector. Let $N$ be the number of source occurrences, let $Q\geq0$, and let
    $0<\varepsilon\leq1$. Set
    \[
        k=\left\lceil\left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2\right\rceil+1.
    \]
    If every nonempty set $S$ of source occurrences satisfies
    \[
        \mathbb E\,\|F_S\|_1\leq Q^{|S|}k^{-|S|/2},
    \]
    then there exists a global Gaussian sample for which the sampled
    physical operator differs from the exact physical operator in trace norm
    by at most $\varepsilon/4$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_sample_count_bound,thm:peps_sampled_error_integral_bound,def:peps_source_sample_count}
    Expanding the difference between the sampled and exact operators over
    nonempty subsets $S$ bounds the expected trace-norm error by
    $\sum_{S\neq\varnothing}\mathbb E\,\|F_S\|_1$. By the choice of $k$, this sum
    is at most $\varepsilon/4$, so some sample realization achieves error at most
    $\varepsilon/4$.
\end{proof}

\begin{theorem}[A single sample for the source circuit]
    \label{thm:peps_actual_source_sampling}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.exists_sourceSample_lifetime_error_le}
    \leanok
    Let $W$ be an allowed circuit on a finite set of parties, listed without
    repetition, with physical output $\bigotimes_p\mathbb C^{d_p}$ followed
    by finite-dimensional discarded registers. Let $p_0$ be an allowed
    source-free scalar-input preparation, with input
    $|p_0(1)\rangle\langle p_0(1)|$.

    Suppose every party participates in at most $b$ gate occurrences, every gate
    has sum of absolute branch coefficients at most $B\geq1$, and $d_p\leq D$
    with $D\geq1$. Let $N$ be the number of source occurrences. For
    $0<\varepsilon\leq1$, set
    \begin{align}
        Q & = B^{4b}D^4,
          & k            & = \left\lceil
        \left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2
        \right\rceil+1.
        \label{eq:peps_actual_sample_count}
    \end{align}
    Then there exist normalized Schmidt decompositions for every source occurrence
    and local branch label, and a global Gaussian sample $\omega$ of size $k$ per
    occurrence and ordered local label pair, such that
    \begin{align}
        \|\widehat\rho(\omega)-\rho\|_1 & \leq \varepsilon/4,
        \label{eq:peps_actual_sample_error}
    \end{align}
    where $\rho$ is the physical output operator and $\widehat\rho(\omega)$ is
    obtained by replacing each mixed source operator with its $k$-sample Gaussian
    average and tracing out the discarded registers.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_source_lifetime_bound,thm:peps_local_schmidt_sources,thm:peps_source_sample_choice_conditional}
    Choose normalized Schmidt decompositions for all source vectors. For every
    nonempty set $S$ of source occurrences, the circuit participation bound gives
    $\mathbb E\,\|F_S\|_1\leq Q^{|S|}k^{-|S|/2}$. With $k$ given
    by~(\ref{eq:peps_actual_sample_count}), the conditional sampling theorem yields
    a realization $\omega$ satisfying~(\ref{eq:peps_actual_sample_error}).
\end{proof}

\begin{theorem}[Power bounds for the sample count]
    \label{thm:peps_source_sampling_polynomial}
    \lean{TNLean.PEPS.Approximation.sourceSamplingCount_lt_of_power_bounds,TNLean.PEPS.Approximation.sourceSamplingCount_lt_inv_pow,TNLean.PEPS.Approximation.sourceSamplingCount_lt_of_gate_power_bounds}
    \leanok
    Let $L\geq1$, $Q\geq0$, and $\varepsilon>0$, and suppose
    \[
        N\leq C_NL^n,\qquad Q\leq C_QL^q,\qquad
        \varepsilon^{-1}\leq C_\varepsilon L^p,
    \]
    for nonnegative integers $n,q,p$ and constants $C_N\geq1$ and $C_Q\geq0$.
    Then the sample count satisfies
    \[
        k<\bigl(64C_N^2C_Q^2C_\varepsilon^2+2\bigr)
        L^{2(n+q+p)}.
    \]
    When $\varepsilon=L^{-p}$, one may take $C_\varepsilon=1$.
    In particular, if $0\leq B\leq C_BL^m$ and $0\leq D\leq C_DL^r$ with
    $C_B\geq0$, then $Q=B^{4b}D^4$ gives
    \[
        k<\bigl[64C_N^2(C_B^{4b}C_D^4)^2+2\bigr]
        L^{2(n+4bm+4r+p)}.
    \]
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_source_sample_count}
    Since $C_NL^n\geq1$, we have $\max(1,N)\leq C_NL^n$. Substituting the three
    power bounds into $k<(8\max(1,N)Q/\varepsilon)^2+2$ and using
    $L^{2(n+q+p)}\geq1$ gives the first inequality; setting $Q=B^{4b}D^4$ gives
    the second.
\end{proof}

\begin{theorem}[Polynomial sampling for real accuracy exponents]
    \label{thm:peps_actual_source_polynomial}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.exists_sourceSample_polynomial_error_le}
    \leanok
    Under the assumptions of
    Theorem~\ref{thm:peps_actual_source_sampling}, let $L\geq1$ and suppose
    \begin{align}
        N & \leq C_N L^n,
          & B             & \leq C_B L^m,
          & D             & \leq C_D L^r,
        \label{eq:peps_actual_resource_powers}
    \end{align}
    for nonnegative integers $n,m,r$ and constants $C_N\geq1$ and $C_B\geq0$.
    For any real exponent $p$, set $q=\max(0,\lceil p\rceil)$ and define $k$
    by~(\ref{eq:peps_actual_sample_count}) at accuracy $\varepsilon=L^{-q}$.
    Then
    \begin{align}
        k & < \left[64C_N^2\bigl(C_B^{4b}C_D^4\bigr)^2+2\right]
        L^{2(n+4bm+4r+q)},
        \label{eq:peps_actual_polynomial_count}
    \end{align}
    and there exist Schmidt decompositions and a sample of size $k$
    satisfying $\|\widehat\rho-\rho\|_1\leq L^{-p}/4$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_actual_source_sampling,thm:peps_source_sampling_polynomial,def:peps_source_sample_count}
    Since $L\geq1$ and $q\geq p\vee 0$, the accuracy $\varepsilon=L^{-q}$ lies in
    $(0,1]$ and satisfies $L^{-q}\leq L^{-p}$. Applying the sampling theorem at
    $\varepsilon=L^{-q}$ and bounding $k$ via~(\ref{eq:peps_actual_resource_powers})
    yields~(\ref{eq:peps_actual_polynomial_count}) and the error bound.
\end{proof}
\begin{theorem}[Polynomial stack length at the per-gate error]
    \label{thm:peps_effect_stack_polynomial}
    \lean{TNLean.PEPS.PairEffect.stackLength_sourceGateBudget_eq}
    \lean{TNLean.PEPS.PairEffect.stackLength_sourceGateBudget_lt_of_power_bounds}
    \leanok
    Let $r,N$ be nonnegative integers, let $S\in\mathbb R$, and let $\varepsilon>0$.
    For the per-gate error $\delta=\varepsilon/(8\max(1,N))$, the stack length is
    \begin{align}
        m & = \left\lceil(rS/\delta)^2\right\rceil+1
        = \left\lceil
        \bigl(8\max(1,N)rS/\varepsilon\bigr)^2
        \right\rceil+1.
        \label{eq:peps_effect_stack_count}
    \end{align}
    If $L\geq1$, $N\leq C_NL^n$, and $0\leq S\leq C_SL^s$ with
    $C_N\geq1$, $C_S\geq0$, and nonnegative integers $n,s,p$, then at
    $\varepsilon=L^{-p}$ we have $m<H L^{2(n+s+p)}$ with
    $H=64C_N^2(rC_S)^2+2$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_sampling_polynomial,def:peps_source_sample_count}
    Substituting $\delta=\varepsilon/(8\max(1,N))$ into the stack-length formula
    gives the same ceiling expression as the sample count with $Q=rS$. The
    power bound on $k$ with $rS\leq rC_SL^s$ then gives $m<H L^{2(n+s+p)}$.
\end{proof}

\begin{theorem}[Polynomial number of replacement monomials]
    \label{thm:peps_effect_replacement_polynomial}
    \lean{TNLean.PEPS.PairEffect.replacementMonomialBound_le_of_power_bounds}
    \lean{TNLean.PEPS.PairEffect.OriginalCircuit.card_branchLabels_replacement_le_power}
    \leanok
    Under the preceding power bounds, let $K$ be a nonnegative integer satisfying
    $K\leq C_KL^a$ with $C_K\geq0$ and a nonnegative integer $a$. Then
    \begin{align}
        K m^r & \leq C_K H^r L^{a+2r(n+s+p)}.
        \label{eq:peps_effect_monomial_count}
    \end{align}
    In particular, if every nonprivate gate of the circuit has at most $K$
    monomials, each with at most $r$ pair effects, and absolute coefficient sum at
    most $S$, then every gate occurrence of the source-only replacement has at most
    the number of branch labels in~(\ref{eq:peps_effect_monomial_count}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_effect_stack_polynomial,thm:peps_source_only_resources}
    Raise $m\leq HL^{2(n+s+p)}$ to the power $r$ and multiply by $K\leq C_KL^a$.
    Replacing up to $r$ pair effects in each of the at most $K$ monomials by a
    stack of length $m$ produces at most $Km^r$ branch labels per gate occurrence.
\end{proof}

\begin{theorem}[Sampled physical operator as a sum of party tensor products]
    \label{thm:peps_actual_pure_sampled_physical_sum}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.physicalSourceReplacedDensity_pure_sampled_eq_scalar_smul_physicalSum}
    \leanok
    \uses{thm:peps_actual_pure_physical_source_replacement,
        thm:peps_actual_source_column_physical_trace,thm:peps_actual_sampled_physical_scalar,
        def:peps_selected_physical_private_basis}
    Let $\psi$ be the product input state. Fix the product physical basis with
    local dimensions $d_p$ and an orthonormal basis of the private output space.
    Choose orthonormal bases of the local source-input and private output spaces,
    and form the grouped local physical-and-private bases as in
    Definition~\ref{def:peps_selected_physical_private_basis}.
    Let $c_\xi$ be the chronological branch coefficients and let
    $\lambda_W$ be the scalar factor from gates with no participating parties.
    For the sampled source matrices $\widehat X$, write
    $\widehat\rho_{\mathrm{phys},\psi}(\widehat X)$ for the resulting physical
    operator. If $n$ is the number of source occurrences, then
    \begin{align}
        \widehat\rho_{\mathrm{phys},\psi}(\widehat X)
         & =k^{-n}\lambda_W\overline{\lambda_W}
        \sum_{\xi,\zeta}c_\xi\overline{c_\zeta}
        \sum_{j\in[k]^n}\bigotimes_{p\in\mathcal P}M_{\xi,\zeta,j,p},
        \label{eq:peps_actual_pure_sampled_physical_sum}
    \end{align}
    where $M_{\xi,\zeta,j,p}$ is the local mixed physical trace of
    Theorem~\ref{thm:peps_actual_sampled_physical_scalar} at party $p$, and each
    $j\in[k]^n$ assigns a sample index to every source occurrence. The case
    $k=0$ is included with the convention $0^{-1}=0$: if $n\ge1$ both sides
    vanish, since $[0]^n$ is empty, and if $n=0$ the factor $k^{-n}$ equals $1$.
    This is the party-factored expansion of the sampled physical operator in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 279--355, 409--434 and 565--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_actual_pure_physical_source_replacement,
        thm:peps_actual_source_column_physical_trace,thm:peps_actual_sampled_physical_scalar}
    Expand the sampled operator over independent ket and bra branch choices
    $\xi,\zeta$ by
    Theorem~\ref{thm:peps_actual_pure_physical_source_replacement}.
    For each pair $(\xi,\zeta)$,
    Theorem~\ref{thm:peps_actual_source_column_physical_trace} replaces the
    private output basis by the party-grouped basis without changing the partial
    trace over the private registers. Applying
    Theorem~\ref{thm:peps_actual_sampled_physical_scalar} term by term factors
    each sample $j\in[k]^n$ across the parties $p\in\mathcal P$ and yields
    (\ref{eq:peps_actual_pure_sampled_physical_sum}).
\end{proof}
