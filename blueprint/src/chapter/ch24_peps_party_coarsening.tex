\section{Grouping parties into two sides}
\label{sec:peps_party_coarsening}

For an arbitrary map $f:P\to Q$ of party labels, we group all parties with
the same image while retaining every register and operation, providing the
owner identification for the affected and exterior systems in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
(\texttt{04-compression.tex}, lines 383--427).

% The two-side result below assumes the explicit internal-source condition.
% The chronological construction that frees corrected and crossing registers
% and retains untouched aggregate gates must still establish that condition.
% Neither that construction nor the full compression theorem is claimed here.

\begin{definition}[Changing register owners]
    \label{def:peps_owner_layout}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwner_nil}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwner_cons}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwner_append}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwnerIso}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwnerIso_nil_apply}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwnerIso_cons_tmul}
    \leanok
    \uses{def:peps_party_registers}
    For a register list $\ell=((p_1,H_1),\ldots,(p_n,H_n))$ (with no
    injectivity or finiteness assumption on $f,P,Q$), put
    $\ell^f=((f(p_1),H_1),\ldots,(f(p_n),H_n))$, preserving empty lists and
    concatenation. The owner-relabelling isometry
    $R_{f,\ell}:\mathcal H_\ell\to\mathcal H_{\ell^f}$ acts as $\Id_{\C}$ on the
    empty list and satisfies
    \begin{align}
        R_{f,(p,H)\ell}(x\otimes y)
         & =x\otimes R_{f,\ell}y.
        \label{eq:peps_owner_head}
    \end{align}
\end{definition}

\begin{theorem}[Compatibility with concatenation]
    \label{thm:peps_owner_concatenation}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwnerIso_append_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mapOwnerIso_append}
    \leanok
    \uses{def:peps_owner_layout}
    With $C_{\ell,m}:\mathcal H_{\ell m}\to\mathcal H_\ell\otimes\mathcal H_m$
    the concatenation isometry and $(\ell m)^f=\ell^f m^f$, every
    $x\in\mathcal H_\ell$ and $y\in\mathcal H_m$ satisfy
    \begin{align}
        R_{f,\ell m}C_{\ell,m}^{-1}(x\otimes y)
         & =C_{\ell^f,m^f}^{-1}
        (R_{f,\ell}x\otimes R_{f,m}y),
        \label{eq:peps_owner_concatenation}
    \end{align}
    or equivalently $R_{f,\ell m}C_{\ell,m}^{-1}=C_{\ell^f,m^f}^{-1}(R_{f,\ell}\otimes R_{f,m})$.
\end{theorem}

\begin{proof}\leanok
    Induction on $\ell$ using (\ref{eq:peps_owner_head}) gives
    (\ref{eq:peps_owner_concatenation}) on elementary tensors, and linearity
    gives the operator identity.
\end{proof}

\begin{definition}[Sources remaining between distinct owners]
    \label{def:peps_owner_sources}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_nil}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_cons}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_append}
    \leanok
    \uses{def:peps_source_inventory}
    A source $s=(p,q;U,V;\eta)$ remains between distinct owners iff
    $f(p)\ne f(q)$, with new record $(f(p),f(q);U,V;\eta)$. For an ordered
    source list $S$, let $S_f$ be the sublist of surviving records in their
    original order, preserving the empty list and concatenation.
\end{definition}

\begin{theorem}[Surviving pair labels after grouping]
    \label{thm:peps_owner_source_pairs}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.map_partyPair_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_partyPair_mapOwner}
    \leanok
    \uses{def:peps_owner_sources}
    The unordered pair labels of $S_f$ are obtained (with multiplicity) by
    applying $f$ to both endpoints of each pair in $S$ and discarding diagonal
    images $f(p)=f(q)$; in particular, a pair $k$ occurs in $S_f$ iff $k$ is
    non-diagonal and $k=\{f(p),f(q)\}$ for some $\{p,q\}$ occurring in $S$.
\end{theorem}

\begin{proof}\leanok
    By induction on $S$ using Definition~\ref{def:peps_owner_sources}, a
    source with endpoints $p,q$ contributes $\{f(p),f(q)\}$ iff $f(p)\ne f(q)$.
\end{proof}

\begin{theorem}[Normalization and the absence of crossing sources]
    \label{thm:peps_owner_sources_internal}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.IsNormalized.mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_eq_nil_iff}
    \leanok
    \uses{def:peps_owner_sources}
    If the vectors of $S$ are normalized, then those of $S_f$ are normalized,
    and $S_f$ is empty iff $f(p)=f(q)$ for every source $(p,q;U,V;\eta)$ in $S$.
\end{theorem}

\begin{proof}\leanok
    By Definition~\ref{def:peps_owner_sources}, surviving records keep the
    same vectors $\eta$, and a record is removed iff $f(p)=f(q)$.
\end{proof}

\begin{definition}[Preparing an internal pair]
    \label{def:peps_internal_pair_preparation}
    \lean{TNLean.Lattice.PartyCircuit.Word.localPairSource}
    \leanok
    \uses{def:peps_party_layout}
    For $q\in Q$ and $\eta\in U\otimes V$, prepare two fresh registers $U,V$
    at owner $q$ by tensoring the local map $\C\to U\otimes(V\otimes\C)$,
    $c\mapsto\eta\otimes c$ (under standard associativity) with the identity
    on all spectator registers.
\end{definition}

\begin{theorem}[The internal preparation operator]
    \label{thm:peps_internal_pair_preparation}
    \lean{TNLean.Lattice.PartyCircuit.Word.localPairSource_spec}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_localPairSource}
    \leanok
    \uses{def:peps_internal_pair_preparation}
    The operation of Definition~\ref{def:peps_internal_pair_preparation} maps
    every spectator vector $x$ to $\eta\otimes x$, and when $\|\eta\|=1$ it is
    an allowed local operation creating no pair source between distinct owners.
\end{theorem}

\begin{proof}\leanok
    The map $c\mapsto\eta\otimes c$ has norm $\|\eta\|$, so tensoring with the
    spectator identity and the associativity isometry preserves contractivity
    when $\|\eta\|=1$ and acts as $x\mapsto\eta\otimes x$ by linearity on
    elementary tensors of $\eta$.
\end{proof}

\begin{definition}[Changing the owners of a composition]
    \label{def:peps_owner_composition}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapOwner}
    \leanok
    \uses{def:peps_owner_layout,def:peps_internal_pair_preparation}
    For a composition $F:\mathcal H_\ell\to\mathcal H_{\ell'}$, define
    $F^f:\mathcal H_{\ell^f}\to\mathcal H_{(\ell')^f}$ in the same
    chronological order by replacing each local map $A:\mathcal H_a\to\mathcal H_b$
    with $R_{f,b}AR_{f,a}^{-1}$, retaining exchanges and untouched registers on
    their original spaces, keeping each pair source $(p,q;U,V;\eta)$ with
    $f(p)\ne f(q)$ as $(f(p),f(q);U,V;\eta)$, and replacing each pair source
    with $f(p)=f(q)$ by the internal preparation of
    Definition~\ref{def:peps_internal_pair_preparation}.
\end{definition}

\begin{theorem}[Exact preservation of an allowed composition]
    \label{thm:peps_owner_composition}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_mapOwner_comp}
    \leanok
    \uses{def:peps_owner_composition,def:peps_owner_sources}
    Without assuming norm bounds, $F$ and $F^f$ satisfy
    \begin{align}
        R_{f,\ell'}F & =F^fR_{f,\ell}.
        \label{eq:peps_owner_composition}
    \end{align}
    If $F$ is allowed with pair-source list $S$, then $F^f$ is allowed with
    pair-source list $S_f$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_owner_concatenation,thm:peps_internal_pair_preparation}
    By induction on $F$, Theorem~\ref{thm:peps_owner_concatenation} intertwines
    local maps, Theorem~\ref{thm:peps_internal_pair_preparation} intertwines
    internal pair preparations, and surviving sources, exchanges, and untouched
    registers commute with $R_{f,\cdot}$, giving (\ref{eq:peps_owner_composition}),
    preservation of local contraction bounds, and source list $S_f$.
\end{proof}

\begin{theorem}[Two contractions when every source is internal]
    \label{thm:peps_internal_source_factorization}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_partition_of_sources_internal}
    \leanok
    % The statement below is an auxiliary result with its internal-source hypothesis explicit.
    \uses{def:peps_owner_composition,def:peps_party_partition}
    Let $f:P\to\{0,1\}$, let $F:\mathcal H_\ell\to\mathcal H_{\ell'}$ be an
    allowed composition with $f(p)=f(q)$ for every pair source $(p,q;U,V;\eta)$
    in $F$, and let
    $J_{f,\ell}:\mathcal H_\ell\to\mathcal H_{\ell,+}\otimes\mathcal H_{\ell,-}$
    compose $R_{f,\ell}$ with the binary partition of
    Definition~\ref{def:peps_party_partition} into the memories of owner images
    $1$ and $0$. Then the owner restrictions $A_+,A_-$ of $F^f$ are allowed
    source-free compositions with $\|A_+\|,\|A_-\|\le1$ and
    \begin{align}
        J_{f,\ell'}F
         & =(A_+\otimes A_-)J_{f,\ell}.
        \label{eq:peps_internal_source_factorization}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_owner_sources_internal,thm:peps_owner_composition,
        thm:peps_party_binary_factorization,thm:peps_party_restriction}
    By Theorems~\ref{thm:peps_owner_sources_internal} and
    \ref{thm:peps_owner_composition}, $F^f$ is allowed with empty source list
    $S_f$; applying Theorems~\ref{thm:peps_party_binary_factorization} and
    \ref{thm:peps_party_restriction} to $F^f$ and composing with
    (\ref{eq:peps_owner_composition}) yields
    (\ref{eq:peps_internal_source_factorization}) with $\|A_+\|,\|A_-\|\le1$.
\end{proof}
