\section{Corrected source inputs and local outputs}
\label{sec:peps_corrected_input_frames}

The proof of Theorem~5.2 in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS} expands each partial output through local input isometries.

Source occurrences retain their chronological positions, even when their
endpoint parties coincide with those of another occurrence.
A register list $L$ has memory $\mathcal H_L$; concatenation is identified
with tensor product by the concatenation isometry
$\alpha_{L,M}\colon\mathcal H_{L+M}\to\mathcal H_L\otimes\mathcal H_M$.
For a division of the parties into two classes, write $\Pi_L$ for the
isometry gathering the two classes in their inherited orders.
Relabelling parties changes neither the order nor the Hilbert spaces.

    \uses{thm:peps_basis_memory_equalities}

    \lean{TNLean.Lattice.PartyCircuit.Word.eval_castLayouts_apply_heq}
    \lean{TNLean.Lattice.PartyCircuit.Layout.memCongr_append_tmul}

    % coordinate-chapter home; cite that entry rather than duplicate its tag.
    \uses{thm:peps_basis_memory_equalities}
    Identifying two equal register lists $L=L'$ preserves vectors in $\mathcal H_L$ and word evaluations, and commutes with source
    preparation and with $\alpha_{L,M}\colon\mathcal H_{L+M}\to\mathcal H_L\otimes\mathcal H_M$.
\end{theorem}
\begin{proof}
    \leanok
    Substitution along $L=L'$ reduces each claim to reflexivity.
\end{proof}

\begin{definition}[Ordered source positions]
    \label{def:peps_ordered_source_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.sourcePositionsLayout}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareSourcePositions}

    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.sourcePositionsLayout_filter}
    \leanok
    For a family of pair sources $r_i$ and a finite ordered list $J$ of positions,
    let $L_J$ be the concatenation of their two-register lists and
    $P_J(\eta)$ the word preparing the vectors $\eta_i$ ($i\in J$) in order.
    Then $L_{J+K}=L_J+L_K$; filtering $J$ by a predicate retains the
    inherited order, endpoint parties and spaces, and replacing $\eta_i$ by $0$ preserves $L_J$.
\end{definition}

\begin{theorem}[Preparation in an ordered list]
    \label{thm:peps_ordered_source_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSourcePositions_heq}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSourcePositions_append}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSourcePositions_filter}
    \leanok
    Preparation indexed by $J$ equals preparation of the corresponding
    ordered source records; under Theorem~\ref{thm:peps_register_identifications},
    $P_{J+K}(\eta)=P_J(\eta)P_K(\eta)$, and filtering $J$ by a predicate prepares the filtered sublist of sources.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_ordered_source_positions,thm:peps_register_identifications}
    Induction on $J$, prepending $\eta_i$ at each step and applying Theorem~\ref{thm:peps_register_identifications} on the spectator registers.
\end{proof}

\begin{theorem}[Uniform permutation of source registers]
    \label{thm:peps_reorder_source_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_reorderSourcePositions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_reorderFreeSlots}
    \leanok
    If $J'$ is a permutation of $J$, then for any spectator register list there is an allowed source-free composition $U$ (independent of $\eta$, without normalization assumptions) such that for every $\eta$,
    \begin{align}
        U P_J(\eta) & = P_{J'}(\eta).
        \label{eq:peps_frames_reorder}
    \end{align}
    In particular, any filtered sublist of free positions can be placed before
    its complement while preserving the internal order of each part.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_ordered_source_preparation}
    Induction on a decomposition of the permutation into adjacent transpositions of two-register source blocks using Theorem~\ref{thm:peps_ordered_source_preparation}; specializing $J'$ to the filtered sublist followed by its complement gives the second claim.
\end{proof}

\begin{definition}[The joint vector of selected sources]
    \label{def:peps_free_source_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.freeSourceVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.crossingSourceVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_freeSlotLayout}
    \leanok
    For selected positions $J$, let $\eta_J\in\mathcal H_{L_J}$ be the ordered
    tensor product of $\eta_i$ ($i\in J$), with $\eta_\emptyset=1$; each register in $L_J$ is one of the two endpoint registers of some $i\in J$.
    For a bipartition of the parties, the crossing vector is $\Pi_{L_J}\eta_J$ with the registers of $L_J$ relabelled by their side of the bipartition.
\end{definition}

\begin{theorem}[Preparation and normalization of source vectors]
    \label{thm:peps_free_source_norm}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareFreeSlots_eq_appendIso_symm}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.norm_freeSourceVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.norm_crossingSourceVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.freeSourceVector_congr}
    \leanok
    For every spectator vector $x\in\mathcal H_M$,
    \begin{align}
        P_J(\eta)x & = \alpha_{L_J,M}^{-1}(\eta_J\otimes x).
        \label{eq:peps_frames_prepare}
    \end{align}
    Moreover, $\eta_J$ depends only on $(\eta_i)_{i\in J}$, and if $\|\eta_i\|=1$ for all $i\in J$, then $\|\eta_J\|=\|\Pi_{L_J}\eta_J\|=1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_free_source_vector}
    Equation~(\ref{eq:peps_frames_prepare}) follows by induction on $J$; $\|\eta_J\|=\prod_{i\in J}\|\eta_i\|=1$, and $\Pi_{L_J}$ is an isometry.
\end{proof}

\begin{definition}[Corrected and crossing blocks]
    \label{def:peps_corrected_crossing_blocks}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSlotMask}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.crossingSlotMask}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceLayout}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.crossingSourceLayout}
    \leanok
    For a set $S$ of original source occurrences of a circuit $W$, let
    $A(S)$ be the set of all endpoint parties of occurrences in $S$. Among the surviving partial
    source positions, a position is corrected if its original occurrence lies in $S$, and crossing if its two endpoints lie on opposite sides of $A(S)$.
    Write $C$ and $T$ for the corrected and crossing register lists in the inherited original order, along with their Boolean-labelled versions recording whether each register lies in $A(S)$.
\end{definition}

\begin{theorem}[The free positions are corrected or crossing]
    \label{thm:peps_corrected_crossing_masks}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSlotMask_endpoints}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedFreeSlots_eq_or}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.crossingSlotMask_eq_false_of_corrected}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedFreeSlots_masks}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceLayout_owners}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_correctedSourceLayout_append}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedMask_meets}
    \leanok
    Both endpoints of every corrected position lie in $A(S)$, so every occurrence in $S$ meets $A(S)$, the corrected
    and crossing positions are disjoint, and their union is the set of free positions. Since all registers of $C$ have owners in $A(S)$, the affected and exterior restrictions of $C+T$ are $(C+T)_A=C+T_A$ and $(C+T)_E=T_E$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_corrected_crossing_blocks}
    By Definition~\ref{def:peps_corrected_crossing_blocks}, every occurrence in $S$ has both endpoints in $A(S)$ and therefore cannot cross between $A(S)$ and its complement.
\end{proof}

\begin{theorem}[A common corrected-source rearrangement]
    \label{thm:peps_corrected_reordering}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_correctedInputReordering}
    \leanok
    There is an allowed source-free composition $U$ from $C+T$ (followed by the original input) to the
    original free-register order satisfying~(\ref{eq:peps_frames_reorder})
    for every free-source vector family, chosen before the partial branch $\xi$ and any corrected-source coordinate.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_reorder_source_positions,thm:peps_corrected_crossing_masks}
    Apply Theorem~\ref{thm:peps_reorder_source_positions} to the free positions with the corrected positions selected; by Theorem~\ref{thm:peps_corrected_crossing_masks}, the complement is the crossing list.
\end{proof}

\begin{theorem}[Preparing vectors from the scalar unit]
    \label{thm:peps_source_scalar_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepare_nil_heq}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.vector_append_heq}

    Evaluating $P_J(\eta)$ on $1\in\mathbb C$ gives $\eta_J$, and concatenating two inventories gives
    $\alpha_{L,M}^{-1}(\eta_L\otimes\eta_M)$. A scalar-input composition
    $p_0$ acting with $L+M$ as spectators acts as $\id$ on $\alpha_{L,M}^{-1}(\eta_L\otimes\eta_M)$ and prepares $p_0(1)$ in the original input memory.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_free_source_norm,thm:peps_register_identifications}
    Specializing~(\ref{eq:peps_frames_prepare}) of Theorem~\ref{thm:peps_free_source_norm} to $x=1$ and commuting spectator identity maps past source preparations via Theorem~\ref{thm:peps_register_identifications} gives all claims.
\end{proof}

\begin{definition}[Absorbing the original input preparation]
    \label{def:peps_prepared_reordered_word}

    source vectors, and a residual composition $V$, let $\widetilde V$ be the word with input $C+T$ that

    fixed sources, and applies $V$ to the input vector

    For a partial circuit branch $\xi$, let $\widetilde V_\xi$ be the word formed from the residual and fixed-source vectors of $\xi$.
\end{definition}

\end{proof}

\begin{theorem}[Schmidt isometries for the crossing vector]
    \label{thm:peps_crossing_schmidt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.norm_sourceCrossingVector}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_sourceCrossing_schmidt}
    \leanok
    For every partial branch $\xi$, $\|\beta_\xi\|=1$, and there exist a finite index set
    $J$, probabilities $\tau_j\geq0$ with $\sum_{j\in J}\tau_j=1$, and linear
    isometries $L\colon\ell^2(J)\to\mathcal H_{T_A}$ and
    $R\colon\ell^2(J)\to\mathcal H_{T_E}$ such that
    \begin{align}
        \beta_\xi & = \sum_{j\in J}\sqrt{\tau_j}\,Le_j\otimes Re_j.
        \label{eq:peps_frames_crossing_schmidt}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_free_source_norm,def:peps_actual_crossing_vector}
    By Theorem~\ref{thm:peps_free_source_norm}, $\|\beta_\xi\|=1$, and the finite-dimensional Schmidt decomposition of $\beta_\xi\in\mathcal H_{T_A}\otimes\mathcal H_{T_E}$ supplies $(J,\tau,L,R)$ satisfying~(\ref{eq:peps_frames_crossing_schmidt}).
\end{proof}

\begin{definition}[Schmidt input isometries]
    \label{def:peps_schmidt_input_frames}
    \lean{TNLean.Lattice.PartyCircuit.HasSchmidtInputFrames}
    \leanok
    Given $Q\colon\ell^2(I)\to\mathcal H_C$ and $\beta\in\mathcal H_{T_A}\otimes\mathcal H_{T_E}$, a Schmidt input representation consists of probabilities $\tau_j$ ($j\in J$) and linear isometries
    $A\colon\ell^2(I\times J)\to\mathcal H_{(C+T)_A}$ and $B\colon\ell^2(J)\to\mathcal H_{(C+T)_E}$, chosen before $i\in I$, such that for all $i\in I$,
    \begin{align}
        \Pi_{C+T}\alpha_{C,T}^{-1}(Qe_i\otimes\Pi_T^{-1}\beta)
         & = \sum_{j\in J}\sqrt{\tau_j}\,Ae_{(i,j)}\otimes Be_j.
        \label{eq:peps_frames_input_schmidt}
    \end{align}
\end{definition}

\begin{theorem}[Construction of the input isometries]
    \label{thm:peps_actual_schmidt_input_frames}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_correctedInputFrames}
    \leanok
    For each partial branch $\xi$ and orthonormal-column endpoint matrices, $Q_\xi$ and $\beta_\xi$ admit a Schmidt input representation~(\ref{eq:peps_frames_input_schmidt}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_corrected_crossing_masks,thm:peps_crossing_schmidt,def:peps_corrected_source_frame}
    Using $(C+T)_A=C+T_A$ and $(C+T)_E=T_E$ from Theorem~\ref{thm:peps_corrected_crossing_masks}, define $Ae_{(i,j)}=\alpha_{C,T_A}^{-1}(Q_\xi e_i\otimes Le_j)$ and $Be_j=Re_j$ from~(\ref{eq:peps_frames_crossing_schmidt}) of Theorem~\ref{thm:peps_crossing_schmidt}; linearity across the tensor product gives~(\ref{eq:peps_frames_input_schmidt}).
\end{proof}

\begin{definition}[A local-word Schmidt output representation]
    \label{def:peps_schmidt_local_output}
    \lean{TNLean.Lattice.PartyCircuit.Word.HasSchmidtOutput}
    \leanok
    A family $z_i$ ($i\in I$) in the affected/exterior output tensor product has a
    local-word Schmidt representation if there exist allowed source-free
    words $D_A,D_E$ with $\|D_A\|,\|D_E\|\le1$, probabilities
    $\tau_j$, and linear input isometries $A,B$ (all chosen before $i$) such that
    \begin{align}
        z_i & = \sum_j\sqrt{\tau_j}\,(D_AAe_{(i,j)})\otimes(D_EBe_j).
        \label{eq:peps_frames_output_schmidt}
    \end{align}
\end{definition}

\begin{theorem}[Applying the two local factors]
    \label{thm:peps_local_schmidt_output}

\end{theorem}
\begin{proof}
    \leanok

\end{proof}

\begin{theorem}[Schmidt expansion of the partial output]

    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_correctedSchmidtOutput}
