\section{Coarse information outside a sublinear collar}

The first information estimate of \cite[Section~3]{OpenAI2026PolynomialPEPS}
concerns the lattice sample $T$ of a bounded union of rectangles. Its proof
reads $T$ as a template of the area-law companion, with the whole domain as
the cut, and then applies the sublinear collar bound and its amplification.
The physical collar theorem provides the required bound from the original
Hamiltonian and its full-system gap. Conditional forms are stated separately.

\begin{definition}[Bounded unions of lattice boxes]
    \label{def:peps_box_union}
    \lean{TNLean.PEPS.Approximation.IsBoxUnion}
    \leanok
    A finite set $T\subseteq\mathbb Z^2$ is a union of at most $N_0$ boxes when
    there are $k\le N_0$ and points $\ell_a,h_a\in\mathbb Z^2$ with
    $T=\bigcup_{a<k}[\ell_a,h_a]$, where $[\ell,h]$ is the set of lattice
    points $x$ with $\ell\le x\le h$ in both coordinates. A box may be empty.
\end{definition}

\begin{lemma}[Lattice samples of rectangles]
    \label{lem:peps_box_union_of_intervals}
    \lean{TNLean.PEPS.Approximation.isBoxUnion_of_ordConnected}
    \leanok
    \uses{def:peps_box_union}
    Let $I_a,J_a\subseteq\mathbb R$, $a<k\le N_0$, be intervals, each of
    which may be open, closed or half-open. If a finite set $T$ is the set of
    lattice points of $\bigcup_a I_a\times J_a$, then $T$ is a union of at most
    $N_0$ boxes.
\end{lemma}
\begin{proof}
    \leanok
    The integers in a real interval form an interval of integers. If both
    integer intervals of $I_a\times J_a$ are nonempty, they are bounded by the
    finiteness of $T$, so each is the closed interval between its infimum and
    supremum. Otherwise the sample of $I_a\times J_a$ is empty, and an empty
    box replaces it.
\end{proof}

\begin{lemma}[Translations and templates]
    \label{lem:peps_box_union_template}
    \lean{TNLean.PEPS.Approximation.IsBoxUnion.translate,
        TNLean.PEPS.Approximation.IsBoxUnion.exists_template}
    \leanok
    \uses{def:peps_box_union,def:area_law_template}
    An integer translate of a union of at most $N_0$ boxes is again such a union. Let
    $T$ be a nonempty union of at most $N_0$ boxes with sup-norm diameter at
    most $D$. If $D+1\le s_0$, $D\le n$ and
    $\max(C_{\rm tpl},0)N_0(s_0+1)\le n$, then $T$ is a template with
    parameters $(n,s_0)$ whose pieces are closed rectangles.
\end{lemma}
\begin{proof}
    \leanok
    A nonempty box $[\ell,h]$ is the lattice sample of the closed rectangle
    with corners $\ell-(\tfrac14,\tfrac14)$ and $h+(\tfrac14,\tfrac14)$, whose
    diameter is at most $D+\tfrac12$. An empty box is the lattice sample of the
    square $[-\tfrac14,-\tfrac18]^2$. The template inequalities are the
    hypotheses.
\end{proof}

\begin{theorem}[Coarse information outside a sublinear collar]
    \label{thm:peps_coarse_information}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le_square,
        TNLean.PEPS.Approximation.coarse_mutualInformation_le_translated_square}
    \leanok
    \uses{def:peps_box_union,def:polynomial_peps_model,
        thm:peps_coarse_information_of_collar,thm:area_law_sublinear_collar}
    In Proposition~3.1 of \cite{OpenAI2026PolynomialPEPS}, fix an integer
    $N_0\ge1$ and a constant $A_0\ge1$. There are $\beta<1$,
    $C<\infty$ and $u_0<\infty$ such that the following holds. Let $u\ge u_0$
    and let $T$ be the full ambient lattice sample of a union of at most $N_0$
    axis-parallel rectangles, with $\operatorname{diam}_\infty T\le A_0u$. For
    every physical set $E\subseteq\Lambda$ with
    $\operatorname{dist}_\infty(E,T)>u^\beta$,
    \begin{align*}
        I_\Omega(\Lambda\cap T:E) & \le C.
    \end{align*}
    The same holds for every subset of $\Lambda\cap T$ in the first argument,
    for rectangles with consistently assigned open or closed sides, and after a
    common translation of the lattice coordinates.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:area_law_collar_bound,thm:area_law_sublinear_collar,
        thm:peps_coarse_information_of_collar,
        thm:peps_coarse_information_square,lem:peps_box_union_of_intervals,
        lem:peps_box_union_template}
    By Lemma~\ref{lem:peps_box_union_of_intervals}, the lattice sample $T$ is
    a union of at most $N_0$ boxes. For a common real translation $d$,
    pulling both the physical coordinates and the rectangles back by $-d$
    before taking lattice samples retains the same interval
    sides and preserves all diameters and separation distances. The
    square-grid estimate together with the physical
    collar theorem, Proposition~9.6 of \cite{OpenAI2026AreaLaw}, at
    $D_0=13$ and $C_{\rm tpl}=24$ then yields the bound.
\end{proof}

\begin{lemma}[The coarse exponent is below one]
    \label{lem:peps_coarse_exponent_lt_one}
    \lean{TNLean.PEPS.Approximation.beta_lt_one}
    \leanok
    The exponent $\beta_0$ fixed by the amplification exponents satisfies
    $\beta_0<1$.
\end{lemma}

\begin{proof}
    \leanok
    This is a numerical evaluation of the amplification exponents.
\end{proof}

\begin{definition}[Constants of the coarse information estimate]
    \label{def:peps_coarse_information_constants}
    \lean{TNLean.PEPS.Approximation.coarseScaleFactor,
        TNLean.PEPS.Approximation.coarseInformationThreshold}
    \leanok
    \uses{thm:area_law_uniform_collar_threshold}
    Fix a template constant $C_{\rm tpl}$, an integer $N_0$ and $A_0\in\mathbb R$.
    The scale factor is
    \begin{align}
        K_{\rm c}(C_{\rm tpl},N_0,A_0) & =(\max\{C_{\rm tpl},0\}+1)(N_0+1)
        (\max\{A_0,1\}+4).
        \notag
    \end{align}
    For a collar threshold $N$, an amplification constant $C_a$ and an
    amplification threshold $N_a$, let $N_r$ be the threshold of
    Theorem~\ref{thm:area_law_uniform_collar_threshold} for the constant $C_a$
    and $\eta=K_{\rm c}^{-\beta_0}$, so that
    $\lfloor n^{1-\varepsilon}\rfloor+2r<K_{\rm c}^{-\beta_0}n^{\beta_0}$ for
    $n\ge N_r$ and every integer $0\le r\le C_an^{(1-\varepsilon)/\alpha}$.
    The coarse threshold is
    \begin{align}
        u_0(C_{\rm tpl},N_0,A_0,N,C_a,N_a) & =
        \max\{N,N_a,N_r,2,K_{\rm c}^{1/\varepsilon}\}.
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Coarse information with explicit constants]
    \label{thm:peps_coarse_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:peps_box_union,def:area_law_collar_bound,
        def:area_law_amplification_bound,def:peps_coarse_information_constants}
    Fix $q,R,J,\Delta$ and template constants $D_0,C_{\rm tpl}$. Suppose that
    the sublinear collar bound holds with constant $C$ and threshold $N$, and
    that the amplification bound holds at $(\max\{C,1\},1)$ with constant $C_a$
    and threshold $N_a$. Fix $N_0$ and $A_0$, and let
    $u_0=u_0(C_{\rm tpl},N_0,A_0,N,C_a,N_a)$ be the coarse threshold of
    Definition~\ref{def:peps_coarse_information_constants}. Let $u\ge u_0$, let
    $\Omega$ be a unit ground vector with full-system gap $\Delta$ of an
    admissible Hamiltonian of range $R$ and bound $J$ on a finite induced
    domain $\Lambda$, and let $T$ be a union of at most $N_0$ boxes with
    $\operatorname{diam}_\infty T\le A_0u$. Then
    $I_\Omega(X:E)\le\max\{C_a,0\}$ for every $X\subseteq\Lambda\cap T$ and
    every $E\subseteq\Lambda$ with
    $\operatorname{dist}_\infty(E,T)>u^{\beta_0}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_box_union_template,def:peps_coarse_information_constants,
        thm:area_law_uniform_collar_threshold}
    Set $\beta=\beta_0$. If $T$ is empty, so is $X$, and the information
    vanishes. Otherwise put $A=\max(A_0,1)$, $s_0=\lceil Au\rceil+1$ and
    $n=\lceil c_1(Au+3)\rceil$ with
    $c_1=(\max(C_{\rm tpl},0)+1)(N_0+1)$. Then $T$ is a template with
    parameters $(n,s_0)$, and $u\le n\le Ku$ with $K=c_1(A+4)=K_{\rm c}$ once
    $u\ge2$.
    The cut is $A=\Lambda$, whose crossing-edge endpoint set is empty, so the
    template is separated from it. When $u^\varepsilon\ge K$,
    \begin{align*}
        n^{1-\varepsilon}\le K^{1-\varepsilon}u^{1-\varepsilon}
        \le u^\varepsilon u^{1-\varepsilon}=u\le s_0,
    \end{align*}
    so the collar bound applies and satisfies every hypothesis of the
    amplification bound with $M=1$ and the common constant $\max(C,1)$. Its
    radius $r$ satisfies
    $\lfloor n^{1-\varepsilon}\rfloor+2r<K^{-\beta_0}n^{\beta_0}\le u^{\beta_0}$
    because $n\ge u\ge N_r$. A walk of length at most $2r$ moves by at most $2r$ in
    sup norm, so every point of the graph neighborhood $N_{2r}(S_0)$ lies within
    $\lfloor n^{1-\varepsilon}\rfloor+2r$ of $T$. Hence $E$ avoids
    $N_{2r}(S_0)$, and $I_\Omega(X:E)\le C_an^{-1}\le\max(C_a,0)$.
\end{proof}

\begin{theorem}[Coarse information, assuming the sublinear collar bound]
    \label{thm:peps_coarse_information_of_collar}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le_of_sublinearCollarBound}
    \leanok
    \uses{def:peps_box_union,def:area_law_collar_bound,
        thm:area_law_amplification,lem:peps_box_union_template,
        thm:area_law_uniform_collar_threshold}
    Fix $q,R,J$ and $\Delta>0$, and assume the sublinear collar bound for some
    template constants $D_0,C_{\rm tpl}$. Fix $N_0$ and $A_0$. There are
    $\beta<1$, $C$ and $u_0$ with the following property. Let $u\ge u_0$, let
    $\Omega$ be a unit ground vector with full-system gap $\Delta$ of an
    admissible Hamiltonian of range $R$ and bound $J$ on a finite induced
    domain $\Lambda$, and let $T$ be a union of at most $N_0$ boxes with
    $\operatorname{diam}_\infty T\le A_0u$. Then $I_\Omega(X:E)\le C$ for every
    $X\subseteq\Lambda\cap T$ and every $E\subseteq\Lambda$ with
    $\operatorname{dist}_\infty(E,T)>u^\beta$. The constants are chosen before
    $u$, $\Lambda$, the Hamiltonian, $T$, $X$ and $E$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_coarse_information_of_collar_explicit,thm:area_law_amplification,
        def:peps_coarse_information_constants,lem:peps_coarse_exponent_lt_one}
    Choose a constant $C$ and a threshold $N$ for the collar bound, and by
    Theorem~\ref{thm:area_law_amplification} a constant $C_a$ and a threshold
    $N_a$ for the amplification bound at $(\max\{C,1\},1)$.
    Theorem~\ref{thm:peps_coarse_information_of_collar_explicit} gives the
    claim with $\beta=\beta_0<1$, the constant $\max\{C_a,0\}$ and the coarse
    threshold of Definition~\ref{def:peps_coarse_information_constants}.
\end{proof}

\begin{theorem}[Coarse information on the square grid, assuming the collar bound]
    \label{thm:peps_coarse_information_square}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le_square_of_sublinearCollarBound,
        TNLean.PEPS.Approximation.finiteProductMutualInformation_relabel}
    \leanok
    \uses{thm:peps_coarse_information_of_collar,def:polynomial_peps_model,
        lem:peps_box_union_template}
    Fix $q$, $J\ge0$ and $\Delta>0$, and assume the sublinear collar bound at
    range one. Fix $N_0$ and $A_0$. There are $\beta<1$, $C$ and $u_0$ such
    that for every $u\ge u_0$, every open square grid with a Hamiltonian of the
    model, every unit ground vector $\Omega$ with full-system gap $\Delta$,
    every translation $c\in\mathbb Z^2$ of the lattice coordinates, every union
    $T$ of at most $N_0$ boxes with $\operatorname{diam}_\infty T\le A_0u$,
    every set $X$ of sites whose translated coordinates lie in $T$, and every
    set $E$ of sites whose translated coordinates are at distance greater than
    $u^\beta$ from $T$, one has $I_\Omega(X:E)\le C$.
\end{theorem}
\begin{proof}
    \leanok
    Translating $T$ by $-c$ and placing the Hamiltonian on the integer
    rectangle preserves the ground vector, the gap, and the entropies of
    relabelled regions, so mutual information is preserved as well and
    Theorem~\ref{thm:peps_coarse_information_of_collar} applies.
\end{proof}

\begin{theorem}[Coarse information on finite induced domains]
    \label{thm:peps_coarse_information_finite_domain}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le}
    \leanok
    \uses{def:peps_box_union,def:area_law_hamiltonian}
    Fix integers $q\ge1$, $R\ge0$ and $N_0\ge0$, real numbers
    $J\ge0$, $\Delta>0$ and $A_0$. There are $\beta<1$, $C$ and $u_0$
    with the following property. Let $u\ge u_0$, let $\Lambda$ be a finite
    induced domain, and let $\Omega$ be a normalized ground vector with
    full-system gap $\Delta$ of an admissible Hamiltonian of range $R$
    and bound $J$ on $\Lambda$. Let $T$ be a union of at most $N_0$ lattice
    boxes with $\operatorname{diam}_\infty T\le A_0u$. Then
    $I_\Omega(X:E)\le C$ whenever $X\subseteq\Lambda\cap T$,
    $E\subseteq\Lambda$ and $\operatorname{dist}_\infty(T,E)>u^\beta$.
    The constants precede the scale, domain, Hamiltonian, ground vector,
    rectangle union and physical regions. Empty sampled regions are included.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_coarse_information_of_collar,thm:area_law_sublinear_collar}
    With $D_0=2R+11$ and $C_{\rm tpl}=24$ in the physical collar theorem,
    Theorem~\ref{thm:peps_coarse_information_of_collar} applies to the cut
    given by the whole physical domain, whose crossing-edge endpoint set is
    empty.
\end{proof}

    \label{def:peps_closed_collar_constants}
    \lean{TNLean.PEPS.Approximation.collarBoxConstantClosed,
        TNLean.PEPS.Approximation.collarConstantClosed,
        TNLean.PEPS.Approximation.collarThresholdClosed}
    \leanok
    \uses{def:area_law_collar_bound}
    For $q,R\in\mathbb N$ and $J,\Delta\in\mathbb R$, let
    $C_{\rm box}(q,R,J,\Delta)$ be the safe-box constant of
    \cite[Proposition~9.5]{OpenAI2026AreaLaw} at range $R$, at the transport
    constants with threshold $1/8$, entropy constant $200000$ with exponent
    $2$, and energy constant $3(18\pi^2+6144e^{\pi^2+1})$ with exponent $4$.
    Let $C(q,R,J,\Delta)$ and $N(q,R,J,\Delta)$ be the constant and threshold
    of the sublinear collar bound of \cite[Proposition~9.6]{OpenAI2026AreaLaw}
    at range $R$, at this safe-box constant, the truncation constant at the
    factor $10$ and these transport constants.
\end{definition}

\begin{theorem}[The sublinear collar bound at range $R$ at closed-form constants]
    \label{thm:peps_collar_closed}
    \lean{TNLean.PEPS.Approximation.sublinearCollarBoundAt_closed}
    \leanok
    \uses{def:area_law_collar_bound,def:peps_closed_collar_constants}
    Fix $q,R\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$ and $D_0>2R+10$.
    The sublinear collar bound at range $R$, clearance $D_0$ and template
    constant $24$ holds with constant $C(q,R,J,\Delta)$ and threshold
    $N(q,R,J,\Delta)$. Neither depends on $D_0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_sublinear_collar_explicit,
        thm:area_law_transport_bound_explicit}
    The stated transport constants satisfy the common step-error bound by
    Theorem~\ref{thm:area_law_transport_bound_explicit}, and
    Theorem~\ref{thm:area_law_sublinear_collar_explicit} at these constants
    and $C_{\rm tpl}=24$ gives the collar bound with constant $C$ and
    threshold $N$.
\end{proof}

\begin{definition}[Closed-form coarse constants at range $R$]
    \label{def:peps_closed_coarse_constants}
    \lean{TNLean.PEPS.Approximation.coarseInformationConstantClosed,
        TNLean.PEPS.Approximation.coarseInformationThresholdClosed}
    \leanok
    \uses{def:peps_closed_collar_constants,
        def:peps_collar_amplification_constants,
        def:peps_coarse_information_constants}
    For $q,R,N_0\in\mathbb N$ and $J,\Delta,A_0\in\mathbb R$, put
    $C=C(q,R,J,\Delta)$ and $N=N(q,R,J,\Delta)$, and let
    $C_a=C_a(q,R,J,\Delta,C)$ and $N_a=N_a(q,R,J,\Delta,C)$ be the
    amplification constants at the collar constant. The closed coarse constant
    is $C_{\rm coarse}(q,R,J,\Delta)=\max\{C_a,0\}$, and the closed coarse
    threshold $u_0(q,R,J,\Delta,N_0,A_0)$ is the coarse threshold at the
    template constant $24$, the threshold $N$ and $(C_a,N_a)$.
\end{definition}

\begin{theorem}[Coarse information at range $R$ at closed-form constants]
    \label{thm:peps_coarse_information_closed}
    \lean{TNLean.PEPS.Approximation.coarse_mutualInformation_le_closed}
    \leanok
    \uses{def:peps_box_union,def:area_law_hamiltonian,
        def:peps_closed_coarse_constants}
    Fix $q,R,N_0\in\mathbb N$ with $q\ge1$, real numbers $J\ge0$, $\Delta>0$
    and $A_0$, and put $u_0=u_0(q,R,J,\Delta,N_0,A_0)$. Let $u\ge u_0$, let
    $\Lambda$ be a finite induced domain, and let $\Omega$ be a normalized
    ground vector with full-system gap $\Delta$ of an admissible Hamiltonian
    of range $R$ and bound $J$ on $\Lambda$. Let $T$ be a union of at most
    $N_0$ lattice boxes with $\operatorname{diam}_\infty T\le A_0u$. Then
    $I_\Omega(X:E)\le C_{\rm coarse}(q,R,J,\Delta)$ whenever
    $X\subseteq\Lambda\cap T$, $E\subseteq\Lambda$ and
    $\operatorname{dist}_\infty(T,E)>u^{\beta_0}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_coarse_information_of_collar_explicit,thm:peps_collar_closed,
        thm:peps_collar_amplification_bound}
    By Theorem~\ref{thm:peps_collar_closed} at the clearance $2R+11$, the
    collar bound holds with constant $C$ and threshold $N$, and by
    Theorem~\ref{thm:peps_collar_amplification_bound} the amplification bound
    at $(\max\{C,1\},1)$ holds with constant $C_a$ and threshold $N_a$.
    Theorem~\ref{thm:peps_coarse_information_of_collar_explicit} then gives
    the claim.
\end{proof}
