\section{Removing finitely many exceptional sizes}
\label{sec:peps_exact_finite_exception}

For natural numbers $q,L$, real numbers $C,c$, and a vector $\Omega$ in the
physical space $(\mathbb C^q)^{\otimes L^2}$ of the open $L\times L$ square,
write $\mathcal A_{C,c}(L,\Omega)$ if there are a
nonzero PEPS $\Phi$ on the original square graph and a real phase $\theta$
such that every edge dimension satisfies $1\le D_e\le CL^c$ and
\begin{align}
    \bigl\|\|\Phi\|^{-1}\Phi-e^{i\theta}\Omega\bigr\|\le L^{-1}.
    \label{eq:peps_finite_error}
\end{align}
For $L=0$ the square has no edges, so the edge bound is vacuous, and the
right-hand side $L^{-1}$ is read with $0^{-1}=0$.

For real $J,\Delta$, let $\mathcal G_{q,J,\Delta}(L)$ be the set of triples
$(H,E_0,\Omega)$ with the following properties. The Hamiltonian is a sum of
site terms and nearest-neighbor edge terms supported on their designated
sites or endpoints, each of operator norm at most $J$. Its total sum is
Hermitian; individual terms need not be. The remaining conditions are
\begin{align}
    \|\Omega\| & =1,\label{eq:peps_finite_ground}                           \\
    H\Omega    & =E_0\Omega,\notag                                          \\
    H-E_0I     & \ge\Delta\bigl(I-|\Omega\rangle\langle\Omega|\bigr).\notag
\end{align}

\begin{theorem}[Increasing the polynomial prefactor]
    \label{thm:peps_prefactor_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_prefactor}
    \leanok
    Let $q,L$ be natural numbers, $C,C',c$ real numbers with $C\le C'$, and
    $\Omega$ a vector on the open square. If $\mathcal A_{C,c}(L,\Omega)$,
    then $\mathcal A_{C',c}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Keep the same PEPS and phase. For every edge, $D_e\le CL^c\le C'L^c$
    because $L^c\ge0$. The normalized vector and its error are unchanged.
\end{proof}

\begin{theorem}[A uniform prefactor for every size]
    \label{thm:peps_remove_finite_exceptions}
    \lean{TNLean.PEPS.Approximation.exists_uniform_approximation_of_sufficiently_large}
    \leanok
    Fix integers $q\ge1$ and $L_0\ge0$, real $J,\Delta,C$, and $c\ge0$.
    Suppose $\mathcal A_{C,c}(L,\Omega)$ holds for every integer
    $L\ge\max\{2,L_0\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$.
    There is one $C'>0$ with $C'\ge C$ such that
    $\mathcal A_{C',c}(L,\Omega)$ holds for every integer $L\ge2$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$.
    The exponent $c$ is unchanged, and $C'$ is independent of $L,H,E_0,\Omega$.
    This is the finite-size argument
    in~\cite[Section~8]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 203--214).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_uniform_small_size_approximation,thm:peps_prefactor_monotone}
    Theorem~\ref{thm:peps_uniform_small_size_approximation} gives a positive
    $C'\ge C$ covering all $2\le L<L_0$ and all unit vectors. Its choice is
    $C'=\max\{C,q^{L_0^2}\}$. For $L<L_0$, apply that theorem using
    $\|\Omega\|=1$ from~(\ref{eq:peps_finite_ground}). For $L\ge L_0$,
    apply the assumed approximation and
    Theorem~\ref{thm:peps_prefactor_monotone}. Both cases use the same $C'$.
\end{proof}

\begin{theorem}[Equivalence with the sufficiently large size statement]
    \label{thm:peps_approximation_iff_large}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_iff_sufficiently_large}
    \leanok
    The uniform polynomial approximation assertion
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;\label{eq:peps_finite_uniform} \\
         & \quad\forall L\ge2\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag                           \\
         & \qquad\mathcal A_{C,c}(L,\Omega)\notag
    \end{align}
    is equivalent to
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;
        \exists L_0\in\mathbb N\;\label{eq:peps_finite_large}     \\
         & \quad\forall L\ge\max\{2,L_0\}\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag \\
         & \qquad\mathcal A_{C,c}(L,\Omega).\notag
    \end{align}
    In both assertions, the constants $C,c$ are independent of the size,
    Hamiltonian, ground energy, and ground vector.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_remove_finite_exceptions}
    Assertion~(\ref{eq:peps_finite_uniform}) gives
    (\ref{eq:peps_finite_large}) with $L_0=0$ and the same constants.
    Conversely, fix $q,J,\Delta$ and choose $C,c,L_0$ from
    (\ref{eq:peps_finite_large}).
    Theorem~\ref{thm:peps_remove_finite_exceptions} produces one positive
    $C'\ge C$ for every $L\ge2$, with the same positive exponent $c$.
    These constants prove~(\ref{eq:peps_finite_uniform}).
\end{proof}

\subsection{Approximation from a compressed dyadic operator network}
\label{sec:peps_ground_state_assembly}

For large sizes, \cite[Section~8.1]{OpenAI2026PolynomialPEPS}
(\texttt{07-assembly.tex}, lines 9--63) obtains from the protocol of its
Proposition~7.1 and its compression theorem a party operator network with
polynomial link dimensions whose operator is within $CL^{-5}$ of
$|\Omega\rangle\langle\Omega|$ in trace norm. The statement below records this
output as an assumption and derives the uniform approximation from it. The
assumption is not proved here, so the result is a conditional form of
Theorem~\ref{thm:openai_polynomial_peps}, not that theorem.

\begin{theorem}[Increasing the polynomial exponent]
    \label{thm:peps_exponent_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_exponent}
    \leanok
    Let $L\ge1$, $C\ge0$ and $c\le c'$. If $\mathcal A_{C,c}(L,\Omega)$, then
    $\mathcal A_{C,c'}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Keep the same PEPS and phase; $CL^c\le CL^{c'}$ because $L\ge1$ and
    $C\ge0$.
\end{proof}

\begin{definition}[Compressed dyadic operator networks for large sizes]
    \label{def:peps_large_size_dyadic_operator_network}
    \lean{TNLean.PEPS.Approximation.LargeSizeDyadicOperatorNetwork}
    \leanok
    \uses{def:peps_dyadic_hv_routing, def:peps_party_operator_fix_input}
    Say that compressed dyadic operator networks exist for large sizes if for
    every integer $q\ge2$ and all reals
    $J,\Delta>0$ there are reals $C,C_1,k$ and natural numbers $B,\Delta_0,c,L_0$
    such that for every integer $L\ge\max\{2,L_0\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$ there is an operator party
    network with the following properties. Every party is attached to a dyadic
    block inside the padded square, with at most $B$ parties per block; every
    party has at most $\Delta_0$ links; the two anchors of every link differ in
    each coordinate by at most $c$ times the side of either endpoint block; every
    link dimension is at most $C_1L^k$; and the operator $\sigma$ of the network,
    with the physical pair of each party read at the fold of its anchor, satisfies
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le CL^{-5}.
    \end{align}
    The constants are independent of $L,H,E_0,\Omega$. This is the conclusion
    of~\cite[Section~8.1]{OpenAI2026PolynomialPEPS} (\texttt{07-assembly.tex},
    lines 53--63), with the anchors of lines 102--114 and the network properties
    of Proposition~7.1 used in the proof of Lemma~8.2 (lines 128--131).
\end{definition}

\begin{theorem}[Uniform approximation from compressed dyadic operator networks]
    \label{thm:peps_ground_state_approximation_conditional}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_of_largeSizeDyadicOperatorNetwork}
    \leanok
    \uses{def:peps_large_size_dyadic_operator_network}
    If the assertion of
    Definition~\ref{def:peps_large_size_dyadic_operator_network} holds, then
    the uniform polynomial approximation assertion~(\ref{eq:peps_finite_uniform})
    holds: for every $q\ge2$ and $J,\Delta>0$ there are $C',c'>0$ such that
    $\mathcal A_{C',c'}(L,\Omega)$ for every $L\ge2$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$. This is the argument
    of~\cite[Section~8]{OpenAI2026PolynomialPEPS} (\texttt{07-assembly.tex},
    lines 53--214) after the compressed network has been obtained.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_party_column_routing_polynomial, thm:peps_exponent_monotone, thm:peps_approximation_iff_large}
    By Theorem~\ref{thm:peps_approximation_iff_large} it suffices to treat large
    $L$. Fix $q,J,\Delta$ and take the constants of the assumption. Replace
    $C_1$ by $\max\{C_1,1\}$ and $k$ by $\max\{k,0\}$; the link bound persists
    since $L\ge1$. For $L\ge\max\{2,L_0,2C\}$ one has $2C\le L\le L^4$, hence
    $2CL^{-5}\le L^{-1}\le1/2$ and $CL^{-5}<1$.
    Theorem~\ref{thm:peps_party_column_routing_polynomial} with $\eta=CL^{-5}$
    gives $\mathcal A_{C_1^\chi,k\chi}(L,\Omega)$ with $\chi=8(2c+1)B\Delta_0$,
    and Theorem~\ref{thm:peps_exponent_monotone} raises the exponent to
    $k\chi+1>0$. The constants $C_1^\chi$, $k\chi+1$ and the threshold depend only
    on $q,J,\Delta$.
\end{proof}

The compressed network itself comes from the protocol of
\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} and the compression theorem
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}. The following statements carry out
this step of~\cite[Section~8.1]{OpenAI2026PolynomialPEPS}
(\texttt{07-assembly.tex}, lines 45--63) from the protocol assertion of
Definition~\ref{def:peps_dyadic_protocol}. That assertion is weaker than
Proposition~7.1 and is not proved, so the final statement remains a
conditional form of Theorem~\ref{thm:openai_polynomial_peps}.

\begin{lemma}[Operator party networks from compressed protocol densities]
    \label{lem:peps_protocol_compressed_party_network}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_partyNetwork_of_contractedOperator,
        TNLean.PEPS.Approximation.DyadicProtocol.toGateMapConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalDensity_toGateMapConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalOut_foldVertex}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_dyadic_hv_routing,
        def:compression_operator_finite_coordinates}
    Let a protocol satisfy the bounds of
    Definition~\ref{def:peps_dyadic_protocol} with $C\ge0$ on a square of side
    $L\ge1$, for a unit vector $\Omega$. Composing its changes gives one
    distributed construction, with the gate operators equal to their recorded
    expansions, whose physical density is that of the final vector, the
    physical configurations of the parties being identified with the
    configurations of the square. Let $\tau$ be a distributed operator contraction
    on the parties of the protocol with
    $\lVert\tau-\Tr_{\rm private}|\Psi_n\rangle\langle\Psi_n|\rVert_1\le\varepsilon$,
    link alphabets of size at most $C_1L^\alpha$, at most $\Delta_0$ links at each
    party, and links only between two parties of a common nonprivate gate. Then
    enumerating the link alphabets gives an operator party network on the dyadic
    anchors of the protocol with the same link bounds, separation constant $b$,
    and, with the physical pair of each party read at the fold of its anchor,
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le\varepsilon+(2C^2+2C)L^{-10}.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_dyadic_protocol_consequences, thm:compression_operator_finite_coordinates}
    The unit-square party of a genuine site has anchor equal to that site, which
    the fold fixes, and every other party has physical dimension one. Hence
    reading each party at the fold of its anchor recovers the configuration of
    the square, and Theorem~\ref{thm:compression_operator_finite_coordinates}
    identifies $\sigma$ with $\tau$ in these coordinates. The trace norm is
    invariant under this relabelling, so the triangle inequality and the density
    estimate of Theorem~\ref{thm:peps_dyadic_protocol_consequences} give the bound.
    The endpoints of a link are participants of a common nonprivate gate of some
    change, so the separation bound of the same theorem applies, and each party
    lies on at most $\Delta_0$ links.
\end{proof}

\begin{theorem}[Compressed dyadic operator networks from the protocol assertion]
    \label{thm:peps_large_size_network_from_protocol}
    \lean{TNLean.PEPS.Approximation.largeSizeDyadicOperatorNetwork_of_largeSizeDyadicProtocolConsequences}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_large_size_dyadic_operator_network}
    If the protocol assertion of Definition~\ref{def:peps_dyadic_protocol} holds,
    then the assertion of
    Definition~\ref{def:peps_large_size_dyadic_operator_network} holds.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_uniform_approximate_operator_network, lem:peps_protocol_compressed_party_network}
    Fix $q,J,\Delta$ and take the constants $C,b,k,r,L_0$ of the assertion. The
    protocols of all sides $L\ge\max\{2,L_0\}$, Hamiltonians and ground vectors
    form one family of distributed constructions indexed by their size. In this
    family the number of nonprivate gates is at most $bCL^3$, each gate involves
    at most $b$ parties, each party participates in at most $b$ gates, every
    physical dimension is at most $q$, and every gate is exactly the sum of its
    recorded expansion, with at most $CL^k$ monomials, coefficients of modulus at
    most $CL^k$, and at most $r$ pair effects per monomial. These expansions serve
    at every accuracy. Theorem~\ref{thm:peps_uniform_approximate_operator_network}
    with $p=5$ then gives constants $C_p,\alpha$, chosen before the family member,
    and for each protocol a distributed operator contraction within $L^{-5}$ of its
    physical density, with link alphabets of size at most $C_pL^\alpha$, at most
    $2b(b-1)$ links at each party, and links only between parties of a common
    gate. Lemma~\ref{lem:peps_protocol_compressed_party_network} turns it into an
    operator party network with at most $b$ parties per block, degree $2b(b-1)$,
    separation constant $b$, link dimensions at most $C_pL^\alpha$, and
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le L^{-5}+(2C^2+2C)L^{-10}\le(1+2C^2+2C)L^{-5}.
    \end{align}
    Here $C\ge0$, since there are at least one and at most $CL^2$ parties.
\end{proof}

\begin{corollary}[Uniform approximation from the protocol assertion]
    \label{cor:peps_ground_state_approximation_from_protocol}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_of_largeSizeDyadicProtocolConsequences}
    \leanok
    \uses{def:peps_dyadic_protocol}
    If the protocol assertion of Definition~\ref{def:peps_dyadic_protocol} holds,
    then the uniform polynomial approximation
    assertion~(\ref{eq:peps_finite_uniform}) holds.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:peps_large_size_network_from_protocol, thm:peps_ground_state_approximation_conditional}
    Combine Theorem~\ref{thm:peps_large_size_network_from_protocol} with
    Theorem~\ref{thm:peps_ground_state_approximation_conditional}.
\end{proof}
