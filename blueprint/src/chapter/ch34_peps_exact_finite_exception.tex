\section{Removing finitely many exceptional sizes}
\label{sec:peps_exact_finite_exception}

For natural numbers $q,L$, real numbers $C,c$, and a vector $\Omega$ in the
physical space $(\mathbb C^q)^{\otimes L^2}$ of the open $L\times L$ square,
write $\mathcal A_{C,c}(L,\Omega)$ if there are a
nonzero PEPS $\Phi$ on the original square graph and a real phase $\theta$
such that every edge dimension satisfies $1\le D_e\le CL^c$ and
\begin{align}
    \bigl\|\|\Phi\|^{-1}\Phi-e^{i\theta}\Omega\bigr\|\le L^{-1}.
    \label{eq:peps_finite_error}
\end{align}
For $L=0$ the square has no edges, so the edge bound is vacuous, and the
right-hand side $L^{-1}$ is read with $0^{-1}=0$.

For real $J,\Delta$, let $\mathcal G_{q,J,\Delta}(L)$ be the set of triples
$(H,E_0,\Omega)$ with the following properties. The Hamiltonian is a sum of
site terms and nearest-neighbor edge terms supported on their designated
sites or endpoints, each of operator norm at most $J$. Its total sum is
Hermitian; individual terms need not be. The remaining conditions are
\begin{align}
    \|\Omega\| & =1,\label{eq:peps_finite_ground}                           \\
    H\Omega    & =E_0\Omega,\notag                                          \\
    H-E_0I     & \ge\Delta\bigl(I-|\Omega\rangle\langle\Omega|\bigr).\notag
\end{align}

\begin{theorem}[Increasing the polynomial prefactor]
    \label{thm:peps_prefactor_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_prefactor}
    \leanok
    Let $q,L$ be natural numbers, $C,C',c$ real numbers with $C\le C'$, and
    $\Omega$ a vector on the open square. If $\mathcal A_{C,c}(L,\Omega)$,
    then $\mathcal A_{C',c}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Retaining the same PEPS and phase preserves the normalized vector and its
    approximation error, while $L^c\ge0$ implies $D_e\le CL^c\le C'L^c$ on
    every edge.
\end{proof}

\begin{lemma}[Removing the small sizes at a named prefactor]
    \label{lem:peps_remove_small_sizes_named}
    \lean{TNLean.PEPS.Approximation.hasPEPSApproximation_of_sufficiently_large_le}
    \leanok
    Fix integers $q\ge1$, $L\ge2$ and $L_0\ge0$, real $C$ and $c\ge0$, and a
    unit vector $\Omega$ on the square of side $L$. If
    $\mathcal A_{C,c}(L,\Omega)$ holds whenever $L\ge L_0$, then
    $\mathcal A_{C',c}(L,\Omega)$ holds with $C'=\max\{C,q^{L_0^2}\}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_exact_small_size_approximation,thm:peps_prefactor_monotone}
    If $L<L_0$, Theorem~\ref{thm:peps_exact_small_size_approximation} gives
    $\mathcal A_{C',c}(L,\Omega)$. Otherwise the assumed approximation
    combines with Theorem~\ref{thm:peps_prefactor_monotone}, since $C\le C'$.
\end{proof}

\begin{theorem}[A uniform prefactor for every size]
    \label{thm:peps_remove_finite_exceptions}
    \lean{TNLean.PEPS.Approximation.exists_uniform_approximation_of_sufficiently_large}
    \leanok
    Following the finite-size argument
    in~\cite[Section~8]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 203--214), fix integers $q\ge1$ and
    $L_0\ge0$, real $J,\Delta,C$, and $c\ge0$.
    Suppose $\mathcal A_{C,c}(L,\Omega)$ holds for every integer
    $L\ge\max\{2,L_0\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$.
    There is one $C'>0$ with $C'\ge C$ such that
    $\mathcal A_{C',c}(L,\Omega)$ holds for every integer $L\ge2$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$.
    The exponent $c$ is unchanged, and $C'$ is independent of $L,H,E_0,\Omega$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_remove_small_sizes_named}
    Take $C'=\max\{C,q^{L_0^2}\}$, which is positive and at least $C$.
    Since $\|\Omega\|=1$ by~(\ref{eq:peps_finite_ground}),
    Lemma~\ref{lem:peps_remove_small_sizes_named} applies to every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$ with $L\ge2$.
\end{proof}

\begin{theorem}[Equivalence with the sufficiently large size statement]
    \label{thm:peps_approximation_iff_large}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_iff_sufficiently_large}
    \leanok
    The uniform polynomial approximation assertion
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;\label{eq:peps_finite_uniform} \\
         & \quad\forall L\ge2\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag                           \\
         & \qquad\mathcal A_{C,c}(L,\Omega)\notag
    \end{align}
    is equivalent to
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;
        \exists L_0\in\mathbb N\;\label{eq:peps_finite_large}     \\
         & \quad\forall L\ge\max\{2,L_0\}\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag \\
         & \qquad\mathcal A_{C,c}(L,\Omega).\notag
    \end{align}
    In both assertions, the constants $C,c$ are independent of the size,
    Hamiltonian, ground energy, and ground vector.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_remove_finite_exceptions}
    Assertion~(\ref{eq:peps_finite_uniform}) implies
    (\ref{eq:peps_finite_large}) with $L_0=0$ and the same constants.
    Conversely, for fixed $q,J,\Delta$ and constants $C,c,L_0$ satisfying
    (\ref{eq:peps_finite_large}),
    Theorem~\ref{thm:peps_remove_finite_exceptions} produces a single positive
    prefactor $C'\ge C$ valid for every $L\ge2$ with the same positive exponent
    $c$, establishing~(\ref{eq:peps_finite_uniform}).
\end{proof}

\subsection{Approximation from a compressed dyadic operator network}
\label{sec:peps_ground_state_assembly}

For large sizes, \cite[Section~8.1]{OpenAI2026PolynomialPEPS}
(\texttt{07-assembly.tex}, lines 9--63) obtains from the protocol of its
Proposition~7.1 and its compression theorem a party operator network with
polynomial link dimensions whose operator is within $CL^{-5}$ of
$|\Omega\rangle\langle\Omega|$ in trace norm. The statement below records this
output, at constants fixed in advance, as an assumption and derives from it the
approximation for all sufficiently large sizes. It is one step of the proof of
Theorem~\ref{thm:openai_polynomial_peps}, not that theorem.

\begin{theorem}[Increasing the polynomial exponent]
    \label{thm:peps_exponent_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_exponent}
    \leanok
    Let $L\ge1$, $C\ge0$ and $c\le c'$. If $\mathcal A_{C,c}(L,\Omega)$, then
    $\mathcal A_{C,c'}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Retaining the same PEPS and phase preserves the approximation error, while
    $L\ge1$ and $C\ge0$ yield $CL^c\le CL^{c'}$.
\end{proof}

\begin{definition}[Compressed dyadic operator networks at given constants]
    \label{def:peps_large_size_dyadic_operator_network}
    \lean{TNLean.PEPS.Approximation.DyadicOperatorNetworkAt}
    \leanok
    \uses{def:peps_dyadic_hv_routing, def:peps_party_operator_fix_input}
    Capturing the conclusion of~\cite[Section~8.1]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 53--63), together with the anchors of
    lines 102--114 and the network properties of Proposition~7.1 used in the
    proof of Lemma~8.2 (lines 128--131), fix an integer $q\ge2$, reals
    $J,\Delta>0$, reals $C,C_1,k$ and natural numbers $B,\Delta_0,c,L_0$.
    Compressed dyadic operator networks exist at these constants if for every
    integer $L\ge\max\{2,L_0\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$ there is an operator party
    network with the following properties. Every party is attached to a dyadic
    block inside the padded square, with at most $B$ parties per block; every
    party has at most $\Delta_0$ links; the two anchors of every link differ in
    each coordinate by at most $c$ times the side of either endpoint block; every
    link dimension is at most $C_1L^k$; and the operator $\sigma$ of the network,
    with the physical pair of each party read at the fold of its anchor, satisfies
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le CL^{-5}.
    \end{align}
\end{definition}

\begin{theorem}[Large-size approximation from compressed dyadic operator networks]
    \label{thm:peps_ground_state_approximation_conditional}
    \lean{TNLean.PEPS.Approximation.hasPEPSApproximation_of_dyadicOperatorNetworkAt}
    \leanok
    \uses{def:peps_large_size_dyadic_operator_network}
    Following~\cite[Section~8]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 53--206) once the compressed network has
    been obtained, if compressed dyadic operator networks exist at the constants
    $C,C_1,k,B,\Delta_0,c,L_0$ in the sense of
    Definition~\ref{def:peps_large_size_dyadic_operator_network}, and
    $\chi=8(2c+1)B\Delta_0$, then
    $\mathcal A_{\max\{C_1,1\}^\chi,\max\{k,0\}\chi+1}(L,\Omega)$ for every
    $L\ge2$ with $L\ge\max\{L_0,\lceil2C\rceil\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_party_column_routing_polynomial, thm:peps_exponent_monotone}
    Replacing $C_1$ by $\max\{C_1,1\}$ and $k$ by $\max\{k,0\}$ preserves the
    link bound since $L\ge1$. For $L\ge\max\{2,L_0,\lceil2C\rceil\}$ one has
    $2C\le L\le L^4$, hence $2CL^{-5}\le L^{-1}\le1/2$ and $CL^{-5}<1$.
    Theorem~\ref{thm:peps_party_column_routing_polynomial} with $\eta=CL^{-5}$
    gives $\mathcal A_{C_1^\chi,k\chi}(L,\Omega)$ with $\chi=8(2c+1)B\Delta_0$,
    and Theorem~\ref{thm:peps_exponent_monotone} raises the exponent to
    $k\chi+1>0$.
\end{proof}

The compressed network itself comes from the protocol of
\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} and the compression theorem
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}. The following statements carry out
this step of~\cite[Section~8.1]{OpenAI2026PolynomialPEPS}
(\texttt{07-assembly.tex}, lines 45--63) from the protocol bounds of
Definition~\ref{def:peps_dyadic_protocol} at given constants, which
Proposition~\ref{prop:peps_dyadic_protocol} supplies.

\begin{lemma}[Operator party networks from compressed protocol densities]
    \label{lem:peps_protocol_compressed_party_network}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_partyNetwork_of_contractedOperator,
        TNLean.PEPS.Approximation.DyadicProtocol.toDistributedConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.toDistributedConstruction_circuit_induction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalDensity_toDistributedConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalOut_foldVertex}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_dyadic_hv_routing,
        def:compression_operator_finite_coordinates}
    Let a protocol satisfy the bounds of
    Definition~\ref{def:peps_dyadic_protocol} with count constant $C_N$ and error
    constant $C_E\ge0$ on a square of side $L\ge1$, for a unit vector $\Omega$.
    Composing its changes gives one
    distributed construction, with the recorded expansions of its gates, whose
    physical density is that of the final vector, the
    physical configurations of the parties being identified with the
    configurations of the square. Let $\tau$ be a distributed operator contraction
    on the parties of the protocol with
    $\lVert\tau-\Tr_{\rm private}|\Psi_n\rangle\langle\Psi_n|\rVert_1\le\varepsilon$,
    link alphabets of size at most $C_1L^\alpha$, at most $\Delta_0$ links at each
    party, and links only between two parties of a common nonprivate gate. Then
    enumerating the link alphabets gives an operator party network on the dyadic
    anchors of the protocol with the same link bounds, separation constant $b$,
    and, with the physical pair of each party read at the fold of its anchor,
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le\varepsilon+(2C_NC_E+2C_E)L^{-10}.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_dyadic_protocol_consequences, thm:compression_operator_finite_coordinates}
    The unit-square party of a genuine site has anchor equal to that site, which
    the fold fixes, and every other party has physical dimension one. Hence
    reading each party at the fold of its anchor recovers the configuration of
    the square, and Theorem~\ref{thm:compression_operator_finite_coordinates}
    identifies $\sigma$ with $\tau$ in these coordinates. The trace norm is
    invariant under this relabelling, so the triangle inequality and the density
    estimate of Theorem~\ref{thm:peps_dyadic_protocol_consequences} give the bound.
    The endpoints of a link are participants of a common nonprivate gate of some
    change, so the separation bound of the same theorem applies, and each party
    lies on at most $\Delta_0$ links.
\end{proof}

\begin{lemma}[Bounds on the composed protocol circuit]
    \label{lem:peps_protocol_circuit_bounds}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_participants_toDistributedConstruction_circuit,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.nonprivateCount_toDistributedConstruction_circuit_le,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.participationCount_toDistributedConstruction_circuit_le,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.isMonomialBounded_toDistributedConstruction_circuit,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.isExpansionBounded_toDistributedConstruction_circuit}
    \leanok
    \uses{def:peps_dyadic_protocol}
    Let $W$ be the circuit obtained by composing the $T$ changes
    $W_0,\ldots,W_{T-1}$ of a protocol, with its initial and final register
    layouts identified with the input and output layouts of the protocol. Repeated
    occurrences of one gate are counted separately. Every nonprivate gate of $W$
    has the set of participating parties of a nonprivate gate of some $W_t$ with
    $t<T$. If the protocol satisfies the bounds of
    Definition~\ref{def:peps_dyadic_protocol} with expansion constant $C_K$ and
    constants $b,k,r$, then $W$ has at most $Tb$ nonprivate gates, every party
    participates in at most $b$ of them, and every nonprivate gate of $W$ has at
    most $\lfloor C_KL^k\rfloor$ monomials, at most $r$ pair effects in each
    monomial, and absolute coefficient sum at most $C_KL^k$.
\end{lemma}
\begin{proof}
    \leanok
    Identifying equal layouts changes neither the gates nor their participating
    sets, so it suffices to treat the composite of the changes. Its nonprivate
    gates are those of the changes, listed in order. Write $N(X)$ for the number of
    nonprivate gates of a circuit $X$ and $n_p(X)$ for the number of them in which
    the party $p$ participates. Hence $N(W)=\sum_{t<T}N(W_t)\le Tb$, and
    $n_p(W)=\sum_{t<T}n_p(W_t)\le b$ is the lifetime bound of the protocol. The monomial
    and expansion bounds hold for every change and therefore for every gate of
    the composite.
\end{proof}

\begin{theorem}[Compressed dyadic operator networks from the protocol bounds]
    \label{thm:peps_large_size_network_from_protocol}
    \lean{TNLean.PEPS.Approximation.dyadicOperatorNetworkAt_of_dyadicProtocolBoundsAt}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_large_size_dyadic_operator_network}
    Fix $q\ge2$, $J,\Delta>0$, a count constant $C_N$, an expansion constant
    $C_K$, an error constant $C_E$ and constants $b,k,r,L_0$. Suppose that for
    every $L\ge\max\{2,L_0\}$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$ there is a protocol
    satisfying the Section~8 bounds of
    Definition~\ref{def:peps_dyadic_protocol} with these constants. Then
    compressed dyadic operator networks exist, in the sense of
    Definition~\ref{def:peps_large_size_dyadic_operator_network}, at the error
    constant $1+2C_NC_E+2C_E$, the link prefactor
    $C_\Sigma(r,b,\max\{C_N,0\}b,q,\max\{C_K,0\},C_K)$, the link exponent
    $\alpha_{\mathrm{net}}(r,b,3,0,k,k,5)$, the block bound $b$, the degree
    bound $2b(b-1)$, the separation constant $b$ and the threshold $L_0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_distributed_construction_compression_of_le, def:peps_compression_constants,
        lem:peps_protocol_compressed_party_network,
        lem:peps_protocol_circuit_bounds}
    For fixed $q,J,\Delta$, the
    protocols across all sides $L\ge\max\{2,L_0\}$, Hamiltonians, and ground
    vectors form a single family of distributed constructions indexed by their
    size. In this
    family the number of nonprivate gates is at most $bC_NL^3$, each gate involves
    at most $b$ parties, each party participates in at most $b$ gates, every
    physical dimension is at most $q$, and every gate is exactly the sum of its
    recorded expansion, with at most $C_KL^k$ monomials, absolute coefficient sum
    at most $C_KL^k$, and at most $r$ pair effects per monomial.
    Theorem~\ref{thm:peps_distributed_construction_compression_of_le} in its
    coefficient-sum form, with $p=5$ and coefficient-sum degree $s=k$, then gives
    constants $C_p=C_\Sigma(r,b,\max\{C_N,0\}b,q,\max\{C_K,0\},C_K)$ and
    $\alpha=\alpha_{\mathrm{net}}(r,b,3,0,k,k,5)$, chosen before the family member,
    and for each protocol a distributed operator contraction within $L^{-5}$ of its
    physical density, with link alphabets of size at most $C_pL^\alpha$, at most
    $2b(b-1)$ links at each party, and links only between parties of a common
    gate. Lemma~\ref{lem:peps_protocol_compressed_party_network} turns it into an
    operator party network with at most $b$ parties per block, degree $2b(b-1)$,
    separation constant $b$, link dimensions at most $C_pL^\alpha$, and
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le L^{-5}+(2C_NC_E+2C_E)L^{-10}\le(1+2C_NC_E+2C_E)L^{-5}.
    \end{align}
    Here $C_N\ge0$, since there are at least one and at most $C_NL^2$ parties,
    and $C_E\ge0$, since $1-C_EL^{-57}\le\lVert\Xi\rVert\le1$. The error constant
    $1+2C_NC_E+2C_E$ does not involve the expansion constant.
\end{proof}
