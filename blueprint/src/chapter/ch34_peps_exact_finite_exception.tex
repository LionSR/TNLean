\section{Removing finitely many exceptional sizes}
\label{sec:peps_exact_finite_exception}

For $q,L\in\N$, $C,c\in\R$, and $\Omega\in(\mathbb C^q)^{\otimes L^2}$ on the
open $L\times L$ square, write $\mathcal A_{C,c}(L,\Omega)$ if there exist a
nonzero PEPS $\Phi$ on the square graph with bond dimensions
$1\le D_e\le CL^c$ and $\theta\in\R$ such that (with $0^{-1}=0$)
\begin{align}
    \bigl\|\|\Phi\|^{-1}\Phi-e^{i\theta}\Omega\bigr\|\le L^{-1}.
    \label{eq:peps_finite_error}
\end{align}

For $J,\Delta\in\R$, let $\mathcal G_{q,J,\Delta}(L)$ be the set of triples
$(H,E_0,\Omega)$ where $H=H^\dagger=\sum_X h_X$ is a sum of site and
nearest-neighbor edge terms on the $L\times L$ square with $\|h_X\|\le J$, and
\begin{align}
    \|\Omega\| & =1,\label{eq:peps_finite_ground}                           \\
    H\Omega    & =E_0\Omega,\notag                                          \\
    H-E_0I     & \ge\Delta\bigl(I-|\Omega\rangle\langle\Omega|\bigr).\notag
\end{align}
A family of unit states $\mathcal F$ assigns to each $L$ a set
$\mathcal F(L)\subseteq\{\Omega\in(\mathbb C^q)^{\otimes L^2}:\|\Omega\|=1\}$;
in particular,
$\mathcal F_{q,J,\Delta}(L)=\{\Omega:\exists H,E_0,\,(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)\}$.

\begin{theorem}[Increasing the polynomial prefactor]
    \label{thm:peps_prefactor_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_prefactor}
    \leanok
    If $C\le C'$ and $\mathcal A_{C,c}(L,\Omega)$, then
    $\mathcal A_{C',c}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Since $L^c\ge0$, the same $(\Phi,\theta)$ satisfies
    $1\le D_e\le CL^c\le C'L^c$ and~(\ref{eq:peps_finite_error}).
\end{proof}

\begin{lemma}[Removing the small sizes at a named prefactor]
    \label{lem:peps_remove_small_sizes_named}
    \lean{TNLean.PEPS.Approximation.hasPEPSApproximation_of_sufficiently_large_le}
    \leanok
    Fix $q\ge1$, $L\ge2$, $L_0\ge0$, $C\in\R$, $c\ge0$, and unit vector
    $\Omega\in(\mathbb C^q)^{\otimes L^2}$. If
    $L\ge L_0\implies\mathcal A_{C,c}(L,\Omega)$, then
    $\mathcal A_{C',c}(L,\Omega)$ with $C'=\max\{C,q^{L_0^2}\}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_exact_small_size_approximation,thm:peps_prefactor_monotone}
    For $L<L_0$, Theorem~\ref{thm:peps_exact_small_size_approximation} gives
    $\mathcal A_{C',c}(L,\Omega)$; for $L\ge L_0$, combine
    $\mathcal A_{C,c}(L,\Omega)$ with Theorem~\ref{thm:peps_prefactor_monotone}
    using $C\le C'$.
\end{proof}

\begin{theorem}[A uniform prefactor for every size]
    \label{thm:peps_remove_finite_exceptions}
    \lean{TNLean.PEPS.Approximation.exists_uniform_approximation_of_sufficiently_large}
    \leanok
    Following~\cite[Section~8]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 203--214), fix $q\ge1$, $L_0\ge0$, a
    unit-state family $\mathcal F$, $C\in\R$, and $c\ge0$. If
    $\mathcal A_{C,c}(L,\Omega)$ holds for all $L\ge\max\{2,L_0\}$ and
    $\Omega\in\mathcal F(L)$, then there exists $C'\ge\max\{C,q^{L_0^2}\}>0$
    (independent of $L$ and $\Omega$) such that $\mathcal A_{C',c}(L,\Omega)$
    holds for all $L\ge2$ and $\Omega\in\mathcal F(L)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_remove_small_sizes_named}
    Apply Lemma~\ref{lem:peps_remove_small_sizes_named} with
    $C'=\max\{C,q^{L_0^2}\}>0$ to each $\Omega\in\mathcal F(L)$ for $L\ge2$.
\end{proof}

\begin{theorem}[Equivalence with the sufficiently large size statement]
    \label{thm:peps_approximation_iff_large}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_iff_sufficiently_large}
    \leanok
    The uniform polynomial approximation assertion
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;\label{eq:peps_finite_uniform} \\
         & \quad\forall L\ge2\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag                           \\
         & \qquad\mathcal A_{C,c}(L,\Omega)\notag
    \end{align}
    is equivalent to
    \begin{align}
         & \forall q\ge2\;\forall J,\Delta>0\;\exists C,c>0\;
        \exists L_0\in\mathbb N\;\label{eq:peps_finite_large}     \\
         & \quad\forall L\ge\max\{2,L_0\}\;
        \forall(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L),\notag \\
         & \qquad\mathcal A_{C,c}(L,\Omega).\notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_remove_finite_exceptions}
    Taking $L_0=0$ gives
    (\ref{eq:peps_finite_uniform})$\implies$(\ref{eq:peps_finite_large}).
    Conversely, applying Theorem~\ref{thm:peps_remove_finite_exceptions} to
    $\mathcal F_{q,J,\Delta}$ converts $(C,c,L_0)$ from
    (\ref{eq:peps_finite_large}) into $(C',c)$ satisfying
    (\ref{eq:peps_finite_uniform}).
\end{proof}

\subsection{Approximation from a compressed dyadic operator network}
\label{sec:peps_ground_state_assembly}

For large $L$, \cite[Section~8.1]{OpenAI2026PolynomialPEPS}
(\texttt{07-assembly.tex}, lines 9--63) combines Proposition~7.1 and the
compression theorem to obtain a party operator network with polynomial link
dimensions within $CL^{-5}$ of $|\Omega\rangle\langle\Omega|$ in trace norm.
The statements below derive the large-size approximation from this network
assumption as one step toward Theorem~\ref{thm:openai_polynomial_peps}.

\begin{theorem}[Increasing the polynomial exponent]
    \label{thm:peps_exponent_monotone}
    \lean{TNLean.PEPS.Approximation.HasPEPSApproximation.mono_exponent}
    \leanok
    For $L\ge1$, $C\ge0$, and $c\le c'$,
    $\mathcal A_{C,c}(L,\Omega)\implies\mathcal A_{C,c'}(L,\Omega)$.
\end{theorem}
\begin{proof}
    \leanok
    Since $L\ge1$ and $C\ge0$, the same $(\Phi,\theta)$ satisfies
    $D_e\le CL^c\le CL^{c'}$ and~(\ref{eq:peps_finite_error}).
\end{proof}

\begin{definition}[Compressed dyadic operator networks at given constants]
    \label{def:peps_large_size_dyadic_operator_network}
    \lean{TNLean.PEPS.Approximation.DyadicOperatorNetworkAt}
    \leanok
    \uses{def:peps_dyadic_hv_routing, def:peps_party_operator_fix_input}
    Following~\cite[Section~8.1, Proposition~7.1, and Lemma~8.2]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 53--63, 102--114, 128--131), fix $q\ge2$,
    a state family $\mathcal F$, $C,C_1,k\in\R$, and
    $B,\Delta_0,c,L_0\in\N$. Compressed dyadic operator networks exist for
    $\mathcal F$ at $(C,C_1,k,B,\Delta_0,c,L_0)$ if for every
    $L\ge\max\{2,L_0\}$ and $\Omega\in\mathcal F(L)$ there is an operator
    party network on the padded square with at most $B$ parties per dyadic
    block, degree at most $\Delta_0$, link anchor separation at most $c$ times
    either endpoint block side in each coordinate, link dimensions at most
    $C_1L^k$, and contracted operator $\sigma$ (with each party's physical
    pair read at the fold of its anchor) satisfying
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le CL^{-5}. \notag
    \end{align}
\end{definition}

\begin{theorem}[Large-size approximation from compressed dyadic operator networks]
    \label{thm:peps_ground_state_approximation_conditional}
    \lean{TNLean.PEPS.Approximation.hasPEPSApproximation_of_dyadicOperatorNetworkAt}
    \leanok
    \uses{def:peps_large_size_dyadic_operator_network}
    Following~\cite[Section~8]{OpenAI2026PolynomialPEPS}
    (\texttt{07-assembly.tex}, lines 53--206), if compressed dyadic operator
    networks exist for a unit-state family $\mathcal F$ at
    $(C,C_1,k,B,\Delta_0,c,L_0)$
    (Definition~\ref{def:peps_large_size_dyadic_operator_network}) and
    $\chi=8(2c+1)B\Delta_0$, then
    $\mathcal A_{\max\{C_1,1\}^\chi,\max\{k,0\}\chi+1}(L,\Omega)$ holds for
    all $L\ge\max\{2,L_0,\lceil2C\rceil\}$ and $\Omega\in\mathcal F(L)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_party_column_routing_polynomial, thm:peps_exponent_monotone}
    Replace $C_1,k$ by $\max\{C_1,1\},\max\{k,0\}$. For
    $L\ge\max\{2,L_0,\lceil2C\rceil\}$, $2C\le L\le L^4$ gives
    $2CL^{-5}\le L^{-1}\le1/2$ and $CL^{-5}<1$.
    Theorem~\ref{thm:peps_party_column_routing_polynomial} with $\eta=CL^{-5}$
    yields $\mathcal A_{C_1^\chi,k\chi}(L,\Omega)$, and
    Theorem~\ref{thm:peps_exponent_monotone} raises the exponent to
    $k\chi+1>0$.
\end{proof}

The next three statements derive
Definition~\ref{def:peps_large_size_dyadic_operator_network} from the protocol
bounds of Definition~\ref{def:peps_dyadic_protocol}
(supplied by Proposition~\ref{prop:peps_dyadic_protocol}) and
\cite[Proposition~7.1, Theorem~5.2, and Section~8.1]{OpenAI2026PolynomialPEPS}.

\begin{lemma}[Operator party networks from compressed protocol densities]
    \label{lem:peps_protocol_compressed_party_network}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_partyNetwork_of_contractedOperator,
        TNLean.PEPS.Approximation.DyadicProtocol.toDistributedConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.toDistributedConstruction_circuit_induction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalDensity_toDistributedConstruction,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalOut_foldVertex}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_dyadic_hv_routing,
        def:compression_operator_finite_coordinates}
    Let a protocol for a unit vector $\Omega$ on the $L\times L$ square
    ($L\ge1$) satisfy Definition~\ref{def:peps_dyadic_protocol} with constants
    $C_N$, $C_E\ge0$, and $b$, and final state $\Psi_n$. If $\tau$ is a
    distributed operator contraction on the protocol parties with
    $\lVert\tau-\Tr_{\rm private}|\Psi_n\rangle\langle\Psi_n|\rVert_1\le\varepsilon$,
    link dimensions at most $C_1L^\alpha$, degree at most $\Delta_0$, and links
    only between parties of a common nonprivate gate, then enumerating the link
    alphabets yields an operator party network on the protocol's dyadic anchors
    with the same link and degree bounds, separation constant $b$, and
    contracted operator $\sigma$ satisfying
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le\varepsilon+(2C_NC_E+2C_E)L^{-10}. \notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_dyadic_protocol_consequences, thm:compression_operator_finite_coordinates}
    Reading each unit-square party at its fold-fixed site anchor and every
    other party at physical dimension $1$ identifies $\sigma$ with $\tau$ by
    Theorem~\ref{thm:compression_operator_finite_coordinates}. The triangle
    inequality and the density and separation bounds of
    Theorem~\ref{thm:peps_dyadic_protocol_consequences} then yield the trace
    norm and anchor-separation bounds.
\end{proof}

\begin{lemma}[Bounds on the composed protocol circuit]
    \label{lem:peps_protocol_circuit_bounds}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_participants_toDistributedConstruction_circuit,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.nonprivateCount_toDistributedConstruction_circuit_le,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.participationCount_toDistributedConstruction_circuit_le,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.isMonomialBounded_toDistributedConstruction_circuit,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.isExpansionBounded_toDistributedConstruction_circuit}
    \leanok
    \uses{def:peps_dyadic_protocol}
    Let $W$ be the composition of the $T$ changes $W_0,\ldots,W_{T-1}$ of a
    protocol satisfying Definition~\ref{def:peps_dyadic_protocol} with
    constants $C_K,b,k,r$. Every nonprivate gate of $W$ has the participant set
    of a nonprivate gate of some $W_t$ ($t<T$), $W$ has at most $Tb$
    nonprivate gates, each party participates in at most $b$ of them, and each
    nonprivate gate has at most $\lfloor C_KL^k\rfloor$ monomials, at most $r$
    pair effects per monomial, and absolute coefficient sum at most $C_KL^k$.
\end{lemma}
\begin{proof}
    \leanok
    Write $N(X)$ for the nonprivate gate count of $X$ and $n_p(X)$ for the
    participation count of party $p$. Then
    $N(W)=\sum_{t<T}N(W_t)\le Tb$ and $n_p(W)=\sum_{t<T}n_p(W_t)\le b$, while
    each gate inherits its participant set, monomial count, and coefficient sum
    from its change $W_t$.
\end{proof}

\begin{theorem}[Compressed dyadic operator networks from the protocol bounds]
    \label{thm:peps_large_size_network_from_protocol}
    \lean{TNLean.PEPS.Approximation.dyadicOperatorNetworkAt_of_dyadicProtocolBoundsAt}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_large_size_dyadic_operator_network}
    Fix $q\ge2$, a unit-state family $\mathcal F$, and constants
    $C_N,C_K,C_E,b,k,r,L_0$. If every $L\ge\max\{2,L_0\}$ and
    $\Omega\in\mathcal F(L)$ admit a protocol satisfying
    Definition~\ref{def:peps_dyadic_protocol}, then compressed dyadic operator
    networks exist for $\mathcal F$
    (Definition~\ref{def:peps_large_size_dyadic_operator_network}) with error
    constant $1+2C_NC_E+2C_E$, link prefactor
    $C_\Sigma(r,b,\max\{C_N,0\}b,q,\max\{C_K,0\},C_K)$, link exponent
    $\alpha_{\mathrm{net}}(r,b,3,0,k,k,5)$, block bound $b$, degree bound
    $2b(b-1)$, separation constant $b$, and threshold $L_0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_distributed_construction_compression_of_le, def:peps_compression_constants,
        lem:peps_protocol_compressed_party_network,
        lem:peps_protocol_circuit_bounds}
    By Lemma~\ref{lem:peps_protocol_circuit_bounds}, the composed circuits for
    $L\ge\max\{2,L_0\}$ and $\Omega\in\mathcal F(L)$ have at most $bC_NL^3$
    nonprivate gates, $\le b$ parties per gate, $\le b$ gates per party,
    physical dimensions $\le q$, and gate expansions with $\le C_KL^k$
    monomials, coefficient sum $\le C_KL^k$, and $\le r$ pair effects per
    monomial. Applying
    Theorem~\ref{thm:peps_distributed_construction_compression_of_le} with
    $p=5$ and $s=k$ produces a contraction $\tau$ within $\varepsilon=L^{-5}$
    of the physical density with link size $\le C_pL^\alpha$ and degree
    $\le 2b(b-1)$, where
    $C_p=C_\Sigma(r,b,\max\{C_N,0\}b,q,\max\{C_K,0\},C_K)$ and
    $\alpha=\alpha_{\mathrm{net}}(r,b,3,0,k,k,5)$. Since $1\le C_NL^2$ and
    $1-C_EL^{-57}\le\lVert\Xi\rVert\le1$ give $C_N,C_E\ge0$,
    Lemma~\ref{lem:peps_protocol_compressed_party_network} yields
    \begin{align}
        \bigl\lVert\sigma-|\Omega\rangle\langle\Omega|\bigr\rVert_1
         & \le L^{-5}+(2C_NC_E+2C_E)L^{-10}\le(1+2C_NC_E+2C_E)L^{-5}. \notag
    \end{align}
\end{proof}
