\section{Finite-group unitary averages}

\begin{definition}[Finite-group unitary average]
    \label{def:mpug_finite_group_unitary_average}
    \mathlibok % Mathlib: Representation.averageMap, whose matrix is P_rho.
    \discussion{7458}
    Let $G$ be a finite group and let $\rho\colon G\to\mathrm U(n)$ be a
    unitary representation. Define
    \begin{align}
        P_\rho & = \frac{1}{|G|}\sum_{g\in G}\rho(g).
    \end{align}
    At a fixed lattice site $j$ in
    \cite[lines~459--461]{FrancoRubio2025MPUGauging}, the specialization
    $\rho(g)=\mathcal G_g^j$ yields the local Gauss-law average $P_\rho=P_j$.
\end{definition}

\begin{theorem}[Finite-group unitary average is an orthogonal projection]
    \label{thm:mpug_finite_group_unitary_average_projection}
    \lean{TNLean.Algebra.isStarProjection_inv_card_smul_sum,
        TNLean.Algebra.isOrthogonalProjection_inv_card_smul_sum}
    \leanok
    \discussion{7458}
    \uses{def:mpug_finite_group_unitary_average}
    For every finite-dimensional unitary representation $\rho$ of a finite
    group,
    \begin{align}
        P_\rho^\dagger & =P_\rho, \\
        P_\rho^2       & =P_\rho.
    \end{align}
    Hence $P_\rho$ is an orthogonal projection.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_finite_group_unitary_average}
    When each matrix $\rho(g)$ is viewed as the corresponding linear map, the
    normalized group-algebra average $e_G=|G|^{-1}\sum_{g\in G}g$ satisfies the
    idempotence calculation
    \begin{align}
        h e_G & =e_G \quad (h\in G),                  &
        e_G^2 & =\frac{1}{|G|}\sum_{h\in G}h e_G=e_G.
    \end{align}
    Applying the linear representation and passing back to matrices gives
    $P_\rho^2=P_\rho$.
    Unitarity gives $\rho(g)^\dagger=\rho(g^{-1})$. Since inversion permutes
    $G$,
    \begin{align}
        P_\rho^\dagger
         & = \frac{1}{|G|}\sum_{g\in G}\rho(g)^\dagger
        = \frac{1}{|G|}\sum_{g\in G}\rho(g^{-1})
        = P_\rho.
    \end{align}
\end{proof}

\section{Finite-group representations by matrix product unitaries}
\label{sec:mpug_group_representations}

Let $G$ be a group, let $d\in\N$, and let
$D\colon G\to\N_{>0}$ assign a virtual bond dimension to each group element.
For every $g\in G$, let $\mathcal U_g$ be an MPO tensor with physical input and
output dimension $d$ and virtual bond dimension $D_g$. Write $U_g^{(N)}$ for
the periodic operator obtained by closing a chain of $N$ copies of
$\mathcal U_g$. Throughout this section the chain length is positive: $N\geq1$.

\begin{definition}[Matrix product unitarity on nonempty chains]
    \label{def:mpug_positive_mpu}
    \lean{MPOTensor.IsMPUPos}
    \leanok
    Following the system-size convention of
    \cite[line~207]{FrancoRubio2025MPUGauging}, an MPO tensor $\mathcal U$ is a
    matrix product unitary on nonempty chains when $U^{(N)}$ is unitary for
    every $N\geq1$.
\end{definition}

\begin{theorem}[Comparison with the $N>1$ MPU convention]
    \label{thm:mpug_positive_mpu_to_mpu}
    \lean{MPOTensor.IsMPUPos.isMPU}
    \leanok
    \uses{def:mpug_positive_mpu, def:mpu}
    Every matrix product unitary on nonempty chains is a matrix product unitary
    in the sense of Definition~\ref{def:mpu}.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_positive_mpu, def:mpu}
    The nonempty-chain condition applies in particular to every $N>1$.
\end{proof}

\begin{theorem}[Physical blocking preserves nonempty-chain unitarity]
    \label{thm:mpug_positive_mpu_blocking}
    \lean{MPOTensor.IsMPUPos.blockTensor}
    \leanok
    \uses{def:mpug_positive_mpu, def:mpdo_physical_blocking}
    If $\mathcal U$ is a matrix product unitary on nonempty chains and $L>0$,
    then $\mathcal U^{[L]}$ is also a matrix product unitary on nonempty chains.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpu_reindex_unitary, thm:mpdo_closed_chain_physical_blocking}
    At chain length $N\geq1$, the canonical reindexing of configurations gives
    \begin{align}
        O_N(\mathcal U^{[L]}) & \cong U^{(NL)}.
    \end{align}
    Since $N,L>0$, the length $NL$ is positive. Thus $U^{(NL)}$ is unitary,
    and simultaneous reindexing preserves unitarity.
\end{proof}

\begin{definition}[Group representation by simple injective MPUs]
    \label{def:mpug_group_representation}
    \lean{MPOTensor.GroupFamily,
        MPOTensor.GroupFamily.IsRepresentation}
    \leanok
    \uses{def:injective, def:mpu_simple, def:mpug_positive_mpu}
    A group representation by simple injective MPUs consists of the
    dependent-bond family $\{\mathcal U_g\}_{g\in G}$ above such that every
    $\mathcal U_g$ is injective and simple, every $\mathcal U_g$ is a matrix
    product unitary on nonempty chains, and
    \begin{align}
        U_e^{(N)}          & =\Id,
        \label{eq:mpug_rep_identity}       \\
        U_g^{(N)}U_h^{(N)} & =U_{gh}^{(N)}
        \label{eq:mpug_rep_multiplication}
    \end{align}
    for all $g,h\in G$ and all $N\geq1$.
    In~\cite[Subsection ``MPU group representations'']
    {FrancoRubio2025MPUGauging}, the multiplicative law serves as the
    representation-level input, and the identity law follows as its standard
    consequence at positive lengths; under the stronger convention of
    \cite[Subsection ``Assumptions for the rest of the paper'']
    {FrancoRubio2025MPUGauging}, this identity law is also compatible with
    taking $\mathcal U_e=\Id$ of bond dimension one.
    Because the dimensions $D_gD_h$ and $D_{gh}$ need not agree,
    (\ref{eq:mpug_rep_multiplication}) is an identity of periodic operators,
    not an equality of local tensors.
\end{definition}

\begin{definition}[The operator representation on one chain]
    \label{def:mpug_operator_representation}
    \lean{MPOTensor.GroupFamily.operatorRepresentation,
        MPOTensor.GroupFamily.operatorRepresentation_coe}
    \leanok
    \uses{def:mpug_positive_mpu}
    Let $\{\mathcal U_g\}_{g\in G}$ be a dependent-bond family in which every
    $\mathcal U_g$ is a matrix product unitary on nonempty chains, and which
    satisfies~(\ref{eq:mpug_rep_identity})
    and~(\ref{eq:mpug_rep_multiplication}) for all $N\geq1$. For a fixed
    $N\geq1$, the map $g\mapsto U_g^{(N)}$ is then a unitary representation
    of $G$ on $(\C^d)^{\otimes N}$. Injectivity and simplicity of the
    tensors are not required.
\end{definition}

\begin{theorem}[Group fusion rules]
    \label{thm:mpug_group_fusion_rules}
    \lean{MPOTensor.GroupFamily.isMPOFusionAlgebra,
        MPOTensor.GroupFamily.IsNormalRepresentation.isMPOFusionAlgebra,
        MPOTensor.GroupFamily.isInvertibleLabel_groupFusion}
    \leanok
    \uses{def:mpo_fusion_algebra, def:fusion_unit_invertible}
    Let $G$ be a finite group and $\{\mathcal U_g\}_{g\in G}$ matrix product
    operator tensors whose periodic operators satisfy
    $U_g^{(N)}U_h^{(N)}=U_{gh}^{(N)}$ for all $g,h\in G$ and $N\geq1$. Then
    $\{\mathcal U_g\}$ is a fusion algebra with structure constants
    \begin{align}
        N_{gh}^k & =\delta_{k,gh},
        \label{eq:mpug_group_fusion_rules}
    \end{align}
    and, as in \cite[Section~4]{GarreRubio2023MPOSym}, every label $g$ is
    invertible with respect to the unit $e$, with inverse $g^{-1}$. Since only
    the multiplicative law~(\ref{eq:mpug_rep_multiplication}) enters, this
    conclusion holds in particular for any representation in the sense of
    Definition~\ref{def:mpug_group_representation}.
    % Source anchor: arXiv:2203.12563, line 660.
\end{theorem}

\begin{proof}
    \leanok
    By the multiplicative law,
    $U_g^{(N)}U_h^{(N)}=U_{gh}^{(N)}=\sum_k\delta_{k,gh}U_k^{(N)}$. For the
    inverse, $N_{g,g^{-1}}^k=\delta_{k,gg^{-1}}=\delta_{k,e}$ and likewise
    $N_{g^{-1},g}^k=\delta_{k,e}$.
\end{proof}

\begin{theorem}[Powers of represented operators]
    \label{thm:mpug_operator_power}
    \lean{MPOTensor.GroupFamily.IsRepresentation.operator_pow}
    \leanok
    \discussion{7445}
    \uses{def:mpug_group_representation}
    For every $g\in G$, every $n\in\N$, and every $N\geq1$,
    \begin{align}
        \bigl(U_g^{(N)}\bigr)^n & =U_{g^n}^{(N)}.
        \label{eq:mpug_operator_power}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_group_representation}
    We proceed by induction on $n$. For $n=0$, the identity law gives
    $U_e^{(N)}=\Id$. For the induction step, multiplying the induction
    hypothesis by $U_g^{(N)}$ and applying the representation multiplication law
    gives
    \begin{align}
        \bigl(U_g^{(N)}\bigr)^{n+1}
         & =U_{g^n}^{(N)}U_g^{(N)}
        =U_{g^{n+1}}^{(N)}.
    \end{align}
\end{proof}

\begin{theorem}[Finite order of represented operators]
    \label{thm:mpug_operator_finite_order}
    \lean{MPOTensor.GroupFamily.IsRepresentation.operator_pow_orderOf_eq_one}
    \leanok
    \discussion{7445}
    \uses{def:mpug_group_representation}
    Suppose that $G$ is finite. As in
    \cite[line~1547]{FrancoRubio2025MPUGauging}, for every $g\in G$ and every
    $N\geq1$, the represented operator has finite order dividing the order of
    $g$:
    \begin{align}
        \bigl(U_g^{(N)}\bigr)^{\operatorname{ord}(g)} & =\Id.
    \end{align}
    % Source anchor: Papers/2502.20257/main.tex, line 1547.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_group_representation, thm:mpug_operator_power}
    Evaluating (\ref{eq:mpug_operator_power}) at $n=\operatorname{ord}(g)$,
    together with the finite-order identity $g^{\operatorname{ord}(g)}=e$ and
    the representation identity law, gives
    $\bigl(U_g^{(N)}\bigr)^{\operatorname{ord}(g)}=U_e^{(N)}=\Id$.
\end{proof}

\begin{theorem}[Product-tensor operator identity]
    \label{thm:mpug_product_tensor_operator}
    \lean{MPOTensor.GroupFamily.IsRepresentation.sameMPV₂Pos_mulTensor}
    \leanok
    \uses{def:mpug_group_representation, def:mpdo_operator_product}
    Let $\mathcal U_g\mathbin{\cdot}\mathcal U_h$ denote the product MPO tensor,
    whose virtual bond dimension is $D_gD_h$. For every $N\geq1$,
    \begin{align}
        O_N(\mathcal U_g\mathbin{\cdot}\mathcal U_h)
         & =U_g^{(N)}U_h^{(N)}
        =U_{gh}^{(N)}.
        \label{eq:mpug_product_tensor_positive_chain}
    \end{align}
    Thus the product tensor and the selected tensor $\mathcal U_{gh}$ generate
    the same nonempty-chain periodic operators, although their virtual bond
    dimensions may differ.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_group_representation, thm:mpdo_operator_product_mpo}
    Multiplicativity of the periodic operator under product-tensor contraction gives
    $O_N(\mathcal U_g\mathbin{\cdot}\mathcal U_h) = U_g^{(N)}U_h^{(N)}$, which equals $U_{gh}^{(N)}$ by~(\ref{eq:mpug_rep_multiplication}).
\end{proof}

\begin{theorem}[The physical adjoint represents the inverse]
    \label{thm:mpug_physical_adjoint_inverse}
    \lean{MPOTensor.GroupFamily.IsRepresentation.mpo_physicalAdjointTensor_eq_inv}
    \leanok
    \uses{def:mpug_group_representation, def:mpdo_physical_adjoint}
    As in the first assertion of
    \cite[line~1552]{FrancoRubio2025MPUGauging}, without any finiteness
    assumption on $G$, for every $g\in G$ and every $N\geq1$ the physical
    adjoint of $\mathcal U_g$ generates the operator representing $g^{-1}$:
    \begin{align}
        O_N(\mathcal U_g^\sharp)
         & =U_{g^{-1}}^{(N)}.
        \label{eq:mpug_physical_adjoint_inverse}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_group_representation, def:mpdo_physical_adjoint}
    The representation law and unitarity give
    \begin{align}
        U_{g^{-1}}^{(N)}U_g^{(N)}              & =U_e^{(N)}=\Id, \\
        U_g^{(N)}\bigl(U_g^{(N)}\bigr)^\dagger & =\Id.
    \end{align}
    Hence
    \begin{align}
        U_{g^{-1}}^{(N)}
         & =U_{g^{-1}}^{(N)}
        \bigl(U_g^{(N)}\bigl(U_g^{(N)}\bigr)^\dagger\bigr)
        =\bigl(U_{g^{-1}}^{(N)}U_g^{(N)}\bigr)
        \bigl(U_g^{(N)}\bigr)^\dagger
        =\bigl(U_g^{(N)}\bigr)^\dagger.
    \end{align}
    Closing the physical-adjoint tensor gives the conjugate transpose, which
    proves (\ref{eq:mpug_physical_adjoint_inverse}).
\end{proof}

\begin{theorem}[Positive-length matrix product vector consequence]
    \label{thm:mpug_physical_adjoint_inverse_mpv}
    \lean{MPOTensor.GroupFamily.IsRepresentation.sameMPV₂Pos_physicalAdjointTensor_inv}
    \leanok
    \uses{def:mpug_group_representation, def:mpdo_physical_adjoint,
        def:mpo_to_mps, def:same_mpv2_pos}
    When $\mathcal U_g^\sharp$ and $\mathcal U_{g^{-1}}$ are viewed as MPS
    tensors by pairing their two physical indices, they generate the same
    matrix product vectors at every positive length:
    \begin{align}
        \mc{V}^+\bigl((\mathcal U_g^\sharp)^{\mathrm{MPS}}\bigr)
         & =\mc{V}^+\bigl(\mathcal U_{g^{-1}}^{\mathrm{MPS}}\bigr).
        \label{eq:mpug_physical_adjoint_inverse_mpv}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_physical_adjoint_inverse,
        lem:mpdo_three_first_site_contraction_coefficients}
    For a doubled physical configuration $\rho$, decoding each site as a pair of
    ket and bra indices $(\sigma_n,\tau_n)$ identifies the corresponding matrix
    product vector coefficient with the $(\sigma,\tau)$ matrix entry of the
    periodic operator. Equality
    (\ref{eq:mpug_physical_adjoint_inverse}) therefore gives equality of every
    coefficient in (\ref{eq:mpug_physical_adjoint_inverse_mpv}).
\end{proof}

\begin{definition}[Exchange of product bond factors]
    \label{def:mpug_product_bond_swap}
    \lean{MPOTensor.productBondSwapEquiv}
    \leanok
    For bond dimensions $D_M$ and $D_N$, identify a product bond index with
    its pair of factor indices. The exchange of product bond factors is the
    equivalence
    \begin{align}
        \{0,\ldots,D_M-1\}\times\{0,\ldots,D_N-1\}
                                                    & \longrightarrow
        \{0,\ldots,D_N-1\}\times\{0,\ldots,D_M-1\}, &
        (\alpha,\gamma)                             & \longmapsto (\gamma,\alpha).
        \label{eq:mpug_product_bond_swap}
    \end{align}
\end{definition}

\begin{theorem}[Physical adjunction reverses the operator product]
    \label{thm:mpug_physical_adjoint_product_reversal}
    \lean{MPOTensor.physicalAdjointTensor_mulTensor}
    \leanok
    \uses{def:mpug_product_bond_swap, def:mpdo_operator_product,
        def:mpdo_physical_adjoint}
    Let $M$ and $N$ be MPO tensors with bond dimensions $D_M$ and $D_N$.
    Write a left bond index of the product tensor as a pair
    $(\alpha,\gamma)$, with $\alpha$ in the bond space of $M$ and $\gamma$ in
    the bond space of $N$, and a right bond index as $(\beta,\delta)$
    likewise.
    % Source anchor: arXiv:2502.20257, line 1662 and prop:zetas,
    % lines 1662--1677.
    As in the inverse-product step for the inversion identities of $\zeta$ and
    $\omega$ in~\cite[lines~1662--1677]{FrancoRubio2025MPUGauging} (where the
    dagger operation conjugates every entry and exchanges the two legs on the
    same side of a tensor), and without assuming $D_M=D_N$, every pair of
    physical indices $i,k$ satisfies
    \begin{align}
        \bigl((M\mathbin{\cdot}N)^\sharp\bigr)^{ik}
            _{(\alpha,\gamma),(\beta,\delta)}
         & =\bigl(N^\sharp\mathbin{\cdot}M^\sharp\bigr)^{ik}
        _{(\gamma,\alpha),(\delta,\beta)}.
        \label{eq:mpug_physical_adjoint_product_reversal}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_product_bond_swap, def:mpdo_operator_product,
        def:mpdo_physical_adjoint}
    Expanding the left side gives the complex conjugate of the
    $\bigl((\alpha,\gamma),(\beta,\delta)\bigr)$ entry of
    $\sum_j M^{kj}\otimes N^{ji}$, namely
    $\sum_j\overline{M^{kj}_{\alpha\beta}}\,
        \overline{N^{ji}_{\gamma\delta}}$.
    Expanding the right side gives the
    $\bigl((\gamma,\alpha),(\delta,\beta)\bigr)$ entry of
    $\sum_j (N^\sharp)^{ij}\otimes(M^\sharp)^{jk}$, namely
    $\sum_j\overline{N^{ji}_{\gamma\delta}}\,
        \overline{M^{kj}_{\alpha\beta}}$,
    which agrees with the left side term by term.
\end{proof}

\begin{theorem}[Unitary gauges for the dagger and the group inverse]
    \label{thm:mpug_dagger_inverse_gauge}
    \lean{MPOTensor.GroupFamily.IsRepresentation.bondDim_inv,
        MPOTensor.GroupFamily.inverseFlattening,
        MPOTensor.GroupFamily.IsRepresentation.exists_daggerInverseGauge,
        MPOTensor.GroupFamily.IsRepresentation.daggerInverseGauge,
        MPOTensor.GroupFamily.IsRepresentation.physicalAdjointTensor_eq_daggerInverseGauge}
    \leanok
    \uses{def:mpug_group_representation, def:mpu_normalized_flattening,
        def:left_canonical_tensor, def:mpdo_physical_adjoint}
    Let $G$ be a group, and let
    $\{\mathcal U_g\}_{g\in G}$ be a representation by simple injective MPU
    tensors whose normalized flattenings are left-canonical.  Then
    \begin{align}
        D_g & =D_{g^{-1}}.
        \label{eq:mpug_dagger_inverse_bond_dimension}
    \end{align}
    After identifying these bond spaces, as in
    \cite[lines~1552--1557]{FrancoRubio2025MPUGauging}, for every $g\in G$ the
    physical-adjoint tensor $\mathcal U_g^\dagger$ is related to
    $\mathcal U_{g^{-1}}$ by a unitary matrix
    $T_g\in\operatorname{Mat}_{D_g}(\C)$ satisfying
    \begin{align}
        \mathcal U_g^\dagger
         & =T_g^\dagger\,\mathcal U_{g^{-1}}T_g.
        \label{eq:mpug_dagger_inverse_gauge}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_physical_adjoint_inverse_mpv,
        thm:mps_equality_reduction_exists,
        thm:mps_rectangular_reduction_identities, thm:ft_single,
        thm:injective_irreducible,
        lem:left_canonical_irreducible_same_phase_unitary_gauge,
        thm:mpu_physical_adjoint_normalized_flattening}
    Applying the equality-case reduction theorem first to
    $\mathcal U_g^\dagger$ and $\mathcal U_{g^{-1}}$, and then with $g$ and
    $g^{-1}$ exchanged, gives $D_{g^{-1}}\leq D_g$ and $D_g\leq D_{g^{-1}}$,
    which proves (\ref{eq:mpug_dagger_inverse_bond_dimension}). Since their
    positive-length matrix product vectors agree and this bond-dimension
    identification matches their length-zero values as well, the single-block
    Fundamental Theorem produces an invertible gauge. Because both normalized
    flattenings are left-canonical and injectivity implies irreducibility, the
    canonical-form unitary-gauge theorem refines this gauge to a unitary one,
    whose reversed orientation gives (\ref{eq:mpug_dagger_inverse_gauge}).
\end{proof}

    \leanok
    \uses{thm:mpug_dagger_inverse_gauge}
    As in \cite[lines~1557--1567]{FrancoRubio2025MPUGauging}, the gauges in
    Theorem~\ref{thm:mpug_dagger_inverse_gauge} may be chosen together with
    scalars $\sigma_g\in\C$ satisfying
    \begin{align}
        T_gT_{g^{-1}}^* & =\sigma_g\mathbf 1.
        \label{eq:mpug_dagger_inverse_scalar}
    \end{align}
    Every such scalar lies in $U(1)$, with
    \begin{align}
        |\sigma_g|              & =1, \\
        \sigma_g\sigma_{g^{-1}} & =1.
    \end{align}
    When $g^{-1}=g$, the chosen gauge at $g^{-1}$, written in the bond
    coordinates at $g$, equals $T_g$, so the same chosen scalar satisfies
    $T_gT_g^*=\sigma_g\mathbf 1$ and $\sigma_g\in\{1,-1\}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_dagger_inverse_gauge,
        thm:mpu_physical_adjoint_identities,
        lem:gauge_unique_up_to_scalar,
        thm:mpug_unitary_entrywise_scalar_phase,
        thm:mpug_unitary_entrywise_paired_scalars,
        thm:mpug_unitary_entrywise_self_sign}
    Applying physical adjunction to (\ref{eq:mpug_dagger_inverse_gauge}) for
    $g^{-1}$, and using its involutivity and unitarity, produces a second gauge
    $T_{g^{-1}}^*$ from $\mathcal U_{g^{-1}}$ to $\mathcal U_g^\dagger$. By
    injective gauge uniqueness, $T_{g^{-1}}^*$ is a nonzero scalar multiple of
    $T_g^\dagger$, and left-multiplication by $T_g$ yields
    (\ref{eq:mpug_dagger_inverse_scalar}) with the displayed orientation.
    Theorems~\ref{thm:mpug_unitary_entrywise_scalar_phase}
    and~\ref{thm:mpug_unitary_entrywise_paired_scalars} then give the
    unit-modulus and inverse-pair identities. When $g^{-1}=g$, rewriting the
    chosen gauge along this equality gives $T_g$ in the current bond
    coordinates, so substituting into (\ref{eq:mpug_dagger_inverse_scalar})
    gives $T_gT_g^*=\sigma_g\mathbf 1$ and
    Theorem~\ref{thm:mpug_unitary_entrywise_self_sign} forces
    $\sigma_g\in\{1,-1\}$.
\end{proof}

\begin{definition}[Gauge matrices for the first cut]

    \end{align}
\end{definition}

\begin{definition}[Candidate first-cut factors]
    \label{def:mpug_inverse_cut_factors}
    \lean{MPOTensor.inverseCompatibleX₁, MPOTensor.inverseCompatibleY₁}

    dimension $D$, and let $T$ be unitary on $\C^D$. Let
    $\ell=\operatorname{rank}M_2(\mathcal U)$. For the chosen compact
    singular-value decomposition $M_2(\mathcal U)=V_2^\dagger D_2U_2$, set
    \begin{align}
        X_2 & =V_2^\dagger,          \\
        Y_2 & =D_2U_2,               \\
        Z_2 & =U_2^\dagger D_2^{-1}.
    \end{align}
    With $A,B$ from Definition~\ref{def:mpug_inverse_cut_gauges} and without any
    choice of a positive weight on the second cut, the candidate factors of
    \cite[line~5432]{FrancoRubio2025MPUGauging} with intermediate coordinate
    space $\C^\ell$ indexed by $\{0,\ldots,\ell-1\}$ are
    \begin{align}
        X & =AY_2^\dagger\colon\C^\ell\longrightarrow\C^d\otimes\C^D,  \\
        Y & =X_2^\dagger B\colon\C^D\otimes\C^d\longrightarrow\C^\ell.
    \end{align}
\end{definition}

\begin{theorem}[Reversing a unitary adjoint gauge]
    \label{thm:mpug_inverse_cut_reverse_gauge}
    \lean{MPOTensor.eq_unitary_mul_physicalAdjointTensor_mul_conjTranspose}
    \leanok
    \uses{def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a tensor of physical dimension $d$ and bond
    dimension $D$. Suppose that $T$ is unitary on $\C^D$ and that, for
    every physical pair $i,j$,
    $(\mathcal U^\sharp)^{ij}=T^\dagger\mathcal U^{ij}T$. Reversing the gauge
    relation of \cite[lines~5391--5410]{FrancoRubio2025MPUGauging} then gives,
    for every physical pair $i,j$,
    $\mathcal U^{ij}=T(\mathcal U^\sharp)^{ij}T^\dagger$.
\end{theorem}
\begin{proof}\leanok
    Unitarity gives
    $T(\mathcal U^\sharp)^{ij}T^\dagger
        =(TT^\dagger)\mathcal U^{ij}(TT^\dagger)=\mathcal U^{ij}$.
\end{proof}

\begin{theorem}[The first cut as a dressed second cut]
    \label{thm:mpug_inverse_cut_dressed}
    \lean{MPOTensor.sourceCutM₁_eq_inverseCompatible_dressed_cut}
    \leanok
    \uses{def:mpug_inverse_cut_gauges, def:mpu_source_matrix_one,
        def:mpu_source_matrix_two, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a tensor of physical dimension $d$ and bond
    dimension $D$. Suppose that $T$ is unitary and that, for every
    physical pair $i,j$, $(\mathcal U^\sharp)^{ij}=T^\dagger\mathcal U^{ij}T$.
    As in the cut identity of
    \cite[lines~5391--5431]{FrancoRubio2025MPUGauging}, the matrices
    $A=I_d\otimes\overline T$ and $B=T^{\mathsf T}\otimes I_d$ relate the two
    source cuts by
    $M_1(\mathcal U)=AM_2(\mathcal U)^\dagger B$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_cut_reverse_gauge,
        thm:mpu_physical_adjoint_source_cuts, def:mpug_inverse_cut_gauges,
        def:mpu_source_matrix_one}
    Physical adjunction gives
    $M_2(\mathcal U)^\dagger=M_1(\mathcal U^\sharp)$.
    For physical indices $i,j$ and virtual indices $a,b$, expanding the matrix
    product and applying Theorem~\ref{thm:mpug_inverse_cut_reverse_gauge} yields
    \begin{align}
        (AM_1(\mathcal U^\sharp)B)_{(i,b),(a,j)}
         & =\sum_{x,y=0}^{D-1}\overline{T_{bx}}
        (\mathcal U^\sharp)^{ij}_{yx}T_{ay}           \\
         & =(T(\mathcal U^\sharp)^{ij}T^\dagger)_{ab}
        =\mathcal U^{ij}_{ab}.
    \end{align}
\end{proof}

\begin{theorem}[Factorization through the second-cut space]

\subsection{Rank equality and the invertible first-cut comparison}

The invertible-comparison step of
\cite[lines~5432--5443]{FrancoRubio2025MPUGauging} relates the source factors
via their one-sided inverses, with inverse comparison matrix $J$.
No unitarity of $K$, equality $J=K^\dagger$, or full inverse-compatible
decomposition is asserted here.

\begin{theorem}[Equality of the two source-cut ranks]
    \label{thm:mpug_inverse_cut_rank_equality}
    \lean{MPOTensor.rightRank_eq_leftRank_of_physicalAdjointTensor_eq_unitary_gauge}
    \leanok
    \uses{def:mpu_right_rank, def:mpu_left_rank, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be an MPO tensor of physical dimension $d$ and bond
    dimension $D$. Suppose that $T$ is unitary on $\C^D$ and
    $(\mathcal U^\sharp)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    $0\leq i,j<d$.
    Then its source-cut ranks satisfy
    $r=\operatorname{rank}M_1(\mathcal U)
        =\operatorname{rank}M_2(\mathcal U)=\ell$.
    No positivity, simplicity, or positive-dimension assumption is required.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_cut_factorization,
        thm:mpug_inverse_cut_left_inverse, thm:mpug_inverse_cut_right_inverse,
        def:mpug_inverse_cut_factors}
    With $X=AY_2^\dagger$, $Y=X_2^\dagger B$, $L=Z_2^\dagger A^\dagger$, and
    $R=B^\dagger X_2$ formed from the gauge matrices and second-cut factors of
    Definition~\ref{def:mpug_inverse_cut_factors}, the factorization $M_1=XY$
    through $\C^\ell$ gives $r\leq\ell$. Conversely, $LX=I_\ell$ and
    $YR=I_\ell$ imply $LM_1R=I_\ell$, so
    $\ell=\operatorname{rank}(LM_1R)
        \leq\operatorname{rank}M_1=r$.
\end{proof}

\begin{theorem}[Both source-cut ranks equal the physical dimension]
    \label{thm:mpug_inverse_source_ranks_physical}
    \lean{MPOTensor.inverseCompatible_sourceRanks_eq_physical,
        MPOTensor.inverseCompatiblePhysicalEquiv}
    \leanok
    \uses{def:mpu_right_rank, def:mpu_left_rank, def:mpu_simple,
        def:mpu_canonical_form_ii, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a simple MPU of physical dimension $d$ and bond
    dimension $D$ in canonical form II. Suppose that $T$ is unitary on
    $\C^D$ and that $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$
    for every $0\leq i,j<d$. In this self-inverse setting of
    \cite[lines~1552--1563]{FrancoRubio2025MPUGauging} (where the general
    finite-group index argument of line~1547 is not needed, and neither the
    phase relation $T\overline T=\sigma I_D$, nor simplicity of
    $\mathcal U^\dagger$, nor any further virtual gauge enters), the source-cut
    ranks satisfy $\ell=r=d$, and the resulting bijection of
    $\{0,\ldots,\ell-1\}$ with $\{0,\ldots,d-1\}$ carries the source-cut index
    to the physical index.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_cut_rank_equality, ThmFund1}
    Theorem~\ref{thm:mpug_inverse_cut_rank_equality} gives $r=\ell$ from
    the adjoint gauge relation alone. Simplicity of a canonical-form-II
    tensor is equivalent to $r\ell=d^2$ by Theorem~\ref{ThmFund1}.
    Substituting $r=\ell$ gives $\ell^2=d^2$, hence $\ell=d$ and $r=d$.
\end{proof}

\begin{definition}[The weighted comparison matrix]
    \label{def:mpug_inverse_cut_comparison_k}
    \lean{MPOTensor.inverseCompatibleComparisonK}
    \leanok
    \uses{def:mpug_inverse_cut_factors, def:mpu_weighted_source_factors}
    Let $\mathcal U$ be an MPO tensor of dimensions $d,D$, let $T$ be
    unitary on $\C^D$, and let $\rho$ be a positive-definite matrix on
    $\C^D$. Put $r=\operatorname{rank}M_1(\mathcal U)$ and
    $\ell=\operatorname{rank}M_2(\mathcal U)$. Let
    $\widetilde X,\widetilde Y,\widetilde Z$ be the first source factors
    constructed with weight $\rho$, and let $X=AY_2^\dagger$ be the
    candidate first factor. Define
    \begin{align}
        \widetilde L & =\widetilde X^\dagger(I_d\otimes\rho),           \\
        K            & =\widetilde L X\colon\C^\ell\longrightarrow\C^r.
    \end{align}
    Here $\widetilde L$ is the weighted left inverse given by
    $\widetilde X^\dagger(I_d\otimes\rho)\widetilde X=I_r$.
    The definition retains the weight $\rho$ and requires no adjoint gauge
    relation.
\end{definition}

\begin{definition}[The reverse comparison matrix]
    \label{def:mpug_inverse_cut_comparison_j}
    \lean{MPOTensor.inverseCompatibleComparisonJ}
    \leanok
    \uses{def:mpug_inverse_cut_factors, def:mpug_inverse_cut_gauges,
        def:mpu_weighted_source_factors}
    Let $\mathcal U$ be an MPO tensor of dimensions $d,D$, let $T$ be
    unitary on $\C^D$, and let $\rho$ be positive definite on $\C^D$.
    Let $\widetilde X$ be its first source factor with weight $\rho$,
    and let $Y_2,Z_2$ be its second-cut factors with $Y_2Z_2=I_\ell$,
    where $r=\operatorname{rank}M_1(\mathcal U)$ and
    $\ell=\operatorname{rank}M_2(\mathcal U)$. For
    $A=I_d\otimes\overline T$, define
    \begin{align}
        L & =Z_2^\dagger A^\dagger,                         \\
        J & =L\widetilde X\colon\C^r\longrightarrow\C^\ell.
    \end{align}
    No adjoint gauge relation is needed to define $J$.
\end{definition}

\begin{theorem}[Mutually inverse rectangular comparisons]
    \label{thm:mpug_inverse_cut_comparison}
    \lean{MPOTensor.inverseCompatibleComparison}
    \leanok
    \uses{def:mpug_inverse_cut_comparison_k, def:mpug_inverse_cut_comparison_j,
        def:mpug_inverse_cut_factors, def:mpu_weighted_source_factors,
        def:mpdo_physical_adjoint}
    Let $\mathcal U$ be an MPO tensor of dimensions $d,D$, let $T$ be
    unitary on $\C^D$, and suppose that
    $(\mathcal U^\sharp)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    $0\leq i,j<d$. Let $\rho$ be positive definite on $\C^D$.
    Write $\widetilde X,\widetilde Y,\widetilde Z$ for the first
    source factors with weight $\rho$, and $X_2,Y_2,Z_2$ for the
    second-cut factors. Set
    \begin{align}
        A            & =I_d\otimes\overline T,                \\
        B            & =T^{\mathsf T}\otimes I_d,             \\
        X            & =AY_2^\dagger,                         \\
        Y            & =X_2^\dagger B,                        \\
        L            & =Z_2^\dagger A^\dagger,                \\
        R            & =B^\dagger X_2,                        \\
        \widetilde L & =\widetilde X^\dagger(I_d\otimes\rho), \\
        K            & =\widetilde L X,                       \\
        J            & =L\widetilde X.
    \end{align}
    With $r=\operatorname{rank}M_1(\mathcal U)$ and
    $\ell=\operatorname{rank}M_2(\mathcal U)$, the matrices
    $K\colon\C^\ell\to\C^r$ and $J\colon\C^r\to\C^\ell$ satisfy
    \begin{align}
        K            & =\widetilde YR, \notag \\
        X            & =\widetilde XK, \notag \\
        \widetilde Y & =KY, \notag            \\
        J            & =Y\widetilde Z, \notag \\
        \widetilde X & =XJ, \notag            \\
        Y            & =J\widetilde Y, \notag \\
        KJ           & =I_r, \notag           \\
        JK           & =I_\ell.
        \label{eq:mpug_inverse_cut_comparison}
    \end{align}
    The two source-coordinate spaces are retained separately, without
    identifying their bases along the equality $r=\ell$. These are inverse
    identities, not adjoint or unitarity identities.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_factorization_comparison,
        thm:mpug_inverse_cut_factorization, thm:mpug_inverse_cut_left_inverse,
        thm:mpug_inverse_cut_right_inverse, thm:mpu_source_factorizations,
        thm:mpu_source_factor_normalizations, thm:mpu_source_factor_right_inverses}
    The two first-cut factorizations give $XY=\widetilde X\widetilde Y$, while
    their one-sided inverses satisfy $\widetilde L\widetilde X=I_r$,
    $LX=I_\ell$, $YR=I_\ell$, and $\widetilde Y\widetilde Z=I_r$. Applying
    Theorem~\ref{thm:mpug_factorization_comparison} to these two factorizations
    first with $\widetilde L,R$ and then in the reverse direction with
    $L,\widetilde Z$ establishes the factor relations in
    (\ref{eq:mpug_inverse_cut_comparison}), which in turn imply
    \begin{align}
        KJ & =KY\widetilde Z=\widetilde Y\widetilde Z=I_r, \\
        JK & =J\widetilde YR=YR=I_\ell.
    \end{align}
\end{proof}
\begin{theorem}[The second source factor from the adjoint comparison]
    \label{thm:mpug_inverse_cut_adjoint_comparison}
    \lean{MPOTensor.sourceY₂_eq_inverseCompatibleComparisonK_adjoint}
    \leanok
    \uses{def:mpug_inverse_cut_comparison_k, def:mpug_inverse_cut_gauges,
        def:mpu_weighted_source_factors, def:mpdo_physical_adjoint}

    $T$ be unitary on $\C^D$. Suppose that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair $i,j$. Write $\widetilde X_1,\widetilde Y_1$ for the
    first source factors with weight $\rho$, and $X_2,Y_2$ for the
    second source factors. Put $A=I_d\otimes\overline T$ and take the
    weighted comparison
    $K=\widetilde X_1^\dagger(I_d\otimes\rho)AY_2^\dagger$ of
    Definition~\ref{def:mpug_inverse_cut_comparison_k}. As in the adjointed
    comparison of \cite[lines~5440--5443]{FrancoRubio2025MPUGauging} in the
    maintained physical--virtual column order, the second source factor
    satisfies
    \begin{align}
        Y_2 & =K^\dagger\widetilde X_1^\dagger A.
    \end{align}
\end{theorem}
\begin{proof}\leanok

    Theorem~\ref{thm:mpug_inverse_cut_comparison} gives
    $AY_2^\dagger=\widetilde X_1K$. Since $A^\dagger A=I$,

         & =(AY_2^\dagger)^\dagger A

\end{proof}

\begin{theorem}[The inverse-compatible source-gate identity]
    \label{thm:mpug_inverse_cut_gate_identity}
    \lean{MPOTensor.sourceU_eq_smul_inverseCompatibleComparisonK_kronecker_sourceV_adjoint}
    \leanok
    \uses{def:mpug_inverse_cut_comparison_k, def:mpug_inverse_cut_gauges,
        def:mpu_weighted_source_factors, def:mpdo_physical_adjoint,
        def:mpu_source_uv}
    Let $\mathcal U$ be an MPO tensor of physical dimension $d$ and bond
    dimension $D$, let $\rho$ be positive definite on $\C^D$, and let
    $T$ be unitary on $\C^D$. Suppose that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair $i,j$, and that $T\overline T=\sigma I_D$ for a scalar
    $\sigma\in\C$. Write $\widetilde X_1,\widetilde Y_1$ for the first
    source factors constructed with this $\rho$, and $X_2,Y_2$ for the
    second source factors. Set $r=\operatorname{rank}M_1(\mathcal U)$,
    $\ell=\operatorname{rank}M_2(\mathcal U)$, and
    $K=\widetilde X_1^\dagger(I_d\otimes\rho)
        (I_d\otimes\overline T)Y_2^\dagger\colon\C^\ell\to\C^r$.
    As in the contraction of \cite[lines~5444--5487]{FrancoRubio2025MPUGauging}
    retaining the diagram's $\widetilde Y_1$
    \cite{gap:fbc25_inverse_compatible_tilde_omission}, without assuming
    canonical form, simplicity, gate unitarity, or $|\sigma|=1$, the source
    gates $u,v$ of Definition~\ref{def:mpu_source_uv} formed from
    $\widetilde X_1,\widetilde Y_1,X_2,Y_2$ satisfy the algebraic identity
    \begin{align}
        u & =\sigma(K^\dagger\otimes K)v^\dagger,
    \end{align}
    where the Kronecker product maps directly between the row order $(r,\ell)$
    of $v^\dagger$ and the row order $(\ell,r)$ of $u$ without any additional
    swap or identification of the rank coordinates.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_cut_adjoint_comparison,
        thm:mpug_inverse_cut_comparison, def:mpug_inverse_cut_factors,
        def:mpug_inverse_cut_gauges, def:mpu_source_uv}
    Conjugating the phase relation by $T$ gives
    $\overline T T=T^\dagger(T\overline T)T=\sigma I_D$.
    Theorem~\ref{thm:mpug_inverse_cut_adjoint_comparison} and
    Theorem~\ref{thm:mpug_inverse_cut_comparison} give
    \begin{align}
        Y_2            & =K^\dagger\widetilde X_1^\dagger(I_d\otimes\overline T), \\
        \widetilde Y_1 & =KX_2^\dagger(T^{\mathsf T}\otimes I_d).
    \end{align}
    For fixed physical indices $i,j$, the slices
    $C\in\operatorname{Mat}_{r\times D}(\C)$ and
    $E\in\operatorname{Mat}_{D\times\ell}(\C)$ with entries
    $C_{a\gamma}=\overline{(\widetilde X_1)_{(i,\gamma),a}}$ and
    $E_{\delta b}=\overline{(X_2)_{(\delta,j),b}}$ rewrite the matrix entries
    at every virtual index $\beta$ and source indices $l,s$ as
    \begin{align}
        (Y_2)_{l,(i,\beta)}
         & =(K^\dagger C\overline T)_{l\beta}, \\
        (\widetilde Y_1)_{s,(\beta,j)}
         & =(TEK^{\mathsf T})_{\beta s}.
    \end{align}
    Contracting the virtual index $\beta$ in the defining expression for $u$
    then gives
    \begin{align}
        u_{(l,s),(i,j)}
         & =\sum_{\beta=0}^{D-1}(Y_2)_{l,(i,\beta)}
        (\widetilde Y_1)_{s,(\beta,j)}                      \\
         & =(K^\dagger C\overline T T E K^{\mathsf T})_{ls}
        =\sigma(K^\dagger C E K^{\mathsf T})_{ls},
    \end{align}
    and expanding the matrix product $K^\dagger CEK^{\mathsf T}$ yields
    \begin{align}
        (K^\dagger C E K^{\mathsf T})_{ls}
         & =\sum_{a=0}^{r-1}\sum_{b=0}^{\ell-1}
        (K^\dagger)_{la}K_{sb}
        \overline{\sum_{\gamma=0}^{D-1}
        (\widetilde X_1)_{(i,\gamma),a}(X_2)_{(\gamma,j),b}} \\
         & =((K^\dagger\otimes K)v^\dagger)_{(l,s),(i,j)}.
    \end{align}
\end{proof}

    \end{align}

\end{theorem}
\begin{proof}\leanok
    \uses{lemuisometry, ThmFund1, thm:mpug_inverse_cut_gate_identity,
        thm:mpug_unitary_entrywise_scalar_phase,
        thm:matrix_unitary_adjoint_kronecker}

    $d^2\leq r\ell$, so both rank spaces are nonempty. By
    Theorem~\ref{thm:mpug_unitary_entrywise_scalar_phase},
    $\sigma\overline\sigma=1$. Theorem~\ref{ThmFund1} gives unitary
    source gates $u,v$, constructed with the recorded $\rho$. Put
    $H=K^\dagger\otimes K\colon\C^r\otimes\C^\ell
        \to\C^\ell\otimes\C^r$.

    $u=\sigma Hv^\dagger$. Cancelling the unitary right factor makes
    $\sigma H$ unitary. Its two Gram identities give
    \begin{align}
        I_{r\ell}  & =(\sigma H)^\dagger(\sigma H)
        =\overline\sigma\sigma H^\dagger H=H^\dagger H, \\
        I_{\ell r} & =(\sigma H)(\sigma H)^\dagger
        =\sigma\overline\sigma HH^\dagger=HH^\dagger.
    \end{align}
    Thus $H$ is unitary. Theorem~\ref{thm:matrix_unitary_adjoint_kronecker}
    implies that $K$ is unitary without identifying the two rank bases.
\end{proof}

\begin{theorem}[Unitary change of basis for the first source factors]
    \label{thm:mpug_inverse_source_unitary_transport}
    \lean{MPOTensor.inverseCompatibleSourceFactors_unitary_transport,
        MPOTensor.inverseCompatibleSourceFactors_gram_eq}
    \leanok
    \uses{def:mpug_inverse_compatible_source_factors,

    form II, with recorded positive-definite matrix $\rho$. Suppose
    that $T$ is unitary on $\C^D$, that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair $i,j$, and that $T\overline T=\sigma I_D$ for a
    scalar $\sigma\in\C$. Let $\widetilde X_1,\widetilde Y_1$ be the
    chosen first source factors at this same weight, and let
    $\widehat X_1,\widehat Y_1$ be the inverse-compatible first factors
    of Definition~\ref{def:mpug_inverse_compatible_source_factors}.
    For $r=\operatorname{rank}M_1(\mathcal U)$ and
    $\ell=\operatorname{rank}M_2(\mathcal U)$, let $K\colon\C^\ell\to\C^r$ be
    the weighted comparison of Definition~\ref{def:mpug_inverse_cut_comparison_k}
    and $P_e\colon\C^\ell\to\C^r$ the permutation matrix of
    Definition~\ref{def:mpug_inverse_cut_rank_equiv}. As in the basis change of
    \cite[lines~5432--5443]{FrancoRubio2025MPUGauging} (once unitarity is
    established at lines~5444--5487, without asserting any transformation
    formula for the chosen first right inverse), the square matrix
    $C=KP_e^\dagger$ is unitary on $\C^r$ and satisfies
    \begin{align}
        \widehat X_1 & =\widetilde X_1C,         \\
        \widehat Y_1 & =C^\dagger\widetilde Y_1,
    \end{align}
    so the Gram matrices on the external first-cut spaces are preserved:
    \begin{align}
        \widehat X_1\widehat X_1^\dagger
         & =\widetilde X_1\widetilde X_1^\dagger, \\
        \widehat Y_1^\dagger\widehat Y_1
         & =\widetilde Y_1^\dagger\widetilde Y_1.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_cut_comparison_unitary,
        thm:mpug_inverse_cut_comparison, thm:mpug_inverse_cut_unitary_consequences,
        def:mpug_inverse_cut_rank_equiv, def:mpug_inverse_compatible_source_factors}

    $X=\widetilde X_1K$, and $Y=K^\dagger\widetilde Y_1$ for the
    untransported candidates $X,Y$. Reindexing the columns of $K$
    along $e$ gives $C=KP_e^\dagger$. Since $P_e$ identifies the two
    coordinate bases,
    \begin{align}
        C^\dagger C & =P_eK^\dagger KP_e^\dagger=I_r, \\
        CC^\dagger  & =KP_e^\dagger P_eK^\dagger=I_r.
    \end{align}
    Reindexing the two candidate factors gives
    \begin{align}
        \widehat X_1 & =XP_e^\dagger
        =\widetilde X_1KP_e^\dagger=\widetilde X_1C, \\
        \widehat Y_1 & =P_eY
        =P_eK^\dagger\widetilde Y_1=C^\dagger\widetilde Y_1.
    \end{align}
    Cancelling $CC^\dagger=I_r$ in the external Gram matrices then gives
    \begin{align}
        \widehat X_1\widehat X_1^\dagger
         & =\widetilde X_1(CC^\dagger)\widetilde X_1^\dagger
        =\widetilde X_1\widetilde X_1^\dagger,               \\
        \widehat Y_1^\dagger\widehat Y_1
         & =\widetilde Y_1^\dagger(CC^\dagger)\widetilde Y_1
        =\widetilde Y_1^\dagger\widetilde Y_1.
    \end{align}
\end{proof}

                                         & =\frac{d}{\ell}I_\ell.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_source_unitary_transport,
        thm:mpug_source_y_one_normalization,
        thm:mpug_source_y_two_normalization,
        def:mpug_inverse_compatible_source_factors}
    Let $\widetilde Y_1,Y_2$ be the original factors. The unitary change of basis
    gives $\widehat Y_1=C^\dagger\widetilde Y_1$ with $C^\dagger C=I_r$.
    Conjugating the original scalar Gram matrix yields
    \begin{align}
        \widehat Y_1\widehat Y_1^\dagger
         & =C^\dagger(\widetilde Y_1\widetilde Y_1^\dagger)C
        =C^\dagger\left(\frac{d}{r}I_r\right)C
        =\frac{d}{r}I_r.
    \end{align}
    The second normalization is unchanged because $\widehat Y_2=Y_2$.
\end{proof}

\begin{theorem}[Physical closures of the compatible left factors]
    \label{thm:mpug_inverse_source_x_physical}
    \lean{MPOTensor.inverseCompatibleSourceFactors_X_physical_contractions}
    \leanok
    \uses{def:mpug_inverse_compatible_source_factors, def:mpu_canonical_form_ii,
        def:mpu_simple, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be an MPU of dimensions $d,D$ in canonical form II,
    with recorded positive-definite matrix $\rho$. Assume that both
    $\mathcal U$ and $\mathcal U^\dagger$ are simple. Suppose that $T$
    is unitary, that $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$
    for every physical pair, and that $T\overline T=\sigma I_D$ for
    $\sigma\in\C$. Let $\widehat X_1,\widehat X_2$ be its
    inverse-compatible source factors at $\rho$, with cut ranks $r,\ell$.
    As in the physical left-factor closures of
    \cite[lines~1120--1162]{FrancoRubio2025MPUGauging} transferred at
    lines~5486--5487 while retaining the canonical weight $\rho$, every pair of
    physical indices $i,j$ satisfies
    \begin{align}
        \sum_{s=0}^{r-1}\sum_{\alpha,\beta=0}^{D-1}
        (\widehat X_1)_{(i,\alpha),s}\rho_{\beta\alpha}
        \overline{(\widehat X_1)_{(j,\beta),s}}
         & =\frac{r}{d}\delta_{i,j},    \\
        \sum_{t=0}^{\ell-1}\sum_{x=0}^{D-1}
        (\widehat X_2)_{(x,i),t}\overline{(\widehat X_2)_{(x,j),t}}
         & =\frac{\ell}{d}\delta_{i,j}.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_source_unitary_transport,
        thm:mpug_source_x_one_physical_normalization,
        thm:mpug_source_x_two_physical_normalization,
        def:mpug_inverse_compatible_source_factors}
    For the original factors $\widetilde X_1,X_2$, multiplying the
    $((i,\alpha),(j,\beta))$ entries of the external Gram equality by
    $\rho_{\beta\alpha}$ and summing over the virtual indices gives
    \begin{align}
        \sum_{\alpha,\beta=0}^{D-1}
        (\widehat X_1\widehat X_1^\dagger)_{(i,\alpha),(j,\beta)}
        \rho_{\beta\alpha}
         & =\sum_{\alpha,\beta=0}^{D-1}
        (\widetilde X_1\widetilde X_1^\dagger)_{(i,\alpha),(j,\beta)}
        \rho_{\beta\alpha}              \\
         & =\frac{r}{d}\delta_{i,j}.
    \end{align}
    Expanding each Gram entry and regrouping the finite sums gives the
    first closure. The second is the original physical closure because
    $\widehat X_2=X_2$.
\end{proof}

These six identities concern only the stated closures and normalizations.
They do not assert movement identities, truncated-gate identities, transformation
of the chosen first right inverse, or the complete set of pleasant properties.

\begin{theorem}[Transformation of the movement gates]
    \label{thm:mpug_inverse_movement_transport}
    \lean{MPOTensor.inverseCompatibleSourceFactors_movement_transport}
    \leanok
    \uses{def:mpug_inverse_compatible_source_factors,
        def:mpug_inverse_cut_rank_equiv, def:mpug_inverse_cut_comparison_k,
        def:mpug_staircase_gates, def:mpu_weighted_source_factors,
        def:mpu_canonical_form_ii, def:mpu_simple, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a simple MPU of dimensions $d,D$ with a
    canonical-form-II presentation and recorded positive-definite matrix
    $\rho$. Let $T$ be unitary on $\C^D$ and let $\sigma\in\C$ satisfy
    $T\overline T=\sigma I_D$. Suppose that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair $i,j$. Let $S_0$ be the chosen source factors at this
    $\rho$, and let $S$ be the inverse-compatible source factors of
    Definition~\ref{def:mpug_inverse_compatible_source_factors} for the
    same data. Put $r=\operatorname{rank}M_1(\mathcal U)$ and
    $\ell=\operatorname{rank}M_2(\mathcal U)$. Let
    $e\colon\{0,\ldots,\ell-1\}\to\{0,\ldots,r-1\}$ be the
    rank identification of Definition~\ref{def:mpug_inverse_cut_rank_equiv}.
    Reindexing the columns of the comparison $K$ by $e$ gives
    $C\in\operatorname{Mat}_r(\C)$ with $C_{a,b}=K_{a,e^{-1}(b)}$. As in the
    crossed contractions of \cite[lines~811--869]{FrancoRubio2025MPUGauging}
    conjugated by the unitary comparison of
    \cite[lines~5486--5487]{FrancoRubio2025MPUGauging}, the crossed movement
    gates (where the right gate has row order $(\text{physical},\text{first rank})$
    and column order $(\text{first rank},\text{physical})$ with no leg
    permutation) satisfy
    \begin{align}
        w_L(S) & =w_L(S_0),                                     \\
        w_R(S) & =(I_d\otimes C^\dagger)w_R(S_0)(C\otimes I_d).
        \label{eq:mpug_inverse_movement_transport}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_source_unitary_transport,
        def:mpug_inverse_compatible_source_factors, def:mpug_staircase_gates}
    The second factors are unchanged, giving the left-gate equality.
    For the first factors, Theorem~\ref{thm:mpug_inverse_source_unitary_transport}
    gives $X_1(S)=X_1(S_0)C$ and $Y_1(S)=C^\dagger Y_1(S_0)$.
    For physical indices $i,j$ and first-rank indices $t,t'$, substituting
    these relations into the crossed contraction gives
    \begin{align}
        (w_R(S))_{(i,t),(t',j)}
         & =\sum_{a,b=0}^{r-1}\overline{C_{b,t}}
        \left(\sum_{\beta=0}^{D-1}
        (X_1(S_0))_{(i,\beta),a}(Y_1(S_0))_{b,(\beta,j)}\right)C_{a,t'} \\
         & =\sum_{a,b=0}^{r-1}\overline{C_{b,t}}
        (w_R(S_0))_{(i,b),(a,j)}C_{a,t'}.
    \end{align}
    Regrouping the finite sums gives the right-gate formula in
    (\ref{eq:mpug_inverse_movement_transport}).
\end{proof}

\begin{theorem}[Unitarity of the compatible movement gates]
    \label{thm:mpug_inverse_movement_unitary}
    \lean{MPOTensor.inverseCompatibleSourceFactors_movement_isUnitaryBetween}
    \leanok
    \uses{def:mpug_inverse_compatible_source_factors, def:mpug_staircase_gates,
        def:mpu_canonical_form_ii, def:mpu_simple, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a simple MPU of dimensions $d,D$ with a
    canonical-form-II presentation and recorded positive-definite matrix
    $\rho$. Suppose that $T$ is unitary on $\C^D$, that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair $i,j$, and that $T\overline T=\sigma I_D$ for a scalar
    $\sigma\in\C$. Let $S$ be its inverse-compatible source factors
    constructed with this same $\rho$. With source-cut ranks $r,\ell$,
    both movement gates are unitary maps between their indicated spaces:
    \begin{align}
        w_L(S) & \colon\C^d\otimes\C^\ell\longrightarrow\C^\ell\otimes\C^d, \\
        w_R(S) & \colon\C^r\otimes\C^d\longrightarrow\C^d\otimes\C^r.
    \end{align}
    No simplicity hypothesis on the physical adjoint is required.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_source_unitary_transport,
        thm:mpug_staircase_gates_unitary, thm:mpug_inverse_movement_transport,
        thm:matrix_unitary_between_kronecker}
    Theorem~\ref{thm:mpug_inverse_source_unitary_transport} makes $C$
    unitary, and Theorem~\ref{thm:mpug_staircase_gates_unitary} makes the
    chosen gates $w_L(S_0),w_R(S_0)$ unitary. While the left gate is unchanged,
    the comparison identity (\ref{eq:mpug_inverse_movement_transport}) as in
    \cite[lines~5486--5487]{FrancoRubio2025MPUGauging} expresses the right gate
    as $(I_d\otimes C^\dagger)w_R(S_0)(C\otimes I_d)$, whose two outer factors
    are unitary by Theorem~\ref{thm:matrix_unitary_between_kronecker} and whose
    middle factor $w_R(S_0)$ is unitary, so their composition $w_R(S)$ is
    unitary as well.
\end{proof}

\begin{definition}[Gates on the common second-cut space]

    $Y=X_2^\dagger(T^{\mathsf T}\otimes I_d)$ and the second factors
    $X_2,Y_2$, the new gates
    $u_{\mathrm{new}}\colon\C^d\otimes\C^d\to\C^\ell\otimes\C^\ell$
    and $v_{\mathrm{new}}\colon\C^\ell\otimes\C^\ell\to\C^d\otimes\C^d$
    of \cite[lines~5486--5487]{FrancoRubio2025MPUGauging} are defined by direct

\end{definition}

\end{theorem}
\begin{proof}\leanok

    \uses{def:mpug_inverse_cut_factors, def:mpug_inverse_cut_gauges}
    Unitarity gives $A^\dagger A=I$, so
    $X^\dagger A=Y_2A^\dagger A=Y_2$.
\end{proof}

\begin{definition}[The two endpoint matrices along a word]
    \label{def:mpug_inverse_word_endpoints}
    \lean{MPOTensor.inverseCompatibleWordA, MPOTensor.inverseCompatibleWordB,
        MPOTensor.inverseCompatibleWordA_zero_apply,
        MPOTensor.inverseCompatibleWordB_zero_apply}
    \leanok
    \uses{def:mpug_inverse_cut_factors, def:mpu_weighted_source_factors,
        def:mpo_eval_word, def:mpug_inverse_compatible_gates}
    Let $\mathcal U$ be an MPO tensor of dimensions $d,D$ and $T$ a
    unitary virtual matrix. With the second source factors $X_2,Y_2$
    and the candidates $X=(I_d\otimes\overline T)Y_2^\dagger$,
    $Y=X_2^\dagger(T^{\mathsf T}\otimes I_d)$, put
    $\ell=\operatorname{rank}M_2(\mathcal U)$ and
    $\Omega_N=\{0,\ldots,d-1\}^{\{0,\ldots,N-1\}}$ for $N\geq0$.
    Define
    \begin{align}
        \mathcal H_N & =\C^d\otimes(\C^{\Omega_N}\otimes\C^d),       \\
        \mathcal R_N & =\C^\ell\otimes(\C^{\Omega_N}\otimes\C^\ell).
    \end{align}
    Extending the endpoint contractions of
    \cite[lines~5444--5487]{FrancoRubio2025MPUGauging} to all words read in
    their given order, the matrices $A_N\colon\mathcal H_N\to\mathcal R_N$ and
    $B_N\colon\mathcal R_N\to\mathcal H_N$ have entries, for $p,q\in\Omega_N$,
    physical indices $a,b$, and source indices $l,s$,
    \begin{align}
        (A_N)_{(l,(p,s)),(a,(q,b))}
         & =\sum_{\alpha,\beta=0}^{D-1}(Y_2)_{l,(a,\alpha)}
        \mathcal U[p,q]_{\alpha\beta}Y_{s,(\beta,b)},       \\
        (B_N)_{(a,(p,b)),(l,(q,s))}
         & =\sum_{\alpha,\beta=0}^{D-1}X_{(a,\alpha),l}
        \mathcal U[p,q]_{\alpha\beta}(X_2)_{(\beta,b),s}.
    \end{align}
    With $\varnothing$ the unique empty configuration, the zero-length
    entries recover the new gates:
    \begin{align}
        (A_0)_{(l,(\varnothing,s)),(a,(\varnothing,b))}
         & =(u_{\mathrm{new}})_{(l,s),(a,b)}, \\
        (B_0)_{(a,(\varnothing,b)),(l,(\varnothing,s))}
         & =(v_{\mathrm{new}})_{(a,b),(l,s)}.
    \end{align}
\end{definition}
\begin{proof}\leanok
    \uses{def:mpo_eval_word, def:mpug_inverse_compatible_gates}
    At length zero, $\mathcal U[\varnothing,\varnothing]=I_D$.
    The entry $\delta_{\alpha\beta}$ contracts the two sums, respectively, to
    \begin{align}
         & \sum_\alpha(Y_2)_{l,(a,\alpha)}Y_{s,(\alpha,b)},  \\
         & \sum_\alpha X_{(a,\alpha),l}(X_2)_{(\alpha,b),s}.
    \end{align}
\end{proof}

\begin{theorem}[All-word adjunction of the endpoints]
    \label{thm:mpug_inverse_word_adjoint}
    \lean{MPOTensor.inverseCompatibleWordA_eq_smul_wordB_adjoint}
    \leanok
    \uses{def:mpug_inverse_word_endpoints, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be an MPO tensor and $T$ a unitary virtual matrix.
    Suppose that $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$
    for every physical pair and that $T\overline T=\sigma I_D$ for a
    scalar $\sigma\in\C$. Extending the local relation of
    \cite[lines~5444--5487]{FrancoRubio2025MPUGauging} to all words (before the
    final truncated-chain sewing identity, and without requiring canonical
    form, simplicity, a comparison matrix, or rank reindexing), for every
    $N\geq0$ the endpoint matrices of
    Definition~\ref{def:mpug_inverse_word_endpoints} satisfy
    $A_N=\sigma B_N^\dagger$, where the scalar $\sigma$ occurs once
    independently of $N$ and adjunction exchanges the physical configurations
    without reversing either word or inserting an additional swap.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_word_gauge, thm:mpug_inverse_endpoint_recovery,
        def:mpug_inverse_word_endpoints, def:mpug_inverse_cut_factors,
        def:mpug_inverse_cut_gauges, def:mpdo_physical_adjoint,
        def:mpo_eval_word}
    Fix $p,q\in\Omega_N$ and physical indices $a,b$. Put
    $M=\mathcal U[p,q]$ and $W=\overline{\mathcal U[q,p]}$, and define
    slices $C_{t\gamma}=\overline{X_{(a,\gamma),t}}$ and
    $E_{\delta t}=\overline{(X_2)_{(\delta,b),t}}$.
    Physical adjunction conjugates each letter and exchanges its two
    physical indices, so
    $W=(\mathcal U^\dagger)[p,q]=T^\dagger MT$ by
    Theorem~\ref{thm:mpug_inverse_word_gauge}. The phase relation gives
    $\overline T=\sigma T^\dagger$. Endpoint recovery and the definition
    of $Y$ give, for virtual indices $\alpha,\beta$,
    \begin{align}
        (Y_2)_{l,(a,\alpha)} & =(C\overline T)_{l\alpha}, \\
        Y_{s,(\beta,b)}      & =(TE)_{\beta s}.
    \end{align}
    The two virtual sums therefore contract as
    \begin{align}
        (A_N)_{(l,(p,s)),(a,(q,b))}
         & =(C\overline T MTE)_{ls}
        =\sigma(C(T^\dagger MT)E)_{ls}
        =\sigma(CWE)_{ls}.
    \end{align}
    Expanding the remaining product gives
    \begin{align}
        (CWE)_{ls}
         & =\overline{\sum_{\alpha,\beta=0}^{D-1}
        X_{(a,\alpha),l}\mathcal U[q,p]_{\alpha\beta}(X_2)_{(\beta,b),s}} \\
         & =\overline{(B_N)_{(a,(q,b)),(l,(p,s))}}.
    \end{align}
    This is precisely the entry of $B_N^\dagger$ at the original row
    and column, with the spatial order unchanged.
\end{proof}
\begin{definition}[Common positive physical blocking]
    \label{def:mpug_common_blocking}
    \lean{MPOTensor.GroupFamily.block}
    \leanok
    \uses{def:mpug_group_representation, def:mpdo_physical_blocking}
    For a positive integer $L$, the common length-$L$ physical blocking is the
    family
    \begin{align}
        \widetilde{\mathcal U}_g & =\mathcal U_g^{[L]}.
        \label{eq:mpug_common_blocked_family}
    \end{align}
    Each blocked tensor has physical dimension $d^L$ and the same virtual bond
    dimension $D_g$ as $\mathcal U_g$. The physical alphabet is replaced
    simultaneously for every group element by words of length $L$.
\end{definition}

\begin{theorem}[Common blocking preserves the representation laws]
    \label{thm:mpug_common_blocking_representation}
    \lean{MPOTensor.GroupFamily.IsRawRepresentation.block,
        MPOTensor.GroupFamily.IsRepresentation.block}
    \leanok
    \uses{def:mpug_common_blocking}
    Let $L>0$. The commonly blocked family
    $\{\mathcal U_g^{[L]}\}_{g\in G}$ is again a group representation by simple
    injective MPUs. Under the identification of blocked
    configurations with words of length $NL$,
    \begin{align}
        O_N(\mathcal U_g^{[L]})
         & \cong U_g^{(NL)},
        \label{eq:mpug_blocked_operator} \\
        O_N(\mathcal U_e^{[L]})
         & =\Id,
        \label{eq:mpug_blocked_identity} \\
        O_N(\mathcal U_g^{[L]})O_N(\mathcal U_h^{[L]})
         & =O_N(\mathcal U_{gh}^{[L]})
        \label{eq:mpug_blocked_multiplication}
    \end{align}
    for every $N\geq1$. Physical blocking also commutes with tensor products:
    \begin{align}
        (\mathcal U_g\mathbin{\cdot}\mathcal U_h)^{[L]}
         & =\mathcal U_g^{[L]}\mathbin{\cdot}\mathcal U_h^{[L]}.
        \label{eq:mpug_block_product_tensor}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_positive_mpu_blocking,
        thm:mpdo_blocking_doubled_index,
        thm:mpu_bounded_simple_blocking,
        thm:mpdo_closed_chain_physical_blocking,
        thm:mpdo_blocking_product}
    Physical blocking preserves nonempty-chain unitarity, injectivity, and
    simplicity. By (\ref{eq:mpug_blocked_operator}), the blocked periodic operator is
    the original operator at length $NL$ up to the canonical simultaneous reindexing.
    Since $N,L>0$,
    the length $NL$ is positive, so (\ref{eq:mpug_rep_identity}) and
    (\ref{eq:mpug_rep_multiplication}) apply. Simultaneous reindexing preserves
    identities and products, which gives (\ref{eq:mpug_blocked_identity}) and
    (\ref{eq:mpug_blocked_multiplication}). Expanding one blocked letter gives
    (\ref{eq:mpug_block_product_tensor}).
\end{proof}

Assume from now until the end of this section that $G$ is finite and $d>0$.

\begin{theorem}[A finite MPU family has a common simple blocking]
    \label{thm:mpug_finite_common_simple_blocking}
    \lean{MPOTensor.GroupFamily.commonSimpleLength_pos,
        MPOTensor.GroupFamily.IsRawRepresentation.block_common,
        MPOTensor.GroupFamily.IsRawRepresentation.block_common_then_block}
    \leanok
    \uses{def:mpug_raw_group_representation,
        def:mpug_common_simple_length, def:mpug_group_representation,
        def:mpu_simple, def:mpdo_physical_blocking}
    Let $\{\mathcal U_g\}_{g\in G}$ be an injective MPU representation without
    an assumed simplicity condition. Then $L_*>0$,
    and every $\mathcal U_g^{[L_*]}$ is simple. For every $K>0$, each further
    block $(\mathcal U_g^{[L_*]})^{[K]}$ is simple. Consequently the blocked
    family at length $L_*$ satisfies
    Definition~\ref{def:mpug_group_representation}, and every further common
    positive blocking does as well.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_positive_mpu_blocking,
        thm:mpug_positive_mpu_to_mpu,
        thm:mpu_common_pow_four_simple_blocking,
        thm:mpu_all_later_simple_blockings,
        thm:mpdo_blocking_doubled_index,
        thm:mpdo_closed_chain_physical_blocking,
        thm:mpug_common_blocking_representation}
    Positivity of the bond dimensions gives $L_*>0$. For each $g$, the direct
    block of length $D_g^4$ is simple, and $D_g^4\leq L_*$ by
    (\ref{eq:mpug_common_simple_length}). Simplicity holds at every later direct
    blocking, hence at the common length $L_*$. Nonempty-chain unitarity and
    injectivity are preserved by physical blocking. Reindexing the closed chain
    at length $NL_*$ carries the identity and multiplication laws of the given
    representation to the blocked family. The length-$L_*$ family is therefore
    a group representation by simple injective MPUs. Applying
    Theorem~\ref{thm:mpug_common_blocking_representation} to this established
    representation proves the assertion for every further positive blocking.
\end{proof}

\begin{definition}[Selected reductions for the pairs of a representation]
    \label{def:mpug_pair_reductions}
    \lean{MPOTensor.GroupFamily.productTensor,
        MPOTensor.GroupFamily.targetTensor,
        MPOTensor.GroupFamily.ReductionFamily,
        MPOTensor.GroupFamily.ReductionFamily.nilpotencyLength,
        MPOTensor.GroupFamily.ReductionFamily.HasExteriorBufferLength}
    \leanok
    \uses{def:mpug_group_representation, def:mpdo_operator_product,
        def:mpo_to_mps, def:mps_rectangular_reduction,
        def:mps_reduction_residual_nilpotency_length,
        def:mps_reduction_exterior_buffer_length}
    For $g,h\in G$, write
    \begin{align}
        B_{g,h} & =\mathcal U_g\mathbin{\cdot}\mathcal U_h,
                & A_{g,h}                                   & =\mathcal U_{gh}
        \label{eq:mpug_pair_tensors}
    \end{align}
    for the product tensor and the target tensor, both regarded as MPS
    tensors whose physical letters are the ket--bra pairs $(i,j)$, of bond
    dimensions $D_gD_h$ and $D_{gh}$.  A \emph{family of selected reductions}
    consists of matrices
    \begin{align}
        V_{g,h} & \in\C^{D_{gh}\times D_gD_h},
                & W_{g,h}                      & \in\C^{D_gD_h\times D_{gh}}
        \label{eq:mpug_pair_reduction_matrices}
    \end{align}
    such that $(V_{g,h},W_{g,h})$ is a reduction from $B_{g,h}$ to $A_{g,h}$
    for every pair: $V_{g,h}W_{g,h}=\Id$ and
    $V_{g,h}B_{g,h}^{\mathbf a}W_{g,h}=A_{g,h}^{\mathbf a}$ for every word
    $\mathbf a$.  The identity on words is the interior fusion equation in
    \cite[equation labelled \emph{eq:fusion\_2}, lines 1459--1497]
    {FrancoRubio2025MPUGauging}, with $F^<_{g,h}=V_{g,h}$ and
    $F^>_{g,h}=W_{g,h}$.    Write $N_0(g,h)=\ell(B_{g,h},A_{g,h};V_{g,h},W_{g,h})$ for the residual
    nilpotency length of the selected reduction of $(g,h)$. Following the
    exterior fusion equation in
    \cite[equation labelled \emph{eq:fusion\_1}, lines 1406--1457, with the
        quantifiers on line 1498]{FrancoRubio2025MPUGauging} (where
    $m\geq\ell$), the family has \emph{exterior buffer length} $m$ if $m$ is an
    exterior buffer length for every selected reduction:
    \begin{align}
        B_{g,h}^{\mathbf p}W_{g,h}A_{g,h}^{\mathbf c}V_{g,h}B_{g,h}^{\mathbf q}
         & =B_{g,h}^{\mathbf p\mathbf c\mathbf q}
        \label{eq:mpug_pair_exterior_identity}
    \end{align}
    for all $g,h\in G$, every nonempty word $\mathbf c$, and all words with
    $|\mathbf p|,|\mathbf q|\geq m$.
\end{definition}

\begin{lemma}[Exterior buffers from the nilpotency lengths]
    \label{lem:mpug_pair_reductions_exterior}

\end{lemma}

\begin{proof}

    preserves (\ref{eq:mpug_pair_exterior_identity}).
\end{proof}

\begin{definition}[Blocked selected reductions and the common buffer length]
    \label{def:mpug_pair_reductions_block}

         & \mathcal U_{gh}^{[L]}
         & \cong A_{g,h}^{[L]},

    so a family of selected reductions $(V_{g,h},W_{g,h})$ of

\begin{theorem}[Common blocking for the one-site fusion identities]
    \label{thm:mpug_common_blocking_one_site}
    \lean{MPOTensor.GroupFamily.ReductionFamily.block_hasExteriorBufferLength_one,
        MPOTensor.GroupFamily.ReductionFamily.block_commonBufferLength_hasExteriorBufferLength_one,
        MPOTensor.GroupFamily.IsRepresentation.exists_block_reductionFamily_one}
    \leanok
    \discussion{7396}
    \uses{def:mpug_pair_reductions_block, def:mpug_group_representation}
    Let $\{\mathcal U_g\}_{g\in G}$ be an MPU representation by simple
    injective tensors with a family of selected reductions, and let $L>0$
    satisfy $N_0(g,h)\leq L+1$ for every pair.  Then the blocked family of
    selected reductions of $\{\mathcal U_g^{[L]}\}_{g\in G}$ has exterior
    buffer length $1$: for all $g,h\in G$ and all nonempty blocked words
    $\mathbf p,\mathbf c,\mathbf q$,
    \begin{align}
        (B_{g,h}^{[L]})^{\mathbf p}W_{g,h}(A_{g,h}^{[L]})^{\mathbf c}
        V_{g,h}(B_{g,h}^{[L]})^{\mathbf q}
         & =(B_{g,h}^{[L]})^{\mathbf p\mathbf c\mathbf q}.
        \label{eq:mpug_common_blocking_one_site}
    \end{align}
    The common buffer length is such an $L$.  Consequently, every MPU
    representation of a finite group by simple injective tensors admits a
    positive $L$ such that $\{\mathcal U_g^{[L]}\}_{g\in G}$ is again an MPU
    representation by simple injective tensors and has a family of selected
    reductions with exterior buffer length $1$.

    In \cite{FrancoRubio2025MPUGauging}, \texttt{main.tex} line 1498, and the
    footnote to Theorem~1 of \cite{GarreRubioSchuch2024DomainWall}, the
    assertion that blocking reduces the nilpotency length to $1$ corresponds to
    the one-site exterior identity (\ref{eq:mpug_common_blocking_one_site}); as
    recorded in \cite{gap:mgsc18_nilpotency_length_one_terminology}, blocking
    need not make a blocked residual letter vanish, and the blocked nilpotency
    length $\ell(B_{g,h}^{[L]},A_{g,h}^{[L]};V_{g,h},W_{g,h})$ of Definition~8
    in \cite{Molnar2018SemiInjective} need not equal $1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:mpug_pair_reductions_exterior, thm:mps_reduction_blocking,
        thm:mpdo_blocking_doubled_index, thm:mpdo_blocking_product,
        thm:mpug_common_blocking_representation,
        thm:mpug_pair_reductions_exist}
    By Lemma~\ref{lem:mpug_pair_reductions_exterior}, $N_0(g,h)-1$ is an
    exterior buffer length for the pair $(g,h)$, and $N_0(g,h)-1\leq L$.
    Theorem~\ref{thm:mps_reduction_blocking} with $k=1$ makes $1$ an
    exterior buffer length for $(V_{g,h},W_{g,h})$ as a reduction from
    $B_{g,h}^{[L]}$ to $A_{g,h}^{[L]}$, and the pairing

\begin{theorem}[Reciprocal scalar rescaling of the fusion tensors]
    \label{thm:mpug_fusion_scalar_gauge}
    \lean{MPSTensor.IsReduction.reciprocal_smul,
        MPSTensor.reductionResidual_reciprocal_smul,
        MPSTensor.isReductionResidualNilpotencyBound_reciprocal_smul,
        MPSTensor.reductionResidualNilpotencyLength_reciprocal_smul,
        MPSTensor.IsReductionExteriorBufferLength.reciprocal_smul,
        MPSTensor.IsReductionExteriorBufferLength.reciprocal_smul_iff,
        MPOTensor.GroupFamily.ReductionFamily.smul,
        MPOTensor.GroupFamily.ReductionFamily.hasExteriorBufferLength_smul,
        MPOTensor.GroupFamily.ReductionFamily.block_smul}
    \leanok
    \discussion{7397}
    \uses{def:mpug_fusion_tensors, def:mpug_pair_reductions,
        def:mpug_pair_reductions_block, def:scalar_cocycle,
        def:mps_rectangular_reduction,
        def:mps_reduction_residual_nilpotency_length,
        def:mps_reduction_exterior_buffer_length}
    Let $(V,W)$ be a reduction from $B$ to $A$ and let
    $\beta\in\C^\times$.  Then $(\beta^{-1}V,\beta W)$ is again a reduction
    from $B$ to $A$; its residual letters $N^i=B^i-WA^iV$ are unchanged, hence
    so are its residual-word nilpotency bounds and its residual nilpotency
    length $\ell(B,A;V,W)$; and a natural number is an exterior buffer length
    for $(\beta^{-1}V,\beta W)$ if and only if it is one for $(V,W)$.

    Consequently, under the identification (\ref{eq:mpug_fusion_tensors}) and
    with uniqueness up to boundary dressing given by
    Theorem~\ref{thm:mps_reduction_boundary_dressed_uniqueness}, the scalar
    gauge freedom of the fusion tensors in
    \cite[lines~1500--1504]{FrancoRubio2025MPUGauging} acts by any scalar
    $2$-cochain $\beta\colon G^2\to\C^\times$: when $\{\mathcal U_g\}_{g\in G}$
    has a family of selected reductions, the rescaled matrices of
    (\ref{eq:mpug_fusion_gauge}), namely $W_{g,h}\mapsto\beta_{g,h}W_{g,h}$ and
    $V_{g,h}\mapsto\beta_{g,h}^{-1}V_{g,h}$, again form a family of selected
    reductions of $\{\mathcal U_g\}_{g\in G}$ that has exterior buffer length
    $m$ exactly when the original family does and commutes with blocking by any
    length $L$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:mps_rectangular_reduction_identities}
    In each identity the two scalars occur in one of the combinations
    $(\beta^{-1}V)(\beta W)$, $(\beta W)A^{\mathbf c}(\beta^{-1}V)$, and
    $(\beta^{-1}V)B^{\mathbf a}(\beta W)$; in every case they multiply to
    $\beta\beta^{-1}=1$ and disappear.  The residual letters and the exterior
    identity are therefore unchanged, and equality of the two sets of
    residual-word nilpotency bounds gives equality of their infima.  The
    statements for a family follow pair by pair, and blocking changes only the
    physical alphabet of the tensors, leaving $V_{g,h}$ and $W_{g,h}$
    untouched.
\end{proof}

\begin{theorem}[Scalar phase from a unitary relation]
    \label{thm:mpug_unitary_entrywise_scalar_phase}
    \lean{Matrix.scalar_mul_star_eq_one_of_mul_map_star_eq_smul_one}
    \leanok
    Let $I$ be a nonempty finite set, let $T,S\in\operatorname{Mat}_I(\C)$ be
    unitary, and let $\sigma\in\C$. As in
    \cite[lines~1557--1562]{FrancoRubio2025MPUGauging}, if
    \begin{align}
        TS^* & =\sigma\mathbf 1,
        \label{eq:mpug_unitary_entrywise_scalar_relation}
    \end{align}
    then the scalar $\sigma$ has unit modulus:
    \begin{align}
        \sigma\overline\sigma & =1.
        \label{eq:mpug_unitary_entrywise_scalar_phase}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    The matrix $S^*$ is unitary, so $TS^*=\sigma\mathbf 1$ is unitary. Hence
    \begin{align}
        (\sigma\mathbf 1)(\sigma\mathbf 1)^\dagger
         & =\sigma\overline\sigma\,\mathbf 1=\mathbf 1.
    \end{align}
    Evaluating a diagonal entry gives
    (\ref{eq:mpug_unitary_entrywise_scalar_phase}).
\end{proof}

\begin{theorem}[Reciprocal scalars in the paired relations]
    \label{thm:mpug_unitary_entrywise_paired_scalars}
    \lean{Matrix.paired_scalars_mul_eq_one_of_mul_map_star_eq_smul_one}
    \leanok
    Let $I$ be a nonempty finite set, let $T,S\in\operatorname{Mat}_I(\C)$ be
    unitary, and let $\sigma,\tau\in\C$. As in the matrix-algebra relation
    underlying $\sigma_g\sigma_{g^{-1}}=1$ in
    \cite[lines~1559--1563]{FrancoRubio2025MPUGauging}, if
    \begin{align}
        TS^* & =\sigma\mathbf 1,
        \label{eq:mpug_unitary_entrywise_rel_ts} \\
        ST^* & =\tau\mathbf 1,
        \label{eq:mpug_unitary_entrywise_rel_st}
    \end{align}
    then
    \begin{align}
        \sigma\tau & =1.
        \label{eq:mpug_unitary_entrywise_paired_scalars}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_unitary_entrywise_scalar_phase}
    Multiplying~(\ref{eq:mpug_unitary_entrywise_rel_ts}) by $T^{-1}$ on the left and $T$ on the
    right gives $S^*T=\sigma\mathbf 1$. Entrywise conjugation of~(\ref{eq:mpug_unitary_entrywise_rel_st})
    gives $S^*T=\overline\tau\,\mathbf 1$. A diagonal entry therefore
    gives $\sigma=\overline\tau$. Applying
    Theorem~\ref{thm:mpug_unitary_entrywise_scalar_phase} to~(\ref{eq:mpug_unitary_entrywise_rel_st})
    yields $\tau\overline\tau=1$, and hence $\sigma\tau=1$.
\end{proof}

\begin{theorem}[Sign dichotomy in the self relation]
    \label{thm:mpug_unitary_entrywise_self_sign}
    \lean{Matrix.scalar_eq_one_or_neg_one_of_mul_map_star_self_eq_smul_one}
    \leanok
    Let $I$ be a nonempty finite set, let $T\in\operatorname{Mat}_I(\C)$ be
    unitary, and let $\sigma\in\C$. As in the sign step
    $\Id=xx^\dagger=e^{2i\phi}\Id$ of \cite[lines~1036--1038]{Cirac2017MPU} and
    the involution conclusion of
    \cite[lines~1563--1567]{FrancoRubio2025MPUGauging}, if
    \begin{align}
        TT^* & =\sigma\mathbf 1,
        \label{eq:mpug_unitary_entrywise_self_relation}
    \end{align}
    then
    \begin{align}
        \sigma & \in\{1,-1\}.
        \label{eq:mpug_unitary_entrywise_self_sign}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_unitary_entrywise_paired_scalars}
    Applying Theorem~\ref{thm:mpug_unitary_entrywise_paired_scalars} with $S=T$
    and $\tau=\sigma$ gives $\sigma^2=1$, so
    $\sigma=1$ or $\sigma=-1$.
\end{proof}

\section{A nondegenerate alternating-network comparison}
\label{sec:mpug_alternating_network}

\begin{definition}[Connected open alternating network]
    \label{def:mpug_open_alternating_network}
    \lean{MPSTensor.openAlternatingNetwork}
    \leanok
    Let $n\geq 0$, and let $D_0,\ldots,D_n$ be horizontal bond dimensions.  At each
    $0\leq i<n$, let $A_i$ have square bond dimension $D_i$, and let $B_i$ have
    left and right bond dimensions $D_i$ and $D_{i+1}$.  Following the
    connected open diagram in
    \cite[lines~2299--2361]{FrancoRubio2025MPUGauging}, where $y_i$ joins $A_i$
    to $B_i$, $x_{i+1}$ joins $B_i$ to $A_{i+1}$ when $i+1<n$, and $x_0,x_n$
    are the open end indices, the connected network with open horizontal ends
    has entries at fixed physical indices $\sigma=(\sigma_i)_i$ and
    $\tau=(\tau_i)_i$ given by
    \begin{align}
        \mathcal N_{A,B}(\sigma,\tau)_{a,b}
         & :=\sum_{\substack{x_0,\ldots,x_n \\
                x_0=a,\,x_n=b}}
        \ \sum_{y_0,\ldots,y_{n-1}}
        \prod_{i=0}^{n-1}
        (A_i^{\sigma_i})_{x_i,y_i}
        (B_i^{\tau_i})_{y_i,x_{i+1}}.
        \notag
    \end{align}
\end{definition}

\begin{lemma}[One-sided-inverse contraction of an alternating network]
    \label{lem:mpug_inverse_contraction_open_alternating_network}
    \lean{MPSTensor.inverseContraction_openAlternatingNetwork}
    \leanok
    \uses{def:injective, def:decomposition_map,
        def:mpug_open_alternating_network}
    Suppose that every $A_i$ is injective.  For the matrix unit $E_{z_i,q_i}$,
    let $c_i^{\sigma_i}(z_i,q_i)$ be its coefficients under a one-sided
    inverse of $A_i$, so that
    \begin{align}
        \sum_{\sigma_i}c_i^{\sigma_i}(z_i,q_i)A_i^{\sigma_i}
         & =E_{z_i,q_i}.
        \notag
    \end{align}
    As in the simultaneous inverse-tensor contraction of
    \cite[lines~2365--2392]{FrancoRubio2025MPUGauging}, every horizontal path
    $z=(z_i)_{i=0}^{n}$, intermediate path $q=(q_i)_{i=0}^{n-1}$, and $\tau$
    then satisfy
    \begin{align}
        \sum_{\sigma_0,\ldots,\sigma_{n-1}}
        \left(\prod_{i=0}^{n-1}c_i^{\sigma_i}(z_i,q_i)\right)
        \mathcal N_{A,B}(\sigma,\tau)_{z_0,z_n}
         & =\prod_{i=0}^{n-1}(B_i^{\tau_i})_{q_i,z_{i+1}}.
        \notag
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Expanding the connected network and interchanging the finite sums turns the
    sum over $\sigma_i$ at site $i$ into the $(x_i,y_i)$ entry of
    $E_{z_i,q_i}$.  Its two Kronecker deltas force $x_i=z_i$ and $y_i=q_i$, so
    every internal sum collapses to leave the displayed product of entries of
    the $B_i$.
\end{proof}

\begin{theorem}[Nondegenerate corrected alternating-network proportionality]
    \label{thm:mpug_alternating_network_nondegenerate}
    \lean{MPSTensor.exists_scalars_of_openAlternatingNetwork_eq}
    \leanok
    \uses{def:injective, def:mpug_open_alternating_network}
    Let $n\geq 0$, and let $D_0,\ldots,D_n$ be horizontal bond dimensions.
    For each $0\leq i<n$, let $A_i$ be an injective MPS tensor with square bond
    dimension $D_i$, and let $B_i,B'_i$ be tensors whose left and right
    horizontal bond dimensions are $D_i$ and $D_{i+1}$.  Suppose that every
    $B_i$ is nonzero and that the connected open networks agree:
    \begin{align}
        A_0B_0A_1B_1\cdots A_{n-1}B_{n-1}
         & =A_0B'_0A_1B'_1\cdots A_{n-1}B'_{n-1}.
        \label{eq:mpug_alternating_network_equality}
    \end{align}
    Here the equality retains the two horizontal end legs and all physical
    legs.  Then there are scalars $\beta_i\in\C$, for $0\leq i<n$, such that
    \begin{align}
        B_i                      & =\beta_iB'_i,
        \label{eq:mpug_alternating_network_proportionality} \\
        \prod_{i=0}^{n-1}\beta_i & =1.
        \label{eq:mpug_alternating_network_scalar_product}
    \end{align}

    In Lemma~\texttt{lemma:1} of
    \cite[lines~2296--2395]{FrancoRubio2025MPUGauging}, whose omitted
    nonvanishing hypothesis is recorded in
    \cite{gap:fbc25_alternating_network_nondegeneracy}, the truncated symmetry
    $U^L_g$ is unitary at line~2099, lines~2444--2466 identify the network side
    at line~2567 with $(U^L_{g^{-1}})^\dagger$, and lines~3033--3053 identify
    the network side at line~3157 with $\sigma_g(U^L_{g^{-1}})^\dagger$ for
    $\sigma_g\in\{1,-1\}$; because both networks are nonzero and any vanishing
    intervening factor would annihilate the entire network, the factors on
    these sides satisfy $B_i\ne 0$ at both applications.
\end{theorem}

\begin{proof}\leanok
    \uses{def:decomposition_map,
        lem:mpug_inverse_contraction_open_alternating_network,
        thm:periodic_product_tensor_phase_extraction}
    Applying Lemma~\ref{lem:mpug_inverse_contraction_open_alternating_network}
    to both sides of (\ref{eq:mpug_alternating_network_equality}) with matrix
    units chosen to prescribe arbitrary entries of all $B_i$ isolates those
    entries in the resulting connected horizontal contractions, which yields
    the pure-tensor identity
    \begin{align}
        \bigotimes_{i=0}^{n-1}B_i
         & =\bigotimes_{i=0}^{n-1}B'_i.
        \label{eq:mpug_alternating_network_pure_tensor_equality}
    \end{align}
    Applying Theorem~\ref{thm:periodic_product_tensor_phase_extraction} to
    (\ref{eq:mpug_alternating_network_pure_tensor_equality}) with global scalar
    $1$ and the nonzero reference entries of the $B_i$ then gives the
    factorwise proportionalities and
    (\ref{eq:mpug_alternating_network_scalar_product}), and inverting the
    resulting nonzero scalars gives the orientation
    (\ref{eq:mpug_alternating_network_proportionality}).
\end{proof}

\subsection{The second network and unitarity of the staircase gates}

\begin{theorem}[Isometries on the middle two legs]
    \label{thm:mpug_staircase_middle_isometry}
    \lean{MPOTensor.staircaseMiddle_isIsometry}
    \leanok
    \uses{def:mpug_staircase_middle}
    Let $A,B,C,E$ be finite index sets, let $B',C'$ be index sets, and let
    $M$ be a complex matrix with rows indexed by $B\times C$ and columns
    by $B'\times C'$. If $M^\dagger M=I$, then
    $(M_{23})^\dagger M_{23}=I$, with the middle-leg placement of
    Definition~\ref{def:mpug_staircase_middle}.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_staircase_middle}
    Tensoring an isometry with identity matrices preserves its Gram
    matrix, and regrouping the four indices preserves this identity.
\end{proof}

\begin{theorem}[The second staircase network]
    \label{thm:mpug_staircase_second_network}
    \lean{MPOTensor.SourceFactors.sourceWL_second_network_apply,
        MPOTensor.SourceFactors.sourceWL_second_network}
    \leanok
    \discussion{7314}
    \uses{def:mpug_staircase_gates, def:mpu_source_uv}
    Let $\mathcal U$ have source factors as in
    Definition~\ref{def:mpug_staircase_gates}, and form $u,v,w_L$ from
    those same factors. Write $d_l,d_r$ for the source-cut ranks. Under
    the natural associativity identifications of the three tensor factors,
    \begin{align}
        (w_L\otimes I_d)(I_d\otimes w_L)
         & =(I_{d_l}\otimes v)(u\otimes I_{d_l}).
    \end{align}
    Both sides map $\C^d\otimes\C^d\otimes\C^{d_l}$ to
    $\C^{d_l}\otimes\C^d\otimes\C^d$. Explicitly, for
    $0\leq l,k<d_l$ and $0\leq i,j,a,b<d$,
    \begin{align}
        \sum_{t=0}^{d_l-1}(w_L)_{(l,i),(a,t)}(w_L)_{(t,j),(b,k)}
         & =\sum_{r=0}^{d_r-1}u_{(l,r),(a,b)}v_{(i,j),(r,k)}.
    \end{align}
    Following the second network in
    \cite[lines~977--1011]{FrancoRubio2025MPUGauging}, this identity uses
    two copies of $w_L$ and requires no simplicity or unitarity hypothesis.
\end{theorem}

\begin{proof}\leanok
    \uses{def:mpug_staircase_gates, def:mpu_source_uv,
        thm:mpu_source_factorization_entries}
    Expanding the crossed gates and using $M_2=X_2Y_2$ reduces the
    left-hand entry to
    \begin{align}
        \sum_{\alpha,\beta=0}^{D-1}
        (Y_2)_{l,(a,\alpha)}\mathcal U^{i,b}_{\alpha\beta}
        (X_2)_{(\beta,j),k}.
    \end{align}
    Expanding $u,v$ and using $M_1=X_1Y_1$ gives the same expression on
    the right. The displayed matrix equality is this entry identity
    after regrouping the three external indices.
\end{proof}

\begin{theorem}[Isometry of the first staircase network]
    \label{thm:mpug_staircase_first_network_isometry}
    \lean{MPOTensor.SourceFactors.sourceWL_kronecker_sourceWR_isIsometry}
    \leanok
    \uses{def:mpug_staircase_gates, def:mpu_source_uv}
    Let $\mathcal U$ have source factors as in
    Definition~\ref{def:mpug_staircase_gates}. If the associated operators
    $u$ and $v$ are unitary between their respective spaces, then
    \begin{align}
        (w_L\otimes w_R)^\dagger(w_L\otimes w_R) & =I.
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_staircase_network_cancelled,
        thm:mpug_staircase_middle_isometry}
    Theorem~\ref{thm:mpug_staircase_network_cancelled} expresses
    $w_L\otimes w_R$ as $v_{23}(u\otimes u)(u^\dagger)_{23}$.
    Each factor is an isometry, so their product is an isometry.
\end{proof}

\begin{theorem}[Isometry of the second staircase network]
    \label{thm:mpug_staircase_second_network_isometry}
    \lean{MPOTensor.SourceFactors.sourceWL_second_network_isIsometry}
    \leanok
    \uses{def:mpug_staircase_gates, def:mpu_source_uv}
    Let $\mathcal U$ have source factors as in
    Definition~\ref{def:mpug_staircase_gates}, with left source-cut rank
    $d_l$. If $u^\dagger u=I$ and $v^\dagger v=I$, then
    $K=(w_L\otimes I_d)(I_d\otimes w_L)$ satisfies $K^\dagger K=I$,
    with the three-leg associativity identifications understood.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_staircase_second_network}
    By Theorem~\ref{thm:mpug_staircase_second_network},
    $K=(I_{d_l}\otimes v)(u\otimes I_{d_l})$.
    Tensoring with an identity, regrouping indices, and composition
    preserve isometries.
\end{proof}

\begin{theorem}[Unitary staircase gates]
    \label{thm:mpug_staircase_gates_unitary}
    \lean{MPOTensor.IsMPUCanonicalFormII.sourceWL_sourceWR_isUnitaryBetween}
    \leanok
    \discussion{7314}
    \uses{def:mpu_canonical_form_ii, def:mpu_simple,
        def:mpug_staircase_gates}
    Let $\mathcal U$ be a simple MPU in canonical form II, with recorded
    positive diagonal fixed matrix $\rho$. Construct the source factors
    from $\mathcal U$ and this same $\rho$, and let $d_l,d_r$ be their
    source-cut ranks. Then, as in part~(a) of the staircase-gate
    corollary in \cite[lines~811--1012]{FrancoRubio2025MPUGauging}, the
    staircase gates are unitary maps
    \begin{align}
        w_L & :\C^d\otimes\C^{d_l}\longrightarrow\C^{d_l}\otimes\C^d, \\
        w_R & :\C^{d_r}\otimes\C^d\longrightarrow\C^d\otimes\C^{d_r}.
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{ThmFund1, thm:mpug_staircase_first_network_isometry,
        thm:mpug_kronecker_identity_factors,
        thm:mpug_staircase_second_network_isometry}
    Theorem~\ref{ThmFund1} gives unitary $u,v$ and $d_rd_l=d^2>0$.
    Theorem~\ref{thm:mpug_staircase_first_network_isometry} gives
    \begin{align}
        (w_L^\dagger w_L)\otimes(w_R^\dagger w_R) & =I.
    \end{align}
    By Theorem~\ref{thm:mpug_kronecker_identity_factors}, the two Gram
    matrices are $\delta I$ and $\delta^{-1}I$ for a nonzero scalar
    $\delta$. Positivity of the first Gram matrix makes $\delta\geq0$.
    Each of the two left-gate layers in
    $K=(w_L\otimes I_d)(I_d\otimes w_L)$ has Gram matrix $\delta I$,
    so $K^\dagger K=\delta^2 I$. By
    Theorem~\ref{thm:mpug_staircase_second_network_isometry}, this equals
    $I$. Thus $\delta^2=1$, equivalently $\delta^{-2}=1$, and positivity
    gives $\delta=1$. Both gates are isometries between spaces of equal
    dimension, hence unitary.
\end{proof}

\begin{definition}[Truncated symmetry]
    \label{def:mpug_truncated_symmetry}
    \lean{MPOTensor.truncatedSymmetryOfEndpoints,
        MPOTensor.SourceFactors.truncatedSymmetry}
    \leanok
    \discussion{7319}
    \uses{def:mpu_source_matrix_one, def:mpu_source_matrix_two,
        def:mpo_eval_word, def:mpu_weighted_source_factors}
    Let $\mathcal U$ be an MPO tensor of physical dimension $d$ and bond
    dimension $D$, and let $d_l=\operatorname{rank}M_2$ and
    $d_r=\operatorname{rank}M_1$. Choose arbitrary complex endpoint matrices
    $Y_2\colon\C^d\otimes\C^D\to\C^{d_l}$ and
    $Y_1\colon\C^D\otimes\C^d\to\C^{d_r}$.
    For $N\geq0$ and physical words $\sigma,\tau$ of length $N$, put
    \begin{align}
        \mathcal U^{\sigma,\tau}
         & =\mathcal U^{\sigma_0,\tau_0}\cdots\mathcal U^{\sigma_{N-1},\tau_{N-1}}, \\
        (U^{N+2})_{(l,\sigma,r),(a,\tau,b)}
         & =\sum_{\alpha,\beta=0}^{D-1}
        (Y_2)_{l,(a,\alpha)}
        (\mathcal U^{\sigma,\tau})_{\alpha,\beta}
        (Y_1)_{r,(\beta,b)}.
        \label{eq:mpug_truncated_symmetry_contraction}
    \end{align}
    Here $0\leq l<d_l$, $0\leq r<d_r$, and $a,b$ are physical
    indices. The empty ordered matrix product is $I_D$. Thus
    \begin{align}
        U^{N+2}\colon \C^d\otimes(\C^d)^{\otimes N}\otimes\C^d
         & \longrightarrow
        \C^{d_l}\otimes(\C^d)^{\otimes N}\otimes\C^{d_r}.
    \end{align}
    As in the endpoint contraction of
    \cite[lines~2062--2100]{FrancoRubio2025MPUGauging} with the group
    label suppressed, the source superscript is the total length
    $L=N+2\geq2$, including the two endpoint triangles, while $N$ counts
    only the bulk tensors; no cut factorization is needed.
    For the normalized source factors of
    Definition~\ref{def:mpu_weighted_source_factors}, choosing their
    matrices $Y_2,Y_1$ recovers the source-factor endpoint contraction.
    No group representation or unitarity assertion is included in this
    definition; arbitrary endpoint matrices need not give a unitary map.
\end{definition}

\begin{theorem}[The zero-bulk contraction]
    \label{thm:mpug_truncated_symmetry_zero}
    \lean{MPOTensor.truncatedSymmetryOfEndpoints_zero,
        MPOTensor.SourceFactors.truncatedSymmetry_zero}
    \leanok
    \uses{def:mpug_truncated_symmetry, def:mpu_source_uv}
    Let $\mathcal U$ and the arbitrary endpoint matrices $Y_2,Y_1$ have
    the dimensions in Definition~\ref{def:mpug_truncated_symmetry}.
    For endpoint indices $l,r,a,b$ and the empty words $\varnothing$,
    \begin{align}
        (U^2)_{(l,\varnothing,r),(a,\varnothing,b)}
         & =\sum_{\alpha=0}^{D-1}
        (Y_2)_{l,(a,\alpha)}(Y_1)_{r,(\alpha,b)}.
        \label{eq:mpug_truncated_symmetry_zero}
    \end{align}
    This is total length two and requires no cut factorization.
    For normalized source factors,
    the right-hand side is $u_{(l,r),(a,b)}$, with $u$ as in
    Definition~\ref{def:mpu_source_uv}.
\end{theorem}
\begin{proof}\leanok
    \uses{def:mpug_truncated_symmetry, def:mpo_eval_word, def:mpu_source_uv}
    The bulk product $\mathcal U^{\varnothing,\varnothing}$ is the empty
    ordered product $I_D$, so $(\mathcal U^{\varnothing,\varnothing})_{\alpha\beta}
        =\delta_{\alpha\beta}$. Instantiating
    (\ref{eq:mpug_truncated_symmetry_contraction}) at $N=0$ gives
    \begin{align}
        (U^2)_{(l,\varnothing,r),(a,\varnothing,b)}
         & =\sum_{\alpha,\beta=0}^{D-1}(Y_2)_{l,(a,\alpha)}
        \delta_{\alpha\beta}(Y_1)_{r,(\beta,b)}
        =\sum_{\alpha=0}^{D-1}(Y_2)_{l,(a,\alpha)}(Y_1)_{r,(\alpha,b)}.
    \end{align}
    For normalized source factors this is the defining contraction of $u$.
\end{proof}

\begin{theorem}[Left growth from the second cut]
    \label{thm:mpug_truncated_symmetry_left_growth}
    \lean{MPOTensor.truncatedSymmetryOfEndpoints_cons,
        MPOTensor.SourceFactors.truncatedSymmetry_cons}
    \leanok
    \discussion{7319}
    \uses{def:mpug_truncated_symmetry, def:mpu_source_matrix_two,
        def:mpug_staircase_gates}
    Let $\mathcal U$ be an MPO tensor with physical dimension $d$ and bond
    dimension $D$, and put $d_l=\operatorname{rank}M_2$ and
    $d_r=\operatorname{rank}M_1$. Let
    $Y_1\colon\C^D\otimes\C^d\to\C^{d_r}$,
    $Y_2\colon\C^d\otimes\C^D\to\C^{d_l}$, and
    $X_2\colon\C^{d_l}\to\C^D\otimes\C^d$ be complex matrices
    satisfying only $M_2=X_2Y_2$, that is,
    \begin{align}
        \mathcal U^{i,b}_{\gamma,\alpha}
         & =\sum_{t=0}^{d_l-1}(X_2)_{(\gamma,i),t}(Y_2)_{t,(b,\alpha)}.
    \end{align}
    Define the crossed contraction
    \begin{align}
        (w_L)_{(l,i),(a,t)}
         & =\sum_{\gamma=0}^{D-1}(Y_2)_{l,(a,\gamma)}(X_2)_{(\gamma,i),t}.
    \end{align}
    For $N\geq0$, physical words $\sigma,\tau$ of length $N$,
    source indices $l,r$, and physical indices $i,a,b,c$,
    \begin{align}
        (U^{N+3})_{(l,i\sigma,r),(a,b\tau,c)}
         & =\sum_{t=0}^{d_l-1}(w_L)_{(l,i),(a,t)}
        (U^{N+2})_{(t,\sigma,r),(b,\tau,c)}.
        \label{eq:mpug_truncated_symmetry_left_growth}
    \end{align}
    In the left-end factorization step of
    \cite[lines~2101--2174]{FrancoRubio2025MPUGauging} via the second
    decomposition of \cite[lines~479--530]{Cirac2017MPU}, the right
    endpoint $Y_1$ is arbitrary and unchanged, no first-cut factorization
    is assumed, and for normalized source factors $w_L$ recovers the
    movement operator of Definition~\ref{def:mpug_staircase_gates}.
\end{theorem}
\begin{proof}\leanok
    \uses{def:mpug_truncated_symmetry, def:mpu_source_matrix_two,
        def:mpo_eval_word}
    The second cut factorization gives, for each virtual index $\alpha$,
    \begin{align}
        \sum_{\gamma=0}^{D-1}(Y_2)_{l,(a,\gamma)}
        \mathcal U^{i,b}_{\gamma,\alpha}
         & =\sum_{\gamma=0}^{D-1}\sum_{t=0}^{d_l-1}
        (Y_2)_{l,(a,\gamma)}(X_2)_{(\gamma,i),t}
        (Y_2)_{t,(b,\alpha)}                                           \\
         & =\sum_{t=0}^{d_l-1}(w_L)_{(l,i),(a,t)}(Y_2)_{t,(b,\alpha)}.
    \end{align}
    Expanding $\mathcal U^{i\sigma,b\tau}
        =\mathcal U^{i,b}\mathcal U^{\sigma,\tau}$, substituting this equality
    into the endpoint contraction, and regrouping the finite sums yields
    \begin{align}
        (U^{N+3})_{(l,i\sigma,r),(a,b\tau,c)}
         & =\sum_{t=0}^{d_l-1}(w_L)_{(l,i),(a,t)}
        \sum_{\alpha,\beta=0}^{D-1}(Y_2)_{t,(b,\alpha)}
        (\mathcal U^{\sigma,\tau})_{\alpha,\beta}(Y_1)_{r,(\beta,c)} \\
         & =\sum_{t=0}^{d_l-1}(w_L)_{(l,i),(a,t)}
        (U^{N+2})_{(t,\sigma,r),(b,\tau,c)}.
    \end{align}

\begin{theorem}[Two-sided movement-operator growth]
    \label{thm:mpug_truncated_symmetry_growth}
    \lean{MPOTensor.truncatedSymmetryOfEndpoints_cons_snoc,

    \leanok
    \discussion{7319}
    \uses{def:mpug_truncated_symmetry, def:mpu_source_matrix_one,

    Let $\mathcal U$ be an MPO tensor of physical dimension $d$ and bond
    dimension $D$, with $d_l=\operatorname{rank}M_2$ and
    $d_r=\operatorname{rank}M_1$. Let
    $X_1\colon\C^{d_r}\to\C^d\otimes\C^D$,
    $Y_1\colon\C^D\otimes\C^d\to\C^{d_r}$,
    $X_2\colon\C^{d_l}\to\C^D\otimes\C^d$, and
    $Y_2\colon\C^d\otimes\C^D\to\C^{d_l}$ be complex matrices
    satisfying only $M_1=X_1Y_1$ and $M_2=X_2Y_2$. In coordinates,
    \begin{align}
        \mathcal U^{j,c}_{\beta,\delta}
         & =\sum_{s=0}^{d_r-1}(X_1)_{(j,\delta),s}(Y_1)_{s,(\beta,c)},  \\
        \mathcal U^{i,b}_{\gamma,\alpha}
         & =\sum_{t=0}^{d_l-1}(X_2)_{(\gamma,i),t}(Y_2)_{t,(b,\alpha)}.
    \end{align}
    Form $U^{N+2}$ using the endpoints $Y_2,Y_1$, and define the crossed
    contractions
    \begin{align}
        (w_L)_{(l,i),(a,t)}
         & =\sum_{\gamma=0}^{D-1}(Y_2)_{l,(a,\gamma)}(X_2)_{(\gamma,i),t}, \\
        (w_R)_{(j,r),(s,e)}
         & =\sum_{\delta=0}^{D-1}(X_1)_{(j,\delta),s}(Y_1)_{r,(\delta,e)}.
    \end{align}
    For $N\geq0$, physical words $\sigma,\tau$ of length $N$, source
    indices $l,r$, and physical indices $i,j,a,b,c,e$, one has
    \begin{align}
        (U^{N+4})_{(l,i\sigma j,r),(a,b\tau c,e)}
         & =\sum_{t=0}^{d_l-1}\sum_{s=0}^{d_r-1}
        (w_L)_{(l,i),(a,t)}
        (U^{N+2})_{(t,\sigma,s),(b,\tau,c)}
        (w_R)_{(j,r),(s,e)}.
        \label{eq:mpug_truncated_symmetry_growth}
    \end{align}
    Here concatenation $i\sigma j$ inserts $i$ before the word and $j$ after
    it. As in the growth diagram of
    \cite[lines~2101--2174]{FrancoRubio2025MPUGauging}, this
    source-factorization identity adds one bulk site at each end using
    only the two decompositions of \cite[lines~479--530]{Cirac2017MPU}.
    For normalized source factors, the crossed contractions yield the
    movement operators of Definition~\ref{def:mpug_staircase_gates},
    extending the identity to the source-factor endpoint contraction.
    No simplicity or unitarity assumption on $\mathcal U$ is needed. These
    algebraic identities do not assert unitarity for arbitrary factors, the
    finite-group physical specialization, or the complementary finite-chain
    movement relation.
\end{theorem}
\begin{proof}\leanok
    \uses{def:mpu_source_matrix_one, def:mpu_source_matrix_two,
        def:mpug_truncated_symmetry, def:mpo_eval_word}
    The enlarged bulk product expands as
    $\mathcal U^{i\sigma j,b\tau c}
        =\mathcal U^{i,b}\mathcal U^{\sigma,\tau}\mathcal U^{j,c}$.
    The second source-cut factorization gives
    \begin{align}
        \sum_\gamma(Y_2)_{l,(a,\gamma)}\mathcal U^{i,b}_{\gamma,\alpha}
         & =\sum_t(w_L)_{(l,i),(a,t)}(Y_2)_{t,(b,\alpha)}.
    \end{align}
    The first source-cut factorization similarly gives
    \begin{align}
        \sum_\delta\mathcal U^{j,c}_{\beta,\delta}(Y_1)_{r,(\delta,e)}
         & =\sum_s(Y_1)_{s,(\beta,c)}(w_R)_{(j,r),(s,e)}.
    \end{align}
    Substituting both equalities into the endpoint contraction and
    regrouping the finite sums leaves the interior contraction $U^{N+2}$.
\end{proof}

\begin{theorem}[Kronecker product of unitary maps]
    \label{thm:matrix_unitary_between_kronecker}
    \lean{Matrix.IsUnitaryBetween.kronecker}
    \leanok
    Let $A,B,C,E$ be finite sets, possibly empty, and let
    $M\colon\C^B\to\C^A$ and $P\colon\C^E\to\C^C$ be unitary maps:
    $M^\dagger M=I_B$, $MM^\dagger=I_A$, $P^\dagger P=I_E$, and
    $PP^\dagger=I_C$. Then $M\otimes P$ is a unitary map from
    $\C^{B\times E}$ to $\C^{A\times C}$.
\end{theorem}

\begin{proof}\leanok
    The adjoint and multiplication identities for Kronecker products give
    \begin{align}
        (M\otimes P)^\dagger(M\otimes P)
         & =(M^\dagger M)\otimes(P^\dagger P)
        =I_B\otimes I_E=I_{B\times E},        \\
        (M\otimes P)(M\otimes P)^\dagger
         & =(MM^\dagger)\otimes(PP^\dagger)

\end{proof}

\subsection{Periodic sewing of the two source arcs}

The periodic chain carries two marked sites. Cutting the marked tensors by
the two source cuts produces two open arcs that share both marked sites,
and the shared rank indices sew them back to the periodic trace. The arcs
have lengths $n+2$ and $m+2$ on a ring of $n+m+2$ sites; the second is not
the disjoint complement of the first.

\begin{definition}[The two sewn source arcs]
    \label{def:mpug_source_periodic_arcs}
    \lean{MPOTensor.sourcePeriodicArcA, MPOTensor.sourcePeriodicArcB}
    \leanok
    \uses{def:mpo_eval_word, def:mpu_right_rank, def:mpu_left_rank}
    Let $\mathcal U$ be an MPO tensor of dimensions $d,D$, and let
    $X_1,Y_1$ and $X_2,Y_2$ be any factorizations of its two source cuts
    through $\C^r$ and $\C^\ell$. For $N\geq0$ and configurations
    $p,q\in\Omega_N$, define matrices $A^{\mathrm{sew}}_N$ and
    $B^{\mathrm{sew}}_N$ by
    \begin{align}
        (A^{\mathrm{sew}}_N)_{(s,(p,t)),(a,(q,b))}
         & =\sum_{\alpha,\beta=0}^{D-1}(Y_2)_{s,(a,\alpha)}
        \mathcal U[p,q]_{\alpha\beta}(Y_1)_{t,(\beta,b)},   \\
        (B^{\mathrm{sew}}_N)_{(a,(p,b)),(t,(q,s))}
         & =\sum_{\alpha,\beta=0}^{D-1}(X_1)_{(a,\alpha),t}
        \mathcal U[p,q]_{\alpha\beta}(X_2)_{(\beta,b),s},
    \end{align}
    with $0\leq s<\ell$, $0\leq t<r$, and physical indices $a,b$. In the
    local algebra of the sewing referred to in
    \cite[lines~5486--5487]{FrancoRubio2025MPUGauging} via the source
    decompositions at lines~5390--5432, both words are read in their given
    order without reversal, and the row order $(\ell,r)$ of the first arc is
    the order in which its endpoints are traversed.
\end{definition}

\begin{theorem}[Sewing the periodic trace at two marked sites]
    \label{thm:mpug_source_periodic_sewing}
    \lean{MPOTensor.trace_two_marked_evalWord_eq_sourcePeriodic_sewing,
        MPOTensor.mpo_two_marked_entry_eq_sourcePeriodic_sewing}
    \leanok
    \uses{def:mpug_source_periodic_arcs, def:mpo_eval_word, def:mpo_family,
        def:mpu_source_matrix_one, def:mpu_source_matrix_two}
    Let $\mathcal U$ be an MPO tensor with $M_1(\mathcal U)=X_1Y_1$ and
    $M_2(\mathcal U)=X_2Y_2$. Fix bulk lengths $n,m$, marked physical
    indices $a,a',b,b'$, and configurations $p,p'\in\Omega_n$,
    $q,q'\in\Omega_m$. Then
    \begin{align}
        \tr\!\left(\mathcal U^{aa'}\mathcal U[p,p']
        \mathcal U^{bb'}\mathcal U[q,q']\right)
         & =\sum_{t=0}^{r-1}\sum_{s=0}^{\ell-1}
        (B^{\mathrm{sew}}_n)_{(a,(p,b)),(t,(p',s))}
        (A^{\mathrm{sew}}_m)_{(s,(q,t)),(b',(q',a'))}.
        \label{eq:mpug_source_periodic_sewing}
    \end{align}
    The same sum is the entry of the periodic operator
    $\mathcal U^{(n+m+2)}$ at the concatenated configurations
    $(a,p,b,q)$ and $(a',p',b',q')$. The ring has $n+m+2$ sites, including
    two distinct marked sites even when $n=m=0$, and the two
    endpoint-inclusive arcs overlap at both marked sites. This identity
    requires only the two cut factorizations, without assuming canonicality,
    simplicity, normalization, or unitarity.
\end{theorem}
\begin{proof}\leanok
    \uses{def:mpug_source_periodic_arcs, def:mpu_source_matrix_one,
        def:mpu_source_matrix_two, def:mpo_eval_word, def:mpo_family}
    With $P=\mathcal U[p,p']$ and $Q=\mathcal U[q,q']$, define the slices
    \begin{align}
        C_{t\alpha}  & =(X_1)_{(a,\alpha),t},  \\
        H_{\delta t} & =(Y_1)_{t,(\delta,a')}, \\
        E_{\beta s}  & =(X_2)_{(\beta,b),s},   \\
        F_{s\gamma}  & =(Y_2)_{s,(b',\gamma)}.
    \end{align}
    Reading the two cut factorizations entrywise gives
    $\mathcal U^{aa'}=HC$ and $\mathcal U^{bb'}=EF$. Cyclicity of the
    trace and associativity then give
    \begin{align}
        \tr\!\left(HCPEFQ\right)
         & =\tr\!\left((CPE)(FQH)\right).
    \end{align}
    The entries of the two products are
    $(CPE)_{ts}=(B^{\mathrm{sew}}_n)_{(a,(p,b)),(t,(p',s))}$ and
    $(FQH)_{st}=(A^{\mathrm{sew}}_m)_{(s,(q,t)),(b',(q',a'))}$, so
    expanding the trace of the product as a double sum over $t$ and $s$
    gives~(\ref{eq:mpug_source_periodic_sewing}). For the operator entry,
    expanding word evaluation along the concatenation $(a,p,b,q)$ by
    splitting off the two marked letters and the two bulk words returns the
    same trace.
\end{proof}

    physical pair. For all bulk lengths $n,m$,
    \begin{align}
        M_{n,m} & =\left(B_n\otimes I_{\Omega_m}\right)C_{n,m}.
        \label{eq:mpug_inverse_periodic_factorization}
    \end{align}
    Neither simplicity, canonical form, nor a phase relation is used; the
    complementary arc is the independently defined $C_{n,m}$, not a
    rearrangement of $M_{n,m}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_source_periodic_sewing, def:mpug_inverse_periodic_grouping,
        def:mpug_inverse_word_endpoints, def:mpug_inverse_cut_factors,
        thm:mpug_inverse_cut_factorization, thm:mpu_source_factorizations}
    Applying Theorem~\ref{thm:mpug_source_periodic_sewing} to the
    factorization $M_1=\widehat X_1\widehat Y_1$ of
    Theorem~\ref{thm:mpug_inverse_cut_factorization} and the factorization
    $M_2=X_2Y_2$ of Theorem~\ref{thm:mpu_source_factorizations} shows that
    the two sewn arcs of Definition~\ref{def:mpug_source_periodic_arcs}
    built from these factors are $B_n$ and $A_m$. On the right-hand side of
    (\ref{eq:mpug_inverse_periodic_factorization}) the identity factor
    contributes $\delta_{q,q'}$ on the second bulk and the Kronecker
    delta of $C_{n,m}$ contributes $\delta_{p,p'}$, so the matrix product
    collapses to the double sum of
    (\ref{eq:mpug_source_periodic_sewing}) at the grouped configurations.
\end{proof}

\begin{theorem}[One phase in the periodic composition]
    \label{thm:mpug_inverse_periodic_one_phase}
    \lean{MPOTensor.inverseCompatibleFirstArcEmbedding_mul_groupedMPO}
    \leanok
    \uses{def:mpug_inverse_periodic_grouping, def:mpu_canonical_form_ii,
        def:mpu_simple, def:mpdo_physical_adjoint}
    Let $\mathcal U$ be a simple MPU of dimensions $d,D$ in canonical form
    II. Suppose that $T$ is unitary, that
    $(\mathcal U^\dagger)^{ij}=T^\dagger\mathcal U^{ij}T$ for every
    physical pair, and that $T\overline T=\sigma I_D$ for $\sigma\in\C$.
    For all bulk lengths $n,m$,
    \begin{align}
        \widehat A_{n,m}M_{n,m} & =\sigma C_{n,m}.
    \end{align}
    Exactly one phase occurs, independently of $n$ and $m$. The two
    endpoint-inclusive arcs share both marked sites, so this is a
    shared-boundary composition and not the disjoint-complement identity
    of \cite[lines~5486--5487]{FrancoRubio2025MPUGauging}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:mpug_inverse_periodic_factorization,
        thm:mpug_inverse_word_phase_cancellation,
        def:mpug_inverse_periodic_grouping}
    Substituting (\ref{eq:mpug_inverse_periodic_factorization}) and
    using $(A_n\otimes I_{\Omega_m})(B_n\otimes I_{\Omega_m})
        =(A_nB_n)\otimes I_{\Omega_m}$ gives
    \begin{align}
        \widehat A_{n,m}M_{n,m}
         & =\left((A_nB_n)\otimes I_{\Omega_m}\right)C_{n,m}.
    \end{align}
    Theorem~\ref{thm:mpug_inverse_word_phase_cancellation} gives
    $A_nB_n=\sigma I$ on the second-cut source space, and the scalar passes
    through the Kronecker product. The cancellation used is the one on
    the source space, not the reverse product $B_nA_n$.
\end{proof}

\subsection{Physical closures of the source factors}

\begin{theorem}[The canonical one-letter physical contraction]
    \label{thm:mpug_one_letter_physical_contraction}
    \lean{MPOTensor.IsMPUCanonicalFormII.oneLetter_physical_contraction}
    \leanok
    \discussion{7314}
    \uses{def:mpu_canonical_form_ii, def:mpu_simple}
    Let $\mathcal U$ be a simple MPU in canonical form II, with physical
    dimension $d$, bond dimension $D$, and recorded fixed matrix $\rho$.
    For $0\leq p,q<d$, the one-letter diagram in the proof of part~(b) of
    the staircase-gate corollary in
    \cite[lines~1013--1030]{FrancoRubio2025MPUGauging} gives
    \begin{align}
        \sum_{x,\alpha,\beta=0}^{D-1}\sum_{z=0}^{d-1}
        \overline{\mathcal U^{z,p}_{x\alpha}}
        \mathcal U^{z,q}_{x\beta}\rho_{\beta\alpha}
         & =\delta_{p,q}.
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_simple_canonical_boundary_contractions,
        thm:mpu_all_later_simple_blockings}
    Theorem~\ref{thm:mpug_simple_canonical_boundary_contractions}
    gives the two-letter insertion identity with the recorded canonical
    boundaries $(\Phi|_{(a,b)}=\delta_{a,b}$ and
    $|\rho)_{(a,b)}=\rho_{ba}$. Applying
    Theorem~\ref{thm:mpu_all_later_simple_blockings} to this fixed pair
    gives the one-letter contraction. Expanding its coefficient yields
    \begin{align}
        W^{pq}W^{kl} & =W^{pq}|\rho)(\Phi|W^{kl},                     \\
        (\Phi|W^{pq}|\rho)
                     & =\sum_{x,\alpha,\beta=0}^{D-1}\sum_{z=0}^{d-1}
        \overline{\mathcal U^{z,p}_{x\alpha}}
        \mathcal U^{z,q}_{x\beta}\rho_{\beta\alpha}
        =\delta_{p,q}.
    \end{align}
    The left boundary identifies the two bond indices $x$, while the
    right boundary contributes $\rho_{\beta\alpha}$.
\end{proof}

\begin{theorem}[Physical closures of the two source factors]
    \label{thm:mpug_source_y_physical_contractions}
    \lean{MPOTensor.IsMPUCanonicalFormII.sourceY₂_physical_contraction,
        MPOTensor.IsMPUCanonicalFormII.sourceY₁_physical_contraction}
    \leanok
    \discussion{7314}
    \uses{def:mpu_canonical_form_ii, def:mpu_simple,
        def:mpu_weighted_source_factors}
    Let $\mathcal U$ be a simple MPU in canonical form II, with physical
    dimension $d$, bond dimension $D$, and recorded positive diagonal
    fixed matrix $\rho$. Let $Y_1,Y_2$ be the source factors constructed
    from this tensor and this same $\rho$, with source-cut ranks $d_r,d_l$.
    For $0\leq p,q<d$,
    \begin{align}
        \sum_{l=0}^{d_l-1}\sum_{\alpha,\beta=0}^{D-1}
        \overline{(Y_2)_{l,(p,\alpha)}}(Y_2)_{l,(q,\beta)}
        \rho_{\beta\alpha}
         & =\delta_{p,q}, \\
        \sum_{r=0}^{d_r-1}\sum_{\alpha=0}^{D-1}
        \overline{(Y_1)_{r,(\alpha,p)}}(Y_1)_{r,(\alpha,q)}
         & =\delta_{p,q}.
    \end{align}
    In the two diagrams of part~(b) of the staircase-gate corollary in
    \cite[lines~869--917]{FrancoRubio2025MPUGauging}, the open indices are
    physical, so these are identities on $\C^d$ rather than on the
    source-cut spaces.
    % Together with thm:mpug_staircase_gates_unitary this establishes
    % cor:mpu for the source factors of the canonical-form-II presentation.
    % No adjoint-simplicity hypothesis or subsequent eq:MPUnice3/4
    % source-factor constants are asserted here.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:mpug_one_letter_physical_contraction,
        thm:mpu_source_y_two_rotated_gram,
        thm:mpu_source_v_four_u_expansions}
    Theorem~\ref{thm:mpu_source_y_two_rotated_gram} at $a=\alpha,b=\beta$
    gives
    \begin{align}
        \sum_{l=0}^{d_l-1}\overline{(Y_2)_{l,(p,\alpha)}}(Y_2)_{l,(q,\beta)}
         & =\sum_{x=0}^{D-1}\sum_{z=0}^{d-1}
        \overline{\mathcal U^{z,p}_{x\alpha}}\mathcal U^{z,q}_{x\beta}.
    \end{align}
    Multiplying by $\rho_{\beta\alpha}$ and summing over $\alpha,\beta$
    rewrites the first sum as the one-letter contraction of
    Theorem~\ref{thm:mpug_one_letter_physical_contraction}. Theorem~\ref{thm:mpu_source_v_four_u_expansions}
    at the diagonal case $a_1=b_1=\alpha$, $p_1=p$, $q_1=q$ gives
    \begin{align}
        \sum_{r=0}^{d_r-1}\overline{(Y_1)_{r,(\alpha,p)}}(Y_1)_{r,(\alpha,q)}
         & =\sum_{\beta,\gamma=0}^{D-1}\sum_{z=0}^{d-1}
        \overline{\mathcal U^{z,p}_{\alpha\beta}}\rho_{\beta\gamma}
        \mathcal U^{z,q}_{\alpha\gamma}.
    \end{align}
    Summing over $\alpha$ rewrites the second sum in the same way, after
    renaming $(\alpha,\beta,\gamma)$ to $(x,\alpha,\beta)$; diagonality of
    $\rho$ gives $\rho_{\alpha\beta}=\rho_{\beta\alpha}$. Both sums
    therefore equal $\delta_{p,q}$. These Gram identities cancel the $X_2$
    and $X_1$ factors using their respective normalizations, as in
    \cite[lines~1030--1050]{FrancoRubio2025MPUGauging}.
\end{proof}
