\section{Whole-group tensor bound and simultaneous truncation}
\label{sec:peps_group_truncation}
This section presents the whole-group tensor bound of
\cite[Lemma~6.2]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 125--179, together with the two
dimension-free matrix inequalities used in its proof. All index sets are
finite. For a matrix $M$, write $\lVert M\rVert_1$ for the sum of its
singular values and $\lVert M\rVert_2$ for its Hilbert--Schmidt norm.

\begin{definition}[Nuclear norm]
    \label{def:peps_group_nuclear_norm}
    \lean{Matrix.rectangularTraceNorm}
    \leanok
    For a finite rectangular complex matrix $M$, the nuclear norm $\lVert M\rVert_1$
    is the sum of the singular values of $M$.
\end{definition}

\begin{lemma}[H\"older bound for a product with a transpose]
    \label{lem:peps_group_nuclear_holder}
    \lean{Matrix.rectangularTraceNorm_mul_transpose_le}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    For finite complex matrices $A$ and $B$ with the same column index set,
    $\lVert AB^{\mathsf T}\rVert_1\le\lVert A\rVert_2\lVert B\rVert_2$.
\end{lemma}

\begin{proof}\leanok
    Let $w_i$ be an orthonormal basis of right singular vectors of
    $M=AB^{\mathsf T}$, and let $u_i$ be the normalized images $Mw_i$ when
    $Mw_i\ne0$. The $i$th singular value of $M$ is
    $\langle u_i, A(B^{\mathsf T}w_i)\rangle
        =\langle A^{*}u_i, B^{\mathsf T}w_i\rangle
        \le\lVert A^{*}u_i\rVert\,\lVert B^{\mathsf T}w_i\rVert$.
    Summing over $i$ and applying the Cauchy--Schwarz inequality gives
    $\lVert M\rVert_1\le(\sum_i\lVert A^{*}u_i\rVert^2)^{1/2}(\sum_i\lVert B^{\mathsf T}w_i\rVert^2)^{1/2}
        \le\lVert A\rVert_2\lVert B\rVert_2$.
\end{proof}

\begin{lemma}[Truncation to leading singular directions]
    \label{lem:peps_group_orthonormal_truncation}
    \lean{Matrix.exists_orthonormal_truncation, Matrix.orthonormalProjector}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    Let $M$ be a finite complex matrix and $k\ge0$ an integer. There exists an
    orthonormal family of at most $k$ vectors in the column space of $M$
    whose orthogonal projection $Q$ satisfies
    $(k+1)\lVert M-MQ\rVert_2^2\le\lVert M\rVert_1^2$. If $M$ has at most $k$
    columns, then $MQ=M$.
\end{lemma}

\begin{proof}\leanok
    Let $Q$ be the orthogonal projection onto the first $\min(k,n)$ right
    singular vectors of $M$, where $n$ is the number of columns. Then
    $\lVert M-MQ\rVert_2^2=\sum_{j>k}s_j^2$, where
    $s_1\ge s_2\ge\cdots\ge0$ are the singular values of $M$. Since the
    singular values are nonincreasing,
    $(k+1)s_{k+1}\le\sum_j s_j$ and $\sum_{j>k}s_j^2\le s_{k+1}\sum_j s_j$,
    so $(k+1)\sum_{j>k}s_j^2\le(\sum_j s_j)^2=\lVert M\rVert_1^2$.
\end{proof}

\begin{definition}[Coordinate contraction of a tensor graph]
    \label{def:peps_group_contraction}
    \lean{TNLean.PEPS.WholeGroup.contraction, TNLean.PEPS.WholeGroup.vertexNormSq}
    \leanok
    Let $(V,E)$ be a finite graph whose edges $e\in E$ have endpoints
    $t(e),h(e)\in V$ and finite index sets $D_e$. At each vertex $v\in V$, let
    $A_v$ be a tensor with one leg for each edge incident to $v$ and one
    group of open legs with finite configuration set $P_v$. The coordinate
    contraction is
    \[
        Z(\sigma)=\sum_{\beta}\prod_{v\in V}A_v\bigl(\beta|_{v},\sigma_v\bigr),
    \]
    where $\beta$ runs over all edge labellings and $\beta|_v$ is the
    restriction of $\beta$ to the edges incident to $v$. The norm
    $\lVert A_v\rVert$ is its Euclidean norm over all incident and open legs.
\end{definition}

\begin{theorem}[Whole-group tensor bound, norm part]
    \label{thm:peps_group_contraction_norm}
    \lean{TNLean.PEPS.WholeGroup.contraction_norm_le_one,
        TNLean.PEPS.WholeGroup.partialNormSq_le_one}
    \leanok
    \uses{def:peps_group_contraction}
    Suppose that no edge joins a vertex to itself and every vertex tensor
    satisfies $\lVert A_v\rVert\le1$. Then $\lVert Z\rVert\le1$.
    The same bound holds for the partial contraction over any subset of
    vertices, with the open legs of the subset and the edges leaving it kept
    open.
\end{theorem}

\begin{proof}\leanok
    Induct on the vertex subset by adding one vertex at a time.
    Contracting the edges between two disjoint vertex subsets amounts to
    multiplying their partial contractions, reshaped as matrices with the
    shared edges as the contracted index, so submultiplicativity
    $\lVert AB\rVert_2\le\lVert A\rVert_2\lVert B\rVert_2$ preserves the norm
    bound. A single vertex has no self-loops and thus has norm
    $\lVert A_v\rVert\le1$, while the empty subset has norm $1$.
\end{proof}

\begin{theorem}[Whole-group tensor bound, nuclear part]
    \label{thm:peps_group_contraction_nuclear}
    \lean{TNLean.PEPS.WholeGroup.rectangularTraceNorm_flattening_le_one,
        TNLean.PEPS.WholeGroup.flattening,
        TNLean.PEPS.WholeGroup.flattening_contraction}
    \leanok
    \uses{def:peps_group_contraction, def:peps_group_nuclear_norm}
    Under the hypotheses of Theorem~\ref{thm:peps_group_contraction_norm},
    for every subset $T\subseteq V$, the matrix reshaping of $Z$ with rows
    indexed by the open-leg groups of $T$ and columns indexed by those of
    $V\setminus T$ has nuclear norm at most $1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_group_contraction_norm, lem:peps_group_nuclear_holder}
    By Theorem~\ref{thm:peps_group_contraction_norm}, the partial contractions
    $A$ on $T$ and $B$ on $V\setminus T$ satisfy $\lVert A\rVert_2\le1$ and
    $\lVert B\rVert_2\le1$. Grouping the edges between $T$ and $V\setminus T$
    into a single column index writes the reshaping of $Z$ as $AB^{\mathsf T}$,
    so Lemma~\ref{lem:peps_group_nuclear_holder} gives
    $\lVert AB^{\mathsf T}\rVert_1\le\lVert A\rVert_2\lVert B\rVert_2\le1$.
\end{proof}

\begin{lemma}[Simultaneous truncation of open-leg groups]
    \label{lem:peps_group_truncation_abstract}
    \lean{TNLean.PEPS.WholeGroup.exists_group_truncation,
        TNLean.PEPS.WholeGroup.norm_sub_factorMatrix_le,
        TNLean.PEPS.WholeGroup.factorMatrix,
        TNLean.PEPS.WholeGroup.productVector}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    Let $Z\in\bigotimes_{v\in V}\mathbb C^{P_v}$ be a tensor with
    $\lVert Z\rVert\le1$ whose matrix reshaping across each single factor $v$
    and $V\setminus\{v\}$ has nuclear norm at most $1$. Let $G\subseteq V$
    contain every $v$ with $|P_v|>1$, and let $k\ge1$. Then there exist
    orthonormal families of at most $k$ vectors in the factors $\mathbb C^{P_v}$
    whose tensor-product projection maps $Z$ to a tensor $Z_k$ satisfying
    $\lVert Z-Z_k\rVert\le|G|/\sqrt{k}$. Expanded in these orthonormal
    families, $Z_k$ is a sum of at most $k^{|G|}$ product terms with
    coefficients of modulus at most $1$.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_group_orthonormal_truncation}
    For each $v\in V$, applying Lemma~\ref{lem:peps_group_orthonormal_truncation}
    to the reshaping across $\{v\}$ and $V\setminus\{v\}$ yields an orthogonal
    projection $Q_v$ of rank at most $k$ on $\mathbb C^{P_v}$ with
    $\lVert Z-Q_v Z\rVert\le1/\sqrt{k+1}\le1/\sqrt{k}$, and $Q_v Z=Z$ whenever
    $|P_v|\le1$. Because the projections $Q_v$ act on distinct tensor factors,
    they commute and have operator norm at most $1$, so telescoping over
    $v\in G$ gives $\lVert Z-Z_k\rVert\le|G|/\sqrt k$. Since
    $\lVert Z_k\rVert\le\lVert Z\rVert\le1$, every coefficient of $Z_k$ in the
    orthonormal product basis has modulus at most $1$.
\end{proof}

\begin{theorem}[Whole-group tensor bound]
    \label{thm:peps_group_tensor}
    \lean{TNLean.PEPS.WholeGroup.exists_simultaneous_openLeg_truncation,
        TNLean.PEPS.WholeGroup.exists_simultaneous_group_truncation}
    \leanok
    \uses{def:peps_group_contraction}
    Let a finite loopless graph carry at each vertex $v$ a tensor $A_v$ of norm
    at most $1$, and let $Z$ be its coordinate contraction, with open legs
    grouped by their vertex $v\in V$. Then $\lVert Z\rVert\le1$, and $Z$ has
    nuclear norm at most $1$ across every bipartition of $V$. If at most $g$
    vertices have nonempty open-leg groups, then for every integer $k\ge1$
    there exist subspaces of dimension at most $k$ in the respective group
    spaces whose tensor-product projection maps $Z$ to a tensor $Z_k$ with
    $\lVert Z-Z_k\rVert\le g/\sqrt{k}$. In orthonormal bases of these
    subspaces, $Z_k$ is a sum of at most $k^g$ product terms with coefficients
    of modulus at most $1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_group_contraction_norm, thm:peps_group_contraction_nuclear,
        lem:peps_group_truncation_abstract}
    Theorems~\ref{thm:peps_group_contraction_norm}
    and~\ref{thm:peps_group_contraction_nuclear} give $\lVert Z\rVert\le1$ and
    the nuclear-norm bound across every bipartition of $V$. A vertex with no
    open legs has $|P_v|=1$, so the set $G$ of vertices with nonempty open-leg
    groups contains every $v$ with $|P_v|>1$, and
    Lemma~\ref{lem:peps_group_truncation_abstract} gives the truncation.
\end{proof}

\begin{corollary}[Whole-group tensor bound for networks with edge matrices]
    \label{cor:peps_group_tensor_network}
    \lean{TNLean.PEPS.WholeGroup.network_norm_le_one,
        TNLean.PEPS.WholeGroup.rectangularTraceNorm_flattening_network_le_one,
        TNLean.PEPS.WholeGroup.exists_simultaneous_network_truncation,
        TNLean.PEPS.WholeGroup.network_one_eq_contraction}
    \leanok
    \uses{thm:peps_group_tensor}
    Consider a loopless tensor network in which each vertex carries one leg per
    incident edge endpoint and each edge carries the identity matrix. If every
    vertex tensor has norm at most $1$, then the conclusions of
    Theorem~\ref{thm:peps_group_tensor} hold for the network contraction.
\end{corollary}

\begin{proof}\leanok
    \uses{thm:peps_group_tensor}
    When every edge matrix is the identity and the graph has no self-loops,
    the network contraction coincides with the coordinate contraction, and the
    vertex norms agree, so the whole-group tensor bound applies directly.
\end{proof}
