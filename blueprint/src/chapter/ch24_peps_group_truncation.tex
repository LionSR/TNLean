\section{Whole-group tensor bound and simultaneous truncation}
\label{sec:peps_group_truncation}
This section presents the whole-group tensor bound of
\cite[Lemma~6.2]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
lines 125--179) and the two dimension-free matrix inequalities used in its
proof. All index sets are finite; for a matrix $M$, $\lVert M\rVert_1$ denotes
the sum of its singular values and $\lVert M\rVert_2$ its Hilbert--Schmidt norm.

\begin{definition}[Nuclear norm]
    \label{def:peps_group_nuclear_norm}
    \lean{Matrix.rectangularTraceNorm}
    \leanok
    For a finite rectangular complex matrix $M$, the nuclear norm
    $\lVert M\rVert_1$ is the sum of the singular values of $M$.
\end{definition}

\begin{lemma}[H\"older bound for a product with a transpose]
    \label{lem:peps_group_nuclear_holder}
    \lean{Matrix.rectangularTraceNorm_mul_transpose_le}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    For finite complex matrices $A,B$ with the same column index set,
    $\lVert AB^{\mathsf T}\rVert_1\le\lVert A\rVert_2\lVert B\rVert_2$.
\end{lemma}

\begin{proof}\leanok
    Let $(w_i)$ be an orthonormal basis of right singular vectors of
    $M=AB^{\mathsf T}$, and let $u_i=Mw_i/\lVert Mw_i\rVert$ when $Mw_i\ne0$.
    The $i$th singular value of $M$ satisfies
    $s_i(M)=\langle u_i,A(B^{\mathsf T}w_i)\rangle
        =\langle A^{*}u_i,B^{\mathsf T}w_i\rangle
        \le\lVert A^{*}u_i\rVert\,\lVert B^{\mathsf T}w_i\rVert$,
    and Cauchy--Schwarz over $i$ gives
    $\lVert M\rVert_1\le(\sum_i\lVert A^{*}u_i\rVert^2)^{1/2}(\sum_i\lVert B^{\mathsf T}w_i\rVert^2)^{1/2}
        \le\lVert A\rVert_2\lVert B\rVert_2$.
\end{proof}

\begin{lemma}[Truncation to leading singular directions]
    \label{lem:peps_group_orthonormal_truncation}
    \lean{Matrix.exists_orthonormal_truncation, Matrix.orthonormalProjector}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    For any finite complex matrix $M$ with $n$ columns and $k\in\N$, there
    exists an orthonormal family of at most $k$ vectors in $\C^n$ whose
    orthogonal projection $Q$ satisfies
    $(k+1)\lVert M-MQ\rVert_2^2\le\lVert M\rVert_1^2$, with $MQ=M$ when
    $n\le k$.
\end{lemma}

\begin{proof}\leanok
    Let $Q$ be the orthogonal projection onto the first $\min(k,n)$ right
    singular vectors of $M$, with singular values $s_1\ge s_2\ge\cdots\ge0$.
    Then $\lVert M-MQ\rVert_2^2=\sum_{j>k}s_j^2$, and since
    $(k+1)s_{k+1}\le\sum_j s_j$ and $\sum_{j>k}s_j^2\le s_{k+1}\sum_j s_j$,
    one has $(k+1)\sum_{j>k}s_j^2\le(\sum_j s_j)^2=\lVert M\rVert_1^2$.
\end{proof}

\begin{definition}[Coordinate contraction of a tensor graph]
    \label{def:peps_group_contraction}
    \lean{TNLean.Lattice.WholeGroup.contraction, TNLean.Lattice.WholeGroup.vertexNormSq}
    \leanok
    Let $(V,E)$ be a finite graph whose edges $e\in E$ have endpoints
    $t(e),h(e)\in V$ and finite index sets $D_e$, and let each $v\in V$ carry
    a tensor $A_v$ with one leg per incident edge and open-leg configuration
    set $P_v$, with Euclidean norm $\lVert A_v\rVert$. The coordinate
    contraction is
    \begin{align}
        Z(\sigma) & =\sum_{\beta}\prod_{v\in V}A_v\bigl(\beta|_{v},\sigma_v\bigr),
        \label{eq:peps_group_contraction}
    \end{align}
    where $\beta$ runs over all edge labellings and $\beta|_v$ restricts
    $\beta$ to edges incident to $v$.
\end{definition}

\begin{theorem}[Whole-group tensor bound, norm part]
    \label{thm:peps_group_contraction_norm}
    \lean{TNLean.Lattice.WholeGroup.contraction_norm_le_one,
        TNLean.Lattice.WholeGroup.partialNormSq_le_one}
    \leanok
    \uses{def:peps_group_contraction}
    If $(V,E)$ has no self-loops and $\lVert A_v\rVert\le1$ for all $v\in V$,
    then $\lVert Z\rVert\le1$, and the same bound holds for the partial
    contraction over any $T\subseteq V$ (keeping open the open legs of $T$ and
    edges leaving $T$).
\end{theorem}

\begin{proof}\leanok
    By induction on $|T|$: the empty subset has norm $1$, a single loopless
    vertex has norm $\lVert A_v\rVert\le1$, and contracting shared edges
    between disjoint vertex subsets multiplies their reshaped partial
    contractions, preserving the bound by
    $\lVert AB\rVert_2\le\lVert A\rVert_2\lVert B\rVert_2$.
\end{proof}

\begin{theorem}[Whole-group tensor bound, nuclear part]
    \label{thm:peps_group_contraction_nuclear}
    \lean{TNLean.Lattice.WholeGroup.rectangularTraceNorm_flattening_le_one,
        TNLean.Lattice.WholeGroup.flattening,
        TNLean.Lattice.WholeGroup.flattening_contraction}
    \leanok
    \uses{def:peps_group_contraction, def:peps_group_nuclear_norm}
    Under the hypotheses of Theorem~\ref{thm:peps_group_contraction_norm}, for
    every $T\subseteq V$, the matrix reshaping $Z_T$ of $Z$ with rows indexed
    by $\prod_{v\in T}P_v$ and columns by $\prod_{v\in V\setminus T}P_v$
    satisfies $\lVert Z_T\rVert_1\le1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_group_contraction_norm, lem:peps_group_nuclear_holder}
    By Theorem~\ref{thm:peps_group_contraction_norm}, the partial contractions
    $A$ on $T$ and $B$ on $V\setminus T$ satisfy $\lVert A\rVert_2,\lVert B\rVert_2\le1$,
    and grouping the edges between $T$ and $V\setminus T$ gives
    $Z_T=AB^{\mathsf T}$, so Lemma~\ref{lem:peps_group_nuclear_holder} yields
    $\lVert Z_T\rVert_1\le\lVert A\rVert_2\lVert B\rVert_2\le1$.
\end{proof}

\begin{lemma}[Simultaneous truncation of open-leg groups]
    \label{lem:peps_group_truncation_abstract}
    \lean{TNLean.Lattice.WholeGroup.exists_group_truncation,
        TNLean.Lattice.WholeGroup.norm_sub_factorMatrix_le,
        TNLean.Lattice.WholeGroup.factorMatrix,
        TNLean.Lattice.WholeGroup.productVector}
    \leanok
    \uses{def:peps_group_nuclear_norm}
    Let $Z\in\bigotimes_{v\in V}\C^{P_v}$ satisfy $\lVert Z\rVert\le1$ and
    $\lVert Z_{\{v\}}\rVert_1\le1$ for all $v\in V$, let $G\subseteq V$ contain
    every $v$ with $|P_v|>1$, and let $k\ge1$. Then there exist orthonormal
    families of at most $k$ vectors in each $\C^{P_v}$ whose tensor-product
    projection $Z_k=\bigl(\bigotimes_{v\in V}Q_v\bigr)Z$ satisfies
    $\lVert Z-Z_k\rVert\le|G|/\sqrt{k}$ and expands in the orthonormal product
    basis as at most $k^{|G|}$ product terms with coefficients of modulus at
    most $1$.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_group_orthonormal_truncation}
    For each $v\in V$, Lemma~\ref{lem:peps_group_orthonormal_truncation}
    provides an orthogonal projection $Q_v$ on $\C^{P_v}$ of rank at most $k$
    with $\lVert Z-Q_v Z\rVert\le1/\sqrt{k+1}\le1/\sqrt{k}$ and $Q_v Z=Z$ when
    $|P_v|\le1$. Since the $Q_v$ commute and have $\lVert Q_v\rVert\le1$,
    telescoping over $v\in G$ gives $\lVert Z-Z_k\rVert\le|G|/\sqrt k$, and
    $\lVert Z_k\rVert\le\lVert Z\rVert\le1$ bounds each product-basis
    coefficient by $1$.
\end{proof}

\begin{theorem}[Whole-group tensor bound]
    \label{thm:peps_group_tensor}
    \lean{TNLean.Lattice.WholeGroup.exists_simultaneous_openLeg_truncation,
        TNLean.Lattice.WholeGroup.exists_simultaneous_group_truncation}
    \leanok
    \uses{def:peps_group_contraction}
    Let a finite loopless graph carry at each $v\in V$ a tensor $A_v$ with
    $\lVert A_v\rVert\le1$, and let $Z$ be its coordinate contraction
    (\ref{eq:peps_group_contraction}). Then $\lVert Z\rVert\le1$,
    $\lVert Z_T\rVert_1\le1$ for every $T\subseteq V$, and if at most $g$
    vertices have nonempty open-leg groups, then for every integer $k\ge1$
    there exist subspaces of dimension at most $k$ in $\C^{P_v}$ whose
    tensor-product projection $Z_k$ satisfies
    $\lVert Z-Z_k\rVert\le g/\sqrt{k}$ and expands in orthonormal bases of
    these subspaces as at most $k^g$ product terms with coefficients of
    modulus at most $1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_group_contraction_norm, thm:peps_group_contraction_nuclear,
        lem:peps_group_truncation_abstract}
    Theorems~\ref{thm:peps_group_contraction_norm} and
    \ref{thm:peps_group_contraction_nuclear} give $\lVert Z\rVert\le1$ and
    $\lVert Z_T\rVert_1\le1$. Since $|P_v|=1$ when $v$ has no open legs, the
    set $G$ of vertices with nonempty open-leg groups contains every $v$ with
    $|P_v|>1$, and Lemma~\ref{lem:peps_group_truncation_abstract} gives $Z_k$.
\end{proof}

\begin{corollary}[Whole-group tensor bound for networks with edge matrices]
    \label{cor:peps_group_tensor_network}
    \lean{TNLean.Lattice.WholeGroup.network_norm_le_one,
        TNLean.Lattice.WholeGroup.rectangularTraceNorm_flattening_network_le_one,
        TNLean.Lattice.WholeGroup.exists_simultaneous_network_truncation,
        TNLean.Lattice.WholeGroup.network_one_eq_contraction}
    \leanok
    \uses{thm:peps_group_tensor}
    For a loopless tensor network with one leg per incident edge endpoint,
    identity edge matrices, and $\lVert A_v\rVert\le1$ for all $v\in V$, the
    conclusions of Theorem~\ref{thm:peps_group_tensor} hold for the network
    contraction.
\end{corollary}

\begin{proof}\leanok
    \uses{thm:peps_group_tensor}
    With identity edge matrices and no self-loops, the network contraction and
    vertex norms coincide with those of (\ref{eq:peps_group_contraction}), so
    Theorem~\ref{thm:peps_group_tensor} applies.
\end{proof}
