\section{The three-way splitting in Hastings' approximate ground-state projector}
\label{sec:hastings_three_way_splitting}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 763--790 (step one of the
% proof of Lemma 1); setting at lines 82--91, statement at lines 246--267.

Step one of the proof of \cite[Lemma~1]{Hastings2007AreaLaw} splits a range-$R$
Hamiltonian $H=\sum_Xh_X$ on the chain of $L$ sites across a cut $j$ with
buffer half-width $m\in\N$ ($l/3$ in the source) and centers each part in the
ground state $\Psi_0$.

\begin{definition}[Three-way splitting at a cut]
    \label{def:hastings_three_way_splitting}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.leftSupports,
        TNLean.MPS.AreaLaw.Hastings.bufferSupports,
        TNLean.MPS.AreaLaw.Hastings.rightSupports,
        TNLean.MPS.AreaLaw.Hastings.sum_supports_eq_left_add_buffer_add_right,
        TNLean.MPS.AreaLaw.ChainHamiltonian.termSum,
        TNLean.MPS.AreaLaw.ChainHamiltonian.termSum_isHermitian,
        TNLean.MPS.AreaLaw.ChainHamiltonian.leftPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.bufferPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.rightPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.leftPart_isHermitian,
        TNLean.MPS.AreaLaw.ChainHamiltonian.bufferPart_isHermitian,
        TNLean.MPS.AreaLaw.ChainHamiltonian.rightPart_isHermitian,
        TNLean.MPS.AreaLaw.Hastings.groundCentered,
        TNLean.MPS.AreaLaw.Hastings.groundCentered_add,
        Matrix.IsHermitian.groundCentered,
        Matrix.IsHermitian.ofReal_re_inner
    }
    \leanok
    \uses{def:chain_model}
    For $H=\sum_Xh_X$ of range $R$ on $\{0,\dots,L-1\}$, a cut $j$, and
    $m\in\N$, define the Hermitian parts and ground-state centering in $\Psi_0$
    \cite[lines 774--782]{Hastings2007AreaLaw}:
    \begin{align}
        H_L          & =\sum_{X\subseteq\lbrack0,j-m)}h_X,\qquad
        H_R=\sum_{X\subseteq\lbrack j+m,L)}h_X,\qquad
        H_B=\sum_{X\not\subseteq\lbrack0,j-m),\,X\not\subseteq\lbrack j+m,L)}h_X,
        \label{eq:hastings_three_way_parts}                                 \\
        \widetilde M & =M-\operatorname{Re}\langle\Psi_0,M\Psi_0\rangle\,1.
        \label{eq:hastings_ground_centered}
    \end{align}
\end{definition}

\begin{theorem}[The splitting is exact]
    \label{thm:hastings_three_way_splitting_sum}
    \lean{
        TNLean.MPS.AreaLaw.ChainHamiltonian.operator_eq_leftPart_add_bufferPart_add_rightPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.groundCentered_parts_sum,
        TNLean.MPS.AreaLaw.ChainHamiltonian.groundCentered_parts_sum_of_eigen,
        TNLean.MPS.AreaLaw.ChainHamiltonian.isGappedGroundState_groundCentered_parts_sum,
        TNLean.MPS.AreaLaw.Hastings.inner_groundCentered,
        TNLean.MPS.AreaLaw.Hastings.groundCentered_of_eigen
    }
    \leanok
    \uses{def:hastings_three_way_splitting}
    For every $(j,m)$, $H=H_L+H_B+H_R$. If $\|\Psi_0\|=1$ and
    $H\Psi_0=E_0\Psi_0$, then $\langle\Psi_0,\widetilde H_\alpha\Psi_0\rangle=0$
    for $\alpha\in\{L,B,R\}$ and
    \begin{align}
        \widetilde H_L+\widetilde H_B+\widetilde H_R=H-E_0.
        \label{eq:hastings_centered_sum}
    \end{align}
    If $H-E_0\ge\Delta(1-|\Psi_0\rangle\langle\Psi_0|)$, then $\Psi_0$ is a
    ground vector of $\widetilde H_L+\widetilde H_B+\widetilde H_R$ with ground
    energy $0$ and gap $\Delta$ \cite[lines 766--782]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Since $X\ne\emptyset$ and $\lbrack0,j-m)\cap\lbrack j+m,L)=\emptyset$, the three support
    classes partition the terms of $H$, giving $H=H_L+H_B+H_R$. Additivity of
    (\ref{eq:hastings_ground_centered}) and
    $\operatorname{Re}\langle\Psi_0,H\Psi_0\rangle=E_0$ give
    (\ref{eq:hastings_centered_sum}) and
    $\langle\Psi_0,\widetilde M\Psi_0\rangle=0$ for $M=M^\dagger$.
\end{proof}

\begin{theorem}[Supports of the three parts]
    \label{thm:hastings_three_way_splitting_supports}
    \lean{
        TNLean.MPS.AreaLaw.ChainHamiltonian.isSupportedOn_termSum,
        TNLean.MPS.AreaLaw.ChainHamiltonian.isSupportedOn_leftPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.isSupportedOn_rightPart,
        TNLean.MPS.AreaLaw.ChainHamiltonian.isSupportedOn_bufferPart,
        TNLean.MPS.AreaLaw.Hastings.subset_window_of_mem_bufferSupports,
        Entropy.IsSupportedOn.groundCentered
    }
    \leanok
    \uses{def:hastings_three_way_splitting}
    The operators $H_L,\widetilde H_L$ are supported on $\lbrack0,j-m)$,
    $H_R,\widetilde H_R$ on $\lbrack j+m,L)$, and $H_B,\widetilde H_B$ on
    $\lbrack j-m-R,j+m+R)\cap\lbrack0,L)$ \cite[lines 774--778]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Each buffer support $X$ in~(\ref{eq:hastings_three_way_parts}) meets both
    $\lbrack j-m,L)$ and $\lbrack0,j+m)$; since $\operatorname{diam}(X)\le R$,
    $X\subseteq\lbrack j-m-R,j+m+R)$. Subtracting a scalar multiple of $1$ in
    (\ref{eq:hastings_ground_centered}) preserves supports.
\end{proof}

\begin{theorem}[Norm of the buffer part]
    \label{thm:hastings_three_way_splitting_norm}
    \lean{
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_termSum_le,
        TNLean.MPS.AreaLaw.card_supports_containing_le,
        TNLean.MPS.AreaLaw.Hastings.card_bufferSupports_le,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_bufferPart_le,
        TNLean.MPS.AreaLaw.Hastings.norm_groundCentered_le,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_groundCentered_bufferPart_le
    }
    \leanok
    \uses{def:hastings_three_way_splitting,
        thm:hastings_three_way_splitting_supports}
    If $\|h_X\|\le J$ for all $X$, then for any unit vector $\Psi_0$,
    \begin{align}
        \|H_B\|\le2(m+R)\,2^{2R+1}J,\qquad
        \|\widetilde H_B\|\le4(m+R)\,2^{2R+1}J.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Each site $x$ lies in at most $2^{2R+1}$ supports $X\subseteq[x-R,x+R]$ of
    diameter at most $R$, and by Theorem~\ref{thm:hastings_three_way_splitting_supports}
    every buffer support meets $[j-m-R,j+m+R)$ (at most $2(m+R)$ sites), giving
    at most $2(m+R)2^{2R+1}$ buffer terms. The triangle inequality and
    $\|\widetilde M\|\le 2\|M\|$ conclude.
\end{proof}
