\section{Dimensions and locality of the original source network}
\label{sec:peps_source_network_dimensions}

Let $\mathcal W:a\to b$ be a chronological source circuit on a party set $P$.
Let $G$ be its gate occurrences with nonempty participant sets $C_g$, finite
branch sets $\Xi_g$, and roots $r_g\in C_g$. At each $g\in G$, the star joins
$r_g$ to $C_g\setminus\{r_g\}$ with virtual alphabet $\Xi_g\times\Xi_g$, and
each unordered pair in $C_g$ has its source sample link with alphabet
$[k]=\{0,\ldots,k-1\}$. For each tagged link $e$, let $g(e)\in G$ be its owner
occurrence, $D_e$ its virtual alphabet, and $\partial e$ its incident party set.

\begin{theorem}[Exact dimensions of the original density links]
    \label{thm:peps_source_density_link_card}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_alphabet_source_density}
    \leanok
    \uses{def:peps_source_gate_root,def:compression_distributed_links,
        def:compression_operator_alphabets}
    For any $k\in\N$ and any constructed link $e$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \begin{align}
        |D_e| & =
        \begin{cases}
            |\Xi_{g(e)}|^2, & e\text{ is a star link},   \\
            k,              & e\text{ is a sample link}.
        \end{cases}
        \label{eq:peps_source_density_link_card}
    \end{align}
    % Source: 04-compression.tex, lines 565--579.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:compression_operator_alphabet_card}
    Apply Theorem~\ref{thm:compression_operator_alphabet_card} with star
    alphabet $\Xi_{g(e)}\times\Xi_{g(e)}$ of cardinality $|\Xi_{g(e)}|^2$ and
    sample alphabet $[k]$ of cardinality $k$.
\end{proof}

\begin{theorem}[A conditional uniform bound for the original density links]
    \label{thm:peps_source_density_link_card_bound}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_alphabet_source_density_le}
    \leanok
    \uses{def:peps_source_gate_root,def:compression_distributed_links,
        def:compression_operator_alphabets}
    For $k,M\in\N$ in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, if
    $|\Xi_g|\le M$ for all $g\in G$, then every constructed link $e$ satisfies
    $|D_e|\le\max\{M^2,k\}$.
    % Source: 04-compression.tex, lines 565--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:compression_operator_alphabet_bound}
    Since $|\Xi_g\times\Xi_g|=|\Xi_g|^2\le M^2$,
    Theorem~\ref{thm:compression_operator_alphabet_bound} gives
    $|D_e|\le\max\{M^2,k\}$.
\end{proof}

\begin{theorem}[The endpoints are original participants of the owner gate]
    \label{thm:peps_source_density_link_locality}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.distributedLinkParties_source_subset}
    \leanok
    \uses{def:peps_source_gate_root,def:compression_distributed_links,
        def:compression_distributed_link_parties}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, every constructed
    link $e$ satisfies $\partial e\subseteq C_{g(e)}$.
    % Source: 04-compression.tex, lines 581--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_source_gate_root_mem,thm:compression_original_gate_locality}
    Since $r_g\in C_g$ for all $g\in G$ by Theorem~\ref{thm:peps_source_gate_root_mem},
    Theorem~\ref{thm:compression_original_gate_locality} yields
    $\partial e\subseteq C_{g(e)}$.
\end{proof}

\section{Incidence of the density links}
\label{sec:peps_source_density_link_incidence}

Let $\mathcal W:a\to c$ be a chronological source circuit on $P$ with nonempty
gate occurrences $G$, participant sets $C_g$, and roots $r_g\in C_g$.

\begin{theorem}[The retained occurrences preserve participation counts]
    \label{thm:peps_nonempty_gate_participation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_incidentNonemptyGates_eq_participationCount}
    \leanok
    \uses{def:compression_lifetime_gates}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, for every $p\in P$,
    $|\{g\in G:p\in C_g\}|$ equals the full participation count of $p$ in
    $\mathcal W$ (Definition~\ref{def:compression_lifetime_gates}).
    % Source: 04-compression.tex, lines 140--142 and 581--588.
\end{theorem}

\begin{proof}
    \leanok
    Every occurrence $g$ with $p\in C_g$ satisfies $C_g\neq\varnothing$, so
    filtering out $C_g=\varnothing$ preserves $\{g:p\in C_g\}$.
\end{proof}

\begin{theorem}[Constant incidence of the density links]
    \label{thm:peps_actual_density_link_incidence}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_incidentSourceDensityLinks_le}
    \leanok
    \uses{def:peps_source_gate_root,def:compression_distributed_links,
        thm:peps_nonempty_gate_participation}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, if $b\in\N$,
    $|C_g|\le b$ for all $g\in G$, and every party participates in at most $b$
    gate occurrences, then at most $2b(b-1)$ density-network links are incident
    to any party $p\in P$.
    % Source: 04-compression.tex, lines 581--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_source_gate_root_mem,thm:peps_nonempty_gate_participation,
        thm:compression_distributed_incidence_bound}
    By Theorems~\ref{thm:peps_source_gate_root_mem}
    and~\ref{thm:peps_nonempty_gate_participation}, $r_g\in C_g$ and
    $|\{g\in G:p\in C_g\}|\le b$, so
    Theorem~\ref{thm:compression_distributed_incidence_bound} gives the bound
    $2b(b-1)$.
\end{proof}

\begin{theorem}[Distributed physical approximation from local gate expansions]
    \label{thm:peps_approximate_sampled_distributed_error}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.exists_sampled_contractedOperator_approximate_error_le}
    \leanok
    \uses{def:peps_approximate_original_operators,
        thm:peps_approximate_source_sample,
        thm:peps_actual_sampled_distributed_contraction,
        def:peps_all_party_local_source_basis,def:peps_all_party_local_discard_basis}
    Let $P$ be a finite party set listed without repetition, with
    finite-dimensional initial memories $H_p$, unit vectors $x_p\in H_p$, input
    $\psi=\bigotimes_{p\in P}x_p$, contraction gates $G_g$, physical dimensions
    $d_p\le D$ ($D\ge 1$), and finite-dimensional discarded private registers.
    Suppose there are at most $M$ nonprivate gate occurrences, each party
    participates in at most $\ell$ of them, and for $0<\varepsilon\le 1$ and
    $\delta_a=\varepsilon/(16\max\{1,M\})$ each gate expansion satisfies
    Definition~\ref{def:peps_approximate_original_operators} at accuracy
    $\delta_a$ with at most $r$ pair effects per monomial and coefficient sum at
    most $B\ge 1$.

    Rescale each expansion by $(1+\delta_a)^{-1}$ to form $W$, and let $R$ be
    its physical-first source replacement with $N_R$ sources, stack length $m$,
    and sample size $k$ given by
    \begin{align}
        \delta_e & =\frac{\varepsilon}{8\max\{1,M\}},                                            &
        m        & =\left\lceil\left(\frac{rB}{\delta_e}\right)^2\right\rceil+1, \notag            \\
        Q        & =B^{4\ell}D^4,                                                                &
        k        & =\left\lceil\left(\frac{8\max\{1,N_R\}Q}{\varepsilon}\right)^2\right\rceil+1.
        \label{eq:peps_approximate_sampled_distributed_params}
    \end{align}
    In the local bases of Definitions~\ref{def:peps_all_party_local_source_basis}
    and~\ref{def:peps_all_party_local_discard_basis}, there exist normalized
    Schmidt data and a Gaussian sample $\omega$ such that the distributed
    contraction $\mathcal C_R(\omega)$ of
    Theorem~\ref{thm:peps_actual_sampled_distributed_contraction} with weights
    $c_g(\xi,\zeta)=k^{-\binom{|C_g|}{2}}a_g(\xi)\overline{a_g(\zeta)}$ and
    zero-party scalar factor $\gamma_R=\lambda_R\overline{\lambda_R}$ satisfies,
    as in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 17--19, 137--175, 199--229,
    342--381, 565--588 and 590--600),
    \begin{align}
        \bigl\|\gamma_R\mathcal C_R(\omega)-\rho_G\bigr\|_1
        \leq\varepsilon,
        \label{eq:peps_approximate_sampled_distributed_error}
    \end{align}
    where $\rho_G$ is the physical partial trace of the original circuit output
    on $\psi$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_approximate_source_sample,
        thm:peps_actual_sampled_distributed_contraction,
        thm:peps_finite_whole_party_input,thm:peps_auxiliary_register_finiteness,
        def:peps_all_party_local_source_basis,def:peps_all_party_local_discard_basis}
    By Theorems~\ref{thm:peps_finite_whole_party_input}
    and~\ref{thm:peps_auxiliary_register_finiteness}, the initial and discarded
    memories are finite-dimensional. Theorem~\ref{thm:peps_approximate_source_sample}
    provides normalized Schmidt data and a sample $\omega$ whose sampled
    physical operator $\widehat\rho$ satisfies
    $\|\widehat\rho-\rho_G\|_1\le\varepsilon$, and
    Theorem~\ref{thm:peps_actual_sampled_distributed_contraction} identifies
    $\widehat\rho=\gamma_R\mathcal C_R(\omega)$ in the bases of
    Definitions~\ref{def:peps_all_party_local_source_basis}
    and~\ref{def:peps_all_party_local_discard_basis}.
\end{proof}

\begin{theorem}[Real power bounds for the density alphabets]
    \label{thm:peps_actual_density_alphabet_power}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_alphabet_source_density_le_of_power_bounds}
    \leanok

    $u,v,k\in\N$, $|\Xi_g|\leq C_M L^u$ for all $g\in G$, and $k\leq C_k L^v$,
    then every link alphabet $D_e$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 565--588) satisfies
    $|D_e|\leq(C_M^2+C_k)L^{\max\{2u,v\}}$.

\begin{proof}
    \leanok
    \uses{thm:peps_source_density_link_card}

\begin{theorem}[Original polynomial bounds for the contraction alphabets]

    participants per gate, at most $K$ monomials per gate, at most $r$ pair

    \begin{align}
        C_s & =\max\{1,C_N\tbinom b2\},                  &

        C_k & =64C_s^2(C_B^{4\ell}C_D^4)^2+2,            &
