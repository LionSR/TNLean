\section{The closed fine-cell cover of a primary region}
\label{sec:area_law_primary_fine_cell_cover}

Each primary birth region in \cite[Section~11]{OpenAI2026AreaLaw} decomposes
into a finite union of closed non-belt fine cells.

% Source: 10-geometry.tex, prop:two-families, lines 212--218 and 299--323,
% revision adc7f1241b42e322a6451854ab7e4b4c146bf78a.
% This is a local decomposition, not the full two-family proposition.

\begin{theorem}[Primary region as a finite union of closed fine cells]
    \label{thm:area_law_primary_fine_cell_cover}
    \lean{TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_eq_iUnion_nonbeltCell_closure}
    \leanok
    \uses{def:area_law_primary_regions,def:area_law_belt_cells,
        def:area_law_nonbelt_pitch_index}
    Fix an origin $o$, a finite endpoint set, and a nonnegative integer
    neighborhood width. Let $k,\ell,p$ be natural exponents with
    $\ell\le k$ and $\ell\le p$, and set $m=2^{p-\ell}$. Choose
    residues $0\le a,b<m$. Denote the fine-layer index set by
    $F_{k,\ell}$ and its belt subset by $\mathcal B$.
    For each pitch index $J\in\mathbb Z^2$, the primary birth region
    satisfies
    \begin{align}
        \overline{D_k\cap U_J}
        =\bigcup_{\substack{z\in F_{k,\ell}\setminus\mathcal B \\j(z)=J}}
        \overline{Q_\ell(z)}.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_nonbelt_primary_containment,thm:area_law_fine_layer_count}
    Every closed non-belt fine cell $\overline{Q_\ell(z)}$ with
    $z\in F_{k,\ell}\setminus\mathcal B$ and $j(z)=J$ lies in
    $\overline{D_k\cap U_J}$. For the reverse inclusion,
    let $x\in D_k\cap U_J$ and let $Q_\ell(z)$ with $z\in F_{k,\ell}$ be the
    half-open fine cell containing $x$. In each coordinate, with residue shift
    $a$ and pitch index $J$, the strict open-square inequalities for $U_J$ and
    the half-open cell inequalities for $Q_\ell(z)$ imply $1\le z-a-mJ<m$.
    Thus $\lfloor(z-a)/m\rfloor=J$ and $z\not\equiv a\pmod m$, so
    $z\in F_{k,\ell}\setminus\mathcal B$ and $j(z)=J$. Since the union on the
    right-hand side is finite, it is closed and therefore contains
    $\overline{D_k\cap U_J}$.
\end{proof}
