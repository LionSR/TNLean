\subsection{Restrictions with excluded registers}

For party label sets $P,Q$, an ordered finite register list $\ell$ (each register consisting of an owner and a complex inner product space) has right-associated tensor memory $\mathcal H_\ell$ with terminal factor $\C$ ($\mathcal H_\varnothing=\C$; Definition~\ref{def:peps_party_registers}).
For a selection $s:Q\to\{0,1\}$ with $A=\{p\in Q:s(p)=1\}$ (Definition~\ref{def:peps_party_partition}) and an owner map $q:P\to Q$ (Definition~\ref{def:peps_owner_layout}), let $\ell|_A$ be the ordered sublist with owners in $A$ and $\ell^q$ the list obtained by relabeling owners via $q$.

\begin{lemma}[An owner map with no selected image]
    \label{thm:peps_excluded_mapped_owners}
    \lean{TNLean.Lattice.PartyCircuit.Layout.restrict_mapOwner_eq_nil}
    \leanok
    \uses{def:peps_party_registers,def:peps_owner_layout,def:peps_party_partition}
    If $q:P\to Q$ and $s:Q\to\{0,1\}$ satisfy $s(q(p))=0$ for all $p\in P$, then $(\ell^q)|_A=\varnothing$ for every register list $\ell$ with owners in $P$.
\end{lemma}

\begin{proof}\leanok
    Any register in $(\ell^q)|_A$ has an original owner $p\in P$ with $s(q(p))=1$, contradicting $s\circ q=0$.
\end{proof}

A source-free composition $W:\ell\to m$ is a finite composite of local linear operations, register exchanges, and identities without pair-source preparations; write $W|_A:\mathcal H_{\ell|_A}\to\mathcal H_{m|_A}$ for its restriction to $A$ (Lemma~\ref{thm:peps_party_restriction}), and for a spectator list $\rho$ write $\widetilde W:\rho\ell\to\rho m$ for $W$ framed by $\rho$ (with list concatenation denoted by juxtaposition).

\begin{lemma}[Excluded spectator registers preserve the restriction]
    \label{thm:peps_excluded_spectator_restriction}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_frameList_eval_heq_of_eq_nil}
    \leanok
    \uses{def:peps_party_registers,def:peps_party_partition,
        thm:peps_party_restriction}
    For any selection $s:Q\to\{0,1\}$, spectator list $\rho$ with $\rho|_A=\varnothing$, and source-free composition $W:\ell\to m$ with owners in $Q$, under $(\rho\ell)|_A=\ell|_A$ and $(\rho m)|_A=m|_A$, one has $\widetilde W|_A=W|_A$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_party_restriction}
    Proceed by induction on $\rho$: for $\rho=\varnothing$, $\widetilde W=W$; for $\rho=r\sigma$ with $\rho|_A=\varnothing$, the owner of $r$ lies outside $A$, so framing by $r$ acts as $\operatorname{id}_{H_r}\otimes W_\sigma$ and Lemma~\ref{thm:peps_party_restriction} gives
    \begin{align}
        (\operatorname{id}_{H_r}\otimes W_\sigma)|_A
         & =W_\sigma|_A = W|_A.
        \label{eq:peps_excluded_head}
    \end{align}
\end{proof}

These register-selection identities underlie \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 233--267 and 351--381.

\begin{lemma}[Successive restrictions to nested party sets]
    \label{thm:peps_nested_party_restrictions}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_restrict_heq_of_implies,
        TNLean.Lattice.PartyCircuit.Word.restrict_restrict_eval_heq_of_implies}
    \leanok
    \uses{thm:peps_party_restriction}
    For any source-free composition $W:\ell\to m$ and $B\subseteq A\subseteq P$, under the natural register-list identifications $(\tau|_A)|_B=\tau|_B$, both as compositions and as linear operators,
    \begin{align}
        (W|_A)|_B=W|_B.
        \label{eq:peps_nested_party_restrictions}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_party_restriction}
    Since $B\subseteq A$, $(\tau|_A)|_B=\tau|_B$ for every list $\tau$.
    By structural induction on $W$ (using Lemma~\ref{thm:peps_party_restriction}), identities and compositions preserve (\ref{eq:peps_nested_party_restrictions}), while for a local map $\Lambda_p(M;\tau)$, exchange $S_{r,s}(\tau)$ with owners $o(r),o(s)$, and framed word $F_r(w)$ satisfying $(w|_A)|_B=w|_B$:
    \begin{align}
        (\Lambda_p(M;\tau)|_A)|_B
         & =\begin{cases}
                \Lambda_p(M;\tau|_B),        & p\in B,    \\
                \operatorname{id}_{\tau|_B}, & p\notin B,
            \end{cases}
        =\Lambda_p(M;\tau)|_B, \notag                                   \\
        (S_{r,s}(\tau)|_A)|_B
         & =\begin{cases}
                S_{r,s}(\tau|_B),               & o(r),o(s)\in B,           \\
                \operatorname{id}_{r\,\tau|_B}, & o(r)\in B,\ o(s)\notin B, \\
                \operatorname{id}_{s\,\tau|_B}, & o(r)\notin B,\ o(s)\in B, \\
                \operatorname{id}_{\tau|_B},    & o(r),o(s)\notin B,
            \end{cases}
        =S_{r,s}(\tau)|_B, \notag                                       \\
        (F_r(w)|_A)|_B
         & =\begin{cases}
                F_r((w|_A)|_B), & o(r)\in B,    \\
                (w|_A)|_B,      & o(r)\notin B,
            \end{cases}
        =\begin{cases}
             F_r(w|_B), & o(r)\in B,    \\
             w|_B,      & o(r)\notin B,
         \end{cases}
        =F_r(w)|_B. \notag
    \end{align}
\end{proof}

\begin{lemma}[A singleton restriction after excluding another party]
    \label{thm:peps_singleton_after_excluded_party}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_withoutParty_atParty_eval_heq}
    \leanok
    \uses{thm:peps_nested_party_restrictions,def:peps_single_party_selector}
    For distinct $p\neq q$ in $P$ and any source-free composition $W$, under the natural register identifications,
    \begin{align}
        (W|_{P\setminus\{p\}})|_{\{q\}}=W|_{\{q\}}.
        \label{eq:peps_singleton_after_excluded_party}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_nested_party_restrictions,def:peps_single_party_selector}
    Since $q\ne p$ implies $\{q\}\subseteq P\setminus\{p\}$, (\ref{eq:peps_nested_party_restrictions}) of Lemma~\ref{thm:peps_nested_party_restrictions} yields (\ref{eq:peps_singleton_after_excluded_party}).
\end{proof}

\begin{lemma}[Scalar restrictions under composition and framing]
    \label{thm:peps_scalar_residual_composition}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_localPart_comp_scalar}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_localPart_frame_scalar}
    \leanok
    \uses{def:peps_canonical_source_residual,thm:peps_party_restriction}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS} (\texttt{04-compression.tex}, lines 246--251 and 351--381), let $W:\ell\to m$ and $V:m\to n$ be chronological compositions with source-free residuals $W^{\mathrm r},V^{\mathrm r}$ (Definition~\ref{def:peps_canonical_source_residual}).
    If a selection $A$ satisfies $\ell|_A=m|_A=n|_A=\varnothing$, excludes all source registers of $W$ and $V$, and satisfies $W^{\mathrm r}|_A=\alpha\operatorname{id}_{\mathbb C}$ and $V^{\mathrm r}|_A=\beta\operatorname{id}_{\mathbb C}$, then $(VW)^{\mathrm r}|_A=\alpha\beta\operatorname{id}_{\mathbb C}$, and framing $W$ by an untouched register $r$ with $o(r)\notin A$ yields $\widetilde W^{\mathrm r}|_A=\alpha\operatorname{id}_{\mathbb C}$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_canonical_source_residual,
        thm:peps_excluded_spectator_restriction,thm:peps_party_restriction}
    In $(VW)^{\mathrm r}$, the excluded source registers of $V$ act as spectators alongside $W^{\mathrm r}$, so Lemma~\ref{thm:peps_excluded_spectator_restriction} and Lemma~\ref{thm:peps_party_restriction} give
    \begin{align}
        (VW)^{\mathrm r}|_A
         & =V^{\mathrm r}|_A\circ W^{\mathrm r}|_A
        =\beta\operatorname{id}_{\mathbb C}\circ\alpha\operatorname{id}_{\mathbb C}
        =\alpha\beta\operatorname{id}_{\mathbb C}. \notag
    \end{align}
    For $\widetilde W$, moving $r$ past the excluded sources via an exchange $E$ gives $\widetilde W^{\mathrm r}=(\operatorname{id}_{H_r}\otimes W^{\mathrm r})\circ E$ with $E|_A=\operatorname{id}_{\mathbb C}$, so (\ref{eq:peps_excluded_head}) yields $\widetilde W^{\mathrm r}|_A=W^{\mathrm r}|_A\circ E|_A=\alpha\operatorname{id}_{\mathbb C}$.
\end{proof}
