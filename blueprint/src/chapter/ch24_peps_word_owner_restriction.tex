\section{Party restrictions after changing owner labels}

Let $P$ and $Q$ be sets of owner labels. A finite composition $W$ consists of
local operations, register exchanges, and operations acting alongside spectator
registers. Its participating parties are the owners of all registers that occur
together with the assigned party of each local operation (including scalar
operations on an empty register list). For a map $f:P\to Q$, write $W^f$ for
the composition obtained by relabeling each owner $p$ to $f(p)$ while
preserving the register spaces and order of operations.

\begin{definition}[Selection of one party]
    \label{def:peps_single_party_selector}
    \lean{TNLean.PEPS.PairEffect.partySelector}
    \leanok
    For $q\in Q$, define $\chi_q:Q\to\{\mathrm{false},\mathrm{true}\}$
    by $\chi_q(x)=\mathrm{true}$ if and only if $x=q$.
\end{definition}

\begin{theorem}[Participating parties after changing owner labels]
    \label{thm:peps_source_free_owner_parties}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties_mapOwner_of_sources_nil}
    \leanok
    \uses{def:peps_owner_composition,def:peps_party_registers}
    If $W$ contains no pair-source preparation, then the set of participating
    parties of $W^f$ is the image under $f$ of the set of participating parties
    of $W$, without assuming that $f$ is injective.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_owner_composition,def:peps_party_registers}
    By induction on $W$. The owners of a relabeled register list
    are the images of its original owners. Under composition, if $W_1$ and
    $W_2$ have participating party sets $S_1$ and $S_2$, then $(W_1;W_2)^f$ has
    participating party set $f(S_1)\cup f(S_2)=f(S_1\cup S_2)$.
    Local maps, exchanges, and operations alongside spectator registers use
    $f(\{p\}\cup S)=\{f(p)\}\cup f(S)$, including scalar local operations on
    empty register lists; pair-source preparations are excluded by hypothesis.
\end{proof}

\begin{theorem}[Restriction to an owner outside the relabelling range]
    \label{thm:peps_owner_disjoint_restriction}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_mapOwner_eval_eq_id_of_disjoint}
    \leanok
    \uses{def:peps_single_party_selector,def:peps_owner_composition,
        def:peps_party_registers,thm:peps_owner_composition,
        thm:peps_party_restriction}
    Suppose that $W$ contains no pair-source preparation and that
    $f(p)\ne q$ for every $p\in P$. Let $V$ be the restriction of $W^f$
    to the owner $q$. Then $V$ has no participating parties, and its operator is
    $\operatorname{id}_{\C}$ under the identification of the
    empty input and output register memories with $\C$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_free_owner_parties,thm:peps_party_restriction,
        def:peps_single_party_selector,thm:peps_party_factorization,
        thm:peps_owner_composition}
    Since $W^f$ has the same pair-source list as $W$ up to owner relabeling,
    $W^f$ contains no pair-source preparation and its restriction $V$ is
    well-defined. If $S$ is the set of participating parties of $W$, then that
    of $V$ is $\{x\in f(S):x=q\}=\varnothing$. A composition with no
    participating parties has empty input and output register lists, contains no
    local operations, and acts as the identity $\operatorname{id}_{\C}$.
\end{proof}

These lemmas are used to separate local operations by party in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
When all original parties are embedded into $P\sqcup\{\bot\}$ away from the
exterior label $\bot$, the restriction of the all-party private preparation to
$\bot$ is the identity $\operatorname{id}_{\C}$ whenever no adjoined source
register is owned by $\bot$.
