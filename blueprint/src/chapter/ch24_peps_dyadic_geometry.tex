\section{Geometry of the dyadic distribution protocol}
\label{sec:peps_dyadic_geometry}
This section treats the geometric part of~\cite[Section~7]{OpenAI2026PolynomialPEPS},
September 24, 2026, \texttt{06-geometry.tex}: the bands along the edges of a
repainted dyadic square, the angular separation of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}
(lines 254--319), the point-treatment estimates (lines 357--415), the absence of
changed true vertices in the bulk operations and the tags (lines 375--398 and
610--631), and parts of the geometry of the small-patch covers, of the direct
repaintings and of the hole resizings (lines 341--351 and 447--533), and the counts of the padded
dyadic hierarchy behind items 1--3
of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} (lines 12--53 and 633--652).
The protocol of Proposition~7.1 itself, whose changes are implemented by the
information estimate and the contraction gates of the earlier sections, is not
treated here.

The clearance estimates are first proved for every guide with the properties that
the proof of Lemma~7.2 uses: the normal word of the operation on the band
$\{-8<x<2\}$ and no change off it, the label $A$ on the open square, and the label
$C$ on the exterior band $\{-8<x<0\}$. The schedule of guides is then constructed,
these properties are derived from the disjointness of the bands, and the estimates
are moved to every square of the hierarchy. This gives
Lemma~\ref{thm:peps_dyadic_angular_separation}, which is Lemma~7.2 for the
labellings that the schedule specifies; Lemma~7.2 for the guides sampled under the
source's tie convention is Lemma~\ref{thm:peps_dyadic_angular_separation_sampled}. A guide of the
source is a partition of the plane into open chambers, with labels on boundaries
sampled after one small generic displacement; here a guide is a labelling of the
whole plane, and two labellings that agree off finitely many interface curves
represent the same guide. The source's sampled guides are identified with readings
of these labellings in Lemma~\ref{lem:peps_dyadic_chamber_sampling}. The earlier scope
restriction and this convention are recorded
in~\cite{gap:openai26_dyadic_geometry_guide_properties}. Distances are sup
distances in the plane, and labels are elements of an arbitrary set; the labels
$A$, $B$, $C$ below are permitted to coincide.

\begin{definition}[Edge coordinates, bands and the lens]
    \label{def:peps_dyadic_edge_band}
    \lean{TNLean.PEPS.Approximation.SquareEdge, TNLean.PEPS.Approximation.bandWidth, TNLean.PEPS.Approximation.edgeBand, TNLean.PEPS.Approximation.closedEdgeBand, TNLean.PEPS.Approximation.bandCoord, TNLean.PEPS.Approximation.edgeLens, TNLean.PEPS.Approximation.edgeNeighbor, TNLean.PEPS.Approximation.edgeMarkDist, TNLean.PEPS.Approximation.cornerMarkDist}
    \leanok
    Let $S=[0,n]^2$. For an edge of $S$ let $s$ be the coordinate parallel to the
    edge, measured from one endpoint, and $d$ the normal coordinate, positive
    into $S$. Put
    \begin{align}
        w(s)=10^{-3}\min(s,n-s),\quad x=d/w(s).
        \label{eq:pdg_band}
    \end{align}
    The band $\{\alpha<x<\beta\}$ is the set of points with $0<s<n$ and
    $\alpha w(s)<d<\beta w(s)$; its closed version has $0\le s\le n$ and
    $\alpha w(s)\le d\le\beta w(s)$. The lens of the edge is the band
    $Y=\{-7<x<-7/2\}$, and the exterior neighboring square is
    $\{0<s<n,\ -n<d<0\}$. For a point $y$, $d_{\mathcal V}(y)$ is its distance from
    the two endpoints of the edge, or from the four corners of $S$.
\end{definition}

\begin{lemma}[Disjointness of the edge bands]
    \label{lem:peps_dyadic_band_disjoint}
    \lean{TNLean.PEPS.Approximation.disjoint_edgeBand, TNLean.PEPS.Approximation.edgeBand_neg_subset_edgeNeighbor, TNLean.PEPS.Approximation.dist_le_of_mem_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.changed_subset_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.dist_le_of_mem_closure_changed, TNLean.PEPS.Approximation.dist_le_of_mem_closure_centralRegion}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    The bands $\{-8<x<2\}$ along distinct edges of $S$ are disjoint, and
    $\{-8<x<0\}\subset\{0<s<n,\ -n<d<0\}$. Every closed band
    $\{\alpha\le x\le\beta\}$ with $-8\le\alpha\le\beta\le8$, the closure of the
    changed set of each elementary edge birth or death, and the closure of the
    central region of Lemma~\ref{thm:peps_dyadic_central_clearance} have
    diameter at most $n$.
\end{lemma}

\begin{proof}\leanok
    By~(\ref{eq:pdg_band}), $w(s)\le\min(s,n-s)/1000\le n/2000$, so bands along
    opposite edges lie within $n/250$ of their edges, while a common point of
    adjacent bands at distances $r_1,r_2>0$ from the two edge lines would satisfy
    $r_1<0.002\,r_2$ and $r_2<0.002\,r_1$. On $\{-8<x<0\}$ one has
    $0>d>-8w(s)\ge-n/250>-n$. In $\{\alpha\le x\le\beta\}$ one has
    $|d|\le8w(s)\le n/250$, giving $\operatorname{diam}\le n$; the changed set of
    an elementary operation lies in such a closed band, and the central region
    lies in $S=[0,n]^2$.
\end{proof}

\begin{definition}[Normal words of the edge construction]
    \label{def:peps_dyadic_normal_words}
    \lean{TNLean.PEPS.Approximation.bandWord, TNLean.PEPS.Approximation.edgeStartWord, TNLean.PEPS.Approximation.auxWord, TNLean.PEPS.Approximation.mainWordOne, TNLean.PEPS.Approximation.edgeOperations, TNLean.PEPS.Approximation.BandOperation}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    A normal word lists labels in increasing $x$ separated by interfaces, taking
    the right-hand label at each interface curve (differing from the source's
    generic-displacement sampling only on the interface curves). An edge operation
    starts from the main word $C\mid A\mid B$ at $0,1$. On an auxiliary sheet
    uniform at $C$, the births of $B$ on $(-6,-1)$, $A$ on $(-5,-2)$, and $C$ on
    $(-4,-3)$ give
    \begin{align}
        C\mid B\mid A\mid C\mid A\mid B\mid C\quad\text{at } -6,-5,-4,-3,-2,-1.
        \label{eq:pdg_aux_word}
    \end{align}
    Exchanging on the lens $Y=\{-7<x<-7/2\}$ gives the main word
    $C\mid B\mid A\mid C\mid A\mid B$ at $-6,-5,-4,0,1$. Then $C$ on $(-4,0)$
    dies into $A$, $A$ on $(-5,1)$ dies into $B$, $C$ is born on $(-3,0)$, and
    $B$ on $(-6,-3)$ dies into $C$, leaving $C\mid B$ at $0$. Each of these seven
    births and deaths changes its word only on a closed interval
    $[\alpha,\beta]$, and its surrounding word (before a birth, after a death) is
    constant on $(\alpha',\beta')\subset(-8,2)$ with
    $\alpha'\le\alpha-1/2$ and $\beta+1/2\le\beta'$. The seven normal words are
    drawn below at fixed $0<s<n$ over $-8\le x\le2$: row~(1) is the starting main
    word, row~(2) the auxiliary word~(\ref{eq:pdg_aux_word}) with the lens $Y$,
    row~(3) the main word after the exchange, and rows~(4)--(7) the main words
    after the four steps on $(-4,0)$, $(-5,1)$, $(-3,0)$, and $(-6,-3)$.
    \input{figures/peps_dyadic_normal_words}
\end{definition}

\begin{lemma}[The lens exchange produces the main word]
    \label{lem:peps_dyadic_lens_word}
    \lean{TNLean.PEPS.Approximation.lens_exchange_word}
    \leanok
    \uses{def:peps_dyadic_normal_words}
    Replacing $C\mid A\mid B$ at $0,1$ by~(\ref{eq:pdg_aux_word}) on
    $Y=\{-7<x<-7/2\}$ yields $C\mid B\mid A\mid C\mid A\mid B$ at
    $-6,-5,-4,0,1$ for every $x$.
\end{lemma}

\begin{proof}\leanok
    Both $-7$ and $-7/2$ lie in $C$-intervals of both words, so the words agree on
    each interval.
\end{proof}

\begin{lemma}[Conical band estimate]
    \label{lem:peps_dyadic_conical_band}
    \lean{TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt, TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt_markDist, TNLean.PEPS.Approximation.edgeMarkDist_le_of_mem_closedEdgeBand}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $a_0=1/5000$, $\alpha\le\beta$, $\alpha'+1/2\le\alpha$, $\beta+1/2\le\beta'$
    and $-8\le\alpha'$, $\beta'\le8$. If $y\in\{\alpha\le x\le\beta\}$ and
    $|y-z|<a_0\min(n,d_{\mathcal V}(y))$, then $z\in\{\alpha'<x<\beta'\}$.
\end{lemma}

\begin{proof}\leanok
    At $y$, with $m=\min(s,n-s)$,~(\ref{eq:pdg_band}) gives $w(s_y)=m/1000$ and
    $|d_y|\le8w(s_y)\le m$, so $d_{\mathcal V}(y)\le m$. For $|y-z|<m/5000$, both
    $s$ and $d$ change by less than $m/5000$, and $10^{-3}$-Lipschitz continuity
    of $w$ gives $|\beta'(w(s_z)-w(s_y))|\le8m/(5\cdot10^{6})$. Thus
    $d_z<\beta w(s_y)+m/5000\le\beta' w(s_z)-m/2000+m/5000+8m/(5\cdot10^6)<\beta'w(s_z)$,
    and symmetrically $d_z>\alpha'w(s_z)$.
\end{proof}

\begin{lemma}[Angular separation of the central birth]
    \label{thm:peps_dyadic_central_clearance}
    \lean{TNLean.PEPS.Approximation.centralRegion, TNLean.PEPS.Approximation.centralBirth_clearance}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $R\subset(0,n)^2$ be the central region where $x>1$ for all four edges, and let
    $f$ be a labelling equal to $A$ on $(0,n)^2$. For every $y\in\overline R$ and
    every $z$ with $f(z)\ne A$,
    \begin{align}
        |y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\},
        \label{eq:pdg_central}
    \end{align}
    where $\mathcal V$ is the set of corners of $S$ and $a_0=1/5000$
    (\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}).
    The guide after the central birth is sketched below, with the bands
    $\{0<x<1\}$ and their corner taperings magnified and $\mathcal V$ marked by dots.
    \input{figures/peps_dyadic_central_clearance}
\end{lemma}

\begin{proof}\leanok
    With $u=\min(y_1,n-y_1)$ and $v=\min(y_2,n-y_2)$, the four bounds $d\ge w(s)$
    on $\overline R$ give $v\ge u/1000$ and $u\ge v/1000$, so
    $d_{\mathcal V}(y)=\max(u,v)\le1000\min(u,v)$. Hence
    $|y-z|<a_0\min(n,d_{\mathcal V}(y))\le\min(u,v)/5$ implies $z\in(0,n)^2$,
    where $f(z)=A$.
\end{proof}

\begin{lemma}[Angular separation of the edge births and deaths]
    \label{thm:peps_dyadic_band_clearance}
    \lean{TNLean.PEPS.Approximation.BandOperation.clearance, TNLean.PEPS.Approximation.auxBirthB, TNLean.PEPS.Approximation.auxBirthA, TNLean.PEPS.Approximation.auxBirthC, TNLean.PEPS.Approximation.mainDeathC, TNLean.PEPS.Approximation.mainDeathA, TNLean.PEPS.Approximation.mainBirthC, TNLean.PEPS.Approximation.mainDeathB}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    For any of the seven elementary births and deaths of
    Definition~\ref{def:peps_dyadic_normal_words}, let $f_b$, $f_a$, $f_s$ carry
    its before, after, and surrounding normal words on $\{-8<x<2\}$ with
    $f_b=f_a$ off $\{-8<x<2\}$, and let $P_\circ$ be its surrounding label. For
    every $y\in\overline{\{f_b\ne f_a\}}$ and every $z$ with $f_s(z)\ne P_\circ$,
    $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$, where $\mathcal V$ is the pair of
    endpoints of the edge (\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    Since $\{f_b\ne f_a\}\subset\{\alpha\le x\le\beta\}$ is contained in a closed
    band, if $|y-z|<a_0\min\{n,d_{\mathcal V}(y)\}$ then
    Lemma~\ref{lem:peps_dyadic_conical_band} places $z$ in the surrounding open
    band $\{\alpha'<x<\beta'\}$, where $f_s(z)=P_\circ$.
\end{proof}

\begin{lemma}[Angular separation of the lens exchange]
    \label{thm:peps_dyadic_lens_clearance}
    \lean{TNLean.PEPS.Approximation.lensExchange_clearance}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    Let $f_m=C$ on $\{-8<x<0\}$, and let $f_{\rm aux}$ carry the word
    (\ref{eq:pdg_aux_word}) on $\{-8<x<2\}$ and equal $C$ off it. Every
    $y\in\overline{\{p:\neg(f_m(p)=f_{\rm aux}(p)=C)\}}$ and $z\in\partial Y$
    satisfy $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$
    (\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    A point $z\in\partial Y$ is either in $\mathcal V$ (where the bound is
    immediate) or satisfies $d_z\in\{-7w(s_z),-\tfrac72w(s_z)\}$ with $0<s_z<n$.
    If $y\notin\{-8<x<0\}$, then $d_{\mathcal V}(y)\le d_{\mathcal V}(z)+|y-z|$
    and $d_{\mathcal V}(z)\le m_z=\min(s_z,n-s_z)$, so
    $|y-z|<a_0d_{\mathcal V}(y)$ implies $|y-z|<m_z/2000$, which forces
    $y\in\{-8<x<0\}$ since $-7m_z/1000\le d_z\le-\tfrac72m_z/1000$ and $w$ is
    $10^{-3}$-Lipschitz. If instead
    $y\in\{-6\le x\le-4\}\cup\{-3\le x\le-1\}$, then
    Lemma~\ref{lem:peps_dyadic_conical_band} places $z$ in
    $\{-\tfrac{13}2<x<-\tfrac72\}\cup\{-\tfrac72<x<-\tfrac12\}$, disjoint from
    $\partial Y$.
\end{proof}

\begin{definition}[The schedule of guides of a repainting]
    \label{def:peps_dyadic_schedule}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline, TNLean.PEPS.Approximation.openSquare, TNLean.PEPS.Approximation.closedSquare, TNLean.PEPS.Approximation.bandUpdate, TNLean.PEPS.Approximation.RepaintingBaseline.centralGuide, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide, TNLean.PEPS.Approximation.RepaintingBaseline.edgeInterface, TNLean.PEPS.Approximation.auxSteps, TNLean.PEPS.Approximation.mainSteps}
    \leanok
    \uses{def:peps_dyadic_normal_words, thm:peps_dyadic_central_clearance}
    A repainting of $S=[0,n]^2$ ($n>0$) starts from a guide $g$ equal to $A$ on
    $(0,n)^2$ and to $C_e$ on the exterior neighbor $\{0<s<n,\ -n<d<0\}$ across
    each edge $e$. The central guide equals $B$ on the central region $R$ of
    Lemma~\ref{thm:peps_dyadic_central_clearance} and $g$ off $R$. For a list
    $E$ of completed edges, the completed guide $g_E$ replaces the band
    $\{-8<x<2\}$ of each $e\in E$ in the central guide by the word $C_e\mid B$ at
    $0$. Along the next edge $e\notin E$, the main guide $g_{E,e}(W)$ replaces the
    band of $e$ in $g_E$ by the normal word $W$, and the auxiliary guide
    $g_{{\rm aux},e}(W)$ replaces the band of $e$ in the constant guide $C_e$ by
    $W$, stepping through the three auxiliary births and four main steps of
    Definition~\ref{def:peps_dyadic_normal_words}. At the start of edge $e$, the
    schedule passes from $g_E$ to $g_{E,e}(C_e\mid A\mid B)$, which agrees with
    $g_E$ off the curves $x=0$ and $x=1$ of $e$.
\end{definition}

\begin{lemma}[The guides of the schedule]
    \label{lem:peps_dyadic_schedule_guides}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_edgeBand, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_edgeStartWord, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_mainGuide_start, TNLean.PEPS.Approximation.RepaintingBaseline.lensExchange_mainGuide, TNLean.PEPS.Approximation.auxSteps_chain, TNLean.PEPS.Approximation.mainSteps_chain, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide_mainWordFive, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide_uniform, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_notMem_closedSquare, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_openSquare, TNLean.PEPS.Approximation.exists_edgeBand_of_notMem_centralRegion, TNLean.PEPS.Approximation.mem_centralRegion_of_forall_notMem}
    \leanok
    \uses{def:peps_dyadic_schedule, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_lens_word}
    For $e\notin E$:
    \begin{enumerate}
        \item On the band $\{-8<x<2\}$ of $e$ off $x\in\{0,1\}$, $g_E$ carries
              $C_e\mid A\mid B$ at $0,1$, so $g_E=g_{E,e}(C_e\mid A\mid B)$ off
              those curves.
        \item Exchanging $g_{E,e}(C_e\mid A\mid B)$ with the auxiliary guide
              carrying~(\ref{eq:pdg_aux_word}) on $Y$ gives
              $g_{E,e}(C_e\mid B\mid A\mid C_e\mid A\mid B)$ at
              $-6,-5,-4,0,1$.
        \item The auxiliary births start from the constant guide $C_e$, the main
              steps start from the exchanged main guide, consecutive steps match,
              and the final main death leaves $g_{E\cup\{e\}}$.
        \item When $E$ contains all four edges, $g_E=B$ on $(0,n)^2$ and $g_E=g$
              on $\mathbb R^2\setminus[0,n]^2$.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_dyadic_band_disjoint}, the band of $e\notin E$ is
    disjoint from the bands of $E$; its regions $x<0$, $0<x<1$, and $1<x<2$ lie
    respectively in the exterior neighbor, in $(0,n)^2\setminus R$, and in $R$,
    giving item~1. Item~2 follows from Lemma~\ref{lem:peps_dyadic_lens_word} on
    the band and identity off it, and item~3 follows by inspection of the words in
    Definition~\ref{def:peps_dyadic_normal_words}. For item~4, every point of
    $(0,n)^2\setminus R$ lies in $\{0<x\le1\}$ for some edge (where the final word
    is $B$), and every point of $\mathbb R^2\setminus[0,n]^2$ lies either outside
    all bands or in $\{-8<x<0\}$ for some edge $e$ (where the final word is
    $C_e=g$).
\end{proof}

\begin{lemma}[Translation of the estimates]
    \label{lem:peps_dyadic_translation}
    \lean{TNLean.PEPS.Approximation.clearance_shift, TNLean.PEPS.Approximation.dist_le_shift, TNLean.PEPS.Approximation.le_dist_of_mem_closure, TNLean.PEPS.Approximation.cornerMarkDist_sub, TNLean.PEPS.Approximation.edgeMarkDist_sub, TNLean.PEPS.Approximation.sub_mem_closure, TNLean.PEPS.Approximation.sub_mem_frontier, TNLean.PEPS.Approximation.mem_closure_shift}
    \leanok
    \uses{def:peps_dyadic_block_guide}
    Translating guides by $o\in\mathbb R^2$ translates their changed sets,
    obstructing sets, and marks by $o$ while preserving sup distances; any lower
    bound $|y-z|\ge c$ for $z\in E$ extends to $z\in\overline E$.
\end{lemma}

\begin{proof}\leanok
    Translation $p\mapsto p+o$ is an isometry, hence a homeomorphism commuting
    with closures and boundaries, and $\{z:|y-z|\ge c\}$ is closed.
\end{proof}

\begin{lemma}[Separation of the unmodified bands]
    \label{thm:peps_dyadic_angular_separation}
    \lean{TNLean.PEPS.Approximation.blockRepainting_central, TNLean.PEPS.Approximation.blockRepainting_auxStep, TNLean.PEPS.Approximation.blockRepainting_mainStep, TNLean.PEPS.Approximation.blockRepainting_exchange, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_central, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_central, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_auxStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_mainStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_exchange, TNLean.PEPS.Approximation.BandOperation.clearance_bandUpdate, TNLean.PEPS.Approximation.BandOperation.dist_le_bandUpdate, TNLean.PEPS.Approximation.dist_le_of_mem_closure_edgeLens, TNLean.PEPS.Approximation.RepaintingBaseline.exchange_changed_subset, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_exchange}
    \leanok
    \uses{thm:peps_dyadic_central_clearance, thm:peps_dyadic_band_clearance, thm:peps_dyadic_lens_clearance, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_schedule_guides, lem:peps_dyadic_translation}
    Let $a_0=1/5000$ and $n>0$, and consider the repainting of an $n$-block $S$
    from a block guide via Definition~\ref{def:peps_dyadic_schedule}.
    \begin{enumerate}
        \item Every elementary birth or death (the central birth and the seven
              steps along each edge) has changed region $R$ with
              $\operatorname{diam}(\overline R)\le n$, and its surrounding guide
              $f$ satisfies, for all $y\in\overline R$,
              \begin{align}
                  \operatorname{dist}\bigl(y,\overline{\{f\ne P_\circ\}}\bigr)
                  \ge a_0\min\{n,d_{\mathcal V}(y)\},
                  \label{eq:pdg_angular_separation}
              \end{align}
              where $\mathcal V$ is the four corners of $S$ for the central birth
              and the two endpoints of the edge otherwise.
        \item For each lens exchange on $Y$, the set $N$ of points not equal to
              $C_e$ on both sheets before the exchange satisfies
              $\operatorname{dist}(y,\partial Y)\ge a_0\min\{n,d_{\mathcal V}(y)\}$
              for all $y\in\overline N$, and the exchange changes only points in
              $N\cap Y$, with $\operatorname{diam}(\overline{N\cap Y})\le n$.
    \end{enumerate}
    This proves~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS} (and the diameter bound
    of lines 421--424) for the schedule labellings; the sampled version is
    Lemma~\ref{thm:peps_dyadic_angular_separation_sampled}.
\end{lemma}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_dyadic_schedule_guides}, the schedule guides satisfy the
    hypotheses of Lemmas~\ref{thm:peps_dyadic_central_clearance},
    \ref{thm:peps_dyadic_band_clearance}, and~\ref{thm:peps_dyadic_lens_clearance}.
    Since $\overline Y\subset\{-7\le x\le-7/2\}$,

\begin{lemma}[The compactness argument]
    \label{lem:peps_dyadic_footprint}
    \lean{TNLean.PEPS.Approximation.exists_pos_le_max_dist_of_inter_eq_empty, TNLean.PEPS.Approximation.exists_footprint_two_owners, TNLean.PEPS.Approximation.exists_footprint_two_owners_param, TNLean.PEPS.Approximation.exists_footprint_two_owners_guide}
    \leanok
    \uses{def:peps_dyadic_incident_labels}
    If finitely many compact sets $F_c$ in a metric space satisfy
    $F_{c_1}\cap F_{c_2}\cap F_{c_3}=\emptyset$ for distinct $c_1,c_2,c_3$, then
    there exists $\delta>0$ such that no closed ball of radius $\delta$ meets
    three distinct $F_c$ (uniformly in a parameter when $F_c$ are compact subsets
    of a parameter-point product). In particular, if a finite-valued guide has all
    its true vertices in a compact set $W$ contained in the union $U$ of open
    inner hole squares, then there exists $\delta>0$ such that every closed
    radius-$\delta$ footprint meets at most two labels on $W\setminus U$
    (\cite[Section~7, lines 453--469, 475--479]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    On $F_{c_1}\times F_{c_2}\times F_{c_3}$, the continuous function
    $\max(|y_1-y_2|,|y_1-y_3|)$ attains a positive minimum $3\delta_{c_1,c_2,c_3}>0$;
    take $\delta=\min\delta_{c_1,c_2,c_3}$. Applying this to the compact sets
    $F_c=\overline{f^{-1}(c)}\cap(W\setminus U)$ (whose triple intersections are
    empty because any common point is a true vertex in $W\setminus U$) gives the
    guide claim.
\end{proof}

\begin{lemma}[Footprint ratios uniform in the scale]
    \label{lem:peps_dyadic_footprint_uniform}
    \lean{TNLean.PEPS.Approximation.HasTwoOwnerFootprints, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.mono, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.relabel, TNLean.PEPS.Approximation.similarity, TNLean.PEPS.Approximation.dist_similarity, TNLean.PEPS.Approximation.similarity_inv, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.image_similarity, TNLean.PEPS.Approximation.isCompact_footprintJointSet, TNLean.PEPS.Approximation.exists_hasTwoOwnerFootprints_Icc, TNLean.PEPS.Approximation.exists_footprintRatio, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.of_template, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.of_le_radii, TNLean.PEPS.Approximation.norm_smul_inv_sub_le_iff}
    \leanok
    \uses{lem:peps_dyadic_footprint}
    A guide $f$ satisfies the two-owner condition $(W,H,r,\delta)$ if every closed
    radius-$\delta$ sup square meets at most two labels of $f$ on
    $W\setminus\bigcup_{h\in H}B(h,r)$. This condition is preserved under
    shrinking $\delta$, relabelling $f$, enlarging hole radii above $r$, and
    applying a similarity $u\mapsto v+su$ ($s>0$, scaling $r,\delta$ to
    $sr,s\delta$). If $f$ is finite-valued, $W$ is compact, and every true vertex
    in $W$ lies in $\bigcup_{h\in H}B(h,r_0)$ ($r_0>0$), then for any $r_1\ge r_0$

    finite family of such template guides there exists $\nu>0$ such that every
    relabelling of a similarity image with ratio $s>0$ satisfies
    $(sW+v,sH+v,sr,\nu s)$ for all $r\in[r_0,r_1]$
    (\cite[Section~7, lines 464--479]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    For each label $c$,
    $F_c=\{(r,y)\in[r_0,r_1]\times W:y\in\overline{f^{-1}(c)},\ \forall h\in H,\ |y-h|\ge r\}$
    is compact, and $F_{c_1}\cap F_{c_2}\cap F_{c_3}=\emptyset$ for distinct
    $c_1,c_2,c_3$ since any common $(r,y)$ would give a true vertex $y\in W$ with
    $|y-h|\ge r\ge r_0$ for all $h\in H$. The parametric form of
    Lemma~\ref{lem:peps_dyadic_footprint}, minimized over the finitely many
    templates and rescaled by $s$, yields $\nu$.
\end{proof}

\begin{lemma}[Explicit footprint ratios for the readings of block guides]
    \label{thm:peps_dyadic_block_reading_ratio}
    \lean{TNLean.PEPS.Approximation.IsDisplacedReading.blockGuide_eq_of_mem,
        TNLean.PEPS.Approximation.IsDisplacedReading.blockCorner_mem_closure,
        TNLean.PEPS.Approximation.floor_div_adjacent,

        TNLean.PEPS.Approximation.resizeRatio_le_div,
        TNLean.PEPS.Approximation.resizing_footprintRatio_mixed_reading_of_le}
    \leanok
    \uses{def:peps_dyadic_block_guide, lem:peps_dyadic_readings_exist}
    Let $g'$ be a labelling that reads, after any displacement, the guide uniform
    on the blocks of side $s>0$, and let every true vertex of $g'$ within $s$ of
    the working set $W$ be a hole center. Then every footprint radius $\delta$
    with $2\delta<r$ and $2\delta<s$ gives the two-owner condition with working
    set $W$ and inner radius $r$.

    In particular, let $\epsilon_0>0$, $\nu_1=\min(\epsilon_0,1)/8$ and
    $\nu\le\nu_1$. For $0<n<t$, every reading of a guide uniform on the
    $2n$-blocks satisfies the two-owner condition with working set the closed
    square of radius $2n$ about a grid corner, holes at its true vertices within
    $4n$ of that corner, each of radius $h_n$ or $h_{2n}$, and footprint radius
    $2\nu n$. Hence the ratio of Lemma~\ref{thm:peps_dyadic_sampled_ratio} for
    the resizings can be taken equal to $\nu_1$, in place of the minimum given by
    the compactness observation of~\cite[Section~7]{OpenAI2026PolynomialPEPS},
    lines 447--479 and 529--541.
\end{lemma}

\begin{proof}\leanok
    A reading of a block guide carries the label of a block on the open interior
    of that block, so a grid corner lies in the closure of the label of each of
    its four blocks. Let three points of a footprint outside the holes carry
    three labels. Move them by the displacement, by less than
    $\min(r-2\delta,s-2\delta)/4$; the moved points are pairwise closer than $s$,
    so their block indices differ by at most one in each coordinate. Three
    distinct labels need three distinct blocks among the four blocks about one
    grid corner, and two of them are diagonal to each other. That corner is a
    true vertex, and it lies within $2\delta$ plus the displacement of a
    footprint point, hence within $r$ and within $s$ of it. So the corner is a
    hole center and the footprint point lies in its hole, a contradiction. For
    the resizings, $s=2n$ and the smallest radius is $h_n=\epsilon_0n$, so
    $\nu\le\nu_1$ gives $2\cdot2\nu n\le\epsilon_0n/2<h_n$ and
    $2\cdot2\nu n\le n/2<2n$. A true vertex within $2n$ of the working set lies
    within $4n$ of the corner. Holes of radius $h_{2n}\ge h_n$ only enlarge the
    excluded set.
\end{proof}

    uses both owners of every block $(j',q,t)$ with $\max(|q-r'|,|t-s'|)\le1$. The
    resize at the junction $(j,u,v)$ uses the new owners $(j,q,t)$ of the nine
    blocks with $\max(|q-(u-1)|,|t-(v-1)|)\le1$.
\end{definition}

\begin{lemma}[Lifetime participation of block parties]
    \label{thm:peps_dyadic_lifetime_parties}
    \lean{TNLean.PEPS.Approximation.RepaintingUses.near, TNLean.PEPS.Approximation.JunctionUses.near, TNLean.PEPS.Approximation.card_dyadicJunction_uses_le, TNLean.PEPS.Approximation.card_gates_le_of_uses}
    \leanok

    most $32$ such repaintings), and every junction resize using $(j,r,s)$ is
    $1$-near $(j,r,s)$ and is one of at most $9$ junctions. Consequently, if each
    event has at most $G$ gates and each gate involves only parties used by its
    event, every block party participates in at most $41G\le64G$ gates
    (\cite[Proposition~7.1, item~2, lines 597--618]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_lifetime_incidence}
    A repainting at scale $j'\in\{j-1,j\}$ using $(j,r,s)$ has block interval
    touching $[2^jr,2^j(r+1)]$ in each coordinate, so it is $0$-near $(j,r,s)$,
    giving at most $2(4\cdot0+4)^2=32$ repaintings by
    Lemma~\ref{thm:peps_dyadic_lifetime_incidence}. A junction resize using
    $(j,r,s)$ has scale $j$ and indices $(u,v)\in\{r,r+1,r+2\}\times\{s,s+1,s+2\}$,
    giving at most $9$ junctions, and $32+9=41\le64$.
\end{proof}

\begin{definition}[Events and changes of the protocol schedule]
    \label{def:peps_dyadic_protocol_schedule}
    \lean{TNLean.PEPS.Approximation.ProtocolEvent, TNLean.PEPS.Approximation.ProtocolEvent.level, TNLean.PEPS.Approximation.ProtocolEvent.IsScheduled, TNLean.PEPS.Approximation.ProtocolEvent.rank, TNLean.PEPS.Approximation.eventSchedule, TNLean.PEPS.Approximation.BulkOp, TNLean.PEPS.Approximation.BulkOp.marks, TNLean.PEPS.Approximation.BulkOp.sheets, TNLean.PEPS.Approximation.RepaintOp, TNLean.PEPS.Approximation.BulkOp.treated, TNLean.PEPS.Approximation.BulkOp.ops, TNLean.PEPS.Approximation.SquareEdge.list, TNLean.PEPS.Approximation.edgeOps, TNLean.PEPS.Approximation.largeRepaintingOps, TNLean.PEPS.Approximation.eventOps, TNLean.PEPS.Approximation.protocolSchedule}

    junction resizes of scale $j+1$ followed by the square repaintings of scale
    $j$. Each resize or direct repainting is $1$ change. Each large-scale
    repainting consists of the central birth followed, along each of the four
    edges, by $1$ auxiliary-sheet initialization, $3$ auxiliary births, the lens
    exchange, and $4$ main births/deaths, where each bulk operation with $m$ marks
    ($m=4$ for the central birth, $m=2$ along an edge) and $\sigma$ sheets
    ($\sigma=2$ for the exchange, $\sigma=1$ otherwise) is bracketed by $m\sigma$
    point-treatment homogenizations and $m\sigma$ removals ($2m\sigma+1$ changes)
    (\cite[Proposition~7.1, \texttt{06-geometry.tex}, lines 82--221, 328--339, 507--541]{OpenAI2026PolynomialPEPS}).
\end{definition}

\begin{definition}[Parties of the protocol and their labels]

\begin{definition}[Patch data of a guide rewrite]

              label, $H_{\rm old},H_{\rm new}$ are corners of $S$, and

\subsection{The distribution protocol of Proposition 7.1}
\label{sec:peps_dyadic_protocol}

The following definition states the clauses
of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} (\texttt{06-geometry.tex},
lines 12--53) as a statement about distributed constructions in the sense of
Definition~\ref{def:peps_distributed_construction} whose chronology is cut into
changes; see~\cite{gap:polypeps_prop71_consequences_interface}. The existence
of such a protocol is Lemma~\ref{prop:peps_dyadic_protocol}.

\begin{definition}[Dyadic distribution protocols and the Section 8 bounds]
    \label{def:peps_dyadic_protocol}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol,
        TNLean.PEPS.Approximation.DyadicProtocol.input,
        TNLean.PEPS.Approximation.DyadicProtocol.trajectory,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv,
        TNLean.PEPS.Approximation.DyadicProtocol.readout,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalDensity,
        TNLean.PEPS.Approximation.DyadicProtocol.participation,
        TNLean.PEPS.Approximation.HierarchySquare,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.exists_originalCircuit_initialCopies,
        TNLean.PEPS.Approximation.LargeSizeDyadicProtocolConsequences,
        TNLean.PEPS.Approximation.DyadicProtocolBoundsAt,
        TNLean.PEPS.Approximation.LargeSizeDyadicProtocolBounds,
        TNLean.PEPS.Approximation.LargeSizeDyadicProtocolConsequences.largeSizeDyadicProtocolBounds,
        TNLean.PEPS.Approximation.LargeSizeDyadicProtocolBounds.largeSizeDyadicProtocolConsequences,
        TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.mono}
    \leanok
    \uses{def:peps_distributed_construction, def:peps_original_circuit,
        def:peps_dyadic_hv_routing}
    A dyadic distribution protocol on the square of side $L$ with local
    dimension $q$ consists of a finite set $P$ of parties, each attached to a
    dyadic square by its scale and block indices, a finite-dimensional private
    memory $H_p$ and a unit vector $x_p\in H_p$ for each party, a number $n$ of
    changes, a register layout $M_t$ of the memory at each stage $t\le n$, and
    for each $t<n$ a contraction circuit $W_t\colon M_t\to M_{t+1}$ whose
    nonprivate gates carry their monomial expansions and their participating
    parties. The first memory is $\bigotimes_pH_p$, and the last consists of one
    physical register of dimension $d_p$ for each party, followed by the private
    registers to be discarded. Each genuine site $v$ has its own party, attached
    to the unit square with address $v$, with $d_p=q$; every other party has
    $d_p=1$. Finally there is a reference vector $r_t\in M_t$ at each stage, with
    $r_0=\bigotimes_px_p$. The protocol assigns a party to each aligned square of
    the padded hierarchy, and records for each change the square of the
    repainting it belongs to, if any, and a list of encoded frames at each stage,
    the encoded copies of $\Omega$. The circuit vector is
    $\Psi_0=r_0$ and
    $\Psi_{t+1}=W_t\Psi_t$.

    For a count constant $C_N$, an expansion constant $C_K$, an error constant
    $C_E$, constants $b$, $k$, $r$ and a unit vector $\Omega$, the protocol
    satisfies the Section~8 bounds if the following hold.
    \begin{enumerate}
        \item Every party is attached to a square of the padded hierarchy,
              there are at most $C_NL^2$ parties, and at most $b$ of them are
              attached to any square. A different party is assigned to each
              aligned square of the padded hierarchy, including the root
              square, at the scale and address of that square. Every other
              party belongs to the repainting of a square of the hierarchy
              other than the root square: it has the scale and address of
              that square, and it participates only in changes of that
              repainting. The changes of the repainting of a square $S$ of
              side $2^j$ form one contiguous segment of the schedule, and
              every party of such a change has scale $j$ or an adjacent
              scale and an address within $b\,2^j$ of the address of $S$ in
              each coordinate. The initial encoded frames are copies of
              $\Omega$ without holes, each held entirely by one party, and
              the initial memory and vector of each party are those of the
              copies it holds; the party holding the first frame holds the
              main copy.
        \item There are at most $C_NL^3$ changes. Each change has at most $b$
              nonprivate gates, each involving at most $b$ parties. Every
              nonprivate gate is a sum of at most $C_KL^k$ monomials with
              absolute coefficient sum at most $C_KL^k$, at most $r$ pair
              effects and at most $r$ pair sources in each monomial. Every
              party participates in at most
              $b$ nonprivate gates over the whole schedule.
        \item Two parties of a nonprivate gate have equal or adjacent scales,
              and their addresses differ in each coordinate by at most $b$
              times the side of the square of either party.
        \item $\lVert W_tr_t-r_{t+1}\rVert\le C_EL^{-20}$ for every $t<n$, and in
              the physical and private coordinates of the last memory
              $r_n=\Omega\otimes\Xi$ with $1-C_EL^{-57}\le\lVert\Xi\rVert\le1$.
              Every intermediate reference vector is the product over the
              encoded frames $F$ of stage $t$, at least one,
              \[
                  r_t=\bigotimes_F\Omega_F,\qquad \Omega_F=K_F\Omega,
                  \qquad \lVert\Omega_F\rVert\le1,\qquad 0<t<n,
              \]
              with the initial frames prepared from $r_0$ by private maps alone and
              the raw registers of the main copy held by the genuine-site parties
              before the final readout change.
    \end{enumerate}
    The existence assertion is that for every $q\ge2$ and $J,\Delta>0$ there are
    constants $C$, $b$, $k$, $r$ and $L_0$, chosen before the size, the
    Hamiltonian and the ground vector, such that for every $L\ge\max\{2,L_0\}$
    and every $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$ there is a protocol
    with these bounds at $C_N=C_K=C_E=C$, the single constant of the source. The
    same requirement at given constants and a given threshold is stated for any
    family of states, consisting for each side $L$ of some vectors on the
    square of side $L$; the existence assertion concerns the family of ground
    vectors of the model. Every
    bound is monotone in each of the three constants, so this assertion is
    equivalent to the one with separate constants $C_N$, $C_K$, $C_E$: given
    those, take $C=\max\{C_N,C_K,C_E\}$. The constants are kept apart because the
    prefactor of the polynomial expansions is much larger than the counts and
    the errors.

    Under the endpoint-coordinate convention just described, these bounds carry
    every clause of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}: strict
    intermediate references are products of encoded frames, whereas the terminal
    reference is expressed in physical and private coordinates as
    $\Omega\otimes\Xi$, with the canonical regrouping of the same auxiliary stack
    proved for the constructed witness rather than retained as a separate field
    of the public protocol predicate. Every intermediate
    reference vector $r_t$, $0<t<n$, has at least one frame
    (\texttt{06-geometry.tex}, lines 38--40), and read per stage, being the
    main copy carries no content beyond the designations at the first stage and
    before the last change. The pair-source bound
    of~\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}, item~3
    (\texttt{04-compression.tex}, lines 143--147), is part of item~2 above, with
    the same constant $r$ as the pair effects; the recorded monomials contain no
    messages, since messages are expanded before a gate is recorded
    (\texttt{04-compression.tex}, lines 42--45). The gates $W_t$ are the
    selected contractions, as in Section~8 (\texttt{07-assembly.tex}, lines
    29--34). Each nonprivate gate $G$ of $W_t$ equals its recorded expansion,
    $G=\sum_\xi c_\xi M_\xi$, so
    $\lVert G-\sum_\xi c_\xi M_\xi\rVert=0\le L^{-a}$ for every $a$, as
    required of expansions accurate to every fixed inverse power of $L$
    (\texttt{06-geometry.tex}, lines 29--30).
\end{definition}

        TNLean.PEPS.Approximation.DyadicProtocol.participation_eq_card,
        TNLean.PEPS.Approximation.DyadicProtocol.participation_le_of_uses}
    \leanok
    \uses{def:peps_dyadic_protocol, def:peps_dyadic_used_parties}
    Let a protocol satisfy the Section~8 bounds of
    Definition~\ref{def:peps_dyadic_protocol} with $C_E\ge0$, on a square of side
    $L\ge1$, for a unit vector $\Omega$. Then:
    \begin{enumerate}
        \item the physical density of the final circuit vector satisfies
              $\lVert\Tr_{\rm private}|\Psi_n\rangle\langle\Psi_n|
                  -|\Omega\rangle\langle\Omega|\rVert_1\le(2C_NC_E+2C_E)L^{-10}$;
        \item every party network whose links join two parties of a common
              nonprivate gate has link anchors differing in each coordinate by
              at most $b$ times the side of either endpoint square.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_protocol_density, thm:peps_dyadic_lifetime_parties}
    The changes are contractions, $\lVert r_0\rVert=1$, and the circuit and
    reference vectors agree at the first stage, so the first item is
    Lemma~\ref{thm:peps_protocol_density} with
    $C_{\rm ch}=C_N$ and $C_{\rm step}=C_{\rm hole}=C_E$. The second is the address bound of
    item~3 applied to the two endpoints of a link. For the last, the participation
    of a party is the number of nonprivate gates of all changes in which it
    participates, and Lemma~\ref{thm:peps_dyadic_lifetime_parties} bounds it.
\end{proof}

\begin{lemma}[Retained norm of a protocol with the Section 8 bounds]
    \label{thm:peps_dyadic_protocol_retained_norm}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.IsDistributionProtocol.retained_norm_and_trace_le}
    \leanok
    \uses{def:peps_dyadic_protocol}
    Let a protocol satisfy the Section~8 bounds of
    Definition~\ref{def:peps_dyadic_protocol} with $C_E\ge0$, on a square of side
    $L\ge1$, for a unit vector $\Omega$. Then the final circuit vector $\Psi_n$
    and its physical density
    $\rho=\Tr_{\rm private}|\Psi_n\rangle\langle\Psi_n|$ satisfy
    \begin{align}
        \lVert\Psi_n\rVert & \ge1-C_EL^{-57}-C_NC_EL^{-17},             &
        1-\tr\rho          & \le2\left(C_EL^{-57}+C_NC_EL^{-17}\right).
    \end{align}
    without renormalizing any intermediate vector
    (\cite[Section~8.1]{OpenAI2026PolynomialPEPS}).
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_protocol_retained_norm}
    As in the proof of Lemma~\ref{thm:peps_dyadic_protocol_consequences}, the
    hypotheses of Lemma~\ref{thm:peps_protocol_density} hold with
    $C_{\rm ch}=C_N$ and $C_{\rm step}=C_{\rm hole}=C_E$, and
    Lemma~\ref{thm:peps_protocol_retained_norm} gives both estimates.
\end{proof}
