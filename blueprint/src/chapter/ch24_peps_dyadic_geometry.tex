\section{Geometry of the dyadic distribution protocol}
\label{sec:peps_dyadic_geometry}
This section treats the geometric part of~\cite[Section~7]{OpenAI2026PolynomialPEPS},
September 24, 2026, \texttt{06-geometry.tex}: the bands along the edges of a
repainted dyadic square, the angular separation of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}
(lines 254--319), the point-treatment estimates (lines 357--415), the absence of
changed true vertices in the bulk operations and the tags (lines 375--398 and
610--631), and parts of the geometry of the small-patch covers, of the direct
repaintings and of the hole resizings (lines 341--351 and 447--533), and the counts of the padded
dyadic hierarchy behind items 1--3
of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} (lines 12--53 and 633--652).
The protocol of Proposition~7.1 itself, whose changes are implemented by the
information estimate and the contraction gates of the earlier sections, is not
treated here.

The clearance estimates are first proved for every guide with the properties that
the proof of Lemma~7.2 uses: the normal word of the operation on the band
$\{-8<x<2\}$ and no change off it, the label $A$ on the open square, and the label
$C$ on the exterior band $\{-8<x<0\}$. The schedule of guides is then constructed,
these properties are derived from the disjointness of the bands, and the estimates
are moved to every square of the hierarchy. This gives
Theorem~\ref{thm:peps_dyadic_angular_separation}, which is Lemma~7.2 for the
labellings that the schedule specifies; Lemma~7.2 for the guides sampled under the
source's tie convention is Theorem~\ref{thm:peps_dyadic_angular_separation_sampled}. A guide of the
source is a partition of the plane into open chambers, with labels on boundaries
sampled after one small generic displacement; here a guide is a labelling of the
whole plane, and two labellings that agree off finitely many interface curves
represent the same guide. The source's sampled guides are identified with readings
of these labellings in Lemma~\ref{lem:peps_dyadic_chamber_sampling}. The earlier scope
restriction and this convention are recorded
in~\cite{gap:openai26_dyadic_geometry_guide_properties}. Distances are sup
distances in the plane, and labels are elements of an arbitrary set; the labels
$A$, $B$, $C$ below are permitted to coincide.

\begin{definition}[Edge coordinates, bands and the lens]
    \label{def:peps_dyadic_edge_band}
    \lean{TNLean.PEPS.Approximation.SquareEdge, TNLean.PEPS.Approximation.bandWidth, TNLean.PEPS.Approximation.edgeBand, TNLean.PEPS.Approximation.closedEdgeBand, TNLean.PEPS.Approximation.bandCoord, TNLean.PEPS.Approximation.edgeLens, TNLean.PEPS.Approximation.edgeNeighbor, TNLean.PEPS.Approximation.edgeMarkDist, TNLean.PEPS.Approximation.cornerMarkDist}
    \leanok
    Let $S=[0,n]^2$. For an edge of $S$ let $s$ be the coordinate parallel to the
    edge, measured from one endpoint, and $d$ the normal coordinate, positive
    into $S$. Put
    \begin{align}
        w(s)=10^{-3}\min(s,n-s),\quad x=d/w(s).
        \label{eq:pdg_band}
    \end{align}
    The band $\{\alpha<x<\beta\}$ is the set of points with $0<s<n$ and
    $\alpha w(s)<d<\beta w(s)$; its closed version has $0\le s\le n$ and
    $\alpha w(s)\le d\le\beta w(s)$. The lens of the edge is the band
    $Y=\{-7<x<-7/2\}$, and the exterior neighboring square is
    $\{0<s<n,\ -n<d<0\}$. For a point $y$, $d_{\mathcal V}(y)$ is its distance from
    the two endpoints of the edge, or from the four corners of $S$.
\end{definition}

\begin{lemma}[Disjointness of the edge bands]
    \label{lem:peps_dyadic_band_disjoint}
    \lean{TNLean.PEPS.Approximation.disjoint_edgeBand, TNLean.PEPS.Approximation.edgeBand_neg_subset_edgeNeighbor, TNLean.PEPS.Approximation.dist_le_of_mem_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.changed_subset_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.dist_le_of_mem_closure_changed, TNLean.PEPS.Approximation.dist_le_of_mem_closure_centralRegion}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    The bands $\{-8<x<2\}$ along distinct edges of $S$ are disjoint, and the band
    $\{-8<x<0\}$ lies in the exterior neighboring square of its edge. Two points
    of a closed band with $-8\le\alpha\le\beta\le8$ are at distance at most $n$.
    Consequently the closure of the changed set of each elementary edge birth or
    death, and the closure of the central region of
    Theorem~\ref{thm:peps_dyadic_central_clearance}, have diameter at most $n$.
\end{lemma}

\begin{proof}\leanok
    Write $w\le s/1000$ and $w\le(n-s)/1000$. For opposite edges the bands lie
    within $n/250$ of their edges. For adjacent edges a common point at distances
    $r_1,r_2>0$ from the two edge lines would satisfy $r_1<0.002\,r_2$ and
    $r_2<0.002\,r_1$. On $\{-8<x<0\}$ one has $0>d>-8w\ge-n/250$. The diameter
    bound follows from $|d|\le8w\le n/250$ in each closed band. The changed set of
    an elementary operation lies in the closed band of its changed interval, and
    the central region lies in $S$.
\end{proof}

\begin{definition}[Normal words of the edge construction]
    \label{def:peps_dyadic_normal_words}
    \lean{TNLean.PEPS.Approximation.bandWord, TNLean.PEPS.Approximation.edgeStartWord, TNLean.PEPS.Approximation.auxWord, TNLean.PEPS.Approximation.mainWordOne, TNLean.PEPS.Approximation.edgeOperations, TNLean.PEPS.Approximation.BandOperation}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    A normal word lists labels in increasing $x$ separated by interfaces; at an
    interface curve the label on its right is taken, whereas the source samples after
    one generic displacement; the two conventions differ only on the interface
    curves. An edge operation starts from the
    main word $C\mid A\mid B$ with interfaces $0,1$. On an auxiliary sheet, uniform
    at $C$, the births of $B$ in $(-6,-1)$, of $A$ in $(-5,-2)$ and of $C$ in
    $(-4,-3)$ give
    \begin{align}
        C\mid B\mid A\mid C\mid A\mid B\mid C\quad\text{at } -6,-5,-4,-3,-2,-1.
        \label{eq:pdg_aux_word}
    \end{align}
    After the lens exchange the main word is $C\mid B\mid A\mid C\mid A\mid B$ at
    $-6,-5,-4,0,1$. The $C$ band $(-4,0)$ then dies into $A$, the $A$ band
    $(-5,1)$ dies into $B$, $C$ is born in $(-3,0)$, and the $B$ band $(-6,-3)$
    dies into $C$, leaving $C\mid B$ at $0$. Each of these seven births and deaths
    changes its word only on a closed interval, and its surrounding word (the word
    before a birth, after a death) carries one label on a strictly larger open
    interval inside $(-8,2)$, with margins at least $1/2$.
    The seven normal words are drawn below at one fixed $0<s<n$ over
    $-8\le x\le2$: row~(1) is the starting main word, row~(2) the auxiliary
    word~(\ref{eq:pdg_aux_word}) with the lens $Y$, row~(3) the main word after the
    exchange, and rows~(4)--(7) the main words after the death of $C$ on $(-4,0)$,
    the death of $A$ on $(-5,1)$, the birth of $C$ on $(-3,0)$ and the death of $B$
    on $(-6,-3)$.
    \input{figures/peps_dyadic_normal_words}
\end{definition}

\begin{lemma}[The lens exchange produces the main word]
    \label{lem:peps_dyadic_lens_word}
    \lean{TNLean.PEPS.Approximation.lens_exchange_word}
    \leanok
    \uses{def:peps_dyadic_normal_words}
    Replacing the starting main word by the auxiliary word (\ref{eq:pdg_aux_word})
    on $-7<x<-7/2$ gives the main word $C\mid B\mid A\mid C\mid A\mid B$ at
    $-6,-5,-4,0,1$, at every value of $x$.
\end{lemma}

\begin{proof}\leanok
    Both thresholds $-7$ and $-7/2$ lie in $C$ intervals of both words; compare
    the two words on each interval.
\end{proof}

\begin{lemma}[Conical band estimate]
    \label{lem:peps_dyadic_conical_band}
    \lean{TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt, TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt_markDist, TNLean.PEPS.Approximation.edgeMarkDist_le_of_mem_closedEdgeBand}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $a_0=1/5000$, $\alpha\le\beta$, $\alpha'+1/2\le\alpha$, $\beta+1/2\le\beta'$
    and $-8\le\alpha'$, $\beta'\le8$. If $y$ lies in the closed band
    $\{\alpha\le x\le\beta\}$ and $|y-z|<a_0\min(n,d_{\mathcal V}(y))$, then $z$
    lies in the open band $\{\alpha'<x<\beta'\}$.
\end{lemma}

\begin{proof}\leanok
    Put $m=\min(s,n-s)$ at $y$, so $w=m/1000$ and $|d|\le m$; hence
    $d_{\mathcal V}(y)\le m$. If $|y-z|<m/5000$, the parallel and normal
    coordinates move by less than $m/5000$, and $w$ is $10^{-3}$-Lipschitz, so
    $|\beta'(w(s_z)-w(s_y))|\le8m/(5\cdot10^{6})$. Then
    $d_z<\beta w(s_y)+m/5000\le\beta' w(s_z)-m/2000+m/5000+8m/(5\cdot10^6)$, and
    symmetrically at $\alpha'$.
\end{proof}

\begin{theorem}[Angular separation of the central birth]
    \label{thm:peps_dyadic_central_clearance}
    \lean{TNLean.PEPS.Approximation.centralRegion, TNLean.PEPS.Approximation.centralBirth_clearance}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $R$ be the set of points of $(0,n)^2$ with $x>1$ for every edge, and let
    $f$ be a labelling of the plane equal to $A$ on $(0,n)^2$. For every $y$ in
    the closure of $R$ and every $z$ with $f(z)\ne A$,
    \begin{align}
        |y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\},
        \label{eq:pdg_central}
    \end{align}
    where $\mathcal V$ is the set of corners of $S$ and $a_0=1/5000$. This is the
    clearance estimate for the central birth in the proof
    of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any surrounding guide equal
    to $A$ on the interior of $S$.
    The guide after the central birth is sketched below, with the bands
    $\{0<x<1\}$ drawn far wider than they are and their narrowing towards the corners
    drawn as four strings; the dots are the corners in $\mathcal V$.
    \input{figures/peps_dyadic_central_clearance}
\end{theorem}

\begin{proof}\leanok
    With $u=\min(y_1,n-y_1)$ and $v=\min(y_2,n-y_2)$, the four conditions $d\ge w$
    give $v\ge u/1000$ and $u\ge v/1000$. The nearest corner is at distance
    $\max(u,v)\le1000\min(u,v)$, so a point within $a_0\min(n,d_{\mathcal V}(y))$
    of $y$ is within $\min(u,v)/5$ of $y$ and lies in $(0,n)^2$.
\end{proof}

\begin{theorem}[Angular separation of the edge births and deaths]
    \label{thm:peps_dyadic_band_clearance}
    \lean{TNLean.PEPS.Approximation.BandOperation.clearance, TNLean.PEPS.Approximation.auxBirthB, TNLean.PEPS.Approximation.auxBirthA, TNLean.PEPS.Approximation.auxBirthC, TNLean.PEPS.Approximation.mainDeathC, TNLean.PEPS.Approximation.mainDeathA, TNLean.PEPS.Approximation.mainBirthC, TNLean.PEPS.Approximation.mainDeathB}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    Consider one of the seven elementary births and deaths of an edge operation,
    and labellings $f_b$, $f_a$ and $f_s$ of the plane whose normal words on
    $\{-8<x<2\}$ are its words before, after and surrounding, with $f_b=f_a$ off
    that band. Let $P_\circ$ be its surrounding label. For every $y$ in the closure
    of $\{f_b\ne f_a\}$ and every $z$ with $f_s(z)\ne P_\circ$,
    $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$, with $\mathcal V$ the two endpoints
    of the edge. These are the clearance estimates for the edge births and deaths in
    the proof of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any guides with
    the stated words.
\end{theorem}

\begin{proof}\leanok
    The set $\{f_b\ne f_a\}$ lies in the closed band of the changed interval,
    which is closed. If $|y-z|<a_0\min\{n,d_{\mathcal V}(y)\}$, the conical band
    estimate puts $z$ in the open band of the surrounding sector, where $f_s$
    equals $P_\circ$.
\end{proof}

\begin{theorem}[Angular separation of the lens exchange]
    \label{thm:peps_dyadic_lens_clearance}
    \lean{TNLean.PEPS.Approximation.lensExchange_clearance}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    Let the main labelling be $C$ on the exterior band $\{-8<x<0\}$, and let the
    auxiliary labelling have the word (\ref{eq:pdg_aux_word}) on $\{-8<x<2\}$ and
    be $C$ off it. Every point $y$ in the closure of the positions that are not
    $C$ on both labellings satisfies
    $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$ for every $z$ on the boundary of the
    lens $Y$. This is the clearance estimate for the lens exchange in the proof
    of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any guides with the stated
    properties.
\end{theorem}

\begin{proof}\leanok
    A boundary point of $Y$ is an endpoint of the edge, where the bound is
    immediate, or satisfies $d=-7w$ or $d=-\tfrac72w$ with $0<s<n$. If $y$ lies
    outside the exterior band $\{-8<x<0\}$, then $d_{\mathcal V}(y)\le
        d_{\mathcal V}(z)+|y-z|$ and $d_{\mathcal V}(z)\le m_z=\min(s_z,n-s_z)$; a
    distance below $a_0d_{\mathcal V}(y)$ is then below $m_z/2000$, which keeps
    $y$ in that band because $-7m_z/1000\le d_z\le-\tfrac72m_z/1000$ and $w$ is
    $10^{-3}$-Lipschitz. If $y$ lies in one of
    the closed auxiliary bands $\{-6\le x\le-4\}$ or $\{-3\le x\le-1\}$, apply the
    conical band estimate with the open bands $\{-\tfrac{13}2<x<-\tfrac72\}$ or
    $\{-\tfrac72<x<-\tfrac12\}$, neither of which meets the two boundary curves.
\end{proof}

\begin{definition}[The schedule of guides of a repainting]
    \label{def:peps_dyadic_schedule}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline, TNLean.PEPS.Approximation.openSquare, TNLean.PEPS.Approximation.closedSquare, TNLean.PEPS.Approximation.bandUpdate, TNLean.PEPS.Approximation.RepaintingBaseline.centralGuide, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide, TNLean.PEPS.Approximation.RepaintingBaseline.edgeInterface, TNLean.PEPS.Approximation.auxSteps, TNLean.PEPS.Approximation.mainSteps}
    \leanok
    \uses{def:peps_dyadic_normal_words, thm:peps_dyadic_central_clearance}
    A repainting of $S=[0,n]^2$, $n>0$, starts from a guide $g$ equal to $A$ on the
    open square and to one label $C_e$ on the open exterior neighboring square across
    each edge $e$. The guide after the central birth is $B$ on the central region and
    $g$ elsewhere. For a list $E$ of edges, the completed guide replaces, for each
    $e\in E$, the band $\{-8<x<2\}$ of $e$ by the word $C_e\mid B$ at $0$. During the
    construction along $e$ after the edges of $E$, the main guide with normal word $W$
    is the completed guide with the band of $e$ replaced by $W$, and the auxiliary
    guide with normal word $W$ is the uniform guide $C_e$ with the band of $e$
    replaced by $W$. The three auxiliary births and the four main births and deaths
    are those of Definition~\ref{def:peps_dyadic_normal_words}. Guides are
    labellings of the plane; at the start of the construction along $e$ the schedule
    passes from the completed guide to the main guide with the starting word, which
    agrees with it off the curves $x=0$ and $x=1$ of the band of $e$.
\end{definition}

\begin{lemma}[The guides of the schedule]
    \label{lem:peps_dyadic_schedule_guides}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_edgeBand, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_edgeStartWord, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_mainGuide_start, TNLean.PEPS.Approximation.RepaintingBaseline.lensExchange_mainGuide, TNLean.PEPS.Approximation.auxSteps_chain, TNLean.PEPS.Approximation.mainSteps_chain, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide_mainWordFive, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide_uniform, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_notMem_closedSquare, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_openSquare, TNLean.PEPS.Approximation.exists_edgeBand_of_notMem_centralRegion, TNLean.PEPS.Approximation.mem_centralRegion_of_forall_notMem}
    \leanok
    \uses{def:peps_dyadic_schedule, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_lens_word}
    Let $e\notin E$. On the band of $e$, off the curves $x=0$ and $x=1$, the
    completed guide carries the word $C_e\mid A\mid B$ at $0,1$; so it agrees with the
    main guide with that word off those curves. Exchanging the starting main guide
    and the auxiliary guide with word~(\ref{eq:pdg_aux_word}) on the lens $Y$ gives
    the main guide with word $C_e\mid B\mid A\mid C_e\mid A\mid B$ at
    $-6,-5,-4,0,1$. The auxiliary births start from the uniform word $C_e$, the main
    operations start from that main word, consecutive operations on one sheet match,
    and the last main death leaves the completed guide for the list $E$ with $e$
    added. Once all four edges are completed, the open square carries $B$, and every
    point outside the closed square keeps its starting label.
\end{lemma}

\begin{proof}\leanok
    On the band of $e$, the bands of the completed edges are absent by the band
    disjointness. Where $x<0$ the point lies in the exterior neighboring square;
    where $0<x<1$ it lies in the open square and outside the central region; where
    $1<x<2$ it lies in the central region, since for every other edge it lies in the
    open square and outside that edge's band, so its normal ratio there is at least
    $2$. The exchange is Lemma~\ref{lem:peps_dyadic_lens_word} on the band, and the
    identity off it. A point of the open square outside the central region lies in
    the band $\{0<x\le1\}$ of some edge, where the final word is $B$; a point outside
    the closed square lies in no band or in the part $x<0$ of a band, where the final
    word is $C_e$.
\end{proof}

\begin{lemma}[Translation of the estimates]
    \label{lem:peps_dyadic_translation}
    \lean{TNLean.PEPS.Approximation.clearance_shift, TNLean.PEPS.Approximation.dist_le_shift, TNLean.PEPS.Approximation.le_dist_of_mem_closure, TNLean.PEPS.Approximation.cornerMarkDist_sub, TNLean.PEPS.Approximation.edgeMarkDist_sub, TNLean.PEPS.Approximation.sub_mem_closure, TNLean.PEPS.Approximation.sub_mem_frontier, TNLean.PEPS.Approximation.mem_closure_shift}
    \leanok
    \uses{def:peps_dyadic_block_guide}
    Moving all guides by a vector $o$ moves the changed regions and the obstructing
    sets by $o$, and the mark distances become the distances from the moved marks;
    sup distances are translation invariant. A lower bound on the distance from a
    point to every point of a set holds for its closure.
\end{lemma}

\begin{proof}\leanok
    Translation is a homeomorphism, so it commutes with closures and boundaries,
    and $|(y-o)-(z-o)|=|y-z|$. The set of points at distance at least $c$ from $y$
    is closed.
\end{proof}

\begin{theorem}[Separation of the unmodified bands]
    \label{thm:peps_dyadic_angular_separation}
    \lean{TNLean.PEPS.Approximation.blockRepainting_central, TNLean.PEPS.Approximation.blockRepainting_auxStep, TNLean.PEPS.Approximation.blockRepainting_mainStep, TNLean.PEPS.Approximation.blockRepainting_exchange, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_central, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_central, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_auxStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_mainStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_exchange, TNLean.PEPS.Approximation.BandOperation.clearance_bandUpdate, TNLean.PEPS.Approximation.BandOperation.dist_le_bandUpdate, TNLean.PEPS.Approximation.dist_le_of_mem_closure_edgeLens, TNLean.PEPS.Approximation.RepaintingBaseline.exchange_changed_subset, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_exchange}
    \leanok
    \uses{thm:peps_dyadic_central_clearance, thm:peps_dyadic_band_clearance, thm:peps_dyadic_lens_clearance, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_schedule_guides, lem:peps_dyadic_translation}
    Let $a_0=1/5000$, $n>0$, and consider the repainting of a block $S$ of side $n$
    from a block guide, with any block labels and any final label, through the
    guides that the schedule specifies at each operation. Every elementary
    birth or death of its schedule, the central birth and the seven births and
    deaths along each edge after any completed edges, has changed region $R$ of
    diameter at most $n$, and its surrounding guide $f$ satisfies, at every point
    $y\in\overline R$,
    \begin{align}
        \operatorname{dist}\bigl(y,\overline{\{f\ne P_\circ\}}\bigr)
        \ge a_0\min\{n,d_{\mathcal V}(y)\},
        \label{eq:pdg_angular_separation}
    \end{align}
    where $\mathcal V$ is the set of corners of $S$ for the central birth and the two
    endpoints of the edge otherwise. For each lens exchange, every point $y$ in the
    closure of the positions that are not common $C_e$ on its two guides before the
    exchange satisfies
    $\operatorname{dist}(y,\partial Y)\ge a_0\min\{n,d_{\mathcal V}(y)\}$. This
    is~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS} for the labellings that the schedule
    specifies; for the guides sampled under the source's tie convention it is
    Theorem~\ref{thm:peps_dyadic_angular_separation_sampled}. The positions inside $Y$ that
    are not common $C_e$ before the exchange, which contain the positions the
    exchange changes, also form a set whose closure has diameter at most $n$; the
    source uses this bound in its proof of the exchange step, lines 421--424.
\end{theorem}

\begin{proof}\leanok
    The guides of an edge construction are obtained from one guide of their sheet by
    replacing the band of the edge by a word, so they have the properties assumed in
    Theorems~\ref{thm:peps_dyadic_central_clearance},
    \ref{thm:peps_dyadic_band_clearance} and~\ref{thm:peps_dyadic_lens_clearance}:
    such a guide carries its word on the band, two such guides of one sheet agree off
    the band, and the starting main word is $C_e$ on $\{-8<x<0\}$; the starting
    guide of the central birth is $A$ on the open square.
    Lemma~\ref{lem:peps_dyadic_schedule_guides} identifies these guides with the
    guides reached by the schedule. The diameter bound is
    Lemma~\ref{lem:peps_dyadic_band_disjoint}. The exchange changes the main guide

\begin{lemma}[Incident labels under local changes]
    \label{lem:peps_dyadic_incident_change}
    \lean{TNLean.PEPS.Approximation.isTrueVertex_congr, TNLean.PEPS.Approximation.not_isTrueVertex_of_subset_pair, TNLean.PEPS.Approximation.incidentLabels_congr, TNLean.PEPS.Approximation.incidentLabels_subset, TNLean.PEPS.Approximation.mem_range_of_mem_incidentLabels, TNLean.PEPS.Approximation.incidentLabels_eq_or_subset_pair, TNLean.PEPS.Approximation.isTrueVertex_iff_of_two_labels, TNLean.PEPS.Approximation.incidentLabels_piecewise, TNLean.PEPS.Approximation.isTrueVertex_piecewise_iff, TNLean.PEPS.Approximation.incidentLabels_homogenize_subset, TNLean.PEPS.Approximation.IsTrueVertex.of_homogenize}
    \leanok
    \uses{def:peps_dyadic_incident_labels}
    Two guides that agree near $c$ have the same incident labels at $c$, and a point
    with at most two incident labels is not a true vertex.
    \begin{enumerate}
        \item Let the closure of the set where two guides differ lie in an open set $U$
              on which both take only the values $P$ and $Q$. Then the two guides have
              the same true vertices; each lies outside $U$ and has the same incident
              labels in both guides.
        \item Let two guides $h_1,h_2$ both equal $C$ on an open set $U$ containing
              $\partial Y$, and let $h$ be $h_2$ on $Y$ and $h_1$ off $Y$. Then $h$ has a
              true vertex at $c$ if and only if $c\in Y$ and $h_2$ has one there, or
              $c\notin Y$ and $h_1$ has one there; it then has the incident labels of
              that guide at $c$.
        \item A true vertex of a guide homogenized to $P$ on an open set $U$ is either a
              true vertex of the guide off $\overline U$, or a point of $\partial U$ at
              which two distinct labels other than $P$ of the guide meet.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    Closures at $c$ depend only on a neighborhood of $c$. In the first case a point
    outside $\overline{\{f\ne g\}}$ has a neighborhood where the guides agree, and a
    point of $U$ has one where both take only two values. In the second, a point of
    $U$ has only the incident label $C$, and a point off $\partial Y$ lies in the
    interior of $Y$ or of its complement. In the third, a point of $U$ has only the
    incident label $P$, a point off $\overline U$ has a neighborhood where
    homogenization does nothing, and homogenization adds at most the label $P$ to
    the incident labels.
\end{proof}

\begin{lemma}[The compactness argument]
    \label{lem:peps_dyadic_footprint}
    \lean{TNLean.PEPS.Approximation.exists_pos_le_max_dist_of_inter_eq_empty, TNLean.PEPS.Approximation.exists_footprint_two_owners, TNLean.PEPS.Approximation.exists_footprint_two_owners_param, TNLean.PEPS.Approximation.exists_footprint_two_owners_guide}
    \leanok
    \uses{def:peps_dyadic_incident_labels}
    Let finitely many compact sets $F_c$ in a metric space be such that no three with
    distinct labels have a common point. Then there is $\delta>0$ such that no
    closed ball of radius $\delta$ meets three of them with distinct labels. If the
    $F_c$ are compact sets of pairs of a parameter and a point, one $\delta$ works
    for all values of the parameter. In particular, let a guide take finitely many
    values, let $W$ be a compact working neighborhood, and let every true vertex in
    $W$ lie in one of the open inner hole squares. Then there is $\delta>0$ such that
    every closed footprint of radius $\delta$ meets at most two labels at its points
    of $W$ outside the open inner holes. This is the abstract compactness argument
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 453--462 and 475--479, and
    its parametric form, lines 464--469, for compact joint sets.
\end{lemma}

\begin{proof}\leanok
    For three compact sets without a common point,
    $\max(|y_1-y_2|,|y_1-y_3|)$ attains a positive minimum on their product; take a
    third of the smallest of these minima over the finitely many triples. For the
    parameters, apply this in the product space, where points with the same
    parameter are at the distance of their second coordinates. For the guide, take
    $F_c$ the closure of the region of $c$ in $W$ with the open inner holes removed;
    a common point of three of them would be a true vertex outside every inner hole.
\end{proof}

\begin{lemma}[Footprint ratios uniform in the scale]
    \label{lem:peps_dyadic_footprint_uniform}

    $u\mapsto v+su$, $s>0$, to the image guide, working set and hole centers with the

\begin{proof}\leanok

    \lean{TNLean.PEPS.Approximation.bandWord_eventually_eq, TNLean.PEPS.Approximation.edgeCurves, TNLean.PEPS.Approximation.mainWords_eventually_eq, TNLean.PEPS.Approximation.auxWords_eventually_eq, TNLean.PEPS.Approximation.eventually_bandUpdate_eq, TNLean.PEPS.Approximation.RepaintingBaseline.eventually_centralGuide_eq, TNLean.PEPS.Approximation.RepaintingBaseline.eventually_completedGuide_eq, TNLean.PEPS.Approximation.RepaintingBaseline.IsUnmodifiedGuide.eventually_eq, TNLean.PEPS.Approximation.rimPattern, TNLean.PEPS.Approximation.rimPattern_finite, TNLean.PEPS.Approximation.dist_of_mem_rimPattern, TNLean.PEPS.Approximation.le_dist_of_mem_rimPattern, TNLean.PEPS.Approximation.mem_rimPattern_of_grid, TNLean.PEPS.Approximation.mem_rimPattern_of_curve, TNLean.PEPS.Approximation.mem_rimPattern_of_incidentLabels}
    \leanok
    \uses{lem:peps_dyadic_unmodified_vertices, lem:peps_dyadic_choice_K0}
    For a point $v$ and $t>0$, the rim pattern about $v$ consists of the points
    $v+(\sigma t,jt/1000)$ and $v+(jt/1000,\sigma t)$ with $\sigma=\pm1$ and
    $j\in\mathbb Z$, $|j|\le6$: on each side of the rim of the square of radius $t$
    about $v$, thirteen points at spacing $t/1000$ about the midpoint of the side. It
    is finite, its points are at distance $t$ from $v$, and distinct points are at
    distance at least $t/1000$. Off the grid lines and off the curves $x=k$,
    $k\in\{-6,\dots,1\}$, of the bands of the four edges, every unmodified guide is
    constant near each point. Let $2t<n$. A point at distance $t$ from a grid corner
    that lies on a grid line, or on one of these curves of an edge of the block, lies
    in the rim pattern of that corner: the grid line passes through the corner, and the
    curve belongs to one of the two edges ending at the corner. Hence every point of
    the rim about a grid corner at which an unmodified guide has two distinct incident
    labels lies in the rim pattern. This is the finite pattern of rim intersection
    points of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 344--349, with the rays
    of both adjacent edges and the grid lines.
    The source's illustration of a point treatment for three distinct labels is
    redrawn below in units of $t$: homogenizing the open square of radius $t$ about
    the true vertex to $P$ moves the meeting point of $P$, $A$ and $B$ to the
    midpoint of the right side of the rim, a point of the rim pattern.
    \input{figures/peps_dyadic_rim_pattern}
\end{lemma}

\begin{proof}\leanok
    A word is constant near every point other than its interfaces, and the interfaces
    of the main and auxiliary words are integers from $-6$ to $1$; the boundary of
    the central region lies on the curves $x=1$. For a point on a grid line, the line
    is within $t<n/2$ of the corner, so it passes through the corner, and the point is
    the midpoint of a side of the rim. For a point on a curve of an edge, its normal
    coordinate is at most $6\cdot10^{-3}\min(s,n-s)$ in absolute value, so the corner
    has normal coordinate $0$ and parallel coordinate $0$ or $n$; the straight-ray
    chart of Lemma~\ref{lem:peps_dyadic_choice_K0} then places the point at parallel
    distance $t$ and normal coordinate $kt/1000$. Distinct pattern points on one side
    differ by at least $t/1000$ along it, and points on different sides differ by at
    least $(1-6/1000)t$ in one coordinate.
\end{proof}

\begin{theorem}[Disjoint outer holes in the temporary guides]
    \label{thm:peps_dyadic_temporary_holes}
    \lean{TNLean.PEPS.Approximation.isTrueVertex_homogenize_cases, TNLean.PEPS.Approximation.le_dist_of_isTrueVertex_homogenize, TNLean.PEPS.Approximation.homogenize_outerHoles_disjoint, TNLean.PEPS.Approximation.IsDisplacedReading.preimage_subset_closure, TNLean.PEPS.Approximation.IsDisplacedReading.isTrueVertex, TNLean.PEPS.Approximation.homogenize_reading_outerHoles_disjoint}

    homogenize an unmodified guide to $P$ on the open squares of radius $t$ about the
    corners of $\mathcal M$. Every true vertex of the resulting guide is a grid corner
    or a point of the rim pattern of a corner of $\mathcal M$, and distinct true
    vertices are at distance at least $t/1000$. Hence for $\epsilon_0<1/4000$ the
    closed outer hole squares of radius $2\epsilon_0t$ about distinct true vertices are
    disjoint. The same holds for every reading of the homogenized guide after a
    displacement, and such a reading exists after every displacement. This is the
    conclusion of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 347--351, that the
    outer holes are disjoint in the temporary guides, for the separations of the true
    vertices; the further smallness of $\epsilon_0$
    compared with $a_0$ enters through Lemma~\ref{lem:peps_dyadic_birth_enlargement}.
\end{theorem}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_dyadic_incident_change} a true vertex of the homogenized
    guide is an old true vertex off the closed treated squares, a grid corner by
    Lemma~\ref{lem:peps_dyadic_unmodified_vertices}, or a point of a treated rim with
    two incident labels, in the rim pattern by Lemma~\ref{lem:peps_dyadic_rim_pattern}.
    Two rim points about one corner are separated by the pattern. Otherwise the two
    points belong to distinct grid corners, at distance at least $n$, and each lies
    within $t$ of its corner, so they are at distance at least $n-2t>t/1000$. A point
    of the region of a label in a reading is a limit of points of that region in the
    guide, so a true vertex of a reading is a true vertex of the guide; readings exist
    by Lemma~\ref{lem:peps_dyadic_readings_exist}.
\end{proof}

    distance $4t$ of the corner, which contains the working set and the holes.

\begin{theorem}[One patch ratio for direct repaintings and resizings]

    \leanok

    Fix $\rho$ and $0<r_0\le r_1$. There is a ratio $\nu>0$ such that every guide
    uniform on the blocks of side $s>0$, about every grid corner $v$ and for every
    inner radius $r\in[r_0s,r_1s]$, satisfies the two-owner condition with working set
    the closed square of radius $s\rho$ about $v$, holes at its true vertices within
    $s(\rho+1)$ of $v$, and footprint radius $\nu s$. In particular, for fixed
    $\epsilon_0>0$ and $K_0$, one ratio serves every direct repainting of a square of
    side $0<n\le K_0t$, before and after, with inner radius $h_n$ and working set the
    closed square of radius $2n$ about a corner of the square; and one ratio serves
    every resizing with $0<n<t$, with both radii $h_n$ and $h_{2n}$ on the guide uniform
    on the blocks of side $2n$ and working set of radius $2n$, and also when some holes
    have radius $h_n$ and the others $h_{2n}$. These are the compactness arguments
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 515--520 and 534--538, in units
    of $n$, with the intermediate configurations of lines 538--540.
\end{theorem}

\begin{proof}\leanok
    In units of $s$ about a grid corner, a block guide is the block guide of side one
    whose labels are those of a fixed square window of blocks, so it is an injective
    relabelling of one of finitely many templates, one for each label-equality type of
    the window. Apply Lemma~\ref{lem:peps_dyadic_footprint_uniform} and
    Lemma~\ref{lem:peps_dyadic_small_radius} for the radius ratios; when $n<t$ the
    radius $h_n=\epsilon_0n$ and $h_{2n}=\epsilon_0\min(2n,t)$ lie between
    $\epsilon_0n$ and $2\epsilon_0n$. Since $h_n\le h_{2n}$, a configuration with some
    holes resized has larger holes than the one with all radii $h_n$.
\end{proof}

\begin{theorem}[Patch ratios for the sampled guides]
    \label{thm:peps_dyadic_sampled_ratio}

    \leanok
    \uses{thm:peps_dyadic_point_ratio, thm:peps_dyadic_block_ratio, lem:peps_dyadic_readings_exist, lem:peps_dyadic_chamber_sampling}
    There is a ratio $\nu>0$, depending only on the fixed constants, such that for every
    generic displacement $v$ the conclusions of
    Theorems~\ref{thm:peps_dyadic_point_ratio} and~\ref{thm:peps_dyadic_block_ratio}
    hold for every labelling that reads the guides concerned after $v$, with holes at
    its own true vertices. These are the source's sampled guides, by
    Lemma~\ref{lem:peps_dyadic_chamber_sampling}; the point-treated guides are also
    constant off finitely many lines with normals in $\mathcal N$ near every point, so
    the same holds for them. As in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines
    73--77 and 559--563, the ratio is chosen before the displacement, which is chosen
    only after all the guides are specified.
\end{theorem}

\begin{proof}\leanok
    A reading takes only values of the labelling. If a guide agrees on an open square
    about $c$ with $u\mapsto\sigma(T((u-c)/s))$ for a template $T$, then every reading
    of the guide after $v$ agrees there with $u\mapsto\sigma(T'((u-c)/s))$ for the
    reading $T'$ of $T$ after $v$, since $(u+\varepsilon v-c)/s=(u-c)/s+(\varepsilon/s)v$.
    The templates of the point treatment and of the block guides are eventually constant
    along rays, so they have readings after $v$. They are also constant off walls on
    lines with normals in the finite set $\mathcal N$. Let $v$ and $w$ lie strictly on
    the same side of each line through the origin with normal in $\mathcal N$. For a
    point $p$ and small $\varepsilon>0$, the segment from $p+\varepsilon v$ to
    $p+\varepsilon w$ stays near $p$ and avoids the walls that miss $p$; a wall through
    $p$ with normal $N$ meets it only where $N\cdot((1-a)v+aw)=0$ for some
    $a\in[0,1]$, which is impossible because $N\cdot v$ and $N\cdot w$ have the same
    sign. So the template is locally constant along the segment, hence constant on it,
    and its readings after $v$ and after $w$ agree. A template therefore has at most one
    reading for each choice of sides of the finitely many wall directions, and only
    finitely many readings after generic displacements. Apply the compactness argument
    of Lemma~\ref{lem:peps_dyadic_footprint_uniform} to these finitely many readings of
    the finitely many templates.
\end{proof}

\begin{definition}[Padded dyadic hierarchy, junctions and addresses]
    \label{def:peps_dyadic_hierarchy}
    \lean{TNLean.PEPS.Approximation.DyadicSquare, TNLean.PEPS.Approximation.DyadicJunction}
    \leanok
    \uses{def:peps_dyadic_anchor}
    For $N=2^k$ the hierarchy consists of the aligned squares of $[0,N]^2$ of side
    $2^j$, $0\le j\le k$, indexed by $j$ and two block indices below $2^{k-j}$; the
    junctions at scale $j$ are the grid corners, indexed by two integers at most
    $2^{k-j}$. A square of side $2^j$ with block index $r$ has address coordinate
    $2^jr+2^{j-1}$ for $j\ge1$ and $r$ for $j=0$, the anchor coordinate of
    Definition~\ref{def:peps_dyadic_anchor}.
\end{definition}

\begin{theorem}[Counts of the hierarchy and the schedule]
    \label{thm:peps_dyadic_hierarchy_counts}
    \lean{TNLean.PEPS.Approximation.three_mul_card_dyadicSquare_add_one, TNLean.PEPS.Approximation.three_mul_card_dyadicSquare_lt, TNLean.PEPS.Approximation.three_mul_card_dyadicJunction_le, TNLean.PEPS.Approximation.three_mul_schedule_le, TNLean.PEPS.Approximation.three_mul_schedule_le_cube}
    \leanok
    \uses{def:peps_dyadic_hierarchy}
    The hierarchy has $(4^{k+1}-1)/3$ squares, and three times the number of
    junctions over all scales is at most $4^{k+2}$. If $2^k<2L$, the hierarchy has
    fewer than $16L^2/3$ squares, and $M$ times the number of squares plus
    junctions is at most $80ML^2/3$, and for $L\ge1$ at most $80ML^3/3$. This is
    the count behind a schedule using at most $M$ changes for each repainting of a
    square and each junction resize; the schedule of changes and gates of the
    protocol is not constructed here.
\end{theorem}

\begin{proof}\leanok
    The scale $j$ contributes $4^{k-j}$ squares and $(2^{k-j}+1)^2\le4\cdot4^{k-j}$
    junctions, and $3\sum_{i<k+1}4^i+1=4^{k+1}$ by induction. Finally
    $4^k=(2^k)^2<4L^2$.
\end{proof}

\begin{lemma}[Addresses and locality]
    \label{lem:peps_dyadic_address_locality}
    \lean{TNLean.PEPS.Approximation.dyadicAnchor_lt_two_pow, TNLean.PEPS.Approximation.dyadicAnchor_lt_add, TNLean.PEPS.Approximation.dyadicAnchor_dist_lt, TNLean.PEPS.Approximation.dyadicAnchor_dist_lt_of_adjacent, TNLean.PEPS.Approximation.card_blocks_meeting_le}
    \leanok
    \uses{def:peps_dyadic_hierarchy, lem:peps_dyadic_active_labels}
    In each coordinate: an address coordinate lies in $[2^jr,2^jr+2^j)$, hence
    below $2^k$. If the intervals $[2^jr,2^j(r+1)]$ and $[2^{j'}r',2^{j'}(r'+1)]$
    are within distance $g$ and $j'\le j+1$, the address coordinates differ by less
    than $g+3\cdot2^j$. In particular, for two blocks of side $2^j$ with equal or
    adjacent indices, each block or its parent of side $2^{j+1}$ has an address
    coordinate within $3\cdot2^{j+1}$ of the other's. The intervals of side $b$
    meeting $[V,U]$ number at most $\lfloor U/b\rfloor+2-\lfloor V/b\rfloor$. With
    Lemma~\ref{lem:peps_dyadic_active_labels}, these are ingredients of item~3 and
    of the bounded lifetime participation
    in~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}; the gates and parties of
    the protocol are not treated here.
\end{lemma}

\begin{proof}\leanok
    The first two claims follow from the position of the address in its block and
    $2^{j'}\le2\cdot2^j$. For adjacent blocks, the interval of a block or of its
    parent contains the block's own interval, and the two blocks' intervals touch,
    so the second claim applies with $g=0$. For the count, $r\mapsto r+1$ maps the meeting blocks
    injectively into $[\lfloor V/b\rfloor,\lfloor U/b\rfloor+1]$.
\end{proof}

\begin{definition}[Squares and junctions near a block]
    \label{def:peps_dyadic_near_block}
    \lean{TNLean.PEPS.Approximation.BlockNear, TNLean.PEPS.Approximation.CornerNear, TNLean.PEPS.Approximation.DyadicSquare.Near, TNLean.PEPS.Approximation.DyadicJunction.Near}

\end{definition}

    \leanok

    For every depth $k$, every block and every constant $c$, at most $2(4c+4)^2$
    squares and at most $2(4c+4)^2$ junctions of the hierarchy are near the block.
    Consequently, if every repainting of a square and every junction resize consists
    of at most $G$ gates, a party all of whose gates belong to repaintings and
    junction resizes near its block participates in at most $4G(4c+4)^2$ gates.
    This is the final count of the bounded lifetime participation
    in~\cite[Proposition~7.1, item~2]{OpenAI2026PolynomialPEPS} (lines 611--618):
    a party of side $m$ occurs in linked gates only at levels $m$ and $m/2$, and at
    these scales there are only a fixed number of repaintings or junctions within
    distance $O(m)$ of its block. That a block party occurs only in such gates is
    the hypothesis here; the gates of the protocol are not constructed.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_dyadic_address_locality}
    In one coordinate, with $b=2^{j'}\ge2^{j-1}$ and $g\le c\,2^j$, the blocks of
    side $b$ meeting $[2^jr-g,2^j(r+1)+g]$ number at most
    $\lfloor U/b\rfloor+2-\lfloor V/b\rfloor$ by
    Lemma~\ref{lem:peps_dyadic_address_locality}, and
    $b\lfloor U/b\rfloor\le U\le V+2^j+2g<b(\lfloor V/b\rfloor+1)+b(4c+2)$ bounds
    this by $4c+4$. A junction coordinate within distance $g$ of the interval lies
    in such a block. There are at most two scales $j'\in\{j-1,j\}$, so the near
    squares and the near junctions number at most $2(4c+4)^2$ each. Finally, the
    gates of the party are distributed over the at most $4(4c+4)^2$ near
    repaintings and junction resizes, at most $G$ to each.
\end{proof}

\begin{definition}[Parties used by a repainting or a resize]
    \label{def:peps_dyadic_used_parties}
    \lean{TNLean.PEPS.Approximation.OwnsLevelBlock, TNLean.PEPS.Approximation.RepaintingUses, TNLean.PEPS.Approximation.JunctionUses, TNLean.PEPS.Approximation.DyadicSquare.Uses, TNLean.PEPS.Approximation.DyadicJunction.Uses, TNLean.PEPS.Approximation.oldParty, TNLean.PEPS.Approximation.newParty}

    $2^{j'}$ the block of side $2^{j'}$ with indices $(q,t)$ is owned by its own party
    $(j',q,t)$ and by the party $(j'+1,\lfloor q/2\rfloor,\lfloor t/2\rfloor)$ of its
    parent; these are its new and its old label. The repainting of the square of
    scale $j'$ with indices $(r',s')$ uses the owners of the blocks whose indices

    corner $(2^ju,2^jv)$ uses the parties of the four blocks of side $2^j$ at the
    corner.
\end{definition}

\begin{theorem}[Active labels in party form]
    \label{thm:peps_dyadic_active_label_parties}
    \lean{TNLean.PEPS.Approximation.levelLabels_eq_party, TNLean.PEPS.Approximation.OwnsLevelBlock.scale, TNLean.PEPS.Approximation.levelLabels_empty_eq_party, TNLean.PEPS.Approximation.levelLabels_eq_party_of_subset, TNLean.PEPS.Approximation.repaintingUses_of_levelLabels, TNLean.PEPS.Approximation.incidentLabels_blockGuide_corner, TNLean.PEPS.Approximation.junctionUses_of_incidentLabels}
    \leanok
    \uses{def:peps_dyadic_used_parties, def:peps_dyadic_block_guide}
    Label the blocks by an injective assignment of parties, none of them the
    placeholder. At the start and end of each repainting of level $2^{j'}$, when
    every block carries its old or its new label, a block party that labels a block
    of side $2^{j'}$ labels a block of the root square and owns it; in particular it
    has scale $j'$ or $j'+1$. At the start of the level only parties of scale $j'+1$
    occur, and once every block of the root square is repainted only parties of scale
    $j'$. At these times the block party labelling a block adjacent to a square, or
    the square itself, is used by the repainting of that square; this covers the
    labels $A$, $B$, $C_e$ of its edge constructions and the labels at its corners.
    At the end of the pass to level $2^j$, the block parties incident at a grid
    corner of the block guide, among them its tag holder, are used by the resize of
    the hole there. This is the invariant on active labels
    in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 597--609, at the transition
    times between repaintings, where the guide is a block guide, together with the
    block-guide part of lines 585--594. The labels and tags inside a repainting (the
    edge and point diagrams, the auxiliary initial owners, the tag updates by corner
    rewrites) are not covered, because the repainting gates are not constructed.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_dyadic_active_labels}
    A block label of the level is the placeholder, the new label of the block, or
    the old label of its parent, and the new labels do not occur at the start nor
    the old ones at the end of the level; injectivity of the party assignment
    identifies the party. Near a grid corner $(nu,nv)$, within distance $n/2$, every
    point lies in one of the four blocks with indices $u-1$ or $u$ and $v-1$ or $v$,
    so every incident label is one of their labels.
\end{proof}

\begin{theorem}[Lifetime participation of block parties]
    \label{thm:peps_dyadic_lifetime_parties}
    \lean{TNLean.PEPS.Approximation.RepaintingUses.near, TNLean.PEPS.Approximation.JunctionUses.near, TNLean.PEPS.Approximation.card_gates_le_of_uses}
    \leanok
    \uses{def:peps_dyadic_used_parties, def:peps_dyadic_near_block}
    A block party used by a repainting or by a junction resize has the scale of the
    event or the next larger one, and the event is near its block with constant
    $c=0$. Consequently, if every repainting and every junction resize consists of at
    most $G$ gates and each gate involves only parties its event uses, every block
    party participates in at most $64G$ gates. This is the bounded lifetime
    participation for block parties in~\cite[Proposition~7.1,
        item~2]{OpenAI2026PolynomialPEPS}, lines 597--618, under the stated hypothesis on
    the gates; the gates of the protocol are not constructed.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_lifetime_incidence}
    A party of side $2^j$ owning a block adjacent to a square of scale $j$ has index
    within one of the square's in each coordinate, so the closed intervals touch.
    For a square of scale $j-1$, a block adjacent to it has a parent with index
    $\lfloor q/2\rfloor$, and $2\lfloor q/2\rfloor\le q\le 2\lfloor q/2\rfloor+1$
    gives touching intervals again. A corner of a block lies in its closed interval.
    Theorem~\ref{thm:peps_dyadic_lifetime_incidence} with $c=0$ bounds the number of
    gates by $4G\cdot4^2=64G$.
\end{proof}

\begin{definition}[Events and changes of the protocol schedule]
    \label{def:peps_dyadic_protocol_schedule}
    \lean{TNLean.PEPS.Approximation.ProtocolEvent, TNLean.PEPS.Approximation.ProtocolEvent.level, TNLean.PEPS.Approximation.ProtocolEvent.IsScheduled, TNLean.PEPS.Approximation.ProtocolEvent.rank, TNLean.PEPS.Approximation.eventSchedule, TNLean.PEPS.Approximation.BulkOp, TNLean.PEPS.Approximation.BulkOp.marks, TNLean.PEPS.Approximation.BulkOp.sheets, TNLean.PEPS.Approximation.RepaintOp, TNLean.PEPS.Approximation.BulkOp.treated, TNLean.PEPS.Approximation.BulkOp.ops, TNLean.PEPS.Approximation.SquareEdge.list, TNLean.PEPS.Approximation.edgeOps, TNLean.PEPS.Approximation.largeRepaintingOps, TNLean.PEPS.Approximation.eventOps, TNLean.PEPS.Approximation.protocolSchedule}
    \leanok
    \uses{def:peps_dyadic_hierarchy}
    An event of the protocol on the hierarchy of $[0,2^k]^2$ is the repainting of a
    square of scale $j<k$, performed at level $2^j$, or the resize of the hole at a
    grid corner of scale $j+1\ge1$, performed on entering level $2^j$. The events are
    performed by descending level, and within a level the resizes precede the
    repaintings; the order of the blocks within a level is arbitrary. A resize and a
    direct repainting are one change each. A large-scale repainting consists of the
    central birth and then, along each of the four edges, the initialization of a fresh
    auxiliary sheet, three auxiliary births, the lens exchange and four main births or
    deaths. Each of these bulk operations is preceded by the homogenization of each
    involved sheet at each of its marks, one mark and one sheet at a time, and followed
    by the removal of these homogenizations; the central birth has four marks and the
    edge operations two, and the exchange involves two sheets and the others one. This
    is the schedule of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}
    (\texttt{06-geometry.tex}, lines 82--100, 127--131, 142--221, 328--339 and
    507--541), with the set of squares repainted by the large-scale construction left
    as a parameter.
\end{definition}

\begin{theorem}[Counts of the protocol schedule]
    \label{thm:peps_dyadic_protocol_schedule_counts}
    \lean{TNLean.PEPS.Approximation.mem_eventSchedule, TNLean.PEPS.Approximation.nodup_eventSchedule, TNLean.PEPS.Approximation.pairwise_rank_eventSchedule, TNLean.PEPS.Approximation.length_eventSchedule_le, TNLean.PEPS.Approximation.BulkOp.length_ops, TNLean.PEPS.Approximation.length_edgeOps, TNLean.PEPS.Approximation.length_largeRepaintingOps, TNLean.PEPS.Approximation.length_eventOps_le, TNLean.PEPS.Approximation.length_protocolSchedule_le, TNLean.PEPS.Approximation.length_protocolSchedule_le_cube, TNLean.PEPS.Approximation.card_filter_event_le, TNLean.PEPS.Approximation.RepaintOp.IsAuxInit, TNLean.PEPS.Approximation.BulkOp.countP_isAuxInit_ops, TNLean.PEPS.Approximation.countP_isAuxInit_edgeOps, TNLean.PEPS.Approximation.countP_isAuxInit_eventOps_le, TNLean.PEPS.Approximation.countP_isAuxInit_protocolSchedule_le}
    \leanok
    \uses{def:peps_dyadic_protocol_schedule, thm:peps_dyadic_hierarchy_counts}
    Every scheduled event, that is every repainting of a non-root square and every
    resize at a junction of positive scale, occurs exactly once in the schedule, in
    the order of the definition, and no other event occurs. A
    bulk operation with $m$ marks on $\sigma$ sheets has $2m\sigma+1$ changes, an edge
    has $45$, and a large-scale repainting $189$; so every event has at most $189$
    changes, at most $189$ changes of the schedule belong to any one event, and if
    $2^k<2L$ the schedule has at most $5040L^3$ changes. Every event initializes at
    most four auxiliary sheets, so the schedule initializes at most four times as many
    auxiliary sheets as there are squares and junctions. This is the bound on the
    number of changes in~\cite[Proposition~7.1, item~2]{OpenAI2026PolynomialPEPS} for
    this schedule.
\end{theorem}

\begin{proof}\leanok
    The schedule sorts the events by their position in the level order, so it contains
    each event once. The schedule of changes lists the changes of each event in turn,
    so its length is at most $189$ times the number of squares and junctions, and
    Theorem~\ref{thm:peps_dyadic_hierarchy_counts} with $M=189$ gives
    $3\cdot189\,(\#\text{squares}+\#\text{junctions})\le80\cdot189\,L^3$.
\end{proof}

\begin{definition}[Parties of the protocol and their labels]

    additional party that represents the exterior label during that repainting. Every
    party is attached to its square: a placeholder has the scale and address of the
    square whose repainting it serves. The guides carry the block labels $(j,r,s)$ and
    one placeholder label. During the repainting of $S$, the label of a square of the
    hierarchy is held by its block party and the placeholder label by the placeholder

    use its block (Definition~\ref{def:peps_dyadic_used_parties}), and the placeholder
    party of $S$ only by the repainting of $S$. This follows~\cite[Proposition~7.1,

\begin{theorem}[Parties of the protocol]
    \label{thm:peps_dyadic_protocol_parties}
    \lean{TNLean.PEPS.Approximation.DyadicSquare.toTriple_injective, TNLean.PEPS.Approximation.DyadicSquare.ofTriple_toTriple, TNLean.PEPS.Approximation.protocolPlacement_inGrid, TNLean.PEPS.Approximation.protocolPlacement_blockBounded, TNLean.PEPS.Approximation.card_protocolParty_le, TNLean.PEPS.Approximation.three_mul_card_protocolParty_lt, TNLean.PEPS.Approximation.inl_ne_guidePlaceholder, TNLean.PEPS.Approximation.labelParty_toTriple, TNLean.PEPS.Approximation.labelParty_guidePlaceholder, TNLean.PEPS.Approximation.labelParty_injOn, TNLean.PEPS.Approximation.levelLabels_mem_validLabels}
    \leanok

        lem:peps_dyadic_active_labels}
    Every party lies in the padded square $[0,2^k]^2$, at most two parties are attached
    to any square, and there are at most twice as many parties as squares; if
    $2^k<2L$ there are fewer than $32L^2/3$ parties. During a repainting, distinct
    labels of squares of the hierarchy and the placeholder label are held by distinct
    parties. At every level $2^j$ with $j<k$, every block label of the level is the
    placeholder label or the label of a square of the hierarchy. These are the counts
    of~\cite[Proposition~7.1, item~1]{OpenAI2026PolynomialPEPS} for these parties.
\end{theorem}

\begin{proof}\leanok
    A block of side $2^j$ with index below $2^{k-j}$ ends before $2^j\cdot2^{k-j}=2^k$.
    The parties at a square are its block party and its placeholder. The count follows
    from Theorem~\ref{thm:peps_dyadic_hierarchy_counts}. A block of the root square at
    level $2^j$ has indices below $2^{k-j}$, and its parent has scale $j+1\le k$ and
    indices below $2^{k-j-1}$.
\end{proof}

\begin{theorem}[Locality of the parties of an event]
    \label{thm:peps_dyadic_protocol_locality}
    \lean{TNLean.PEPS.Approximation.EventUses.scale_le, TNLean.PEPS.Approximation.EventUses.anchor_dist_le}
    \leanok
    \uses{def:peps_dyadic_protocol_parties, lem:peps_dyadic_address_locality}
    Two parties used by one event have equal or adjacent scales, and in each coordinate
    their addresses differ by at most $12$ times the side of the square of either
    party. This is the locality of~\cite[Proposition~7.1,
        item~3]{OpenAI2026PolynomialPEPS} for the parties an event uses.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_lifetime_parties}
    A party used by the repainting of a square $S$ of scale $j$ owns a block of side
    $2^j$ adjacent to $S$ or its parent, or is the placeholder of $S$; by
    Lemma~\ref{lem:peps_dyadic_address_locality} its address is within $3\cdot2^{j+1}$
    of the address of $S$ in each coordinate, and its scale is $j$ or $j+1$. A party
    used by the resize at a junction of scale $j$ owns one of the four blocks at the
    corner, so its address is within $3\cdot2^{j+1}$ of the address of a fixed block
    there. The triangle inequality gives $12\cdot2^j$.
\end{proof}

\begin{definition}[Patch data of a guide rewrite]
    \label{def:peps_dyadic_guide_patch_rewrite}
    \lean{TNLean.PEPS.Approximation.GuidePatchRewrite, TNLean.PEPS.Approximation.GuidePatchRewrite.footprint, TNLean.PEPS.Approximation.GuidePatchRewrite.parties, TNLean.PEPS.Approximation.GuidePatchRewrite.IsJunctionResize, TNLean.PEPS.Approximation.GuidePatchRewrite.IsDirectRepainting}
    \leanok
    \uses{def:peps_dyadic_block_guide, def:peps_dyadic_incident_labels}
    The patch data of a candidate small-patch rewrite of one sheet consists of its
    old and new guide, the centres of finitely many patches with a common inner
    radius $u$, and the centres of its affected old and new holes with one outer
    radius covering each of them. Its footprint is the union of the outer patch
    squares, of radius $2u$, and of the outer squares of the affected holes. Its
    parties are the labels of the old and of the new guide on the footprint, and the
    tag holders of both guides at the affected holes; these are the raw owners and
    the affected tag owners named in condition~4 of the small-patch rewrite lemma
    in~\cite[Section~6]{OpenAI2026PolynomialPEPS}. Conditions~1--3 of that lemma,
    the two-owner tests outside the old and new inner holes and the avoidance of the
    untouched hole footprints, are not part of the patch data.

    The patch data is that of the resize at the grid corner $(2^ju,2^jv)$ when both
    guides are the block guide uniform on blocks of side $2^j$ at the end of the pass
    to level $2^j$, its only affected hole is at the corner, and its footprint lies
    within sup distance $2^j$ of the corner. It is that of the direct repainting of
    the square $S$ with indices $(r,s)$ and side $2^{j'}$ when its old guide is the block
    guide of level $2^{j'}$ at a transition time, its new guide is the one in which
    $S$ also carries its new label, its affected holes lie at corners of $S$, and its
    footprint lies within sup distance $2^{j'}$ of the closed square $S$.
\end{definition}

\begin{theorem}[Block indices near a block or a corner]
    \label{thm:peps_dyadic_block_index_near}
    \lean{TNLean.PEPS.Approximation.floor_div_mem_Icc, TNLean.PEPS.Approximation.blockIndex_of_dist_blockCorner_lt, TNLean.PEPS.Approximation.abs_blockIndex_sub_le}
    \leanok
    \uses{def:peps_dyadic_block_guide}

    a block with indices $u-1$ or $u$ and $v-1$ or $v$. A point within sup distance
    $n$ of the closed block with indices $(r,s)$ lies in a block whose indices differ
    from $(r,s)$ by at most one in each coordinate.
\end{theorem}

\begin{proof}\leanok
    If $na-n<y<nb+n$ then $a-1<y/n<b+1$, so $a-1\le\lfloor y/n\rfloor\le b$. Apply
    this to each coordinate, with $a=b=u$ near a corner and with $a=r$, $b=r+1$ near
    the closed block.
\end{proof}

\begin{theorem}[Parties of the junction resizes and the direct repaintings]
    \label{thm:peps_dyadic_patch_rewrite_parties}
    \lean{TNLean.PEPS.Approximation.GuidePatchRewrite.parties_subset, TNLean.PEPS.Approximation.junctionUses_of_blockGuide_eq, TNLean.PEPS.Approximation.repaintingUses_of_blockGuide_eq, TNLean.PEPS.Approximation.GuidePatchRewrite.IsJunctionResize.uses, TNLean.PEPS.Approximation.GuidePatchRewrite.IsDirectRepainting.uses}
    \leanok
    \uses{def:peps_dyadic_guide_patch_rewrite, def:peps_dyadic_used_parties}
    Label the blocks by an injective assignment of parties, none of them the
    placeholder. If the outer radius of the affected holes is positive, the parties
    of the patch data are labels of its old or new guide on its footprint. Every
    block party of the patch data of the resize at a grid corner is one of the four
    block parties at the corner, and every block party of the patch data of the
    direct repainting of a square is used by the repainting of that square. This is
    ``Resizing uses the current block labels near its junction'' and the locality of the direct repaintings
    in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 500--541 and 585--594.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_block_index_near, thm:peps_dyadic_active_label_parties}
    An outer hole square of positive radius is a neighbourhood of its centre, so an
    incident label there, in particular a tag holder, is a label on the footprint. A
    point of the footprint of a resize lies in one of the four blocks at the corner,
    and a point of the footprint of a direct repainting lies in a block adjacent to
    the square or in the square itself, by
    Theorem~\ref{thm:peps_dyadic_block_index_near}. The party labelling such a block
    is identified by Theorem~\ref{thm:peps_dyadic_active_label_parties}; at the end of
    a pass only the parties of the blocks themselves occur.
\end{proof}

\begin{theorem}[Patch nets of the junction resizes and the direct repaintings]
    \label{thm:peps_dyadic_patch_rewrite_nets}
    \lean{TNLean.PEPS.Approximation.exists_junctionResize, TNLean.PEPS.Approximation.exists_directRepainting}
    \leanok
    \uses{def:peps_dyadic_guide_patch_rewrite}
    Let $\epsilon_0,\nu,t>0$ with $2\epsilon_0+3\nu<1$. The resize at the grid corner
    $c=(2^ju,2^jv)$ of the block guide at the end of the pass to level $2^j$, from
    the hole radius $\epsilon_0\min(2^j,t)$ to $\epsilon_0\min(2^j/2,t)$, has
    patch data whose patches form a square net of mesh $u=\nu2^j$. There are at
    most $(2(2\epsilon_0+\nu)/\nu+1)^2$ patches, and their inner squares cover both
    outer squares of the hole. The direct repainting of a square $S$ of side
    $n=2^{j'}$ at a transition time of level $n$, with affected holes at the four
    corners of $S$ and hole radius $\epsilon_0\min(n,t)$, has patch data whose
    patches form a square net of mesh $\nu n$. There are at most
    $(2(1/2+2\epsilon_0+\nu)/\nu+1)^2$ patches, and their inner squares cover the
    closed square $S$ and the outer squares of the four holes. Both counts are
    independent of the scale. These are the fine nets of the direct repainting and
    of the hole resizing in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines
    500--541. Conditions~1--3 of the small-patch rewrite lemma are not verified for
    these nets. The source repaints directly only for $n\le K_0t$; that restriction
    enters the two-owner condition on the patches, not the nets or their parties.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_dyadic_nets, lem:peps_dyadic_small_radius,
        lem:peps_dyadic_direct_repainting}
    The hole radius at scale $m$ is at most $\epsilon_0m$, and the new radius of a
    resize is at most the old one. The set to be covered lies in the closed square of
    radius $2\epsilon_0\,2^j$ about the corner, or of radius $(1/2+2\epsilon_0)n$
    about the centre of $S$, and Lemma~\ref{lem:peps_dyadic_nets} gives a net of mesh
    $u$ with the stated number of patches and with centres within $u$ of the covered
    set. An outer patch square is then within $3u$ of a point of the covered set, and
    every point of the covered set is within $2\epsilon_0m$ of the corner or of the
    closed square $S$. Since $2\epsilon_0+3\nu<1$, the footprint lies within sup
    distance $2^j$ of the corner, or $n$ of the closed square $S$.
\end{proof}

\begin{theorem}[Lifetime participation for small-patch rewrites]
    \label{thm:peps_dyadic_patch_rewrite_lifetime}
    \lean{TNLean.PEPS.Approximation.card_gates_le_of_guidePatchRewrites}
    \leanok
    \uses{thm:peps_dyadic_patch_rewrite_parties}
    Let the gates of a schedule on the hierarchy belong to repaintings and junction
    resizes, at most $G$ to each event, and let each gate carry the patch data of its
    event: a direct repainting of its square or the resize at its junction, at a
    level $j$ whose root square $[0,2^k]^2$ has $2^{k-j}$ blocks per side. If a block
    party participates in a gate only when it is a party of the gate's patch data, it
    participates in at most $64G$ gates. The bound is conditional on this hypothesis,
    which is the conclusion of the small-patch rewrite lemma and is not derived here
    from it. This is a conditional form of the bounded lifetime participation
    of~\cite[Proposition~7.1, item~2]{OpenAI2026PolynomialPEPS} for schedules
    consisting of direct repaintings and junction resizes; the large-scale
    repaintings, with their births, deaths, exchanges and point treatments, are not
    covered.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_lifetime_parties}
    By Theorem~\ref{thm:peps_dyadic_patch_rewrite_parties} every gate in which the
    party participates is one of an event that uses the party, and
    Theorem~\ref{thm:peps_dyadic_lifetime_parties} bounds the number of such gates by
    $64G$.
\end{proof}

\begin{definition}[Local guides of a large-scale repainting]
    \label{def:peps_dyadic_local_guides}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline.localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide, TNLean.PEPS.Approximation.levelBaseline, TNLean.PEPS.Approximation.GuidePatchRewrite.IsLargeRepaintingStep}
    \leanok
    \uses{def:peps_dyadic_schedule, def:peps_dyadic_guide_patch_rewrite}
    Let a repainting of $S=[0,n]^2$ start from the guide $g$ with final label $B$.
    Its local labels are $B$ and the values of $g$ within sup distance $n$ of the
    closed square. A guide is local over a base guide, the starting guide $g$ of the
    main sheet or the uniform guide $C_e$ of an auxiliary sheet, if it takes only
    local labels within sup distance $n$ of the closed square and agrees with the
    base guide outside the open neighbourhood of radius $n/2$ of the closed square.
    For the repainting of the square with indices $(r,s)$ and side $2^{j'}$ during the
    pass to level $2^{j'}$, the starting guide is the block guide of the level at a
    transition time, moved to the origin, with final label the new party of the
    square. Patch data is that of a gate of this large-scale repainting when its old
    and new guides are local guides over one base, moved back to the square, and its
    footprint lies within sup distance $2^{j'}$ of the closed square.
\end{definition}

\begin{theorem}[The guides of a repainting are local]
    \label{thm:peps_dyadic_local_guides}
    \lean{TNLean.PEPS.Approximation.SquareEdge.point_mem_closedSquare, TNLean.PEPS.Approximation.edgeBand_subset_thickening, TNLean.PEPS.Approximation.edgeTreated_subset_thickening, TNLean.PEPS.Approximation.cornerTreated_subset_thickening, TNLean.PEPS.Approximation.edgeLens_subset_thickening, TNLean.PEPS.Approximation.RepaintingBaseline.finalLabel_mem_localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.guide_mem_localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.oldLabel_mem_localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.nbrLabel_mem_localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.edgeWords_mem_localLabels, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_guide, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_const, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide.update, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide.homogenize, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide.piecewise, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide.bandUpdate, TNLean.PEPS.Approximation.RepaintingBaseline.IsLocalGuide.changed_subset, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_centralGuide, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_completedGuide, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_mainGuide, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_auxGuide, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_treatedStartMain, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_treatedAux, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_exchangedMain, TNLean.PEPS.Approximation.RepaintingBaseline.isLocalGuide_exchangedAux}
    \leanok
    \uses{def:peps_dyadic_local_guides, lem:peps_dyadic_floor}
    The labels $A$, $B$ and $C_e$ are local labels, and so is every value of a normal
    word of an edge construction. The starting guide and the guide after the central
    birth, every completed guide, and every main guide of an edge construction with
    any of its normal words are local over the starting guide; the uniform guide
    $C_e$ and every auxiliary guide of the construction along $e$ with any of its
    normal words are local over the uniform guide $C_e$. Homogenizing a local guide
    to a local label on a set inside the neighbourhood of radius $n/2$, as the point
    treatment does on the treated squares of radius $t\le n/2$, and replacing it on a
    set inside that neighbourhood by a local guide over another base give local
    guides over the same base. The lens lies in that neighbourhood, so after the lens
    exchange along $e$, for $t\le n/2$, the main sheet is local over the starting
    guide and the auxiliary sheet is local over the uniform guide $C_e$. Two local guides over one base differ only
    inside the neighbourhood of radius $n/2$. This is the locality of the large-scale
    repaintings in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 585--596: ``An
    auxiliary uses only the labels of the local edge and point diagrams; its uniform
    initial owner $C$ is the actual neighboring-block owner.''
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_dyadic_schedule_guides, lem:peps_dyadic_active_labels}
    A point of the band $\{-8<x<2\}$ of an edge lies in the closed square when
    $x\ge0$, and otherwise within $8w(s)\le n/250$ of the edge. The treated squares of
    radius $t\le n/2$ are centred at corners of the square, and the central region
    lies in the open square. The labels $A$ and $C_e$ are values of the starting
    guide at the centre of the square and at distance $n/2$ across the edge $e$. Each
    guide of the schedule is obtained from the starting guide or from the uniform
    guide $C_e$ by finitely many changes, on the central region or on a band, to
    labels among $A$, $B$, $C_e$, and a change inside the neighbourhood of radius
    $n/2$ to local labels preserves locality. The lens $\{-7<x<-7/2\}$ lies in the
    band, and on it each exchanged sheet takes the values of the other sheet, which
    are local labels.
\end{proof}

\begin{theorem}[Parties of the large-scale repaintings]
    \label{thm:peps_dyadic_large_repainting_parties}
    \lean{TNLean.PEPS.Approximation.repaintingUses_of_mem_localLabels, TNLean.PEPS.Approximation.GuidePatchRewrite.IsLargeRepaintingStep.uses}
    \leanok
    \uses{def:peps_dyadic_local_guides, def:peps_dyadic_used_parties}
    Label the blocks by an injective assignment of parties, none of them the
    placeholder. In the repainting of the square with indices $(r,s)$ and side
    $2^{j'}$, every local label that is a block party is used by the repainting, and
    so is every block party of the patch data of a gate of the large-scale
    repainting. This is ``every active main owner is an $n$- or $2n$-block owner. On
    the affected footprints those blocks lie at distance $O(n)$ from the repainted
    square'' in~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 585--596.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_patch_rewrite_parties, thm:peps_dyadic_local_guides}
    The final label is the new party of the square. A value of the starting guide
    within sup distance $2^{j'}$ of the closed square is the label of a block whose
    indices differ from $(r,s)$ by at most one in each coordinate, and such a label
    is used by the repainting by Theorem~\ref{thm:peps_dyadic_patch_rewrite_parties}.
    A party of the patch data is a label of its old or new guide on its footprint,
    hence a local label.
\end{proof}

\begin{theorem}[Patch nets of the large-scale repaintings]
    \label{thm:peps_dyadic_large_repainting_nets}
    \lean{TNLean.PEPS.Approximation.exists_largeRepaintingStep_of_cover, TNLean.PEPS.Approximation.exists_pointTreatment, TNLean.PEPS.Approximation.exists_bulkStep}
    \leanok
    \uses{def:peps_dyadic_local_guides}
    Let $n=2^{j'}$ and consider two local guides over one base of the repainting of
    a square of side $n$. If $0<t$, $2t\le n$, $\epsilon_0,\nu>0$,
    $2\epsilon_0+3\nu<1$ and $c$ is a point of the closed square, a homogenization or
    removal of a point treatment about $c$ with affected holes centred within $t$ of
    $c$ and of outer radius $2\epsilon_0t$ has patch data of a gate of the
    large-scale repainting whose patches form a square net of mesh $\nu t$. There are
    at most $(2(1+2\epsilon_0+\nu)/\nu+1)^2$ patches, and their inner squares cover
    the treated square of radius $t$ about $c$ and the affected outer holes. If
    $0<\nu\le1/12$ and $0<R\le n/4$, a bulk birth, death or exchange with affected
    holes of outer radius $R$ centred within $n/2$ of the square has patch data of a
    gate of the large-scale repainting whose patches form a square net of mesh
    $\nu n$, at most $(2(5/4+\nu)/\nu+1)^2$ of them, covering the changed set and the
    affected outer holes. The point-treatment nets are those
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 482--505. The bulk step is
    implemented by the birth or exchange lemma, not by its net; the net bounds its
    parties, the owners on the changed set ``compactly located in diameter $O(n)$''
    and on the affected holes, lines 383--387 and 421--430.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_dyadic_nets, thm:peps_dyadic_local_guides}
    Lemma~\ref{lem:peps_dyadic_nets} gives a net of mesh $u$ over the set to be
    covered, with centres within $u$ of it, so an outer patch square lies within $3u$
    of the covered set. For a point treatment the covered set lies within
    $(1+2\epsilon_0)t$ of $c$, and $(1+2\epsilon_0+3\nu)t<2t\le n$. For a bulk step
    the covered set is the neighbourhood of radius $3n/4$ of the square, which
    contains the changed set by Theorem~\ref{thm:peps_dyadic_local_guides} and the
    affected outer holes, and $3n/4+3\nu n\le n$.
\end{proof}

\begin{theorem}[Lifetime participation with large-scale repaintings]
    \label{thm:peps_dyadic_large_repainting_lifetime}
    \lean{TNLean.PEPS.Approximation.card_gates_le_of_repaintingRewrites}
    \leanok
    \uses{def:peps_dyadic_local_guides, def:peps_dyadic_guide_patch_rewrite,
        def:peps_dyadic_used_parties}
    Let the gates of a schedule on the hierarchy belong to repaintings and junction
    resizes, at most $G$ to each event, and let each gate carry patch data: for a
    repainting, the direct repainting of its square or a gate of its large-scale
    repainting; for a junction, the resize of its hole. If a block party
    participates in a gate only when it is a party of the gate's patch data, it
    participates in at most $64G$ gates. The bound is conditional on this
    hypothesis, the conclusion of the small-patch rewrite lemma and of the birth and
    exchange lemmas for the parties of the patch data. This is a conditional form of
    the bounded lifetime participation
    of~\cite[Proposition~7.1, item~2]{OpenAI2026PolynomialPEPS}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_patch_rewrite_parties, thm:peps_dyadic_large_repainting_parties,
        thm:peps_dyadic_lifetime_parties}
    By Theorems~\ref{thm:peps_dyadic_patch_rewrite_parties}
    and~\ref{thm:peps_dyadic_large_repainting_parties} every gate in which the party
    participates is one of an event that uses the party, and
    Theorem~\ref{thm:peps_dyadic_lifetime_parties} bounds the number of such gates by
    $64G$.
\end{proof}

\begin{theorem}[Lifetime participation of the protocol parties]
    \label{thm:peps_dyadic_protocol_lifetime}
    \lean{TNLean.PEPS.Approximation.participation_le_of_eventUses, TNLean.PEPS.Approximation.participation_le_of_repaintingRewrites, TNLean.PEPS.Approximation.card_gates_event_le, TNLean.PEPS.Approximation.participation_le_of_schedule}
    \leanok
    \uses{def:peps_dyadic_protocol_parties, def:peps_dyadic_guide_patch_rewrite,
        def:peps_dyadic_local_guides}
    Let the gates of a schedule belong to events, at most $G$ gates to each. If every
    gate in which a party participates belongs to an event that uses the party, then the
    party participates in at most $64G$ gates. In particular this holds when each gate
    carries the patch data of its event, a block party participates in a gate only when
    its label is a party of the gate's patch data, and a placeholder party only in
    gates of its own repainting. The bound is conditional on this hypothesis on the
    gates; it is a conditional form of the bounded lifetime participation
    of~\cite[Proposition~7.1, item~2]{OpenAI2026PolynomialPEPS}. If the gates belong
    to the changes of the protocol schedule, at most $b$ to each change, then at most
    $189b$ belong to each event, and the bound is $64\cdot189b$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_dyadic_lifetime_parties, thm:peps_dyadic_large_repainting_lifetime,
        thm:peps_dyadic_protocol_schedule_counts}
    For a block party this is Theorem~\ref{thm:peps_dyadic_lifetime_parties}, or
    Theorem~\ref{thm:peps_dyadic_large_repainting_lifetime} for the patch data. A
    placeholder party participates only in gates of one event, at most $G\le64G$.
    Along the schedule, Theorem~\ref{thm:peps_dyadic_protocol_schedule_counts} gives at
    most $189$ changes for each event.
\end{proof}
