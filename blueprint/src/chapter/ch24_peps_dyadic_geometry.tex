\section{Geometry of the dyadic distribution protocol}
\label{sec:peps_dyadic_geometry}
This section treats the geometric part of~\cite[Section~7]{OpenAI2026PolynomialPEPS},
September 24, 2026, \texttt{06-geometry.tex}: the bands along the edges of a
repainted dyadic square, the angular separation of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}
(lines 254--319), the point-treatment estimates (lines 357--415), the absence of
changed true vertices in the bulk operations and the tags (lines 375--398 and
610--631), and parts of the geometry of the small-patch covers, of the direct
repaintings and of the hole resizings (lines 341--351 and 447--533), and the counts of the padded
dyadic hierarchy behind items 1--3
of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} (lines 12--53 and 633--652).
The protocol of Proposition~7.1 itself, whose changes are implemented by the
information estimate and the contraction gates of the earlier sections, is not
treated here.

The clearance estimates are first proved for every guide with the properties that
the proof of Lemma~7.2 uses: the normal word of the operation on the band
$\{-8<x<2\}$ and no change off it, the label $A$ on the open square, and the label
$C$ on the exterior band $\{-8<x<0\}$. The schedule of guides is then constructed,
these properties are derived from the disjointness of the bands, and the estimates
are moved to every square of the hierarchy. This gives
Theorem~\ref{thm:peps_dyadic_angular_separation}, which is Lemma~7.2. A guide of the
source is a partition of the plane into open chambers, with labels on boundaries
sampled after one small generic displacement; here a guide is a labelling of the
whole plane, and two labellings that agree off finitely many interface curves
represent the same guide. The earlier scope restriction and this convention are
recorded in~\cite{gap:openai26_dyadic_geometry_guide_properties}. Distances are sup
distances in the plane, and labels are elements of an arbitrary set; the labels
$A$, $B$, $C$ below are permitted to coincide.

\begin{definition}[Edge coordinates, bands and the lens]
    \label{def:peps_dyadic_edge_band}
    \lean{TNLean.PEPS.Approximation.SquareEdge, TNLean.PEPS.Approximation.bandWidth, TNLean.PEPS.Approximation.edgeBand, TNLean.PEPS.Approximation.closedEdgeBand, TNLean.PEPS.Approximation.bandCoord, TNLean.PEPS.Approximation.edgeLens, TNLean.PEPS.Approximation.edgeNeighbor, TNLean.PEPS.Approximation.edgeMarkDist, TNLean.PEPS.Approximation.cornerMarkDist}
    \leanok
    Let $S=[0,n]^2$. For an edge of $S$ let $s$ be the coordinate parallel to the
    edge, measured from one endpoint, and $d$ the normal coordinate, positive
    into $S$. Put
    \begin{align}
        w(s)=10^{-3}\min(s,n-s),\quad x=d/w(s).
        \label{eq:pdg_band}
    \end{align}
    The band $\{\alpha<x<\beta\}$ is the set of points with $0<s<n$ and
    $\alpha w(s)<d<\beta w(s)$; its closed version has $0\le s\le n$ and
    $\alpha w(s)\le d\le\beta w(s)$. The lens of the edge is the band
    $Y=\{-7<x<-7/2\}$, and the exterior neighboring square is
    $\{0<s<n,\ -n<d<0\}$. For a point $y$, $d_{\mathcal V}(y)$ is its distance from
    the two endpoints of the edge, or from the four corners of $S$.
\end{definition}

\begin{lemma}[Disjointness of the edge bands]
    \label{lem:peps_dyadic_band_disjoint}
    \lean{TNLean.PEPS.Approximation.disjoint_edgeBand, TNLean.PEPS.Approximation.edgeBand_neg_subset_edgeNeighbor, TNLean.PEPS.Approximation.dist_le_of_mem_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.changed_subset_closedEdgeBand, TNLean.PEPS.Approximation.BandOperation.dist_le_of_mem_closure_changed, TNLean.PEPS.Approximation.dist_le_of_mem_closure_centralRegion}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    The bands $\{-8<x<2\}$ along distinct edges of $S$ are disjoint, and the band
    $\{-8<x<0\}$ lies in the exterior neighboring square of its edge. Two points
    of a closed band with $-8\le\alpha\le\beta\le8$ are at distance at most $n$.
    Consequently the closure of the changed set of each elementary edge birth or
    death, and the closure of the central region of
    Theorem~\ref{thm:peps_dyadic_central_clearance}, have diameter at most $n$.
\end{lemma}

\begin{proof}\leanok
    Write $w\le s/1000$ and $w\le(n-s)/1000$. For opposite edges the bands lie
    within $n/250$ of their edges. For adjacent edges a common point at distances
    $r_1,r_2>0$ from the two edge lines would satisfy $r_1<0.002\,r_2$ and
    $r_2<0.002\,r_1$. On $\{-8<x<0\}$ one has $0>d>-8w\ge-n/250$. The diameter
    bound follows from $|d|\le8w\le n/250$ in each closed band. The changed set of
    an elementary operation lies in the closed band of its changed interval, and
    the central region lies in $S$.
\end{proof}

\begin{definition}[Normal words of the edge construction]
    \label{def:peps_dyadic_normal_words}
    \lean{TNLean.PEPS.Approximation.bandWord, TNLean.PEPS.Approximation.edgeStartWord, TNLean.PEPS.Approximation.auxWord, TNLean.PEPS.Approximation.mainWordOne, TNLean.PEPS.Approximation.edgeOperations, TNLean.PEPS.Approximation.BandOperation}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    A normal word lists labels in increasing $x$ separated by interfaces; at an
    interface curve the label on its right is taken, whereas the source samples after
    one generic displacement; the two conventions differ only on the interface
    curves. An edge operation starts from the
    main word $C\mid A\mid B$ with interfaces $0,1$. On an auxiliary sheet, uniform
    at $C$, the births of $B$ in $(-6,-1)$, of $A$ in $(-5,-2)$ and of $C$ in
    $(-4,-3)$ give
    \begin{align}
        C\mid B\mid A\mid C\mid A\mid B\mid C\quad\text{at } -6,-5,-4,-3,-2,-1.
        \label{eq:pdg_aux_word}
    \end{align}
    After the lens exchange the main word is $C\mid B\mid A\mid C\mid A\mid B$ at
    $-6,-5,-4,0,1$. The $C$ band $(-4,0)$ then dies into $A$, the $A$ band
    $(-5,1)$ dies into $B$, $C$ is born in $(-3,0)$, and the $B$ band $(-6,-3)$
    dies into $C$, leaving $C\mid B$ at $0$. Each of these seven births and deaths
    changes its word only on a closed interval, and its surrounding word (the word
    before a birth, after a death) carries one label on a strictly larger open
    interval inside $(-8,2)$, with margins at least $1/2$.
\end{definition}

\begin{lemma}[The lens exchange produces the main word]
    \label{lem:peps_dyadic_lens_word}
    \lean{TNLean.PEPS.Approximation.lens_exchange_word}
    \leanok
    \uses{def:peps_dyadic_normal_words}
    Replacing the starting main word by the auxiliary word (\ref{eq:pdg_aux_word})
    on $-7<x<-7/2$ gives the main word $C\mid B\mid A\mid C\mid A\mid B$ at
    $-6,-5,-4,0,1$, at every value of $x$.
\end{lemma}

\begin{proof}\leanok
    Both thresholds $-7$ and $-7/2$ lie in $C$ intervals of both words; compare
    the two words on each interval.
\end{proof}

\begin{lemma}[Conical band estimate]
    \label{lem:peps_dyadic_conical_band}
    \lean{TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt, TNLean.PEPS.Approximation.mem_edgeBand_of_dist_lt_markDist, TNLean.PEPS.Approximation.edgeMarkDist_le_of_mem_closedEdgeBand}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $a_0=1/5000$, $\alpha\le\beta$, $\alpha'+1/2\le\alpha$, $\beta+1/2\le\beta'$
    and $-8\le\alpha'$, $\beta'\le8$. If $y$ lies in the closed band
    $\{\alpha\le x\le\beta\}$ and $|y-z|<a_0\min(n,d_{\mathcal V}(y))$, then $z$
    lies in the open band $\{\alpha'<x<\beta'\}$.
\end{lemma}

\begin{proof}\leanok
    Put $m=\min(s,n-s)$ at $y$, so $w=m/1000$ and $|d|\le m$; hence
    $d_{\mathcal V}(y)\le m$. If $|y-z|<m/5000$, the parallel and normal
    coordinates move by less than $m/5000$, and $w$ is $10^{-3}$-Lipschitz, so
    $|\beta'(w(s_z)-w(s_y))|\le8m/(5\cdot10^{6})$. Then
    $d_z<\beta w(s_y)+m/5000\le\beta' w(s_z)-m/2000+m/5000+8m/(5\cdot10^6)$, and
    symmetrically at $\alpha'$.
\end{proof}

\begin{theorem}[Angular separation of the central birth]
    \label{thm:peps_dyadic_central_clearance}
    \lean{TNLean.PEPS.Approximation.centralRegion, TNLean.PEPS.Approximation.centralBirth_clearance}
    \leanok
    \uses{def:peps_dyadic_edge_band}
    Let $R$ be the set of points of $(0,n)^2$ with $x>1$ for every edge, and let
    $f$ be a labelling of the plane equal to $A$ on $(0,n)^2$. For every $y$ in
    the closure of $R$ and every $z$ with $f(z)\ne A$,
    \begin{align}
        |y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\},
        \label{eq:pdg_central}
    \end{align}
    where $\mathcal V$ is the set of corners of $S$ and $a_0=1/5000$. This is the
    clearance estimate for the central birth in the proof
    of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any surrounding guide equal
    to $A$ on the interior of $S$.
\end{theorem}

\begin{proof}\leanok
    With $u=\min(y_1,n-y_1)$ and $v=\min(y_2,n-y_2)$, the four conditions $d\ge w$
    give $v\ge u/1000$ and $u\ge v/1000$. The nearest corner is at distance
    $\max(u,v)\le1000\min(u,v)$, so a point within $a_0\min(n,d_{\mathcal V}(y))$
    of $y$ is within $\min(u,v)/5$ of $y$ and lies in $(0,n)^2$.
\end{proof}

\begin{theorem}[Angular separation of the edge births and deaths]
    \label{thm:peps_dyadic_band_clearance}
    \lean{TNLean.PEPS.Approximation.BandOperation.clearance, TNLean.PEPS.Approximation.auxBirthB, TNLean.PEPS.Approximation.auxBirthA, TNLean.PEPS.Approximation.auxBirthC, TNLean.PEPS.Approximation.mainDeathC, TNLean.PEPS.Approximation.mainDeathA, TNLean.PEPS.Approximation.mainBirthC, TNLean.PEPS.Approximation.mainDeathB}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    Consider one of the seven elementary births and deaths of an edge operation,
    and labellings $f_b$, $f_a$ and $f_s$ of the plane whose normal words on
    $\{-8<x<2\}$ are its words before, after and surrounding, with $f_b=f_a$ off
    that band. Let $P_\circ$ be its surrounding label. For every $y$ in the closure
    of $\{f_b\ne f_a\}$ and every $z$ with $f_s(z)\ne P_\circ$,
    $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$, with $\mathcal V$ the two endpoints
    of the edge. These are the clearance estimates for the edge births and deaths in
    the proof of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any guides with
    the stated words.
\end{theorem}

\begin{proof}\leanok
    The set $\{f_b\ne f_a\}$ lies in the closed band of the changed interval,
    which is closed. If $|y-z|<a_0\min\{n,d_{\mathcal V}(y)\}$, the conical band
    estimate puts $z$ in the open band of the surrounding sector, where $f_s$
    equals $P_\circ$.
\end{proof}

\begin{theorem}[Angular separation of the lens exchange]
    \label{thm:peps_dyadic_lens_clearance}
    \lean{TNLean.PEPS.Approximation.lensExchange_clearance}
    \leanok
    \uses{lem:peps_dyadic_conical_band, def:peps_dyadic_normal_words}
    Let the main labelling be $C$ on the exterior band $\{-8<x<0\}$, and let the
    auxiliary labelling have the word (\ref{eq:pdg_aux_word}) on $\{-8<x<2\}$ and
    be $C$ off it. Every point $y$ in the closure of the positions that are not
    $C$ on both labellings satisfies
    $|y-z|\ge a_0\min\{n,d_{\mathcal V}(y)\}$ for every $z$ on the boundary of the
    lens $Y$. This is the clearance estimate for the lens exchange in the proof
    of~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}, for any guides with the stated
    properties.
\end{theorem}

\begin{proof}\leanok
    A boundary point of $Y$ is an endpoint of the edge, where the bound is
    immediate, or satisfies $d=-7w$ or $d=-\tfrac72w$ with $0<s<n$. If $y$ lies
    outside the exterior band $\{-8<x<0\}$, then $d_{\mathcal V}(y)\le
        d_{\mathcal V}(z)+|y-z|$ and $d_{\mathcal V}(z)\le m_z=\min(s_z,n-s_z)$; a
    distance below $a_0d_{\mathcal V}(y)$ is then below $m_z/2000$, which keeps
    $y$ in that band because $-7m_z/1000\le d_z\le-\tfrac72m_z/1000$ and $w$ is
    $10^{-3}$-Lipschitz. If $y$ lies in one of
    the closed auxiliary bands $\{-6\le x\le-4\}$ or $\{-3\le x\le-1\}$, apply the
    conical band estimate with the open bands $\{-\tfrac{13}2<x<-\tfrac72\}$ or
    $\{-\tfrac72<x<-\tfrac12\}$, neither of which meets the two boundary curves.
\end{proof}

\begin{definition}[The schedule of guides of a repainting]
    \label{def:peps_dyadic_schedule}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline, TNLean.PEPS.Approximation.openSquare, TNLean.PEPS.Approximation.closedSquare, TNLean.PEPS.Approximation.bandUpdate, TNLean.PEPS.Approximation.RepaintingBaseline.centralGuide, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide, TNLean.PEPS.Approximation.RepaintingBaseline.edgeInterface, TNLean.PEPS.Approximation.auxSteps, TNLean.PEPS.Approximation.mainSteps}
    \leanok
    \uses{def:peps_dyadic_normal_words, thm:peps_dyadic_central_clearance}
    A repainting of $S=[0,n]^2$, $n>0$, starts from a guide $g$ equal to $A$ on the
    open square and to one label $C_e$ on the open exterior neighboring square across
    each edge $e$. The guide after the central birth is $B$ on the central region and
    $g$ elsewhere. For a list $E$ of edges, the completed guide replaces, for each
    $e\in E$, the band $\{-8<x<2\}$ of $e$ by the word $C_e\mid B$ at $0$. During the
    construction along $e$ after the edges of $E$, the main guide with normal word $W$
    is the completed guide with the band of $e$ replaced by $W$, and the auxiliary
    guide with normal word $W$ is the uniform guide $C_e$ with the band of $e$
    replaced by $W$. The three auxiliary births and the four main births and deaths
    are those of Definition~\ref{def:peps_dyadic_normal_words}. Guides are
    labellings of the plane; at the start of the construction along $e$ the schedule
    passes from the completed guide to the main guide with the starting word, which
    agrees with it off the curves $x=0$ and $x=1$ of the band of $e$.
\end{definition}

\begin{lemma}[The guides of the schedule]
    \label{lem:peps_dyadic_schedule_guides}
    \lean{TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_edgeBand, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_edgeStartWord, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_eq_mainGuide_start, TNLean.PEPS.Approximation.RepaintingBaseline.lensExchange_mainGuide, TNLean.PEPS.Approximation.auxSteps_chain, TNLean.PEPS.Approximation.mainSteps_chain, TNLean.PEPS.Approximation.RepaintingBaseline.mainGuide_mainWordFive, TNLean.PEPS.Approximation.RepaintingBaseline.auxGuide_uniform, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_notMem_closedSquare, TNLean.PEPS.Approximation.RepaintingBaseline.completedGuide_of_mem_openSquare, TNLean.PEPS.Approximation.exists_edgeBand_of_notMem_centralRegion, TNLean.PEPS.Approximation.mem_centralRegion_of_forall_notMem}
    \leanok
    \uses{def:peps_dyadic_schedule, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_lens_word}
    Let $e\notin E$. On the band of $e$, off the curves $x=0$ and $x=1$, the
    completed guide carries the word $C_e\mid A\mid B$ at $0,1$; so it agrees with the
    main guide with that word off those curves. Exchanging the starting main guide
    and the auxiliary guide with word~(\ref{eq:pdg_aux_word}) on the lens $Y$ gives
    the main guide with word $C_e\mid B\mid A\mid C_e\mid A\mid B$ at
    $-6,-5,-4,0,1$. The auxiliary births start from the uniform word $C_e$, the main
    operations start from that main word, consecutive operations on one sheet match,
    and the last main death leaves the completed guide for the list $E$ with $e$
    added. Once all four edges are completed, the open square carries $B$, and every
    point outside the closed square keeps its starting label.
\end{lemma}

\begin{proof}\leanok
    On the band of $e$, the bands of the completed edges are absent by the band
    disjointness. Where $x<0$ the point lies in the exterior neighboring square;
    where $0<x<1$ it lies in the open square and outside the central region; where
    $1<x<2$ it lies in the central region, since for every other edge it lies in the
    open square and outside that edge's band, so its normal ratio there is at least
    $2$. The exchange is Lemma~\ref{lem:peps_dyadic_lens_word} on the band, and the
    identity off it. A point of the open square outside the central region lies in
    the band $\{0<x\le1\}$ of some edge, where the final word is $B$; a point outside
    the closed square lies in no band or in the part $x<0$ of a band, where the final
    word is $C_e$.
\end{proof}

\begin{definition}[Block guides and the labels of a level]
    \label{def:peps_dyadic_block_guide}
    \lean{TNLean.PEPS.Approximation.blockIndex, TNLean.PEPS.Approximation.blockGuide, TNLean.PEPS.Approximation.blockCorner, TNLean.PEPS.Approximation.SquareEdge.nbrBlock, TNLean.PEPS.Approximation.blockBaseline, TNLean.PEPS.Approximation.blockIndex_of_mem_openSquare, TNLean.PEPS.Approximation.blockIndex_of_mem_edgeNeighbor, TNLean.PEPS.Approximation.blockIndex_ne_of_notMem_closedSquare, TNLean.PEPS.Approximation.shiftGuide, TNLean.PEPS.Approximation.shiftGuide_blockBaseline_guide, TNLean.PEPS.Approximation.InRoot, TNLean.PEPS.Approximation.levelLabels}
    \leanok
    \uses{def:peps_dyadic_schedule}
    For $n>0$ and labels $\ell(Q)$ of the blocks $Q\in\mathbb Z^2$, the block guide
    has label $\ell(\lfloor p_1/n\rfloor,\lfloor p_2/n\rfloor)$ at $p$. The repainting
    of the block $S=(r,s)$ to the label $B$ starts from the block guide moved by
    $-(nr,ns)$, which is $\ell(S)$ on the open square and $\ell$ of the neighboring
    block across each edge; all its guides are then moved back by $(nr,ns)$. During
    the pass to level $n$, with a set of already repainted blocks, a repainted block
    of the root square carries its new label, another block of the root square the
    label of its parent $2n$-block, and a block outside the root square the
    placeholder.
\end{definition}

\begin{lemma}[Translation of the estimates]
    \label{lem:peps_dyadic_translation}
    \lean{TNLean.PEPS.Approximation.clearance_shift, TNLean.PEPS.Approximation.dist_le_shift, TNLean.PEPS.Approximation.le_dist_of_mem_closure, TNLean.PEPS.Approximation.cornerMarkDist_sub, TNLean.PEPS.Approximation.edgeMarkDist_sub, TNLean.PEPS.Approximation.sub_mem_closure, TNLean.PEPS.Approximation.sub_mem_frontier, TNLean.PEPS.Approximation.mem_closure_shift}
    \leanok
    \uses{def:peps_dyadic_block_guide}
    Moving all guides by a vector $o$ moves the changed regions and the obstructing
    sets by $o$, and the mark distances become the distances from the moved marks;
    sup distances are translation invariant. A lower bound on the distance from a
    point to every point of a set holds for its closure.
\end{lemma}

\begin{proof}\leanok
    Translation is a homeomorphism, so it commutes with closures and boundaries,
    and $|(y-o)-(z-o)|=|y-z|$. The set of points at distance at least $c$ from $y$
    is closed.
\end{proof}

\begin{theorem}[Separation of the unmodified bands]
    \label{thm:peps_dyadic_angular_separation}
    \lean{TNLean.PEPS.Approximation.blockRepainting_central, TNLean.PEPS.Approximation.blockRepainting_auxStep, TNLean.PEPS.Approximation.blockRepainting_mainStep, TNLean.PEPS.Approximation.blockRepainting_exchange, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_central, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_central, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_auxStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_mainStep, TNLean.PEPS.Approximation.RepaintingBaseline.clearance_exchange, TNLean.PEPS.Approximation.BandOperation.clearance_bandUpdate, TNLean.PEPS.Approximation.BandOperation.dist_le_bandUpdate, TNLean.PEPS.Approximation.dist_le_of_mem_closure_edgeLens, TNLean.PEPS.Approximation.RepaintingBaseline.exchange_changed_subset, TNLean.PEPS.Approximation.RepaintingBaseline.dist_le_exchange}
    \leanok
    \uses{thm:peps_dyadic_central_clearance, thm:peps_dyadic_band_clearance, thm:peps_dyadic_lens_clearance, lem:peps_dyadic_band_disjoint, lem:peps_dyadic_schedule_guides, lem:peps_dyadic_translation}
    Let $a_0=1/5000$, $n>0$, and consider the repainting of a block $S$ of side $n$
    from a block guide, with any block labels and any final label, through the
    guides that the schedule specifies at each operation. Every elementary
    birth or death of its schedule, the central birth and the seven births and
    deaths along each edge after any completed edges, has changed region $R$ of
    diameter at most $n$, and its surrounding guide $f$ satisfies, at every point
    $y\in\overline R$,
    \begin{align}
        \operatorname{dist}\bigl(y,\overline{\{f\ne P_\circ\}}\bigr)
        \ge a_0\min\{n,d_{\mathcal V}(y)\},
        \label{eq:pdg_angular_separation}
    \end{align}
    where $\mathcal V$ is the set of corners of $S$ for the central birth and the two
    endpoints of the edge otherwise. For each lens exchange, every point $y$ in the
    closure of the positions that are not common $C_e$ on its two guides before the
    exchange satisfies
    $\operatorname{dist}(y,\partial Y)\ge a_0\min\{n,d_{\mathcal V}(y)\}$. This
    is~\cite[Lemma~7.2]{OpenAI2026PolynomialPEPS}. The positions inside $Y$ that
    are not common $C_e$ before the exchange, which contain the positions the
    exchange changes, also form a set whose closure has diameter at most $n$; the
    source uses this bound in its proof of the exchange step, lines 421--424.
\end{theorem}

\begin{proof}\leanok
    The guides of an edge construction are obtained from one guide of their sheet by
    replacing the band of the edge by a word, so they have the properties assumed in
    Theorems~\ref{thm:peps_dyadic_central_clearance},
    \ref{thm:peps_dyadic_band_clearance} and~\ref{thm:peps_dyadic_lens_clearance}:
    such a guide carries its word on the band, two such guides of one sheet agree off
    the band, and the starting main word is $C_e$ on $\{-8<x<0\}$; the starting
    guide of the central birth is $A$ on the open square.
    Lemma~\ref{lem:peps_dyadic_schedule_guides} identifies these guides with the
    guides reached by the schedule. The diameter bound is
    Lemma~\ref{lem:peps_dyadic_band_disjoint}. The exchange changes the main guide
    from the starting guide to the guide with the main word, and the auxiliary guide
    to the starting main guide on $Y$, so it changes only points of $Y$ where the two
    sheets differ; these are not common $C_e$. The closure of $Y$ lies in the closed
    band $\{-7\le x\le-7/2\}$, of diameter at most $n$ by
    Lemma~\ref{lem:peps_dyadic_band_disjoint}. Translate by
    Lemma~\ref{lem:peps_dyadic_translation}.
\end{proof}

\begin{lemma}[Levels and active labels]
    \label{lem:peps_dyadic_active_labels}

\begin{lemma}[The compactness argument]
    \label{lem:peps_dyadic_footprint}
    \lean{TNLean.PEPS.Approximation.exists_pos_le_max_dist_of_inter_eq_empty, TNLean.PEPS.Approximation.exists_footprint_two_owners, TNLean.PEPS.Approximation.exists_footprint_two_owners_param, TNLean.PEPS.Approximation.exists_footprint_two_owners_guide}
    \leanok
    \uses{def:peps_dyadic_incident_labels}
    Let finitely many compact sets $F_c$ in a metric space be such that no three with
    distinct labels have a common point. Then there is $\delta>0$ such that no
    closed ball of radius $\delta$ meets three of them with distinct labels. If the
    $F_c$ are compact sets of pairs of a parameter and a point, one $\delta$ works
    for all values of the parameter. In particular, let a guide take finitely many
    values, let $W$ be a compact working neighborhood, and let every true vertex in
    $W$ lie in one of the open inner hole squares. Then there is $\delta>0$ such that
    every closed footprint of radius $\delta$ meets at most two labels at its points
    of $W$ outside the open inner holes. This is the abstract compactness argument
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 453--462 and 475--479, and
    its parametric form, lines 464--469, for compact joint sets.
\end{lemma}

\begin{proof}\leanok
    For three compact sets without a common point,
    $\max(|y_1-y_2|,|y_1-y_3|)$ attains a positive minimum on their product; take a
    third of the smallest of these minima over the finitely many triples. For the
    parameters, apply this in the product space, where points with the same
    parameter are at the distance of their second coordinates. For the guide, take
    $F_c$ the closure of the region of $c$ in $W$ with the open inner holes removed;
    a common point of three of them would be a true vertex outside every inner hole.
\end{proof}

\begin{lemma}[Footprint ratios uniform in the scale]
    \label{lem:peps_dyadic_footprint_uniform}
    \lean{TNLean.PEPS.Approximation.HasTwoOwnerFootprints, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.mono, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.relabel, TNLean.PEPS.Approximation.similarity, TNLean.PEPS.Approximation.dist_similarity, TNLean.PEPS.Approximation.similarity_inv, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.image_similarity, TNLean.PEPS.Approximation.isCompact_footprintJointSet, TNLean.PEPS.Approximation.exists_hasTwoOwnerFootprints_Icc, TNLean.PEPS.Approximation.exists_footprintRatio, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.of_template, TNLean.PEPS.Approximation.HasTwoOwnerFootprints.of_le_radii}
    \leanok
    \uses{lem:peps_dyadic_footprint}

    centers $H$, inner radius $r$ and footprint radius $\delta$ if every closed
    footprint of radius $\delta$ meets at most two labels at its points of $W$ outside
    the open inner holes of radius $r$ about the points of $H$. This condition survives
    shrinking $\delta$, identifying labels, and enlarging some of the inner holes, so it
    holds as well when the holes have different radii, all at least $r$. It is carried
    by the similarity
    $u\mapsto v+su$, $s>0$, to the image guide, working set and hole centers with the
    radii $sr$ and $s\delta$. For a guide with finitely many values and a compact $W$,
    the pairs of a radius $r\in[r_0,r_1]$ and a point of $W$ in the closure of the

    set. Hence, if every true vertex in $W$ lies in an open inner hole of radius

    footprint radius $\nu s$. This is the uniformity argument
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 464--479, with the joint
    sets of lines 465--468 and the minimum over finitely many label-equality types of
    lines 469--471.
\end{lemma}

\begin{proof}\leanok

    closed in the pair $(r,y)$; with $r\in[r_0,r_1]$ and $y\in W$ the joint sets are
    compact. Three of them with distinct labels have no common point, since a common
    point $y$ would be a true vertex with $r_0\le r\le|y-h|<r_0$ for some $h$. Apply
    the parametric form of Lemma~\ref{lem:peps_dyadic_footprint}, take the minimum
    over the finitely many templates, and transport by the similarity, which
    multiplies all sup distances by $s$.
\end{proof}

\begin{proof}\leanok

    the block of side $n$ to those of the unit square, so every unmodified guide is an
    unmodified guide of the repainting of the unit square composed with the scaling.
    Relabelling commutes with replacing a band by a word and with the central birth,
    so near the corner the guide is the relabelling by the window labels of an
    unmodified guide of one reference repainting, whose labels are the nine blocks
    around $(0,0)$ and the final label. On the unit square the bands, the normal
    ratios, the central region and the block guide are conical about each corner
    within distance $1/2$, by the straight-ray charts of
    Lemma~\ref{lem:peps_dyadic_choice_K0}; so are the unmodified guides, and the scale
    $t/n<1/8$ can be replaced by $1/32$. The reference repainting has finitely many
    unmodified guides, since a guide after completed edges depends only on which edges
    are completed. Choosing for each label-equality type a representative map of the
    finitely many slots, the guide near $v$ is an injective relabelling of one of
    finitely many templates, and injective relabellings and similarities preserve true
    vertices. Apply Lemma~\ref{lem:peps_dyadic_footprint_uniform}. Near one corner of
    $\mathcal M$ the other treated squares are at distance more than $n-5t>3t$, so
    homogenizing on all of them agrees with homogenizing on its square alone within
    distance $4t$ of the corner, which contains the working set and the holes.
\end{proof}

\begin{theorem}[One patch ratio for direct repaintings and resizings]
    \label{thm:peps_dyadic_block_ratio}
    \lean{TNLean.PEPS.Approximation.blockWindow, TNLean.PEPS.Approximation.BlockSlot, TNLean.PEPS.Approximation.blockSlot, TNLean.PEPS.Approximation.blockSlotLabel, TNLean.PEPS.Approximation.blockSlotLabel_blockSlot, TNLean.PEPS.Approximation.floor_mem_Icc, TNLean.PEPS.Approximation.blockGuide_eq_blockSlot, TNLean.PEPS.Approximation.exists_blockGuide_footprintRatio, TNLean.PEPS.Approximation.exists_directRepainting_footprintRatio, TNLean.PEPS.Approximation.exists_resizing_footprintRatio, TNLean.PEPS.Approximation.exists_resizing_footprintRatio_mixed}
    \leanok
    \uses{lem:peps_dyadic_footprint_uniform, lem:peps_dyadic_small_radius, lem:peps_dyadic_direct_repainting}
    Fix $\rho$ and $0<r_0\le r_1$. There is a ratio $\nu>0$ such that every guide
    uniform on the blocks of side $s>0$, about every grid corner $v$ and for every
    inner radius $r\in[r_0s,r_1s]$, satisfies the two-owner condition with working set
    the closed square of radius $s\rho$ about $v$, holes at its true vertices within
    $s(\rho+1)$ of $v$, and footprint radius $\nu s$. In particular, for fixed
    $\epsilon_0>0$ and $K_0$, one ratio serves every direct repainting of a square of
    side $0<n\le K_0t$, before and after, with inner radius $h_n$ and working set the
    closed square of radius $2n$ about a corner of the square; and one ratio serves
    every resizing with $0<n<t$, with both radii $h_n$ and $h_{2n}$ on the guide uniform
    on the blocks of side $2n$ and working set of radius $2n$, and also when some holes
    have radius $h_n$ and the others $h_{2n}$. These are the compactness arguments
    of~\cite[Section~7]{OpenAI2026PolynomialPEPS}, lines 515--520 and 534--538, in units
    of $n$, with the intermediate configurations of lines 538--540.
\end{theorem}

\begin{proof}\leanok
    In units of $s$ about a grid corner, a block guide is the block guide of side one
    whose labels are those of a fixed square window of blocks, so it is an injective
    relabelling of one of finitely many templates, one for each label-equality type of
    the window. Apply Lemma~\ref{lem:peps_dyadic_footprint_uniform} and
    Lemma~\ref{lem:peps_dyadic_small_radius} for the radius ratios; when $n<t$ the
    radius $h_n=\epsilon_0n$ and $h_{2n}=\epsilon_0\min(2n,t)$ lie between
    $\epsilon_0n$ and $2\epsilon_0n$. Since $h_n\le h_{2n}$, a configuration with some
    holes resized has larger holes than the one with all radii $h_n$.
\end{proof}

\begin{lemma}[Trivial holes at the last level]

\begin{definition}[Padded dyadic hierarchy, junctions and addresses]
    \label{def:peps_dyadic_hierarchy}
    \lean{TNLean.PEPS.Approximation.DyadicSquare, TNLean.PEPS.Approximation.DyadicJunction}
    \leanok
    \uses{def:peps_dyadic_anchor}
    For $N=2^k$ the hierarchy consists of the aligned squares of $[0,N]^2$ of side
    $2^j$, $0\le j\le k$, indexed by $j$ and two block indices below $2^{k-j}$; the
    junctions at scale $j$ are the grid corners, indexed by two integers at most
    $2^{k-j}$. A square of side $2^j$ with block index $r$ has address coordinate
    $2^jr+2^{j-1}$ for $j\ge1$ and $r$ for $j=0$, the anchor coordinate of
    Definition~\ref{def:peps_dyadic_anchor}.
\end{definition}

\begin{theorem}[Counts of the hierarchy and the schedule]
    \label{thm:peps_dyadic_hierarchy_counts}
    \lean{TNLean.PEPS.Approximation.three_mul_card_dyadicSquare_add_one, TNLean.PEPS.Approximation.three_mul_card_dyadicSquare_lt, TNLean.PEPS.Approximation.three_mul_card_dyadicJunction_le, TNLean.PEPS.Approximation.three_mul_schedule_le, TNLean.PEPS.Approximation.three_mul_schedule_le_cube}
    \leanok
    \uses{def:peps_dyadic_hierarchy}
    The hierarchy has $(4^{k+1}-1)/3$ squares, and three times the number of
    junctions over all scales is at most $4^{k+2}$. If $2^k<2L$, the hierarchy has
    fewer than $16L^2/3$ squares, and $M$ times the number of squares plus
    junctions is at most $80ML^2/3$, and for $L\ge1$ at most $80ML^3/3$. This is
    the count behind a schedule using at most $M$ changes for each repainting of a
    square and each junction resize; the schedule of changes and gates of the
    protocol is not constructed here.
\end{theorem}

\begin{proof}\leanok
    The scale $j$ contributes $4^{k-j}$ squares and $(2^{k-j}+1)^2\le4\cdot4^{k-j}$
    junctions, and $3\sum_{i<k+1}4^i+1=4^{k+1}$ by induction. Finally
    $4^k=(2^k)^2<4L^2$.
\end{proof}

\begin{lemma}[Addresses and locality]
    \label{lem:peps_dyadic_address_locality}
    \lean{TNLean.PEPS.Approximation.dyadicAnchor_lt_two_pow, TNLean.PEPS.Approximation.dyadicAnchor_lt_add, TNLean.PEPS.Approximation.dyadicAnchor_dist_lt, TNLean.PEPS.Approximation.dyadicAnchor_dist_lt_of_adjacent, TNLean.PEPS.Approximation.card_blocks_meeting_le}
    \leanok
    \uses{def:peps_dyadic_hierarchy, lem:peps_dyadic_active_labels}
    In each coordinate: an address coordinate lies in $[2^jr,2^jr+2^j)$, hence
    below $2^k$. If the intervals $[2^jr,2^j(r+1)]$ and $[2^{j'}r',2^{j'}(r'+1)]$
    are within distance $g$ and $j'\le j+1$, the address coordinates differ by less
    than $g+3\cdot2^j$. In particular, for two blocks of side $2^j$ with equal or
    adjacent indices, each block or its parent of side $2^{j+1}$ has an address
    coordinate within $3\cdot2^{j+1}$ of the other's. The intervals of side $b$
    meeting $[V,U]$ number at most $\lfloor U/b\rfloor+2-\lfloor V/b\rfloor$. With
    Lemma~\ref{lem:peps_dyadic_active_labels}, these are ingredients of item~3 and
    of the bounded lifetime participation
    in~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}; the gates and parties of
    the protocol are not treated here.
\end{lemma}

\begin{proof}\leanok
    The first two claims follow from the position of the address in its block and
    $2^{j'}\le2\cdot2^j$. For adjacent blocks, the interval of a block or of its
    parent contains the block's own interval, and the two blocks' intervals touch,
    so the second claim applies with $g=0$. For the count, $r\mapsto r+1$ maps the meeting blocks
    injectively into $[\lfloor V/b\rfloor,\lfloor U/b\rfloor+1]$.
\end{proof}
