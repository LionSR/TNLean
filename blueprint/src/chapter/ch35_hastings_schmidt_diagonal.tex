\section{Correlations of Schmidt-diagonal operators in Hastings' area law}
\label{sec:hastings_schmidt_diagonal}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 627--692 (Lemma 3,
% equation fwdback).

\cite[Lemma~3]{Hastings2007AreaLaw} bounds the correlation between a
Schmidt-diagonal operator across a cut and an operator supported outside a
buffer around the cut, using only the approximate projector factorization of
Lemma~1.

\begin{theorem}[Correlation bound under an approximate factorization]
    \label{thm:hastings_schmidt_diagonal_operator}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_inner_sub_le_of_projector_factorization}
    \leanok
    Let $\Omega$ be a unit vector and $P=|\Omega\rangle\langle\Omega|$. Let
    $0\le O_B,O_L,O_R\le1$ satisfy $\|O_BO_LO_R-P\|\le\varepsilon$, and let
    $A,B_L,B_R$ have norm at most $1$. If
    $[O_B,A]=[O_R,B_L]=[B_R,O_L]=[O_L,O_R]=0$ and $B_L\Omega=B_R\Omega$, then
    \begin{align}
        \bigl|\langle\Omega,AB_L\Omega\rangle-\langle\Omega,A\Omega\rangle
        \langle\Omega,B_L\Omega\rangle\bigr|
         & \le3\sqrt{2\varepsilon}+\varepsilon.
        \label{eq:hastings_schmidt_diagonal_operator}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    From $\|O_BO_LO_R\Omega-\Omega\|\le\varepsilon$ and $0\le O_B,O_L,O_R\le1$,
    each factor satisfies $\|(O_X-1)\Omega\|\le\delta=\sqrt{2\varepsilon}$.
    Writing $x\approx_\eta y$ for $|x-y|\le\eta$, the commutation relations,
    $B_L\Omega=B_R\Omega$, and $\|O_BO_LO_R-P\|\le\varepsilon$ yield
    \begin{align}
        \langle\Omega,AB_L\Omega\rangle
         & \approx_\delta\langle\Omega,AB_LO_R\Omega\rangle
        \approx_\delta\langle\Omega,O_BAB_LO_R\Omega\rangle
        =\langle\Omega,AO_BO_RB_R\Omega\rangle
        \notag                                                    \\
         & \approx_\delta\langle\Omega,AO_BO_RB_RO_L\Omega\rangle
        =\langle\Omega,A(O_BO_LO_R)B_R\Omega\rangle
        \approx_\varepsilon\langle\Omega,APB_R\Omega\rangle
        =\langle\Omega,A\Omega\rangle\langle\Omega,B_L\Omega\rangle,
        \notag
    \end{align}
    giving~(\ref{eq:hastings_schmidt_diagonal_operator}).
\end{proof}

\begin{theorem}[Schmidt-diagonal operators are contractions]
    \label{thm:hastings_schmidt_diagonal_norm}
    \lean{TNLean.MPS.AreaLaw.Hastings.l2_opNorm_schmidtDiagonal_le_one}
    \leanok
    For an orthonormal family $(\psi(\alpha))_\alpha$ and $|O(\alpha)|\le1$,
    $\bigl\|\sum_\alpha O(\alpha)\,|\psi(\alpha)\rangle\langle\psi(\alpha)|\bigr\|\le1$.
\end{theorem}
\begin{proof}\leanok
    By Bessel's inequality,
    $\|\sum_\alpha O(\alpha)\langle\psi(\alpha),v\rangle\psi(\alpha)\|^2
        =\sum_\alpha|O(\alpha)|^2|\langle\psi(\alpha),v\rangle|^2\le\|v\|^2$.
\end{proof}

\begin{theorem}[Correlations of a Schmidt-diagonal operator under an
        approximate factorization]
    \label{thm:hastings_schmidt_diagonal_chain}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_inner_mul_sub_le_of_factorization}
    \leanok
    \uses{def:chain_model}
    Let $\Omega=\sum_\alpha a_\alpha\,\psi_L(\alpha)\otimes\psi_R(\alpha)$ be the
    Schmidt decomposition of a unit vector on a chain of $L$ sites across cut
    $j$. Let
    $B_L=\sum_\alpha O(\alpha)\,|\psi_L(\alpha)\rangle\langle\psi_L(\alpha)|\otimes1$
    with $|O(\alpha)|\le1$, and let $A$ with $\|A\|\le1$ be supported outside
    the buffer $[j-l,j+l)\cap[0,L)$. Suppose $0\le O_B,O_L,O_R\le1$ are
    supported on $[j-l,j+l)\cap[0,L)$, $[0,j)$, and $[j,L)$ respectively, with
    $\|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|\le\varepsilon$. Then
    (\ref{eq:hastings_schmidt_diagonal_operator}) holds. With Lemma~1 of
    \cite{Hastings2007AreaLaw} this yields \cite[Lemma~3, eq.~(fwdback)]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_schmidt_diagonal_operator,
        thm:hastings_schmidt_diagonal_norm}
    Set $B_R=1\otimes\sum_\alpha O(\alpha)\,|\psi_R(\alpha)\rangle\langle\psi_R(\alpha)|$.
    By orthonormality and Theorem~\ref{thm:hastings_schmidt_diagonal_norm},
    \begin{align}
        B_L\Omega=\sum_\alpha a_\alpha O(\alpha)\,\psi_L(\alpha)\otimes\psi_R(\alpha)=B_R\Omega,
        \qquad \|B_L\|\le1,\qquad \|B_R\|\le1.
        \notag
    \end{align}
    Disjoint supports give $[O_B,A]=[O_R,B_L]=[B_R,O_L]=[O_L,O_R]=0$, so
    Theorem~\ref{thm:hastings_schmidt_diagonal_operator} applies.
\end{proof}

\begin{theorem}[The correlation lemma]
    \label{thm:hastings_fwdback}
    \lean{TNLean.MPS.AreaLaw.exists_norm_inner_mul_sub_le}
    \leanok
    \uses{def:chain_model, thm:hastings_lemma1_endpoint}
    For every range $R\in\N$ there is a constant $K\ge0$ depending only on $R$
    with the following property. Let $H$ be a range-$R$ Hamiltonian on a chain
    of $L$ sites with term norms at most $J\ge0$, unique ground state
    $\Psi_0=\sum_\alpha a_\alpha\,\psi_L(\alpha)\otimes\psi_R(\alpha)$ across
    cut $j$, and gap $\Delta E>0$, and let $\tilde\xi$ be the length of
    Theorem~\ref{thm:hastings_lemma1_endpoint}. For
    $B_L=\sum_\alpha O(\alpha)\,|\psi_L(\alpha)\rangle\langle\psi_L(\alpha)|\otimes1$
    with $|O(\alpha)|\le1$ and $A$ with $\|A\|\le1$ supported outside
    $[j-l,j+l)$,
    \begin{align}
        \bigl|\langle\Psi_0,AB_L\Psi_0\rangle-\langle\Psi_0,A\Psi_0\rangle
        \langle\Psi_0,B_L\Psi_0\rangle\bigr|
         & \le3\sqrt{2\epsilon(l)}+\epsilon(l),
         & \epsilon(l)                          & =K\tilde\xi^3e^{-l/(12\tilde\xi)}.
        \notag
    \end{align}
    This is \cite[Lemma~3, eq.~(fwdback), lines 627--646]{Hastings2007AreaLaw}
    with rate $12\tilde\xi$ (twice $\xi'=6\tilde\xi$) from the positive repair
    of Lemma~1~\cite{gap:hastings07_lemma1_positivity_rate}, polynomial
    prefactor $K\tilde\xi^3$, and $\tilde\xi$ from the chain Lieb--Robinson
    estimate with raised velocity~\cite{gap:hastings07_lemma1_decay_length,
        gap:hastings07_gaussian_average_exponent}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint,
        thm:hastings_schmidt_diagonal_chain}
    Theorem~\ref{thm:hastings_lemma1_endpoint} supplies $0\le O_B,O_L,O_R\le1$
    at $(j,l)$ with $\|O_BO_LO_R-P_0\|\le\epsilon(l)$, so
    Theorem~\ref{thm:hastings_schmidt_diagonal_chain} applies with
    $\varepsilon=\epsilon(l)$.
\end{proof}
