\section{The anomalous \texorpdfstring{$\Z_3$}{Z3} symmetry and its condensation defect}%
\label{sec:asymex_z3anomalous}

A finite group acting on a spin chain by matrix product operators may be
anomalous: the obstruction to gauging it is a three-cocycle, and for a
cyclic group Garre-Rubio and Schuch~\cite{GarreRubioSchuch2024DomainWall}
give the cocycle in closed form together with the interchange statistics
of the domain walls it governs. That source prints no local tensor: the
phase-decorated shift defined below is a construction of this development
motivated by their cocycle formula for $\Z_3$, and no identification of
its anomaly class is made here. The general construction of a matrix
product operator representation from a finite group is given in Garre-Rubio,
Lootens and Molnár~\cite{GarreRubio2023MPOSym}, and matrix product operator
symmetries are reviewed in~\cite[Section~III.A.6]{Cirac2021Matrix}. The sum of the
three group operators is a condensation defect in the terminology of
Roumpedakis, Seifnashri and Shao~\cite{Roumpedakis2023Condensation}.

Let $\omega=(-1+\mathrm{i}\sqrt{3})/2$ be the primitive cube root of unity.
Thus $\omega^2+\omega+1=0$, $\omega^3=1$, and
$\overline{\omega}=\omega^{-1}=\omega^2$. On a periodic chain of $N>0$
qutrits, use basis vectors $\ket{t}$ indexed by
$t=(t_0,\ldots,t_{N-1})\in\{0,1,2\}^N$. Set $\chi(a)=[a\neq0]$ and
\begin{align}
    F(t) & =\sum_{j=0}^{N-1}\chi(t_j)t_{j+1},
    \label{eq:asymex_z3_phase}
\end{align}
where $t_N=t_0$ and the exponent uses the integer representatives
$0,1,2$. Write $X\ket{a}=\ket{a+1}$, with addition modulo three, and
$(Xt)_j=t_j+1$. The phase-decorated shift acts by
\begin{align}
    U_N\ket{t}
     & =\omega^{F(t)}\ket{Xt}.
    \label{eq:asymex_z3anomalous_operator}
\end{align}
The diagonal phase acts first, followed by $X^{\otimes N}$. Every cyclic
bond is included: at $N=1$ there is a self-loop, and at $N=2$ both
oriented bonds contribute.

The group-cocycle context for anomalous symmetries is discussed in
Garre-Rubio and Schuch~\cite{GarreRubioSchuch2024DomainWall}; see also
Sun~\cite{Sun2025Anomalous}.
The physical operators and their group law are derived directly from the
tensor entries, and the compression of their stacked products is
determined afterwards.

\begin{definition}[The tensors of the representation]
    \label{def:asymex_z3anomalous_tensors}
    \lean{Z3Anomalous.uTensor, Z3Anomalous.uDagTensor, Z3Anomalous.identityTensor,
        Z3Anomalous.uMPS, Z3Anomalous.uDagMPS, Z3Anomalous.identityMPS}
    \leanok
    \uses{def:mpo_tensor, def:mpo_to_mps, def:mpo_family}
    The tensor $M$ has bond dimension two, with
    \begin{align}
        M^{1,0} & =\begin{pmatrix}1&0\\1&0\end{pmatrix},        &
        M^{2,1} & =\begin{pmatrix}0&1\\0&\omega\end{pmatrix},   &
        M^{0,2} & =\begin{pmatrix}0&1\\0&\omega^2\end{pmatrix},
        \notag
    \end{align}
    and all other matrices zero. The independently specified tensor
    denoted by $M^\dagger$ has bond dimension two, with
    \begin{align}
        (M^\dagger)^{0,1} & =\begin{pmatrix}1&0\\1&0\end{pmatrix},        &
        (M^\dagger)^{1,2} & =\begin{pmatrix}0&1\\0&\omega^2\end{pmatrix}, &
        (M^\dagger)^{2,0} & =\begin{pmatrix}0&1\\0&\omega\end{pmatrix},
        \notag
    \end{align}
    and all other matrices zero. The first physical index is the output
    and the second is the input. The identity tensor $\delta$ has bond
    dimension one and scalar matrices $\delta^{o,i}=\delta_{o,i}$,
    where $\delta_{o,i}$ is the Kronecker delta.
    Read over the alphabet of index pairs, $M$, $M^\dagger$ and $\delta$
    are matrix product state tensors of physical dimension nine, with
    the matrix at $(o,i)$ assigned to the letter $3o+i$.

    For any such tensor $T$, its periodic operator uses the ordinary,
    unnormalized trace:
    \begin{align}
        O_N(T)(s,t)
         & =\tr(T^{s_0,t_0}\cdots T^{s_{N-1},t_{N-1}}).
        \label{eq:asymex_z3_periodic_kernel}
    \end{align}
    Set $U_N=O_N(M)$ and $V_N=O_N(M^\dagger)$. The next lemma verifies
    that this definition of $U_N$ gives
    (\ref{eq:asymex_z3anomalous_operator}).
    The contraction below represents $U_L$ on a ring drawn with $L$
    sites and physical input labels $s_1,\ldots,s_L$.

            \tn[skin=mpo, ports={90:physical:$s_L{+}1$, 270:physical:$s_L$}]{M}

    \end{center}
\end{definition}

\begin{theorem}[Physical kernels and unitarity]
    \label{thm:asymex_z3_physical_kernels}
    \lean{Z3Anomalous.mpo_uTensor_apply,
        Z3Anomalous.mpo_uDagTensor_apply_zpow,
        Z3Anomalous.mpo_uTensor_mulVec_single,
        Z3Anomalous.mpo_uDagTensor_mulVec_single,
        Z3Anomalous.mpo_uTensor_mem_unitaryGroup,
        Z3Anomalous.mpo_uDagTensor_mem_unitaryGroup,
        Z3Anomalous.mpo_uDagTensor_eq_conjTranspose,
        Z3Anomalous.uTensor_isMPUPos,
        Z3Anomalous.uDagTensor_isMPUPos}
    \leanok
    \uses{def:asymex_z3anomalous_tensors}

    For every $N>0$, the two bond-two tensors have kernels
    \begin{align}
        U_N(s,t) & =\delta_{s,Xt}\omega^{F(t)},             \\
        V_N(s,t) & =\delta_{s,X^{-1}t}\omega^{-F(X^{-1}t)}.
        \label{eq:asymex_z3_inverse_kernel}
    \end{align}
    Both operators are unitary, and $V_N=U_N^\dagger=U_N^{-1}$.
\end{theorem}

\begin{proof}
    \leanok
    For physical labels $o,i\in\{0,1,2\}$ and bond labels
    $\alpha,\beta\in\{0,1\}$, the defining entries give
    \begin{align}
        M^{o,i}_{\alpha,\beta}
         & =\delta_{o,i+1}\delta_{\beta,\chi(i)}\omega^{\alpha i},  \\
        (M^\dagger)^{o,i}_{\alpha,\beta}
         & =\delta_{i,o+1}\delta_{\beta,\chi(o)}\omega^{-\alpha o}.
        \label{eq:asymex_z3_local_entries}
    \end{align}
    The physical equalities in the Kronecker deltas are modulo three.
    Expanding the trace in~(\ref{eq:asymex_z3_periodic_kernel}) gives
    \begin{align}
        U_N(s,t)
         & =\sum_{g_0,\ldots,g_{N-1}\in\{0,1\}}
        \prod_{j=0}^{N-1}M^{s_j,t_j}_{g_j,g_{j+1}},
        \label{eq:asymex_z3_cyclic_trace}
    \end{align}
    with $g_N=g_0$. A nonzero summand requires $s=Xt$.
    The bond constraints then force $g_{j+1}=\chi(t_j)$, so there is
    exactly one surviving bond configuration. Its weight is
    $\omega^{\sum_j\chi(t_{j-1})t_j}=\omega^{F(t)}$, by cyclic
    reindexing.

    For $M^\dagger$, the physical constraint forces $s=q=X^{-1}t$,
    and the bonds are $g_{j+1}=\chi(q_j)$. The surviving weight is
    $\omega^{-\sum_j\chi(q_{j-1})q_j}=\omega^{-F(q)}$, proving
    (\ref{eq:asymex_z3_inverse_kernel}).

    Since $X$ permutes configurations and $|\omega|=1$, $U_N$ maps an
    orthonormal basis to an orthonormal basis. The same argument applies
    to $V_N$. Conjugate transposition of the physical kernel yields
    \begin{align}
        U_N^\dagger(s,t)
         & =\overline{U_N(t,s)}
        =\delta_{t,Xs}\omega^{-F(s)}
        =\delta_{s,X^{-1}t}\omega^{-F(X^{-1}t)}
        =V_N(s,t).
    \end{align}
    Thus $V_N$ is also the inverse of $U_N$.
\end{proof}

\begin{lemma}[The cube of the phase-decorated shift]
    \label{lem:asymex_z3_physical_cube}
    For every $N>0$, $U_N^3=\Id$ and $V_N=U_N^2$.
\end{lemma}

\begin{proof}
    For $a,b\in\{0,1,2\}$, direct evaluation gives
    \begin{align}
        \left[\sum_{r=0}^{2}\chi(a+r\bmod3)(b+r\bmod3)\right]_{a,b=0}^{2}
         & =\begin{pmatrix}3&2&1\\1&3&2\\2&1&3\end{pmatrix}.
    \end{align}
    The $(a,b)$ entry is congruent to $a-b$ modulo three.
    Summing over cyclic bonds therefore gives
    \begin{align}
        F(t)+F(Xt)+F(X^2t)
         & \equiv\sum_{j=0}^{N-1}(t_j-t_{j+1})
        =0\pmod3.
        \label{eq:asymex_z3_phase_telescope}
    \end{align}
    Applying~(\ref{eq:asymex_z3anomalous_operator}) three times and using
    (\ref{eq:asymex_z3_phase_telescope}) gives
    $U_N^3\ket{t}=\ket{t}$. Since $V_N=U_N^{-1}$, it follows that
    $V_N=U_N^2$.
\end{proof}

\begin{theorem}[Normality of the three blocks up to normalization]
    \label{thm:asymex_z3anomalous_normal}
    \lean{Z3Anomalous.uMPS_isNormal, Z3Anomalous.uDagMPS_isNormal,
        Z3Anomalous.identityMPS_isNormal}
    \leanok
    \uses{def:asymex_z3anomalous_tensors, def:normal}
    The length-two words of $M$ span $\MN{2}$, as do those of $M^\dagger$.
    The length-one words of $\delta$ span $\MN{1}$. Thus each tensor is proportional
    to a normal tensor (NT); the NT convention additionally normalizes the spectral
    radius of the associated completely positive map to one.
\end{theorem}

\begin{proof}
    \leanok
    For $M$, four length-two words in the two letters $(0,2)$ and $(1,0)$
    combine, with coefficients in the Eisenstein integers $\Z[\omega]$,
    to $(\omega-1)$ times each matrix unit of $\MN{2}$.
    Since $\omega-1\neq0$, these words span $\MN{2}$ over $\C$.
    The same calculation holds for $M^\dagger$ with the letters
    $(0,1)$ and $(1,2)$. For the identity tensor, a diagonal physical
    index pair gives the nonzero scalar $1$. These spanning identities
    establish injectivity after blocking, independently of any
    normalization of the transfer spectral radius.
\end{proof}

\begin{remark}[Notation for the group inverse]
    \label{rem:asymex_z3anomalous_notation}
    The symbol $M^\dagger$ denotes the tensor in
    Definition~\ref{def:asymex_z3anomalous_tensors}, not the conjugate
    transpose of each bond matrix of $M$. Its entries satisfy
    $(M^\dagger)^{o,i}=\overline{M^{i,o}}$: the physical indices are
    exchanged, while the bond matrices are conjugated without
    transposition. Theorem~\ref{thm:asymex_z3_physical_kernels}
    and Lemma~\ref{lem:asymex_z3_physical_cube} show that its periodic operator
    is $U_N^\dagger=U_N^{-1}=U_N^2$ for $N>0$.

    At $N=1$, the self-loop gives $F(t)=t_0$, so
    $U_1\ket{a}=\omega^a\ket{a+1}$.
    At $N=2$, $F(t)=\chi(t_0)t_1+\chi(t_1)t_0$.
    These short rings obey the same unitary group law. At $N=0$, however,
    the trace of the empty bond product gives
    $O_0(M)=O_0(M^\dagger)=[2]$ and $O_0(\delta)=[1]$.
    The positive-length hypothesis cannot be omitted.
\end{remark}

\subsection{Fusion of the anomalous representation}%
\label{sec:asymex_z3anomalous_fusion}

Write $M_0=\delta$, $M_1=M$ and $M_2=M^\dagger$, with group indices
added modulo three. The physical calculations give
$O_N(M_g)O_N(M_h)=O_N(M_{g+h})$ for every $N>0$.
The stacked product tensor has entries
\begin{align}
    (M_g\cdot M_h)^{o,i}
     & =\sum_{k=0}^{2}M_g^{o,k}\otimes M_h^{k,i}.
    \label{eq:asymex_z3_stacked_product}
\end{align}
Its bond dimension is the product of the two bond dimensions, and its
periodic operator is the physical product
(Theorem~\ref{thm:asym_mpo_product}). Consequently, the group law gives
equality of nonempty-word traces before any compression pair is chosen.
The asymmetric reductions realizing these equalities on the virtual
spaces are described next. In the statements and diagrams below, $L$
denotes the ring length, as $N$ did above.

The four nontrivial products have targets $M^\dagger$ for $M\cdot M$,
$\delta$ for $M\cdot M^\dagger$ and $M^\dagger\cdot M$, and $M$ for
$M^\dagger\cdot M^\dagger$. The compressions come first, and the periodic
fusion rules follow from them.

\begin{theorem}[Compression of \texorpdfstring{$M\cdot M$ and $M^\dagger\cdot M^\dagger$}{the two equal-factor products}]
    \label{thm:asymex_z3anomalous_uu_compression}
    \lean{Z3Anomalous.uu_compression, Z3Anomalous.uu_isReduction, Z3Anomalous.uu_right_eq,
        Z3Anomalous.uu_left_eq, Z3Anomalous.uu_dim_eq,
        Z3Anomalous.uu_evalWord_remainder_eq_zero, Z3Anomalous.uu_remainder_mul_ne_zero,
        Z3Anomalous.dd_compression, Z3Anomalous.dd_isReduction, Z3Anomalous.dd_right_eq,
        Z3Anomalous.dd_left_eq, Z3Anomalous.dd_dim_eq,
        Z3Anomalous.dd_evalWord_remainder_eq_zero, Z3Anomalous.dd_remainder_mul_ne_zero}
    \leanok
    \uses{def:asymex_z3anomalous_tensors, thm:asym_multiblock}
    The stacked tensor $B=M\cdot M$ has the block triangular form of
    Theorem~\ref{thm:asym_multiblock} for the single target $M^\dagger$ with $z=2$
    one-dimensional zero diagonal blocks, $4=2+2$. The compression pair is
    \begin{align}
        V & =\begin{pmatrix}0&0\\0&-\omega\\0&1+\omega\\-1&0\end{pmatrix}, &
        W & =\begin{pmatrix}0&0&0&-1\\0&0&-\omega&0\end{pmatrix},
        \notag
    \end{align}
    with $WV=\Id$ and $WB^wV=(M^\dagger)^w$ for every word.
    The symbol $V$ here denotes a virtual compression matrix, not the
    physical operator $V_L$.
    The remainder $R^{ij}=B^{ij}-V(M^\dagger)^{ij}W$ is nilpotent of
    order exactly three: some product of two letters is nonzero, but
    every product of three vanishes.

    The stacked tensor $M^\dagger\cdot M^\dagger$ has the analogous block
    triangular form for the single target $M$, again with $z=2$
    one-dimensional zero diagonal blocks and a remainder of order exactly three.
    Its compression pair is
    \begin{align}
        V_{22} & =\begin{pmatrix}0&0\\0&-1\\0&-\omega\\1+\omega&0\end{pmatrix}, &
        W_{22} & =\begin{pmatrix}0&0&0&-\omega\\0&0&1+\omega&0\end{pmatrix}.
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    Choose bond coordinates adapted to the common kernel, the common image
    modulo that kernel, and a completion. Conjugation by this change of
    coordinates makes every letter block upper triangular with $M^\dagger$
    on the matched diagonal block. The entries and the inverse change of
    coordinates are checked over $\Z[\omega]$, as in
    Theorem~\ref{thm:asymex_explicit_gauge}.
    The rows and columns belonging to the matched block give $W$ and $V$.
    Multiplication of upper triangular block matrices gives
    $WB^wV=(M^\dagger)^w$, including $WV=\Id$ for the empty word.
    The remainder identities are
    $R^aR^bR^c=0$ for all letters $a,b,c$ and $R^1R^5\neq0$,
    using the pair-alphabet encoding $a=3o+i$.
    For $M^\dagger\cdot M^\dagger$, the corresponding calculation has
    target $M$ and gives $(R^2)^2\neq0$, with all triple products zero.
\end{proof}

\begin{theorem}[Periodic fusion \texorpdfstring{$U\cdot U=U^\dagger$ and $U^\dagger\cdot U^\dagger=U$}{of the nonidentity group elements}]
    \label{thm:asymex_z3anomalous_uu_trace}
    \lean{Z3Anomalous.uu_trace_evalWord, Z3Anomalous.dd_trace_evalWord, Z3Anomalous.mpo_uu,
        Z3Anomalous.mpo_dd}
    \leanok
    \uses{thm:asymex_z3anomalous_uu_compression, prop:asym_compression_trace,
        thm:asym_mpo_product}
    For every nonempty word $w$, $\tr(B^w)=\tr((M^\dagger)^w)$, where $B=M\cdot M$.
    Similarly, $\tr((M^\dagger\cdot M^\dagger)^w)=\tr(M^w)$. Hence
    $O_L(M)O_L(M)=O_L(M^\dagger)$ and
    $O_L(M^\dagger)O_L(M^\dagger)=O_L(M)$ at every positive length $L$.
\end{theorem}

\begin{proof}
    \leanok
    Proposition~\ref{prop:asym_compression_trace} applied to
    Theorem~\ref{thm:asymex_z3anomalous_uu_compression} gives the trace identities, and
    Theorem~\ref{thm:asym_mpo_product} gives the stated identities of periodic operators.
\end{proof}

\begin{theorem}[Compression of \texorpdfstring{$M\cdot M^\dagger$ and $M^\dagger\cdot M$}{the inverse-factor products}]
    \label{thm:asymex_z3anomalous_ud_compression}
    \lean{Z3Anomalous.ud_compression, Z3Anomalous.ud_isReduction, Z3Anomalous.ud_right_eq,
        Z3Anomalous.ud_left_eq, Z3Anomalous.ud_dim_eq,
        Z3Anomalous.ud_evalWord_remainder_eq_zero, Z3Anomalous.ud_remainder_mul_ne_zero,

    \leanok
    \uses{def:asymex_z3anomalous_tensors, thm:asym_multiblock}
    The stacked tensor $B=M\cdot M^\dagger$ has the block triangular form of
    Theorem~\ref{thm:asym_multiblock} for the single target $\delta$ with $z=3$
    one-dimensional zero diagonal blocks, $4=1+3$. The compression pair satisfies
    $WV=\Id$ and $WB^wV=\delta^w$ for every word, and the remainder is nilpotent of
    order exactly three, one short of the general bound $|S|+z=4$.
    Here $|S|=1$ is the number of target slots.
    The stacked tensor $M^\dagger\cdot M$ has the analogous block triangular
    form for the same target $\delta$. Both products use the compression pair
    \begin{align}
        V & =\begin{pmatrix}1\\0\\0\\1\end{pmatrix}, &
        W & =\begin{pmatrix}0&0&0&1\end{pmatrix}.
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    Use the conjugation argument of
    Theorem~\ref{thm:asymex_z3anomalous_uu_compression}, over $\Z[\omega]$,
    with two one-dimensional zero diagonal blocks before the target and
    one after. The target rows and columns give $WV=\Id$ and
    $WB^wV=\delta^w$. Direct multiplication gives zero for every triple
    product of remainder letters. The products $R^0R^4$ for
    $M\cdot M^\dagger$ and $R^0R^0$ for $M^\dagger\cdot M$ are nonzero,
    proving that the order is exactly three in both cases.
\end{proof}

\begin{theorem}[The exact \texorpdfstring{$\Z_3$}{Z3} representation]
    \label{thm:asymex_z3anomalous_group_law}
    \lean{Z3Anomalous.ud_trace_evalWord, Z3Anomalous.du_trace_evalWord, Z3Anomalous.mpo_ud,
        Z3Anomalous.mpo_du, Z3Anomalous.mpo_u_mul_uDag, Z3Anomalous.mpo_uDag_mul_u,
        Z3Anomalous.mpo_u_pow_three, Z3Anomalous.mpo_identityTensor}
    \leanok
    \uses{thm:asymex_z3anomalous_ud_compression, thm:asymex_z3anomalous_uu_trace,
        prop:asym_compression_trace, thm:asym_mpo_product}
    The periodic operator of the identity tensor is the identity map at every length,
    $O_L(\delta)=\Id$. For every nonempty word $w$, both
    $\tr((M\cdot M^\dagger)^w)$ and $\tr((M^\dagger\cdot M)^w)$ equal $\tr(\delta^w)$.
    Thus $O_L(M)O_L(M^\dagger)=O_L(M^\dagger)O_L(M)=\Id$ at every positive length.
    Together with $O_L(M)O_L(M)=O_L(M^\dagger)$, this gives $O_L(M)^3=\Id$:
    the symmetry gives a representation of $\Z_3$ without any length-dependent phase.
\end{theorem}

\begin{proof}
    \leanok
    The word evaluation of $\delta$ is the scalar $1$ when the input and
    output configurations agree, and $0$ otherwise. Hence $O_L(\delta)=\Id$,
    including at length zero. Proposition~\ref{prop:asym_compression_trace}
    applied to Theorem~\ref{thm:asymex_z3anomalous_ud_compression} gives the
    nonempty-word trace identities. By Theorem~\ref{thm:asym_mpo_product},
    they imply $O_L(M)O_L(M^\dagger)=\Id$ and
    $O_L(M^\dagger)O_L(M)=\Id$. Finally,
    Theorem~\ref{thm:asymex_z3anomalous_uu_trace} gives
    $O_L(M)^3=O_L(M)^2O_L(M)=O_L(M^\dagger)O_L(M)=\Id$.
\end{proof}

\begin{theorem}[Both sitewise intertwiner spaces vanish on every nontrivial block]
    \label{thm:asymex_z3anomalous_no_intertwiner}
    \lean{Z3Anomalous.uu_right_intertwiner_eq_zero, Z3Anomalous.uu_left_intertwiner_eq_zero,
        Z3Anomalous.dd_right_intertwiner_eq_zero, Z3Anomalous.dd_left_intertwiner_eq_zero,
        Z3Anomalous.ud_right_intertwiner_eq_zero, Z3Anomalous.ud_left_intertwiner_eq_zero,
        Z3Anomalous.du_right_intertwiner_eq_zero, Z3Anomalous.du_left_intertwiner_eq_zero}
    \leanok
    \uses{def:asymex_z3anomalous_tensors}
    For each of the four stacked products $M\cdot M$, $M\cdot M^\dagger$,
    $M^\dagger\cdot M$, $M^\dagger\cdot M^\dagger$, let $B$ denote the stacked tensor
    and let $C$ denote its respective target $M^\dagger$, $\delta$, $\delta$, $M$.
    There is no nonzero matrix $X$ with $B^aX=XC^a$ for every letter $a$,
    and no nonzero matrix $Y$ with $YB^a=C^aY$ for every letter $a$.
    In particular, the compression identities $WB^wV=C^w$ do not give
    sitewise intertwiners.
\end{theorem}

\begin{proof}
    \leanok
    Select eight scalar equations for the equal-factor products and
    four for the inverse-factor products. Write the selected sitewise
    intertwining equations as $Kx=0$, where $x$ lists
    the entries of $X$ or $Y$. The coefficient matrix $K$ has entries in
    $\Z[\omega]$ and admits a matrix $H$ with $HK=c\Id$ for a nonzero
    scalar $c$. These identities are verified over $\Z[\omega]$.
    Consequently $cx=HKx=0$, and all entries of the intertwiner vanish.
\end{proof}

\begin{remark}[Failure of closure under conjugate transposition]
    \label{rem:asymex_z3anomalous_anomaly}
    \uses{cor:asym_anomaly_not_star}
    By Corollary~\ref{cor:asym_anomaly_not_star},
    Theorem~\ref{thm:asymex_z3anomalous_no_intertwiner} shows that the
    algebra generated by each stacked tensor is not closed under
    conjugate transposition. This concerns the virtual matrices, rather than
    the physical operator or a $3$-cocycle associator.
\end{remark}

\subsection{The condensation defect}%
\label{sec:asymex_z3anomalous_defect}

The condensation defect of the anomalous symmetry is the sum of its three
group operators, $\Id+U_N+U_N^\dagger$, in the terminology
of~\cite{Roumpedakis2023Condensation}. It is realized by the direct sum
of the three tensors. On the physical side the defect squares to three
times itself; on the virtual side the stacked square splits into nine
pair blocks, each compressing onto one group tensor.

\begin{definition}[The condensation defect]
    \label{def:asymex_z3anomalous_defect}
    \lean{Z3Anomalous.summandDim, Z3Anomalous.summandMPS, Z3Anomalous.defectTensor,
        Z3Anomalous.defectMPS}
    \leanok
    \uses{def:asymex_z3anomalous_tensors, def:mpdo_operator_product}
    The condensation defect is the block-diagonal tensor
    $A=\delta\oplus M\oplus M^\dagger$ of bond dimension five.
    Its three diagonal blocks have bond dimensions one, two and two
    and represent $\Id$, $U_L$ and $U_L^\dagger$, respectively,
    at positive ring length $L$.
\end{definition}

\begin{theorem}[The defect is the sum of the three group elements]
    \label{thm:asymex_z3anomalous_defect_sum}
    \lean{Z3Anomalous.defectMPS_trace_evalWord, Z3Anomalous.mpo_defectTensor}
    \leanok
    \uses{def:asymex_z3anomalous_defect}
    For every word $w$, $\tr(A^w)=\tr(\delta^w)+\tr(M^w)+\tr((M^\dagger)^w)$; that is,
    $O_L(A)=O_L(\delta)+O_L(M)+O_L(M^\dagger)$ at every length $L$.
\end{theorem}

\begin{proof}
    \leanok
    Products of block-diagonal matrices remain block diagonal, so
    $A^w=\delta^w\oplus M^w\oplus(M^\dagger)^w$.
    Taking the trace gives the word identity and hence the periodic
    operator identity.
\end{proof}

\begin{lemma}[The normalized defect projects onto invariant states]
    \label{lem:asymex_z3_defect_projector}
    For $L>0$, the operator
    $P_L=O_L(A)/3=(\Id+U_L+U_L^2)/3$ is the orthogonal projector onto
    $\{\psi:U_L\psi=\psi\}$. Its basis action is
    \begin{align}
        P_L\ket{t}
         & =\frac{1}{3}\left(\ket{t}+\omega^{F(t)}\ket{Xt}
        +\omega^{-F(X^{-1}t)}\ket{X^{-1}t}\right).
    \end{align}
    In particular, $O_L(A)^2=3O_L(A)$ and $\rank P_L=3^{L-1}$.
\end{lemma}

\begin{proof}
    Theorem~\ref{thm:asymex_z3anomalous_defect_sum} and
    Theorem~\ref{thm:asymex_z3_physical_kernels}
    and Lemma~\ref{lem:asymex_z3_physical_cube} give the formula for $P_L$.
    Since $U_L^3=\Id$, each power $\Id,U_L,U_L^2$ occurs three times
    in $(\Id+U_L+U_L^2)^2$, so $P_L^2=P_L$.
    Also $U_L^\dagger=U_L^2$, hence $P_L^\dagger=P_L$.
    The identities $U_LP_L=P_L$ and
    $P_L\psi=\psi$ whenever $U_L\psi=\psi$ identify its range.
    Neither $X$ nor $X^2$ fixes a configuration when $L>0$.
    Thus $\tr U_L=\tr U_L^2=0$, and
    $\rank P_L=\tr P_L=3^L/3$.
\end{proof}

The physical identity has a corresponding virtual decomposition.
The stacked square has nine pair blocks, each reducing to one of the
three group tensors. The asymmetric compression records these nine
targets separately.

\begin{theorem}[Block decomposition of the stacked square]
    \label{thm:asymex_z3anomalous_defect_square_blocks}
    \lean{Z3Anomalous.defectSquare, Z3Anomalous.defectSquareEis_eq, Z3Anomalous.pairStackEis,
        Z3Anomalous.squareBond}
    \leanok
    \uses{def:asymex_z3anomalous_defect, def:mpdo_operator_product}
    Write $M_0=\delta$, $M_1=M$, $M_2=M^\dagger$, and
    $B_{gh}=(M_g\cdot M_h)^a$ for the pair block at a fixed letter $a$.
    Reindex the bond coordinate $p=5p_1+p_2$ by the summand pair and its
    internal coordinates. The stacked tensor $A\cdot A$ of bond dimension
    twenty-five is then block diagonal over the nine pairs
    $(g,h)\in\{0,1,2\}^2$, with $(g,h)$ block $M_g\cdot M_h$.
    In these coordinates, for every letter $a$,
    \begin{align}
        (A\cdot A)^{a} & =
        \begin{pmatrix}
            B_{00}                                                                   \\
             & B_{01}                                                                \\
             &        & B_{02}                                                       \\
             &        &        & B_{10}                                              \\
             &        &        &        & B_{11}                                     \\
             &        &        &        &        & B_{12}                            \\
             &        &        &        &        &        & B_{20}                   \\
             &        &        &        &        &        &        & B_{21}          \\
             &        &        &        &        &        &        &        & B_{22}
        \end{pmatrix},
        \label{eq:asymex_z3anomalous_defect_square_blocks}
    \end{align}
    with all off-diagonal blocks zero.
\end{theorem}

\begin{proof}
    \leanok
    Each letter of $A$ preserves its three summands. The tensor product
    of two such letters therefore preserves each of the nine summand
    pairs. In the pair coordinates, its restriction is
    $M_g^{o,k}\otimes M_h^{k,i}$. Summing over $k$ gives
    $(M_g\cdot M_h)^{o,i}$, as in
    (\ref{eq:asymex_z3_stacked_product}). The specified reindexing and
    the resulting entry identities are verified over $\Z[\omega]$.
\end{proof}

\begin{theorem}[The defect squares to three times itself]
    \label{thm:asymex_z3anomalous_defect_square}
    \lean{Z3Anomalous.defectSquare_trace_evalWord, Z3Anomalous.mpo_defect_mul_defect}
    \leanok
    \uses{thm:asymex_z3anomalous_defect_square_blocks, thm:asymex_z3anomalous_defect_sum,
        prop:asym_compression_trace}
    For every nonempty word $w$, $\tr((A\cdot A)^w)=3\bigl(\tr(\delta^w)+\tr(M^w)+
        \tr((M^\dagger)^w)\bigr)$; that is, $O_L(A)O_L(A)=3\,O_L(A)$ at every positive length $L$.
\end{theorem}

\begin{proof}
    \leanok
    By Theorem~\ref{thm:asymex_z3anomalous_defect_square_blocks},
    the trace of $(A\cdot A)^w$ is the sum of the nine pair-block
    traces. Theorems~\ref{thm:asymex_z3anomalous_uu_trace}
    and~\ref{thm:asymex_z3anomalous_group_law} give
    $\tr((M_g\cdot M_h)^w)=\tr(M_{g+h}^w)$ on the four nontrivial
    pairs. For $g=0$ or $h=0$, the pair block equals the target under
    the natural bond identification.
    Each group element occurs as $g+h$ exactly three times, so
    \begin{align}
        \tr((A\cdot A)^w)
         & =\sum_{g,h=0}^{2}\tr(M_{g+h}^w)
        =3\sum_{k=0}^{2}\tr(M_k^w).
    \end{align}
    Theorem~\ref{thm:asym_mpo_product} converts this trace identity into
    the periodic operator identity.
\end{proof}

\begin{theorem}[Compression of the stacked square]
    \label{thm:asymex_z3anomalous_defect_compression}
    \lean{Z3Anomalous.pairTarget, Z3Anomalous.defectSquare_compression,
        Z3Anomalous.defectSquare_isReduction, Z3Anomalous.defectSquare_left_mul_right_of_ne,
        Z3Anomalous.defectSquare_dim_eq, Z3Anomalous.defectSquare_evalWord_remainder_eq_zero,
        Z3Anomalous.sum_pairTargetDim}
    \leanok
    \uses{thm:asymex_z3anomalous_defect_square_blocks, thm:asym_multiblock,
        thm:asymex_z3anomalous_uu_compression, thm:asymex_z3anomalous_ud_compression,
        thm:asymex_explicit_gauge}
    The stacked tensor $A\cdot A$ has the block triangular form of
    Theorem~\ref{thm:asym_multiblock} for the nine targets $M_{g+h}$,
    $(g,h)\in\{0,1,2\}^2$, with $z=10$ one-dimensional zero diagonal
    blocks and $25=15+10$. The nine compression pairs are mutually
    biorthogonal, and the remainder is nilpotent of order at most nineteen.
    Explicitly, writing $s=(g,h)$ and $C_s=M_{g+h}$, the pairs satisfy
    $W_s(A\cdot A)^wV_s=C_s^w$ for every word and $W_sV_t=0$ for $s\neq t$.
    The remainder is
    \begin{align}
        R^a & =(A\cdot A)^a-\sum_s V_sC_s^aW_s,
    \end{align}
    and $R^w=0$ whenever $|w|\geq19$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    On the five pairs with $g=0$ or $h=0$, the pair block equals its
    target under the natural bond identification and requires no zero
    diagonal block. On the other four pairs, use the gauges of
    Theorem~\ref{thm:asymex_z3anomalous_uu_compression} or
    Theorem~\ref{thm:asymex_z3anomalous_ud_compression}.
    Their direct sum, transported along the bond reindexing of
    Theorem~\ref{thm:asymex_z3anomalous_defect_square_blocks}, gives
    the required triangular form. Distinct pair blocks have disjoint
    coordinates, which gives $W_sV_t=0$ for $s\neq t$.

    The original block dimensions sum to
    $1+2+2+2+4+4+2+4+4=25$.
    The target dimensions sum to $1+2+2+2+2+1+2+1+2=15$,
    and the zero diagonal blocks number $0+0+0+0+2+3+0+3+2=10$.
    Clause~(vi) of Theorem~\ref{thm:asym_multiblock} bounds the
    nilpotency order by $|S|+z=9+10=19$, where $S$ is the set of
    nine target slots.
\end{proof}
