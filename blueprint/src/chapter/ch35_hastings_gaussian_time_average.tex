\section{The Gaussian time average in Hastings' approximate ground-state projector}
\label{sec:hastings_gaussian_time_average}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 843--864 (step five of the
% proof of Lemma 1). Finite complex matrices; no lattice content.

The last step of the proof of Lemma~1 of \cite{Hastings2007AreaLaw} replaces
the ground-state projector by a Gaussian average of the evolution, and then
replaces the Hamiltonian in that average by a nearby operator. This section
states both estimates for finite complex matrices, with explicit constants in
the parameter $q$. Norms are operator norms.

\begin{definition}[Gaussian time average]
    \label{def:hastings_gaussian_time_average}
    \lean{TNLean.MPS.AreaLaw.Hastings.timeAverageVariance,
        TNLean.MPS.AreaLaw.Hastings.gaussianTimeAverage}
    \leanok
    Let $q$ and $\Delta E$ be real numbers and put $h=\max(q/\Delta E^2,0)$ if
    $\Delta E\ne0$, and $h=0$ if $\Delta E=0$.
    For a complex square matrix $H$ let
    \begin{align*}
        P_q(H)=\int e^{iHt}\,d\gamma_h(t),
    \end{align*}
    where $\gamma_h$ is the centred Gaussian measure of variance $h$, and
    $\gamma_0$ is the point mass at zero. For $q>0$ and $\Delta E>0$ this is
    the operator
    $P_q=\frac{\Delta E}{\sqrt{2\pi q}}\int e^{iHt}e^{-(t\Delta E)^2/2q}\,dt$
    of \cite[lines 843--847]{Hastings2007AreaLaw}.
\end{definition}

\begin{theorem}[Gaussian time average across the gap]
    \label{thm:hastings_gaussian_time_average_gap}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_gaussianTimeAverage_sub_groundProjector_le}
    \leanok
    \uses{def:hastings_gaussian_time_average}
    Let $H$ be Hermitian, let $\Omega$ be a unit vector with $H\Omega=0$, let
    $\Delta E>0$, and suppose that
    $H\ge\Delta E\,(1-|\Omega\rangle\langle\Omega|)$. Then for every real $q$,
    \begin{align*}
        \bigl\|P_q(H)-|\Omega\rangle\langle\Omega|\bigr\|\le e^{-q/2}.
    \end{align*}
    This is the estimate of $\|P_q-P_0\|$ at
    \cite[line 848]{Hastings2007AreaLaw}, for the unique gapped ground state
    with energy zero. With the source's choice $q=(l/3)\Delta E/(2v)$ it
    gives only $O(e^{-l/6\xi})$, not the printed $O(e^{-l/3\xi})$; the same
    factor enters the two preceding estimates on the filtered pieces of the
    Hamiltonian and the margin of the spectral cut-offs. Taking
    $q=(l/3)\Delta E/v$ recovers the printed rate,
    see~\cite{gap:hastings07_gaussian_average_exponent}.
\end{theorem}
\begin{proof}\leanok
    \uses{def:hastings_gaussian_time_average}
    Let $(e_k)$ be an orthonormal eigenbasis of $H$ with eigenvalues
    $\lambda_k$. The Gaussian characteristic function gives
    $\langle e_k,P_q(H)x\rangle=e^{-h\lambda_k^2/2}\langle e_k,x\rangle$.
    Each $e_k$ is either a multiple of $\Omega$ with $\lambda_k=0$, or is
    orthogonal to $\Omega$ with $\lambda_k\ge\Delta E$. Hence the coefficients
    of $P_q(H)x-\langle\Omega,x\rangle\Omega$ are bounded by
    $e^{-h\Delta E^2/2}$ times those of $x$, and Parseval's identity gives
    the bound $e^{-h\Delta E^2/2}\|x\|$. Finally $h\Delta E^2\ge q$.
\end{proof}

\begin{theorem}[Comparison of two unitary evolutions]
    \label{thm:hastings_unitary_path_difference}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_hermitianUnitaryPath_sub_le}
    \leanok
    For Hermitian matrices $M$ and $H$ and every real $t$,
    \begin{align*}
        \|e^{iMt}-e^{iHt}\|\le\|M-H\|\,|t|.
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    By the Duhamel formula,
    $e^{-iMt}e^{iHt}-1=\int_0^te^{-iMs}\,i(H-M)\,e^{iHs}\,ds$. The integrand
    has norm $\|M-H\|$, and multiplication by the unitary $e^{iMt}$ preserves
    the norm.
\end{proof}

\begin{theorem}[First absolute moment of a Gaussian tail]
    \label{thm:hastings_gaussian_absolute_moment_tail}
    \lean{TNLean.Lattice.integral_compl_Icc_abs_gaussianReal_le}
    \leanok
    For $h,T\ge0$,
    \begin{align*}
        \int_{|t|>T}|t|\,d\gamma_h(t)
        \le \sqrt{2h/\pi}\,e^{-T^2/(2h)},
    \end{align*}
    where the right-hand side is interpreted as zero when $h=0$.
\end{theorem}
\begin{proof}\leanok
    If $h=0$, the measure is concentrated at zero and the integral vanishes.
    For $h>0$, write $\varphi_h(t)=(2\pi h)^{-1/2}e^{-t^2/(2h)}$.
    Since $(-h\varphi_h)'(t)=t\varphi_h(t)$ and $\varphi_h(t)\to0$ as
    $t\to\infty$, one has $\int_T^\infty t\varphi_h(t)\,dt=h\varphi_h(T)$.
    Reflection gives the same contribution on $(-\infty,-T)$, hence the
    asserted bound, with equality for $h>0$.
\end{proof}

\begin{theorem}[First absolute moment of a Gaussian]
    \label{thm:hastings_gaussian_absolute_moment}
    \lean{TNLean.Lattice.integral_abs_gaussianReal_le_sqrt_two_mul_div_pi,
        TNLean.Lattice.integral_abs_gaussianReal_le}
    \leanok
    For every $h\ge0$,
    $\int|t|\,d\gamma_h(t)\le\sqrt{2h/\pi}\le\sqrt h$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_gaussian_absolute_moment_tail}
    Specialize Theorem~\ref{thm:hastings_gaussian_absolute_moment_tail} to $T=0$.
    Since $|t|$ vanishes at zero, removing that point does not change the
    integral, even when $h=0$. Finally, $2\le\pi$ gives
    $\sqrt{2h/\pi}\le\sqrt h$.
\end{proof}

\begin{theorem}[Stability of the Gaussian time average]
    \label{thm:hastings_gaussian_time_average_stability}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_gaussianTimeAverage_sub_le,
        TNLean.MPS.AreaLaw.Hastings.norm_gaussianTimeAverage_sub_groundProjector_le_add}
    \leanok
    \uses{def:hastings_gaussian_time_average,
        thm:hastings_gaussian_time_average_gap}
    Let $M$ and $H$ be Hermitian, let $\Delta E>0$, let $q$ be real and write
    $\sqrt q$ for $\sqrt{\max(q,0)}$. Then
    \begin{align*}
        \|P_q(M)-P_q(H)\|\le\frac{\sqrt q}{\Delta E}\,\|M-H\|.
    \end{align*}
    If moreover $\Omega$ and $H$ satisfy the hypotheses of
    Theorem~\ref{thm:hastings_gaussian_time_average_gap}, then
    \begin{align*}
        \bigl\|P_q(M)-|\Omega\rangle\langle\Omega|\bigr\|
        \le e^{-q/2}+\frac{\sqrt q}{\Delta E}\,\|M-H\|.
    \end{align*}
    With $M=M_L+M_B+M_R$ this is the first inequality of the display of
    \cite[lines 849--864]{Hastings2007AreaLaw}, before the evolution of $M$
    is written in the interaction picture.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_unitary_path_difference,
        thm:hastings_gaussian_absolute_moment,
        thm:hastings_gaussian_time_average_gap}
    Integrating Theorem~\ref{thm:hastings_unitary_path_difference} against
    $\gamma_h$ gives
    $\|P_q(M)-P_q(H)\|\le\|M-H\|\int|t|\,d\gamma_h$, and
    Theorem~\ref{thm:hastings_gaussian_absolute_moment} bounds the integral
    by $\sqrt h=\sqrt q/\Delta E$. The second bound follows from the triangle
    inequality and Theorem~\ref{thm:hastings_gaussian_time_average_gap}.
\end{proof}
