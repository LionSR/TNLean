For a normal PEPS, the tensors become injective after blocking around every edge,
and each site separates two injective regions whose complements are also injective.
With union closure in hand, these hypotheses yield the Fundamental
Theorem for normal PEPS (Theorem~\ref{thm:fundamentalTheorem_normalPEPS_connected})
on a connected graph \cite[Section~4]{Molnar2018NormalPEPS}, with gauges unique
modulo balanced edge scalars.
Corollaries specialize the result to closed chains, translation-invariant matrix
product states, and the square lattice.

    \lean{TNLean.PEPS.%

    \lean{TNLean.PEPS.%

    \lean{TNLean.PEPS.%

    \lean{TNLean.PEPS.%

    \lean{TNLean.PEPS.%
        normalSquarePEPSBlockingHypotheses_mem_withSite_iff_of_marginCovers}
    \leanok
    \uses{thm:peps_normalSquareTranslatedEdgeBlockingHypotheses_of_windows, def:peps_normalPEPSBlockingHypotheses, def:peps_normalOneSiteSeparationHypotheses}
    Given rectangular injectivity hypotheses, finite union closure, oriented
    margin-and-cover data at every edge $e$, and one-site separation
    (Definition~\ref{def:peps_normalOneSiteSeparationHypotheses}), the finite
    square lattice satisfies Definition~\ref{def:peps_normalPEPSBlockingHypotheses}:
    at each edge $e=\{u_e,v_e\}$ the blocks $(R_e,B_e,C_e)$ satisfy
    $\inj(R_e)$, $\inj(B_e)$, $\inj(C_e)$, $u_e\in R_e$, $v_e\in B_e$,
    $R_e\cap B_e=R_e\cap C_e=B_e\cap C_e=\varnothing$, and
    $R_e\cup B_e\cup C_e=V$; at each site $v$ the regions $R_v,S_v$ satisfy
    $w\in S_v\Longleftrightarrow w=v\lor w\in R_v$.
\end{theorem}
\begin{proof}\leanok
    Pair the edge-blocking hypotheses of
    Theorem~\ref{thm:peps_normalSquareTranslatedEdgeBlockingHypotheses_of_windows}
    with the given one-site separation hypotheses
    (Definition~\ref{def:peps_normalOneSiteSeparationHypotheses}).
\end{proof}

\begin{theorem}[Fundamental Theorem for normal PEPS]%
    \label{thm:fundamentalTheorem_normalPEPS}
    \notready
    \uses{thm:peps_normalEdgeBlockingHypotheses_injectiveChain, def:peps_normalPEPSBlockingHypotheses, thm:peps_oneVertexComplementComparison, def:GaugeEquivPEPS}
    Suppose two normal PEPS $A$ and $B$ generate the same state, satisfy the
    three-site injective-chain blocking around every edge, and at every site $v$
    admit injective regions $R_v$ and $S_v=R_v\cup\{v\}$ ($v\notin R_v$) with
    injective complements. Then $A$ and $B$ are related by a local gauge
    unique up to the scalar freedom of the graph
    \cite[Section~4, following Theorem~3]{Molnar2018NormalPEPS}.
    The connected-graph form with shared blocking and single-crossing edges is
    Theorem~\ref{thm:fundamentalTheorem_normalPEPS_connected}, with per-edge
    uniqueness in Theorem~\ref{thm:fundamentalTheorem_normalPEPS_gauge_unique}.
    The source's remark that translationally invariant systems admit
    translationally invariant gauges when the proportionality scalar $\lambda$
    is not absorbed is realized on the torus by
    Theorem~\ref{thm:fundamentalTheorem_normalTorusPEPS_unconditional}.
\end{theorem}

\begin{definition}[Doubly injective regions]%
    \label{def:peps_regionInjectivityDataPair}
    \lean{TNLean.PEPS.regionInjectivityDataPair}
    \leanok
    \uses{def:peps_regionInjectivityData}
    For two region-injectivity assignments $\inj_1,\inj_2$ on $V$, the doubly
    injective assignment $\inj_{1,2}$ is defined by
    $\inj_{1,2}(R)\Longleftrightarrow\inj_1(R)\wedge\inj_2(R)$.
\end{definition}

\begin{definition}[Two normal PEPS tensors]%
    \label{def:peps_normalPEPSPairHypotheses}
    \lean{TNLean.PEPS.NormalPEPSPairHypotheses}
    \leanok
    \uses{def:peps_normalPEPSBlockingHypotheses, def:peps_regionInjectivityDataPair}
    Two PEPS tensors $A$ and $B$ on a finite simple graph form a
    \emph{pair of normal PEPS tensors} if they satisfy the normal blocking
    hypotheses (Definition~\ref{def:peps_normalPEPSBlockingHypotheses}) for the
    doubly injective assignment $\inj_{A,B}$
    (Definition~\ref{def:peps_regionInjectivityDataPair}) and have positive
    bond dimensions $D_A(e)>0$ and $D_B(e)>0$ for every edge $e$, following
    \cite[Section~3]{Molnar2018NormalPEPS}. Because a zero-dimensional boundary
    bond leaves a region with no boundary configurations so that injectivity
    holds vacuously, positivity is recorded explicitly
    \cite{gap:peps_injective_ft_section3_route}.
\end{definition}

\begin{theorem}[Equal bond dimensions from the normal blocking hypotheses]%
    \label{thm:peps_bondDim_eq_of_normalBlocking}
    \lean{TNLean.PEPS.bondDim_eq_of_normalBlocking}
    \lean{TNLean.PEPS.bondDim_apply_eq_of_blockingData}
    \leanok
    \uses{def:peps_normalPEPSPairHypotheses, def:peps_normalPEPSBlockingHypotheses,
        def:peps_regionInjectivityDataPair}
    Let $A$ and $B$ be a pair of normal PEPS tensors
    (Definition~\ref{def:peps_normalPEPSPairHypotheses}) on a finite simple
    graph with physical dimension $d>0$, generating the same state, such that
    at every edge $e$ the single-crossing condition
    $\{g\in E:g\text{ joins }R_e\text{ and }B_e\}=\{e\}$ holds. Then
    $D_A(e)=D_B(e)$ for every edge $e$. The one-edge form assumes only the
    blocking datum at a single edge $e$ with $D_A,D_B>0$.
    In \cite[Section~3]{Molnar2018NormalPEPS}, this follows because the
    insertion correspondence $X\mapsto Y$ on the blocked three-site chain is an
    algebra isomorphism $\MN{D_A(e)}\cong\MN{D_B(e)}$.
    \begin{proof}
        \leanok
        \uses{thm:peps_edgeBlockedThreeSiteInjective, thm:peps_edgeBlockedInsertionAlgebraIsomorphism}
        Fix $e\in E$. Blocking $A$ and $B$ into $(R_e,B_e,C_e)$ yields two
        three-site chains on $K_3$ with injective super-sites
        (Theorem~\ref{thm:peps_edgeBlockedThreeSiteInjective}). The coarse state
        coefficients satisfy
        $\Coeff_{\mathrm{coarse}}(A)=c_A\Coeff_A$ and
        $\Coeff_{\mathrm{coarse}}(B)=c_B\Coeff_B$ with positive merge-collapse
        constants $c_A,c_B>0$ (products of bond dimensions on edges non-incident
        to the blocks). Rescaling one super-site of $A$ by $c_B$ and of $B$ by
        $c_A$ equates the coarse states while preserving bond dimensions and
        super-site injectivity. By
        Theorem~\ref{thm:peps_edgeBlockedInsertionAlgebraIsomorphism}, matched
        insertions on the coarse bond between $R_e$ and $B_e$ define an algebra
        isomorphism between the full coarse bond matrix algebras, forcing equal
        coarse bond dimensions. Under $\{g\in E:g\text{ joins }R_e\text{ and }B_e\}=\{e\}$, the coarse bond dimension is $D_A(e)$ (respectively
        $D_B(e)$), so $D_A(e)=D_B(e)$.
    \end{proof}
\end{theorem}

\begin{theorem}[Fundamental Theorem for normal PEPS on a connected graph]%
    \label{thm:fundamentalTheorem_normalPEPS_connected}
    \lean{TNLean.PEPS.fundamentalTheorem_normalPEPS}
    \leanok
    % Reading note: this is the general normal theorem stated after Theorem 3
    % of \cite{Molnar2018NormalPEPS} on a connected graph.  The shared
    % blockings and the single-crossing hypothesis are the source's blocking
    % hypothesis made explicit, not restrictions; both readings are recorded
    % in \path{docs/paper-gaps/peps_normal_ft_section3_route.tex}.
    \uses{def:peps_normalPEPSPairHypotheses, def:peps_normalPEPSBlockingHypotheses,
        def:peps_regionInjectivityDataPair, def:GaugeEquivPEPS}
    Let $A$ and $B$ be a pair of normal PEPS tensors
    (Definition~\ref{def:peps_normalPEPSPairHypotheses}) on a connected finite
    simple graph with physical dimension $d>0$, generating the same state, and
    satisfying the single-crossing condition
    $\{g\in E:g\text{ joins }R_e\text{ and }B_e\}=\{e\}$ at every edge $e$.
    Then $A$ and $B$ are gauge equivalent (Definition~\ref{def:GaugeEquivPEPS}):
    there exist invertible matrices $X_e$ ($e\in E$) such that $B_v=(A^X)_v$
    for every $v\in V$.
    At each edge $e$, the three-site isomorphism transfer $T_e$ on $R_1=R_e$
    satisfies
    \begin{align}
        C_A(R_1,e,X;\sigma,\tau)
         & =C_B(R_1,e,T_e(X);\sigma,\tau),
         &
        T_e(X)
         & =Z_e\,\iota_e(X)\,Z_e^{-1},
        \label{eq:peps_normal_ft_transfer}
    \end{align}
    for an invertible $Z_e$ and every insertion $X$ on $e$, where $D_A=D_B$
    follows from Theorem~\ref{thm:peps_bondDim_eq_of_normalBlocking}
    \cite[Section~4]{Molnar2018NormalPEPS} (cf.\ Theorem~\ref{thm:fundamentalTheorem_PEPS}).
    % Source-faithful reading: Section 4 of \cite{Molnar2018NormalPEPS} uses
    % one common blocking of the two states and enlarges the endpoint regions
    % so that they meet only along the distinguished edge.
    \begin{proof}
        \leanok
        \uses{thm:peps_bondDim_eq_of_normalBlocking, lem:peps_injectiveRegion_union_pos, thm:peps_region_edge_gauge_of_coeff_transfer, thm:gaugeConsistency}
        By Theorem~\ref{thm:peps_bondDim_eq_of_normalBlocking}, $D_A=D_B$, and
        by Lemma~\ref{lem:peps_injectiveRegion_union_pos}, injective regions of
        $A$ and $B$ are closed under unions. At each $e\in E$,
        Theorem~\ref{thm:peps_region_edge_gauge_of_coeff_transfer} yields an
        invertible gauge $X_e$ via (\ref{eq:peps_normal_ft_transfer}) whose
        orientation-adapted absorption $\widetilde B=B^X$ satisfies
        $C_A(e,M;\sigma)=C_{\widetilde B}(e,M;\sigma)$ for all $M$ and $\sigma$.
        Since gauge absorption preserves region injectivity, comparing $R_v$ and
        $S_v=R_v\cup\{v\}$ at each $v\in V$ yields nonzero scalars $c_R(v),c_S(v)$
        and $\lambda_v=c_S(v)/c_R(v)\ne 0$ such that
        $A_v=\lambda_v\widetilde B_v$ (with scalar $1$ when $R_v=\varnothing$ or
        $S_v=V$, and a boundary edge guaranteed by connectivity otherwise).
        Contracting against the nonzero closed state gives
        $\prod_{v\in V}\lambda_v=1$; a spanning-tree factorization on the
        connected graph absorbs $\lambda_v^{-1}$ into the edge gauges as in
        Theorem~\ref{thm:gaugeConsistency}.
    \end{proof}
\end{theorem}

\begin{theorem}[Uniqueness of the normal-PEPS gauges up to a constant]%
    \label{thm:fundamentalTheorem_normalPEPS_gauge_unique}
    \lean{TNLean.PEPS.fundamentalTheorem_normalPEPS_gauge_unique}
    \leanok
    \uses{def:peps_normalPEPSPairHypotheses, def:peps_normalPEPSBlockingHypotheses,
        def:peps_regionInjectivityDataPair, def:gaugeVertex}
    Let $A$ and $B$ be a pair of normal PEPS tensors
    (Definition~\ref{def:peps_normalPEPSPairHypotheses}) on a finite simple
    graph with $D_A=D_B$. If two families of invertible edge matrices $X$ and
    $X'$ satisfy $(A^X)_v=B_v=(A^{X'})_v$ for all $v\in V$
    (Definition~\ref{def:gaugeVertex}), then for every $e\in E$ there exists
    $c_e\in\C^\times$ such that $X'_e=c_e X_e$
    \cite[Section~4, following Theorem~3]{Molnar2018NormalPEPS}. Neither same-state
    equality nor graph connectivity is needed.
    \begin{proof}
        \leanok
        \uses{thm:peps_gaugeConj_eq_of_coeffIdentities, lem:peps_injectiveRegion_union_pos}
        Multilinearity of the edge-inserted coefficient in the site tensors gives
        $C_B(e,M;\sigma)=C_{A^X}(e,M;\sigma)=C_{A^{X'}}(e,M;\sigma)$ at every
        $e\in E$. By the blocking hypotheses and
        Lemma~\ref{lem:peps_injectiveRegion_union_pos}, both $R_e$ and
        $V\setminus R_e=B_e\cup C_e$ are injective for $A$. By
        Theorem~\ref{thm:peps_gaugeConj_eq_of_coeffIdentities}, the orientation-adapted
        absorbing gauges $X_{\mathrm{abs},e}$ and $X'_{\mathrm{abs},e}$ induce
        equal conjugations on $\MN{D_A(e)}$, so
        $X'_{\mathrm{abs},e}=c_e X_{\mathrm{abs},e}$ for some $c_e\in\C^\times$;
        unwinding the orientation adaptation gives $X'_e=c_e X_e$.
    \end{proof}
\end{theorem}
