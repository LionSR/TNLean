\section{Hastings' area law from the approximate factorization}
\label{sec:hastings_area_law}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 124--232 (main theorem) and
% 478--503 (the window condition of the entropy gain).

The contradiction argument of Section~\ref{sec:hastings_main_stage} takes the
window facts and the strengthened doubling inequality as hypotheses. This
section discharges all of them from the approximate factorization of the
ground-state projector, the first lemma of~\cite{Hastings2007AreaLaw}. The
removal of sites and the bounds on the window entropy come from
Section~\ref{ch:hastings_window}, and the doubling inequality comes from the
entropy gain across a cut of Section~\ref{sec:hastings_relent}, whose proof
contains the bootstrap lemma.

The entropy gain is proved when the entropy of the cut is large compared
with the remainder $F(\xi',q)$ of the bootstrap lemma
(Definition~\ref{def:hastings_bootstrap_remainder}). The source meets this
condition at~\cite[lines 480--484]{Hastings2007AreaLaw}: when it fails, the
bound $S_{\max}$ is small, and the constant $c_0$ in $S_{\max}$ is chosen
large enough. Here the window is lengthened instead, which needs the
contradiction argument for every window at least as long as the doubling
window.

\begin{theorem}[The contradiction argument for a longer window]
    \label{thm:hastings_main_stage_long_window}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_prefix_le_of_windowClaim_of_le}
    \leanok
    \uses{def:chain_model,def:hastings_max_entropy_bound}
    Theorem~\ref{thm:hastings_main_stage} holds for every window length
    $l_0\ge\lceil\xi_0\rceil_+2^M$, with $S_{\max}=3l_0\ln q$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_doubling_window,thm:hastings_main_stage}
    The proof of Theorem~\ref{thm:hastings_main_stage} goes through
    unchanged. At its end, Theorem~\ref{thm:hastings_doubling_window} gives
    $l_0<\lceil\xi_0\rceil_+2^M$, which contradicts the choice of $l_0$.
\end{proof}

\begin{definition}[The entropy bound of the area law]
    \label{def:hastings_area_law_bound}
    \lean{TNLean.MPS.AreaLaw.Hastings.areaLawWindowLength,
        TNLean.MPS.AreaLaw.Hastings.areaLawBound}
    \leanok
    \uses{def:hastings_max_entropy_bound,def:hastings_bootstrap_remainder}
    Let $\xi'>0$, $C_1\ge1$ and $q\ge2$. Let $\lceil\xi_0\rceil_+2^M$ be the
    doubling window of Definition~\ref{def:hastings_max_entropy_bound} for
    the constants $2C_1$ and $C_2=7$ of the entropy gain, and put
    \begin{align*}
        l_0      & =\max\Bigl\{\lceil\xi_0\rceil_+2^M,\
        \Bigl\lceil\frac23\Bigl(\xi'\ln C_1+\frac{F(\xi',q)}{\ln q}\Bigr)
        \Bigr\rceil_+\Bigr\},                           \\
        S_{\max} & =3l_0\ln q .
    \end{align*}
\end{definition}

\begin{theorem}[Size of the entropy bound of the area law]
    \label{thm:hastings_area_law_bound_le}
    \lean{TNLean.MPS.AreaLaw.Hastings.areaLawBound_le}
    \leanok
    \uses{def:hastings_area_law_bound}
    For $q\ge2$, the bound $S_{\max}$ of
    Definition~\ref{def:hastings_area_law_bound} is at most the bound of
    Definition~\ref{def:hastings_max_entropy_bound} for the constants $2C_1$
    and $7$, plus $2(\xi'\ln C_1\ln q+F(\xi',q))+3\ln q$. The added term is the
    part that~\cite[lines 480--484]{Hastings2007AreaLaw} absorbs into the
    constant $c_0$.
\end{theorem}
\begin{proof}\leanok
    The maximum of two nonnegative numbers is at most their sum, and
    $3\ln q\,\lceil x\rceil_+\le 3x\ln q+3\ln q$ for $x\ge0$.
\end{proof}

\begin{theorem}[Hastings' area law, with the approximate factorization as
        hypothesis]
    \label{thm:hastings_area_law_from_factorization}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_prefix_le_of_factorization}
    \leanok
    \uses{def:chain_model,def:hastings_area_law_bound}
    Let $\Omega$ be a gapped ground vector of a finite-range Hamiltonian on a
    chain of $L$ sites with local dimension $q$. Let $\xi'>0$ and $C_1\ge1$,
    and suppose that for every cut $c$ and every $l$ with $c+l\le L$ there are
    operators $O_B$, $O_L$, $O_R$ with these properties:
    \begin{itemize}
        \item $O_L$ and $O_R$ are commuting orthogonal projections acting on
              the first $c$ sites and on the remaining sites;
        \item $\|O_B\|\le1$, and $O_B$ acts on the sites of
              $\lbrack c-l,c+l)$ in the chain;
        \item $\|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|\le C_1e^{-l/\xi'}$.
    \end{itemize}
    Then the entropy of the first $i$ sites is at most $S_{\max}$ for every
    $i$, with $S_{\max}$ as in Definition~\ref{def:hastings_area_law_bound}.
    This is the main theorem of~\cite{Hastings2007AreaLaw}, equation (entbd),
    once the first lemma of the source supplies the operators. Requiring
    $C_1\ge1$ loses nothing, since enlarging $C_1$ only weakens that lemma.
    The bound $S_{\max}$ grows as $4^{\xi'\ln q}$, not $2^{\xi'\ln q}$, as
    recorded in~\cite{gap:hastings07_doubling_constants}, and the entropy gain
    enters with the constants $2C_1$ and $7$
    of~\cite{gap:hastings07_relent_constants}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_main_stage_long_window,
        thm:hastings_cut_entropy_slow,thm:hastings_window_entropy_bounds,
        thm:hastings_relent_window}
    For $q\le1$ every entropy vanishes. For $q\ge2$, apply
    Theorem~\ref{thm:hastings_main_stage_long_window} with the window length
    $l_0$ of Definition~\ref{def:hastings_area_law_bound} and the window
    entropy. Theorem~\ref{thm:hastings_cut_entropy_slow} gives the first
    item, and Theorem~\ref{thm:hastings_window_entropy_bounds} the second.
    For the third, fix a window all of whose cuts have entropy at least
    $2S_{\max}/3=2l_0\ln q$, and let $2l\le l_0$. Take an interval
    $\lbrack a,a+2l)$ of largest entropy in the window. The choice of $l_0$
    gives
    \begin{align*}
        (l+\xi'\ln C_1)\ln q+F(\xi',q)\le 2l_0\ln q,
    \end{align*}
    so Theorem~\ref{thm:hastings_relent_window} applies at the cut after
    $a+l$ sites, with the constants $2C_1$ and $7$. It bounds the entropy of
    $\lbrack a,a+2l)$ by the entropies of $\lbrack a,a+l)$ and
    $\lbrack a+l,a+2l)$, each at most the window entropy at length $l$, minus
    the gain, which is the doubling inequality.
\end{proof}
