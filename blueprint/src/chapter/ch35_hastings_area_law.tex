\section{Hastings' area law from the approximate factorization}
\label{sec:hastings_area_law}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 124--232 (main theorem) and
% 478--503 (the window condition of the entropy gain).

The contradiction argument of Section~\ref{sec:hastings_main_stage} takes the
window inequalities and the strengthened doubling bound as hypotheses. Here we
discharge them from the approximate projector factorization of
\cite[Lemma~1]{Hastings2007AreaLaw}, combining the window bounds of
Chapter~\ref{ch:hastings_window} with the entropy gain of
Section~\ref{sec:hastings_relent}. Whereas
\cite[lines 483--486]{Hastings2007AreaLaw} absorbs the bootstrap remainder
$F(\xi',q)$ (Definition~\ref{def:hastings_bootstrap_remainder}) into the
prefactor $c_0$ of $S_{\max}$, we lengthen the window $l_0\ge\lceil\xi_0\rceil_+2^M$.

\begin{lemma}[The contradiction argument for a longer window]
    \label{thm:hastings_main_stage_long_window}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_prefix_le_of_windowClaim_of_le}
    \leanok
    \uses{def:chain_model,def:hastings_max_entropy_bound,thm:hastings_main_stage}
    Lemma~\ref{thm:hastings_main_stage} holds for every window length
    $l_0\ge\lceil\xi_0\rceil_+2^M$, with $S_{\max}=3l_0\ln q$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_doubling_window,thm:hastings_prefix_entropy_le}
    If $i+l_0>L$, Lemma~\ref{thm:hastings_prefix_entropy_le} gives
    $S(\lbrack0,i))\le(L-i)_+\ln q\le l_0\ln q\le S_{\max}$. Otherwise, if
    $S(\lbrack0,i))>S_{\max}$, then for $i\le k\le i+l_0$ the site-removal
    hypothesis gives
    \begin{align}
        S(\lbrack0,k))\ge S(\lbrack0,i))-(k-i)\ln q>2l_0\ln q=\tfrac23\,S_{\max},
        \notag
    \end{align}
    so the strengthened doubling inequality holds on $[i,i+l_0)$ and
    Lemma~\ref{thm:hastings_doubling_window} yields
    $l_0<\lceil\xi_0\rceil_+2^M\le l_0$, a contradiction.
\end{proof}

\begin{definition}[The entropy bound of the area law]
    \label{def:hastings_area_law_bound}
    \lean{TNLean.MPS.AreaLaw.Hastings.areaLawWindowLength,
        TNLean.MPS.AreaLaw.Hastings.areaLawBound}
    \leanok
    \uses{def:hastings_max_entropy_bound,def:hastings_bootstrap_remainder}
    Let $\xi'>0$, $C_1\ge1$, and $q\ge2$. Let $\lceil\xi_0\rceil_+2^M$ be the
    doubling window of Definition~\ref{def:hastings_max_entropy_bound} for
    constants $2C_1$ and $C_2=7$, and define
    \begin{align}
        l_0 & =\max\Bigl\{\lceil\xi_0\rceil_+2^M,
        \Bigl\lceil\frac23\Bigl(\xi'\ln C_1+\frac{F(\xi',q)}{\ln q}\Bigr)
        \Bigr\rceil_+\Bigr\},                          \qquad
        S_{\max} =3l_0\ln q .
        \label{eq:hastings_area_law_bound}
    \end{align}
\end{definition}

\begin{lemma}[Size of the entropy bound of the area law]
    \label{thm:hastings_area_law_bound_le}
    \lean{TNLean.MPS.AreaLaw.Hastings.areaLawBound_le}
    \leanok
    \uses{def:hastings_area_law_bound,def:hastings_max_entropy_bound}
    For $\xi'>0$, $C_1\ge1$, and $q\ge2$, the bound $S_{\max}$ of
    Definition~\ref{def:hastings_area_law_bound} and the bound
    $S_{\max}^{\mathrm{dbl}}=3\lceil\xi_0\rceil_+2^M\ln q$ of
    Definition~\ref{def:hastings_max_entropy_bound} (with constants $2C_1$ and
    $7$) satisfy
    \begin{align}
        S_{\max}\le S_{\max}^{\mathrm{dbl}}+2\bigl(\xi'\ln C_1\ln q+F(\xi',q)\bigr)+3\ln q,
        \notag
    \end{align}
    where~\cite[lines 483--486]{Hastings2007AreaLaw} absorbs the added term into
    $c_0$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:hastings_bootstrap_remainder,def:hastings_max_entropy_bound}
    Use $\max(a,b)\le a+b$ for $a,b\ge0$ and
    $3\ln q\,\lceil x\rceil_+\le 3x\ln q+3\ln q$ for $x\ge0$ in
    (\ref{eq:hastings_area_law_bound}).
\end{proof}

\begin{lemma}[Hastings' area law, with the approximate factorization as
        hypothesis]
    \label{thm:hastings_area_law_from_factorization}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_prefix_le_of_factorization}
    \leanok
    \uses{def:chain_model,def:hastings_area_law_bound,
        def:hastings_factorization_at}
    Let $\Omega$ be a gapped ground vector of a finite-range Hamiltonian on a
    chain of $L$ sites with local dimension $q$, and let $\xi'>0$ and
    $C_1\ge1$. Suppose that for every cut $c$ and $l\in\N$ with $c+l\le L$
    there is an approximate factorization $(O_B,O_L,O_R)$ at $(c,l)$ with error
    $C_1e^{-l/\xi'}$ (Definition~\ref{def:hastings_factorization_at}). Then
    for every $i$, with $S_{\max}$ as in
    Definition~\ref{def:hastings_area_law_bound},
    \begin{align}
        S(\lbrack0,i))\le S_{\max}.
        \label{eq:hastings_area_law_entbd}
    \end{align}
    This derives eq.~(entbd) of \cite{Hastings2007AreaLaw} from Lemma~1, with
    base $4^{\xi'\ln q}$ in place of $2^{\xi'\ln q}$
    \cite{gap:hastings07_doubling_constants} and constants $2C_1$ and $7$ in
    the entropy gain \cite{gap:hastings07_relent_constants}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_main_stage_long_window,
        thm:hastings_cut_entropy_slow,thm:hastings_window_entropy_bounds,
        thm:hastings_relent_window,thm:hastings_window_claim_of_relent,
        lem:chain_entropy_inequalities}
    For $q\le1$, $S(\lbrack0,i))=0\le S_{\max}$. For $q\ge2$, apply
    Lemma~\ref{thm:hastings_main_stage_long_window} with $l_0$ from
    (\ref{eq:hastings_area_law_bound}) and the window entropy $S_l$:
    Lemma~\ref{thm:hastings_cut_entropy_slow} gives hypothesis~(1), and
    Lemma~\ref{thm:hastings_window_entropy_bounds} gives hypothesis~(2). For
    hypothesis~(3), if $S(\lbrack0,k))\ge2S_{\max}/3=2l_0\ln q$ on $[i,i+l_0]$
    and $2l\le l_0$, then~(\ref{eq:hastings_area_law_bound}) implies
    \begin{align}
        (l+\xi'\ln C_1)\ln q+F(\xi',q)\le 2l_0\ln q\le S(\lbrack0,a+l))
        \notag
    \end{align}
    for every $\lbrack a,a+2l)\subseteq\lbrack i,i+l_0)$, so
    Lemma~\ref{thm:hastings_relent_window} and
    Lemma~\ref{thm:hastings_window_claim_of_relent} give the doubling bound.
\end{proof}

\begin{theorem}[Hastings' area law for gapped chains]
    \label{thm:hastings_area_law}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_regionEntropy_prefix_le}
    \leanok
    \uses{def:chain_model,def:hastings_area_law_bound,
        thm:hastings_lemma1_endpoint}
    For every range $R\in\N$ there is a constant $K\ge0$ depending only on $R$
    with the following property. Let $J\ge0$, $\Delta E>0$, and let $\tilde\xi$
    be the decay length of Lemma~\ref{thm:hastings_lemma1_endpoint}
    (depending only on $R$ and $J/\Delta E$). For every chain of $L$ sites with
    local dimension $q$, every range-$R$ Hamiltonian with term norms at most
    $J$, and every ground vector $\Omega$ with spectral gap $\Delta E$,
    (\ref{eq:hastings_area_law_entbd}) holds for all $i$ with $S_{\max}$ from
    Definition~\ref{def:hastings_area_law_bound} at $\xi'=6\tilde\xi$ and
    $C_1=\max\{K\tilde\xi^3,1\}$. Thus $S_{\max}$ depends only on $R$,
    $J/\Delta E$, and $q$, and is independent of $L$, $H$, and $\Omega$.

    This is the main theorem of~\cite{Hastings2007AreaLaw}, eq.~(entbd), at the
    printed rate $\xi'=6\tilde\xi$ with polynomial prefactor
    $C_1=\max\{K\tilde\xi^3,1\}$ (lines 258--262). Here $\tilde\xi$ comes from
    the chain Lieb--Robinson estimate with $J$-dependent prefactor, single-site
    second operator, and velocity raised by $5/4$ rather than a Lieb--Robinson
    pair $(v,\xi_C)$ in the sense of lines
    96--112~\cite{gap:hastings07_lemma1_decay_length,
        gap:hastings07_gaussian_average_exponent}, and $S_{\max}$ grows as
    $4^{\xi'\ln q}$ rather than the printed
    $c_0\xi'\ln(\xi')\ln(D)2^{\xi'\ln D}$~\cite{gap:hastings07_doubling_constants,
        gap:hastings07_relent_constants}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint,
        thm:hastings_area_law_from_factorization,
        thm:hastings_factorization_mono}
    Lemma~\ref{thm:hastings_lemma1_endpoint} gives $K\ge0$ and factorizations
    at every $(c,l)$ with error $K\tilde\xi^3e^{-l/(6\tilde\xi)}\le C_1e^{-l/\xi'}$
    (Lemma~\ref{thm:hastings_factorization_mono}), so
    Lemma~\ref{thm:hastings_area_law_from_factorization} yields
    (\ref{eq:hastings_area_law_entbd}).
\end{proof}
