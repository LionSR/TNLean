\section{Entropy scaling and efficient approximation}

For a unit vector $\psi_N\in(\mathbb C^q)^{\otimes N}$, write
$\rho_{N,k}$ for its marginal on the first $k$ sites. All logarithms are
natural, and $\|\cdot\|_1$ denotes the trace norm.

\begin{definition}[Efficient approximation by matrix product states]
    \label{def:efficient_mps_approximation}
    \lean{TNLean.MPS.AreaLaw.IsEfficientlyApproximableByMPS}
    \leanok
    \uses{def:chain_open_mps}
    Following~\cite{Schuch2008EntropyScaling}, a family
    $(\psi_N)_{N\in\mathcal N}$ ($\mathcal N\subseteq\mathbb N$) is
    \emph{efficiently approximable by matrix product states} if for every
    $\delta>0$ there exist $C_\delta>0$ and $m_\delta\in\mathbb N$ such that
    each $N\in\mathcal N$ admits an (unnormalized) MPS $\phi_N$ of bond
    dimension at most $D_N$ with
    \begin{align}
        D_N                                    & \le C_\delta(N+1)^{m_\delta}, &
        \bigl\||\psi_N\rangle\langle\psi_N|
        -|\phi_N\rangle\langle\phi_N|\bigr\|_1 & \le\delta.
        \label{eq:efficient_mps_approximation}
    \end{align}
\end{definition}

\begin{theorem}[An entropy threshold for the discarded weight]
    \label{thm:renyi_entropy_tail_threshold}
    \lean{TNLean.MPS.AreaLaw.one_sub_kyFanNorm_le_of_renyiEntropy_le}
    \leanok
    For a density matrix $\rho$, $0<\alpha<1$, $D\ge1$, $T>0$, and tail weight
    $\tau_D(\rho)=1-\|\rho\|_{(D)}$, if
    \begin{align}
        S_\alpha(\rho)\le
        \log\frac{D}{1-\alpha}+\frac{\alpha}{1-\alpha}\log T,
        \label{eq:renyi_tail_threshold}
    \end{align}
    then $\tau_D(\rho)\le T$.
\end{theorem}
\begin{proof}
    \leanok
    For $\tau_D(\rho)>0$, Lemma~2 of~\cite{VerstraeteCirac2006Faithfully} and
    (\ref{eq:renyi_tail_threshold}) give
    \begin{align}
        \log\tau_D(\rho)
        \le\frac{1-\alpha}{\alpha}
        \left(S_\alpha(\rho)-\log\frac{D}{1-\alpha}\right)
        \le\log T.
        \label{eq:renyi_tail_log_bound}
    \end{align}
\end{proof}

\begin{theorem}[Normalized approximation on a finite chain]
    \label{thm:renyi_finite_mps_approximation}
    \lean{TNLean.MPS.AreaLaw.exists_normalized_isOpenMPS_of_renyiEntropy_le}
    \leanok
    \uses{def:chain_open_mps}
    Let $\|\psi_N\|=1$, $0<\alpha<1$, $D\ge1$, and $0<\delta\le1$. If for every
    $0<k<N$,
    \begin{align}
        S_\alpha(\rho_{N,k})\le
        \log\frac{D}{1-\alpha}
        +\frac{\alpha}{1-\alpha}\log\frac{\delta^2}{16(N+1)},
        \label{eq:renyi_finite_entropy_threshold}
    \end{align}
    then there exists a unit MPS $\phi_N$ of bond dimension at most $D$ with
    $\||\psi_N\rangle\langle\psi_N|-|\phi_N\rangle\langle\phi_N|\|_1\le\delta$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:renyi_entropy_tail_threshold,thm:chain_sequential_truncation_eigen,
        thm:chain_sequential_truncation_sq}
    By Theorem~\ref{thm:renyi_entropy_tail_threshold} and
    (\ref{eq:renyi_finite_entropy_threshold}), $\tau_D(\rho_{N,k})\le\delta^2/[16(N+1)]$.
    Theorems~\ref{thm:chain_sequential_truncation_eigen}
    and~\ref{thm:chain_sequential_truncation_sq} give an MPS $\Phi\ne0$ with
    $\|\Phi-\psi_N\|^2\le\delta^2/16$. Setting $\phi_N=\Phi/\|\Phi\|$ yields
    \begin{align}
        \|\phi_N-\psi_N\|
         & \le |1-\|\Phi\||+\|\Phi-\psi_N\|
        \le2\|\Phi-\psi_N\|\le\delta/2,
        \label{eq:renyi_normalized_vector_error}
    \end{align}
    and $\||\psi_N\rangle\langle\psi_N|-|\phi_N\rangle\langle\phi_N|\|_1
        \le2\|\phi_N-\psi_N\|\le\delta$.
\end{proof}

\begin{theorem}[Exact representation from marginal ranks]
    \label{thm:renyi_zero_exact_mps}
    \lean{TNLean.MPS.AreaLaw.isOpenMPS_of_regionState_rank_le}
    \leanok
    \uses{def:chain_open_mps}
    If $\|\psi_N\|=1$ and $\operatorname{rank}\rho_{N,k}\le D$ for every
    $0<k<N$, then $\psi_N$ is an MPS of bond dimension at most $D$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:chain_sequential_truncation_eigen,thm:chain_sequential_truncation_sq}
    Since $\tau_D(\rho_{N,k})=0$ for all $0<k<N$, sequential truncation
    (Theorems~\ref{thm:chain_sequential_truncation_eigen}
    and~\ref{thm:chain_sequential_truncation_sq}) gives $\|\Phi-\psi_N\|^2=0$.
\end{proof}

\begin{theorem}[Logarithmic R\'enyi entropy implies efficient approximation]
    \label{thm:renyi_family_mps_approximation}
    \lean{TNLean.MPS.AreaLaw.isEfficientlyApproximableByMPS_of_renyiEntropy_le}
    \leanok
    \uses{def:efficient_mps_approximation}
    If $(\psi_N)_{N\in\mathcal N}$ are unit vectors satisfying
    $S_\alpha(\rho_{N,k})\le c\log N+c'$ for all $0<k<N$, with $0\le\alpha<1$,
    $c\ge0$, and $c'\in\mathbb R$ independent of $N$, then
    $(\psi_N)_{N\in\mathcal N}$ is efficiently approximable by MPS
    \cite{Schuch2008EntropyScaling}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:renyi_finite_mps_approximation,thm:renyi_zero_exact_mps}
    For $\delta>0$, set $\varepsilon=\min(\delta,1)$, $t=\alpha/(1-\alpha)$, and
    \begin{align}
        m                                   & =\lceil c+t\rceil, &
        K                                   & =1+\left\lceil
        \exp\left(c'+t\log\frac{16}{\varepsilon^2}
        +\log(1-\alpha)\right)\right\rceil, &
        D_N                                 & =K(N+1)^m.
        \label{eq:renyi_family_parameters}
    \end{align}
    Then (\ref{eq:renyi_family_parameters}) implies
    \begin{align}
        c\log(N+1)+c'\le
        \log\frac{D_N}{1-\alpha}
        +t\log\frac{\varepsilon^2}{16(N+1)}.
        \label{eq:renyi_family_entropy_threshold}
    \end{align}
    Apply Theorem~\ref{thm:renyi_finite_mps_approximation} for $0<\alpha<1$ and
    Theorem~\ref{thm:renyi_zero_exact_mps} for $\alpha=0$ (where
    $S_0(\rho_{N,k})=\log\operatorname{rank}\rho_{N,k}\le\log D_N$).
\end{proof}

\begin{theorem}[Rényi lower bound at a chain cut]
    \label{thm:chain_cut_renyi_approximation_lower_bound}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.renyiEntropy_leftBlock_add_log_error_le_log}
    \leanok
    Let $\|\psi\|=1$ on $L$ sites, $\phi$ be an (unnormalized) open-boundary MPS
    of bond dimension at most $D$, $0<k<L$, $\alpha>1$, $\delta<1$, and
    $\lVert\ketbra{\psi}{\psi}-\ketbra{\phi}{\phi}\rVert_1\le\delta$. For
    $\rho_k=\operatorname{tr}_{\{k,\dots,L-1\}}\ketbra{\psi}{\psi}$,
    \begin{align}
        \label{eq:chain_cut_renyi_approximation_lower_bound}
        S_\alpha(\rho_k)+\frac{\alpha}{\alpha-1}\log(1-\delta) & \le\log D
    \end{align}
    \cite{Schuch2008EntropyScaling,gap:swvc08_renyi_bond_dimension_sign}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_reduced_state_rank_error}
    By Theorem~\ref{thm:chain_reduced_state_rank_error}, the prefix marginal of
    $\phi$ has rank at most $D$ and trace-norm distance at most $\delta$ from
    $\rho_k$; the finite-dimensional Rényi approximation inequality gives
    (\ref{eq:chain_cut_renyi_approximation_lower_bound}).
\end{proof}

\begin{theorem}[Rényi lower bound on an arbitrary contiguous block]
    \label{thm:chain_block_renyi_approximation_lower_bound}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.rank_regionState_sdiff_leftBlock_le_sq,
        TNLean.MPS.AreaLaw.IsOpenMPS.rank_regionState_leftBlock_le_max,
        TNLean.MPS.AreaLaw.IsOpenMPS.rank_regionState_sdiff_leftBlock_le_max_sq,
        TNLean.MPS.AreaLaw.IsOpenMPS.renyiEntropy_sdiff_leftBlock_add_log_error_le_two_log}
    \leanok
    Let $\|\psi\|=1$, $\phi$ be an (unnormalized) open-boundary MPS of bond
    dimension at most $D\ge0$ (with $\log0=0$), $\alpha>1$, $\delta<1$, and
    $\lVert\ketbra{\psi}{\psi}-\ketbra{\phi}{\phi}\rVert_1\le\delta$. For every
    contiguous block $B$,
    \begin{align}
        \label{eq:chain_block_renyi_approximation_lower_bound}
        S_\alpha(\rho_{\psi,B})
        +\frac{\alpha}{\alpha-1}\log(1-\delta) & \le2\log D
    \end{align}
    \cite{Schuch2008EntropyScaling,gap:swvc08_interior_block_bond_rank}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_reduced_state_rank_error}
    Writing $B=T\setminus R$ for nested prefixes $R\subseteq T$, complementary
    rank equality and submultiplicativity on $B^c=R\sqcup T^c$ give
    \begin{align}
        \label{eq:chain_block_rank_product}
        \operatorname{rank}\rho_{\phi,B}
         & \le\operatorname{rank}\rho_{\phi,R}
        \operatorname{rank}\rho_{\phi,T}\le\max(1,D)^2.
    \end{align}
    Contracting the trace-norm error to $B$ and applying the finite-dimensional
    Rényi inequality yields $\log(\max(1,D)^2)=2\log D$.
\end{proof}

\begin{theorem}[Logarithmic Rényi bound for efficiently approximable families]
    \label{thm:efficient_mps_renyi_logarithmic_bound}
    \lean{TNLean.MPS.AreaLaw.IsEfficientlyApproximableByMPS.exists_logarithmic_renyi_bound,
        TNLean.MPS.AreaLaw.not_isEfficientlyApproximableByMPS_of_superlogarithmic_renyi}
    \leanok
    If $(\psi_N)_{N\in\mathcal N}$ is efficiently approximable by MPS, then for
    every $\alpha>1$ and contiguous blocks $B_N$ there exist $K\in\mathbb R$ and
    $m\in\mathbb N$ independent of $N$ such that
    \begin{align}
        \label{eq:efficient_mps_renyi_logarithmic_bound}
        S_\alpha(\rho_{N,B_N}) & \le K+m\log(N+1)
        \qquad(N\in\mathcal N).
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_block_renyi_approximation_lower_bound}
    Taking $\delta=1/2$ and $D_N\le C(N+1)^r$ in
    (\ref{eq:chain_block_renyi_approximation_lower_bound}) gives
    \begin{align}
        \label{eq:efficient_mps_renyi_fixed_accuracy}
        S_\alpha(\rho_{N,B_N})
         & \le2|\log C|+2r\log(N+1)
        -\frac{\alpha}{\alpha-1}\log(1/2).
    \end{align}
\end{proof}

\begin{theorem}[Power-law Rényi obstruction]
    \label{thm:mps_renyi_power_growth_obstruction}
    \lean{TNLean.MPS.AreaLaw.not_isEfficientlyApproximableByMPS_of_renyi_power_growth}
    \leanok
    Let $\mathcal N\subseteq\mathbb N$ be infinite, $\alpha>1$, and
    $c,\kappa,\eta>0$. If contiguous blocks $B_N$ satisfy, for all sufficiently
    large $N\in\mathcal N$,
    \begin{align}
        \label{eq:mps_renyi_power_growth_hypotheses}
        |B_N| & \ge\eta N,
              & S_\alpha(\rho_{N,B_N}) & \ge c|B_N|^\kappa,
    \end{align}
    then $(\psi_N)_{N\in\mathcal N}$ is not efficiently approximable by MPS
    \cite[lines 309--315]{Schuch2008EntropyScaling}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:efficient_mps_renyi_logarithmic_bound}
    For $N\ge1$, (\ref{eq:mps_renyi_power_growth_hypotheses}) gives
    $S_\alpha(\rho_{N,B_N})\ge c(\eta/2)^\kappa(N+1)^\kappa$, which contradicts
    (\ref{eq:efficient_mps_renyi_logarithmic_bound}) as $N\to\infty$ since
    $\log(N+1)=o((N+1)^\kappa)$.
\end{proof}

\begin{theorem}[Regional entropy from a spectral tail]
    \label{thm:chain_von_neumann_spectral_tail}
    \lean{TNLean.MPS.AreaLaw.regionEntropy_le_log_rank_add_spectral_tail}
    \leanok
    For $\|\psi\|=1$ on a chain of local dimension $q$ and any region $R$, if
    $\tau_D(\rho_R)\le\varepsilon$ ($D\ge0$ with $\log0=0$), then
    \begin{align}
        \label{eq:chain_von_neumann_spectral_tail}
        S_\psi(R) & \le\log D+\varepsilon |R|\log q+\log2
    \end{align}
    \cite[lines 278--305]{Schuch2008EntropyScaling}.
\end{theorem}
\begin{proof}\leanok
    Apply the spectral-concentration entropy bound of
    \cite[``Spectral concentration and von Neumann entropy'']{QICLean2026Blueprint}
    on $\dim\mathcal H_R=q^{|R|}$.
\end{proof}

\begin{theorem}[Single-boundary entropy bound at small approximation error]
    \label{thm:chain_von_neumann_prefix_error}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.regionEntropy_leftBlock_le_log_add_half_error}
    \leanok
    Let $\|\psi\|=1$ on a chain of local dimension $q$ and $\phi$ be an
    (unnormalized) open-boundary MPS of bond dimension at most $D\ge0$ with
    $\|\ketbra{\psi}{\psi}-\ketbra{\phi}{\phi}\|_1\le\delta<1$. For every prefix
    $R$,
    \begin{align}
        \label{eq:chain_von_neumann_prefix_error}
        S_\psi(R) & \le\log D+\frac{\delta}{2}|R|\log q+\log2.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_reduced_state_rank_error,thm:chain_von_neumann_spectral_tail}
    Since $\operatorname{rank}\rho_{\phi,R}\le\max(1,D)$, the half-error
    marginal-tail bound of
    \cite[``Discarded marginal spectrum at small pure-state error'']{QICLean2026Blueprint}
    gives $\tau_{\max(1,D)}(\rho_{\psi,R})\le\delta/2$; apply
    (\ref{eq:chain_von_neumann_spectral_tail}) with $\log\max(1,D)=\log D$.
\end{proof}

\begin{theorem}[Two-boundary entropy bound at small approximation error]
    \label{thm:chain_von_neumann_block_error}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.regionEntropy_sdiff_leftBlock_le_two_log_add_half_error}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:chain_von_neumann_prefix_error},
    every contiguous block $R$ satisfies
    \begin{align}
        \label{eq:chain_von_neumann_block_error}
        S_\psi(R) & \le2\log D+\frac{\delta}{2}|R|\log q+\log2
    \end{align}
    \cite[lines 269--305]{Schuch2008EntropyScaling},
    \cite{gap:swvc08_von_neumann_error_range,gap:swvc08_interior_block_bond_rank}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_von_neumann_spectral_tail,thm:chain_block_renyi_approximation_lower_bound}
    By (\ref{eq:chain_block_rank_product}),
    $\operatorname{rank}\rho_{\phi,R}\le\max(1,D)^2$; apply the half-error
    marginal-tail bound and (\ref{eq:chain_von_neumann_spectral_tail}) with
    $\log(\max(1,D)^2)=2\log D$.
\end{proof}

\begin{theorem}[Arbitrarily small linear entropy bounds]
    \label{thm:efficient_mps_von_neumann_bound}
    \lean{TNLean.MPS.AreaLaw.IsEfficientlyApproximableByMPS.exists_vonNeumann_bound}
    \leanok
    If $(\psi_N)_{N\in\mathcal N}$ is efficiently approximable by MPS and $R_N$
    are contiguous blocks, then for every $\varepsilon>0$ there exist
    $K\in\mathbb R$ and $m\in\mathbb N$ such that for all $N\in\mathcal N$,
    \begin{align}
        \label{eq:efficient_mps_von_neumann_bound}
        S_{\psi_N}(R_N) & \le K+m\log(N+1)+\varepsilon N.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_von_neumann_block_error}
    Apply (\ref{eq:chain_von_neumann_block_error}) with
    $\delta=\min(1/2,\varepsilon/(\log q+1))$, $D_N\le C_\delta(N+1)^{m_\delta}$,
    and $|R_N|\le N$.
\end{proof}

\begin{theorem}[Linear von Neumann entropy obstructs efficient approximation]
    \label{thm:mps_linear_von_neumann_obstruction}
    \lean{TNLean.MPS.AreaLaw.not_isEfficientlyApproximableByMPS_of_linear_vonNeumann_growth}
    \leanok
    Let $\mathcal N\subseteq\mathbb N$ be infinite and $c,\eta>0$. If contiguous
    blocks $R_N$ satisfy, for all sufficiently large $N\in\mathcal N$,
    \begin{align}
        \label{eq:mps_linear_von_neumann_obstruction_hypotheses}
        |R_N| & \ge\eta N,
              & S_{\psi_N}(R_N) & \ge c|R_N|,
    \end{align}
    then $(\psi_N)_{N\in\mathcal N}$ is not efficiently approximable by MPS
    \cite[lines 278--305]{Schuch2008EntropyScaling},
    \cite{gap:swvc08_von_neumann_error_range,gap:swvc08_interior_block_bond_rank}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:efficient_mps_von_neumann_bound}
    Choosing $\varepsilon=c\eta/2$ in (\ref{eq:efficient_mps_von_neumann_bound})
    against (\ref{eq:mps_linear_von_neumann_obstruction_hypotheses}) gives
    $(c\eta/2)N\le K+m\log(N+1)$, impossible as $N\to\infty$.
\end{proof}
