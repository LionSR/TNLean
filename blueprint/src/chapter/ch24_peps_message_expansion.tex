\section{Exact elimination of finite-dimensional messages}
\label{sec:peps_message_expansion}

The identities below justify the preliminary message expansion in
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} on the transmitted register and an
arbitrary spectator memory, with no bound on the spectator dimension.

\begin{definition}[A local bra and a message summand]
    \label{def:peps_local_message_summand}
    \lean{TNLean.Lattice.PartyCircuit.Word.localDiscard}
    \lean{TNLean.Lattice.PartyCircuit.Word.transferTerm}
    \leanok
    \uses{def:peps_party_layout}
    For $u\in H$ at party $a$, the local bra is
    $\langle u|\otimes I_K:H\otimes K\to K$. For parties $a,b$ and an
    orthonormal basis $(e_i)_{i\in I}$ of $H$, the $i$-th message summand
    applies $\langle e_i|_a\otimes I_K$ and then prepares $|e_i\rangle_b$.
\end{definition}

\begin{theorem}[Local message summands]
    \label{thm:peps_local_message_summands}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_localDiscard}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_localDiscard_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_transferTerm}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_transferTerm}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_transferTerm_tmul}
    \leanok
    For $\|u\|=1$, $\|\langle u|\otimes I_K\|\le 1$. Each message summand is an
    allowed source-free composition of local contractions sending
    \begin{align}
        x\otimes y\longmapsto \langle e_i,x\rangle e_i\otimes y,
        \label{eq:peps_message_summand_action}
    \end{align}
    with the first output register owned by $b$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_local_message_summand}
    Since $\|e_i\|=1$, $\langle e_i|_a\otimes I_K$ is a contraction and
    preparing $|e_i\rangle_b$ is an isometry, with composite action
    (\ref{eq:peps_message_summand_action}).
\end{proof}

\begin{theorem}[The exact message identity]
    \label{thm:peps_exact_message_identity}
    \lean{TNLean.Lattice.PartyCircuit.Word.sum_eval_transferTerm}
    \leanok
    Summing the message summands over $(e_i)_{i\in I}$ gives the identity
    transfer from $a$ to $b$:
    \begin{align}
        \sum_{i\in I} |e_i\rangle_b\langle e_i|_a\otimes I_K
        =I_{H\otimes K}.
        \label{eq:peps_exact_message_identity}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_local_message_summands}
    By (\ref{eq:peps_message_summand_action}) and Parseval's identity
    $\sum_{i\in I}\langle e_i,x\rangle e_i=x$, both sides agree on elementary
    tensors $x\otimes y$.
\end{proof}

\begin{definition}[Prepending and composing monomials]
    \label{def:peps_compose_actual_monomials}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.prependWord}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.comp}
    \leanok
    Prepending a word to a monomial composes it with the monomial's initial
    segment. Composing two monomials with matching intermediate registers
    concatenates their effect chains by prepending the first monomial's final
    segment to the second monomial.
\end{definition}

\begin{theorem}[Operators and pair effects of composed monomials]
    \label{thm:peps_compose_actual_monomials}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.eval_prependWord}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.isAllowed_prependWord}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.effectCount_prependWord}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.eval_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.isAllowed_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.effectCount_comp}
    \leanok
    \uses{def:peps_compose_actual_monomials}
    Prepending an allowed source-only word to an allowed monomial preserves its
    allowedness and pair-effect count. For composable monomials $M,N$, their
    composition $N\circ M$ has operator $N\circ M$, pair-effect count
    $r_{N\circ M}=r_N+r_M$, and is allowed whenever $M$ and $N$ are allowed.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_compose_actual_monomials}
    Induct on the effect chain of $M$ using
    Definition~\ref{def:peps_compose_actual_monomials}.
\end{proof}

\begin{theorem}[The finite expansion of a message]
    \label{thm:peps_message_expansion_count}
    \lean{TNLean.Lattice.PartyCircuit.gate_toGate_eq_sum}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.transfer}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.length_transfer}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.isAllowed_effectCount_transfer}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.sum_norm_coeff_transfer}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.eval_transfer}
    \leanok
    The unit-weighted list of message summands for $H$ has length $\dim H$,
    absolute coefficient sum $\dim H$, allowed terms with $0$ pair effects, and
    evaluated operator $I_{H\otimes K}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_exact_message_identity}
    Each of the $\dim H$ basis elements contributes one allowed, effect-free
    summand with coefficient $1$, and (\ref{eq:peps_exact_message_identity})
    gives the sum $I_{H\otimes K}$.
\end{proof}

\begin{definition}[Composition of gate expansions]
    \label{def:peps_composed_gate_expansion}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.mem_comp}
    \leanok
    \uses{def:peps_compose_actual_monomials}
    For ordered gate expansions $L=((c_j,M_j))_j$ and $K=((d_k,N_k))_k$ with
    matching intermediate memories, their chronological composition $K\circ L$
    lists $(c_jd_k,N_k\circ M_j)$ in lexicographic order on $(j,k)$.
\end{definition}

\begin{theorem}[Operator and resources of composed expansions]
    \label{thm:peps_composed_gate_resources}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.gate_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.length_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.sum_norm_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.isAllowed_comp}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.effectCount_comp_le}
    \leanok
    \uses{def:peps_composed_gate_expansion}
    The composition $K\circ L$ evaluates to $G_K G_L$, has length $|L||K|$,
    and has coefficient sum $(\sum_j|c_j|)(\sum_k|d_k|)$. If all terms of $L$
    and $K$ are allowed with at most $r$ and $s$ pair effects, respectively,
    then all terms of $K\circ L$ are allowed with at most $r+s$ pair effects.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_compose_actual_monomials}
    Bilinearity and Theorem~\ref{thm:peps_compose_actual_monomials} give
    $\sum_{j,k}c_jd_k(N_k\circ M_j)=G_K G_L$, $|K\circ L|=|L||K|$,
    $\sum_{j,k}|c_jd_k|=(\sum_j|c_j|)(\sum_k|d_k|)$, allowedness, and the effect
    bound $r+s$.
\end{proof}

\begin{definition}[Monomials with finite-dimensional messages]
    \label{def:peps_message_monomial}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.eval}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.IsAllowed}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.messageDimensions}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.effectCount}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.expand}
    \leanok
    \uses{def:peps_party_monomial}
    A monomial with messages $M$ is a chronological composition of party
    monomials (Definition~\ref{def:peps_party_monomial}) and register transfers
    of dimensions $d_1,\ldots,d_s$, with total pair-effect count $r_M$; it is
    allowed when all its party-monomial segments are allowed. Expanding each
    transfer in an orthonormal basis and composing the resulting gate lists
    defines the message-free expansion $\mathcal E(M)$.
\end{definition}

\begin{theorem}[Exact expansion and resources of bounded messages]
    \label{thm:peps_message_expansion}
    \uses{def:peps_message_monomial}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.eval_expand}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.isAllowed_effectCount_expand}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.expansion_size}
    \leanok
    The expansion $\mathcal E(M)$ evaluates to $M$. If $M$ is allowed, every
    term of $\mathcal E(M)$ is an allowed party monomial with $r_M$ pair
    effects, and
    \begin{align}
        |\mathcal E(M)|
        =\sum_{(c,N)\in\mathcal E(M)}|c|
        =\prod_{j=1}^{s}d_j.
        \label{eq:peps_message_expansion_size}
    \end{align}
\end{theorem}
\begin{proof}
    \uses{thm:peps_message_expansion_count,thm:peps_composed_gate_resources}
    \leanok
    Induct over the segments and transfers of $M$ using
    Theorems~\ref{thm:peps_message_expansion_count}
    and~\ref{thm:peps_composed_gate_resources}.
\end{proof}

\begin{theorem}[Polynomial cost of boundedly many messages]
    \label{thm:peps_message_expansion_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.expansion_size_le}
    \lean{TNLean.Lattice.PartyCircuit.MessageMonomial.expansion_size_le_of_power_bounds}
    \leanok
    If $M$ contains $s\le m$ messages of dimensions $d_j\le D$ ($D\ge 1$), then
    $|\mathcal E(M)|=\sum_{(c,N)\in\mathcal E(M)}|c|\le D^m$. If
    $d_j\le C_DL^v$ ($C_D,L\ge 1$, $v\in\N$), both quantities are at most
    $C_D^mL^{vm}$, as in Theorem~5.2, assumption~3.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_message_expansion}
    By (\ref{eq:peps_message_expansion_size}),
    $\prod_{j=1}^s d_j\le D^s\le D^m$, and setting $D=C_DL^v$ gives $C_D^mL^{vm}$.
\end{proof}

\section{Gates containing bounded messages}
\label{sec:peps_message_gate_expansion}

Let $H_a,H_b$ be the input and output memories of a finite set of participating
parties, using message monomials as in

\cite[Proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.

\begin{definition}[Weighted message monomials]
    \label{def:peps_message_gate}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate,
        TNLean.Lattice.PartyCircuit.MessageGate.eval}
    \leanok
    \uses{def:peps_message_monomial}
    A weighted message-gate list is a finite ordered list
    $L=((c_j,M_j))_{j=0}^{K-1}$ of message monomials $M_j:H_a\to H_b$ with
    $c_j\in\C$ and operator $G_L=\sum_j c_jM_j$.
\end{definition}

\begin{definition}[Expanding all messages in a gate]
    \label{def:peps_message_gate_expand}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.expand,
        TNLean.Lattice.PartyCircuit.MessageGate.mem_expand}
    \leanok
    \uses{def:peps_message_gate}
    For $L=((c_j,M_j))_{j=0}^{K-1}$ with $\mathcal E(M_j)=((a_{j,t},N_{j,t}))_t$,
    define the expanded gate list $\widetilde L$ by concatenating
    $((c_ja_{j,t},N_{j,t}))_t$ over $j=0,\ldots,K-1$.
\end{definition}

% Source: 04-compression.tex, allowed monomials and message expansion,
% lines 32--43; Theorem 5.2, lines 143--149.
% Revision: adc7f1241b42e322a6451854ab7e4b4c146bf78a.

\begin{theorem}[Expansion of a gate without messages]
    \label{thm:peps_message_gate_segment_expansion}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.expand_map_segment}
    \leanok
    \uses{def:peps_party_monomial,def:peps_message_monomial,
        def:peps_message_gate_expand}
    Viewing a list $A=((c_j,M_j))_{j=0}^{K-1}$ of message-free party monomials
    as single-segment message monomials gives $\widetilde A=A$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_message_monomial,def:peps_message_gate_expand}
    Each segment $M_j$ has $\mathcal E(M_j)=((1,M_j))$, so concatenating
    $((c_j\cdot 1,M_j))$ recovers $A$.
\end{proof}

\begin{theorem}[Operator and effect count under message expansion]
    \label{thm:peps_message_gate_eval}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.eval_expand,
        TNLean.Lattice.PartyCircuit.MessageGate.isAllowed_expand,
        TNLean.Lattice.PartyCircuit.MessageGate.effectCount_expand_le}
    \leanok
    \uses{def:peps_message_gate_expand}
    The expanded list $\widetilde L$ evaluates to $G_L$. If every $M_j$ is
    allowed with at most $r$ pair effects, then every term of $\widetilde L$ is
    allowed with at most $r$ pair effects.
\end{theorem}
\begin{proof}
    \uses{thm:peps_message_expansion}
    \leanok
    By Theorem~\ref{thm:peps_message_expansion},
    $\sum_t a_{j,t}N_{j,t}=M_j$ and each $N_{j,t}$ is allowed with $r_{M_j}\le r$
    pair effects, so $\sum_{j,t}c_ja_{j,t}N_{j,t}=\sum_j c_jM_j=G_L$.
\end{proof}

\begin{theorem}[Term count and coefficient sum]
    \label{thm:peps_message_gate_size}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.expansion_size,
        TNLean.Lattice.PartyCircuit.MessageGate.expansion_size_le,
        TNLean.Lattice.PartyCircuit.MessageGate.expansion_size_le_of_coeff_le}
    \leanok
    \uses{def:peps_message_gate_expand}
    If $M_j$ has message dimensions $d_{j,1},\ldots,d_{j,m_j}$ and
    $q_j=\prod_{t=1}^{m_j}d_{j,t}$, then $|\widetilde L|=\sum_j q_j$ and
    $\sum_{(c,N)\in\widetilde L}|c|=\sum_j|c_j|q_j$. If $|L|\le K$,
    $m_j\le m$, and $d_{j,t}\le D$ ($D\ge 1$), then
    \begin{align}
        |\widetilde L| & \le KD^m,                    &
        \sum_{(c,N)\in\widetilde L}|c|
                       & \le D^m\sum_{(c,M)\in L}|c|,
        \label{eq:peps_message_gate_size}
    \end{align}
    and if $|c_j|\le C$ ($C\ge 0$), then $\sum_{(c,N)\in\widetilde L}|c|\le KCD^m$.
\end{theorem}
\begin{proof}
    \uses{thm:peps_message_expansion}
    \leanok
    By (\ref{eq:peps_message_expansion_size}), $|\mathcal E(M_j)|=\sum_t|a_{j,t}|=q_j\le D^m$;
    summing over $j$ gives $|\widetilde L|=\sum_j q_j\le KD^m$ and
    $\sum_{j,t}|c_ja_{j,t}|=\sum_j|c_j|q_j\le D^m\sum_j|c_j|\le KCD^m$.
\end{proof}

\begin{definition}[The corresponding original circuit gate]
    \label{def:peps_message_gate_circuit}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.toOriginalCircuit}
    \leanok
    \uses{thm:peps_message_gate_eval}
    For a participant set $C$ with $|C|\ne1$, embedding $\iota:C\hookrightarrow P$,
    allowed monomials $M_j$, and $\|G_L\|\le1$, define the original-circuit gate
    by inserting $\widetilde L$ as a single nonprivate gate with participant
    embedding $\iota$ and unchanged spectator memory.
\end{definition}

\begin{theorem}[Operator and occurrences of the constructed gate]
    \label{thm:peps_message_gate_circuit_properties}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.eval_toOriginalCircuit,
        TNLean.Lattice.PartyCircuit.MessageGate.nonprivateCount_toOriginalCircuit,
        TNLean.Lattice.PartyCircuit.MessageGate.participants_toOriginalCircuit}
    \leanok
    \uses{def:peps_message_gate_circuit}
    The gate of Definition~\ref{def:peps_message_gate_circuit} has operator
    equal to the relabelling of $G_L\otimes I$, nonprivate gate count $1$, and
    participant set $\iota(C)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_message_gate_eval}
    By Theorem~\ref{thm:peps_message_gate_eval}, $\widetilde L$ evaluates to
    $G_L$ with $\|G_L\|\le 1$, and Definition~\ref{def:peps_message_gate_circuit}
    inserts one nonprivate occurrence on $\iota(C)$.
\end{proof}

\begin{theorem}[Resource bounds for the constructed gate]
    \label{thm:peps_message_gate_circuit_bounds}
    \lean{TNLean.Lattice.PartyCircuit.MessageGate.isExpansionBounded_toOriginalCircuit,
        TNLean.Lattice.PartyCircuit.MessageGate.isMonomialBounded_toOriginalCircuit}
    \leanok
    \uses{def:peps_message_gate_circuit}
    Under the hypotheses of Theorem~\ref{thm:peps_message_gate_size}, if
    $\sum_{(c,M)\in L}|c|\le S$, each $M_j$ has at most $r$ pair effects, and
    $|L|\le K$, then the gate of Definition~\ref{def:peps_message_gate_circuit}
    has at most $r$ pair effects per monomial, coefficient sum at most $SD^m$,
    and at most $\lceil KD^m\rceil$ terms.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_message_gate_eval,thm:peps_message_gate_size}
    Apply Theorems~\ref{thm:peps_message_gate_eval}
    and~\ref{thm:peps_message_gate_size} to $\widetilde L$.
\end{proof}
