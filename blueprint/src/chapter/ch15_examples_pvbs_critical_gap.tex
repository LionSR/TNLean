\subsection{Critical One-Particle Dynamics and the Thermodynamic Gap}
\label{sec:pvbs_critical_gap}

\begin{lemma}[Critical PVBS one-particle dynamics]
    \label{thm:pvbs_one_particle_dynamics}
    \lean{MPSTensor.localTermES_pvbs_critical_oneMagnon,
        MPSTensor.parentHamiltonianES_pvbs_critical_oneMagnon,
        MPSTensor.parentHamiltonianES_pvbs_critical_uniformParticle}
    \leanok
    \uses{def:pvbs_tensor, def:parent_hamiltonian_es}
    For site amplitudes $f_j$, let
    $W(f)=\sum_j f_j|0\cdots1_j\cdots0\rangle$.
    On the one-particle sector, the critical local term is one half
    of identity minus exchange of its two sites. Thus for $N\ge2$,
    \begin{align}
        H_1W(f) & =W(Lf),
                & (Lf)_j  & =f_j-\frac{f_{j+1}+f_{j-1}}2.
        \label{eq:pvbs_one_particle_laplacian}
    \end{align}
    The indices are cyclic, with both oriented edges retained for $N=2$.
    In particular, the constant-amplitude W vector is a zero-energy state.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:pvbs_periodic_occupation, lem:pvbs_cyclic_geometry,
        thm:short_range_parent_gap_counterexample}
    Because a one-particle configuration has no doubly occupied bond, the
    $|11\rangle\langle11|$ part of the local projector vanishes, leaving only
    the singlet exchange term. Summing the exchange differences via the
    one-magnon permutation identity gives the cycle Laplacian $L$, which
    annihilates constant coefficients.
\end{proof}

\begin{lemma}[Positive low-energy Fourier magnons]
    \label{thm:pvbs_critical_fourier_magnon}
    \lean{MPSTensor.pvbs_fourierMagnon_ne_zero,
        MPSTensor.parentHamiltonianES_pvbs_critical_fourierMagnon,
        MPSTensor.pvbs_fourierMagnon_mem_orthogonal_kernel}
    \leanok
    \uses{def:pvbs_tensor, def:parent_hamiltonian_es}
    For $f_j=\exp(2\pi\mathrm{i}j/N)$ and $\psi_N=W(f)$, every $N\ge2$
    satisfies $\psi_N\ne0$ and
    \begin{align}
        H_1\psi_N & =\eta_N\psi_N,
                  & \eta_N         & =1-\cos(2\pi/N)>0.
        \label{eq:pvbs_critical_fourier_energy}
    \end{align}
    The vector $\psi_N$ is orthogonal to the whole kernel of $H_1$,
    including the uniform W state, rather than only to the vacuum.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:pvbs_one_particle_dynamics, lem:pvbs_cyclic_geometry,
        thm:short_range_parent_gap_counterexample, lem:transported_es_positive}
    The cyclic Fourier identity gives $Lf=\eta_N f$ with $\eta_N>0$ for
    $N\ge2$, and $\psi_N\ne0$ because its particle amplitude at site $0$ is
    $1$. For every ground vector $w\in\ker H_1$, self-adjointness gives
    $0=\langle H_1w,\psi_N\rangle=\eta_N\langle w,\psi_N\rangle$, proving
    orthogonality to the full kernel.
\end{proof}

\begin{lemma}[No uniform critical periodic gap]
    \label{thm:pvbs_critical_gaplessness}
    \lean{MPSTensor.not_exists_eventual_uniform_gap_pvbs_critical}
    \leanok
    \uses{def:pvbs_tensor, def:parent_hamiltonian_es}
    There are no $\delta>0$ and $N_0$ such that every ring of length
    $N\ge N_0$ satisfies
    $\delta\|v\|\le\|H_1v\|$ for all $v\perp\ker H_1$.
    This is thermodynamic gaplessness in the finite-periodic-volume
    criterion; no zero gap above the enlarged kernel of a fixed finite
    ring is asserted.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:pvbs_critical_fourier_magnon, thm:short_range_parent_gap_counterexample}
    Because the Fourier energies $\eta_N$ tend to zero, for any $\delta>0$ and
    threshold $N_0$ there exists $N\ge\max\{N_0,2\}$ with $0<\eta_N<\delta$.
    Evaluating the putative bound on the nonzero kernel-orthogonal vector
    $\psi_N$ forces $\delta\le\eta_N$, a contradiction; as recorded
    in~\cite{gap:bn12_pvbs_periodic_gap}, this establishes the critical
    specialization of the one-particle dispersion following equation~(8)
    in~\cite{BachmannNachtergaele2012PVBS}.
\end{proof}
