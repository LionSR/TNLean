\section{Encoded frames of guides}
\label{sec:peps_guide_frames}
The statements below set up the encoded frames used in the proof of
Proposition~7.1 of \cite{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{06-geometry.tex}, lines 62--125 and 669--672: the initial site owners are read
from a guide at the sampled sites, the holes sit at given centers with the
patches provided by Proposition~4.1 of the same source, and the tags are held by
a canonical incident label. Proposition~4.1 itself is stated here and enters as
a hypothesis; its proof is not part of this section.

We keep the setting of Section~\ref{sec:peps_encoded_frames}. For square cylinder
data with radii $r_1,\ldots,r_n$ and vectors $v_{j\ell}$, write
\begin{align}
    \Pi=\sum_{j=1}^{n}\sum_{\ell=1}^{d_j}
    \bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id
    \label{eq:peps_guide_frames_cyl_sum}
\end{align}
for the sum of the cylinder projectors, without the convention for an empty
outer sample.

\begin{definition}[Statement of the adaptive patch constraint]
    \label{def:peps_adaptive_patch_constraint}
    \lean{TNLean.PEPS.EncodedFrame.SquarePatch.cylSum,
        TNLean.PEPS.EncodedFrame.HasAdaptivePatch,
        TNLean.PEPS.EncodedFrame.AdaptivePatchSupply,
        TNLean.PEPS.Approximation.latticePos,
        TNLean.PEPS.Approximation.AdaptivePatchConstraint}
    \leanok
    \uses{def:peps_frame_square_patch}
    Let $\Lambda_L=\{1,\ldots,L\}^2$ with its sites at their integer positions.
    The adaptive patch constraint (\cite[Proposition~4.1]{OpenAI2026PolynomialPEPS},
    \texttt{03-patches.tex}, lines 26--49, with the parameters of the main
    theorem fixed as in \texttt{01-preliminaries.tex}, lines 3--10) is the following assertion. Fix $q\ge2$,
    $J,\Delta>0$, $B_0>0$ and $p>0$. There are constants $C,c>0$ such that for
    every $L\ge2$, every square Hamiltonian with local bounds $J$ and every
    unit ground vector $\Omega$ with gap $\Delta$, every center $z\in\R^2$ and
    every $0<u\le B_0\log L$ admit square cylinder data at $(z,u)$ with
    \begin{align}
        \sum_{j}d_j\le CL^c,
        \qquad
        \lVert(\id-\Pi)\Omega\rVert\le L^{-p}.
        \label{eq:peps_adaptive_patch_constraint}
    \end{align}
    The sum
    $\Pi$ in~(\ref{eq:peps_guide_frames_cyl_sum}) is an orthogonal projection supported on the sample of $Q^+(z,u)$ because the cylinder projectors have orthogonal ranges.
\end{definition}

\begin{lemma}[Adaptive patches on cell centers]
    \label{lem:peps_adaptive_patch_supply}

        TNLean.PEPS.Approximation.AdaptivePatchSupply.cellCenterPos,
        TNLean.PEPS.Approximation.AdaptivePatchConstraint.supply_cellCenterPos}
    \leanok
    \uses{def:peps_adaptive_patch_constraint,
        def:peps_frame_hole_encoder,
        def:peps_angular_information_bound,
        def:peps_squareLatticeCoordinateRegions}
    If the outer sample of square cylinder data is nonempty, its projector is
    $P_{c,h}=\Pi$; in all cases
    $\lVert(\id-P_{c,h})\Omega\rVert\le\lVert(\id-\Pi)\Omega\rVert$ and the
    number of tags is at most $\max\{1,\sum_jd_j\}$. Translating the site
    positions and the center by one vector does not change the samples, so the
    conclusion~(\ref{eq:peps_adaptive_patch_constraint}) of Definition~\ref{def:peps_adaptive_patch_constraint} is translation invariant. Hence,
    if the adaptive patch constraint holds, then with the same constants it holds
    on the sites placed at the cell centers $p_x-(1/2,1/2)=(i+\tfrac12,j+\tfrac12)$,
    $0\le i,j<L$, as in \texttt{06-geometry.tex}, lines 62--63. Fixing one
    such patch for each $(z,u)$, its projector $P_{z,u}$ satisfies
    $\lVert(\id-P_{z,u})\Omega\rVert\le L^{-p}$ and has at most $\max\{1,CL^c\}$
    tags. Because the patch depends only on $(z,u)$, a given center
    and radius receive the same patch on every sheet, as required at
    \texttt{06-geometry.tex}, lines 114--118.
\end{lemma}

\begin{proof}\leanok
    With a nonempty outer sample the tags are the pairs $(j,\ell)$ and $P_{c,h}=\Pi$; with an
    empty one $P_{c,h}=\id$. Translation preserves sup-norm
    distances and shifts each integer site position $p_x$ to $p_x-(1/2,1/2)$.
\end{proof}

\begin{definition}[Canonical tag holder and the frame of a guide]
    \label{def:peps_guide_frame}
    \lean{TNLean.PEPS.Approximation.tagHolder,
        TNLean.PEPS.Approximation.guideFrame}
    \leanok
    \uses{def:peps_dyadic_incident_labels, def:peps_frame}
    In the construction of \cite[Section~7]{OpenAI2026PolynomialPEPS} (\texttt{06-geometry.tex},
    lines 69--80 and 109--125), fix a ranking of the labels by natural numbers and a placeholder label $\ast$. The
    canonical tag holder of a point $c$ of a guide is the incident label at $c$
    other than $\ast$ of least rank. Given a guide $g$ read at the sites, an
    owner map $\varpi$ from labels to parties, a finite set $V$ of centers and a square
    patch $P_c$ at each $c\in V$ with pairwise disjoint outer samples, the frame $\mathcal F(g,V)$ of the
    guide assigns to each site $x$ the initial owner $\varpi(g(p_x))$ and has one hole at each $c\in V$, carrying $P_c$, with tag
    held by $\varpi$ applied to the canonical tag holder at $c$.
\end{definition}

\begin{theorem}[Guide frames with equal patches at their centers]
    \label{thm:peps_guide_frame_patch_congruence}
    \lean{TNLean.PEPS.Approximation.guideFrame_congr_patch,
        TNLean.PEPS.Approximation.guideFrame_congr}
    \leanok
    \uses{def:peps_guide_frame}
    Fix finite physical sites, their positions and local dimension, a guide $g$,
    an owner map $\varpi$, a placeholder $\ast$ and a label ranking.
    Two patch families $(P_c)_{c\in V}$ and $(P'_c)_{c\in V}$ on the same finite set $V$ of hole centers give equal
    guide frames if $P_c=P'_c$ for every $c\in V$ and each family has
    pairwise disjoint outer physical samples. In particular, equal sets $V=V'$ of
    centers and equal patch functions $P=P'$ give equal guide frames, with the same
    remaining data, regardless of the chosen disjointness proofs.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_guide_frame}
    The initial site owners $\varpi(g(p_x))$ agree, and the lists of holes agree term by term because every
    $c\in V$ receives the same patch $P_c=P'_c$ and canonical tag holder. Equality of
    center sets and patch functions then gives the second conclusion from the first.
\end{proof}

\end{proof}

\begin{theorem}[Reference vectors of guide frames]
    \label{thm:peps_guide_frame_norm}
    \lean{TNLean.PEPS.Approximation.disjoint_outer_of_disjoint_closedBall,
        TNLean.PEPS.Approximation.pairwise_disjoint_outer_patchAt,
        TNLean.PEPS.Approximation.mem_holes_guideFrame,
        TNLean.PEPS.Approximation.length_holes_guideFrame,
        TNLean.PEPS.Approximation.one_sub_mul_le_norm_refVec_guideFrame,
        TNLean.PEPS.Approximation.one_sub_mul_le_norm_refVec_guideFrame_patchAt,
        TNLean.PEPS.Approximation.outer_eq_empty_of_isCellCenter,
        TNLean.PEPS.Approximation.guideFrame_unitLevel}
    \leanok
    \uses{def:peps_guide_frame, thm:peps_frame_norm, lem:peps_adaptive_patch_supply}
    Patches whose closed outer squares $Q^+(c,h)$ are disjoint in $\R^2$ have disjoint
    outer samples. Let $\lVert\Omega\rVert=1$. If every patch at $c\in V$
    has projector error $\lVert(\id-P_{c,h})\Omega\rVert\le\varepsilon$, the reference vector $\Omega_{\mathcal F}$ of
    the frame $\mathcal F$ of a guide satisfies
    $1-\lvert V\rvert\varepsilon\le\lVert\Omega_{\mathcal F}\rVert\le1$; with
    adaptive patches (Lemma~\ref{lem:peps_adaptive_patch_supply}) one may take $\varepsilon=L^{-p}$. At the last level
    $n=1$ of \texttt{06-geometry.tex}, lines 669--672, where
    $2h_1\le2\epsilon_0<\tfrac12$, if the sites lie at cell
    centers, $V\subseteq\Z^2$, and every outer
    radius satisfies $2h_c<\tfrac12$, then every hole has empty physical outer
    sample, $K_{\mathcal F}$ is an isometry, and
    $\lVert\Omega_{\mathcal F}\rVert=\lVert\Omega\rVert$.
\end{theorem}

\begin{proof}\leanok
    Disjoint closed outer squares have disjoint site samples by Definition~\ref{def:peps_frame_square_patch}. The norm bound is
    Theorem~\ref{thm:peps_frame_norm} with $r=\lvert V\rvert$.
    Every cell center $(i+\tfrac12,j+\tfrac12)$ is at sup distance at least $\tfrac12>2h_c$ from every $c\in\Z^2$, so at $n=1$ the outer samples are empty and every hole projector is
    $\id$.
\end{proof}

\begin{lemma}[Sites separated from the holes of a guide]
    \label{lem:peps_guide_frame_outer_holes_disjoint}
    \lean{TNLean.PEPS.Approximation.disjoint_outerHoles_guideFrame}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_frame_rewrite_data,
        def:peps_guide_frame}
    Let $I$ be a finite set of sites with positions $p_x\in\mathbb R^2$, and
    let $q\in\N$ be the local dimension. Fix a guide, owner map,
    placeholder label and label ranking. Let $V$ be a finite indexing
    set of plane points, and assign a square patch $P_z$ with center $c_z$ and scale $r_z$ to each $z\in V$,
    with pairwise-disjoint outer samples (where $c_z$ need not equal $z$).
    For any $T\subseteq I$, if
    $2r_z<\operatorname{dist}(c_z,p_x)$ for all $z\in V$ and $x\in T$,
    then $T\cap H^+(\mathcal F)=\emptyset$ for the resulting guide frame $\mathcal F$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_frame_square_patch, def:peps_frame_rewrite_data,
        thm:peps_guide_frame_norm}
    Every hole of $\mathcal F$ is $P_z$ for some $z\in V$, whose outer
    sample consists of sites $x$ with $\operatorname{dist}(p_x,c_z)\le2r_z$, disjoint from $T$ by hypothesis.
\end{proof}

\begin{theorem}[Small-patch rewrites between frames of guides]
    \label{thm:peps_guide_frame_rewrite}
    \lean{TNLean.PEPS.Approximation.GuideFrameRewrite,
        TNLean.PEPS.Approximation.GuideFrameRewrite.PlaneConditions,
        TNLean.PEPS.Approximation.GuideFrameRewrite.PlaneConditions.conditions,
        TNLean.PEPS.EncodedFrame.Frame.exists_permWord,
        TNLean.PEPS.Approximation.GuideFrameRewrite.PlaneConditions.exists_originalCircuit}
    \leanok
    \uses{def:peps_guide_frame, lem:peps_guide_frame_tags,
        thm:peps_frame_small_rewrite_gate}
    Consider old and new guides $g_{\rm old},g_{\rm new}$ with one owner map $\varpi$, a
    finite set $V_{\rm unt}$ of untouched hole centers, finite sets $V_{\rm old},V_{\rm new}$ of old and new affected
    hole centers, hole patches of the two frames $\mathcal F_{\rm old}=\mathcal F(g_{\rm old},V_{\rm unt}\cup V_{\rm old})$ and $\mathcal F_{\rm new}=\mathcal F(g_{\rm new},V_{\rm unt}\cup V_{\rm new})$, additional
    patches $(c_i,u_i)_{i=1}^m$, and a specified list $S$ of at most $b$ parties. Suppose in $\R^2$ that:
    (i) at each $c\in V_{\rm unt}$ the two frames share the same patch and the same incident labels in $g_{\rm old}$ and $g_{\rm new}$;
    (ii) every $Q^+(c_i,u_i)$ and every affected outer hole square $Q^+(c,h_c)$ ($c\in V_{\rm old}\cup V_{\rm new}$) is disjoint from every untouched outer hole square $Q^+(c',h_{c'})$ ($c'\in V_{\rm unt}$);
    (iii) $\bigcup_{i=1}^m Q^-(c_i,u_i)$ contains $\{p\in\R^2:\varpi(g_{\rm old}(p))\ne\varpi(g_{\rm new}(p))\}$ and every affected $Q^+(c,h_c)$;
    (iv) in each $Q^+(c_i,u_i)$, the points outside the open inner holes of $g_{\rm old}$ (respectively $g_{\rm new}$) carry at most two labels of $g_{\rm old}$ (respectively $g_{\rm new}$); and
    (v) the owners $\varpi(g_{\rm old}(p)),\varpi(g_{\rm new}(p))$ on the outer patch and affected outer hole squares, and the
    affected tag holders, lie in $S$. Then conditions (1)--(4) of Definition~\ref{def:peps_frame_rewrite_data} (\cite[Lemma~6.3]{OpenAI2026PolynomialPEPS}) hold on the sampled sites. Under
    the bounds of Theorem~\ref{thm:peps_frame_small_rewrite_gate}, the rewrite
    is a circuit from the registers of $\mathcal F_{\rm old}$ to those of $\mathcal F_{\rm new}$
    with at most one nonprivate gate, every participant in $S$, and reference-vector error at most $(2b+1)L^{-20}$.
\end{theorem}

\begin{proof}\leanok
    A site $x$ lies in a sampled square $Q^\pm(c,h)$ if and only if $p_x\in Q^\pm(c,h)$, and by Lemma~\ref{lem:peps_guide_frame_tags} matching incident labels on $V_{\rm unt}$ give matching canonical tag holders, so (i)--(v) imply conditions (1)--(4) of Definition~\ref{def:peps_frame_rewrite_data}. While Definition~\ref{def:peps_frame_rewrite_data} lists $V_{\rm unt}$ first and Definition~\ref{def:peps_guide_frame} orders $V_{\rm unt}\cup V_{\rm old}$ and $V_{\rm unt}\cup V_{\rm new}$ in a fixed center order, the two listings are intertwined by tag-register permutations that preserve norms and carry one reference vector to the other; composing with Theorem~\ref{thm:peps_frame_small_rewrite_gate} yields the circuit.
\end{proof}

    Theorem~\ref{thm:peps_guide_frame_rewrite}, the rewrite is a circuit with

    tag, then $\Pi=0$ and $\lVert(\id-\Pi)\Omega\rVert=\lVert\Omega\rVert=1>L^{-60}$, a contradiction.
\end{proof}

\begin{theorem}[Direct repaintings, resizes and point treatments as small-patch rewrites]
    \label{thm:peps_guide_frame_rewrite_events}
    \lean{TNLean.PEPS.Approximation.exists_directRepainting_planeConditions,
        TNLean.PEPS.Approximation.directRepainting_step,
        TNLean.PEPS.Approximation.exists_resize_planeConditions,
        TNLean.PEPS.Approximation.exists_local_planeConditions,
        TNLean.PEPS.Approximation.dist_le_or_le_dist_of_isTrueVertex_homogenize,
        TNLean.PEPS.Approximation.exists_pointTreatment_planeConditions}
    \leanok
    \uses{thm:peps_guide_frame_rewrite}
    The plane conditions of Theorem~\ref{thm:peps_guide_frame_rewrite} hold in
    the following three cases, with patches on a square net of mesh $u$:
    \begin{enumerate}
        \item Direct repainting (\cite[Section~7]{OpenAI2026PolynomialPEPS},
              \texttt{06-geometry.tex}, lines 507--527): $g_{\rm old},g_{\rm new}$ are
              readings of guides uniform on $n$-blocks that differ only on
              one block $S$, with holes of radius $r$ at their true vertices and
              $3u+4r<n$; the affected holes lie at the four corners of $S$, and $S$ has specified parties from the nine blocks around $S$.
        \item Resize (lines 529--541): $g_{\rm old}=g_{\rm new}$ is a reading of a guide
              uniform on $N$-blocks, only the hole at one grid corner
              changes, all hole radii are at most $R$, and $4R+3u<N$.
        \item Local rewrite, such as a point treatment (lines 481--505):
              $g_{\rm old}=g_{\rm new}$ outside a closed sup ball of radius $\rho_0$ about $Q_0$, the affected
              holes lie within $\rho_1$ of $Q_0$ and the untouched ones at
              distance at least $\rho_2$, with
              $\max(\rho_0,\rho_1+2r)+3u+4r<\rho_2$. For the point treatment or its removal at
              a grid corner $Q_0$ of a repainted block of side $n$, every true vertex of
              a treated guide lies within $t$ of $Q_0$ or at distance at least $n-t$
              from $Q_0$, so $(\rho_0,\rho_1,\rho_2)=(t,t,n-t)$ applies whenever
              $2t+6r+3u<n$.
    \end{enumerate}
    In each case the two-owner condition for footprints of
    radius $2u$ is assumed as a hypothesis (supplied for readings by Section~\ref{sec:peps_dyadic_geometry} with $2u=\nu n$). In item~1 the rewrite is a circuit as in
    Theorem~\ref{thm:peps_guide_frame_rewrite} between the frames with holes at
    the true vertices, involving at most $18$ parties.
\end{theorem}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_dyadic_sampled_guides}, readings of two guides differ only in the closure of the set where the
    guides differ, so the owner changes only on the closed block $S$ or the closed ball of radius $\rho_0$. The net covers that set and the affected
    outer holes. Each outer patch square lies within $3u$ of the covered set,
    while every untouched hole is at distance at least $n$, $N$ or $\rho_2$, ensuring disjointness from untouched outer holes. The two-owner condition applies because
    every hole center near the event belongs to the frame, and any
    point within $n$ of $S$ lies in one of the nine surrounding blocks by Theorem~\ref{thm:peps_dyadic_block_index_near}.
\end{proof}

\begin{theorem}[The constants of the protocol and the separations of its steps]
    \label{thm:peps_protocol_constants}
    \lean{TNLean.PEPS.Approximation.patchRadius,
        TNLean.PEPS.Approximation.protocolPatchBound,
        TNLean.PEPS.Approximation.IsProtocolConstants,
        TNLean.PEPS.Approximation.exists_isProtocolConstants,
        TNLean.PEPS.Approximation.ProtocolSeparations,
        TNLean.PEPS.Approximation.IsProtocolConstants.protocolSeparations,
        TNLean.PEPS.Approximation.IsProtocolConstants.eventually_protocolSeparations}
    \leanok
    \uses{thm:peps_guide_frame_rewrite_events, lem:peps_dyadic_small_radius,
        lem:peps_dyadic_choice_K0}
    Fix the constants in the order of~\cite[Section~7]{OpenAI2026PolynomialPEPS},
    \texttt{06-geometry.tex}, lines 559--570: given the clearance constant $a_0$, choose $K_0\ge20$, then $\epsilon_0>0$ with
    $2\epsilon_0(a_0+1)\le a_0/2$, then $\nu>0$ with $2\epsilon_0+3\nu<1$, then
    $0<a\le a_0/2$, then $D>0$, and finally $B_0\ge K_0D$; any further bounds on each constant depending only on earlier constants in this chain can be satisfied simultaneously.

    Let $t=D\log L$ with $L\ge2$, $r=h_n=\epsilon_0\min(n,t)$, $u=\nu s/2$ (so $2u=\nu s$), and
    $b=\max\bigl(18,\lceil((4+8\epsilon_0+2\nu)/\nu+1)^2\rceil\bigr)$. Then
    the separations of Theorem~\ref{thm:peps_guide_frame_rewrite_events} hold at
    every scale:
    \begin{enumerate}
        \item for a direct repainting at level $n>0$ with $s=n$:
              $3u+4r<n$ and $(2(n/2+2r+u)/u+1)^2\le b$;
        \item for a resize on $N$-blocks with $R=h_N$ and $s=N$: $4R+3u<N$ and $(2(2R+u)/u+1)^2\le b$;
        \item for a point treatment at level $n>K_0t$ with $s=t$:
              $2t+6r+3u<n$, $\max(t,t+2r)+3u+4r<n-t$ and
              $(2(t+2r+u)/u+1)^2\le b$;
        \item $18\le b$, $2h_n\le B_0\log L$, and $\nu s\le B_0\log L$ whenever $s\le K_0t$.
    \end{enumerate}
\end{theorem}

\begin{proof}\leanok
    Taking $K_0,B_0$ as the maxima of the prescribed lower bounds and $\epsilon_0,\nu,a$ as the minima of the prescribed upper bounds satisfies the chain. Since $h_n\le\epsilon_0n$,
    $h_n\le\epsilon_0t$, and $2\epsilon_0+3\nu<1$, we have
    $3u+4r\le(3\nu/2+4\epsilon_0)n<n$ and $4R+3u\le(4\epsilon_0+3\nu/2)N<N$. For $n>K_0t\ge20t$, $(2+6\epsilon_0+3\nu/2)t<20t<n$. Each patch count is at most $((4+8\epsilon_0+2\nu)/\nu+1)^2\le b$, and $2h_n\le t\le K_0t\le B_0\log L$ as well as $\nu s\le K_0t\le B_0\log L$.
\end{proof}

    Let $2t<n$, let the level labels equal $\ast$ on every block
    outside the root square of $M\times M$ blocks, and let $S$ be a block of
    the root square. Consider an unmodified guide of the repainting of $S$,
    homogenized to one label on the open squares of radius $t$ about a finite set of grid-corner marks, and its reading along a small displacement
    (\texttt{06-geometry.tex}, lines 328--351):
    \begin{enumerate}
        \item Every true vertex of the reading is a grid corner $(na,nb)$ with
              $0\le a,b\le M+1$ or a point of the rim pattern of a mark (at most $52$ points per mark), so the true vertices form a finite set.
        \item At most $53$ true vertices lie within $t$ of any mark $Q_0$ (namely $Q_0$ and its rim pattern).
        \item If two mark sets differ at most at $Q_0$, their readings
              and true vertices coincide at distance strictly greater than $t$ from $Q_0$.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_dyadic_temporary_holes}, every true vertex is a grid
    corner or lies in the rim pattern of a mark. Outside the $n/2$-neighbourhood of $S$, an unmodified guide agrees with
    the starting guide or the uniform auxiliary label, so grid corners outside $0\le a,b\le M+1$ see only $\ast$ and the homogenizing label (at most two labels) and are not true vertices. Distinct grid
    corners are at distance at least $n$, so any rim point of another mark lies at
    distance at least $n-t>t$ from $Q_0$. Outside the closed square of
    radius $t$ about $Q_0$, the two homogenized guides and hence their local readings coincide.
\end{proof}
