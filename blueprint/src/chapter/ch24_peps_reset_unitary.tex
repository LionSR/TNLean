\section{The supported unitary of the information reset}

The information reset of \cite{OpenAI2026PolynomialPEPS} combines a
relative-entropy overlap bound, a two-sided Gaussian energy filter,
Lieb--Robinson localization, and a polar unitary correction.
Throughout, $\mathcal X_D=[q]^D$ denotes the configurations on a region $D$,
and $F_R$ denotes the replica swap on a region $R$.

\begin{lemma}[A unitary polar factor on the same support]
    \label{thm:peps_supported_polar_unitary}
    \lean{QuantumCircuit.exists_supported_unitary_polar_correction}
    \leanok
    For any operator $D$ supported on a site set $S$, there exists a unitary $U$ supported on $S$ such that for all vectors $x,y$,
    \begin{align}
        \|Ux-y\| & \le\|Dx-y\|+\|D^*y-x\|. \notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Write $D=D_S\otimes\id$, take a unitary polar factor $D_S=U_S|D_S|$ on $S$, and set $U=U_S\otimes\id$ so $D=U|D|$ with $|D|\ge0$. With $w=U^*y$, since $\id+|D|\ge\id$,
    \begin{align}
        \|Ux-y\| = \|x-w\| \le \|(\id+|D|)(x-w)\| \le \||D|x-w\|+\||D|w-x\| = \|Dx-y\|+\|D^*y-x\|. \notag
    \end{align}
\end{proof}

\begin{lemma}[Regional coordinates of the replica exchange]
    \label{thm:peps_reset_regional_coordinates}
    \lean{TNLean.Lattice.splitTwoRegions_sheetSwap,
        TNLean.Lattice.submatrix_partialSwap_kronecker_one_eq_sheetSwapOp,
        TNLean.Lattice.quantumRelativeEntropy_splitTwoRegionsState_eq_mutualInformation}
    \leanok
    For disjoint regions $R,C$ and $Q=(R\cup C)^c$, in replica coordinates $\mathcal X_R\times\mathcal X_C\times\mathcal X_Q$, the operator $F_R$ swaps the two $\mathcal X_R$ registers and fixes the other four registers, and any unit vector $\Omega$ satisfies $D(\rho_{RC}\,\|\,\rho_R\otimes\rho_C)=I_\Omega(R:C)$.
\end{lemma}
\begin{proof}
    \leanok
    The action of $F_R$ follows from $R\cap C=R\cap Q=\varnothing$, and $D(\rho_{RC}\,\|\,\rho_R\otimes\rho_C)=S(\rho_R)+S(\rho_C)-S(\rho_{RC})=I_\Omega(R:C)$.
\end{proof}

\begin{definition}[The variance and cutoff of the information reset]
    \label{def:lattice_reset_constants}
    \lean{TNLean.Lattice.resetVariance,
        TNLean.Lattice.resetCutoff,
        TNLean.Lattice.resetVariance_pos,
        TNLean.Lattice.resetCutoff_pos,
        TNLean.Lattice.resetVariance_mul_sq_div_two,
        TNLean.Lattice.resetCutoff_sq_div}
    \leanok
    For $\Delta,r\in\mathbb R$, define
    \begin{align}
        A(\Delta,r) & =\frac{2(30+r)}{\Delta^2}, & T_0(\Delta,r) & =\frac{2(30+r)}{\Delta}.
        \label{eq:lattice_reset_constants}
    \end{align}
    For $\Delta\neq0$ and $r\neq-30$, $A\Delta^2/2=T_0^2/(2A)=30+r$; for $\Delta>0$ and $r\ge0$, $A,T_0>0$.
\end{definition}

\begin{proof}\leanok
    Direct computation from (\ref{eq:lattice_reset_constants}).
\end{proof}

\begin{definition}[The constants of the information reset]
    \label{def:peps_reset_constants}
    \lean{TNLean.PEPS.Approximation.resetRadius,
        TNLean.PEPS.Approximation.resetFilterPrefactor,
        TNLean.PEPS.Approximation.resetUnitaryPrefactor,
        TNLean.PEPS.Approximation.resetRadius_pos,
        TNLean.PEPS.Approximation.resetFilterPrefactor_pos,
        TNLean.PEPS.Approximation.resetUnitaryPrefactor_pos,
        TNLean.PEPS.Approximation.resetRadius_eq,
        TNLean.PEPS.Approximation.resetUnitaryPrefactor_eq}
    \leanok
    \uses{def:peps_square_localization_constants,def:lattice_reset_constants}
    With $C_\square,v_\square(J)$ from Definition~\ref{def:peps_square_localization_constants}, $A(\Delta,r),T_0(\Delta,r)$ from (\ref{eq:lattice_reset_constants}), and $\mu=1/2$ (\cite{OpenAI2026PolynomialPEPS}, \texttt{02-information.tex}, lines 443--450 and 551--554), define for $J,\Delta,r\in\mathbb R$:
    \begin{align}
        K(J,\Delta,r)   & =\frac{v_\square(J)T_0(\Delta,r)+15+r}{\mu} = 1280e^2\frac J\Delta(30+r)+2r+30, \notag                   \\
        C_F(J,\Delta,r) & =C_\square(1+8J)\bigl(1+T_0(\Delta,r)\bigr)+3, \notag                                                    \\
        C_U(J,\Delta,r) & =2C_F(J,\Delta,r) = 2\Bigl(\frac{64e^{3/2}}{1-e^{-1/2}}(1+8J)\Bigl(1+\frac{2(30+r)}\Delta\Bigr)+3\Bigr),
        \label{eq:peps_reset_constants}
    \end{align}
    all positive when $J\geq0$, $\Delta>0$, and $r\geq0$.
\end{definition}

\begin{lemma}[The two-sided Gaussian filter, with explicit constants]
    \label{thm:peps_reset_regional_filter_explicit}
    \lean{TNLean.Lattice.regional_physicalBuffer_filter}
    \leanok
    \uses{def:lattice_reset_constants}
    Fix $\Delta>0$ and $r\geq0$, with $A,T_0$ from (\ref{eq:lattice_reset_constants}). Let $H$ have unit ground vector $\Omega$ ($H\Omega=E_0\Omega$, $H-E_0\geq\Delta(\id-|\Omega\rangle\langle\Omega|)$), let $R,C$ be disjoint with $I_\Omega(R:C)\leq b$ ($b\geq0$), and put $Q=(R\cup C)^c$, $\Psi=\Omega\otimes\Omega$, $\Phi=F_R\Psi$, $\mathbb H=H\otimes\id+\id\otimes H$, and $\mathbb H'=F_R\mathbb HF_R^*$. Then $\|\Psi\|=\|\Phi\|=1$, and there exist a contraction $W$ on the two copies of $\mathcal X_Q$ and $z\geq e^{-2b}$ such that for every $L\geq1$, with $B_*=1+b+\log L$, $h=AB_*$, $T=T_0B_*$, and Gaussian density $g_h$,
    \begin{align}
        M                                   & =\int_{[-T,T]}g_h(t)e^{it\mathbb H'}(\id\otimes W)e^{-it\mathbb H}\,dt, \label{eq:peps_reset_regional_filter} \\
        \|M\Psi-z\Phi\|,\ \|M^*\Phi-z\Psi\| & \le3e^{-10b}L^{-r}. \label{eq:peps_reset_regional_filter_err}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_reset_regional_coordinates,def:lattice_reset_constants}
    By Lemma~\ref{thm:peps_reset_regional_coordinates}, $D(\rho_{RC}\,\|\,\rho_R\otimes\rho_C)=I_\Omega(R:C)\le b$, so the purified relative-entropy overlap bound gives a Hermitian contraction $W$ on the buffer with $z=\langle\Phi,(\id\otimes W)\Psi\rangle\ge e^{-2b}$. Since $\mathbb H,\mathbb H'$ have ground vectors $\Psi,\Phi$ at $2E_0$ with gap $\Delta$ and $A\Delta^2/2=T_0^2/(2A)=30+r$ (Definition~\ref{def:lattice_reset_constants}), the full Gaussian filter residual is at most $e^{-h\Delta^2/2}\le e^{-10b}L^{-r}$ and truncation at $T$ adds at most $2e^{-T^2/(2h)}\le 2e^{-10b}L^{-r}$, giving (\ref{eq:peps_reset_regional_filter_err}).
\end{proof}

\begin{lemma}[The two-sided Gaussian filter of the physical buffer]
    \label{thm:peps_reset_regional_filter}
    \lean{TNLean.Lattice.exists_regional_physicalBuffer_filter}
    \leanok
    For every $\Delta>0$ and $r\ge0$, there exist $A,T_0>0$ such that under the hypotheses of Lemma~\ref{thm:peps_reset_regional_filter_explicit}, some buffer contraction $W$ and $z\ge e^{-2b}$ make $M$ in (\ref{eq:peps_reset_regional_filter}) satisfy (\ref{eq:peps_reset_regional_filter_err}) for all $L\ge1$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_reset_regional_filter_explicit,def:lattice_reset_constants}
    Take $A=A(\Delta,r)>0$ and $T_0=T_0(\Delta,r)>0$ from Definition~\ref{def:lattice_reset_constants} in Lemma~\ref{thm:peps_reset_regional_filter_explicit}.
\end{proof}

\begin{lemma}[The polar step of the information reset]
    \label{thm:peps_reset_polar_step}
    \lean{TNLean.Lattice.exists_supported_unitary_of_two_sided}
    \leanok
    If $M$ is supported on $S$, $b,C,\varepsilon\ge0$, and $z\ge e^{-2b}$ satisfies $\|M\psi-z\varphi\|,\|M^*\varphi-z\psi\|\le Ce^{-10b}\varepsilon$, then there exists a unitary $U$ supported on $S$ with $\|U\psi-\varphi\|\le2C\varepsilon$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_supported_polar_unitary}
    Apply Lemma~\ref{thm:peps_supported_polar_unitary} to $D=M/z$, noting $z^{-1}Ce^{-10b}\varepsilon\le Ce^{-8b}\varepsilon\le C\varepsilon$.
\end{proof}

\begin{lemma}[The localized filter, with explicit constants]
    \label{thm:peps_reset_localized_filter_explicit}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.reset_localized_filter}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_frame_sheet_swap,def:peps_reset_constants}
    Fix $q\ge1$, $J\ge0$, $\Delta>0$, and $r\ge0$, with $C=C_F(J,\Delta,r)$ and $K=K(J,\Delta,r)$ from (\ref{eq:peps_reset_constants}). For an open $L\times L$ square Hamiltonian $H$ ($L\ge1$) with interaction bound $J$, unit ground vector $\Omega$ with gap $\Delta$, disjoint $R,C_0$ with $I_\Omega(R:C_0)\le b$ ($b\ge0$), $Q=(R\cup C_0)^c$, crossing-edge endpoints $Z_R$ of $R$, and the paired-site graph neighbourhood $B$ of radius $\lceil K(1+b+\log L)\rceil$ around $Q\cup Z_R$, the unit vectors $\Psi=\Omega\otimes\Omega$ and $\Phi=F_R\Psi$ admit $z\ge e^{-2b}$ and a contraction $M$ supported on $B$ with
    \begin{align}
        \|M\Psi-z\Phi\|,\ \|M^*\Phi-z\Psi\| & \le Ce^{-10b}L^{-r}.
        \label{eq:peps_reset_localized_filter_err}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_reset_regional_filter_explicit,
        thm:peps_supported_physical_buffer_gaussian,def:peps_reset_constants,
        def:peps_square_localization_constants}
    Combining Lemma~\ref{thm:peps_reset_regional_filter_explicit} with the localized filter $M$ of Lemma~\ref{thm:peps_supported_physical_buffer_gaussian} at radius $\ell=\lceil KB_*\rceil$ and $\mu=1/2$ ($\mu K=v_\square(J)T_0+15+r$) bounds the localization error by
    \begin{align}
        C_\square L^2(1+8JL^2T_0B_*)e^{v_\square(J)T_0B_*-\mu\ell}
        \le C_\square(1+8J)(1+T_0)e^{-10b}L^{-r}; \notag
    \end{align}
    adding (\ref{eq:peps_reset_regional_filter_err}) gives (\ref{eq:peps_reset_localized_filter_err}) with $C=C_F(J,\Delta,r)$.
\end{proof}

\begin{lemma}[The localized filter of the information reset]
    \label{thm:peps_reset_localized_filter}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.exists_reset_localized_filter}

\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_reset_localized_filter_explicit,def:peps_reset_constants}
    Take $C=C_F(J,\Delta,r)>0$ and $K=K(J,\Delta,r)>0$ from Definition~\ref{def:peps_reset_constants} in Lemma~\ref{thm:peps_reset_localized_filter_explicit}.
\end{proof}

\subsection{The supported unitary at interaction range $R$}

The following statements extend Lemmas~\ref{thm:peps_reset_localized_filter_explicit} and~\ref{thm:peps_reset_unitary_explicit} from $R=1$ on the open square to range-$R$ Hamiltonians on a finite domain $\Lambda$ with $|\Lambda|\le L^2$.

\begin{definition}[The constants of the information reset at range $R$]
    \label{def:peps_reset_constants_range}
    \lean{TNLean.PEPS.Approximation.rangeResetRadius,
        TNLean.PEPS.Approximation.rangeResetFilterPrefactor,
        TNLean.PEPS.Approximation.rangeResetUnitaryPrefactor,
        TNLean.PEPS.Approximation.rangeResetRadius_pos,
        TNLean.PEPS.Approximation.rangeResetFilterPrefactor_pos,
        TNLean.PEPS.Approximation.rangeResetUnitaryPrefactor_pos}
    \leanok
    \uses{def:peps_reset_constants,def:area_graph_ball_localization_constants,
        def:peps_crossing_supports}
    With $C_0,v$ from Definition~\ref{def:area_graph_ball_localization_constants}, $c_R$ from Definition~\ref{def:peps_crossing_supports}, $T_0=T_0(\Delta,r)$ from (\ref{eq:lattice_reset_constants}), and $\mu=1/2$, define for $R\in\mathbb N$ and $J,\Delta,r\in\mathbb R$:
    \begin{align}
        K_R(J,\Delta,r)     & =\frac{v(R,2J)T_0+15+r}{\mu}, &
        C_{F,R}(J,\Delta,r) & =2C_0(R)(1+c_RJ)(1+T_0)+3,    &
        C_{U,R}(J,\Delta,r) & =2C_{F,R}(J,\Delta,r),
        \label{eq:peps_reset_constants_range}
    \end{align}
    all positive when $J\ge0$, $\Delta>0$, and $r\ge0$.
\end{definition}

\begin{lemma}[The localized filter and the unitary at range $R$]
    \label{thm:peps_reset_unitary_range}
    \lean{TNLean.Lattice.LocalHamiltonian.reset_localized_filter,
        TNLean.Lattice.LocalHamiltonian.reset_unitary}
    \leanok
    \uses{def:area_law_hamiltonian,def:peps_frame_sheet_swap,
        def:peps_crossing_supports,def:peps_reset_constants_range}
    Fix integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$, $r\ge0$. For $L\ge1$, a finite domain $\Lambda$ with $|\Lambda|\le L^2$, a range-$R$, bound-$J$ Hamiltonian $H$ on $\Lambda$ with unit ground vector $\Omega$ and gap $\Delta$, disjoint $A,C\subseteq\Lambda$ with $I_\Omega(A:C)\le b$ ($b\ge0$), and the paired-site graph neighbourhood $B$ of radius $\lceil K_R(J,\Delta,r)(1+b+\log L)\rceil$ around $(A\cup C)^c\cup Z_R(A)$, the unit vectors $\Psi=\Omega\otimes\Omega$ and $\Phi=F_A\Psi$ admit $z\ge e^{-2b}$, a contraction $M$ supported on $B$, and a unitary $U$ supported on $B$ satisfying
    \begin{align}
        \|M\Psi-z\Phi\|,\ \|M^*\Phi-z\Psi\| & \le C_{F,R}(J,\Delta,r)e^{-10b}L^{-r}, &
        \|U\Psi-\Phi\|                      & \le C_{U,R}(J,\Delta,r)L^{-r}. \notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_reset_regional_filter_explicit,
        thm:peps_supported_physical_buffer_gaussian_range,
        thm:peps_reset_polar_step,def:peps_reset_constants_range}
    Combine Lemma~\ref{thm:peps_reset_regional_filter_explicit} with Lemma~\ref{thm:peps_supported_physical_buffer_gaussian_range} at $\ell=\lceil K_RB_*\rceil$ (using $|\Lambda|\le L^2$ and $\mu K_R=v(R,2J)T_0+15+r$ to bound the localization error by $2C_0(R)(1+c_RJ)(1+T_0)e^{-10b}L^{-r}$), and apply Lemma~\ref{thm:peps_reset_polar_step}.
\end{proof}
