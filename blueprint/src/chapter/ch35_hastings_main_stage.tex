\section{The contradiction argument in Hastings' area law}
\label{sec:hastings_main_stage}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 124--232 (main theorem,
% equations entbd, smd, l0def, sbnd, claim).

The main theorem of~\cite{Hastings2007AreaLaw} bounds the cut entropy $S_i$ by
a chain-independent constant $S_{\max}=3l_0\ln D$: if $S_i>S_{\max}$, then
$S_k\ge2S_{\max}/3$ for $k\in[i,i+l_0]$, so the strengthened doubling
inequality holds on the window and Section~\ref{sec:hastings_doubling} forces
$l_0<\lceil\xi_0\rceil_+2^M=l_0$, with $S_{\max}$ set by the corrected doubling
iteration~\cite{gap:hastings07_doubling_constants}.

\begin{definition}[Window length and entropy bound]
    \label{def:hastings_max_entropy_bound}
    \lean{TNLean.MPS.AreaLaw.Hastings.doublingWindowLength,
        TNLean.MPS.AreaLaw.Hastings.maxEntropyBound,
        TNLean.MPS.AreaLaw.Hastings.maxEntropyBound_nonneg}
    \leanok
    \uses{def:hastings_doubling_constants}
    With $\xi_0$ and $M$ from Definition~\ref{def:hastings_doubling_constants},
    define
    \begin{align}
        l_0 & =\lceil\xi_0\rceil_+\,2^M, \qquad
        S_{\max} =3l_0\ln D\ge0\quad(D\ge1).
        \label{eq:hastings_max_entropy_bound}
    \end{align}
\end{definition}
\begin{proof}\leanok
    Both $l_0\ge0$ and $\ln D\ge0$ for $D\ge1$.
\end{proof}

\begin{theorem}[Size of the entropy bound]
    \label{thm:hastings_max_entropy_bound_le}
    \lean{TNLean.MPS.AreaLaw.Hastings.maxEntropyBound_le,
        TNLean.MPS.AreaLaw.Hastings.maxEntropyBound_le_of_one_le}
    \leanok
    \uses{def:hastings_max_entropy_bound}
    For $\xi'>0$, $2C_1>1$, $D\ge1$, and $C_2\ge-2$,
    \begin{align}
        S_{\max}\le6\ln D\,(\xi_0+1)\,
        2^{2(\xi'\ln D+(2+C_2)\xi'/\xi_0)+1},
        \notag
    \end{align}
    and if $C_1\ge1$,
    \begin{align}
        S_{\max}\le6\ln D\,(\xi_0+1)\,4^{\xi'\ln D}\,
        2^{(2+C_2)/\ln2+1},
        \notag
    \end{align}
    with base $4^{\xi'\ln D}$ in place of $2^{\xi'\ln D}$ in
    \cite[eq.~(smd)]{Hastings2007AreaLaw}~\cite{gap:hastings07_doubling_constants}.
\end{theorem}
\begin{proof}\leanok
    \uses{def:hastings_doubling_constants,def:hastings_max_entropy_bound}
    Substitute $M\le2(\xi'\ln D+(2+C_2)\xi'/\xi_0+1/2)+1$ and
    $\lceil\xi_0\rceil_+\le\xi_0+1$ into~(\ref{eq:hastings_max_entropy_bound});
    for $C_1\ge1$, $(2+C_2)\xi'/\xi_0=(2+C_2)/(2\ln(2C_1))\le(2+C_2)/(2\ln2)$.
\end{proof}

\begin{theorem}[Entropy of an initial segment]
    \label{thm:hastings_prefix_entropy_le}
    \lean{TNLean.MPS.AreaLaw.Hastings.compl_interval_zero,
        TNLean.MPS.AreaLaw.Hastings.regionEntropy_interval_zero_le}
    \leanok
    \uses{def:chain_model}
    For any unit vector on a chain of $L$ sites with local dimension $q\ge1$,
    $S(\lbrack0,i))\le(L-i)\ln q$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:chain_entropy_inequalities}
    Since $\lbrack0,i)^c=\lbrack i,L)$ has $L-i$ sites, purity and dimension bounds
    (Lemma~\ref{lem:chain_entropy_inequalities}) give
    $S(\lbrack0,i))=S(\lbrack i,L))\le(L-i)\ln q$.
\end{proof}

\begin{theorem}[The contradiction argument, with the window facts as
        hypotheses]
    \label{thm:hastings_main_stage}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_prefix_le_of_windowClaim}
    \leanok
    \uses{def:chain_model,def:hastings_max_entropy_bound}
    Let $\Omega$ be a gapped ground vector on a chain of $L$ sites with local
    dimension $q\ge1$, let $\xi'>0$, $2C_1>1$, and $\ln C_1+C_2\ge0$, and let
    $l_0,S_{\max}$ be given by~(\ref{eq:hastings_max_entropy_bound}) with $D=q$.
    Set $S_k=S([0,k))$ and assume $W_i:\N\to\R$ satisfies:
    \begin{enumerate}
        \item $S_i-(k-i)\ln q\le S_k$ for all $0\le i\le k\le L$;
        \item $0\le W_i(l)\le l\ln q$ for all $l\le l_0$ and $i+l_0\le L$;
        \item if $i+l_0\le L$ and $S_k\ge2S_{\max}/3$ for all $k\in[i,i+l_0]$,
              then for all $2l\le l_0$,
              \begin{align}
                  W_i(2l)\le2W_i(l)-\bigl(1-2C_1e^{-l/\xi'}\bigr)
                  \frac{l}{\xi'}+\ln C_1+C_2.
                  \notag
              \end{align}
    \end{enumerate}
    Then $S_i\le S_{\max}$ for every $i$ (\cite[eq.~(entbd)]{Hastings2007AreaLaw}
    with $S_{\max}$ from~\cite{gap:hastings07_doubling_constants}).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_doubling_window,thm:hastings_prefix_entropy_le,
        def:hastings_max_entropy_bound}
    Suppose $S_i>S_{\max}$. If $i+l_0>L$,
    Theorem~\ref{thm:hastings_prefix_entropy_le} gives
    $S_i\le(L-i)\ln q\le l_0\ln q\le S_{\max}$, a contradiction. If
    $i+l_0\le L$, item~(1) gives $S_k\ge S_{\max}-l_0\ln q=2S_{\max}/3$ for
    $k\in[i,i+l_0]$, so items~(2)--(3) and
    Theorem~\ref{thm:hastings_doubling_window} give
    $l_0<\lceil\xi_0\rceil_+2^M=l_0$, a contradiction.
\end{proof}
