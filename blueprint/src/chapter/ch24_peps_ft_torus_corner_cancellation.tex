\begin{definition}[Corner extension of a region insert]
    \label{def:peps_corner_extension}
    \lean{TNLean.PEPS.nestedThreeBlockGeometry}
    \lean{TNLean.PEPS.bareExtendInsert}
    \lean{TNLean.PEPS.extendInsert}
    \leanok
    For regions $R\subseteq S$ in a finite graph, an insert $C$ on $R$, the
    added-block contraction kernel $\Phi_{R,S}$ of $S\setminus R$ between
    $\partial R$ and $\partial S$, and the non-crossing bond-dimension product
    $P_{V\setminus S}$ of $V\setminus S$, define the bare and normalized corner
    extensions \cite[Section~4, proof of Theorem~3]{Molnar2018NormalPEPS} by
    \begin{align}
        \overline{\Ext}_{R\subseteq S}(C)_{\nu,\sigma}
         & =
        \sum_{\mu\in \partial R}
        C_{\mu,\sigma|_R}
        \Phi_{R,S}(\mu,\sigma|_{S\setminus R},\nu),
        \label{eq:peps_bare_corner_extension} \\
        \Ext_{R\subseteq S}(C)_{\nu,\sigma}
         & =
        P_{V\setminus S}^{-1}
        \overline{\Ext}_{R\subseteq S}(C)_{\nu,\sigma}.
        \label{eq:peps_norm_corner_extension}
    \end{align}
\end{definition}

\begin{lemma}[State preservation under corner extension]
    \label{thm:peps_corner_extension_state_preserves}
    \lean{TNLean.PEPS.deformedRegionState_extend}
    \leanok
    \uses{def:peps_corner_extension}
    If all bond dimensions are positive and $R\subseteq S$, then
    $\Psi_{A,R,C}=\Psi_{A,S,\Ext_{R\subseteq S}(C)}$.
\end{lemma}
\begin{proof}\leanok
    Separating $S\setminus R$ from $V\setminus S$ and substituting~(\ref{eq:peps_norm_corner_extension})
    gives, for every $\tau$,
    \begin{align}
        \Psi_{A,S,\Ext_{R\subseteq S}(C)}(\tau)
         & =
        P_{V\setminus S}^{-1}\,P_{V\setminus S}
        \Psi_{A,R,C}(\tau)
        =
        \Psi_{A,R,C}(\tau).
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Consecutive horizontal windows have equal corner extensions]
    \label{thm:peps_horizontal_consecutive_window_extend_eq}
    \lean{TNLean.PEPS.horizontalConsecutiveWindow_extend_eq}
    \leanok
    \uses{def:peps_corner_extension, thm:peps_corner_extension_state_preserves}
    Assume the one-orientation $L\times K$ window-injectivity hypotheses,
    union closure for injective regions, positive bond dimensions, and $2\le L$,
    $2\le K$, $2L+1\le W$, and $2K+1\le H$. If horizontally consecutive
    $L\times K$ windows $W_1,W_2$ with $(L+1)\times K$ union $U=W_1\cup W_2$
    carry inserts $C_1,C_2$ with $\Psi_{A,W_1,C_1}=\Psi_{A,W_2,C_2}$, then
    \begin{align}
        \Ext_{W_1\subseteq U}(C_1)
         & =
        \Ext_{W_2\subseteq U}(C_2).
        \label{eq:peps_horizontal_consecutive_extend_eq}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_corner_extension_state_preserves},
    \begin{align}
        \Psi_{A,U,\Ext_{W_1\subseteq U}(C_1)}
        =\Psi_{A,W_1,C_1}
        =\Psi_{A,W_2,C_2}
        =\Psi_{A,U,\Ext_{W_2\subseteq U}(C_2)},
        \notag
    \end{align}
    and injectivity of $U$ gives~(\ref{eq:peps_horizontal_consecutive_extend_eq}).
\end{proof}

\begin{lemma}[Patch-extended equality of the staircase end windows]
    \label{thm:peps_horizontal_staircase_patch_extend_eq}
    \lean{TNLean.PEPS.horizontalStaircase_patch_extend_eq}
    \leanok
    \uses{def:peps_horizontal_staircase_patch, thm:peps_corner_extension_state_preserves}
    Assume positive bond dimensions, $0<L$, $0<K$, $2L\le W$, and $2K\le H$. If
    $\Psi_{A,W_{\mathrm{right}},C_0}=\Psi_{A,W_{\mathrm{left}},C_{\mathrm{end}}}$,
    then on the staircase patch $P$,
    \begin{align}
        \Psi_{A,P,\Ext_{W_{\mathrm{right}}\subseteq P}(C_0)}
         & =
        \Psi_{A,P,\Ext_{W_{\mathrm{left}}\subseteq P}(C_{\mathrm{end}})}.
        \label{eq:peps_staircase_patch_extend_eq}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_corner_extension_state_preserves},
    \begin{align}
        \Psi_{A,P,\Ext_{W_{\mathrm{right}}\subseteq P}(C_0)}
        =\Psi_{A,W_{\mathrm{right}},C_0}
        =\Psi_{A,W_{\mathrm{left}},C_{\mathrm{end}}}
        =\Psi_{A,P,\Ext_{W_{\mathrm{left}}\subseteq P}(C_{\mathrm{end}})}.
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Two-block merge collapse]
    \label{lem:peps_two_block_merge_collapse}
    \lean{TNLean.PEPS.mergeCollapse2}
    \leanok
    Let $B_1,K_1\subseteq V\setminus T$ and $B_2,H_2\subseteq T$, fix boundary
    data $c_1$ on $V\setminus K_1$ and $c_2$ on $H_2$, and physical labels
    $\sigma_1$ on $B_1$ and $\sigma_2$ on $B_2$. With
    $F_i(\xi)=\prod_{x\in B_i}A_x(\xi|_{\partial x},\sigma_i(x))$ ($i=1,2$) and
    interior-bond product $P_T$ of $T$,
    \begin{align}
        \sum_{\substack{
                \zeta_2|_{\partial(V\setminus K_1)}=c_1,
        \zeta_1|_{\partial H_2}=c_2 \\
                \zeta_1|_{\partial T}=\zeta_2|_{\partial T}}}
        F_1(\zeta_2)F_2(\zeta_1)
         & =
        P_T
        \sum_{\substack{
                \eta|_{\partial(V\setminus K_1)}=c_1,
                \eta|_{\partial H_2}=c_2}}
        F_1(\eta)F_2(\eta).
        \label{eq:peps_two_block_merge_collapse}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $B_1,K_1\subseteq V\setminus T$ and $B_2,H_2\subseteq T$, each merged
    $\eta$ has $P_T$ preimages $(\zeta_1,\zeta_2)$ indexed by virtual bonds
    strictly inside $T$, leaving $F_1,F_2$ and the boundary conditions invariant.
\end{proof}

\begin{lemma}[Composition of corner-extension coefficient kernels]
    \label{thm:peps_corner_extension_coefficient_composition}
    \lean{TNLean.PEPS.threeBlockBlueCoeff_comp}
    \leanok
    \uses{def:peps_corner_extension, lem:peps_two_block_merge_collapse}
    For nested regions $R\subseteq S\subseteq P$
    \cite[Section~4, proof of Theorem~3]{Molnar2018NormalPEPS},
    \begin{align}
        \sum_{\nu\in\partial S}
        \Phi_{R,S}(\mu,\sigma|_{S\setminus R},\nu)
        \Phi_{S,P}(\nu,\sigma|_{P\setminus S},\omega)
         & =
        P_{V\setminus S}
        \Phi_{R,P}(\mu,\sigma|_{P\setminus R},\omega).
        \label{eq:peps_corner_kernel_comp}
    \end{align}
\end{lemma}

\begin{lemma}[Linearity and transitivity of corner extension]
    \label{thm:peps_corner_extension_linear_transitive}
    \lean{TNLean.PEPS.bareExtendInsert_const_smul}
    \lean{TNLean.PEPS.extendInsert_const_smul}
    \lean{TNLean.PEPS.extendInsert_add}
    \lean{TNLean.PEPS.extendInsert_sub}
    \lean{TNLean.PEPS.bareExtendInsert_trans}
    \lean{TNLean.PEPS.extendInsert_trans}
    \leanok
    \uses{def:peps_corner_extension, thm:peps_corner_extension_coefficient_composition}
    Both $\overline{\Ext}_{R\subseteq S}$ and $\Ext_{R\subseteq S}$ are linear
    in $C$, and for $R\subseteq S\subseteq P$,
    \begin{align}
        \overline{\Ext}_{S\subseteq P}
        (\overline{\Ext}_{R\subseteq S}(C))
         & =
        P_{V\setminus S}
        \overline{\Ext}_{R\subseteq P}(C),
        \label{eq:peps_bare_ext_trans} \\
        \Ext_{S\subseteq P}
        (\Ext_{R\subseteq S}(C))
         & =
        \Ext_{R\subseteq P}(C),
        \label{eq:peps_norm_ext_trans}
    \end{align}
    where~(\ref{eq:peps_norm_ext_trans}) assumes positive bond dimensions.
\end{lemma}
\begin{proof}\leanok
    Linearity follows from~(\ref{eq:peps_bare_corner_extension})--(\ref{eq:peps_norm_corner_extension});
    substituting~(\ref{eq:peps_corner_kernel_comp}) of
    Lemma~\ref{thm:peps_corner_extension_coefficient_composition} into~(\ref{eq:peps_bare_corner_extension})
    gives~(\ref{eq:peps_bare_ext_trans}), and cancelling $P_{V\setminus S}>0$
    gives~(\ref{eq:peps_norm_ext_trans}).
\end{proof}

\begin{definition}[Combination of subregion physical configurations]
    \label{def:peps_assemble_subregion_physical_config}
    \lean{TNLean.PEPS.assembleSubRegionσ}
    \leanok
    For $R\subseteq S$, $\sigma_R$ on $R$, and $\sigma_Q$ on $Q=S\setminus R$,
    define $\operatorname{As}_{R,S}(\sigma_R,\sigma_Q)$ on $S$ by
    \begin{align}
        \operatorname{As}_{R,S}(\sigma_R,\sigma_Q)(v)
         & =
        \begin{cases}
            \sigma_R(v), & v\in R,            \\
            \sigma_Q(v), & v\in S\setminus R.
        \end{cases}
        \label{eq:peps_assemble_subregion_config}
    \end{align}
\end{definition}

\begin{lemma}[Restrictions of combined subregion configurations]
    \label{thm:peps_assemble_subregion_physical_config_restricts}
    \lean{TNLean.PEPS.restrictSubRegionσ_assembleSubRegionσ}
    \lean{TNLean.PEPS.restrictSubRegionσ_sdiff_assembleSubRegionσ}
    \leanok
    \uses{def:peps_assemble_subregion_physical_config}
    For~(\ref{eq:peps_assemble_subregion_config}),
    $\operatorname{As}_{R,S}(\sigma_R,\sigma_Q)|_R=\sigma_R$ and
    $\operatorname{As}_{R,S}(\sigma_R,\sigma_Q)|_{S\setminus R}=\sigma_Q$.
\end{lemma}

\begin{lemma}[Kernel triviality of corner extension from the added block]
    \label{thm:peps_extend_insert_kernel_trivial_added_injective}
    \lean{TNLean.PEPS.extendInsert_kernel_trivial_of_addedInjective}
    \leanok
    \uses{def:peps_corner_extension, def:peps_assemble_subregion_physical_config, thm:peps_assemble_subregion_physical_config_restricts}
    For $R\subseteq S$ with positive bond dimensions and $Q=S\setminus R$
    blocked-tensor injective, $\Ext_{R\subseteq S}(C)=0\implies C=0$.
\end{lemma}
\begin{proof}\leanok
    Since $P_{V\setminus S}\ne0$, $\overline{\Ext}_{R\subseteq S}(C)=0$. Fix
    $\sigma_R$, let $\rho:\partial R\simeq\partial(V\setminus R)$ and
    $\tau:\partial S\simeq\partial(V\setminus S)$ be the complement-boundary
    bijections, and set $c_\eta=C_{\rho^{-1}(\eta),\sigma_R}$. Evaluating
    $\overline{\Ext}_{R\subseteq S}(C)=0$ at
    $\operatorname{As}_{R,S}(\sigma_R,\sigma_Q)$ (Lemma~\ref{thm:peps_assemble_subregion_physical_config_restricts})
    gives, for all $\sigma_Q$ and $\omega\in\partial(V\setminus S)$,
    \begin{align}
        \sum_{\eta\in\partial(V\setminus R)}
        c_\eta\,\Phi_{R,S}(\rho^{-1}(\eta),\sigma_Q,\tau^{-1}(\omega))=0.
        \label{eq:peps_bare_ext_zero_eval}
    \end{align}
    For $\omega\in\partial(V\setminus S)$ and $\beta\in\partial Q$, set
    \begin{align}
        f_\omega(\beta)
        =\sum_{\eta\in\partial(V\setminus R)}
        c_\eta\mathbf 1_{\{\exists q:\partial_{V\setminus R}q=\eta,\,\partial_{V\setminus S}q=\omega,\,\partial_Qq=\beta\}}.
        \notag
    \end{align}
    With crossing-bond product $\Gamma>0$ and blocked weights $W_Q$ of $Q$,
    unfolding $\Phi_{R,S}$ and using~(\ref{eq:peps_bare_ext_zero_eval}) yields
    \begin{align}
        (T_Q f_\omega)(\sigma_Q)
        =\sum_{\beta\in\partial Q} f_\omega(\beta)\,W_Q(\beta,\sigma_Q)
        =\Gamma\sum_{\eta\in\partial(V\setminus R)}
        c_\eta\,\Phi_{R,S}(\rho^{-1}(\eta),\sigma_Q,\tau^{-1}(\omega))=0,
        \notag
    \end{align}
    so injectivity of $T_Q$ gives $f_\omega=0$. For any $\mu\in\partial R$,
    choosing a global $q$ with $\partial_{V\setminus R}q=\rho(\mu)$ and setting
    $\omega=\partial_{V\setminus S}q$, $\beta=\partial_Q q$ gives
    $0=f_\omega(\beta)=c_{\rho(\mu)}=C_{\mu,\sigma_R}$.
\end{proof}

\begin{lemma}[Cancellation of a shared injective corner]
    \label{thm:peps_extend_insert_cancel_added_injective}
    \lean{TNLean.PEPS.extendInsert_cancel_addedInjective}
    \leanok
    \uses{thm:peps_corner_extension_linear_transitive, thm:peps_extend_insert_kernel_trivial_added_injective}
    If $R\subseteq S$, all bond dimensions are positive, and $S\setminus R$ is
    blocked-tensor injective, then
    $\Ext_{R\subseteq S}(C_1)=\Ext_{R\subseteq S}(C_2)\implies C_1=C_2$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_corner_extension_linear_transitive},
    $\Ext_{R\subseteq S}(C_1-C_2)=0$, so $C_1-C_2=0$ by
    Lemma~\ref{thm:peps_extend_insert_kernel_trivial_added_injective}.
\end{proof}

\begin{lemma}[Staircase-pair cancellation on the torus]
    \label{thm:peps_staircase_pair_cancel}
    \lean{TNLean.PEPS.staircasePair_cancel}
    \leanok
    \uses{thm:peps_extend_insert_cancel_added_injective}
    On a $W\times H$ torus with $1<W,H$, $2L\le W$, and $2K\le H$, if the
    completed corner $Q$ of the $(L,K)$-staircase end pair $S$ is blocked-tensor
    injective and all bond dimensions are positive, then for any inserts $X,Y$
    on $S$ (without assuming injectivity of $V\setminus S$),
    $\Ext_{S\subseteq S\cup Q}(X)=\Ext_{S\subseteq S\cup Q}(Y)\implies X=Y$.
\end{lemma}
\begin{proof}\leanok
    Since $2L\le W$ and $2K\le H$ imply $S\cap Q=\varnothing$, we have
    $(S\cup Q)\setminus S=Q$, and
    Lemma~\ref{thm:peps_extend_insert_cancel_added_injective} applies.
\end{proof}
