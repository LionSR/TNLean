\begin{theorem}[Local regular parent constraints force trivial holonomy]%
    \label{thm:peps_regular_region_parent_flatness}
    \lean{TNLean.PEPS.regionGroundSpace_regularTwisted_eq_of_internalGaugeFlat,
        TNLean.PEPS.regionPhysicalMap_mem_regularProjectorTwistedRange,
        TNLean.PEPS.regionPhysicalMap_leftInverse_on_regularGroundSpace,
        TNLean.PEPS.regularProjectorOpenRegionRange_apply_eq_zero_of_cycle_ne_one,
        TNLean.PEPS.regionPhysicalMap_apply_eq_zero_of_mem_regularGroundSpace_of_cycle_ne_one,
        TNLean.PEPS.regionPhysicalMap_slice_apply_eq_zero_of_parent_annihilates,
        TNLean.PEPS.disjoint_regularProjectorRegion_ranges_of_cycleResidual_ne_one,
        TNLean.PEPS.disjoint_regionGroundSpace_regularTwisted_of_cycleResidual_ne_one,
        TNLean.PEPS.regularWalkHolonomy_eq_one_of_mem_regionGroundSpace_of_mem_twisted,
        TNLean.PEPS.regularWalkHolonomy_eq_one_of_regionParent_annihilates_twistedState}
    \leanok
    \uses{def:peps_region_ground_space, def:peps_region_parent_interaction}
    On a finite graph with site-dependent regular $G$-injective tensors $A_v$,
    let $R$ induce a connected subgraph, and form $A^u$ by inserting regular
    bond operators $u_e$ at their ordered heads, with regional ground spaces
    $\mathcal G_R=\mathcal G_R(A)$ and $\mathcal G_R^u=\mathcal G_R(A^u)$
    (Definition~\ref{def:peps_region_ground_space}). If an internal closed
    walk $w$ in $R$ has holonomy $\operatorname{hol}_u(w)\ne 1$---or more
    generally if a spanning-tree cycle residual satisfies $r_c(u)\ne 1$---then
    $\mathcal G_R\cap\mathcal G_R^u=\{0\}$.

    Let $F_R=\bigotimes_{v\in R}F_v$ be a product of local $G$-injective
    inverses exposing the regular half-edge coordinates. Every
    $\phi\in\mathcal G_R$ satisfies $(F_R\phi)(\alpha)=0$ whenever some
    spanning-tree cycle label of $\alpha$ is not $1$. In particular, for any
    global vector $\Psi$ annihilated by an original regional parent interaction
    $h_R^{\mathrm{global}}\Psi=0$ (Definition~\ref{def:peps_region_parent_interaction})
    and any exterior configuration $\tau$, the slice $\Psi(\cdot,\tau)\in\mathcal G_R$
    obeys $(F_R\Psi(\cdot,\tau))(\alpha)=0$ at every such $\alpha$, without
    assuming an inserted-bond representation of $\Psi$.

    Consequently, if $0\ne\Psi(A^u)\in\ker h_R^{\mathrm{global}}$, then
    $\operatorname{hol}_u(w)=1$ for every internal closed walk $w$ in $R$.
    If a vertex gauge $g:R\to G$ trivializes all internal insertions
    ($g_{h(e)}u_eg_{t(e)}^{-1}=1$ for all $e\in E_R$), then
    $\mathcal G_R^u=\mathcal G_R$: crossing operators only relabel the boundary
    condition, using only coefficient invariance.
    No $G$-isometry is assumed. This is a necessary local implication
    in the closure argument of \cite[Theorem~5.5]{Schuch2010PEPS};
    it does not assert that every parent-kernel vector admits an
    inserted-bond description.
    This necessary local holonomy condition does not give a global closure representation; see
    \cite{gap:rmp_peps_quantum_double_g_isometry}.
\end{theorem}

\begin{proof}\leanok
    Applying $F_R=\bigotimes_{v\in R}F_v$ exposes the canonical regular
    half-edge coordinates, and the product of the original local coefficient
    maps $A_R=\bigotimes_{v\in R}A_v$ satisfies
    $A_RF_R|_{\mathcal G_R}=\mathrm{id}_{\mathcal G_R}$. In spanning-tree
    coordinates, $F_R(\mathcal G_R)$ is supported on identity cycle labels
    $c=1$, whereas $F_R(\mathcal G_R^u)$ is supported on simultaneous
    conjugates $(gr_c(u)g^{-1})_c$. Thus $r_c(u)\ne 1$ implies
    $F_R(\mathcal G_R)\cap F_R(\mathcal G_R^u)=\{0\}$, and left invertibility
    of $F_R$ on $\mathcal G_R$ yields $\mathcal G_R\cap\mathcal G_R^u=\{0\}$.

    A nonzero $\phi\in\mathcal G_R\cap\mathcal G_R^u$ therefore forces
    $r_c(u)=1$ for all $c$, so evaluating the closed-walk word gives
    $\operatorname{hol}_u(w)=1$. Finally, if $\Psi(A^u)\ne 0$ and
    $h_R^{\mathrm{global}}\Psi(A^u)=0$, any nonzero slice
    $\Psi(A^u)(\cdot,\tau)$ lies in $\mathcal G_R^u$ by contraction and in
    $\mathcal G_R$ by the parent constraint, so
    $\mathcal G_R\cap\mathcal G_R^u\ne\{0\}$.
\end{proof}
