\section{Extracting one ket column from an operator network}
\label{sec:peps_ket_column}
We establish \cite[Lemma~8.1]{OpenAI2026PolynomialPEPS},
September 24, 2026, \texttt{07-assembly.tex}, lines 67--94 (statement
lines 67--78, proof lines 79--94). Throughout, $\Omega$ is a unit vector in a
finite-dimensional space with a distinguished orthonormal basis
$\{\ket z\}$, and $\sigma$ is an operator on that space.

\begin{lemma}[Columns of an operator and the trace norm]
    \label{thm:peps_ket_column_hilbert_schmidt}
    \lean{TNLean.Lattice.Compression.sum_norm_sq_le_traceNorm_reindex_sq}
    \leanok
    For every operator $E$ on a finite-dimensional space with orthonormal
    basis $\{\ket z\}$,
    \begin{align}
        \sum_z \lVert E\ket z\rVert^2 & \le \lVert E\rVert_1^2 .
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $\{\ket z\}$ is orthonormal, $\sum_z\lVert E\ket z\rVert^2=\lVert E\rVert_2^2\le\lVert E\rVert_1^2$.
\end{proof}

\begin{lemma}[Selection of one index]
    \label{thm:peps_ket_column_selection}
    \lean{TNLean.Lattice.Compression.exists_ne_zero_norm_sq_le_of_sum}
    \leanok
    For vectors $f_z$ and scalars $a_z\in\C$ on a finite index set with
    $\sum_z\lVert f_z\rVert^2\le\eta^2$ and $\sum_z|a_z|^2=1$, there exists
    $z$ with $a_z\ne0$ and $\lVert f_z\rVert^2\le\eta^2|a_z|^2$.
\end{lemma}

\begin{proof}\leanok
    The set $S=\{z:a_z\ne0\}$ satisfies $\sum_{z\in S}|a_z|^2=1$; if $\lVert f_z\rVert^2>\eta^2|a_z|^2$ for all $z\in S$, summing over $S$ would give $\eta^2<\sum_{z\in S}\lVert f_z\rVert^2\le\eta^2$.
\end{proof}

\begin{lemma}[Normalization within twice the error]
    \label{thm:peps_ket_column_normalization}
    \lean{TNLean.Lattice.Compression.normalize_sub_le_two_mul,
        TNLean.Lattice.Compression.exists_phase_normalize_sub_le}
    \leanok
    Let $\lVert\Omega\rVert=1$ and $\eta<1$. If $\lVert w-\Omega\rVert\le\eta$,
    then $w\ne0$ and $\lVert w/\lVert w\rVert-\Omega\rVert\le2\eta$.
    Consequently, if $v=f+a\Omega$ with $a\ne0$ and
    $\lVert f\rVert\le\eta|a|$, then $v\ne0$ and some $\theta\in\R$ satisfies
    $\lVert v/\lVert v\rVert-e^{i\theta}\Omega\rVert\le2\eta$.
\end{lemma}

\begin{proof}\leanok
    Since $|\lVert w\rVert-1|\le\lVert w-\Omega\rVert\le\eta<1$, we have $w\ne0$ and $\lVert w/\lVert w\rVert-\Omega\rVert\le|1-\lVert w\rVert|+\lVert w-\Omega\rVert\le2\eta$. Applying this to $w=v/a=\Omega+f/a$ and setting $e^{i\theta}=a/|a|$ gives $\lVert v/\lVert v\rVert-e^{i\theta}\Omega\rVert=\lVert w/\lVert w\rVert-\Omega\rVert\le2\eta$.
\end{proof}

\begin{lemma}[Nonzero column close to a pure state]
    \label{thm:peps_ket_column_vector}
    \lean{TNLean.Lattice.Compression.exists_column_ne_zero_of_traceNorm_sub_pure_le,
        TNLean.Lattice.Compression.exists_eq_column_ne_zero_of_traceNorm_sub_pure_le,
        Matrix.toEuclideanLin_single_one_apply}
    \leanok
    In the vector part of \cite[Lemma~8.1]{OpenAI2026PolynomialPEPS}, if
    $\lVert\Omega\rVert=1$ and
    $\lVert\sigma-\ket\Omega\langle\Omega|\rVert_1\le\eta<1$, then some basis
    column $v=\sigma\ket z\ne0$ and some $\theta\in\R$ satisfy
    \begin{align}
        \Big\lVert\frac{v}{\lVert v\rVert}-e^{i\theta}\Omega\Big\rVert & \le2\eta .
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_ket_column_hilbert_schmidt,thm:peps_ket_column_selection,
        thm:peps_ket_column_normalization}
    With $E=\sigma-\ket\Omega\langle\Omega|$, $f_z=E\ket z$, and $a_z=\braket{\Omega}{z}$, Lemma~\ref{thm:peps_ket_column_hilbert_schmidt} gives $\sum_z\lVert f_z\rVert^2\le\eta^2$ while $\sum_z|a_z|^2=1$. Lemma~\ref{thm:peps_ket_column_selection} selects $z$ with $a_z\ne0$ and $\lVert f_z\rVert\le\eta|a_z|$, and Lemma~\ref{thm:peps_ket_column_normalization} applied to $v=\sigma\ket z=f_z+a_z\Omega$ concludes.
\end{proof}

\begin{definition}[Operator networks and fixed inputs on directed multigraphs]
    \label{def:peps_ket_column_dependent_fix_input}
    \lean{TNLean.Lattice.Compression.operatorNetworkMatrix,
        TNLean.Lattice.Compression.fixInput,
        TNLean.Lattice.Compression.network_fixInput,
        TNLean.Lattice.Compression.operatorNetworkMatrix_mulVec_single}
    \leanok
    \uses{def:peps_dependent_network}
    For a directed-multigraph network (Definition~\ref{def:peps_dependent_network}) with virtual alphabets $X_e$, bond matrices $B_e$, and output-input physical pairs $(s_v,r_v)$ defining the operator $\sigma=(\sigma_{\tau\rho})$, the fixed-input network at input configuration $z$ replaces each vertex tensor $A_v$ by $A_{v,z}(\eta,s):=A_v(\eta,(s,z_v))$ on the same multigraph, alphabets $X_e$, and bond matrices $B_e$, so its contraction is $\tau\mapsto\sigma_{\tau z}=(\sigma\ket z)_\tau$.
\end{definition}

\begin{lemma}[Ket column of an operator network on a directed multigraph]
    \label{thm:peps_ket_column_dependent}
    \lean{TNLean.Lattice.Compression.exists_fixInput_network_of_traceNorm_sub_pure_le}
    \leanok
    \uses{def:peps_ket_column_dependent_fix_input}
    In the directed-multigraph model of \cite[Lemma~8.1]{OpenAI2026PolynomialPEPS},
    let $\sigma$ be the operator of an operator network (Definition~\ref{def:peps_ket_column_dependent_fix_input}) with matching output and input spaces at each vertex, and let $\lVert\Omega\rVert=1$ with
    $\lVert\sigma-\ket\Omega\langle\Omega|\rVert_1\le\eta<1$. Then for some
    product-basis input $z$, the fixed-input network on the same multigraph, virtual alphabets, and bond matrices evaluates to $\psi=\sigma\ket z\ne0$ with $\lVert\psi/\lVert\psi\rVert-e^{i\theta}\Omega\rVert\le2\eta$ for some $\theta\in\R$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_ket_column_vector,def:peps_ket_column_dependent_fix_input}
    Apply Lemma~\ref{thm:peps_ket_column_vector} to $\sigma$ and identify $\sigma\ket z$ with the fixed-input contraction via Definition~\ref{def:peps_ket_column_dependent_fix_input}.
\end{proof}

\begin{lemma}[Ket column of an operator tensor on a simple graph]
    \label{thm:peps_ket_column_tensor}

\end{lemma}

\begin{proof}\leanok

\begin{corollary}[Column extraction for the main theorem]
    \label{cor:peps_ket_column_square_main}
    \lean{TNLean.PEPS.Approximation.Pinned.pos_of_contractPEPS_ne_zero,
        TNLean.PEPS.Approximation.Pinned.exists_peps_of_traceNorm_sub_pure_le,
        TNLean.PEPS.Approximation.Vector.Tensor.exists_peps_of_traceNorm_sub_pure_le}
    \leanok
    \uses{def:peps_ket_column_square_fix_input}
    If an operator network on the open $L\times L$ square has edge dimensions $D_e\le B$ and its operator $\sigma$ satisfies $\lVert\sigma-\ket\Omega\langle\Omega|\rVert_1\le\eta<1$ for a unit vector $\Omega$, then for every $\varepsilon\ge2\eta$ there is a nonzero PEPS $\Phi$ on the grid with edge dimensions $0<D_e\le B$ and
    \begin{align}
        \min_{\theta}\Big\lVert\frac{\Phi}{\lVert\Phi\rVert}
        -e^{i\theta}\Omega\Big\rVert\le\varepsilon .
    \end{align}
    For $B=CL^c$ and $\varepsilon=L^{-1}$ this gives the single-size conclusion of \cite[Theorem~1.1]{OpenAI2026PolynomialPEPS} (\texttt{00-introduction.tex}, lines 36--57; routing before column selection commutes with the source order in \texttt{07-assembly.tex}, lines 96--177, and $L<L_0$ is handled by an exact spanning-tree PEPS in lines 205--212).
\end{corollary}

\begin{proof}\leanok
    \uses{thm:peps_ket_column_square}
    By Lemma~\ref{thm:peps_ket_column_square}, the fixed-input PEPS $\Phi=\sigma\ket z\ne0$ retains $D_e\le B$ and satisfies the error bound; since $\Phi\ne0$, no contraction sum is empty, so $D_e>0$ for all $e$.
\end{proof}

\subsection{Error and normalization of the protocol}

The following estimates formalize \cite[Section~8.1]{OpenAI2026PolynomialPEPS}.

\begin{lemma}[Telescoping a chain of contractions]
    \label{lem:peps_protocol_telescoping}
    \lean{TNLean.Lattice.PartyCircuit.norm_trajectory_sub_reference_le}
    \lean{TNLean.Lattice.PartyCircuit.norm_trajectory_le}
    \leanok
    For normed spaces $E_t$, linear contractions $S_t\colon E_t\to E_{t+1}$ ($\lVert S_tv\rVert\leq\lVert v\rVert$, $t<n$), and $r_t\in E_t$, the sequence $x_0=r_0$, $x_{t+1}=S_tx_t$ satisfies
    \begin{align}
        \lVert x_n-r_n\rVert & \leq\sum_{t<n}\lVert S_tr_t-r_{t+1}\rVert, &
        \lVert x_n\rVert     & \leq\lVert r_0\rVert.
    \end{align}
\end{lemma}

\begin{proof}\leanok
    By induction on $n$, using $\lVert x_{t+1}-r_{t+1}\rVert\le\lVert S_t(x_t-r_t)\rVert+\lVert S_tr_t-r_{t+1}\rVert\le\lVert x_t-r_t\rVert+\lVert S_tr_t-r_{t+1}\rVert$ and $\lVert x_{t+1}\rVert\le\lVert x_t\rVert$.
\end{proof}

\begin{lemma}[Error and normalization of the protocol]
    \label{thm:peps_protocol_density}
    \lean{TNLean.Lattice.PartyCircuit.norm_protocol_sub_reference_le}
    \lean{TNLean.Lattice.PartyCircuit.rectangularTraceNorm_protocol_density_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.rectangularTraceNorm_compressed_density_sub_le}
    \leanok
    Let $L\geq1$ and $C_{\rm step},C_{\rm hole}\geq0$. Suppose that in Lemma~\ref{lem:peps_protocol_telescoping} one has $n\leq C_{\rm ch}L^3$, $\lVert r_0\rVert\leq1$, $\lVert S_tr_t-r_{t+1}\rVert\leq C_{\rm step}L^{-20}$ ($t<n$), and $r_n=\Omega\otimes\Xi$ in physical-private coordinates with
    \begin{align}
        \lVert\Omega\rVert                       & =1,    &
        1-C_{\rm hole}L^{-57}\leq\lVert\Xi\rVert & \leq1.
    \end{align}
    Then $\Psi=x_n$ satisfies $\lVert\Psi\rVert\leq1$ and
    \begin{align}
        \lVert\Psi-\Omega\otimes\Xi\rVert & \leq C_{\rm ch}C_{\rm step}L^{-17}, \label{eq:peps_protocol_vec_err}                  \\
        \big\lVert\Tr_{\rm private}|\Psi\rangle\langle\Psi|
        -|\Omega\rangle\langle\Omega|\big\rVert_1
                                          & \leq(2C_{\rm ch}C_{\rm step}+2C_{\rm hole})L^{-10}. \label{eq:peps_protocol_dens_err}
    \end{align}
    If moreover $\lVert\sigma-\Tr_{\rm private}|\Psi\rangle\langle\Psi|\rVert_1\leq C_{\rm comp}L^{-5}$, then
    \begin{align}
        \lVert\sigma-|\Omega\rangle\langle\Omega|\rVert_1
         & \leq(C_{\rm comp}+2C_{\rm ch}C_{\rm step}+2C_{\rm hole})L^{-5}.
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_protocol_telescoping}
    Lemma~\ref{lem:peps_protocol_telescoping} gives $\lVert\Psi\rVert\le\lVert r_0\rVert\le1$ and $\lVert\Psi-\Omega\otimes\Xi\rVert\le nC_{\rm step}L^{-20}\le C_{\rm ch}C_{\rm step}L^{-17}$. Since $\Tr_{\rm private}|\Omega\otimes\Xi\rangle\langle\Omega\otimes\Xi|=\lVert\Xi\rVert^2|\Omega\rangle\langle\Omega|$ and $1-\lVert\Xi\rVert^2\le 2(1-\lVert\Xi\rVert)\le 2C_{\rm hole}L^{-57}$, contractivity of $\Tr_{\rm private}$ and $L\ge1$ give (\ref{eq:peps_protocol_dens_err}), and the triangle inequality with $\lVert\sigma-\Tr_{\rm private}|\Psi\rangle\langle\Psi|\rVert_1\le C_{\rm comp}L^{-5}$ gives the final bound.
\end{proof}

\begin{lemma}[Retained norm and trace of the protocol]
    \label{thm:peps_protocol_retained_norm}
    \lean{TNLean.Lattice.PartyCircuit.protocol_norm_and_trace_deficit_le}
    \leanok
    Under the hypotheses of Lemma~\ref{thm:peps_protocol_density} (\cite[Section~8.1]{OpenAI2026PolynomialPEPS}), with $\varepsilon = C_{\rm ch}C_{\rm step}L^{-17}$, $\delta = C_{\rm hole}L^{-57}$, and $\rho=\Tr_{\rm private}|\Psi\rangle\langle\Psi|$,
    \begin{align}
        \lVert\Psi\rVert & \geq 1-\delta-\varepsilon,  \\
        1-\tr\rho        & \leq 2(\delta+\varepsilon).
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_protocol_density}
    By (\ref{eq:peps_protocol_vec_err}), $s:=\lVert\Psi\rVert\ge\lVert\Omega\otimes\Xi\rVert-\lVert\Psi-\Omega\otimes\Xi\rVert\ge 1-\delta-\varepsilon$, and hence $1-\tr\rho=1-s^2\le 2(1-s)\le 2(\delta+\varepsilon)$.
\end{proof}

\subsection{Column selection after discarding a product reference}

\begin{lemma}[A column from the compressed density]
    \label{thm:peps_product_reference_column}
    \lean{TNLean.PEPS.Approximation.Pinned.exists_fixInput_contractPEPS_of_product_reference}
    \lean{TNLean.PEPS.Approximation.Pinned.exists_peps_of_product_reference}
    \leanok
    \uses{def:peps_ket_column_square_fix_input}
    Combining \cite[Section~8.1 and Lemma~8.1]{OpenAI2026PolynomialPEPS}, let $\Omega$ be a unit vector on the open $L\times L$ square, and let $\Psi,\Xi$ satisfy
    \begin{align}
        \lVert\Psi\rVert                  & \leq1,           &
        1-\delta\leq\lVert\Xi\rVert       & \leq1,           &
        \lVert\Psi-\Omega\otimes\Xi\rVert & \leq\varepsilon.
    \end{align}
    If the operator $\sigma$ of an operator network on the square with edge dimensions $D_e$ satisfies
    \begin{align}
        \lVert\sigma-\Tr_{\rm private}|\Psi\rangle\langle\Psi|\rVert_1
                                            & \leq\kappa, &
        \alpha:=\kappa+2\varepsilon+2\delta & <1,
    \end{align}
    then some input $z$ yields fixed-input tensors (Definition~\ref{def:peps_ket_column_square_fix_input}) with positive edge dimensions $D_e$ (preserving any bounds $D_e\le B$) whose contraction $\Phi=\sigma\ket z\ne0$ satisfies, for some $\theta\in\R$,
    \begin{align}
        \Big\lVert\frac{\Phi}{\lVert\Phi\rVert}
        -e^{i\theta}\Omega\Big\rVert & \leq2\alpha.
    \end{align}
    (Routing precedes column selection as in Corollary~\ref{cor:peps_ket_column_square_main}; see \cite{gap:polypeps_product_reference_column}.)
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_ket_column_square, cor:peps_ket_column_square_main}
    Since $\Tr_{\rm private}|\Omega\otimes\Xi\rangle\langle\Omega\otimes\Xi|=\lVert\Xi\rVert^2|\Omega\rangle\langle\Omega|$, contractivity of $\Tr_{\rm private}$ gives $\lVert\Tr_{\rm private}|\Psi\rangle\langle\Psi|-|\Omega\rangle\langle\Omega|\rVert_1\le 2\varepsilon+(1-\lVert\Xi\rVert^2)\le 2\varepsilon+2\delta$, so $\lVert\sigma-|\Omega\rangle\langle\Omega|\rVert_1\le\alpha<1$ and Lemma~\ref{thm:peps_ket_column_square} and Corollary~\ref{cor:peps_ket_column_square_main} apply.
\end{proof}
