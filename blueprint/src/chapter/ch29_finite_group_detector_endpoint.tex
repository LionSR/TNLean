\subsection{Endpoint dependence of the finite-group detector}

\begin{lemma}[The circuit realizing the dressed interval family]
    \label{lem:mpug_detector_endpoint_circuit}
    \lean{MPOTensor.FiniteGroupDetector.pairedBond,
        MPOTensor.FiniteGroupDetector.pairedGate,
        MPOTensor.FiniteGroupDetector.pairCircuitBond,
        MPOTensor.FiniteGroupDetector.pairCircuitLayer,
        MPOTensor.FiniteGroupDetector.pairCircuitLayer_op,
        MPOTensor.FiniteGroupDetector.dressedCircuitMatchings,
        MPOTensor.FiniteGroupDetector.pairCircuitLayer_first_retained,
        MPOTensor.FiniteGroupDetector.pairCircuitLayer_second_retained,
        MPOTensor.FiniteGroupDetector.dressedPairCircuit,
        MPOTensor.FiniteGroupDetector.dressedPairCircuit_length,
        MPOTensor.FiniteGroupDetector.circuitOp_dressedPairCircuit_univ,
        MPOTensor.FiniteGroupDetector.circuitOp_dressedPairCircuit_single_entry,
        MPOTensor.FiniteGroupDetector.cellSuccessor,
        MPOTensor.FiniteGroupDetector.cellSuccessor_Icc_single_entry}
    Let $u,w$ be one-site unitaries. On a periodic chain partitioned into
    even--odd cells, apply $(wu)\otimes u$ on each cell, followed by
    $w^{\dagger}\otimes\mathbf 1$ on each pair consisting of a cell's odd
    site and the next cell's even site. The full circuit has depth two and
    is $u^{\otimes N}$. Retain exactly the gates whose nominal pair lies in
    a chosen set $S$ of cells. If $a\in S$ is the unique entry boundary,
    meaning that the predecessor of $k\in S$ is in $S$ exactly when $k\ne a$,
    the retained circuit is
    \begin{align}
        T(S)&=w_{(a,\mathrm{even})}\prod_{s\in S\times\{\mathrm{even},\mathrm{odd}\}}u_s.
        \label{eq:mpug_endpoint_circuit_interval}
    \end{align}
    A nonwrapping interval of cells with positive left endpoint has this
    entry-boundary property.
\end{lemma}

\begin{proof}
    The two matchings partition the sites, so gates in each layer have
    disjoint nominal pairs. Multiplication of their one-site factors gives
    $w^{\dagger}wu=u$ on every even site and $u$ on every odd site.
    After deletion, this cancellation occurs on the even sites whose
    predecessors remain in $S$. The sole uncanceled factor is $w$ at $a$.
\end{proof}

\begin{lemma}[Spatial factors of the chronological detector]
    \label{lem:mpug_detector_spatial_endpoint_word}
    \lean{MPOTensor.FiniteGroupDetector.intervalSiteFactor,
        MPOTensor.FiniteGroupDetector.spatialEndpointWord,
        MPOTensor.FiniteGroupDetector.spatialEndpointWord_eq,
        MPOTensor.FiniteGroupDetector.prod_spatialEndpointWord_eq_pow,
        MPOTensor.FiniteGroupDetector.prod_spatialEndpointWord_even}
    Let $u,w,c$ belong to a group with $cu=uc$ and $w^{-1}uw=cu$.
    Put $T(a,b;k)=w^{[k=a]}u^{[a\leq k\leq b]}$ and define the local
    factor of the complete interval word by
    \begin{align}
        D_m(k)&=T(0,2m+1;k)^{-1}
        \left(\prod_{s=1}^m T(s,2m+1-s;k)^{-1}\right)
        \left(\prod_{s=1}^m T(s,m+1+s;k)\right)T(0,m+1;k),
        \label{eq:mpug_detector_spatial_word}
    \end{align}
    with both products written in increasing $s$. Then
    \begin{align}
        D_m(k)&=\begin{cases}c^{k-1}&1\leq k\leq m,\\1&\text{otherwise},\end{cases}
        &\prod_{k=1}^m D_m(k)&=c^{\binom m2}.
        \label{eq:mpug_detector_spatial_factors}
    \end{align}
    If $m=2j$ and $c^{2j}=1$, the last product is $c^j$.
    This is an identity for the specified interval word; it imposes no
    hypothesis on a reference state or on the value of the full detector.
\end{lemma}

\begin{proof}
    At $1\leq k\leq m$, the forward interval factors multiply to
    $u^{k-1}wu$, and the corresponding inverse factors multiply to
    $u^{-(k-1)}(wu)^{-1}$. Including the two exterior intervals gives
    $u^{-k-1}w^{-1}u^{k-1}wu^2=c^{k-1}$.
    At $k=0$ the two copies of $wu$ cancel. At $k>m$ there is no dressing,
    and the forward and inverse interval multiplicities coincide.
    Multiplying the remaining phases gives the triangular exponent;
    its even-order reduction follows from $\binom{2j}2+j=2j^2$.
\end{proof}

\begin{lemma}[Conjugation phases of local endpoint words]
    \label{lem:mpug_detector_endpoint_phase}
    \lean{MPOTensor.FiniteGroupDetector.conjugation_word_eq_zpow,
        MPOTensor.FiniteGroupDetector.endpointConjugationWord,
        MPOTensor.FiniteGroupDetector.endpointConjugationWord_eq_pow,
        MPOTensor.FiniteGroupDetector.prod_endpointConjugationWord_eq_pow,
        MPOTensor.FiniteGroupDetector.prod_endpointConjugationWord_even}
    \leanok
    Let $u,w,c$ belong to a group, with $cu=uc$ and $w^{-1}uw=cu$.
    For integers $a,b,d$,
    \begin{align}
        u^aw^{-1}u^bwu^d&=c^bu^{a+b+d}.
        \label{eq:mpug_endpoint_conjugation}
    \end{align}
    In particular, for $E_r=u^{-r-2}w^{-1}u^rwu^2$ and $m\geq0$,
    \begin{align}
        E_0E_1\cdots E_{m-1}&=c^{\binom m2}.
        \label{eq:mpug_endpoint_phase_product}
    \end{align}
    If $m=2k$ and $c^{2k}=1$, the product is $c^k$.
\end{lemma}

\begin{proof}\leanok
    Conjugation preserves integer powers, so
    $w^{-1}u^bw=(cu)^b=c^bu^b$. Commuting $c^b$ past $u^a$ proves
    (\ref{eq:mpug_endpoint_conjugation}). Substituting $a=-r-2$, $b=r$ and
    $d=2$ gives $E_r=c^r$. Induction on $m$, using
    $\binom{m+1}2=\binom m2+m$, proves
    (\ref{eq:mpug_endpoint_phase_product}).
    Finally $\binom{2k}2+k=2k^2$, so
    $c^{\binom{2k}2}c^k=1=(c^k)^2$; canceling the common factor gives
    $c^{\binom{2k}2}=c^k$.
\end{proof}

\begin{lemma}[An order-two finite-group detector with nontrivial endpoint phase]
    \label{lem:mpug_finite_group_detector_endpoint}
    \lean{Matrix.leftDressedOnSiteInterval,
        Matrix.cyclicTwoTruncationDetector,
        Matrix.leftDressedOnSiteInterval_conjTranspose_mul_self,
        Matrix.cyclicTwoTruncationDetector_leftDressedOnSiteInterval_pauliX,
        Matrix.star_dotProduct_cyclicTwoTruncationDetector_leftDressedOnSiteInterval_pauliX}
    \leanok
    Let $X,Z$ be the Pauli matrices on a chain, and choose cuts
    $a_0<a_2<a_1<b_0<b_2<b_1$. Put
    $T_{rs}=Z_{a_r}\prod_{a_r\leq k\leq b_s}X_k$. Every $T_{rs}$ is
    unitary, and
    \begin{align}
        T_{01}^{\dagger}T_{22}^{\dagger}T_{10}^{\dagger}
        T_{22}T_{11}T_{00} &=-\mathbf 1.
        \label{eq:mpug_finite_detector_sign}
    \end{align}
    Thus its expectation on every normalized state is $-1$. This is the
    six-factor order-two word of the finite-group detector in
    \cite[Section~V]{GarreRubioSchuch2024DomainWall}, with the truncations
    applied in increasing index order and the cut order printed there.
    The identity concerns the explicitly defined interval operators.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:mpug_onsite_truncation_commutator,
        thm:mpug_truncation_detector_endpoint_dependence}
    Multiply the six tensor products site by site. At $a_1$, the local
    product is $XX(ZX)^{\dagger}X(ZX)X=-\mathbf 1$.
    At $a_0$ it is $(ZX)^{\dagger}(ZX)=\mathbf 1$, and at $a_2$ it is
    $X(ZX)^{\dagger}(ZX)X=\mathbf 1$. Elsewhere the factors are powers of
    $X$, and the forward and backward interval coverage multiplicities agree.
    Consequently every other factor is the identity, proving
    (\ref{eq:mpug_finite_detector_sign}).
\end{proof}
