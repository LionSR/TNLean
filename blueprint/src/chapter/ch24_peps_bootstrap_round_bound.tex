\section{An explicit budget for exponent improvement}
\label{sec:area_law_bootstrap_round_bound}

The finite iteration in the proof of Proposition~9.5
\cite[lines~788--803]{OpenAI2026AreaLaw} contracts an initial exponent
$e_0$ by $1-\ell$ at each round. The following estimates bound the round count
in terms of the cyclic-color recurrence parameters of
Section~\ref{sec:area_law_initial_exponent_gain}.

\begin{lemma}[Logarithmic round bound]
    \label{thm:area_law_bootstrap_round_log_bound}
    \lean{TNLean.PEPS.AreaLaw.Scan.boxBootstrapRounds_le_ceil_log_div}
    \leanok
    For $0<e_0\le1$, $0<\varepsilon\le1$, and $0<\ell<1$,
    \begin{align}
        m:=\left\lceil\frac{\log(\varepsilon/e_0)}{\log(1-\ell)}\right\rceil_+
        \le\left\lceil\frac{\log(1/\varepsilon)}{\ell}\right\rceil_+.
        \label{eq:area_law_bootstrap_round_log_bound}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $\log(1-\ell)\le-\ell<0$, $\log(1/\varepsilon)\ge0$, and $e_0\le 1$,
    \begin{align}
        \frac{\log(1/\varepsilon)}{\ell}\log(1-\ell)
        \le-\log(1/\varepsilon)=\log\varepsilon
        \le\log(\varepsilon/e_0).\notag
    \end{align}
    Dividing by $\log(1-\ell)<0$ and applying $\lceil\cdot\rceil_+$ yields
    (\ref{eq:area_law_bootstrap_round_log_bound}).
\end{proof}

\begin{definition}[Initial round budget]
    \label{def:area_law_initial_round_budget}
    \lean{TNLean.PEPS.AreaLaw.Scan.initialBoxBootstrapRoundBound}
    \leanok
    \uses{thm:area_law_initial_box}
    For the cyclic-color recurrence probability $p$ and scale $M$ of
    Section~\ref{sec:area_law_initial_exponent_gain}, set
    \begin{align}
        B:=\left\lceil\frac{400000\log M\log500000}{p}\right\rceil_+.
        \label{eq:area_law_initial_round_budget}
    \end{align}
\end{definition}

\begin{lemma}[Bound on the physical bootstrap count]
    \label{thm:area_law_initial_round_budget_bound}
    \lean{TNLean.PEPS.AreaLaw.Scan.boxBootstrapRounds_initial_le}
    \leanok
    \uses{def:area_law_initial_round_budget,def:area_law_bootstrap_parameters}
    For $0<e_0<1$ and $(\ell,e_*)$ as in
    Definition~\ref{def:area_law_bootstrap_parameters}, the round count $m$ of
    (\ref{eq:area_law_bootstrap_round_log_bound}) at $\varepsilon=e_*$ satisfies
    $m\le B$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_bootstrap_round_log_bound,
        thm:area_law_initial_exponent_gain_bounds,
        thm:area_law_bootstrap_margins}
    By Lemmas~\ref{thm:area_law_initial_exponent_gain_bounds} and
    \ref{thm:area_law_bootstrap_margins},
    \begin{align}
        \ell                     & \ge\frac{p}{400000\log M}>0,\notag         \\
        \frac{\log(1/e_*)}{\ell} & \le\frac{400000\log M\log500000}{p}.\notag
    \end{align}
    Applying (\ref{eq:area_law_bootstrap_round_log_bound}) and monotonicity of
    $\lceil\cdot\rceil_+$ yields $m\le B$ in (\ref{eq:area_law_initial_round_budget}).
\end{proof}

\begin{lemma}[The round budget reaches the safe-box exponent]
    \label{thm:area_law_initial_round_budget_reaches}
    \lean{TNLean.PEPS.AreaLaw.Scan.pow_initialBoxBootstrapRoundBound_mul_le}
    \leanok
    \uses{def:area_law_initial_round_budget,def:area_law_bootstrap_parameters}
    With $e_0$, $\ell$, and $e_*$ as in
    Definition~\ref{def:area_law_bootstrap_parameters},
    \begin{align}
        (1-\ell)^B e_0\le e_*.
        \label{eq:area_law_initial_round_budget_reaches}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \lean{TNLean.PEPS.AreaLaw.Scan.pow_boxBootstrapRounds_mul_le}
    \uses{thm:area_law_initial_round_budget_bound}
    Since $0<1-\ell<1$ and $m\le B$ by
    Lemma~\ref{thm:area_law_initial_round_budget_bound},
    $(1-\ell)^B e_0\le(1-\ell)^m e_0\le e_*$.
\end{proof}

\begin{lemma}[Physical entropy bound within the round budget]
    \label{thm:area_law_physical_box_bounded_rounds}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_boxEntropy_le_boxError_bounded_rounds}
    \leanok
    \uses{def:area_law_initial_round_budget,thm:arealaw-physical-box-fixed-exponent}
    Let $q\ge1$, $R\in\mathbb N$, $J\ge0$, and $\Delta>0$, and fix transport
    constants $(c_t,C_{\rm ent},e_{\rm ent},C_{\rm en},e_{\rm en})$ with $c_t>0$
    satisfying the finite-family transport bound. With initial safe-box constant
    $C_0$ and one-round improvement $G$ at truncation constant
    $C_{\rm tr}(R,J,\Delta,2)$ as in
    Theorem~\ref{thm:arealaw-physical-box-fixed-exponent}, define
    \begin{align}
        C_{j+1}=G(C_j)+\log q.
        \label{eq:area_law_physical_box_recurrence}
    \end{align}
    Then there exists $m\le B$ (independent of $D_0$ and $r$) such that for all
    $D_0>2R+10$ and $r\ge1$,
    \begin{align}
        F_{\mathrm{box},D_0}(r)\le C_m r^{1+e_*},
        \label{eq:area_law_physical_box_bounded_rounds}
    \end{align}
    evaluated at $C_m$ without comparing $C_m$ and $C_B$.
\end{lemma}
\begin{proof}\leanok
    \lean{TNLean.PEPS.AreaLaw.Scan.boxEntropy_le_boxError_explicit}
    \uses{thm:area_law_initial_round_budget_bound,
        thm:arealaw-physical-box-fixed-exponent}
    In Proposition~9.5 \cite[lines~788--805]{OpenAI2026AreaLaw}
    (Theorem~\ref{thm:arealaw-physical-box-fixed-exponent}), the logarithmic
    round count $m$ satisfies $m\le B$ by
    Lemma~\ref{thm:area_law_initial_round_budget_bound} and yields
    (\ref{eq:area_law_physical_box_bounded_rounds}) with recurrence coefficient
    $C_m$ from (\ref{eq:area_law_physical_box_recurrence}).
\end{proof}
