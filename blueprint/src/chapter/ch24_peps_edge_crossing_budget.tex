\subsection{Exact nearest-neighbor crossing sums}
\label{sec:area_law_edge_crossing_budget}

\begin{definition}[On-site and nearest-neighbor supports]
    \label{def:area_law_nearest_neighbor_support}
    \lean{TNLean.PEPS.AreaLaw.nearestNeighborSupport}
    \leanok
    \uses{def:area_law_domain}
    For finite $\Lambda\subset\mathbb Z^2$ with unordered nearest-neighbor edges $E_\Lambda$, index supports by $I=\Lambda\sqcup E_\Lambda$ with $S_v=\{v\}$ and $S_{\{x,y\}}=\{x,y\}$, and for $A\subseteq\Lambda$ set
    \begin{align}
        C(A) & =\{i\in I:S_i\cap A\ne\varnothing,\ S_i\setminus A\ne\varnothing\}.
        \label{eq:area_law_crossing_supports}
    \end{align}
\end{definition}

\begin{theorem}[One inside endpoint]
    \label{thm:area_law_crossing_inside_endpoint}
    \lean{
        TNLean.PEPS.AreaLaw.not_mem_crossingTerms_singleton,
        TNLean.PEPS.AreaLaw.pair_inter_eq_singleton_of_crossing,
        TNLean.PEPS.AreaLaw.card_pair_insideConfig_of_crossing
    }
    \leanok
    No singleton $\{v\}$ meets both $A$ and $\Lambda\setminus A$. If $\{x,y\}$ meets both $A$ and $\Lambda\setminus A$, then $\{x,y\}\cap A=\{v\}$ is a singleton, and for a $q$-element alphabet ($q\ge0$), the configuration set on $\{x,y\}\cap A$ has cardinality $q$.
\end{theorem}
\begin{proof}
    \leanok
    Since $\{x,y\}\cap A\ne\varnothing$ and $\{x,y\}\setminus A\ne\varnothing$, exactly one of $x,y$ belongs to $A$, so $|\{x,y\}\cap A|=1$ and its $q$-alphabet configuration space has size $q^1=q$.
\end{proof}

\begin{theorem}[Crossing supports and unordered edges]
    \label{thm:area_law_crossing_support_count}
    \lean{
        TNLean.PEPS.AreaLaw.inl_not_mem_crossingTerms,
        TNLean.PEPS.AreaLaw.inr_mem_crossingTerms_iff,
        TNLean.PEPS.AreaLaw.card_nearestNeighbor_insideConfig,
        TNLean.PEPS.AreaLaw.card_crossingTerms_nearestNeighbor
    }
    \leanok
    \uses{def:area_law_nearest_neighbor_support,def:area_law_hamiltonian}
    For every $A\subseteq\Lambda$, $C(A)=\partial_\Lambda A\subseteq E_\Lambda$, so $|C(A)|=|\partial_\Lambda A|$, and for every $i\in C(A)$ and integer $q\ge0$, the $q$-alphabet configuration set on $S_i\cap A$ has cardinality $q$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_crossing_inside_endpoint}
    Apply Theorem~\ref{thm:area_law_crossing_inside_endpoint} to (\ref{eq:area_law_crossing_supports}).
\end{proof}

\begin{theorem}[Exact weighted crossing sum]
    \label{thm:area_law_nearest_neighbor_budget}
    \lean{TNLean.PEPS.AreaLaw.nearestNeighbor_crossing_budget}
    \leanok
    \uses{def:area_law_nearest_neighbor_support,def:area_law_hamiltonian}
    For integers $m,q\ge0$, $A_j\subseteq\Lambda$ ($0\le j<m$), $a_j,J\in\mathbb R$, and $d_{ij}=q^{|S_i\cap A_j|}$,
    \begin{align}
        \sum_{j=0}^{m-1}a_j^2\sum_{i\in C(A_j)}d_{ij}^2J
         & =q^2J\sum_{j=0}^{m-1}a_j^2|\partial_\Lambda A_j|
        \label{eq:area_law_exact_crossing_sum}
    \end{align}
    (\cite[Section~4]{OpenAI2026PolynomialPEPS}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_crossing_support_count}
    By Theorem~\ref{thm:area_law_crossing_support_count}, $d_{ij}=q$ for all $i\in C(A_j)$ and $|C(A_j)|=|\partial_\Lambda A_j|$.
\end{proof}

\begin{theorem}[A square-grid maximizing family with controlled energy]
    \label{thm:square_zero_floor_maximizer_energy}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.exists_zero_floor_maximizer_energy_fidelity}
    \leanok
    \uses{def:polynomial_peps_model,def:area_law_nearest_neighbor_support,
        def:square_area_law_hamiltonian}
    Let $L,q,m\ge0$ be integers, $J,E_0,\Delta\in\mathbb R$, $H$ an original square-grid Hamiltonian with interaction bound $J$, and $\Omega$ a unit vector satisfying $H\Omega=E_0\Omega$ and $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$. Write $\Lambda$ for the integer square domain, $U$ for the canonical configuration isometry, $\widetilde H=UHU^*$, and $\widetilde\Omega=U\Omega$.
    For any nested regions $D_0\subseteq\cdots\subseteq D_{m-1}\subseteq\Lambda$ whose boundaries $\partial_\Lambda D_j$ are pairwise disjoint in $E_\Lambda$, and weights $0<a_j\le1$, let feasible regional filters take the form $K_j=V_j\operatorname{diag}(p_{j,r}^{a_j/2})V_j^*$ ($V_j$ unitary, $p_{j,r}\ge0$, $\sum_r p_{j,r}=1$), extended by $\Id$ outside $D_j$, with $\psi(K)=K_{m-1}\cdots K_0\widetilde\Omega$. Then there exists a feasible family $K$ maximizing $\|\psi(K')\|$ over all feasible $K'$, and with $\psi=\psi(K)$, $\phi=\psi/\|\psi\|$, and $B=q^2J\sum_{j=0}^{m-1}a_j^2|\partial_\Lambda D_j|$,
    \begin{align}
        0 & < \|\psi\|,                                & \|\phi\| & =1, \notag \\
        |\operatorname{Re}\langle\phi,\widetilde H\phi\rangle-E_0|
          & \le B, \notag                                                      \\
        \Delta\bigl(1-|\langle\widetilde\Omega,\phi\rangle|^2\bigr)
          & \le B. \label{eq:square_zero_floor_energy}
    \end{align}
    This specializes~\cite{QICLean2026ZeroFloorEnergy} to~\cite{OpenAI2026PolynomialPEPS}; see~\cite{gap:oai26_zero_floor_energy_scope} and~\cite{gap:polypeps_square_zero_floor_optimizer_scope}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:square_nearest_neighbor_terms,thm:square_area_law_ground_gap,
        thm:area_law_crossing_support_count,thm:area_law_nearest_neighbor_budget}
    Transporting the Hermitian parts of the terms of $H$ by $U$ (Theorem~\ref{thm:square_nearest_neighbor_terms}) gives site and edge terms of norm $\le J$ summing to $\widetilde H$, each split by at most one $D_j$. The zero-floor optimizer bound of~\cite{QICLean2026ZeroFloorEnergy} together with (\ref{eq:area_law_exact_crossing_sum}) gives a global maximizer $K$ with $|\operatorname{Re}\langle\phi,\widetilde H\phi\rangle-E_0|\le B$, and Theorem~\ref{thm:square_area_law_ground_gap} yields $\Delta(1-|\langle\widetilde\Omega,\phi\rangle|^2)\le\operatorname{Re}\langle\phi,\widetilde H\phi\rangle-E_0\le B$.
\end{proof}

\subsection{Positive filters on the original square}
\label{sec:pinned_square_filters}

\begin{definition}[Positive regional filters and ascending output]
    \label{def:pinned_square_filters}
    \lean{
        TNLean.PEPS.Approximation.Pinned.liftLocal,
        TNLean.PEPS.Approximation.Pinned.asMap,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.PositiveFilter,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.tracePower,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.FilterFamily,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.Admissible,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.output,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.IsMaximizer,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.normalizedOutput,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.PositiveFilter.ext
    }
    \leanok
    \uses{def:peps_pinned_regional_matrix}
    On $V_L=\{0,\ldots,L-1\}^2$ with integers $q,m\ge0$ and regions $X_j\subseteq V_L$ ($0\le j<m$), a positive regional filter $F_j\ge0$ is a positive semidefinite matrix on $\{0,\ldots,q-1\}^{X_j}$ with lift
    \begin{align}
        (\widehat F_j)_{x,y}
         & =\begin{cases}
                (F_j)_{x|_{X_j},y|_{X_j}} & x|_{V_L\setminus X_j}=y|_{V_L\setminus X_j}, \\
                0                         & \text{otherwise}.
            \end{cases}\notag
    \end{align}
    For $a_j\in\mathbb R$, a family $F=(F_j)_{j=0}^{m-1}$ is \emph{admissible} if $\sum_r\lambda_r(F_j)^{2/a_j}=1$ for all $j$ (with $2/0=0$, $0^0=1$, $0^t=0$ for $t\ne0$). For a state $\Omega$, set
    \begin{align}
        \psi(F,\Omega) & =\widehat F_{m-1}\cdots\widehat F_0\Omega, &
        \phi(F,\Omega) & =\|\psi(F,\Omega)\|^{-1}\psi(F,\Omega)
        \label{eq:pinned_filter_output}
    \end{align}
    (with $0^{-1}=0$). An admissible family $F$ is \emph{maximizing} if $\|\psi(G,\Omega)\|\le\|\psi(F,\Omega)\|$ for all admissible $G$ (\cite{OpenAI2026PolynomialPEPS}).
\end{definition}

\begin{definition}[Regional coordinates in the integer square]
    \label{def:pinned_filter_coordinates}
    \lean{
        TNLean.PEPS.Approximation.Pinned.NestedFilter.canonicalRegions,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.canonicalRegions_fin,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.regionEquiv,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.CanonicalFamily,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.canonicalMatrices,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.fromCanonical
    }
    \leanok
    \uses{def:pinned_square_filters,thm:area_law_domain_identifications}
    Let $\sigma:V_L\to\Lambda$ be the canonical site bijection with configuration isometry $U$ and $D_j=\sigma(X_j)$ (extended by $\varnothing$ for $j\ge m$). Conjugating $F_j$ by the regional configuration bijection induced by $\sigma|_{X_j}$ defines $K_j(F)$ on $D_j$, with inverse $F_j(K)$.
\end{definition}

\begin{theorem}[All admissible families in either coordinate system]
    \label{thm:pinned_filter_coordinates}
    \lean{
        TNLean.PEPS.Approximation.Pinned.NestedFilter.canonicalMatrices_fromCanonical,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.fromCanonical_canonicalMatrices,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.admissible_iff_feasible,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.admissible_fromCanonical
    }
    \leanok
    \uses{def:pinned_filter_coordinates}
    Every feasible family $K$ satisfies $K(F(K))=K$, and every positive family $F$ with feasible $K(F)$ satisfies $F(K(F))=F$. If $a_j>0$ for all $j$, then $F$ is admissible if and only if each $K_j(F)$ is floor-zero feasible at weight $a_j$.
\end{theorem}
\begin{proof}
    \leanok
    Permutation conjugation preserves positivity and spectra, and for $a_j>0$ floor-zero feasibility is equivalent to $F_j\ge0$ and $\sum_r\lambda_r(F_j)^{2/a_j}=1$ (\cite{QICLean2026FilterFeasibility}).
\end{proof}

\begin{theorem}[Ascending output and global maximality]
    \label{thm:pinned_filter_output_maximum}
    \lean{
        TNLean.PEPS.Approximation.Pinned.liftLocal_eq_localLift,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.squareAreaLawState_eq_relabelVector,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.squareAreaLawState_output,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.squareAreaLawState_normalizedOutput,
        TNLean.PEPS.Approximation.Pinned.NestedFilter.isMaximizer_iff
    }
    \leanok
    \uses{def:pinned_filter_coordinates}
    For every family $F$ and vector $\Omega$,
    \begin{align}
        U\psi(F,\Omega) & =\widehat K_{m-1}(F)\cdots\widehat K_0(F)U\Omega,
        \notag                                                              \\
        U\phi(F,\Omega) & =
        \|\widehat K_{m-1}(F)\cdots\widehat K_0(F)U\Omega\|^{-1}
        \widehat K_{m-1}(F)\cdots\widehat K_0(F)U\Omega.
        \label{eq:pinned_filter_coordinates}
    \end{align}
    If $a_j>0$ for all $j$, then $F$ is a maximizing admissible family if and only if $K(F)$ is feasible and maximizes $\|\widehat K'_{m-1}\cdots\widehat K'_0U\Omega\|$ over all feasible $K'$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:pinned_filter_coordinates}
    Since $U\widehat F_j=\widehat K_j(F)U$ (\cite{QICLean2026FilterOutput}) and $U$ is an isometry, (\ref{eq:pinned_filter_coordinates}) holds; Theorem~\ref{thm:pinned_filter_coordinates} then identifies the admissible and feasible maximization problems.
\end{proof}

\begin{theorem}[An original-grid maximizing family with energy and fidelity bounds]
    \label{thm:pinned_square_maximizer_energy}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.exists_pinned_maximizer_energy_fidelity}
    \leanok
    \uses{def:polynomial_peps_model,def:pinned_square_filters,def:pinned_filter_crossings}
    Let $H$ be an original square-grid Hamiltonian with real interaction bound $J$, and let $\|\Omega\|=1$ satisfy $H\Omega=E_0\Omega$ and $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$ ($E_0,\Delta\in\mathbb R$). For nested regions $X_0\subseteq\cdots\subseteq X_{m-1}\subseteq V_L$ such that each forward edge crosses at most one $X_j$, and weights $0<a_j\le1$, there exists a maximizing admissible family $F$ whose output $\psi=\psi(F,\Omega)$ and normalization $\phi=\phi(F,\Omega)$ from (\ref{eq:pinned_filter_output}) satisfy
    \begin{align}
        0 & <\|\psi\|,                                  & \|\phi\| & =1, \notag \\
        |\operatorname{Re}\langle\phi,H\phi\rangle-E_0|
          & \le q^2J\sum_{j=0}^{m-1}b(X_j)a_j^2, \notag                         \\
        \Delta\bigl(1-|\langle\Omega,\phi\rangle|^2\bigr)
          & \le q^2J\sum_{j=0}^{m-1}b(X_j)a_j^2.
        \label{eq:pinned_square_energy}
    \end{align}
    The adapted source definitions and scope are recorded in~\cite{gap:polypeps_square_zero_floor_optimizer_scope}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:square_zero_floor_maximizer_energy,thm:pinned_filter_output_maximum,
        thm:pinned_filter_crossings,thm:square_area_law_hamiltonian_operator}
    Pull the maximizer $K$ of Theorem~\ref{thm:square_zero_floor_maximizer_energy} back to $F=F(K)$ via Theorem~\ref{thm:pinned_filter_output_maximum}, and transport (\ref{eq:square_zero_floor_energy}) through $U$ using Theorem~\ref{thm:square_area_law_hamiltonian_operator} and Theorem~\ref{thm:pinned_filter_crossings}.
\end{proof}
