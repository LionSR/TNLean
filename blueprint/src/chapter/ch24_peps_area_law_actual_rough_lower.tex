\subsection{Rough lower bounds for the scanner roots}
\label{sec:actual_scan_rough_lower}

The rough comparison of \cite{OpenAI2026AreaLaw} uses a fixed per-band
weight $0<a\leq1/2$, total weight $W=aK$, and inverse exponent $s=1/4$,
retaining the same physical vector, auxiliary labels and energy across
every fill and charge interpolation.
Its terminal family consists of the completed histories through a fixed
horizon together with the pre-charge histories before that horizon;
writing $Q_{jg}$ and $Y_{jg}$ for the near and middle regions of terminal
$j$ in band $g$ gives the far region $(Q_{jg}\cup Y_{jg})^c$.

% The following three entries concern the arbitrary-weight auxiliary estimates.
\begin{theorem}[Scanner-root rough lower estimate at arbitrary weight]
    \label{thm:actual_scan_rough_lower_weight}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_actualScan_rough_lower_norm_of_weight}
    \leanok
    Fix a finite physical domain, dimensions $q,d>0$, an interaction range and
    bound, a band count $K\in\mathbb N$, and $0<a\leq1/2$, and set $W=aK$.
    There are $c,C_f\geq0$ (chosen before the scanner, horizon, physical
    vectors, $k$, $\lambda$, and interpolation parameter) such that for any
    collar scanner with $K$ bands, $m,M>0$, truncated Hamiltonian
    $\widetilde H$, unit vector $\widetilde\Omega$, and $\gamma>0$ satisfying
    \begin{align}
        \widetilde H-e_0I
         & \succeq\gamma
        (I-|\widetilde\Omega\rangle\langle\widetilde\Omega|),
        \label{eq:actual_rough_weight_gap}
    \end{align}
    if $k>0$ and a nonzero copy-symmetric prevector of norm at most $1$ is
    fixed by $\lambda$ on both auxiliary factors with
    $2kS'-\varepsilon\leq2\log\dim\lambda$, and
    \begin{align}
        r_k & =\log(k+1)+c\log(k+2)+a\varepsilon,\notag \\
        R_k & =\frac{K}{2k}
        \left(2|r_k|+C_f\log(k+2)+2\log2+\log(3/2)\right),
        \label{eq:actual_rough_weight_remainder}
    \end{align}
    then at every fill or charge round before the finite horizon and every real
    interpolation parameter, the squared filtered norm $\mathcal N_k$ and mean
    energy $\mu_k$ satisfy $\mathcal N_k>0$, unit normalized filtered norm,
    $\delta_k=(\mu_k-e_0)/\gamma\geq0$, and for every $0<\tau\leq1/2$ (with $h$
    the binary entropy; \cite[Section~8, lines~97--169 and~585--651]{OpenAI2026AreaLaw}),
    \begin{align}
        \frac W4\left[
            4S'-6\left(\tau\log d+\frac{h(\tau)}a\right)
            -3\frac{\delta_k}{\tau}\max\{0,2S'\}\right]-R_k
         & \leq-\frac{\log\mathcal N_k}{k}.
        \label{eq:actual_rough_weight_rate}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_root_compression,thm:al_scan_actual_round_global_energy}
    Here $Q_{jg}\cap Y_{jg}=\varnothing$ and $Q_{jg}\cup Y_{jg}\subseteq Q_{j'h}$
    for $g<h$. Compressing via the symmetry isometry $Z$ and forming the
    ordered band product $M_k$ at $t=a/2$ (Theorem~\ref{thm:al_augmented_root_compression})
    gives
    \begin{align}
        w_k & =M_k^{-1/4}Z^*\mathrm{pre}_k,
            & \mathcal N_k                  & =\|w_k\|^2,
            & v_k                           & =Z\frac{w_k}{\|w_k\|},
        \label{eq:actual_rough_lower_coordinates}
    \end{align}
    with $\mu_k=\operatorname{Re}\langle v_k,\widetilde H_kv_k\rangle$
    (Theorem~\ref{thm:al_scan_actual_round_global_energy}). Since $4t=2a\leq1$,
    the Section~8 rough lower comparison divided by $k$ yields $\mathcal N_k>0$,
    $\|v_k\|=1$, the remainder (\ref{eq:actual_rough_weight_remainder}), and
    (\ref{eq:actual_rough_weight_rate}) with prefactor $aK/4=W/4$, while
    (\ref{eq:actual_rough_weight_gap}) gives $\delta_k\geq0$.
\end{proof}

\begin{theorem}[Vanishing remainder at a fixed band weight]
    \label{thm:actual_rough_weight_remainder_limit}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughLowerRemainderAtWeight,
        TNLean.PEPS.AreaLaw.Scan.tendsto_roughLowerRemainderAtWeight_of_label_dimension}
    \leanok
    For fixed $K\in\mathbb N$, $a,c,C_f,S'\in\mathbb R$, and a label sequence
    $\lambda_k$ with $\varepsilon_k=2|\log\dim\lambda_k-kS'|$, define for $k>0$
    \begin{align}
        R_k^{(a)}
         & =\frac{K}{2k}\left(
        2|\log(k+1)+c\log(k+2)+a\varepsilon_k|
        +C_f\log(k+2)+2\log2+\log(3/2)\right).
        \label{eq:actual_rough_weight_label_remainder}
    \end{align}
    If $\log\dim\lambda_k-kS'=o(k)$, then $R_k^{(a)}\to0$ as $k\to\infty$.
\end{theorem}
\begin{proof}
    \leanok
    Here $\varepsilon_k/k\to0$ and $\log(k+1)/k,\log(k+2)/k,1/k\to0$, so
    $R_k^{(a)}\to0$.
\end{proof}

\begin{theorem}[The physical label sequence at arbitrary weight]
    \label{thm:actual_supplied_rough_lower_weight}
    \lean{
        TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_actualScan_suppliedPrevector_rough_lower_of_weight}
    \leanok
    \uses{thm:actual_scan_rough_lower_weight}
    Under the hypotheses of Theorem~\ref{thm:actual_scan_rough_lower_weight},
    let $\lambda_k$ be a label sequence with $\log\dim\lambda_k-kS'=o(k)$ whose
    physical Schmidt--Bell prevector satisfies $0<\|\mathrm{pre}_k\|\leq1$ for
    all $k>0$ (with $c,C_f\geq0$ chosen before the scanner, physical vectors,
    $\lambda_k$, and $S'$). Then every fill and charge round before $nm$, at
    every real interpolation parameter and $k>0$, satisfies
    (\ref{eq:actual_rough_weight_rate}) with remainder $R_k^{(a)}$ from
    (\ref{eq:actual_rough_weight_label_remainder}), $\mathcal N_k>0$,
    $\|v_k\|=1$, $\delta_k\geq0$, and $R_k^{(a)}\to0$ uniformly over all rounds
    and interpolation parameters.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:actual_scan_rough_lower_weight,
        thm:actual_rough_weight_remainder_limit,thm:actual_rough_lower_label_defect,
        thm:al_augmented_prevector_norm}
    By Theorems~\ref{thm:al_augmented_prevector_norm} and
    \ref{thm:actual_rough_lower_label_defect},
    Theorem~\ref{thm:actual_scan_rough_lower_weight} applies with
    $\varepsilon_k=2|\log\dim\lambda_k-kS'|$, and
    Theorem~\ref{thm:actual_rough_weight_remainder_limit} gives $R_k^{(a)}\to0$
    independently of the round and interpolation parameter.
\end{proof}

The weight-four statements below specialize $a=4/K$ and $W=4$ (with
$0<a\leq1/2$ for $K\geq8$).

\begin{theorem}[Scanner-root rough lower estimate]
    \label{thm:actual_scan_rough_lower_norm}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_actualScan_rough_lower_norm}
    \leanok
    Fix a finite physical site space, dimensions $q,d>0$, and $K\geq8$, and set
    $a=4/K$. There are $c,C_f\geq0$ (chosen before the scanner, horizon,
    physical vectors, $k$, $\ell$, and interpolation parameter) such that for
    every scanner with $K$ bands, $m,M>0$, truncated Hamiltonian
    $\widetilde H$, and unit vector $\widetilde\Omega$ satisfying
    \begin{align}
        \widetilde H-e_0 I
         & \succeq\gamma
        \bigl(I-|\widetilde\Omega\rangle
        \langle\widetilde\Omega|\bigr),
         & \gamma        & >0,
        \label{eq:actual_rough_lower_gap}
    \end{align}
    if $k>0$ and a nonzero copy-symmetric prevector of norm at most $1$ is
    fixed by $\ell$ on both auxiliary factors with
    $2kS'-\varepsilon\leq2\log\dim\ell$, and
    \begin{align}
        r_k & =\log(k+1)+c\log(k+2)+a\varepsilon,\notag \\
        R_k & =\frac{K}{2k}
        \left(2|r_k|+C_f\log(k+2)+2\log2+\log(3/2)\right),
        \label{eq:actual_rough_lower_finite_remainder}
    \end{align}
    then at any fill or charge interpolation before the horizon,
    $\mathcal N_k>0$, $\|v_k\|=1$, $\delta_k=(\mu_k-e_0)/\gamma\geq0$, and for
    every $0<\tau\leq1/2$,
    \begin{align}
        4S'-6\left(\tau\log d+\frac{h(\tau)}a\right)
        -3\frac{\delta_k}{\tau}\max\{0,2S'\}-R_k
         & \leq-\frac1k\log\mathcal N_k.
        \label{eq:actual_rough_lower_rate}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:actual_scan_rough_lower_weight}
    Since $K\geq8$ gives $0<a=4/K\leq1/2$ and $W=aK=4$,
    Theorem~\ref{thm:actual_scan_rough_lower_weight} yields
    (\ref{eq:actual_rough_lower_finite_remainder}) and
    (\ref{eq:actual_rough_lower_rate}).
\end{proof}

\begin{definition}[The same-label rough lower remainder]
    \label{def:actual_rough_lower_remainder}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughLowerLabelDefect,TNLean.PEPS.AreaLaw.Scan.roughLowerRemainder}
    \leanok
    For a label sequence $\ell_k$ and $S'\in\mathbb R$, set
    $\varepsilon_k=2|\log\dim\ell_k-kS'|$ in
    (\ref{eq:actual_rough_lower_finite_remainder}).
\end{definition}

\begin{theorem}[Label-defect entropy inequality]
    \label{thm:actual_rough_lower_label_defect}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughLowerLabelDefect_label_bound}
    \leanok
    For every $k$, $\varepsilon_k\geq0$ and
    $2kS'-\varepsilon_k\leq2\log\dim\ell_k$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:actual_rough_lower_remainder}
    Use $kS'-\log\dim\ell_k\leq|\log\dim\ell_k-kS'|$ in
    Definition~\ref{def:actual_rough_lower_remainder}.
\end{proof}

\begin{theorem}[Vanishing rough remainder and a common threshold]
    \label{thm:actual_rough_lower_remainder_limit}
    \lean{TNLean.PEPS.AreaLaw.Scan.tendsto_roughLowerRemainder_of_label_dimension,TNLean.PEPS.AreaLaw.Scan.exists_roughLowerRemainder_threshold}
    \leanok
    If $\log\dim\ell_k-kS'=o(k)$, then $R_k\to0$ for each fixed $K,c,C_f$, so
    for every $\epsilon_{\rm rep}>0$ there is $k_0$ (independent of the round
    and interpolation parameter) with $R_k<\epsilon_{\rm rep}$ for all
    $k\geq k_0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:actual_rough_weight_remainder_limit,def:actual_rough_lower_remainder}
    Specialize Theorem~\ref{thm:actual_rough_weight_remainder_limit} to
    $a=4/K$.
\end{proof}

\begin{theorem}[Rough lower rate for the physical label sequence]
    \label{thm:actual_scan_supplied_prevector_rough_lower}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_actualScan_suppliedPrevector_rough_lower}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:actual_scan_rough_lower_norm}, for
    a label sequence $\ell_k$ with $\log\dim\ell_k-kS'=o(k)$ whose physical
    prevector satisfies $0<\|\mathrm{pre}_k\|\leq1$ for all $k>0$ (with $c,C_f$
    fixed before the scanner and physical vector), every fill and charge before
    the full horizon satisfies (\ref{eq:actual_rough_lower_rate}) with
    $R_k\to0$, $\|v_k\|=1$, and $\delta_k\geq0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:actual_supplied_rough_lower_weight,def:actual_rough_lower_remainder}
    Specialize Theorem~\ref{thm:actual_supplied_rough_lower_weight} to
    $a=4/K\in(0,1/2]$ and $W=4$.
\end{proof}

In the physical application, $E$ and $\psi$ are chosen with $\widetilde\Omega$
and $S'=S(\operatorname{Tr}_{\rm left}|\psi\rangle\langle\psi|)$.
