\begin{definition}[Characteristic birth lengths]
    \label{def:area_law_repair_characteristic_length}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialRegionCharacteristicLength,
        TNLean.PEPS.AreaLaw.Geometry.repairStageCharacteristicLength}
    \leanok
    \uses{def:area_law_repair_stage}
    For an initial identifier family $I_0$, define the characteristic length
    $\tau_0:I_0\to\mathbb R$ (with layer-cell side $r_k=2^k$ and fine-cell side $t_k$) by
    \begin{align}
        \tau_0(i) & =
        \begin{cases}
            0,   & i\text{ is the dummy},                         \\
            r_k, & i\text{ is a primary in layer }k,              \\
            t_k, & i\text{ is an initial belt region in layer }k.
        \end{cases}
        \label{eq:area_law_initial_characteristic_length}
    \end{align}
    For the recursive family $\mathcal D_j$ (Definition~\ref{def:area_law_repair_stage}) with
    identifier set $I_j$, identify each old $i\in I_j$ with its retained copy in $I_{j+1}$,
    and for each fresh identifier $q=(j+1,p,H)$ with parent mark $p$ of scale $S_p^{(j)}$, define
    \begin{align}
        \tau_{j+1}(i)       & =\tau_j(i),    &
        \tau_{j+1}(j+1,p,H) & =d(S_p^{(j)}).
        \label{eq:area_law_repair_characteristic_length}
    \end{align}
\end{definition}

A replacement with $\tau_j(q)=u$ has removal half-side $u$ and scanner primitive scale $2u$
(\cite[Section~11]{OpenAI2026AreaLaw}). The dummy is omitted from the retained family; birth order
places primaries first, initial belt regions second, and replacements by generation.

\begin{theorem}[A uniform lower bound on nonprimary birth lengths]
    \label{thm:area_law_repair_positive_rank_length_lower_bound}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_uniform_repairStage_positive_rank_length_lower_bound}
    \leanok
    \uses{def:area_law_repair_characteristic_length,
        def:area_law_repair_birth_rank}
    For every $U\in\mathbb R$ there exists $K\in\mathbb N$ (depending only on $U$) such that
    for any origin $o\in\mathbb R^2$, finite domain $\Lambda\subset\mathbb Z^2$, cut $A\subseteq\Lambda$,
    integers $C_0\ge2$ and $k_0\ge\max(K,50{,}000{,}000)$, stopping scale $S_*\ge U$,
    generation $N\in\mathbb N$, and identifier $i\in I_N$ with $\operatorname{rk}_N(i)>0$,
    one has $\tau_N(i)\ge U$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:area_law_fine_belt_scales,
        def:area_law_repair_characteristic_length,
        def:area_law_repair_birth_rank,def:area_law_repair_successor_regions}
    Since $\zeta>0$, $t_k=2^{\lfloor\zeta k\rfloor}\to\infty$ (Definition~\ref{def:area_law_fine_belt_scales});
    choose $K$ so that $t_k\ge U$ for all $k\ge K$, and induct on $N$.
    At $N=0$, $\operatorname{rk}_0(i)>0$ (Definition~\ref{def:area_law_repair_birth_rank}) implies $i$ is a belt
    in layer $k\ge k_0\ge K$, so $\tau_0(i)=t_k\ge U$ by (\ref{eq:area_law_initial_characteristic_length}).
    At $N+1$, old identifiers preserve $\operatorname{rk}$ and $\tau$ by (\ref{eq:area_law_repair_characteristic_length}),
    while a fresh replacement has a nonterminal parent of scale $S$ with $\tau_{N+1}=d(S)\ge S_*\ge U$.
\end{proof}

\begin{theorem}[Templates for the remembered birth regions]
    \label{thm:area_law_repair_birth_templates}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_uniform_repairStage_birth_templates}
    \leanok
    \uses{def:area_law_repair_characteristic_length,
        def:area_law_repair_birth_rank,def:area_law_template,
        def:area_law_hamiltonian}
    Fix $C_{\rm tpl}\ge1$. There exist $C_P\ge1$ and $C_B>0$ such that for every $D_0\in\mathbb R$
    there exists $T_0\ge1$ with the following property: for every $S_*\ge T_0$ and integer
    $C_0>\max(1,4D_0)$, there exists $K\ge0$ such that for every $o\in\mathbb R^2$, finite
    $\Lambda\subset\mathbb Z^2$, cut $A\subseteq\Lambda$ with crossing-edge endpoints $Z$,
    and $k_0\ge\max(K,50{,}000{,}000)$, if $N\ge0$ and $i\in I_N\setminus\{*_N\}$ has
    remembered closed birth region $\mathcal B$ meeting $\mathbb Z^2$, characteristic length
    $u=\tau_N(i)$, birth rank $q=\operatorname{rk}_N(i)$ (Definition~\ref{def:area_law_repair_birth_rank}),
    and scanner scale and primitive size
    \begin{align}
        n   & =
        \begin{cases}
            \bigl\lceil C_Pu^{1+2\delta_0}\bigr\rceil, & q=0, \\
            \lceil C_Bu\rceil,                         & q>0,
        \end{cases}
            &
        s_0 & =
        \begin{cases}
            u,  & q\le1, \\
            2u, & q\ge2,
        \end{cases}
        \label{eq:area_law_birth_template_scales}
    \end{align}
    then $T=\mathcal B\cap\mathbb Z^2$ is a $(C_{\rm tpl},n,s_0)$-template
    (Definition~\ref{def:area_law_template}) and every $y\in T$, $z\in Z$ satisfy
    (\cite[Section~11]{OpenAI2026AreaLaw})
    \begin{align}
        4D_0s_0  & <\|y-z\|_\infty, &
        2n^\beta & <\|y-z\|_\infty.
        \label{eq:area_law_birth_template_clearance}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_primary_scanner_constant,
        def:area_law_primary_birth_template,
        thm:area_law_primary_fragment_union,
        thm:area_law_dyadic_layer_distance,
        thm:area_law_primary_birth_template_points,
        thm:area_law_primary_birth_template_clearance,
        thm:area_law_primary_scale_limits,
        thm:area_law_belt_replacement_templates,
        thm:area_law_belt_birth_template_clearance,
        thm:area_law_descendant_birth_template_clearance,
        def:area_law_repair_characteristic_length}
    Fix $C_P$ from Theorem~\ref{thm:area_law_primary_scanner_constant}, $C_B$ from
    Theorem~\ref{thm:area_law_belt_replacement_templates}, $T_0\ge1$ from
    Theorem~\ref{thm:area_law_descendant_birth_template_clearance}, and $K$ dominating the
    thresholds of Theorems~\ref{thm:area_law_primary_scale_limits},
    \ref{thm:area_law_belt_birth_template_clearance}, and
    \ref{thm:area_law_descendant_birth_template_clearance}. Induct on $N$.
    At $N=0$: for $q=0$ (primary of layer $k$), $\mathcal B\cap\mathbb Z^2\ne\varnothing$ supplies
    the template via Definition~\ref{def:area_law_primary_birth_template} and
    Theorem~\ref{thm:area_law_primary_birth_template_points}, while $4D_0<C_0$,
    Theorem~\ref{thm:area_law_primary_birth_template_clearance} ($\|y-z\|_\infty\ge C_0r_k$),
    and Theorem~\ref{thm:area_law_primary_scale_limits} ($n^\beta\le r_k/4$) yield
    (\ref{eq:area_law_birth_template_clearance}); for $q=1$ (initial belt),
    Theorems~\ref{thm:area_law_belt_replacement_templates} and
    \ref{thm:area_law_belt_birth_template_clearance} give the template with $s_0=t_k$ and
    (\ref{eq:area_law_birth_template_clearance}).
    At $N+1$: old identifiers inherit their witness unchanged by (\ref{eq:area_law_repair_characteristic_length}),
    while a fresh replacement has $q=N+2\ge2$ and $u=d(S_p^{(N)})$, so
    Theorems~\ref{thm:area_law_belt_replacement_templates} and
    \ref{thm:area_law_descendant_birth_template_clearance} supply the template with $s_0=2u$,
    $n=\lceil C_Bu\rceil$, and (\ref{eq:area_law_birth_template_clearance}).
\end{proof}

\begin{theorem}[Positive characteristic lengths]
    \label{thm:area_law_repair_characteristic_length_positive}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStageCharacteristicLength_pos_of_ne_dummy}
    \leanok
    \uses{def:area_law_repair_characteristic_length,def:area_law_repair_stage}
    For any $o\in\mathbb R^2$, finite $\Lambda\subset\mathbb Z^2$, cut $A\subseteq\Lambda$,
    integers $C_0\ge2$ and $k_0\ge50{,}000{,}000$, and $S_*\in\mathbb R$, every non-dummy
    identifier in the repair family satisfies
    \begin{align}
        \tau_N(i) & >0
        \qquad(N\in\mathbb N,\ i\in I_N\setminus\{*_N\}).\notag
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_characteristic_length,
        def:area_law_repair_square,def:area_law_repair_stage}
    By induction on $N$: at $N=0$, $\tau_0(i)\in\{r_k,t_k\}$ is a positive power of $2$ by
    (\ref{eq:area_law_initial_characteristic_length}); at $N+1$, (\ref{eq:area_law_repair_characteristic_length})
    preserves $\tau_N(i)>0$ for old $i\ne *_N$ and assigns $\tau_{N+1}=d(S)>0$ (a power of $2$ by
    Definition~\ref{def:area_law_repair_square}) to fresh identifiers.
\end{proof}

\begin{lemma}[Preservation under retention]
    \label{lem:area_law_repair_retention_identities}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStageOldInclusion_injective,
        TNLean.PEPS.AreaLaw.Geometry.repairStageOldInclusion_dummy,
        TNLean.PEPS.AreaLaw.Geometry.repairStage_color_oldInclusion,
        TNLean.PEPS.AreaLaw.Geometry.repairStageCharacteristicLength_oldInclusion}
    \leanok
    \uses{def:area_law_repair_inclusions,def:area_law_repair_characteristic_length,
        thm:area_law_retained_birth_lattice_point}
    For any $o\in\mathbb R^2$, finite $\Lambda\subset\mathbb Z^2$, cut $A\subseteq\Lambda$,
    $C_0\ge2$, $k_0\ge50{,}000{,}000$, $S_*\in\mathbb R$, and $j,n\in\mathbb N$, the retention map
    $\iota_{j,j+n}:I_j\to I_{j+n}$ (Definition~\ref{def:area_law_repair_inclusions}) of $\mathcal D_j$
    is injective, maps $\iota_{j,j+n}(*_j)=*_{j+n}$, and satisfies for all $i\in I_j$:
    \begin{align}
        B^{(j+n)}_{\iota_{j,j+n}(i)}   & =B^{(j)}_i, &
        \xi^{(j+n)}_{\iota_{j,j+n}(i)} & =\xi^{(j)}_i, &
        \tau_{j+n}(\iota_{j,j+n}(i))   & =\tau_j(i).
        \label{eq:area_law_repair_retention_identities}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_inclusions,def:area_law_repair_stage,
        def:area_law_repair_successor_regions,
        def:area_law_repair_characteristic_length,
        thm:area_law_retained_birth_lattice_point}
    Induct on $n$: $\iota_{j,j}=\mathrm{id}$, and each step $I_{j+n}\hookrightarrow I_{j+n+1}$
    is a disjoint-union summand inclusion preserving dummy, birth, label, and $\tau$ by
    Definitions~\ref{def:area_law_repair_successor_regions} and
    \ref{def:area_law_repair_characteristic_length}.
\end{proof}

\begin{lemma}[Rank-zero ancestors at an earlier generation]
    \label{lem:area_law_repair_rank_zero_ancestry}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_repairStageOldInclusion_of_repairStageBirthRank_eq_zero}
    \leanok
    \uses{def:area_law_repair_inclusions,def:area_law_repair_birth_rank}
    For every $j,n\in\mathbb N$ and $x\in I_{j+n}$ with $\operatorname{rk}_{j+n}(x)=0$,
    there exists $y\in I_j$ with $\operatorname{rk}_j(y)=0$ and $\iota_{j,j+n}(y)=x$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_inclusions,def:area_law_repair_birth_rank}
    Induct on $n$: $n=0$ takes $y=x$, and at $j+n+1$ every fresh identifier has
    rank $j+n+2>0$ (Definition~\ref{def:area_law_repair_birth_rank}), so $\operatorname{rk}_{j+n+1}(x)=0$
    forces $x$ to be old.
\end{proof}

\begin{lemma}[The unique original primary of a rank-zero identifier]
    \label{lem:area_law_repair_rank_zero_primary_ancestor}
    \lean{TNLean.PEPS.AreaLaw.Geometry.existsUnique_initialPrimary_of_repairStageBirthRank_eq_zero}
    \leanok
    \uses{def:area_law_repair_inclusions,def:area_law_repair_birth_rank,
        def:area_law_initial_region_indices,def:area_law_primary_pitch_indices}
    Let $G_k$ be the primary pitch index set at layer $k$
    (Definition~\ref{def:area_law_primary_pitch_indices}). For every $n\in\mathbb N$ and
    $x\in I_n\setminus\{*_n\}$ with $\operatorname{rk}_n(x)=0$, there exists a unique
    pair $(k,J)$ with $k\ge k_0$ and $J\in G_k$ such that $\iota_{0,n}((k,J))=x$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:area_law_repair_rank_zero_ancestry,
        lem:area_law_repair_retention_identities,
        def:area_law_initial_region_indices,def:area_law_repair_birth_rank}
    By Lemma~\ref{lem:area_law_repair_rank_zero_ancestry}, $x=\iota_{0,n}(y)$ for some $y\in I_0$
    with $\operatorname{rk}_0(y)=0$. Since $\iota_{0,n}(*_0)=*_n\ne x$
    (Lemma~\ref{lem:area_law_repair_retention_identities}) and initial belts have rank $1$,
    $y=(k,J)$ is a primary in $I_0$ (Definition~\ref{def:area_law_initial_region_indices});
    uniqueness follows from injectivity of $\iota_{0,n}$.
\end{proof}
