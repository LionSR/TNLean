\section{Product Vacua with Boundary States}
\label{sec:examples_pvbs}

Following~\cite[Section~II, equations~(1)--(8)]{BachmannNachtergaele2012PVBS},
the one-species PVBS chain has a product vacuum and a geometric boundary
particle. In left-to-right matrix products with hopping parameter and phase
$q=\mu e^{i\theta}$, the real parameter of
\cite[Section~IV.C.4]{Cirac2021Matrix} is $q=1+\lambda$; as explained
in~\cite{gap:cpgsv21_pvbs_threshold}, the review's printed right-localization
threshold and unconditional periodic-uniqueness claim require correction. The
finite-chain results below hold for all complex $q$ (with $q=0$ as an
algebraic extension of the original positive-hopping family) and make no
spectral-gap assertion.

\begin{definition}[One-species PVBS tensor and boundary vectors]
    \label{def:pvbs_tensor}
    \lean{MPSTensor.pvbsTensor}
    \lean{MPSTensor.pvbsVacuum}
    \lean{MPSTensor.pvbsEdge}
    \leanok
    The one-species PVBS tensor has matrices $A^0=\operatorname{diag}(q,1)$
    and $A^1=|0\rangle\langle1|$, with vacuum vector
    $\Omega_N=|0\cdots0\rangle$ and boundary-particle vector
    $\Psi_N(q)=\sum_{k=0}^{N-1}q^k|0\cdots1_k\cdots0\rangle$.
\end{definition}

\begin{lemma}[PVBS amplitudes and boundary contraction]
    \label{lem:pvbs_amplitudes}
    \lean{MPSTensor.pvbsVacuum_zero}
    \lean{MPSTensor.pvbsVacuum_nil}
    \lean{MPSTensor.pvbsVacuum_cons}
    \lean{MPSTensor.pvbsEdge_cons_zero}
    \lean{MPSTensor.pvbsEdge_cons_one}
    \lean{MPSTensor.pvbsEdge_zero}
    \lean{MPSTensor.pvbsEdge_first}
    \lean{MPSTensor.pvbsVacuum_first}
    \lean{MPSTensor.evalWord_pvbsTensor}
    \lean{MPSTensor.pvbsTensor_openState}
    \lean{MPSTensor.pvbsEdge_excitedAt}
    \lean{MPSTensor.pvbsVacuum_excitedAt}
    \leanok
    \uses{def:pvbs_tensor}
    For every physical word $\sigma$,
    \begin{align}
        A^{\sigma_1}\cdots A^{\sigma_N}
         & =\begin{pmatrix}
                q^N\Omega_N(\sigma) & \Psi_N(q)(\sigma) \\
                0                   & \Omega_N(\sigma)
            \end{pmatrix}.
        \label{eq:pvbs_word}
    \end{align}
    The standard boundary contraction $\langle0|A^{\sigma_1}\cdots
        A^{\sigma_N}|1\rangle$ is $\Psi_N(q)(\sigma)$.
\end{lemma}
\begin{proof}
    \leanok
    Because multiplication by $A^0$ contributes a factor of $q$ before the
    unique particle, multiplication by $A^1$ creates the particle, and two
    particles give the zero matrix, induction on the word length yields
    (\ref{eq:pvbs_word}).
\end{proof}

\begin{lemma}[Open PVBS matrix-product space]
    \label{thm:pvbs_open_space}
    \lean{MPSTensor.groundSpaceMap_pvbsTensor}
    \lean{MPSTensor.mem_groundSpace_pvbsTensor_iff}
    \lean{MPSTensor.groundSpace_pvbsTensor}
    \lean{MPSTensor.linearIndependent_pvbs_states}
    \lean{MPSTensor.finrank_groundSpace_pvbsTensor}
    \leanok
    \uses{lem:pvbs_amplitudes}
    The full boundary matrix-product space is
    $\mathcal G_N=\operatorname{span}\{\Omega_N,\Psi_N(q)\}$.
    It has dimension two for every $N\ge1$.
\end{lemma}
\begin{proof}
    \leanok
    Taking the trace against a boundary matrix $X$ gives
    $(q^NX_{00}+X_{11})\Omega_N+X_{10}\Psi_N(q)$.
    The entries $X_{11}$ and $X_{10}$ realize arbitrary coefficients.
    Evaluation on the vacuum and on a particle at the first site proves
    linear independence.
\end{proof}

\begin{lemma}[Open PVBS parent-Hamiltonian ground space]
    \label{thm:pvbs_open_parent}
    \lean{MPSTensor.pvbs_pair_constraints}
    \lean{MPSTensor.mem_groundSpace_pvbs_two_iff}
    \lean{MPSTensor.groundSpace_pvbs_intersection}
    \lean{MPSTensor.contiguous_mem_groundSpace_pvbs}
    \lean{MPSTensor.ker_openParentHamiltonianES_pvbs}
    \lean{MPSTensor.finrank_ker_openParentHamiltonianES_pvbs}
    \leanok
    \uses{thm:pvbs_open_space}
    The open nearest-neighbour parent Hamiltonian has kernel $\mathcal G_N$
    and ground-state dimension two for every $N\ge2$.
\end{lemma}
\begin{proof}
    \leanok
    A two-site vector satisfies the local constraint precisely when
    $a_{11}=0$ and $a_{01}=q a_{10}$. For a longer vector whose left and right
    restrictions lie in the shorter vacuum-particle spaces, their shared
    two-site constraints eliminate the two-particle coefficient and relate the
    remaining particle coefficients by $q$, establishing the intersection
    property from length two onward. Iterating this intersection property
    identifies the subspace cut out by all nonwrapping local constraints with
    $\mathcal G_N$, which equals the kernel of the positive parent
    Hamiltonian.
\end{proof}

\begin{lemma}[Periodic PVBS vacuum uniqueness]
    \label{thm:pvbs_periodic_parent}
    \lean{MPSTensor.chainGroundSpace_pvbs_le_groundSpace}
    \lean{MPSTensor.pvbsVacuum_mem_chainGroundSpace}
    \lean{MPSTensor.cyclicTranslateCfg_excitedAt}
    \lean{MPSTensor.pvbs_periodic_particle_coefficient}
    \lean{MPSTensor.chainGroundSpace_pvbs_eq_vacuum}
    \lean{MPSTensor.ker_parentHamiltonian_pvbs}
    \lean{MPSTensor.finrank_ker_parentHamiltonian_pvbs}
    \lean{MPSTensor.pvbs_pow_ne_one_of_norm_ne_one}
    \lean{MPSTensor.ker_parentHamiltonian_pvbs_of_norm_ne_one}
    \leanok
    \uses{thm:pvbs_open_parent}
    If $N\ge2$ and $q^N\ne1$, the periodic parent Hamiltonian has kernel
    $\operatorname{span}\{\Omega_N\}$ and ground-state dimension one.
    In particular, this conclusion holds at every such length if $|q|\ne1$.
\end{lemma}
\begin{proof}
    \leanok
    Every periodic ground vector satisfies the nonwrapping constraints and
    therefore takes the form $a\Omega_N+b\Psi_N(q)$. Because its cyclic
    translate is also a periodic ground vector, comparing particle amplitudes
    at the first two sites gives $b=bq^N$, which forces $b=0$, while the
    vacuum $\Omega_N$ satisfies every cyclic constraint.
\end{proof}

\begin{lemma}[Critical W state and geometric localization]
    \label{thm:pvbs_boundary_regimes}
    \lean{MPSTensor.pvbsTensor_one}
    \lean{MPSTensor.pvbsEdge_one}
    \lean{MPSTensor.pvbsEdge_one_two_mem_chainGroundSpace}
    \lean{MPSTensor.pvbs_critical_periodic_counterexample}
    \lean{MPSTensor.norm_pvbsEdge_excitedAt}
    \lean{MPSTensor.pvbsEdge_left_decay}
    \lean{MPSTensor.pvbsEdge_right_relative}
    \lean{MPSTensor.pvbsEdge_right_decay}
    \leanok
    \uses{lem:pvbs_amplitudes,thm:pvbs_periodic_parent}
    At $q=1$, the tensor and boundary vector are the W-state tensor and
    unnormalized W vector. Already on the two-site ring, the W vector is a
    nonvacuum zero-energy state. For $|q|<1$, amplitudes decay as $|q|^k$
    from the left edge. For $|q|>1$, the amplitude $k$ sites from the right
    edge relative to the last-site amplitude equals $q^{-k}$ and tends to zero.
\end{lemma}
\begin{proof}
    \leanok
    Substitution gives the W tensor at $q=1$ and verifies the two periodic
    constraints. Evaluation on a one-particle configuration separates the
    W vector from the vacuum span. The localization statements follow from
    the exact geometric coefficients and convergence of powers of a scalar
    of modulus less than one.
\end{proof}

\subsection{Pure one-particle windows and cyclic phase}

\begin{lemma}[Overlapping one-dimensional windows with an entangled cut]
    \label{lem:overlapping_one_dimensional_windows_zero}
    \lean{MPSPreparation.eq_zero_of_overlapping_one_dimensional_sections}
    \notready
    Let $W_{ab}$ and $V_{bc}$ be two coefficient arrays. Suppose some
    two-by-two minor of $W$ is nonzero. If a three-part coefficient
    array satisfies
    \begin{align}
        \psi_{abc}=f_cW_{ab}=g_aV_{bc},
    \end{align}
    then $\psi=0$.
\end{lemma}
\begin{proof}
    \notready
    Fix $c$. Multiplying the nonzero minor by $f_c^2$ and substituting
    the two section identities makes its two products cancel. Thus
    $f_c^2=0$, whence $f_c=0$ and every coefficient vanishes.
\end{proof}

\begin{lemma}[Two pure geometric one-particle windows are incompatible]
    \label{lem:pvbs_pure_window_overlap_zero}
    \lean{MPSTensor.pvbsEdge_first_cut_minor,
        MPSTensor.eq_zero_of_pvbsEdge_overlapping_sections}
    \notready
    \uses{def:pvbs_tensor,lem:pvbs_amplitudes,
        lem:overlapping_one_dimensional_windows_zero}
    For $q\ne0$, the geometric one-particle vector on $n\ge2$ sites has
    a first-cut minor equal to $q$. Consequently no nonzero vector on
    $n+1$ sites can have all its first-window sections proportional to
    this vector and all its last-window sections proportional to the
    same vector. These are actual coefficient sections of the vector
    on the union.
\end{lemma}
\begin{proof}
    \notready
    At the first cut take first-site values $1,0$ and tail words equal
    to the vacuum and its first excitation. The coefficient matrix is
    $\left(\begin{smallmatrix}1&0\\0&q\end{smallmatrix}\right)$.
    Apply the preceding overlap obstruction.
\end{proof}

\begin{lemma}[Right-endpoint one-particle coefficients]
    \label{lem:pvbs_right_endpoint_coefficients}
    \lean{MPSTensor.pvbsVacuum_snoc,MPSTensor.pvbsEdge_snoc}
    \notready
    \uses{lem:pvbs_amplitudes}
    Appending a zero site to the vacuum keeps its amplitude, and
    appending a particle annihilates it. For the geometric particle
    vector on $n+1$ sites, a zero at the last site leaves the length-$n$
    particle amplitude; a particle there has amplitude $q^n$ times the
    length-$n$ vacuum amplitude.
\end{lemma}
\begin{proof}
    \notready
    Use the defining first-site recursion and induction on the number
    of sites. The vacuum condition is equality with the all-zero word.
\end{proof}

\begin{lemma}[The cyclic phase of a geometric one-particle vector]
    \label{lem:pvbs_geometric_cyclic_phase}
    \lean{MPSTensor.pvbsEdge_cyclic_step,
        MPSTensor.pvbsEdge_not_cyclic_invariant}
    \notready
    \uses{lem:pvbs_right_endpoint_coefficients}
    On a chain of $n\ge1$ sites, if $q^n=1$, rotating the physical
    coefficients by one site satisfies
    \begin{align}
        q\,W_q(\sigma_1,\ldots,\sigma_{n-1},\sigma_0)=W_q(\sigma).
    \end{align}
    For $q\ne1$, the actual coefficient vector is not invariant,
    although its line is invariant. This is a statement about the
    given one-particle vector, not about a fixed Hamiltonian at all volumes.
\end{lemma}
\begin{proof}
    \notready
    Move the first site to the right endpoint and apply the preceding
    recursion. The case with a particle at the first site uses
    $q\,q^{n-1}=1$. Exact invariance would force $q=1$ by evaluating the
    first-site excitation, whose amplitude is one.
\end{proof}
