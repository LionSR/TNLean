\section{Radial incidence and convexity in an elementary fan}
\label{sec:area_law_fan_radial_incidence_convexity}

Let $o\in\R^2$, $\ell\in\N$, and $z\in\Z^2$, and let
$Q=o+2^\ell(z+[0,1]^2)$ be the corresponding closed dyadic square with
center $c$. On each side of $Q$, take either the full side or its two
midpoint halves, oriented counterclockwise along $\partial Q$. Let
$\mathcal E$ index the resulting directed perimeter segments $[a_i,b_i]$,
and set
\begin{align}
    P_i & =\operatorname{conv}\{c,a_i,b_i\}, \\
    R_i & =[c,b_i].
\end{align}
These are the triangles and radial segments of the elementary fan in
\cite[Section~11]{OpenAI2026AreaLaw}. Every endpoint satisfies
$\|b_i-c\|_\infty=2^\ell/2>0$, and the directed rays in direction
$b_i-c$ for $i\in\mathcal E$ are pairwise distinct.

\begin{theorem}[Noncentral radial incidence]
    \label{thm:area_law_fan_noncentral_radial_incidence}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_mem_iff_of_mem_radial}
    \leanok
    \uses{def:area_law_cell_fan}
    For $i,t\in\mathcal E$ and $x\in R_t$ with $x\ne c$,
    \begin{align}
        x\in P_i & \quad\Longleftrightarrow\quad
        t=i\ \text{or}\ b_t=a_i.
        \label{eq:fan_radial_incidence}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_fan_triangle_contact,
        thm:area_law_cell_fan_end_same_ray}
    By convexity, $R_t=[c,b_t]\subseteq P_t$. If $t=i$, then $x\in P_i$; if
    $b_t=a_i$, then $R_t=[c,a_i]\subseteq P_i$.

    Conversely, suppose $x\in P_i$ and $t\ne i$. Then $P_t\cap P_i$ contains
    the distinct points $c$ and $x$, so the classification of contacts
    between distinct fan triangles gives either $b_t=a_i$ or
    \begin{align}
        b_i         & =a_t, \\
        P_t\cap P_i & =R_i.
    \end{align}
    In the second case, $x\in R_t\cap R_i\setminus\{c\}$, so there exist
    $u,v>0$ with
    \begin{align}
        x-c & =u(b_t-c)=v(b_i-c).
    \end{align}
    Thus $b_t-c$ and $b_i-c$ define the same directed ray, forcing $t=i$, a
    contradiction. This proves~(\ref{eq:fan_radial_incidence}).
\end{proof}

\begin{theorem}[Affine basis of an elementary fan triangle]
    \label{thm:area_law_fan_triangle_affine_basis}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_exists_affineBasis}
    \leanok
    \uses{def:area_law_cell_fan}
    For every $i\in\mathcal E$, the triple $(c,a_i,b_i)$
    is an affine basis of $\R^2$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_cell_fan_regularity}
    Since $P_i$ has nonempty interior, its affine span is $\R^2$, so its three
    vertices $(c,a_i,b_i)$ are affinely independent and span $\R^2$.
\end{proof}

\begin{theorem}[Convexity after removing a radial segment]
    \label{thm:area_law_fan_radial_deletion_convex}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_sdiff_radial_convex}
    \leanok
    \uses{def:area_law_cell_fan}
    For every $i,t\in\mathcal E$, the set $P_i\setminus R_t$ is convex.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_fan_noncentral_radial_incidence,
        thm:area_law_fan_triangle_affine_basis}
    Since $c\in R_t$, (\ref{eq:fan_radial_incidence}) implies
    \begin{align}
        P_i\cap R_t & =
        \begin{cases}
            [c,b_i], & t=i,                         \\
            [c,a_i], & t\ne i\ \text{and}\ b_t=a_i, \\
            \{c\},   & \text{otherwise}.
        \end{cases}
        \label{eq:fan_radial_face_cases}
    \end{align}
    Let $\lambda_0,\lambda_1,\lambda_2$ be the barycentric coordinates on
    $\R^2$ with respect to the affine basis $(c,a_i,b_i)$. On $P_i$ these
    coordinates are nonnegative and sum to $1$, so
    \begin{align}
        P_i\setminus[c,b_i]
         & =P_i\cap\{x:\lambda_1(x)>0\},              \\
        P_i\setminus[c,a_i]
         & =P_i\cap\{x:\lambda_2(x)>0\},              \\
        P_i\setminus\{c\}
         & =P_i\cap\{x:\lambda_1(x)+\lambda_2(x)>0\}.
        \label{eq:fan_radial_convex_deletions}
    \end{align}
    Each set in~(\ref{eq:fan_radial_convex_deletions}) is the intersection of
    $P_i$ with an open half-plane and is therefore convex; by~(\ref{eq:fan_radial_face_cases}),
    $P_i\setminus R_t$ is convex.
\end{proof}

\begin{theorem}[Convexity after removing a family of radial segments]
    \label{thm:area_law_fan_radial_union_deletion_convex}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_sdiff_radials_convex}
    \leanok
    \uses{def:area_law_cell_fan}
    For every $i\in\mathcal E$ and every subset $A\subseteq\mathcal E$,
    the set $P_i\setminus\bigcup_{t\in A}R_t$ is convex.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_fan_radial_deletion_convex}
    Write
    \begin{align}
        P_i\setminus\bigcup_{t\in A}R_t
         & =P_i\cap\bigcap_{t\in A}(P_i\setminus R_t).
    \end{align}
    Since $P_i$ and each $P_i\setminus R_t$ are convex, their intersection is
    convex.
\end{proof}
