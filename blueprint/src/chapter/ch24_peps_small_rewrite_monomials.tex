\section{Small-patch rewrites as allowed monomials}
\label{sec:peps_small_rewrite_monomials}
The statements below formalize the last step of the proof of the second
assertion of Lemma~6.3 of \cite{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 262--273 and 306--316: every product term of the
truncated branch expansion
(Theorem~\ref{thm:peps_frame_small_rewrite_product_terms}) is a monomial
allowed by \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} on the registers of the
frames, one register per site and per tag
(Definition~\ref{def:peps_frame_registers}). The number of monomials is bounded
by the number of branches times $k^{2m}$; it is polynomial in $L$ once the
number of cylinder terms of each patch and each affected hole is polynomial in $L$ and $m$,
$r_{\rm old}$, $r_{\rm new}$ and the specified list of parties are bounded,
as shown in
Theorem~\ref{thm:peps_frame_small_rewrite_polynomial}.

Throughout, $\Lambda$ is a finite set of sites, each with a site register
$\C^q$, and operators on $\mathcal H_\Lambda$ are written in the
coordinates of Lemma~\ref{lem:peps_frame_registers_coordinates}.

\begin{lemma}[A rank-one map on two registers with a change of owners]
    \label{lem:peps_rank_one_chain}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.append,
        TNLean.Lattice.PartyCircuit.PartyChain.eval_append,
        TNLean.Lattice.PartyCircuit.PartyChain.append_props,
        TNLean.PEPS.PairEffect.braLocal,
        TNLean.PEPS.PairEffect.ketLocal,
        TNLean.PEPS.PairEffect.norm_braLocal_le,
        TNLean.PEPS.PairEffect.norm_ketLocal_le,
        TNLean.Lattice.PartyCircuit.owner_of_mem_pair,
        TNLean.Lattice.PartyCircuit.owner_of_mem_nil,
        TNLean.PEPS.PairEffect.braChain,
        TNLean.PEPS.PairEffect.ketWord,
        TNLean.PEPS.PairEffect.rankOneChain,
        TNLean.PEPS.PairEffect.eval_braChain,
        TNLean.PEPS.PairEffect.pairHeadIso_ketWord,
        TNLean.PEPS.PairEffect.act_vecMulVec_star,
        TNLean.PEPS.PairEffect.pairHeadIso_rankOneChain,
        TNLean.PEPS.PairEffect.rankOneChain_props}
    \leanok
    \uses{def:peps_party_monomial,def:peps_monomial_parties,
        thm:peps_monomial_composition_counts,lem:peps_front_registers}
    Two allowed monomials on parties in $S$ applied in succession form an allowed
    monomial on $S$ whose pair-source and pair-effect counts add.
    For unit vectors $v,w\in\C^\alpha\otimes\C^\beta$, the map $\ket v\bra w$ from a register
    $\C^\alpha$ of $p_A$ and a register $\C^\beta$ of $p_B$ to output registers held by $p_A'$ and $p_B'$
    (tensored with $\id$ elsewhere) is an allowed monomial using only
    $\{p_A,p_B,p_A',p_B'\}$: $\bra w$ is a normalized pair effect if
    $p_A\ne p_B$ and a private contraction at $p_A$ if $p_A=p_B$, while
    $\ket v$ is a normalized pair source if $p_A'\ne p_B'$ and a private
    preparation at $p_A'$ if $p_A'=p_B'$. Thus it has at most $1_{p_A'\ne p_B'}$ pair sources and at most $1_{p_A\ne p_B}$ pair effects.
\end{lemma}

\begin{proof}\leanok
    Composing two monomials appends the second to the last
    source-only segment of the first (Theorem~\ref{thm:peps_monomial_composition_counts}). Since $\bra w(y\otimes t)=\langle w,y\rangle t$ and $\ket v(t)=v\otimes t$, their composite is $\langle w,y\rangle v\otimes t=(\ket v\bra w\,y)\otimes t$, and $\lVert v\rVert=\lVert w\rVert=1$ ensures contractive private maps when $p_A=p_B$ or $p_A'=p_B'$.
\end{proof}

\begin{definition}[Rank-one blocks]
    \label{def:peps_rank_one_block}
    \lean{TNLean.PEPS.EncodedFrame.RankOneBlock,
        TNLean.PEPS.EncodedFrame.RankOneBlock.op,
        TNLean.PEPS.EncodedFrame.RankOneBlock.ketAt,
        TNLean.PEPS.EncodedFrame.RankOneBlock.braAt,
        TNLean.PEPS.EncodedFrame.RankOneBlock.op_apply,
        TNLean.PEPS.EncodedFrame.RankOneBlock.ketAt_congr,
        TNLean.PEPS.EncodedFrame.RankOneBlock.braAt_congr,
        TNLean.PEPS.EncodedFrame.zeroPair,
        TNLean.PEPS.EncodedFrame.norm_zeroPair,
        TNLean.PEPS.EncodedFrame.zeroPair_apply,
        TNLean.PEPS.EncodedFrame.zeroBlock,
        TNLean.PEPS.EncodedFrame.zeroBlock_ketAt,
        TNLean.PEPS.EncodedFrame.zeroBlock_braAt,
        TNLean.PEPS.EncodedFrame.emptyBlock,
        TNLean.PEPS.EncodedFrame.emptyBlock_ketAt,
        TNLean.PEPS.EncodedFrame.emptyBlock_braAt,
        TNLean.PEPS.EncodedFrame.emptyBlock_op}
    \leanok
    \uses{lem:peps_grouped_monomial_placement}
    A rank-one block $b=(T,E,v,w)$ consists of disjoint sets $T,E\subseteq\Lambda$, with
    $U=\Lambda\setminus(T\cup E)$, and vectors $v,w\in\C^{[q]^T\times[q]^U}$. Its operator $\mathrm{op}(b)=(\ket v\bra w)_{T\cup U}\otimes\id_E$ has matrix entries
    \begin{align}
        \braket{c'}{\mathrm{op}(b)|c}=1_{c'|_E=c|_E}\,v(c'|_T,c'|_U)\,\overline{w(c|_T,c|_U)}.
        \label{eq:peps_rank_one_block_entry}
    \end{align}
    The one-site zero block at $x$ has $T=\{x\}$, $U=\emptyset$, $v=w=\ket0$, and $\mathrm{op}(b)=(\ket0\bra0)_{\{x\}}\otimes\id$; the trivial block has $T=U=\emptyset$ and $\mathrm{op}(b)=\id$.
\end{definition}

\begin{lemma}[Products of disjoint blocks]
    \label{lem:peps_rank_one_block_products}
    \lean{TNLean.PEPS.EncodedFrame.RankOneBlock.prod_op_apply}
    \leanok
    \uses{def:peps_rank_one_block}
    Let $b_1,\dots,b_r$ be rank-one blocks whose active sets $\Lambda\setminus E_{b_i}$ are pairwise disjoint. Then
    \begin{align}
        \braket{c'}{\mathrm{op}(b_1)\cdots\mathrm{op}(b_r)|c}
        =1_{c'|_{\bigcap_i E_{b_i}}=c|_{\bigcap_i E_{b_i}}}\prod_{i=1}^r v_{b_i}(c')\,\overline{w_{b_i}(c)},
        \label{eq:peps_rank_one_block_prod}
    \end{align}
    where $v_b(c)=v(c|_T,c|_U)$ and $w_b(c)=w(c|_T,c|_U)$.
\end{lemma}

\begin{proof}\leanok
    By induction on $r$ using~(\ref{eq:peps_rank_one_block_entry}): in $\braket{c'}{\mathrm{op}(b_1)X|c}$ with $X=\mathrm{op}(b_2)\cdots\mathrm{op}(b_r)$, only the intermediate configuration matching $c'$ on $E_{b_1}$ and $c$ on $\Lambda\setminus E_{b_1}\subseteq\bigcap_{i\ge2}E_{b_i}$ contributes.
\end{proof}

\begin{lemma}[Rank-one blocks as allowed monomials on a frame]
    \label{lem:peps_rank_one_block_monomial}
    \lean{TNLean.PEPS.EncodedFrame.exists_rankOneBlock_chain,
        TNLean.PEPS.EncodedFrame.exists_pairVec,
        TNLean.PEPS.EncodedFrame.exists_groupBlock,
        TNLean.PEPS.EncodedFrame.zeroGroupVec,
        TNLean.PEPS.EncodedFrame.norm_zeroGroupVec,
        TNLean.PEPS.EncodedFrame.zeroGroupVec_apply,
        TNLean.PEPS.EncodedFrame.exists_braVertexBlock,
        TNLean.PEPS.EncodedFrame.exists_ketVertexBlock,
        TNLean.PEPS.EncodedFrame.prod_map_one_kronecker,
        TNLean.PEPS.EncodedFrame.exists_chain_listProd,
        TNLean.PEPS.EncodedFrame.exists_zeroChain}
    \leanok
    \uses{lem:peps_rank_one_chain,def:peps_rank_one_block,
        lem:peps_grouped_monomial_placement}
    Let $b=(T,E,v,w)$ be a rank-one block with unit vectors $v,w$, where the site registers
    of $T$ have owner $p_A$ before and $p_A'$ after, those of $U$ have $p_B$
    and $p_B'$, and those of $E$ keep their owner. Then
    $\id_{\rm tags}\otimes\mathrm{op}(b)$ is the operator of an allowed
    monomial between the two frame register layouts, using only $\{p_A,p_B,p_A',p_B'\}$, with at most one pair source
    and at most one pair effect. Consequently:
    \begin{enumerate}
        \item if $\emptyset\ne O\subseteq\Lambda$ has site owners in a subset of $S$ of size at most $2$, then for unit vectors $v,w\in\C^{[q]^O}$,
              $\id_{\rm tags}\otimes(\ket v\bra w)_O\otimes\id_{\Lambda\setminus O}$
              is such a monomial on $S$ preserving site owners (while for $O=\emptyset$, a unit covector or vector on $[q]^O$ is a scalar of modulus $\le1$);
        \item for distinct sites $x_1,\dots,x_r$ outside of which two owner
              assignments agree, the product of the one-site zero blocks at $x_1,\dots,x_r$ is the operator of an allowed monomial between the two owner layouts using only the owners of $x_1,\dots,x_r$ and no pair resources;
        \item a product of operators of allowed monomials on $S$ is the operator of an
              allowed monomial on $S$ whose pair-source and pair-effect counts sum.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    Grouping the site registers of $T$ and $U$, applying
    Lemma~\ref{lem:peps_rank_one_chain}, and ungrouping via
    Lemma~\ref{lem:peps_grouped_monomial_placement} proves the main claim; item~1 follows by partitioning $O=T\cup U$ by owner and setting $E=\Lambda\setminus O$, item~2 by contracting with $\bra0$ at the old owner and preparing $\ket0$ at the new owner at each $x_i$, and item~3 by Lemma~\ref{lem:peps_rank_one_chain}.
\end{proof}

\begin{lemma}[A product term as a product of blocks]
    \label{lem:peps_product_term_blocks}
    \lean{TNLean.PEPS.SiteChain.eq_of_not_earlierTouch,
        TNLean.PEPS.SiteChain.eq_of_not_laterTouch,
        TNLean.PEPS.SiteChain.eq_of_isOpen_inl,
        TNLean.PEPS.SiteChain.eq_of_isOpen_inr,
        TNLean.PEPS.SiteChain.not_untouched_of_isOpen,
        TNLean.PEPS.SiteChain.fixedIn_iff,
        TNLean.PEPS.SiteChain.fixedOut_iff,
        TNLean.PEPS.SiteChain.prod_braBlocks_apply,
        TNLean.PEPS.SiteChain.prod_ketBlocks_apply,
        TNLean.PEPS.SiteChain.prod_zeroBlocks_apply,
        TNLean.PEPS.SiteChain.termOp_eq_smul_prod}
    \leanok
    \uses{lem:peps_frame_chain_network,lem:peps_frame_chain_truncation,
        lem:peps_rank_one_block_products}
    In the network of $L_0\cdots L_{n-1}$ from Lemma~\ref{lem:peps_frame_chain_network} with $W=\bigcup_{j=0}^{n-1}S_j$,
    an open input leg at $x\in W$ belongs to the first applied bra containing $x$, an open output
    leg to the last applied ket containing $x$, and a leg at $x\in W$ is fixed if and only if it is not open. For any choice of truncation vectors on the open-leg groups in Lemma~\ref{lem:peps_frame_chain_truncation}, the resulting product term $T$ satisfies
    \begin{align}
        T=\sigma\,K_1\cdots K_n\,Z\,B_1\cdots B_n,
        \label{eq:peps_product_term_blocks}
    \end{align}
    where $B_j$ is the block $(\ket0\bra{w_j})_{O^{\rm in}_j}\otimes\id$ on the open input legs $O^{\rm in}_j$ of the $j$-th bra, $K_j$ is $(\ket{v_j}\bra0)_{O^{\rm out}_j}\otimes\id$ on the open output legs $O^{\rm out}_j$ of the $j$-th ket, $Z$ is the product of the one-site zero
    blocks on $W$, and $\sigma$ is the product of the scalar values at vertices without open legs.
\end{lemma}

\begin{proof}\leanok
    Applying Lemma~\ref{lem:peps_rank_one_block_products} to $B_1\cdots B_n$, to $Z$, and to $K_1\cdots K_n$ shows via~(\ref{eq:peps_rank_one_block_prod}) that the matrix entry $\braket{c'}{\sigma\,K_1\cdots K_n\,Z\,B_1\cdots B_n|c}$ has a single contributing intermediate configuration, which equals $\braket{c'}{T|c}$.
\end{proof}

    $\delta_{\tau_{\rm aff},b}\ket{\tau_{\rm unt},a}\otimes Y\ket c$.
\end{proof}

\begin{theorem}[Small-patch rewrite: product terms as allowed monomials]
    \label{thm:peps_frame_small_rewrite_term_monomial}
    \lean{TNLean.PEPS.EncodedFrame.exists_of_mem_encSteps,
        TNLean.PEPS.EncodedFrame.exists_of_mem_decSteps,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.owners_mem_parties_of_mem_branchSteps,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.owners_mem_parties_of_not_untouched,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.ownerOld_eq_ownerNew_of_untouched,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.card_open_vertices_le,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.exists_termChain}
    \leanok
    \uses{lem:peps_frame_branch_network,lem:peps_frame_wires,
        lem:peps_rank_one_block_monomial,lem:peps_product_term_blocks,
        lem:peps_frame_tag_change}
    Assume conditions~(1)--(4) of Definition~\ref{def:peps_frame_rewrite_data}, fix a branch $\beta=(a,s,b)$, and
    choose a unit vector on every open-leg group of its network. Then the
    placed product term $\ket a\bra b\otimes T\otimes\id_{\rm unt}$
    equals $\sigma M_{\beta,T}$ with $\lvert\sigma\rvert\le1$, where $M_{\beta,T}$ is the operator of an allowed monomial from the registers
    of $\mathcal F_{\rm old}$ to those of $\mathcal F_{\rm new}$ using only the specified parties and
    having at most $2m$ normalized pair sources and at most $2m$ normalized pair
    effects (\texttt{05-frames.tex}, lines 262--273 and 306--316).
\end{theorem}

\begin{proof}\leanok
    Factor $T=\sigma\,K_1\cdots K_n\,Z\,B_1\cdots B_n$ by
    Lemma~\ref{lem:peps_product_term_blocks}. By Lemma~\ref{lem:peps_frame_branch_network}, only the at most $2m$ patch vertices have open legs, each $O^{\rm in}_j$ has at most two old owners in the specified list, and each $O^{\rm out}_j$ has at most two new owners in the specified list; by Lemma~\ref{lem:peps_frame_wires}, all old and new owners on $W$ lie in the specified list and owners agree outside $W$. Applying
    Lemma~\ref{lem:peps_rank_one_block_monomial} to $B_1\cdots B_n$, $Z$, and $K_1\cdots K_n$, and composing with the tag monomial of
    Lemma~\ref{lem:peps_frame_tag_change}, yields $M_{\beta,T}$.
\end{proof}

\begin{theorem}[Small-patch rewrite: expansion into allowed monomials]
    \label{thm:peps_frame_small_rewrite_monomials}
    \lean{TNLean.Lattice.Sheet.act_smul,
        TNLean.Lattice.Sheet.act_sum,
        TNLean.PEPS.EncodedFrame.truncationRank_mono,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.exists_allowed_monomial_approx,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.exists_allowed_monomial_approx_of_card_le}
    \leanok
    \uses{def:peps_party_monomial,thm:peps_frame_small_rewrite_product_terms,
        thm:peps_frame_small_rewrite_term_monomial,lem:peps_frame_branches}
    Assume conditions~(1)--(4) of Definition~\ref{def:peps_frame_rewrite_data}, let $\delta>0$, let $N$ be the
    number of branches, and set $k=\lceil(4Nm/\delta)^2\rceil+1$. There exist a
    contraction $M_\delta$ with $\lVert M_\delta-M\rVert\le\delta$ and an
    expansion $M_\delta=\sum_tc_tM_t$ into at most $Nk^{2m}$ allowed monomials
    $M_t$ from the registers of $\mathcal F_{\rm old}$ to those of
    $\mathcal F_{\rm new}$ (with untouched tags ordered first), using only the specified parties, each with at
    most $2m$ normalized pair sources, at most $2m$ normalized pair
    effects, and $\lvert c_t\rvert\le1$. If every affected hole and
    additional patch has at most $D$ cylinder terms, the number of monomials
    is at most $D^Ek'^{2m}$ with $E=r_{\rm new}+m+r_{\rm old}$ and
    $k'=\lceil(4D^Em/\delta)^2\rceil+1$ (\cite[Lemma~6.3]{OpenAI2026PolynomialPEPS}, \texttt{05-frames.tex},
    lines 214--217; see Theorem~\ref{thm:peps_frame_small_rewrite_polynomial} for the polynomial bound when $D\le CL^c$).
\end{theorem}

\begin{proof}\leanok
    Substituting $\ket a\bra b\otimes T_{\beta,i}\otimes\id_{\rm unt}=\sigma_{\beta,i}M_{\beta,i}$ from
    Theorem~\ref{thm:peps_frame_small_rewrite_term_monomial} into $M_\delta=(1+\tfrac12\delta)^{-1}\sum_\beta\ket a\bra b\otimes
        \bigl(\sum_ic_{\beta,i}T_{\beta,i}\bigr)\otimes\id_{\rm unt}$ from
    Theorem~\ref{thm:peps_frame_small_rewrite_product_terms} gives the expansion with coefficients
    $c_t=(1+\tfrac12\delta)^{-1}c_{\beta,i}\sigma_{\beta,i}$ satisfying $\lvert c_t\rvert\le1$; Lemma~\ref{lem:peps_frame_branches} gives $N\le D^E$ and $k\le k'$.
\end{proof}
