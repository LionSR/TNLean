\begin{lemma}[Additivity of the index on an MPU representation]
    \label{thm:mpug_group_index_additivity}
    \lean{MPOTensor.GroupFamily.IsRawRepresentation.index_mul,
        MPOTensor.GroupFamily.IsRawRepresentation.index_one,
        MPOTensor.GroupFamily.IsRawRepresentation.neZero_phys}
    \leanok
    \uses{def:mpug_group_representation, def:mpu_reduced_representative_index}
    Let $\{\mathcal U_g\}_{g\in G}$ be an injective family of MPUs with positive
    bond dimensions, satisfying~(\ref{eq:mpug_rep_identity}) and
    (\ref{eq:mpug_rep_multiplication}) at every positive chain length, without
    assuming simplicity. As in the finite-order argument of
    \cite[line~1547]{FrancoRubio2025MPUGauging}, the physical dimension satisfies
    $d>0$ and the index obeys
    \begin{align}
        \operatorname{ind}(\mathcal U_e) & =0,                                                                 \\
        \operatorname{ind}(\mathcal U_{gh})
                                         & =\operatorname{ind}(\mathcal U_g)+\operatorname{ind}(\mathcal U_h).
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:mpu_reduced_representative_index_invariance,
        thm:mpu_representative_index_tensor_product,
        thm:mpu_identity_index, thm:mpu_admissible_index_composition}
    Injectivity on a positive-dimensional bond space excludes an empty
    physical alphabet, so $d>0$. On the remaining periodic operators, the
    identity law identifies $\mathcal U_e$ with the identity tensor of zero
    index, while the multiplication law identifies $\mathcal U_{gh}$ with the
    composition of $\mathcal U_g$ and $\mathcal U_h$. Since the index is
    invariant under equality of periodic operators and additive under
    composition, these two identifications yield
    $\operatorname{ind}(\mathcal U_e)=0$ and
    $\operatorname{ind}(\mathcal U_{gh})=\operatorname{ind}(\mathcal U_g)+\operatorname{ind}(\mathcal U_h)$.
\end{proof}

\begin{lemma}[Vanishing of the index for a finite-group representation]
    \label{thm:mpug_finite_group_index_zero}
    \lean{MPOTensor.GroupFamily.IsRawRepresentation.index_eq_zero}
    \leanok
    \uses{thm:mpug_group_index_additivity}
    Under the hypotheses of Lemma~\ref{thm:mpug_group_index_additivity},
    if $G$ is finite then, as in \cite[line~1547]{FrancoRubio2025MPUGauging},
    every $g\in G$ satisfies $\operatorname{ind}(\mathcal U_g)=0$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:mpug_group_index_additivity}
    Additivity under multiplication gives
    $\operatorname{ind}(\mathcal U_{g^n})=n\operatorname{ind}(\mathcal U_g)$
    by induction on $n$. Evaluating this relation at the finite order
    $n=\operatorname{ord}(g)>0$, where $g^n=e$, yields
    $0=n\operatorname{ind}(\mathcal U_g)$ and hence
    $\operatorname{ind}(\mathcal U_g)=0$.
\end{proof}

\begin{lemma}[Source-leg dimensions in a finite-group representation]
    \label{thm:mpug_finite_group_source_ranks}
    \lean{MPOTensor.GroupFamily.IsRepresentation.sourceRanks_eq_physDim}
    \leanok
    \uses{def:mpug_group_representation, def:mpu_right_rank, def:mpu_left_rank}
    Let $G$ be finite and let $\{\mathcal U_g\}_{g\in G}$ be an exact
    representation by simple injective MPU tensors of physical dimension
    $d$. As in \cite[line~1547]{FrancoRubio2025MPUGauging}, every $g\in G$ has
    source-leg dimensions
    \begin{align}
        d_r(g) & =d, \\
        d_l(g) & =d.
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:mpug_finite_group_index_zero,
        thm:mpu_simple_injective_source_index,
        thm:mpug_zero_specified_source_index_ranks}
    For $r=d_r(g)$ and $\ell=d_l(g)$, simplicity and injectivity imply
    $r\ell=d^2$ and
    $\operatorname{ind}(\mathcal U_g)=\frac12(\log_2 r-\log_2\ell)$, with
    $r,\ell>0$ because $d>0$. Combining $r\ell=d^2$ with the finite-order
    vanishing $\operatorname{ind}(\mathcal U_g)=0$ rewrites this index as
    $\log_2(r/d)=0$, which forces $r=d$ and hence $\ell=d$.
\end{proof}
