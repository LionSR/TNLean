\subsection{Comparison at the first endpoint}

\begin{theorem}[Endpoint projector and local interaction comparison]
    \label{thm:mpo_mixed_endpoint_projector_comparison}
    \lean{MPSTensor.MPOSymmetry.mixedEndpointRowSector_isSymmetricProjection,
        MPSTensor.MPOSymmetry.mixedEndpointColumnSector_isSymmetricProjection,
        MPSTensor.MPOSymmetry.mixedEndpointRowSector_commute_columnSector,
        MPSTensor.MPOSymmetry.mixedEndpointLeftTensor_starProjection_eq_outerCorner,
        MPSTensor.MPOSymmetry.mixedEndpointParentInteraction_isPositive,
        MPSTensor.MPOSymmetry.mixedEndpointParentInteraction_zero_le_parentInteractionES,
        MPSTensor.MPOSymmetry.parentInteractionES_mixedEndpointLeftTensor_le,
        MPSTensor.MPOSymmetry.mixedEndpoint_one_sub_columnSector_zero_le_parentInteraction,
        MPSTensor.MPOSymmetry.mixedEndpoint_one_sub_rowSector_one_le_parentInteraction}
    \leanok
    \uses{thm:mpo_mixed_corner_support,thm:mpo_mixed_reducing_sectors,
        thm:mpo_mixed_inner_sectors}
    Let $P'=\Pi_{\mathcal S'_0}$, $P=\Pi_{G_2(B_0)}$, and
    $h'=I-P'$, $h=I-P$. Then
    \begin{align}
        P     & =R_0C_1P',
              & 0\leq h'   & \leq h\leq h'+(I-R_0)+(I-C_1),
        \label{eq:mpo_mixed_endpoint_local_comparison}      \\
        I-C_0 & \leq h',
              & I-R_1      & \leq h'.
        \label{eq:mpo_mixed_endpoint_inner_penalties}
    \end{align}
    These comparisons hold for arbitrary endpoint tensors; injectivity
    is unnecessary for the local statements.
\end{theorem}
\begin{proof}
    \leanok
    The sector selectors are commuting orthogonal projections. The outer
    selectors also commute with $P'$. Their product $R_0C_1P'$ is
    therefore an orthogonal projection whose range is $G_2(B_0)$, giving
    the first identity. Inclusion $G_2(B_0)\subseteq\mathcal S'_0$
    yields $h'\leq h$.
    For commuting orthogonal projections $Q,T$,
    \begin{align}
        (I-Q)+(I-T)-(I-QT)=(I-Q)(I-T)\geq0.
    \end{align}
    Applying this twice yields the upper comparison.
    Finally, $\mathcal S'_0\subseteq\operatorname{range}C_0$ and
    $\mathcal S'_0\subseteq\operatorname{range}R_1$ give
    (\ref{eq:mpo_mixed_endpoint_inner_penalties}) by taking
    complementary projections.
\end{proof}

\begin{definition}[Exchanging endpoint sectors]
    \label{def:mpo_mixed_endpoint_swap}
    \lean{MPSTensor.MPOSymmetry.mixedEndpointBondSwap,
        MPSTensor.MPOSymmetry.mixedEndpointPhysicalSwap,
        MPSTensor.MPOSymmetry.mixedEndpointRightTensor,
        MPSTensor.physicalReindexLinearIsometryEquiv,
        MPSTensor.physicalReindexLinearIsometryEquiv_apply_apply}
    \uses{def:mpo_mixed_endpoint_tensor,def:mpo_mixed_first_endpoint}
    Exchange the summands of $\C^{D_0}\oplus\C^{D_1}$, and exchange
    them in each physical register. Let $U_N$ be the resulting
    coefficient-space isometry from the reversed pair to the original
    pair on $N$ sites. Let $B_1$ equal $A_1$ on $I_1\times I_1$ and
    vanish on the other physical sectors.
\end{definition}

\begin{theorem}[Second endpoint by sector exchange]
    \label{thm:mpo_mixed_second_endpoint_comparison}
    \lean{MPSTensor.MPOSymmetry.mixedEndpointRightTensor_apply,
        MPSTensor.MPOSymmetry.isInjective_mixedEndpointRightTensor,
        MPSTensor.MPOSymmetry.mixedEndpointBase_swap,
        MPSTensor.MPOSymmetry.bondInterpolationMatrix_one_swap,
        MPSTensor.MPOSymmetry.insertedTwoSiteMap_swap_zero,
        MPSTensor.MPOSymmetry.mixedEndpoint_extendedSupport_one_eq_map_swap,
        MPSTensor.MPOSymmetry.mixedEndpointParentInteraction_one_eq_conj_swap,
        MPSTensor.MPOSymmetry.mixedEndpointRowSector_conj_swap,
        MPSTensor.MPOSymmetry.mixedEndpointColumnSector_conj_swap,
        MPSTensor.MPOSymmetry.parentInteractionES_mixedEndpointRightTensor_eq_conj,
        MPSTensor.MPOSymmetry.mixedEndpointParentInteraction_one_le_parentInteractionES,
        MPSTensor.MPOSymmetry.parentInteractionES_mixedEndpointRightTensor_le,
        MPSTensor.MPOSymmetry.mixedEndpoint_columnSector_zero_le_parentInteraction_one,
        MPSTensor.MPOSymmetry.mixedEndpoint_rowSector_one_le_parentInteraction_one,
        MPSTensor.physicalReindexLinearIsometryEquiv_conj_diagonal,
        MPSTensor.physicalReindexLinearIsometryEquiv_groundSpaceMapES,
        MPSTensor.groundSpaceES_reindexPhysical_equiv,
        MPSTensor.parentInteractionES_reindexPhysical_equiv}
    \uses{def:mpo_mixed_endpoint_swap,thm:mpo_mixed_endpoint_projector_comparison}
    Put $h'_1=I-\Pi_{\mathcal S'_1}$ and $h_1=I-\Pi_{G_2(B_1)}$.
    The second endpoint support and interaction are the images of the
    first endpoint support and interaction of the reversed pair under
    $U_2$. Moreover, $B_1$ is injective if $A_1$ is injective, and
    \begin{align}
        h'_1 & \leq h_1\leq h'_1+R_0+C_1,
             & C_0                        & \leq h'_1,
             & R_1                        & \leq h'_1.
        \label{eq:mpo_mixed_second_endpoint_comparison}
    \end{align}
\end{theorem}
\begin{proof}
    If $\Phi$ denotes the simultaneous virtual basis exchange, each
    mixed letter of the reversed pair is $\Phi(A^{ij})$, and its
    first-endpoint insertion is $\Phi(W(1))$. Hence preservation of
    products and trace gives
    \begin{align}
        U_2\Gamma_{\mathrm{rev},0}(\Phi(X))=\Gamma_{A,W(1)}(X).
    \end{align}
    The boundary exchange is bijective, so the supports are isometric
    images. Isometries transport orthogonal projections, while physical
    relabeling transports the canonical boundary spaces of $B_1$.
    Each first-sector selector for the reversed pair becomes the
    complementary selector for the original pair. Conjugate
    (\ref{eq:mpo_mixed_endpoint_local_comparison}) and
    (\ref{eq:mpo_mixed_endpoint_inner_penalties}) to obtain
    (\ref{eq:mpo_mixed_second_endpoint_comparison}).
\end{proof}

\begin{theorem}[Periodic endpoint comparison, ground lines, and gaps]
    \label{thm:mpo_mixed_periodic_endpoint_gaps}
    \lean{MPSTensor.periodicLocalInteractionES_add,
        MPSTensor.periodicLocalInteractionES_sub,
        MPSTensor.periodicLocalInteractionES_one,
        MPSTensor.periodicLocalInteractionES_diagonal_apply,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_mixedEndpointRowSector_previous,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_mixedEndpointColumnSector_next,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_outerRowPenalty_le_previous,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_outerColumnPenalty_le_next,
        MPSTensor.MPOSymmetry.localTermES_mixedEndpointLeftTensor_le_three_extended,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_comparison,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_ker_eq,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_norm_gap,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_groundSpace_eq,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_groundSpace_finrank,
        MPSTensor.MPOSymmetry.exists_uniform_mixedEndpoint_periodic_gap,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_outerRowPenalty_one_le_previous,
        MPSTensor.MPOSymmetry.periodicLocalInteractionES_outerColumnPenalty_one_le_next,
        MPSTensor.MPOSymmetry.localTermES_mixedEndpointRightTensor_le_three_extended,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_one_comparison,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_one_ker_eq,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_one_norm_gap,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_one_groundSpace_eq,
        MPSTensor.MPOSymmetry.mixedEndpoint_periodic_one_groundSpace_finrank,
        MPSTensor.MPOSymmetry.exists_uniform_mixedEndpoint_periodic_one_gap,
        MPSTensor.MPOSymmetry.exists_uniform_mixedEndpoint_periodic_endpoints_gap}
    \uses{thm:mpo_mixed_endpoint_projector_comparison,
        thm:mpo_mixed_second_endpoint_comparison}
    Let $H'_{p,N}$ be the periodic sum of the actual extended interaction
    at $p\in\{0,1\}$, and let $K_{p,N}$ be the canonical nearest-neighbor
    parent Hamiltonian of the embedded tensor $B_p$. For every $N\geq2$,
    \begin{align}
        H'_{p,N}\leq K_{p,N}\leq3H'_{p,N},
        \qquad \ker H'_{p,N}=\ker K_{p,N}.
    \end{align}
    If $A_p$ is injective and $D_p>0$, this kernel is exactly the
    one-dimensional span of its embedded periodic MPS.
    A gap $\delta_p$ of $K_{p,N}$ transfers to a gap $\delta_p/3$ of
    $H'_{p,N}$. Consequently, if both endpoint tensors are injective,
    there is $\delta>0$ such that both endpoint Hamiltonians have gap at
    least $\delta$ on every periodic chain of length at least two.
    Only the two endpoint parameters are quantified in this conclusion.
\end{theorem}
\begin{proof}
    A missing outer row sector on one edge is an inner row sector of
    the previous edge; a missing outer column sector is an inner column
    sector of the next edge. Cyclic extension therefore gives
    \begin{align}
        h'_{p,i}\leq h_{p,i}
        \leq h'_{p,i-1}+h'_{p,i}+h'_{p,i+1}.
    \end{align}
    Summing counts each extended term three times, also when $N=2$.
    Positivity and the two inequalities give equal kernels. On their
    common orthogonal complement, the positive spectral comparison
    transfers the canonical gap with factor $1/3$.
    Each embedded endpoint is one-site injective, so its canonical
    nearest-neighbor parents have a unique periodic ground line and
    a positive gap independent of the chain length. Taking the smaller
    of the two transferred bounds gives the common endpoint bound.
\end{proof}
