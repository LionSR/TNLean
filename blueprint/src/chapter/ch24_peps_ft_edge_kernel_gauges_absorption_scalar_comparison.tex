\begin{lemma}[Absorbing edge gauges into the second PEPS]%
    \label{thm:peps_edge_gauge_absorption}
    \lean{TNLean.PEPS.edge_gauge_absorption}
    \leanok
    \uses{def:peps_absorbEdgeGauges, thm:peps_absorbEdgeGauges_bondDim, thm:peps_absorbEdgeGauges_component}

    $D_{\widetilde B}=D_B$ and, for every vertex $v$,
    \begin{align}
        \widetilde B_v(\eta,\sigma)

        \sum_{\eta'}
        \left(\prod_{ie}M_{ie}(\eta(ie),\eta'(ie))\right)
        B_v(\eta',\sigma),
        \label{eq:peps_edge_gauge_absorption_comp}
    \end{align}
    where $M_{ie}=X_e$ or $M_{ie}=(X_e^{-1})^t$ according to the orientation of
    $ie$.
    % Formula: the preceding component identity is gauge absorption at v.
    % Ink-to-index/type verdict: paper_normal.tex:1212--1314 gives only
    % vertex-specific pictures, each with boundary (V,P)=(deg(v),1); the old
    % degree-four star cannot represent the theorem on an arbitrary graph.
\end{lemma}
\begin{proof}\leanok
    Combine Lemma~\ref{thm:peps_absorbEdgeGauges_bondDim} ($D_{\widetilde B}=D_B$)
    and Lemma~\ref{thm:peps_absorbEdgeGauges_component}
    ((\ref{eq:peps_edge_gauge_absorption_comp})).
\end{proof}

\begin{definition}[Post-absorption inserted-edge comparison]%

    \lean{TNLean.PEPS.PostAbsorptionEdgeInsertionEquality}
    \leanok
    \uses{def:peps_edgeInsertedCoeff}

    when $h:D_A=D_{\widetilde B}$ and, writing
    $\phi_{h_e}:\MN{D_A(e)}\to\MN{D_{\widetilde B}(e)}$ for the isomorphism
    induced by $h_e:D_A(e)=D_{\widetilde B}(e)$,

        C_A(e,X;\sigma) = C_{\widetilde B}(e,\phi_{h_e}(X);\sigma)
        \qquad (\forall\,e,\ X\in\MN{D_A(e)},\ \sigma).

\end{definition}

\begin{lemma}[Gauge consistency]%
    \label{thm:gaugeConsistency}
    \lean{TNLean.PEPS.gaugeConsistency}
    \leanok
    \uses{thm:peps_exists_edgeGaugeFamily, thm:peps_postAbsorptionEdgeInsertionEquality, thm:localGauge_exists, def:gaugeVertex}
    Let $A$ and $B$ be vertex-injective PEPS tensors on a connected graph with
    $\Coeff_A=\Coeff_B$, $D_A=D_B$, and $D_A(e)>0$ for all $e$ (connectivity is
    necessary, as disconnected components can carry reciprocal non-unit scalars).
    Then there exists an edge gauge family $X=(X_e)_e$ such that
    % Formula: one global family X supplies the local gauge equation at every v.
    % Ink-to-index/type verdict: the local component has boundary
    % (V,P)=(deg(v),1), while a two-vertex comparison has
    % (deg(u)+deg(w)-2,2).  The source gives the vertexwise equations in
    % paper_normal.tex:1212--1314, not the leg-suppressed scalar legacy picture.
    \begin{align}
        B_v(\eta,\sigma)
         & =
        (A^X)_v(\eta,\sigma)
        =
        \sum_{\eta'}
        \left(\prod_{e\ni v}X_{v,e}(\eta_e,\eta'_e)\right)
        A_v(\eta',\sigma)
        \qquad (\forall\,v,\eta,\sigma),
        \label{eq:peps_edge_gauge_action}
    \end{align}
    establishing the edge-gauge construction of
    \cite[Section~3]{Molnar2018NormalPEPS}.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_postAbsorptionEdgeInsertionEquality}, $\widetilde B=B^Z$
    satisfies (\ref{eq:peps_post_absorption_insertion}). Comparing each vertex
    $v$ with its complement gives $\widetilde B_v=\lambda_v A_v$ ($\lambda_v\in\C^\times$)
    with $\prod_v\lambda_v=1$; on the connected graph, a spanning-tree edge-scalar
    factorization writes $\lambda_v^{-1}=\prod_{e\ni v}c_{v,e}$, and combining
    $c_e$ with $Z_e^{-1}$ yields $X_e$ satisfying (\ref{eq:peps_edge_gauge_action}).
\end{proof}

\begin{remark}
    In (\ref{eq:peps_edge_gauge_action}), on each ordered edge $e=(u,w)$ the
    endpoint actions $X_e$ at $u$ and $(X_e^{-1})^\top$ at $w$ cancel upon
    contraction \cite[Section~3]{Molnar2018NormalPEPS}.
\end{remark}

\begin{definition}[Left-identity three-leg residual form]%
    \label{def:peps_threeLegLeftIdentityForm}
    \lean{TNLean.PEPS.threeLegLeftIdentityForm}
    \leanok
    For virtual leg spaces $\alpha,\beta,\gamma$ and $Z\in\End(\beta\otimes\gamma)$,
    the left-identity residual form on $\alpha\otimes\beta\otimes\gamma$ is
    $\Id_\alpha\otimes Z$, with entries $\delta_{aa'}Z_{bc,b'c'}$
    \cite[Section~3]{Molnar2018NormalPEPS}.
\end{definition}

\begin{definition}[Right-identity three-leg residual form]%
    \label{def:peps_threeLegRightIdentityForm}
    \lean{TNLean.PEPS.threeLegRightIdentityForm}
    \leanok
    For $U\in\End(\alpha\otimes\beta)$, the right-identity residual form on
    $\alpha\otimes\beta\otimes\gamma$ is $U\otimes\Id_\gamma$, with entries
    $U_{ab,a'b'}\delta_{cc'}$ \cite[Section~3]{Molnar2018NormalPEPS}.
\end{definition}

\begin{definition}[Middle-identity three-leg residual form]%
    \label{def:peps_threeLegMiddleIdentityForm}
    \lean{TNLean.PEPS.threeLegMiddleIdentityForm}
    \leanok
    For $W\in\End(\alpha\otimes\gamma)$, the middle-identity residual form on
    $\alpha\otimes\beta\otimes\gamma$ has entries $\delta_{bb'}W_{ac,a'c'}$
    for $(a,b,c),(a',b',c')\in\alpha\times\beta\times\gamma$
    \cite[Section~3]{Molnar2018NormalPEPS}.
\end{definition}

\begin{lemma}[Scalar reduction of the residual two-tensor gauges]%
    \label{thm:peps_twoInjectiveGaugeScalarReduction}
    \lean{TNLean.PEPS.threeLeg_residual_forms_scalar}
    \leanok
    \uses{def:peps_threeLegLeftIdentityForm, def:peps_threeLegRightIdentityForm, def:peps_threeLegMiddleIdentityForm}
    For nonzero virtual leg spaces $\alpha,\beta,\gamma$, let
    $Z\in\End(\beta\otimes\gamma)$, $U\in\End(\alpha\otimes\beta)$, and
    $W\in\End(\alpha\otimes\gamma)$ have equal induced three-leg residual forms
    (Definitions~\ref{def:peps_threeLegLeftIdentityForm}--\ref{def:peps_threeLegMiddleIdentityForm})
    on $\alpha\otimes\beta\otimes\gamma$:
    \begin{align}
        \delta_{aa'}Z_{bc,b'c'} = U_{ab,a'b'}\delta_{cc'} = W_{ac,a'c'}\delta_{bb'}
        \qquad (\forall\,a,a'\in\alpha,\ b,b'\in\beta,\ c,c'\in\gamma).
        \label{eq:peps_three_leg_compat}
    \end{align}
    Then there exists $\lambda\in\C$ such that
    \begin{align}
        Z = \lambda\,\Id_{\beta\otimes\gamma}, \qquad
        U = \lambda\,\Id_{\alpha\otimes\beta}, \qquad
        W = \lambda\,\Id_{\alpha\otimes\gamma}. \notag
    \end{align}
    \textbf{Local fix (nonzero virtual legs):}
    Requires $\alpha,\beta,\gamma\neq\varnothing$ (if $\alpha=\varnothing$,
    $|\beta|=1$, $|\gamma|=2$, then $Z=\diag(1,2)$ satisfies
    (\ref{eq:peps_three_leg_compat}) vacuously), modeling the nonzero virtual
    bond spaces of \cite[Section~3]{Molnar2018NormalPEPS}
    (\cite{gap:peps_injective_ft_section3_route}).
\end{lemma}
\begin{proof}\leanok
    Taking $a=a'$ in (\ref{eq:peps_three_leg_compat}) gives $Z_{bc,b'c'}=0$ for
    $c\neq c'$ (from the first equality) and for $b\neq b'$ (from the second),
    while $Z_{bc,bc}=U_{ab,ab}=W_{ac,ac}$ is independent of $c$ and $b$. Thus
    $Z=\lambda\Id_{\beta\otimes\gamma}$, and (\ref{eq:peps_three_leg_compat})
    yields $U=\lambda\Id_{\alpha\otimes\beta}$ and
    $W=\lambda\Id_{\alpha\otimes\gamma}$.
\end{proof}

\begin{lemma}[Many-leg scalar factorization]%
    \label{thm:peps_piProductFormsScalar}
    \lean{TNLean.PEPS.piProduct_forms_scalar}
    \leanok
    For a finite family of nonempty index sets with invertible matrices
    $M_i,N_i$ on each leg $i$, if
    $\prod_i(M_i)_{p_iq_i}=\prod_i(N_i)_{p_iq_i}$ for all $(p,q)$, then there
    exist $c_i\in\C^\times$ such that $M_i=c_iN_i$ for all $i$ and
    $\prod_i c_i=1$.
\end{lemma}
\begin{proof}\leanok
    Since each $N_i$ is invertible, choose $(p^0,q^0)$ with
    $(N_i)_{p^0_iq^0_i}\neq 0$ for all $i$; then $(M_i)_{p^0_iq^0_i}\neq 0$ as
    well. Varying $(p_i,q_i)=(a,b)$ while keeping $p_j=p^0_j$, $q_j=q^0_j$ for
    $j\neq i$ gives
    $(M_i)_{ab}\prod_{j\neq i}(M_j)_{p^0_jq^0_j}=(N_i)_{ab}\prod_{j\neq i}(N_j)_{p^0_jq^0_j}$,
    so $M_i=c_i N_i$ with $c_i=(M_i)_{p^0_iq^0_i}/(N_i)_{p^0_iq^0_i}\in\C^\times$
    and $\prod_i c_i=1$.
\end{proof}

\begin{lemma}[Pointwise product cancellation for two injective blocks]%
    \label{thm:peps_twoBlockPointwiseProductCancellation}
    \lean{TNLean.PEPS.twoBlockReciprocalScalarProportional_of_pointwise_mul_eq}
    \leanok
    Let $A_1,A_2,B_1,B_2$ be tensors with nonempty external and shared-boundary
    configuration spaces, with $A_1$ and $A_2$ injective. If
    \begin{align}
        A_1(\eta_1,\mu,\sigma_1)A_2(\eta_2,\nu,\sigma_2)
         & =
        B_1(\eta_1,\mu,\sigma_1)B_2(\eta_2,\nu,\sigma_2)
        \qquad (\forall\,\eta_1,\eta_2,\mu,\nu,\sigma_1,\sigma_2),
        \label{eq:peps_two_block_pointwise_prod}
    \end{align}
    then there exists $\lambda\in\C^\times$ such that
    $A_1=\lambda B_1$ and $A_2=\lambda^{-1}B_2$.
\end{lemma}
% This theorem is the rank-one cancellation step once
% \cite[Section~3]{Molnar2018NormalPEPS} has been reduced to pointwise product
% equality of separated coefficients.
\begin{proof}\leanok
    By injectivity of $A_1,A_2$ and (\ref{eq:peps_two_block_pointwise_prod}),
    choose $(\eta_k^0,\mu^0,\sigma_k^0)$ with
    $A_k(\eta_k^0,\mu^0,\sigma_k^0)\neq 0$ and
    $B_k(\eta_k^0,\mu^0,\sigma_k^0)\neq 0$ ($k=1,2$). Setting
    $\lambda=B_2(\eta_2^0,\mu^0,\sigma_2^0)/A_2(\eta_2^0,\mu^0,\sigma_2^0)\in\C^\times$,
    evaluating (\ref{eq:peps_two_block_pointwise_prod}) at
    $(\eta_2,\nu,\sigma_2)=(\eta_2^0,\mu^0,\sigma_2^0)$ gives $A_1=\lambda B_1$,
    and evaluating at $(\eta_1,\mu,\sigma_1)=(\eta_1^0,\mu^0,\sigma_1^0)$ gives
    $A_2=\lambda^{-1}B_2$.
\end{proof}

\begin{lemma}[Coefficient selection for one shared bond]%
    \label{thm:peps_twoBlockInsertedCoeff_singletonBond}
    \lean{TNLean.PEPS.twoBlockInsertedCoeff_singletonBond_single}
    \leanok

    \begin{align}
        C_{A_1,A_2}
        (b,E_{\mu,\nu};\eta_1,\eta_2,\sigma_1,\sigma_2)
         & =
        A_1(\eta_1,\mu,\sigma_1)A_2(\eta_2,\nu,\sigma_2).

    \end{align}
\end{lemma}

    \begin{align}
        C_{A_1,A_2}(b,E_{\mu,\nu};\eta_1,\eta_2,\sigma_1,\sigma_2)

        \sum_{\mu',\nu'}
        \delta_{\mu(b),\mu'(b)}\delta_{\nu(b),\nu'(b)}
        A_1(\eta_1,\mu',\sigma_1)A_2(\eta_2,\nu',\sigma_2)
        \notag \\
         & =
        A_1(\eta_1,\mu,\sigma_1)A_2(\eta_2,\nu,\sigma_2).
        \notag
    \end{align}
\end{proof}

            \tn[at=0.45 s of midway twoInjBOne and twoInjBTwo,
                name=twoInjBLow, skin=none]{}
            \tnwire{twoInjBOne.20}{twoInjBX.200}
            \tnwire{twoInjBX.-20}{twoInjBTwo.160}
            \tnwire{twoInjBOne.0}{twoInjBTwo.180}
            \tnwire{twoInjBOne.-20}{twoInjBLow}
            \tnwire{twoInjBLow}{twoInjBTwo.200}
        \end{tenkz}
    \end{center}
    Then there exists $\lambda\in\C^\times$ such that $A_1=\lambda B_1$ and
    $A_2=\lambda^{-1}B_2$ \cite[Section~3]{Molnar2018NormalPEPS}.
\end{lemma}
\begin{proof}
    \leanok
    Setting $X=\Id_{D_b}$ in (\ref{eq:peps_two_block_inserted_eq}) and applying
    bipartite Schmidt uniqueness \cite[Section~3]{Molnar2018NormalPEPS} to the
    linearly independent families $A_1,A_2,B_1,B_2$ gives an invertible matrix
    $g$ on shared-bond configurations with
    \begin{align}
        A_1(\eta_1,\mu,\sigma_1)
                                                      & =
        \sum_\nu g_{\mu,\nu}B_1(\eta_1,\nu,\sigma_1), &
        B_2(\eta_2,\nu,\sigma_2)
                                                      & =
        \sum_\mu g_{\mu,\nu}A_2(\eta_2,\mu,\sigma_2).
        \label{eq:peps_two_block_schmidt_gauge}
    \end{align}
    Substituting (\ref{eq:peps_two_block_schmidt_gauge}) into
    (\ref{eq:peps_two_block_inserted_eq}) with $X=E_{p,q}$ on bond $b$ and using
    linear independence of $B_1$ and $A_2$ gives
    \begin{align}
        [\rho_b=q]\,g_{\rho^{b\to p},\nu}
         & =
        [\nu_b=p]\,g_{\rho,\nu^{b\to q}}
        \qquad (\forall\,b,p,q,\nu,\rho),
        \label{eq:peps_residual_equality}
    \end{align}
    where $\rho^{b\to p}$ replaces $\rho_b$ by $p$
    (cf.\ Lemma~\ref{thm:peps_twoInjectiveGaugeScalarReduction}). Evaluating
    (\ref{eq:peps_residual_equality}) shows $g_{\rho,\nu}=0$ for $\rho\neq\nu$
    and $g_{\rho^{b\to p},\rho^{b\to p}}=g_{\rho,\rho}$, so $g=\lambda\Id$ with
    $\lambda\in\C^\times$, and (\ref{eq:peps_two_block_schmidt_gauge}) yields
    $A_1=\lambda B_1$ and $A_2=\lambda^{-1}B_2$.
\end{proof}
