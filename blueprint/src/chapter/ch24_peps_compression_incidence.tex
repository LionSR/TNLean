\section{Finite incidence in distributed PEPS compression}
\label{sec:peps_compression_incidence}
The finite counting statements below are those used in the proof of
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{04-compression.tex}, lines 342--381 and 565--588.
% These entries formalize finite prerequisites only. Full circuit expansion,
% tensor evaluation, virtual dimensions and quantum/sampling composition remain open.

\begin{definition}[Source-position endpoints]
    \label{def:compression_source_endpoints}
    \lean{TNLean.Lattice.Compression.sourceEndpoints}
    \lean{TNLean.Lattice.Compression.mem_sourceEndpoints}
    \leanok
    For source positions $e$ with ordered party pairs $(a_e,b_e)$, set $E_e=\{a_e,b_e\}$.
\end{definition}

\begin{definition}[Corrected parties]
    \label{def:compression_corrected_parties}
    \lean{TNLean.Lattice.Compression.correctedParties}
    \lean{TNLean.Lattice.Compression.mem_correctedParties}
    \leanok
    For a finite set $S$ of source positions, set $A_S=\bigcup_{e\in S}E_e$.
\end{definition}

\begin{definition}[Lifetime gate incidence]
    \label{def:compression_lifetime_gates}
    \lean{TNLean.Lattice.Compression.incidentGates}
    \lean{TNLean.Lattice.Compression.mem_incidentGates}
    \leanok
    For a finite set $G$ of nonprivate gate occurrences $g$ with finite participant sets $P_g$, set $G(p)=\{g\in G:p\in P_g\}$.
\end{definition}

\begin{definition}[Gates touching a party set]
    \label{def:compression_touching_gates}
    \lean{TNLean.Lattice.Compression.gatesTouching}
    \lean{TNLean.Lattice.Compression.mem_gatesTouching}
    \leanok
    For a finite party set $A$, set $\Gamma(A)=\bigcup_{p\in A}G(p)=\{g\in G:P_g\cap A\ne\varnothing\}$.
\end{definition}

\begin{lemma}[Endpoint count at one source position]
    \label{thm:compression_endpoint_count}
    \lean{TNLean.Lattice.Compression.card_sourceEndpoints_le}
    \leanok
    Every source position $e$ satisfies $|E_e|\le2$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_source_endpoints}
    By Definition~\ref{def:compression_source_endpoints}, $E_e=\{a_e,b_e\}$ has at most two elements.
\end{proof}

\begin{lemma}[Corrected-party count]
    \label{thm:compression_corrected_party_count}
    \lean{TNLean.Lattice.Compression.card_correctedParties_le}
    \leanok
    For every finite correction set $S$, $|A_S|\le2|S|$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_corrected_parties,thm:compression_endpoint_count}
    By Definition~\ref{def:compression_corrected_parties} and Lemma~\ref{thm:compression_endpoint_count}, $|A_S|\le\sum_{e\in S}|E_e|\le2|S|$.
\end{proof}

\begin{lemma}[Lifetime bound for touching gates]
    \label{thm:compression_touching_gate_count}
    \lean{TNLean.Lattice.Compression.card_gatesTouching_le}
    \leanok
    If $|G(p)|\le b$ for every $p\in A$, then $|\Gamma(A)|\le b|A|$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_touching_gates}
    By Definition~\ref{def:compression_touching_gates}, $|\Gamma(A)|\le\sum_{p\in A}|G(p)|\le b|A|$.
\end{proof}

\begin{lemma}[Affected-gate count for corrected positions]
    \label{thm:compression_affected_gate_count}
    \lean{TNLean.Lattice.Compression.card_gatesTouching_correctedParties_le}
    \leanok
    If $|G(p)|\le b$ for every $p\in A_S$, then $|\Gamma(A_S)|\le2b|S|$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_touching_gate_count,thm:compression_corrected_party_count}
    Lemmas~\ref{thm:compression_touching_gate_count} and \ref{thm:compression_corrected_party_count} give $|\Gamma(A_S)|\le b|A_S|\le2b|S|$.
\end{proof}

\begin{lemma}[Owner of a corrected source position]
    \label{thm:compression_corrected_owner}
    \lean{TNLean.Lattice.Compression.owner_mem_gatesTouching_correctedParties}
    \leanok
    Let $o(e)$ be the owner of $e$. If $e\in S$, $o(e)\in G$, and $a_e\in P_{o(e)}$, then $o(e)\in\Gamma(A_S)$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_corrected_parties,def:compression_touching_gates}
    Here $a_e\in E_e\subseteq A_S$ and $o(e)\in G(a_e)\subseteq\Gamma(A_S)$.
\end{proof}

\begin{lemma}[Exterior gates avoid corrected parties]
    \label{thm:compression_exterior_disjoint}
    \lean{TNLean.Lattice.Compression.disjoint_participants_of_not_mem_gatesTouching}
    \leanok
    If $g\in G\setminus\Gamma(A)$, then $P_g\cap A=\varnothing$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_touching_gates}
    Immediate from $\Gamma(A)=\{g\in G:P_g\cap A\ne\varnothing\}$.
\end{proof}

\begin{definition}[Pair-sample labels]
    \label{def:compression_pair_labels}
    \lean{TNLean.Lattice.Compression.pairSampleLabels}
    \lean{TNLean.Lattice.Compression.mem_pairSampleLabels}
    \lean{TNLean.Lattice.Compression.mk_mem_pairSampleLabels}
    \leanok
    For a finite party set $P$, set $U(P)=\{\{p,q\}:p,q\in P,\ p\ne q\}$.
\end{definition}

\begin{definition}[Star and sample label sets]
    \label{def:compression_gate_labels}
    \lean{TNLean.Lattice.Compression.gateLinkLabels}
    \leanok
    For a root $r$, define the disjoint union $T(P,r)=(P\setminus\{r\})\sqcup U(P)$ of star and sample labels.
\end{definition}

\begin{definition}[Gate-local link endpoints]
    \label{def:compression_gate_link_parties}
    \lean{TNLean.Lattice.Compression.gateLinkParties}
    \leanok
    Set $F(r,q)=\{r,q\}$ for $q\in P\setminus\{r\}$ and $F(r,\{p,q\})=\{p,q\}$ for $\{p,q\}\in U(P)$.
\end{definition}

\begin{definition}[Links tagged by gate occurrence]
    \label{def:compression_gate_links}
    \lean{TNLean.Lattice.Compression.linksAtGate}
    \lean{TNLean.Lattice.Compression.mem_linksAtGate}
    \leanok
    For chosen roots $r_g$, set $\mathcal L_g=\{(g,t):t\in T(P_g,r_g)\}$.
\end{definition}

\begin{definition}[Distributed link family]
    \label{def:compression_distributed_links}
    \lean{TNLean.Lattice.Compression.distributedLinks}
    \lean{TNLean.Lattice.Compression.mem_distributedLinks}
    \leanok
    Set $\mathcal L=\bigcup_{g\in G}\mathcal L_g=\{(g,t):g\in G,\ t\in T(P_g,r_g)\}$.
\end{definition}

\begin{definition}[Tagged link endpoints]
    \label{def:compression_distributed_link_parties}
    \lean{TNLean.Lattice.Compression.distributedLinkParties}
    \leanok
    For $\ell=(g,t)\in\mathcal L$, set $V(\ell)=F(r_g,t)$.
\end{definition}

\begin{definition}[Distributed party incidence]
    \label{def:compression_distributed_incidence}
    \lean{TNLean.Lattice.Compression.incidentDistributedLinks}
    \leanok
    Set $I(p)=\{\ell\in\mathcal L:p\in V(\ell)\}$.
\end{definition}

\begin{definition}[Gate-local party incidence]
    \label{def:compression_gate_label_incidence}
    \lean{TNLean.Lattice.Compression.incidentGateLinkLabels}
    \leanok
    Set $T_p(P,r)=\{t\in T(P,r):p\in F(r,t)\}$.
\end{definition}

\begin{definition}[Tagged incidence at one gate]
    \label{def:compression_gate_incidence}
    \lean{TNLean.Lattice.Compression.incidentLinksAtGate}
    \leanok
    Set $\mathcal L_g(p)=\{\ell\in\mathcal L_g:p\in V(\ell)\}$.
\end{definition}

\begin{lemma}[Number of pair-sample labels]
    \label{thm:compression_pair_label_count}
    \lean{TNLean.Lattice.Compression.card_pairSampleLabels}
    \leanok
    If $|P|=n$, then $|U(P)|=\binom n2$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_pair_labels}
    Off-diagonal ordered pairs in $P^2$ map two-to-one onto $U(P)$, giving $|U(P)|=n(n-1)/2=\binom n2$.
\end{proof}

\begin{lemma}[Sample links incident to a participant]
    \label{thm:compression_pair_incidence}
    \lean{TNLean.Lattice.Compression.filter_pairSampleLabels_eq_map}
    \lean{TNLean.Lattice.Compression.card_filter_pairSampleLabels}
    \leanok
    For $p\in P$, $q\mapsto\{p,q\}$ bijects $P\setminus\{p\}$ onto $\{u\in U(P):p\in u\}$, which has cardinality $|P|-1$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_pair_labels}
    Each $\{p,q\}\in U(P)$ containing $p$ has a unique second endpoint $q\in P\setminus\{p\}$.
\end{proof}

\begin{lemma}[Number of links at one gate]
    \label{thm:compression_gate_label_count}
    \lean{TNLean.Lattice.Compression.card_gateLinkLabels}
    \lean{TNLean.Lattice.Compression.card_gateLinkLabels_le}
    \leanok
    If $r\in P$ and $|P|=n\le b$, then $|T(P,r)|=n-1+\binom n2\le b-1+\binom b2$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_gate_labels,thm:compression_pair_label_count}
    By Definition~\ref{def:compression_gate_labels} and Lemma~\ref{thm:compression_pair_label_count}, $|T(P,r)|=|P\setminus\{r\}|+|U(P)|=n-1+\binom n2$, which is monotone in $n$.
\end{proof}

\begin{lemma}[Gate-local incidence bound]
    \label{thm:compression_gate_incidence_bound}
    \lean{TNLean.Lattice.Compression.card_incidentGateLinkLabels_le}
    \leanok
    If $r,p\in P$, then $|T_p(P,r)|\le2(|P|-1)$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_gate_label_incidence,thm:compression_pair_incidence}
    The star summand $P\setminus\{r\}$ has size $|P|-1$, and the sample summand contributes $|P|-1$ by Lemma~\ref{thm:compression_pair_incidence}.
\end{proof}

\begin{lemma}[Gate-local endpoint locality]
    \label{thm:compression_gate_locality}
    \lean{TNLean.Lattice.Compression.gateLinkParties_subset}
    \leanok
    If $r\in P$ and $t\in T(P,r)$, then $F(r,t)\subseteq P$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_gate_labels,def:compression_gate_link_parties}
    Immediate from Definitions~\ref{def:compression_gate_labels} and \ref{def:compression_gate_link_parties}.
\end{proof}

\begin{lemma}[Distinct link endpoints]
    \label{thm:compression_gate_two_endpoints}
    \lean{TNLean.Lattice.Compression.card_gateLinkParties}
    \leanok
    For every $t\in T(P,r)$, $|F(r,t)|=2$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_gate_labels,def:compression_gate_link_parties}
    Star labels satisfy $q\ne r$, and sample labels $\{p,q\}\in U(P)$ satisfy $p\ne q$.
\end{proof}

\begin{lemma}[Tagging preserves cardinalities]
    \label{thm:compression_tagging_cardinality}
    \lean{TNLean.Lattice.Compression.card_linksAtGate}
    \lean{TNLean.Lattice.Compression.card_incidentLinksAtGate}
    \leanok
    For every $g$ and $p$, $|\mathcal L_g|=|T(P_g,r_g)|$ and $|\mathcal L_g(p)|=|T_p(P_g,r_g)|$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_gate_links,def:compression_gate_incidence}
    The map $t\mapsto(g,t)$ is injective and commutes with filtering by $p\in F(r_g,t)$.
\end{proof}

\begin{lemma}[Original-gate locality of distributed links]
    \label{thm:compression_original_gate_locality}
    \lean{TNLean.Lattice.Compression.distributedLinkParties_subset}
    \lean{TNLean.Lattice.Compression.exists_common_gate_of_mem_distributedLinkParties}
    \leanok
    Suppose $r_g\in P_g$ for all $g\in G$. Every $\ell=(g,t)\in\mathcal L$ satisfies $V(\ell)\subseteq P_g$, so any $p,q\in V(\ell)$ share $g\in G$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_gate_locality,def:compression_distributed_links}
    Combine Definition~\ref{def:compression_distributed_links} with Lemma~\ref{thm:compression_gate_locality}.
\end{proof}

\begin{lemma}[Two endpoints of each distributed link]
    \label{thm:compression_distributed_two_endpoints}
    \lean{TNLean.Lattice.Compression.card_distributedLinkParties}
    \leanok
    Every $\ell\in\mathcal L$ satisfies $|V(\ell)|=2$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_gate_two_endpoints,def:compression_distributed_links}
    Combine Definition~\ref{def:compression_distributed_links} with Lemma~\ref{thm:compression_gate_two_endpoints}.
\end{proof}

\begin{lemma}[Lifetime containment of incident links]
    \label{thm:compression_incidence_containment}
    \lean{TNLean.Lattice.Compression.incidentDistributedLinks_subset}
    \leanok
    If $r_g\in P_g$ for all $g\in G$, then $I(p)\subseteq\bigcup_{g\in G(p)}\mathcal L_g(p)$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_original_gate_locality,def:compression_gate_incidence}
    If $\ell=(g,t)\in I(p)$, Lemma~\ref{thm:compression_original_gate_locality} gives $p\in V(\ell)\subseteq P_g$, so $g\in G(p)$ and $\ell\in\mathcal L_g(p)$.
\end{proof}

\begin{lemma}[Bounded distributed incidence]
    \label{thm:compression_distributed_incidence_bound}
    \lean{TNLean.Lattice.Compression.card_incidentDistributedLinks_le}
    \leanok
    If $r_g\in P_g$ and $|P_g|\le b$ for all $g\in G$, and $|G(p)|\le b$, then $|I(p)|\le2b(b-1)$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_incidence_containment,thm:compression_gate_incidence_bound,thm:compression_tagging_cardinality}
    By Lemmas~\ref{thm:compression_incidence_containment}, \ref{thm:compression_tagging_cardinality}, and \ref{thm:compression_gate_incidence_bound}, $|I(p)|\le\sum_{g\in G(p)}|\mathcal L_g(p)|\le2b(b-1)$.
\end{proof}

\begin{definition}[Ket and bra coefficient cost]
    \label{def:compression_coefficient_cost}
    \lean{TNLean.Lattice.Compression.ketBraCoefficientCost}
    \leanok
    For finite $\Xi$ and $a_\xi\ge0$, set $W(\Xi,a)=(\sum_{\xi\in\Xi}a_\xi)^2$.
\end{definition}

\begin{lemma}[Double coefficient sum]
    \label{thm:compression_double_coefficient_sum}
    \lean{TNLean.Lattice.Compression.ketBraCoefficientCost_eq_sum}
    \leanok
    $W(\Xi,a)=\sum_{\xi,\zeta\in\Xi}a_\xi a_\zeta$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_coefficient_cost}
    Expand the square in Definition~\ref{def:compression_coefficient_cost}.
\end{proof}

\begin{definition}[Affected-position choice cost]
    \label{def:compression_position_choice_cost}
    \lean{TNLean.Lattice.Compression.correctedPositionChoiceCost}
    \leanok
    For finite $\Xi_g$, weights $a_{g,\xi}\ge0$, and dimensions $n_p\in\mathbb N$, define
    \begin{align}
        \mathcal C(S) & =\left(\prod_{g\in\Gamma(A_S)}W(\Xi_g,a_g)\right)\left(\prod_{p\in A_S}n_p^2\right).
        \label{eq:compression_incidence_choice_cost}
    \end{align}
\end{definition}

\begin{lemma}[Corrected-position weighted choice bound]
    \label{thm:compression_position_choice_bound}
    \lean{TNLean.Lattice.Compression.correctedPositionChoiceCost_le}
    \leanok
    If $B,d\ge1$, $|G(p)|\le b$ and $n_p\le d$ for all $p\in A_S$, and $\sum_{\xi\in\Xi_g}a_{g,\xi}\le B$ for all $g\in\Gamma(A_S)$, then $\mathcal C(S)\le(B^{4b}d^4)^{|S|}$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_position_choice_cost,thm:compression_affected_gate_count,thm:compression_corrected_party_count}
    In (\ref{eq:compression_incidence_choice_cost}), Lemmas~\ref{thm:compression_affected_gate_count} and \ref{thm:compression_corrected_party_count} yield $\mathcal C(S)\le B^{2|\Gamma(A_S)|}d^{2|A_S|}\le(B^{4b}d^4)^{|S|}$.
\end{proof}

\begin{lemma}[Explicit monomial enclosure of the choice base]
    \label{thm:compression_choice_monomial_bound}
    \lean{TNLean.Lattice.Compression.compressionChoiceBase_le_of_monomial_bounds}
    \leanok
    For $L,B,d,C,D\ge0$ and $a,c,b\in\mathbb N$, if $B\le CL^a$ and $d\le DL^c$, then $B^{4b}d^4\le C^{4b}D^4L^{4ba+4c}$.
\end{lemma}

\begin{proof}\leanok
    Multiply $(CL^a)^{4b}$ and $(DL^c)^4$.
\end{proof}
