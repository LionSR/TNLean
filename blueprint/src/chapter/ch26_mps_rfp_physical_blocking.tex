\section{Physical blocking and transfer idempotence}

\begin{definition}[Physical blocking isometry]
    \label{def:physical_blocking_isometry}
    \lean{MPSTensor.HasPhysicalBlockingIsometry}
    \leanok
    Following \cite[Definition~3.2]{Cirac2016MPDO_arXiv}, an MPS tensor $A$ is a
    \emph{renormalization fixed point} if there exists an isometry
    $V\colon \C^d\to\C^{d^2}$ with coefficients $V_{(i_1,i_2),j}$ satisfying
    $V^\dagger V=\Id$ that fuses every two-site product into a single-site
    letter: for all physical indices $i_1,i_2\in\{0,\ldots,d-1\}$,
    \begin{align}
        A^{i_1}A^{i_2}
         & =\sum_{j=0}^{d-1}V_{(i_1,i_2),j}A^j.
        \label{eq:corr_physical_blocking_isometry}
    \end{align}
\end{definition}

\begin{definition}[Idempotent transfer map]\label{def:mps_transfer_idempotence}
    \lean{MPSTensor.IsTransferIdempotent}
    \leanok
    \uses{def:transfer_map}
    The transfer map $\E_A$ of an MPS tensor $A$ satisfies \emph{transfer map idempotence} when
    $\E_A\circ\E_A=\E_A$.
\end{definition}

\begin{lemma}[Eigenvalues of an idempotent transfer map]
    \label{thm:transfer_idempotent_eigenvalue_eq_one}
    \lean{MPSTensor.eq_one_of_hasEigenvalue_of_isTransferIdempotent}
    \leanok
    \uses{def:mps_transfer_idempotence}
    If $\E_A\circ\E_A=\E_A$, then every nonzero eigenvalue of $\E_A$ equals
    $1$, matching the spectral fixed-point condition of
    \cite[Section~III.A]{Cirac2021Matrix}.
\end{lemma}

\begin{proof}\leanok
    If $\E_A(X)=\mu X$ with $X\neq0$, then $\mu^2X=\E_A^2(X)=\E_A(X)=\mu X$,
    so $\mu^2=\mu$ and hence $\mu=1$.
\end{proof}

\begin{lemma}[A physical blocking isometry gives transfer idempotence]
    \label{thm:physical_blocking_implies_transfer_idempotence}
    \lean{MPSTensor.isTransferIdempotent_of_kraus_isometry}
    \leanok
    \uses{def:mps_transfer_idempotence}
    Suppose there exists an isometry
    $V\colon \C^d\to\C^{d^2}$, written as coefficients $V_{(i_1,i_2),j}$, such
    that
    \begin{align}
        A^{i_1}A^{i_2}
         & =\sum_{j=0}^{d-1}V_{(i_1,i_2),j}A^j
        \label{eq:corr_kraus_isometry_decomp}
    \end{align}
    for all $i_1,i_2\in\{0,\ldots,d-1\}$.
    Then $\E_A^2=\E_A$.
\end{lemma}
\begin{proof}\leanok
    Substituting the decomposition (\ref{eq:corr_kraus_isometry_decomp}) into
    the double Kraus sum gives, for every bond matrix $X$,
    \begin{align}
        \E_A^2(X)
         & =\sum_{i_1,i_2}(A^{i_1}A^{i_2})X
        (A^{i_1}A^{i_2})^\dagger
        \notag                                             \\
         & =\sum_{j,k}\left(\sum_{i_1,i_2}
        V_{(i_1,i_2),j}\overline{V_{(i_1,i_2),k}}\right)A^jX(A^k)^\dagger
        \notag                                             \\
         & =\sum_{j,k}(V^\dagger V)_{j,k}A^jX(A^k)^\dagger
        =\sum_{j,k}\delta_{j,k}A^jX(A^k)^\dagger
        =\sum_jA^jX(A^j)^\dagger
        =\E_A(X).
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Kraus-isometry criterion for transfer idempotence]%
    \label{thm:kraus_isometry_iff_transfer_idempotence}
    \lean{MPSTensor.isTransferIdempotent_iff_kraus_isometry}
    \leanok
    \uses{def:mps_transfer_idempotence, thm:physical_blocking_implies_transfer_idempotence}
    The transfer map of $A$ is idempotent if and only if there exists an
    isometry $V\colon \C^d\to\C^{d^2}$, written as coefficients
    $V_{(i_1,i_2),j}$ with $V^\dagger V=\Id$, such that
    \begin{align}
        A^{i_1}A^{i_2}
         & =\sum_{j=0}^{d-1}V_{(i_1,i_2),j}A^j
        \label{eq:corr_AA_isometry}
    \end{align}
    for all $i_1,i_2\in\{0,\ldots,d-1\}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:physical_blocking_implies_transfer_idempotence}
    Lemma~\ref{thm:physical_blocking_implies_transfer_idempotence} proves the
    backward implication. Conversely, expanding $\E_A^2=\E_A$ as a Kraus sum
    gives, for every $X$,
    \begin{align}
        \sum_{i_1,i_2}(A^{i_1}A^{i_2})X(A^{i_1}A^{i_2})^\dagger
         & =\sum_jA^jX(A^j)^\dagger.
        \notag
    \end{align}
    Rectangular Kraus freedom~\cite[``Rectangular Kraus freedom with general finite index types'']{QICLean2026Blueprint} then produces the isometry $V$
    relating the two Kraus families.
\end{proof}

\begin{theorem}[Physical and transfer-map fixed-point criteria]
    \label{thm:physical_blocking_isometry_iff_transfer_idempotence}
    \lean{MPSTensor.isTransferIdempotent_iff_hasPhysicalBlockingIsometry}
    \leanok
    \uses{def:physical_blocking_isometry, def:mps_transfer_idempotence}
    The transfer map $\E_A$ of an MPS tensor $A$ is idempotent, $\E_A^2 = \E_A$, if and only if there exists an isometry $V \colon \C^d \to \C^d \otimes \C^d$ such that
    \begin{align}
        A^{i_1} A^{i_2} = \sum_j V_{(i_1, i_2), j} A^j.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:kraus_isometry_iff_transfer_idempotence}
    Lemma~\ref{thm:kraus_isometry_iff_transfer_idempotence} and
    Definition~\ref{def:physical_blocking_isometry} yield the equivalence
    directly.
\end{proof}

\begin{definition}[Limit of the renormalization flow]
    \label{def:appears_as_renormalization_flow_limit}
    \lean{MPSTensor.AppearsAsRenormalizationFlowLimit}
    \leanok
    \uses{def:transfer_map}
    Let $A=(A^i)_{i=0}^{d-1}$ be an MPS tensor with bond dimension $D$. The
    physical-isometry class of $A$ \emph{appears as a limit of the
        renormalization flow} if there exist a physical dimension $e$ and an MPS
    tensor $B=(B^a)_{a=0}^{e-1}$ of bond dimension $D$ such that
    \begin{align}
        \widehat{\E}_B^{\,2^n}
         & \longrightarrow \widehat{\E}_A.
        \notag
    \end{align}
    Here $\widehat{\E}_A$ and $\widehat{\E}_B$ are the transfer matrices, and
    convergence is taken in the standard topology on the finite-dimensional
    matrix space $\MN{D^2}$. Because transfer matrices represent tensors modulo
    physical isometries \cite[lines~382--394 and 1205--1209]{Cirac2016MPDO_arXiv},
    blocking may change the physical dimension while preserving the bond
    dimension $D$.
\end{definition}

\begin{lemma}[A dyadic power limit is idempotent]
    \label{thm:dyadic_power_limit_idempotent}
    \lean{isIdempotentElem_of_tendsto_pow_two_pow}
    \leanok
    Let $R$ be a Hausdorff topological monoid with continuous multiplication.
    If $E^{2^n}\to P$, then $P^2=P$.
\end{lemma}
\begin{proof}\leanok
    Continuity of multiplication gives
    $E^{2^n}E^{2^n}\to P^2$, while $E^{2^n}E^{2^n}=E^{2^{n+1}}\to P$, so
    uniqueness of limits gives $P^2=P$.
\end{proof}

\begin{theorem}[Limits of the renormalization flow]
    \label{thm:renormalization_flow}
    \lean{MPSTensor.IsTransferIdempotent.of_appearsAsRenormalizationFlowLimit,
        MPSTensor.IsTransferIdempotent.appearsAsRenormalizationFlowLimit,
        MPSTensor.appearsAsRenormalizationFlowLimit_iff_isTransferIdempotent,
        MPSTensor.appearsAsRenormalizationFlowLimit_iff_hasPhysicalBlockingIsometry}
    \leanok
    \uses{def:appears_as_renormalization_flow_limit, def:physical_blocking_isometry}
    As in \cite[Theorem~3.1 and lines~1205--1209]{Cirac2016MPDO_arXiv}, a
    tensor class appears as a limit of the two-site renormalization flow if
    and only if it has a physical blocking isometry: an isometry
    $V\colon\C^d\to\C^{d^2}$ such that
    \begin{align}
        A^{i_1}A^{i_2}
        =\sum_{j=0}^{d-1}V_{(i_1,i_2),j}A^j
        \notag
    \end{align}
    for every $i_1,i_2$.

    % Local fix (renormalization-flow indices): The printed display sums the
    % free indices i_1,i_2 and has the undefined output index j_1. The equation
    % above is the well-formed identity specified by the renormalization step
    % at lines 389--394, the source diagrams, and the Appendix proof. This
    % changes neither the hypotheses nor the conclusion and is documented in
    % docs/paper-gaps/cpsv16_renormalization_flow_index_typo.tex.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:dyadic_power_limit_idempotent, thm:physical_blocking_isometry_iff_transfer_idempotence}
    If $\widehat{\E}_B^{\,2^n}\to\widehat{\E}_A$, then
    Lemma~\ref{thm:dyadic_power_limit_idempotent} gives
    $\widehat{\E}_A^2=\widehat{\E}_A$. Injectivity of the transfer-matrix
    representation then implies $\E_A^2=\E_A$, and
    Theorem~\ref{thm:physical_blocking_isometry_iff_transfer_idempotence}
    produces the physical blocking isometry. Conversely, if $A$ has such an
    isometry, Theorem~\ref{thm:physical_blocking_isometry_iff_transfer_idempotence}
    makes $\E_A$ idempotent, so setting $B=A$ gives the constant dyadic orbit
    $\widehat{\E}_A^{\,2^n}=\widehat{\E}_A\to\widehat{\E}_A$.
\end{proof}
