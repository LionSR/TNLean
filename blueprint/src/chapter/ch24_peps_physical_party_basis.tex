% Auxiliary coordinate statements for polynomial PEPS compression.

\subsection{Physical bases and party grouping}

Let $P$ be a finite party set and let $p_1,\ldots,p_m$ be an
ordering containing each party exactly once. For a register list $a$,
write $M(a)$ for its right-associated tensor memory, including a terminal
factor $\C$, and $a_p$ for the registers owned by $p$, in their original
order. The canonical isometry $G_a$ successively partitions off
$a_{p_1},\ldots,a_{p_m}$, identifying $M(a)$ with
$M(a_{p_1})\otimes\cdots\otimes M(a_{p_m})\otimes\C$.

\begin{definition}[Party tensor bases]
    \label{def:peps_party_tensor_basis}
    \lean{TNLean.PEPS.PairEffect.partyListBasis,
        TNLean.PEPS.PairEffect.partyLabelledBasis}
    \leanok
    \uses{def:peps_party_layout}
    For each $p$, let $b_p$ be an orthonormal basis of $M(a_p)$,
    indexed by a finite set $I_p$. The ordered party basis has column
    $b_{p_1}(x_{p_1})\otimes\cdots\otimes b_{p_m}(x_{p_m})\otimes1$,
    indexed by assignments $x_p\in I_p$. For an empty ordering,
    its single column is the scalar $1$.
\end{definition}

\begin{definition}[Party bases on the original register order]
    \label{def:peps_party_grouped_basis}
    \lean{TNLean.PEPS.PairEffect.partyGroupedBasis}
    \leanok
    \uses{def:peps_party_tensor_basis}
    Transport the ordered party basis by $G_a^{-1}$. Its column on the
    original register memory is
    \begin{align}
        B^a_x=G_a^{-1}
        \bigl(b_{p_1}(x_{p_1})\otimes\cdots
        \otimes b_{p_m}(x_{p_m})\otimes1\bigr).
    \end{align}
\end{definition}

\begin{definition}[Original local physical dimensions]
    \label{def:peps_family_physical_output}
    \lean{TNLean.PEPS.PairEffect.familyPhysicalLayout,
        TNLean.PEPS.PairEffect.familyPhysicalOutputLayout,
        TNLean.PEPS.PairEffect.familyPhysicalListBasis,
        TNLean.PEPS.PairEffect.familyLabelledPhysicalBasis}
    \leanok
    \uses{def:peps_party_layout}
    Let the physical space at $p$ be $H_p=\C^{d_p}$, where $d_p$
    is a nonnegative integer, and denote its standard orthonormal
    basis by $e_{p,j}$. The physical register list has one register
    $H_{p_i}$ owned by $p_i$ in each position. Its labelled basis has
    column $E_x=e_{p_1,x_{p_1}}\otimes\cdots
        \otimes e_{p_m,x_{p_m}}\otimes1$ for each assignment
    $x_p\in\{0,\ldots,d_p-1\}$. The physical output list may be
    followed by any specified list of private registers.
\end{definition}


\begin{theorem}[Prescribed physical columns under an owner map]
    \label{thm:peps_physical_owner_columns}
    \lean{TNLean.PEPS.PairEffect.mapOwnerIso_familyPhysicalListBasis_apply_heq}
    \leanok
    \uses{def:peps_family_physical_output}
    Let $P,Q$ be arbitrary sets, let $f:P\to Q$, and let
    $\delta_q$ be nonnegative integers. Let
    $\mathbf p=(p_0,\ldots,p_{m-1})$ be any ordered list in $P$,
    with repetitions permitted. Fix a prescribed configuration
    $y_q\in\{0,\ldots,\delta_q-1\}$ for every $q\in Q$.
    At position $j$, take the physical space $\C^{\delta_{f(p_j)}}$.
    Changing its owner from $p_j$ to $f(p_j)$ preserves its space
    and its position. Write $U_f$ for the resulting canonical
    memory isometry. Then
    \begin{align}
        U_f\bigl(e_{p_0,y_{f(p_0)}}\otimes\cdots
        \otimes e_{p_{m-1},y_{f(p_{m-1})}}\otimes1\bigr)
        = e_{f(p_0),y_{f(p_0)}}\otimes\cdots
        \otimes e_{f(p_{m-1}),y_{f(p_{m-1})}}\otimes1.
        \label{eq:peps_physical_owner_columns}
    \end{align}
    Here the column on the left belongs to the original register list
    $\mathbf p$, whose party $p$ carries the physical dimension
    $\delta_{f(p)}$, and the column on the right belongs to the
    relabelled register list $(f(p_0),\ldots,f(p_{m-1}))$. No injectivity of $f$,
    distinctness of the listed owners, or coverage of $P$ or $Q$
    is required. The empty list has column $1\in\C$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_basis_owner_memory}
    By Theorem~\ref{thm:peps_basis_owner_memory}, changing owners
    leaves each tensor factor unchanged, and $U_f$ is the tensor
    product of the identities on these factors.
    The prescribed basis column is the ordered tensor product
    of its local standard basis vectors. Induction on the list
    therefore gives (\ref{eq:peps_physical_owner_columns});
    the empty tensor product is the scalar $1$.
\end{proof}

\begin{theorem}[Singleton physical memory of each party]
    \label{thm:peps_singleton_physical_memory}
    \lean{TNLean.PEPS.PairEffect.atParty_familyPhysicalLayout_of_mem}
    \leanok
    Let an ordered list of parties have no repetitions. For any party
    $p$ in this list, selecting its physical registers gives exactly
    its one register $H_p$, with memory $H_p\otimes\C$.
\end{theorem}

\begin{proof}
    \leanok
    Selecting a party filters the given list by equality with $p$.
    Since $p$ occurs exactly once, the filtered register list is the
    singleton containing its physical register.
\end{proof}

\begin{definition}[Physical basis at an individual party]
    \label{def:peps_singleton_physical_basis}
    \lean{TNLean.PEPS.PairEffect.familyPhysicalPartyBasis}
    \leanok
    \uses{thm:peps_singleton_physical_memory}
    Let $p$ occur in an ordered physical list with no repetitions.
    On its physical memory $H_p\otimes\C$, take the orthonormal basis
    with columns $b_p(j)=e_{p,j}\otimes1$.
    This is the standard basis transported by the inverse of the
    right scalar-unit isometry.
\end{definition}

\begin{theorem}[Partitioning a single selected register]
    \label{thm:peps_selected_head_partition}
    \lean{TNLean.PEPS.PairEffect.Layout.partitionIso_selectedHead_of_tail_excluded_heq}
    \leanok
    Let the first register have owner $p$ and Hilbert space $H$, and
    let every remaining register have an owner different from $p$.
    Write $K$ for the memory of the remaining registers. Under the
    canonical partition into the registers of $p$ and their complement,
    every $e\in H$ and $y\in K$ satisfy
    \begin{align}
        e\otimes y\longmapsto(e\otimes1)\otimes y.
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    Partition the remaining registers first. Their selected memory is
    $\C$, so $y$ is sent to $1\otimes y$. Adjoining the first register
    and applying the associativity isometry gives
    $e\otimes y\mapsto(e\otimes1)\otimes y$.
\end{proof}

\begin{theorem}[Grouping prescribed physical columns in arbitrary local coordinates]
    \label{thm:peps_basis_physical_columns_local_coordinates}
    \lean{TNLean.PEPS.PairEffect.groupByPartyIso_familyPhysicalListBasis_eq_partyListBasis}
    \leanok
    \uses{def:peps_family_physical_output,def:peps_party_tensor_basis}
    Let $P$ be an arbitrary set of parties, let $d_p$ be arbitrary
    nonnegative integers, and let $\mathbf p=(p_0,\ldots,p_{m-1})$
    be an ordered list without repetitions. Put
    $a=a_{\mathbf d}(\mathbf p)$, with register $\C^{d_{p_j}}$
    at the party in position $j$. For each party $p$, let $b_p$
    be an orthonormal basis of the complete memory $M(a_p)$,
    indexed by a finite set $I_p$. Let
    $x\in\prod_{p\in P}\{0,\ldots,d_p-1\}$ and
    $y\in\prod_{p\in P} I_p$.

    Suppose that, at every party in $\mathbf p$, the canonical
    singleton-memory identification sends $b_p(y_p)$ to
    $e_{x_p}\otimes1$. Then grouping the prescribed physical tensor
    column gives the ordered column of these local bases:
    \begin{align}
        G_{\mathbf p,a}\,
        b_{\mathbf d,\mathbf p}\bigl((x_{p_j})_{j=0}^{m-1}\bigr)
        = b_{a,\mathbf p}\bigl((y_{p_j})_{j=0}^{m-1}\bigr).
    \end{align}
    Here $b_{\mathbf d,\mathbf p}$ is the ordered Euclidean tensor
    basis, $G_{\mathbf p,a}$ successively separates the complete memories
    in the listed order, and $b_{a,\mathbf p}$ is the ordered tensor
    basis formed from $b_{p_0},\ldots,b_{p_{m-1}}$.
    The local index sets need not be the physical index sets.
    No coverage of all of $P$ is required: the list covers the owners
    of $a$ by construction. The empty list and zero local dimensions
    are included, with quantification over the indicated assignments.
    Both ordered memories retain their terminal scalar factor $\C$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_selected_head_partition,def:peps_party_tensor_basis}
    Induct on the length of the ordered list. For the empty list,
    both columns are $1\in\C$. At a nonempty list, its first
    party does not occur in the tail. The selected-register partition
    therefore separates its physical column as $e_{x_p}\otimes1$.
    Apply the induction hypothesis to the tail. The prescribed physical
    column thus groups as the ordered tensor product of its singleton
    columns. By the given identifications, these are precisely
    $b_{p_j}(y_{p_j})$ at every listed party. The ordered tensor basis
    has the same column,
    \begin{align}
        b_{p_0}(y_{p_0})\otimes\cdots\otimes
        b_{p_{m-1}}(y_{p_{m-1}})\otimes1.
    \end{align}
\end{proof}

\begin{theorem}[Compatibility of physical and party coordinates]
    \label{thm:peps_family_physical_party_basis}
    \lean{TNLean.PEPS.PairEffect.familyLabelledPhysicalBasis_eq_partyGroupedBasis}
    \leanok
    Let the physical register list $a$ and the singleton bases $b_p$
    be as above. For every physical assignment $x$,
    \begin{align}
        G_a E_x
         & =(e_{p_1,x_{p_1}}\otimes1)\otimes\cdots
        \otimes(e_{p_m,x_{p_m}}\otimes1)\otimes1.
    \end{align}
    Thus the labelled physical basis equals the party basis on the
    original register order. No positivity of $d_p$ is required.
    If some $d_p=0$, the assignment set and both bases are empty.
    If $P$ is empty, both tensor products are the scalar $1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_party_grouped_basis,def:peps_family_physical_output,
        def:peps_singleton_physical_basis,thm:peps_selected_head_partition}
    First consider any ordered list without repetitions, retaining
    only its physical registers. Induct on this list. For the empty
    list, both memories are $\C$ with basis vector $1$. In the
    induction step, no remaining register is owned by the first party.
    Theorem~\ref{thm:peps_selected_head_partition} separates its basis
    vector as $e_{p,j}\otimes1$, leaving the ordered physical basis
    vector on the tail. Apply the induction hypothesis to that tail.
    When the list exhausts $P$, transporting the resulting equality
    by $G_a^{-1}$ gives equality of the two bases.
\end{proof}

\begin{definition}[Joint physical and private basis of one party]
    \label{def:peps_party_local_output_basis}
    \lean{TNLean.PEPS.PairEffect.partyLocalOutputBasis}
    \leanok
    For register lists $a,c$, choose orthonormal bases $b_p$ of
    $M(a_p)$ and $f_p$ of $M(c_p)$, indexed respectively by finite
    sets $X_p$ and $F_p$. Let $J_{u,v}:M(u\mathbin{+\!+}v)
        \longrightarrow M(u)\otimes M(v)$ be the canonical concatenation
    isometry. The joint basis at party $p$ has column
    $J_{a_p,c_p}^{-1}(b_p(x_p)\otimes f_p(y_p))$, under the
    identification $(a\mathbin{+\!+}c)_p=a_p\mathbin{+\!+}c_p$.
\end{definition}

\begin{theorem}[Concatenating physical and private party bases]
    \label{thm:peps_party_grouped_basis_append}
    \lean{TNLean.PEPS.PairEffect.partyGroupedBasis_append}
    \leanok
    With the joint local bases just defined, let $B^{a+c}$ denote
    their party basis on $M(a\mathbin{+\!+}c)$, and let $B^a,B^c$
    be the separately transported party bases. Then for all
    assignments $x_p\in X_p$ and $y_p\in F_p$,
    \begin{align}
        B^{a+c}_{(x,y)}=J_{a,c}^{-1}(B^a_x\otimes B^c_y).
    \end{align}
    The index $(x,y)$ denotes the assignment $p\mapsto(x_p,y_p)$;
    equivalently, this is equality of the two bases after the canonical
    separation of the two assignment families.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_party_grouped_basis,def:peps_party_local_output_basis}
    Induct on the ordered party list. Partitioning a concatenation
    first partitions each list and then regroups its four factors,
    joining the two selected memories and the two complementary
    memories. The selected vector is the stated joint local column.
    Apply the induction hypothesis to the two complementary lists,
    and transport by $G_{a+c}^{-1}$. For the empty list all memories
    are scalar and the assertion reduces to multiplication in $\C$.
\end{proof}

These coordinate identities apply to the local contractions in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 137--151 and 565--588.

\begin{theorem}[Party memories after selecting all original parties]
    \label{thm:peps_selected_physical_party_layouts}
    \lean{TNLean.PEPS.PairEffect.Layout.atParty_none_mapOwner_all}
    \lean{TNLean.PEPS.PairEffect.Layout.atParty_some_mapOwner_all}
    \uses{def:peps_affected_owner,def:peps_party_registers,def:peps_owner_layout}
    Let $\mathcal P$ be any party set, let
    $\mathcal Q=\mathcal P\sqcup\{\star\}$, and let $\iota:\mathcal P\to\mathcal Q$
    assign each original party its selected owner. For any register list $a$,
    the registers of $\iota_*a$ at $\star$ form the empty list. At $\iota(p)$,
    their list is $\iota_*(a_p)$, in the original order. This is the
    all-party selection in the proof of
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}.
\end{theorem}

\begin{proof}
    \uses{thm:peps_excluded_mapped_owners,thm:peps_basis_owner_restriction}
    No original party maps to $\star$, so restriction to $\star$ is empty.
    An original register maps to $\iota(p)$ exactly when its owner is $p$.
    Restriction after relabelling therefore selects exactly $a_p$.
\end{proof}

\begin{definition}[Selected physical coordinate families and party bases]
    \label{def:peps_selected_family_physical_basis}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalIndex}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalPartyBasis}
    \uses{thm:peps_selected_physical_party_layouts,def:peps_singleton_physical_basis}
    Let $d:\mathcal P\to\mathbb N$ specify the original physical dimensions,
    and let $\mathbf p$ list every original party exactly once. Set
    $X_\star=\{\bullet\}$ and $X_{\iota(p)}=\{0,\ldots,d(p)-1\}$.
    On the physical memory at $\star$, take the scalar basis with column $1$.
    At $\iota(p)$, transport the original singleton physical basis
    $b_p$ along the canonical owner isometry and the register-list equality
    of Theorem~\ref{thm:peps_selected_physical_party_layouts}.
    Denote the resulting basis by $b^{\mathrm{sel}}_{\iota(p)}$.
    Dimensions are retained individually, including $d(p)=0$; the original
    party set may be empty.
\end{definition}

\begin{theorem}[Original physical columns after selecting all parties]
    \label{thm:peps_selected_family_physical_columns}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalPartyBasis_none_apply_heq}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalPartyBasis_some_apply_heq}
    \uses{def:peps_selected_family_physical_basis}
    Under the canonical memory identifications,
    $b^{\mathrm{sel}}_{\star,\bullet}=1$ and
    $b^{\mathrm{sel}}_{\iota(p),j}=b_{p,j}$ for every $j\in X_{\iota(p)}$.
    No equality between independently chosen bases is required.
\end{theorem}

\begin{proof}
    \uses{def:peps_selected_family_physical_basis,thm:peps_basis_memory_equalities,
        thm:peps_basis_owner_memory}
    The scalar singleton basis has column $1$. Equality identifications of
    register lists preserve vectors. The canonical owner isometry also
    preserves each vector under the equality of the underlying memory spaces,
    so the transported original columns have the stated values.
\end{proof}




\begin{definition}[Actual physical and private columns at a selected owner]
    \label{def:peps_selected_physical_private_basis}
    \lean{TNLean.PEPS.PairEffect.selectedPhysicalPrivateBasis}
    \lean{TNLean.PEPS.PairEffect.selectedPhysicalPrivateBasis_apply_heq}
    \uses{def:peps_selected_family_physical_basis,def:peps_party_local_output_basis}
    Write $a$ for the original physical register list and $c$ for the private
    output list. For each selected owner $q$, choose an orthonormal basis
    $f_q$ of $M((\iota_*c)_q)$, indexed by a finite set $F_q$.
    The local basis on $M((\iota_*(a\mathbin{+\!+}c))_q)$ has indices
    $(x_q,e_q)\in X_q\times F_q$ and column
    \begin{align}
        C_q(x_q,e_q)
         & =J_{(\iota_*a)_q,(\iota_*c)_q}^{-1}
        \bigl(b^{\mathrm{sel}}_q(x_q)\otimes f_q(e_q)\bigr).
        \label{eq:peps_selected_physical_private_column}
    \end{align}
    Here the two memories are identified using
    $(\iota_*(a\mathbin{+\!+}c))_q
        =(\iota_*a)_q\mathbin{+\!+}(\iota_*c)_q$.
    Thus the physical coordinate has its original dimension at each
    $\iota(p)$ and a single scalar coordinate at $\star$. The private bases
    are arbitrary; their columns are retained in~(\ref{eq:peps_selected_physical_private_column}).
\end{definition}



\begin{definition}[Original physical configurations]
    \label{def:peps_selected_physical_configurations}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalConfigEquiv}
    \uses{def:peps_selected_family_physical_basis}
    The canonical bijection
    $e:\prod_{q\in\mathcal Q}X_q\to
        \prod_{p\in\mathcal P}\{0,\ldots,d(p)-1\}$,
    given by $e(x)_p=x_{\iota(p)}$, removes the single exterior coordinate. Its inverse assigns
    $x_\star=\bullet$ and $x_{\iota(p)}=y_p$.
    It neither enlarges a physical space nor requires that a party exist.
\end{definition}



\begin{theorem}[The exterior scalar in original physical configurations]
    \label{thm:peps_selected_physical_product_scalar}
    \lean{TNLean.PEPS.PairEffect.partyProductMatrix_selectedFamilyPhysicalConfig}
    \uses{def:peps_selected_physical_configurations}
    Let $\mathcal P$ be finite, and let $M_q$ be arbitrary complex matrices
    indexed by $X_q$. In the original physical configurations,
    \begin{align}
        \prod_{q\in\mathcal Q}
        (M_q)_{e^{-1}(x)_q,e^{-1}(y)_q}
         & =(M_\star)_{\bullet,\bullet}
        \prod_{p\in\mathcal P}(M_{\iota(p)})_{x_p,y_p}.
        \label{eq:peps_selected_physical_product_scalar}
    \end{align}
    The exterior scalar may vanish. If $\mathcal P$ is empty, the
    product on the right is $1$.
\end{theorem}

\begin{proof}
    \uses{def:peps_selected_physical_configurations}
    Separate the exterior factor from the finite product over
    $\mathcal Q=\mathcal P\sqcup\{\star\}$, and relabel the remaining
    factors by their original owners. No division is used.
\end{proof}




\begin{definition}[The selected party order]
    \label{def:peps_selected_party_order}
    \lean{TNLean.PEPS.PairEffect.selectedPartyOrder}
    \lean{TNLean.PEPS.PairEffect.selectedPartyOrder_nodup}
    \lean{TNLean.PEPS.PairEffect.mem_selectedPartyOrder}
    \uses{thm:peps_selected_physical_party_layouts}
    From the original order $\mathbf p=(p_0,\ldots,p_{m-1})$, form
    $\mathbf q=(\star,\iota(p_0),\ldots,\iota(p_{m-1}))$.
    If $\mathbf p$ lists every original party exactly once, then
    $\mathbf q$ lists every selected owner exactly once. If there is no
    original party, $\mathbf q=(\star)$.
\end{definition}



\begin{theorem}[The grouped basis in original physical configurations]
    \label{thm:peps_selected_physical_basis_grouped}
    \lean{TNLean.PEPS.PairEffect.selectedFamilyPhysicalPartyBasis_grouped}
    \uses{def:peps_selected_party_order,def:peps_selected_family_physical_basis,
        def:peps_selected_physical_configurations,def:peps_party_grouped_basis,
        def:peps_family_physical_output}
    Let $\mathbf p$ list every original party exactly once. Group the
    relabelled physical memory in the selected order $\mathbf q$, using
    the local bases $b_q^{\mathrm{sel}}$. Write $B$ for the resulting basis
    on the original register order, and $U_\iota$ for the canonical owner
    isometry. In each original physical configuration $x$,
    \begin{align}
        B_{e^{-1}(x)} & = U_\iota E_x.
        \label{eq:peps_selected_physical_basis_grouped}
    \end{align}
    This identifies the two constructed bases. The exterior party contributes
    the scalar column $1$, and no additional physical register is inserted.
    Empty party sets and zero physical dimensions are included.
\end{theorem}

\begin{proof}
    \uses{thm:peps_physical_owner_columns,
        thm:peps_basis_physical_columns_local_coordinates,
        thm:peps_selected_family_physical_columns,thm:peps_basis_empty_first_party,
        thm:peps_basis_grouping_equality,thm:peps_basis_memory_equalities,
        thm:peps_basis_party_recursion}
    First group the relabelled physical column along
    $(\iota(p_0),\ldots,\iota(p_{m-1}))$. The owner isometry preserves the
    ordered physical factors. At each listed owner the selected local column
    is the original singleton column, so the grouped vector is their
    ordered tensor product. Prepending $\star$ contributes only $1$,
    since it owns no register. Transport back to the register memory to
    obtain~(\ref{eq:peps_selected_physical_basis_grouped}).
\end{proof}




\begin{theorem}[Separating the actual physical and private grouped bases]
    \label{thm:peps_selected_physical_private_basis_grouped}
    \lean{TNLean.PEPS.PairEffect.selectedPhysicalPrivateBasis_grouped}
    \uses{def:peps_selected_physical_private_basis,def:peps_party_grouped_basis,
        thm:peps_party_grouped_basis_append}
    Choose any order containing every selected owner exactly once.
    Let $C$ be the grouped basis obtained from the actual local columns
    in~(\ref{eq:peps_selected_physical_private_column}), and let
    $B^{\iota_*a}$ and $B^{\iota_*c}$ be the separately grouped physical
    and private bases in the same order. Separating the two coordinate
    families gives
    \begin{align}
        C_{q\mapsto(x_q,e_q)}
         & =J_{\iota_*a,\iota_*c}^{-1}
        \bigl(B^{\iota_*a}_x\otimes B^{\iota_*c}_e\bigr).
        \label{eq:peps_selected_physical_private_basis_grouped}
    \end{align}
    The output memories are identified through
    $\iota_*(a\mathbin{+\!+}c)=\iota_*a\mathbin{+\!+}\iota_*c$.
    All private basis choices are retained, and the physical indices have
    their original dimensions with the exterior singleton coordinate.
\end{theorem}

\begin{proof}
    \uses{thm:peps_party_grouped_basis_append,thm:peps_basis_memory_equalities}
    Apply the concatenation identity to the separately relabelled register
    lists. Transport each local column and the grouped column by the
    displayed equality of output lists. Equality identifications preserve
    these columns, yielding~(\ref{eq:peps_selected_physical_private_basis_grouped}).
\end{proof}
