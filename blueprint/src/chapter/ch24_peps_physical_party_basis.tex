\subsection{Physical and private configurations}

Let $P$ be a finite party set with a repetition-free ordering $p_1,\ldots,p_m$ of $P$.

\begin{definition}[Original local physical dimensions]
    \label{def:peps_family_physical_output}
    \lean{QuantumCircuit.Party.physicalSystem}
    \leanok
    \uses{def:peps_party_layout}
    For local dimensions $d_p\in\N$, let $H_p=\C^{d_p}$ have standard orthonormal basis $(e_{p,j})_{j=0}^{d_p-1}$. The physical register list has register $H_{p_i}$ owned by $p_i$ at position $i$, with labelled basis columns
    \begin{align}
        E_x & = e_{p_1,x_{p_1}}\otimes\cdots\otimes e_{p_m,x_{p_m}}\otimes1, &
        x   & \in \prod_{p\in P}\{0,\ldots,d_p-1\}, \notag
    \end{align}
    and may be followed by a private register list.
\end{definition}

These coordinates apply to the local contractions in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 137--151 and 565--588.

% Polynomial-PEPS Theorem 5.2, 04-compression.tex, lines 137--151 and 565--588.


\begin{definition}[Selected physical coordinate families and party bases]
    \label{def:peps_selected_family_physical_basis}
    \lean{QuantumCircuit.Party.SourceCircuit.blockPhys}
    \leanok
    \uses{def:peps_owner_layout,def:peps_family_physical_output}
    For $d:P\to\mathbb N$, let $Q=P\sqcup\{*\}$ with the inclusion $\iota:P\to Q$ of the all-party selection in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}. Grouping the parties along $\iota$, the block of $\iota(p)$ consists of $p$ alone and the exterior block $*$ is empty. Accordingly, set $X_*:=\{\bullet\}$ and $X_{\iota(p)}:=\{0,\ldots,d(p)-1\}$, the physical configurations of the block $q\in Q$.
\end{definition}

% Polynomial-PEPS Theorem 5.2, 04-compression.tex, lines 137--151, 409--417 and 565--588.


\begin{definition}[Physical and private columns at a selected owner]
    \label{def:peps_selected_physical_private_basis}
    \lean{QuantumCircuit.Party.SourceCircuit.blockReadout,
        QuantumCircuit.Party.SourceCircuit.blockEnv}
    \leanok
    \uses{def:peps_selected_family_physical_basis}
    Let the output configurations at each party $p$ be split by a bijection $\epsilon_p$ into a physical coordinate in $X_{\iota(p)}$ and a private coordinate in a finite set $F_p$, and set $F_*:=\{\bullet\}$ and $F_{\iota(p)}:=F_p$. At each $q\in Q$, define the local readout $C_q$ from the output configurations of the block $q$ to $X_q\times F_q$: for $q=\iota(p)$ and a configuration $z$ of $p$,
    \begin{align}
        C_q\bigl((x_q,e_q),z\bigr)
         & =\begin{cases}
                1 & \text{if }\epsilon_p(z)=(x_q,e_q), \\
                0 & \text{otherwise},
            \end{cases}
        \label{eq:peps_selected_physical_private_column}
    \end{align}
    and $C_*$ is the $1\times1$ matrix $1$.
\end{definition}



\begin{definition}[Original physical configurations]
    \label{def:peps_selected_physical_configurations}
    \lean{QuantumCircuit.Party.SourceCircuit.blockPhysEquiv}
    \leanok
    \uses{def:peps_selected_family_physical_basis}
    The bijection $e:\prod_{q\in Q}X_q\to\prod_{p\in P}\{0,\ldots,d(p)-1\}$ is defined by $e(x)_p:=x_{\iota(p)}$, with inverse $e^{-1}(y)_*=\bullet$ and $e^{-1}(y)_{\iota(p)}=y_p$.
\end{definition}



\begin{lemma}[The exterior scalar in original physical configurations]
    \label{thm:peps_selected_physical_product_scalar}
    \lean{Fintype.prod_option}
    \leanok
    \uses{def:peps_selected_physical_configurations}
    For finite $P$, matrices $M_q\in\C^{X_q\times X_q}$ ($q\in Q$), and $x,y\in\prod_{p\in P}\{0,\ldots,d(p)-1\}$,
    \begin{align}
        \prod_{q\in Q}
        (M_q)_{e^{-1}(x)_q,e^{-1}(y)_q}
         & =(M_*)_{\bullet,\bullet}
        \prod_{p\in P}(M_{\iota(p)})_{x_p,y_p}.
        \label{eq:peps_selected_physical_product_scalar}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_selected_physical_configurations}
    Split the product over $Q=\{*\}\sqcup\iota(P)$ and substitute Definition~\ref{def:peps_selected_physical_configurations}.
\end{proof}
