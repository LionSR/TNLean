\subsection{Finite improvement of the box exponent}
\label{sec:area_law_exponent_bootstrap}

\begin{definition}[Fixed parameters for exponent improvement]
    \label{def:area_law_bootstrap_parameters}
    \lean{
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.g0,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.mu,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.nu,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.ell,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.kappa
    }
    \leanok
    \uses{thm:area_law_exponent_gaps}
    For $e_0\in\mathbb R$, set $e_*=2\cdot10^{-6}$ and define the fixed
    parameters of \cite[Proposition~9.5]{OpenAI2026AreaLaw}:
    \begin{align}
        g_0  & =\frac{1-e_0}{2}, & \mu    & =1-g_0,              & \nu & =\frac{g_0}{1000},
        \label{eq:area_law_bootstrap_parameters}                                            \\
        \ell & =\frac{\nu}{100}, & \kappa & =\min\{\ell,e_*/4\}.
        \label{eq:area_law_bootstrap_separation}
    \end{align}
\end{definition}

\begin{theorem}[Margins in the exponent estimates]
    \label{thm:area_law_bootstrap_margins}
    \lean{
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.parameter_bounds,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.density_exponents,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.energy_exponents,
        TNLean.PEPS.AreaLaw.Scan.BootstrapParameters.comparison_exponents
    }
    \leanok
    \uses{def:area_law_bootstrap_parameters}
    Under (\ref{eq:area_law_bootstrap_parameters})--(\ref{eq:area_law_bootstrap_separation}),
    $0<e_0<1$ gives $0<g_0<1/2$, $0<\ell<1$, and $\kappa>0$. For $e\le e_0<1$,
    \begin{align}
        e_0-\mu+\nu & =-0.999g_0,
                    & -\frac{g_0-\ell}{4}+\nu & =-0.2489975g_0,
        \label{eq:area_law_bootstrap_density_values}            \\
        e-\mu+\nu   & <-0.24g_0,
                    & -\frac{g_0-\ell}{4}+\nu & <-0.24g_0,
        \label{eq:area_law_bootstrap_density_margin}
    \end{align}
    and $p=\ell+2\kappa$ satisfies
    \begin{align}
        p & \le0.00003g_0, & p-\nu & <-\nu/2, & p-0.03g_0 & <-\nu/2.
        \label{eq:area_law_bootstrap_energy_margin}
    \end{align}
    Moreover, $e_0>0$ and $e>e_*$ give $\kappa<(1-\ell)e$ and $g_0-\ell-\nu/4<1$.
\end{theorem}
\begin{proof}\leanok
    Direct substitution of (\ref{eq:area_law_bootstrap_parameters})--(\ref{eq:area_law_bootstrap_separation})
    with $g_0=(1-e_0)/2\in(0,1/2)$, $\ell=g_0/10^5$, and $0<\kappa\le\min\{\ell,e_*/4\}$.
\end{proof}

\begin{theorem}[Principal energy term at the fixed bootstrap scales]
    \label{thm:area_law_bootstrap_principal_energy}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_principal_energy_bound}
    \leanok
    \uses{def:area_law_bootstrap_parameters}
    Fix $0<e_0<1$, the parameters of
    Definition~\ref{def:area_law_bootstrap_parameters}, $C,C_\delta\ge0$, and
    $\gamma,\gamma_\delta\in\mathbb R$. For $n\ge2$ and $e\le e_0$ (with
    $x/0=0$), define the principal energy term of
    \cite[Section~9, lines~731--755]{OpenAI2026AreaLaw} by
    \begin{align}
        L_n         & =\lfloor n^{1-\ell}\rfloor,                 &
        m_n         & =\lfloor n^\mu\rfloor,                      &
        K_n         & =\left\lfloor\frac{L_n}{8m_n}\right\rfloor, &
        D_n         & =\lceil n^\kappa\rceil,                     & a_n & =\frac4{K_n},
        \label{eq:al_principal_energy_scales}                                           \\
        \delta_n(e) & =C_\delta n^\nu
        \bigl(n^{e-\mu}+a_n^{1/4}(\log n)^{\gamma_\delta}\bigr),
        \label{eq:al_principal_energy_density}                                          \\
        A_n(e)      & =16C n^\ell D_n^2(\log n)^\gamma
        \bigl(\delta_n(e)^{1/8}+n^{-\nu}+n^{-100}\bigr).
        \label{eq:al_principal_energy_term}
    \end{align}
    There is $N\ge2$ (depending only on the fixed constants and $e_0$) such
    that $A_n(e)\le n^{-\nu/2}$ for all $n\ge N$ and $e\le e_0$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:al_scanner_scale_facts,thm:area_law_bootstrap_margins}
    By Lemma~\ref{lem:al_scanner_scale_facts}, there is $C_1\ge1$ such that for
    large $n$,
    \begin{align}
        n   & \le C_1n^\ell m_nK_n\le C_1n^{\ell+\mu}K_n, &
        a_n & \le4C_1n^{-(g_0-\ell)}.
        \label{eq:al_principal_energy_band}
    \end{align}
    For $n$ large enough that $\max\{4C_1,2C_\delta,(\log n)^{\gamma_\delta}\}\le n^\nu$
    and $192C(\log n)^\gamma\le n^{\nu/4}$, the uniform bounds $g_0-\ell\ge45\nu$
    and $e-\mu\le-10\nu$ ($e\le e_0$) give $a_n\le n^{-44\nu}$,
    $n^{e-\mu}+a_n^{1/4}(\log n)^{\gamma_\delta}\le2n^{-10\nu}$, and
    $\delta_n(e)\le n^{-8\nu}$. Since $\nu<100$ and $D_n\le2n^\kappa$,
    (\ref{eq:area_law_bootstrap_energy_margin}) yields
    \begin{align}
        A_n(e)
         & \le192C(\log n)^\gamma n^{\ell+2\kappa-\nu}
        \le n^{\ell+2\kappa-3\nu/4}
        \le n^{-\nu/2}. \notag
    \end{align}
\end{proof}

\begin{corollary}[Numerical energy bound at the fixed bootstrap scales]
    \label{cor:area_law_bootstrap_energy_expression}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_energy_expression_bound}
    \leanok
    \uses{thm:area_law_bootstrap_principal_energy}
    Under the hypotheses of Theorem~\ref{thm:area_law_bootstrap_principal_energy},
    there is $N\ge2$ such that for all $n\ge N$ and $e\le e_0$,
    \begin{align}
        E_n(e) & =A_n(e)+Cn^{-1000}
        \label{eq:al_bootstrap_energy_expression} \\
               & \le(1+C)n^{-\nu/2}.
        \label{eq:al_bootstrap_energy_expression_bound}
    \end{align}
\end{corollary}
\begin{proof}\leanok
    \uses{thm:area_law_bootstrap_principal_energy,def:area_law_bootstrap_parameters}
    Since $\nu/2=g_0/2000<1000$, we have $n^{-1000}\le n^{-\nu/2}$ for $n\ge2$;
    combine with Theorem~\ref{thm:area_law_bootstrap_principal_energy}.
\end{proof}

\begin{theorem}[Binary-entropy cost of the rounded bands]
    \label{thm:area_law_bootstrap_binary_entropy_cost}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_binEntropy_cost_bound}
    \leanok
    For fixed scanner exponents (Section~\ref{sec:al_scanner}), with $L_n,m_n,K_n,a_n$
    as in (\ref{eq:al_principal_energy_scales}) and
    $h(\tau)=-\tau\log|\tau|-(1-\tau)\log|1-\tau|$ ($0\log0=0$), there is
    $N\ge1$ such that for all $n\ge N$ and $\tau\in\mathbb R$
    (\cite[Proposition~9.5]{OpenAI2026AreaLaw}),
    \begin{align}
        K_n                 & \ge1,                         & a_n & >0, &
        \frac{h(\tau)}{a_n} & \le\frac{\log2}{4}\,n. \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{lem:al_scanner_scale_facts}
    Since $\mu<1-\ell$, Lemma~\ref{lem:al_scanner_scale_facts} gives
    $1\le K_n\le L_n\le n$ for large $n$, and $h(\tau)\le\log2$ gives
    $h(\tau)/a_n=K_nh(\tau)/4\le(n\log2)/4$.
\end{proof}

\begin{theorem}[Additive comparison costs at fixed bootstrap scales]
    \label{thm:area_law_bootstrap_comparison_budget}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_comparison_budget_bound}
    \leanok
    \uses{def:area_law_bootstrap_parameters}
    Fix $e_*<e_0<1$ ($e_*=2\cdot10^{-6}$), the parameters of
    (\ref{eq:area_law_bootstrap_parameters})--(\ref{eq:area_law_bootstrap_separation}),
    $\eta=1-\ell$, and $L_n,m_n,K_n,D_n,a_n$ as in
    (\ref{eq:al_principal_energy_scales}). There is $N\ge1$ (depending only on
    $e_0$) such that for all $n\ge N$, $e_*<e\le e_0$, and $\tau\in\mathbb R$
    (\cite[Proposition~9.5]{OpenAI2026AreaLaw}), $K_n\ge1$, $a_n>0$, and
    \begin{align}
        nL_n^e+nD_n+n^{3/5}+\frac{h(\tau)}{a_n}+1
         & \le\left(5+\frac{\log2}{4}\right)n^{1+\eta e}. \notag
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:area_law_bootstrap_binary_entropy_cost,thm:area_law_bootstrap_margins}
    Take $N$ from Theorem~\ref{thm:area_law_bootstrap_binary_entropy_cost} and
    set $P=n^{1+\eta e}\ge n\ge1$. Then $nL_n^e\le P$, $n^{3/5}\le P$,
    $h(\tau)/a_n\le(\log2/4)P$, $1\le P$, and by
    Theorem~\ref{thm:area_law_bootstrap_margins} ($\kappa<\eta e$),
    $D_n\le n^\kappa+1\le2n^{\eta e}$, so $nD_n\le2P$.
\end{proof}

\begin{theorem}[Uniform bounds after finitely many exponent improvements]
    \label{thm:area_law_finite_exponent_improvement}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_uniform_boxError_bound_of_eventual_improvement}
    \leanok
    \uses{def:area_law_bootstrap_parameters}
    Fix $0<e_0<1$, $e_*=2\cdot10^{-6}$, and $\ell$ from
    (\ref{eq:area_law_bootstrap_separation}). Let $F_i:\mathbb N\to\mathbb R$
    satisfy, for some $V\ge0$, $C_0>0$, and all $i$ and $r\ge1$,
    \begin{align}
        F_i(r) & \le Vr^2, & F_i(r) & \le C_0r^{1+e_0}.
        \label{eq:area_law_bootstrap_initial}
    \end{align}
    Suppose that for every $e_*<e\le e_0$, having $F_i(r)\le C_er^{1+e}$ for all
    $i$ and $r\ge1$ implies the existence of $C'_e>0$ and $N_e$ with
    \begin{align}
        F_i(r) & \le C'_er^{1+(1-\ell)e}\quad(r\ge N_e).
        \label{eq:area_law_bootstrap_step}
    \end{align}
    Then there is $C>0$ such that $F_i(r)\le Cr^{1+e_*}$ for all $i$ and $r\ge1$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_bootstrap_margins}
    By Theorem~\ref{thm:area_law_bootstrap_margins}, $a=1-\ell\in(0,1)$.
    Using $F_i(r)\le Vr^2$ from (\ref{eq:area_law_bootstrap_initial}) to absorb
    $1\le r<N_e$ at each step (\ref{eq:area_law_bootstrap_step}), iterate until
    $a^ne_0\le e_*$, where $r^{1+a^ne_0}\le r^{1+e_*}$.
\end{proof}
