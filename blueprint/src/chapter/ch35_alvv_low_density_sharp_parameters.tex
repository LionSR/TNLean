\section{Independent scales for sharp low-energy approximation}

The clipping threshold must be chosen independently of the polynomial
window length to retain the three-quarter power of the accuracy logarithm.
The following numerical choice supplies the squared-buffer comparison budget
and the Chebyshev exponent for the same pair of parameters. The physical
approximation and its global matrix product representation are separate.

\begin{lemma}[Sharp scales at a fixed low-energy bandwidth]
    \label{lem:ld_sharp_independent_parameters}
    \lean{Real.exists_sharp_low_density_scalar_parameters}
    \leanok
    For every fixed $\eta>0$ there is $C_\eta\ge21$, chosen before
    $0<g\le1$ and $0<\delta\le1$, with the following property.
    Put $a=16+\log(1/g)+\log(1/\delta)$ and $t=C_\eta a$.
    There is an integer $\ell\ge1$ such that
    \begin{align}
        \ell         & \le\frac8g+(7C_\eta)^{1/4}g^{-1/4}a^{3/4},\notag \\
        19+\eta      & \le t,\notag                                     \\
        \xi          & =288e^{11/8}e^{-(t-\eta)/8}\le1,\notag           \\
        2(6+\eta)\xi & \le\frac\delta{100}g^2.
        \label{eq:ld_sharp_scales}
    \end{align}
    For every $0<R\le3t+4\ell$ one also has
    $\bigl(2e^{-2\ell^2\sqrt{g/R}}\bigr)^2\le\delta/8$.
\end{lemma}
\begin{proof}\leanok
    Let $C_0$ be the existing universal independent-threshold constant,
    $B=6+\eta$ and $C_\eta=\eta+C_0(2+\log B)$.
    Apply its residual estimate at buffer $g^2$ and accuracy
    $\tau=\delta/B$. Its logarithmic scale
    $u=1+2\log(1/g)+\log B+\log(1/\delta)$ satisfies
    $u\le(2+\log B)a$, so $t\ge\eta+C_0u$.
    Consequently $\xi\le\tau g^2/10^6$, which gives the residual
    assertions in (\ref{eq:ld_sharp_scales}).
    The fourth-root integer-window choice for $t=C_\eta a$ gives the
    stated bound on $\ell$ and $a\le\ell^2\sqrt{g/R}$.
    Since $a\ge16$ and $\log(1/\delta)\le a$, the squared Chebyshev
    estimate is at most $\delta/8$.
\end{proof}


\begin{lemma}[Actual sharp regional low-energy polynomial families]
    \label{lem:ld_sharp_regional_polynomial_family}
    \lean{TNLean.MPS.AreaLaw.exists_fixed_bandwidth_sharp_regional_low_energy_family}
    \leanok
    For each fixed bandwidth $\eta_{\rm LD}>0$, choose $C\ge21$
    before $w>0$ and $0<\epsilon\le1$. Put
    $g=\min\{w,1\}$, $u=16+\log(1/g)+\log(1/\epsilon)$ and $t=Cu$.
    There is an integer $\ell\ge1$ with
    $\ell\le8/g+(7C)^{1/4}g^{-1/4}u^{3/4}$, chosen before the physical
    chain, such that the following holds.
    Let $q\ge2$, $m\ge q$, $D=(m\ell^2)^{22\ell}$ and suppose the
    five regions fit at $a,b,\ell$. Let
    $J_0=[a+\ell,b+\ell+1)$ and
    $|J_0\setminus J|+|J\setminus J_0|\le2\ell$.
    For the actual supported positive-contraction bonds, the original
    regional minima $e_j$, common regional ground vector and centered
    original ground energy $0\le E_0\le4$ can be chosen so that
    $H_0=H-(\sum_j e_j)I$ and the actual three-bulk clipping $B_t$
    satisfies $\|B_t\|\le3t+4\ell$.
    Writing $\lambda_0=\min\sigma(H)$, the exact identities
    \begin{align}
        \lambda_0 & =E_0+\sum_j e_j,\notag                         \\
        \mathbf1_{(-\infty,E_0+c]}(H_0)\mathcal H
                  & =\mathbf1_{(-\infty,\lambda_0+c]}(H)\mathcal H
        \label{eq:ld_sharp_original_bands}
    \end{align}
    hold for every real $c$ before the buffered cutoff is chosen.
    There are $N\le D^2$ matrices $W_j$ on $J$, chosen before
    every $6w\le\eta\le\eta_{\rm LD}$ and source space.
    For each such cutoff, let
    $T_-=\mathbf1_{(-\infty,E_0+\eta-2w]}(B_t)\mathcal H$ and
    $T_+=\mathbf1_{(-\infty,E_0+\eta-w]}(B_t)\mathcal H$.
    Then $T_-\subseteq T_+$, $G(\eta)$ is $\epsilon/100$-close
    to $T_+$, and $T_-$ is $\epsilon/100$-close to $G(\eta-3w)$,
    where $G$ is the canonical original band in
    (\ref{eq:ld_sharp_original_bands}).
    A real polynomial $p$ of degree exactly $\ell^2$, with
    $p(E_0+\eta-2w)=1$, and actual matrices $M_j$ on $J^c$ satisfy
    $p(B_t)=\sum_j W_j\otimes M_j$ in the physical tensor coordinates.
    Its squared suppression
    \begin{align}
        \Delta=\left(2\exp\left[-2\ell^2
            \sqrt{\frac{w}{3t+4\ell-(E_0+\eta-2w)}}\right]\right)^2
        \le\epsilon/8
        \label{eq:ld_sharp_actual_suppression}
    \end{align}
    gives, for every $\delta<1$ and every $\delta$-viable $S$ for $T_+$,
    an actual image $V=\sum_j W_jS$ of dimension at most $N\dim S$,
    which is $\Delta/(1-\delta)^2$-viable for $T_-$.
    There is no supplied decomposition, spectral gap or invariance
    hypothesis on an original band. The five-region and collar conditions
    are explicit; arbitrary intervals require a subsequent argument.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:ld_sharp_independent_parameters,lem:ld_buffered_targets_independent,
        thm:actual_clipped_polynomial_family,thm:collar_larger_count_base}
    Apply Lemma~\ref{lem:ld_sharp_independent_parameters} at the capped
    buffer $g$. Since $g\le w$, its squared-buffer residual estimate
    supplies Lemma~\ref{lem:ld_buffered_targets_independent} with
    comparison loss $\epsilon/100$. The actual polynomial family at
    degree $\ell^2$ has collar-adjusted count at most $D^2$.
    At the two clipped cutoffs, apply the nested Chebyshev construction
    to the same matrix $B_t$ and the same regional family.
    The actual spectral denominator is positive and at most $3t+4\ell$;
    replacing its numerator $w$ by $g$ weakens suppression, so the scalar
    estimate gives (\ref{eq:ld_sharp_actual_suppression}).
    The original normalized ground vector and positive lower bound
    identify its canonical minimum and every shifted spectral span as in
    (\ref{eq:ld_sharp_original_bands}). This retains the actual polynomial
    and tensor factors and uses no original-band invariance.
\end{proof}
