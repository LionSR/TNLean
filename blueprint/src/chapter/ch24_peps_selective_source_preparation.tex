\section{Free source inputs and untouched exterior gates}

The separation argument in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 342--434, uses corrected source
registers as free inputs. Exact sources internal to either side are
prepared there, while gates wholly in the exterior retain their
complete operators. We give the corresponding exact constructions.

\subsection{Finite gate expansions}

\begin{definition}[A gate with common finite source positions]
    \label{def:peps_prepared_source_gate}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.branchWord}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.eval}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_registers}
    Fix a finite branch set $\Xi$, coefficients $c_\xi\in\C$, and
    input and output register lists $a,b$ on the gate's participating
    party set $P$. Common finite source data consist of an ordered list
    $E$ containing each unordered pair of distinct parties once,
    spaces $U_e=\C^{u_e}$ and $V_e=\C^{v_e}$, normalized vectors
    $\eta_{\xi,e}\in U_e\otimes V_e$, and allowed compositions
    $T_\xi:\mathcal S\otimes\mathcal H_a\to\mathcal H_b$
    containing no pair preparations. Here $\mathcal S$ is the memory of
    all source registers in their prescribed order. Write
    $F_\xi=T_\xi Q_{\eta_\xi,\mathcal H_a}$ and
    \begin{align}
        G & =\sum_{\xi\in\Xi}c_\xi F_\xi,
        \label{eq:peps_prepared_gate_sum}
    \end{align}
    and require $\|G\|\leq1$. The coefficients are fixed before these
    data are chosen. Completion by unused pairs is taken only over the
    participating parties $P$ of this gate.
\end{definition}

\begin{theorem}[Prepared branches and the complete gate]
    \label{thm:peps_prepared_gate_properties}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.branchWord_isAllowed}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.sources_branchWord}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.norm_eval_le_one}
    \leanok
    \uses{def:peps_prepared_source_gate}
    Each $F_\xi$ is an allowed composition. Its ordered source list is
    the list of prescribed pairs with spaces $U_e,V_e$ and
    vectors $\eta_{\xi,e}$. The complete sum $G$, with its original
    scalar coefficients, is a contraction.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_prepared_source_gate}
    Normalization makes the source preparation allowed, and composing it with
    $T_\xi$, whose source list is empty, leaves only the
    prescribed preparations as sources. The bound on $G$ is part of the gate data.
\end{proof}

\begin{theorem}[Construction from the original branch operators]
    \label{thm:peps_prepared_gate_exists}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_preparedSourceGate}
    \leanok
    \uses{def:peps_prepared_source_gate}
    Suppose $P$ and $\Xi$ are finite, each $W_\xi:a\to b$ is an
    allowed composition, and
    $\|\sum_{\xi\in\Xi}c_\xi W_\xi\|\leq1$.
    There are common finite source data with the same coefficients
    $c_\xi$ for which $G=\sum_{\xi\in\Xi}c_\xi W_\xi$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_finite_source_gate}
    Applying Theorem~\ref{thm:peps_finite_source_gate} to the original
    branch compositions gives common finite spaces, normalized
    vectors, and allowed compositions without sources, with
    $T_\xi Q_{\eta_\xi,\mathcal H_a}=W_\xi$ for each $\xi$.
    Substitution in (\ref{eq:peps_prepared_gate_sum}) preserves the
    original sum and therefore its contraction bound.
\end{proof}

\subsection{Affected parties and the exterior}

\begin{definition}[Keeping affected owners distinct]
    \label{def:peps_affected_owner}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner_eq_none}
    \leanok
    Let $\mathcal A\subseteq P$ have decidable membership. Set
    $Q=\mathcal A\sqcup\{*\}$, where $*$ denotes the exterior, and
    define $g:P\to Q$ by
    \begin{align}
        g(p) & =
        \begin{cases}
            p, & p\in\mathcal A,    \\
            *, & p\notin\mathcal A.
        \end{cases}
        \label{eq:peps_affected_owner}
    \end{align}
    Thus $g(p)=*$ when $p$ is exterior.
\end{definition}

\begin{theorem}[Equality of grouped owners]
    \label{thm:peps_affected_owner_equality}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner_eq_iff}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.affectedOwner_ne_iff}
    \leanok
    \uses{def:peps_affected_owner}
    One has $g(p)=g(q)$ if and only if $p=q$ or both parties are
    exterior. Consequently, for distinct source endpoints $p,q$,
    $g(p)\neq g(q)$ if and only if at least one endpoint belongs to
    $\mathcal A$. Both corrected pairs within $\mathcal A$ and pairs
    crossing to its exterior remain between distinct owners.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_affected_owner}
    Separating the four possibilities for membership of $p,q$ in
    $\mathcal A$, two affected images agree only when their original
    parties agree, and an affected image cannot equal $*$.
\end{proof}

\begin{theorem}[The surviving ordered source occurrences]
    \label{thm:peps_affected_source_occurrences}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_mapOwner_affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.map_partyPair_affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_affectedOwner_eq_nil_iff}
    \leanok
    \uses{def:peps_affected_owner,def:peps_owner_sources}
    A source occurs in $S_g$ if and only if it is obtained from an
    occurrence $s=(p,q;U,V;\eta)$ in $S$ with at least one affected
    endpoint, by replacing its owners with $g(p),g(q)$. Its spaces and
    vector are unchanged. The ordered pair labels are obtained by
    filtering $S$ by this same condition and then applying $g$ to both
    endpoints. Order and multiplicity are retained, even when different
    original pairs acquire the same image. In particular, $S_g$ is
    empty if and only if both endpoints of every original source are
    exterior.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_owner_sources,thm:peps_owner_sources_internal,
        thm:peps_affected_owner_equality}
    Substituting the distinct-owner criterion of
    Theorem~\ref{thm:peps_affected_owner_equality} into the occurrence
    description of $S_g$ gives the first characterization, and induction on the original source list gives
    the ordered pair-label identity. The empty-list criterion follows
    from Theorem~\ref{thm:peps_owner_sources_internal}.
\end{proof}

\begin{definition}[Retaining a complete operation at one grouped owner]
    \label{def:peps_grouped_block_map}
    \lean{TNLean.Lattice.PartyCircuit.Word.groupedBlockMap}
    \leanok
    \uses{def:peps_owner_layout,def:peps_party_layout}
    Let $f:P\to Q$, and suppose every register in $a$ and $b$ has
    owner image $q\in Q$. For $D:\mathcal H_a\to\mathcal H_b$,
    regard $R_{f,b}DR_{f,a}^{-1}$ as a local operation at $q$ and
    leave the spectator list $\ell$ untouched. This gives a
    composition $D^f_{\ell}:(a\ell)^f\to(b\ell)^f$.
\end{definition}

\begin{theorem}[Exact action of an untouched aggregate]
    \label{thm:peps_grouped_block_map}
    \lean{TNLean.Lattice.PartyCircuit.Word.groupedBlockMap_spec}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_groupedBlockMap}
    \leanok
    \uses{def:peps_grouped_block_map}
    The complete operation satisfies
    \begin{align}
        D^f_{\ell}R_{f,a\ell}
         & =R_{f,b\ell}C_{b,\ell}^{-1}
        (D\otimes I_{\mathcal H_\ell})C_{a,\ell}.
        \label{eq:peps_grouped_block_action}
    \end{align}
    If $\|D\|\leq1$, this composition is allowed and contains no pair
    sources. For $f=g$ and $q=*$, this retains an exterior gate as its
    original aggregate contraction, without expanding its branch sum.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_owner_concatenation,def:peps_grouped_block_map}
    Applying the concatenation identity for the owner
    isometries, the two isometries conjugating $D$ cancel with the
    corresponding input and output identifications, giving
    (\ref{eq:peps_grouped_block_action}). Conjugation by isometries
    preserves the contraction bound, and a local operation creates no
    pair sources.
\end{proof}

\begin{theorem}[Grouping preserves the weighted gate sum]
    \label{thm:peps_grouped_gate_sum}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_sum_mapOwner_comp}
    \leanok
    \uses{def:peps_owner_composition}
    For a finite family of compositions $F_\xi:a\to b$ and arbitrary
    coefficients $c_\xi\in\C$,
    \begin{align}
        \left(\sum_\xi c_\xi F_\xi^f\right)R_{f,a}
         & =R_{f,b}\left(\sum_\xi c_\xi F_\xi\right).
        \label{eq:peps_grouped_gate_sum}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_owner_composition}
    Summing the identity $F_\xi^fR_{f,a}=R_{f,b}F_\xi$ with
    coefficients $c_\xi$ and using linearity of composition on both sides gives the claim.
\end{proof}

\begin{definition}[The complete gate after grouping]
    \label{def:peps_prepared_gate_grouped}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.evalAtOwners}
    \leanok
    \uses{def:peps_prepared_source_gate,def:peps_owner_layout}
    For common finite source data and any map $f:P\to Q$, define
    the grouped aggregate by $G^f=R_{f,b}GR_{f,a}^{-1}$.
\end{definition}

\begin{theorem}[The original expansion and aggregate norm after grouping]
    \label{thm:peps_prepared_gate_grouped}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.evalAtOwners_eq_sum}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.norm_evalAtOwners_le_one}
    \leanok
    \uses{def:peps_prepared_gate_grouped,def:peps_owner_composition}
    The branch expansion is $G^f=\sum_\xi c_\xi F_\xi^f$,
    and $\|G^f\|\leq1$. This bound comes from the original
    aggregate contraction and does not involve
    $\sum_\xi|c_\xi|$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_grouped_gate_sum,thm:peps_prepared_gate_properties}
    Composing (\ref{eq:peps_grouped_gate_sum}) with $R_{f,a}^{-1}$ gives the branch expansion, and since
    both owner maps are isometries, conjugating the
    aggregate preserves its norm bound.
\end{proof}

\subsection{Preparing only the fixed sources}

\begin{definition}[The selected original source positions]
    \label{def:peps_selected_source_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.selectedSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.layout_selectedSources_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.freeSlotLayout}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_selectedSources}
    \leanok
    \uses{def:peps_source_slots}
    Let $E$ be an ordered list of pair positions, with halfspaces
    $U_e,V_e$, and let $E_{\rm free}\subseteq E$ be selected. The
    selected source list retains these occurrences in their original
    order, with their original endpoint parties and the prescribed
    spaces and vectors. Its register layout is independent of those
    vectors. Denote its memory by $\mathcal S_{\rm free}$ and the
    memory of all positions by $\mathcal S$.
\end{definition}

\begin{definition}[Selective source preparation]
    \label{def:peps_selective_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareFreeSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.fillSourceVectors}
    \leanok
    \uses{def:peps_selected_source_positions}
    Fix vectors $\eta_e\in U_e\otimes V_e$ only at positions outside
    $E_{\rm free}$. Define
    $\Phi_\eta:\mathcal S_{\rm free}\otimes\mathcal H_\ell
        \to\mathcal S\otimes\mathcal H_\ell$ by preparing the fixed
    sources and retaining all free registers in their original
    positions. This composition is independent of the
    vectors later assigned to the free registers.

    For an arbitrary family $\zeta_e\in U_e\otimes V_e$, write
    $Q^{\rm free}_{\zeta,\mathcal H_\ell}$ for preparation at just
    the free positions. The completed family $\theta$ has
    $\theta_e=\zeta_e$ at free positions and $\theta_e=\eta_e$
    elsewhere.
\end{definition}

\begin{theorem}[Only the fixed sources are prepared]
    \label{thm:peps_selective_preparation_sources}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.isAllowed_prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.sources_prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_sources_prepareSelected}
    \leanok
    \uses{def:peps_selective_preparation}
    The source list of $\Phi_\eta$ is the list of fixed
    original positions, in their original order, with vectors
    $\eta_e$. In particular, each prepared source has the original
    endpoint parties and halfspaces of its fixed position. If every
    fixed vector has norm one, then $\Phi_\eta$ is allowed.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_selective_preparation,def:peps_selected_source_positions}
    Proceeding by induction on the source list, at a free position the construction retains the two
    registers and applies recursively to the remaining positions;
    at a fixed position it prepares its source after the remaining
    positions, so the source inventory records this fixed
    occurrence, and its preparation is allowed by normalization.
\end{proof}

\begin{theorem}[Uniform recovery of every completed preparation]
    \label{thm:peps_selective_preparation_recovery}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSelected_prepareFreeSlots}
    \leanok
    \uses{def:peps_selective_preparation}
    For every family of free vectors,
    \begin{align}
        \Phi_\eta Q^{\rm free}_{\zeta,\mathcal H_\ell}
         & =Q_{\theta,\mathcal H_\ell}.
        \label{eq:peps_selective_recovery}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_selective_preparation,thm:peps_source_preparation}
    Proceeding by induction on the ordered positions, at a free head preparation of
    its vector commutes with the operations on the remaining
    registers, while at a fixed head both sides prepare the same fixed
    vector. In each case the induction hypothesis gives the equality
    for the remaining positions, and the empty source list gives the
    identity on the original input memory.
\end{proof}

\begin{theorem}[Two contractions independent of the free source vectors]
    \label{thm:peps_selective_partition}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_selective_partition}
    \leanok
    \uses{def:peps_selective_preparation,def:peps_party_partition}
    Let $f:P\to\{0,1\}$ specify two sides. Suppose every fixed
    source has both endpoints on the same side and has norm one.
    Let $T:\mathcal S\otimes\mathcal H_\ell\to\mathcal H_{\ell'}$
    be an allowed composition containing no pair sources. There are
    two allowed compositions $A_+,A_-$, containing no pair sources and
    each having norm at most one, such that
    \begin{align}
        J_{f,\ell'}T\Phi_\eta
         & =(A_+\otimes A_-)J_{f,L},
        \label{eq:peps_selective_partition} \\
        J_{f,\ell'}TQ_{\theta,\mathcal H_\ell}
         & =(A_+\otimes A_-)J_{f,L}
        Q^{\rm free}_{\zeta,\mathcal H_\ell}.
        \label{eq:peps_selective_all_vectors}
    \end{align}
    Here $L$ is the free-source register list followed by $\ell$, and
    $J_{f,L}$ is its partition isometry. The same two
    compositions work for every family $\zeta$ of free vectors.
    Thus the contraction bounds apply on all free inputs.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_selective_preparation_sources,
        thm:peps_internal_source_factorization,
        thm:peps_selective_preparation_recovery}
    The only sources in $T\Phi_\eta$ are the fixed original
    positions, which are internal by hypothesis, so
    Theorem~\ref{thm:peps_internal_source_factorization} yields
    (\ref{eq:peps_selective_partition}) and the two contraction
    bounds. Composing that identity with
    $Q^{\rm free}_{\zeta,\mathcal H_\ell}$ and using
    (\ref{eq:peps_selective_recovery}) yields
    (\ref{eq:peps_selective_all_vectors}), with the factors chosen
    before $\zeta$.
\end{proof}

\subsection{Operations beside untouched registers}

\begin{definition}[Extension by spectator registers]
    \label{def:peps_append_spectators}
    \lean{TNLean.Lattice.PartyCircuit.Word.appendTail}
    \leanok
    \uses{def:peps_party_layout,thm:peps_register_exchange}
    Given a composition $W:a\to b$ and a spectator list
    $\ell$, move $\ell$ before $a$, perform $W$ while leaving those
    registers unchanged, and move them back after $b$. Denote the
    resulting composition $a\ell\to b\ell$ by $W_{\ell}$.
\end{definition}

\begin{theorem}[Exact spectator extension]
    \label{thm:peps_append_spectators}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_appendTail}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_appendTail}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_appendTail}
    \leanok
    \uses{def:peps_append_spectators}
    The spectator extension has the same ordered source list
    as $W$, is allowed whenever $W$ is allowed, and satisfies
    \begin{align}
        W_{\ell}C_{a,\ell}^{-1}
         & =C_{b,\ell}^{-1}(W\otimes I_{\mathcal H_\ell}).
        \label{eq:peps_append_spectators}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_register_exchange,thm:peps_source_preparation}
    On an elementary input $x\otimes y$, the first exchange gives
    $y\otimes x$, the framed composition gives $y\otimes Wx$, and
    the last exchange gives $Wx\otimes y$, so linearity gives
    (\ref{eq:peps_append_spectators}). Exchanges create no sources,
    and leaving a register unchanged preserves the source list and
    allowedness of the operations on the other registers.
\end{proof}

\begin{theorem}[Relabelling a composition beside spectators]
    \label{thm:peps_owner_spectator_extension}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapOwner_appendTail_heq}
    \leanok
    \uses{def:peps_append_spectators,def:peps_owner_composition,
        def:peps_owner_layout}
    For the register operations in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $g:P\to Q$ be any map of party labels, let $W:a\to b$
    be a chronological composition, and let $t$ be an
    ordered spectator list. Write $W^g$ and $t^g$ for their
    relabellings. Under the concatenation identifications
    $(at)^g=a^g t^g$ and $(bt)^g=b^g t^g$, the constructed
    compositions satisfy $(W_t)^g=(W^g)_{t^g}$.
    % Source: 04-compression.tex, lines 233--267 and 409--434.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_append_spectators,def:peps_owner_composition,
        def:peps_owner_layout,thm:peps_register_exchange}
    Relabelling preserves the spaces, order and concatenation of
    register lists. For a register $r$, a block $A$ and an adjacent
    exchange, each frame and exchange acquires its relabelled
    owners,
    \begin{align}
        (\mathrm{frame}_r\,w)^g
         & =\mathrm{frame}_{r^g}(w^g),
        \qquad
        (\mathrm{Exch}_{r,A})^g=\mathrm{Exch}_{r^g,A^g},
        \label{eq:peps_owner_frame_exchange}
    \end{align}
    so induction on a prefix shows that moving a register through it
    commutes with relabelling. Induction on the second block then gives
    the block exchange identity
    $(\mathrm{Exch}_{A,B})^g=\mathrm{Exch}_{A^g,B^g}$, and induction on
    $t$ gives $(W_{[t]})^g=(W^g)_{[t^g]}$ for the framing of $W$ by $t$.
    Applying these identities to the initial exchange, the framed core and
    the final exchange defining $W_t$ and composing gives
    \begin{align}
        (W_t)^g
         & =\bigl(\mathrm{Exch}_{t,b}\circ W_{[t]}\circ\mathrm{Exch}_{a,t}\bigr)^g
        =\mathrm{Exch}_{t^g,b^g}\circ(W^g)_{[t^g]}\circ\mathrm{Exch}_{a^g,t^g}
        =(W^g)_{t^g},
    \end{align}
    under the concatenation identifications.
\end{proof}


\begin{theorem}[Congruence under equal register lists]
    \label{thm:peps_word_structural_congruence}
    \lean{TNLean.Lattice.PartyCircuit.Word.comp_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.frame_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.id_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.swap_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.castLayouts_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_heq}
    \leanok
    \uses{def:peps_party_layout,thm:peps_party_restriction}
    For the register operations in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    identify equal ordered input, intermediate and output register
    lists. If chronological compositions $W$ and $W'$ agree under
    these identifications, and so do $V$ and $V'$, then
    $V\circ W=V'\circ W'$. Framing the same register $r$ preserves
    this equality: $\mathrm{frame}_r W=\mathrm{frame}_r W'$.
    Identity compositions on equal lists agree, and exchanging the
    same two registers beside equal spectator lists gives the same
    composition. Identifying equal input and output lists preserves
    the original composition itself. If two constructed words agree
    under equalities of their ordered input and output lists, their
    evaluated operators also agree under the induced
    memory identifications.
    % Source: 04-compression.tex, lines 233--267 and 351--381.
\end{theorem}

\begin{proof}\leanok
    Replacing each equal register list by its identified list and each
    equal composition by the same composition shows that both sides have
    identical chronological operations, including the identity,
    frame and exchange operations, and evaluating the same constructed
    word gives the same operator on the identified input and output memories.
\end{proof}

\begin{theorem}[Relabelling equal register lists]
    \label{thm:peps_owner_layout_identification}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapOwner_castLayouts}
    \leanok
    \uses{def:peps_owner_composition,thm:peps_party_restriction}
    For the register operations in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $f:P\to Q$ be an owner map and $W:a\to b$ a

    Write $h_a^f:a^f=(a')^f$ and $h_b^f:b^f=(b')^f$ for the

    $(W[h_a,h_b])^f=W^f[h_a^f,h_b^f]$.

\begin{proof}\leanok
    After identifying $a'$ with $a$ and $b'$ with $b$, both sides

\end{proof}

\begin{theorem}[Restriction commutes with relabelling owners]
    \label{thm:peps_owner_selection_commutation}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_mapOwner_heq}
    \leanok
    \uses{def:peps_owner_composition,thm:peps_party_restriction}
    For the party restrictions in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $f:P\to Q$ be any map of party labels, let $B\subseteq Q$
    be any selection of new owners, and let $W:a\to b$ be a
    chronological composition containing no pair-source preparations.
    Under the list identifications
    $(a^f)_B=(a_{f^{-1}(B)})^f$ and
    $(b^f)_B=(b_{f^{-1}(B)})^f$, the constructed compositions satisfy
    \begin{align*}
        (W^f)_B & =(W_{f^{-1}(B)})^f.
    \end{align*}
    % Source: 04-compression.tex, lines 383--434 and 565--588.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_word_structural_congruence,
