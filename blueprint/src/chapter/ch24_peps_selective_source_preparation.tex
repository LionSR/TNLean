\section{Free source inputs and untouched exterior gates}

The separation argument in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 342--434, uses corrected source
registers as free inputs. Exact sources internal to either side are
prepared there, while gates wholly in the exterior retain their
complete operators. We give the corresponding exact constructions.

% identifications. The chronological partially expanded circuit and the
% corrected-subset trace-norm estimate are separate obligations.

\subsection{Finite gate expansions}

\begin{definition}[A gate with common finite source positions]
    \label{def:peps_prepared_source_gate}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.branchWord}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.eval}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_registers}
    For a finite branch set $\Xi$, coefficients $c_\xi\in\C$, and boundary
    register lists $a,b$ on the gate's participating party set $P$, common
    finite source data consist of an ordered list $E$ of all unordered pairs of
    distinct parties in $P$, halfspaces $U_e=\C^{u_e}$ and $V_e=\C^{v_e}$ with
    total source memory $\mathcal S$ (Definition~\ref{def:peps_source_slots}),
    unit vectors $\eta_{\xi,e}\in U_e\otimes V_e$ ($\|\eta_{\xi,e}\|=1$), and
    source-free allowed words $T_\xi:\mathcal S\otimes\mathcal H_a\to\mathcal H_b$
    such that $F_\xi=T_\xi Q_{\eta_\xi,\mathcal H_a}$ and
    \begin{align}
        G & =\sum_{\xi\in\Xi}c_\xi F_\xi
        \label{eq:peps_prepared_gate_sum}
    \end{align}
    satisfies $\|G\|\leq1$.
\end{definition}

\begin{lemma}[Prepared branches and the complete gate]
    \label{thm:peps_prepared_gate_properties}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.branchWord_isAllowed}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.sources_branchWord}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.norm_eval_le_one}
    \leanok
    \uses{def:peps_prepared_source_gate}
    In Definition~\ref{def:peps_prepared_source_gate}, each $F_\xi$ is allowed
    with ordered source list $((e;U_e,V_e;\eta_{\xi,e}))_{e\in E}$, and
    $\|G\|\le 1$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_prepared_source_gate}
    Since $\|\eta_{\xi,e}\|=1$ and $T_\xi$ is source-free and allowed,
    Lemma~\ref{thm:peps_source_preparation} and
    Definition~\ref{def:peps_prepared_source_gate} imply that $F_\xi=T_\xi Q_{\eta_\xi,\mathcal H_a}$
    is allowed with source list $((e;U_e,V_e;\eta_{\xi,e}))_{e\in E}$ and $\|G\|\le 1$.
\end{proof}

\begin{lemma}[Construction from the original branch operators]
    \label{thm:peps_prepared_gate_exists}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_preparedSourceGate}
    \leanok
    \uses{def:peps_prepared_source_gate}
    If $P$ and $\Xi$ are finite, each $W_\xi:a\to b$ is an allowed word, and
    $\|\sum_{\xi\in\Xi}c_\xi W_\xi\|\leq1$, then there exist common finite
    source data (Definition~\ref{def:peps_prepared_source_gate}) with the same
    $c_\xi$ such that $G=\sum_{\xi\in\Xi}c_\xi W_\xi$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_finite_source_gate}
    By Lemma~\ref{thm:peps_finite_source_gate}, there exist common finite
    spaces $U_e,V_e$, unit vectors $\eta_{\xi,e}$, and source-free allowed words
    $T_\xi$ with $T_\xi Q_{\eta_\xi,\mathcal H_a}=W_\xi$ for all $\xi\in\Xi$;
    substituting into~(\ref{eq:peps_prepared_gate_sum}) gives
    $G=\sum_{\xi\in\Xi}c_\xi W_\xi$ with $\|G\|\le 1$.
\end{proof}

\subsection{Affected parties and the exterior}

\begin{definition}[Keeping affected owners distinct]
    \label{def:peps_affected_owner}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner_eq_none}
    \leanok
    For $\mathcal A\subseteq P$ with decidable membership, set
    $Q=\mathcal A\sqcup\{*\}$ and define $g:P\to Q$ by
    \begin{align}
        g(p) & =
        \begin{cases}
            p,   & p\in\mathcal A,    \\
            {*}, & p\notin\mathcal A.
        \end{cases}
        \label{eq:peps_affected_owner}
    \end{align}
\end{definition}

\begin{lemma}[Equality of grouped owners]
    \label{thm:peps_affected_owner_equality}
    \lean{TNLean.Lattice.PartyCircuit.affectedOwner_eq_iff}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.affectedOwner_ne_iff}
    \leanok
    \uses{def:peps_affected_owner}
    For $g$ as in Definition~\ref{def:peps_affected_owner}, $g(p)=g(q)$ if and
    only if $p=q$ or $p,q\notin\mathcal A$. Consequently, for $p\neq q$,
    $g(p)\neq g(q)$ if and only if $\{p,q\}\cap\mathcal A\neq\varnothing$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_affected_owner}
    By~(\ref{eq:peps_affected_owner}), $g$ is injective on $\mathcal A$ and maps
    $P\setminus\mathcal A$ to $*\notin\mathcal A$.
\end{proof}

\begin{lemma}[The surviving ordered source occurrences]
    \label{thm:peps_affected_source_occurrences}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_mapOwner_affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.map_partyPair_affectedOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mapOwner_affectedOwner_eq_nil_iff}
    \leanok
    \uses{def:peps_affected_owner,def:peps_owner_sources}
    Under $g$ (Definition~\ref{def:peps_affected_owner}), the surviving source
    list $S_g$ (Definition~\ref{def:peps_owner_sources}) is obtained by
    filtering $S$ to occurrences $s=(p,q;U,V;\eta)$ with
    $\{p,q\}\cap\mathcal A\neq\varnothing$ and replacing $(p,q)$ by
    $(g(p),g(q))$. In particular, $S_g=\varnothing$ if and only if
    $p,q\notin\mathcal A$ for every $(p,q;U,V;\eta)\in S$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_owner_sources,thm:peps_owner_sources_internal,
        thm:peps_affected_owner_equality}
    Combine Definition~\ref{def:peps_owner_sources},
    Lemma~\ref{thm:peps_owner_sources_internal}, and the criterion
    $g(p)\neq g(q)\iff\{p,q\}\cap\mathcal A\neq\varnothing$ of
    Lemma~\ref{thm:peps_affected_owner_equality}.
\end{proof}

\begin{definition}[Retaining a complete operation at one grouped owner]
    \label{def:peps_grouped_block_map}
    \lean{TNLean.Lattice.PartyCircuit.Word.groupedBlockMap}
    \leanok
    \uses{def:peps_owner_layout,def:peps_party_layout}
    Let $f:P\to Q$, and suppose every register in $a$ and $b$ has owner image
    $q\in Q$. For $D:\mathcal H_a\to\mathcal H_b$ and a spectator list $\ell$,
    define $D^f_{\ell}:(a\ell)^f\to(b\ell)^f$ to apply $R_{f,b}DR_{f,a}^{-1}$
    as a local operation at $q$ while leaving $\ell^f$ untouched.
\end{definition}

\begin{lemma}[Exact action of an untouched aggregate]
    \label{thm:peps_grouped_block_map}
    \lean{TNLean.Lattice.PartyCircuit.Word.groupedBlockMap_spec}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_groupedBlockMap}
    \leanok
    \uses{def:peps_grouped_block_map}
    In Definition~\ref{def:peps_grouped_block_map},
    \begin{align}
        D^f_{\ell}R_{f,a\ell}
         & =R_{f,b\ell}C_{b,\ell}^{-1}
        (D\otimes I_{\mathcal H_\ell})C_{a,\ell}.
        \label{eq:peps_grouped_block_action}
    \end{align}
    If $\|D\|\leq1$, then $D^f_{\ell}$ is source-free and allowed.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_owner_concatenation,def:peps_grouped_block_map}
    By Definition~\ref{def:peps_grouped_block_map} and
    Lemma~\ref{thm:peps_owner_concatenation}, $R_{f,b}DR_{f,a}^{-1}$ cancels
    with the factor isometries in $R_{f,a\ell}$ and $R_{f,b\ell}$ to yield~(\ref{eq:peps_grouped_block_action}),
    and $\|R_{f,b}DR_{f,a}^{-1}\|=\|D\|\le 1$.
\end{proof}

\begin{lemma}[Grouping preserves the weighted gate sum]
    \label{thm:peps_grouped_gate_sum}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_sum_mapOwner_comp}
    \leanok
    \uses{def:peps_owner_composition}
    For a finite family of words $F_\xi:a\to b$ and $c_\xi\in\C$,
    \begin{align}
        \left(\sum_\xi c_\xi F_\xi^f\right)R_{f,a}
         & =R_{f,b}\left(\sum_\xi c_\xi F_\xi\right).
        \label{eq:peps_grouped_gate_sum}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_owner_composition}
    Sum $F_\xi^fR_{f,a}=R_{f,b}F_\xi$ (Lemma~\ref{thm:peps_owner_composition})
    with weights $c_\xi$.
\end{proof}

\begin{lemma}[The original expansion and aggregate norm after grouping]

\begin{proof}

\subsection{Preparing only the fixed sources}

\begin{definition}[The selected original source positions]
    \label{def:peps_selected_source_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.selectedSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.layout_selectedSources_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.freeSlotLayout}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_selectedSources}
    \leanok
    \uses{def:peps_source_slots}
    For an ordered pair-position list $E$ with halfspaces $U_e,V_e$ and a subset
    $E_{\rm free}\subseteq E$, the selected source list retains the positions
    $e\in E_{\rm free}$ in their original order with their endpoint parties and
    spaces $U_e,V_e$. Denote its memory by $\mathcal S_{\rm free}$ and the
    full source memory by $\mathcal S$ (Definition~\ref{def:peps_source_slots}).
\end{definition}

\begin{definition}[Selective source preparation]
    \label{def:peps_selective_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareFreeSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.fillSourceVectors}
    \leanok
    \uses{def:peps_selected_source_positions}
    Given $\eta_e\in U_e\otimes V_e$ for $e\in E\setminus E_{\rm free}$, define
    $\Phi_\eta:\mathcal S_{\rm free}\otimes\mathcal H_\ell\to\mathcal S\otimes\mathcal H_\ell$
    by preparing $\eta_e$ at $e\in E\setminus E_{\rm free}$ and retaining all
    free registers $e\in E_{\rm free}$ in order. For $\zeta_e\in U_e\otimes V_e$
    ($e\in E_{\rm free}$), let $Q^{\rm free}_{\zeta,\mathcal H_\ell}$ prepare
    $\zeta_e$ at $e\in E_{\rm free}$, and define the completed family $\theta$ by
    $\theta_e=\zeta_e$ for $e\in E_{\rm free}$ and $\theta_e=\eta_e$ for
    $e\in E\setminus E_{\rm free}$.
\end{definition}

\begin{lemma}[Only the fixed sources are prepared]
    \label{thm:peps_selective_preparation_sources}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.isAllowed_prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.sources_prepareSelected}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.mem_sources_prepareSelected}
    \leanok
    \uses{def:peps_selective_preparation}
    The source list of $\Phi_\eta$ (Definition~\ref{def:peps_selective_preparation})
    is $((e;U_e,V_e;\eta_e))_{e\in E\setminus E_{\rm free}}$ in the order of
    $E$. If $\|\eta_e\|=1$ for all $e\in E\setminus E_{\rm free}$, then
    $\Phi_\eta$ is allowed.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_selective_preparation,def:peps_selected_source_positions}
    By induction on $E$ via Definition~\ref{def:peps_selected_source_positions}
    and Definition~\ref{def:peps_selective_preparation}, positions
    $e\in E_{\rm free}$ frame the recursive word without adding sources, whereas
    positions $e\in E\setminus E_{\rm free}$ append the normalized preparation
    of $\eta_e$.
\end{proof}

\begin{lemma}[Uniform recovery of every completed preparation]
    \label{thm:peps_selective_preparation_recovery}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_prepareSelected_prepareFreeSlots}
    \leanok
    \uses{def:peps_selective_preparation}
    In the notation of Definition~\ref{def:peps_selective_preparation}, for any
    free family $\zeta$,
    \begin{align}
        \Phi_\eta Q^{\rm free}_{\zeta,\mathcal H_\ell}
         & =Q_{\theta,\mathcal H_\ell}.
        \label{eq:peps_selective_recovery}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_selective_preparation,thm:peps_source_preparation}
    Induct on $E$ using Definition~\ref{def:peps_selective_preparation} and
    Lemma~\ref{thm:peps_source_preparation}: at $e\in E_{\rm free}$, preparing
    $\zeta_e$ commutes with $\Phi_\eta$ on the remaining positions, while at
    $e\in E\setminus E_{\rm free}$ both sides prepare $\eta_e$.
\end{proof}

\begin{lemma}[Two contractions independent of the free source vectors]
    \label{thm:peps_selective_partition}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_selective_partition}
    \leanok
    \uses{def:peps_selective_preparation,def:peps_party_partition}
    Let $f:P\to\{0,1\}$ be a party partition (Definition~\ref{def:peps_party_partition}).
    Suppose every fixed position $e=(p_e,q_e)\in E\setminus E_{\rm free}$
    satisfies $f(p_e)=f(q_e)$ and $\|\eta_e\|=1$, and let
    $T:\mathcal S\otimes\mathcal H_\ell\to\mathcal H_{\ell'}$ be a source-free
    allowed word. Then there exist source-free allowed words $A_+,A_-$ with
    $\|A_\pm\|\le 1$, independent of $\zeta$, such that for $L$ the free-source
    register list followed by $\ell$,
    \begin{align}
        J_{f,\ell'}T\Phi_\eta
         & =(A_+\otimes A_-)J_{f,L},
        \label{eq:peps_selective_partition} \\
        J_{f,\ell'}TQ_{\theta,\mathcal H_\ell}
         & =(A_+\otimes A_-)J_{f,L}
        Q^{\rm free}_{\zeta,\mathcal H_\ell}.
        \label{eq:peps_selective_all_vectors}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_selective_preparation_sources,
        thm:peps_internal_source_factorization,
        thm:peps_selective_preparation_recovery}
    By Lemma~\ref{thm:peps_selective_preparation_sources}, all sources of
    $T\Phi_\eta$ belong to $E\setminus E_{\rm free}$ and are internal to $f$, so
    Lemma~\ref{thm:peps_internal_source_factorization} gives
    $A_+,A_-$ with $\|A_\pm\|\le 1$ satisfying~(\ref{eq:peps_selective_partition});
    right-multiplying by $Q^{\rm free}_{\zeta,\mathcal H_\ell}$ and applying~(\ref{eq:peps_selective_recovery})
    gives~(\ref{eq:peps_selective_all_vectors}).
\end{proof}

\subsection{Operations beside untouched registers}

\begin{definition}[Extension by spectator registers]
    \label{def:peps_append_spectators}
    \lean{TNLean.Lattice.PartyCircuit.Word.appendTail}
    \leanok
    \uses{def:peps_party_layout,thm:peps_register_exchange}
    For a word $W:a\to b$ and a spectator list $\ell$, define
    $W_{\ell}=\mathrm{Exch}_{\ell,b}\circ W_{[\ell]}\circ\mathrm{Exch}_{a,\ell}:a\ell\to b\ell$,
    which exchanges $\ell$ in front of $a$, applies $W$ framed by $\ell$, and
    exchanges $\ell$ back after $b$ (Lemma~\ref{thm:peps_register_exchange}).
\end{definition}

\begin{lemma}[Exact spectator extension]
    \label{thm:peps_append_spectators}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_appendTail}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_appendTail}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_appendTail}
    \leanok
    \uses{def:peps_append_spectators}
    The extension $W_{\ell}$ (Definition~\ref{def:peps_append_spectators}) has
    the same ordered source list as $W$, is allowed whenever $W$ is allowed, and
    satisfies
    \begin{align}
        W_{\ell}C_{a,\ell}^{-1}
         & =C_{b,\ell}^{-1}(W\otimes I_{\mathcal H_\ell}).
        \label{eq:peps_append_spectators}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_register_exchange,thm:peps_source_preparation}
    By Lemma~\ref{thm:peps_register_exchange} and
    Lemma~\ref{thm:peps_source_preparation}, on $x\otimes y\in\mathcal H_a\otimes\mathcal H_\ell$
    the three steps of $W_{\ell}$ send $x\otimes y\mapsto y\otimes x\mapsto y\otimes Wx\mapsto Wx\otimes y$,
    giving~(\ref{eq:peps_append_spectators}) while preserving the source list
    and allowedness of $W$.
\end{proof}

\begin{lemma}[Relabelling a composition beside spectators]
    \label{thm:peps_owner_spectator_extension}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapOwner_appendTail_heq}
    \leanok
    \uses{def:peps_append_spectators,def:peps_owner_composition,
        def:peps_owner_layout}
    For an owner map $g:P\to Q$, a chronological word $W:a\to b$, and an
    ordered spectator list $t$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, under the
    concatenation identifications $(at)^g=a^g t^g$ and $(bt)^g=b^g t^g$,
    $(W_t)^g=(W^g)_{t^g}$.
    % Source: 04-compression.tex, lines 233--267 and 409--434.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_append_spectators,def:peps_owner_composition,
        def:peps_owner_layout,thm:peps_register_exchange}
    By Definition~\ref{def:peps_owner_composition},
    Definition~\ref{def:peps_owner_layout}, and
    Lemma~\ref{thm:peps_register_exchange}, for any register $r$ and block
    $A$,
    \begin{align}
        (\mathrm{frame}_r\,w)^g
         & =\mathrm{frame}_{r^g}(w^g),
        \qquad
        (\mathrm{Exch}_{r,A})^g=\mathrm{Exch}_{r^g,A^g},
        \label{eq:peps_owner_frame_exchange}
    \end{align}
    so induction gives $(\mathrm{Exch}_{A,B})^g=\mathrm{Exch}_{A^g,B^g}$ and
    $(W_{[t]})^g=(W^g)_{[t^g]}$. By Definition~\ref{def:peps_append_spectators},
    \begin{align}
        (W_t)^g
         & =\bigl(\mathrm{Exch}_{t,b}\circ W_{[t]}\circ\mathrm{Exch}_{a,t}\bigr)^g
        =\mathrm{Exch}_{t^g,b^g}\circ(W^g)_{[t^g]}\circ\mathrm{Exch}_{a^g,t^g}
        =(W^g)_{t^g}.
    \end{align}
\end{proof}
