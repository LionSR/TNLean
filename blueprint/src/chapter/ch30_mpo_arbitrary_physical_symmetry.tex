\section{Symmetry and endpoint projections in arbitrary alphabets}
\label{sec:mpo_arbitrary_physical_symmetry}

We keep the notation of Section~\ref{sec:mpo_arbitrary_physical_path}:
$A_p=F_pB_p$ is the physical polar factorization of $A_p$, with
$F_p^\dagger F_p=\Pi_p$ the projection onto the support of the positive
factor, $J$ includes $\C^{D^2}$ into
$\mathcal H=\C^{D^2}\oplus\C^{d_0}\oplus\C^{d_1}$, $E_p$ are the full
endpoint embeddings, and $\widetilde A(\gamma)=J\mathcal B(\gamma)$.
For a matrix $G:\C^{d}\to\C^{e}$ and an operator tensor $T$ with
physical space $\C^d$, write $GTG^\dagger$ for the operator tensor with
entries $(GTG^\dagger)^{xy}=\sum_{u,v}G_{xu}\overline{G_{yv}}\,T^{uv}$.

\cite[Section~5]{GarreRubio2023MPOSym} first restricts each local
Hilbert space to the span of the tensor. Here that restriction is
derived from the symmetry: an exact action decomposition already forces
the operator to preserve the physical support.

\begin{lemma}[Physical coordinate changes of the action and fusion]
    \label{thm:mpo_physical_basis_action}
    \lean{MPOTensor.actTensor_apply_finProdFinEquiv,
        MPOTensor.actTensor_changePhysicalBasis,
        MPOTensor.physicalSlice_mulTensor,
        MPOTensor.mulTensor_changePhysicalBasis_of_slice,
        MPOTensor.mulTensor_changePhysicalBasis,
        MPSTensor.IsBiorthogonalDecomposition.rotatePhysical,
        MPSTensor.IsBiorthogonalDecomposition.actTensor_changePhysicalBasis,
        MPSTensor.IsBiorthogonalDecomposition.toMPSTensor_changePhysicalBasis,
        MPSTensor.IsBiorthogonalDecomposition.mulTensor_changePhysicalBasis}
    \leanok
    \uses{def:glm_boundary_decomposition}
    For every matrix $G:\C^d\to\C^e$, operator tensor $T$, and tensor $B$
    with physical space $\C^e$,
    \begin{align}
        \bigl((GTG^\dagger)\cdot B\bigr)^x
         & =\sum_uG_{xu}\bigl(T\cdot(G^\dagger B)\bigr)^u.
        \label{eq:mpo_physical_basis_action}
    \end{align}
    Physical mixing preserves exact biorthogonal decompositions with the
    same analysis and synthesis maps. Consequently an action
    decomposition of $T$ on $G^\dagger B$ gives one of $GTG^\dagger$ on
    $B$ with targets mixed by $G$, and a fusion decomposition of $T$
    into tensors $T_c$ gives one of $GTG^\dagger$ into $GT_cG^\dagger$.
    If every physical slice $s$ of $S$ satisfies
    $G^\dagger GsG^\dagger=sG^\dagger$, then
    $(GTG^\dagger)(GSG^\dagger)=G(TS)G^\dagger$; this holds whenever
    $G^\dagger G=\Id$.
\end{lemma}
\begin{proof}
    \leanok
    Expanding one coefficient of either side of
    (\ref{eq:mpo_physical_basis_action}) gives the same triple sum.
    Mixing the letters $B^x=\sum_cW_cA_c^xV_c$ by $G$ commutes with the
    left and right multiplications. For products, the physical slice of
    a stacked product is the product of the slices, so the intermediate
    factor $G^\dagger G$ is absorbed by the stated identity.
\end{proof}

\begin{lemma}[Support invariance from an exact action]
    \label{thm:mpo_support_invariance}
    \lean{MPOTensor.physicalSlice_mul_eq_of_isBiorthogonalDecomposition,
        MPSTensor.rotatePhysical_conjTranspose_polarIsoMatrix,
        MPSTensor.rotatePhysical_polarIsoMatrix_mul_conjTranspose,
        MPSTensor.linearIndependent_polarPosTensor,
        MPSTensor.polarIsoMatrix_mul_physicalSlice_of_isBiorthogonalDecomposition}
    \leanok
    \uses{thm:mpo_physical_basis_action,thm:mpo_polar_support_injective,
        thm:ldpolar_tensor_decomposition}
    Let $U:\C^e\to\C^d$, let $B$ have linearly independent letters, and
    suppose $T\cdot(UB)=\sum_cW_cA_cV_c$ exactly, where physical mixing
    by $G$ fixes every target $A_c$. Then every physical slice $s$ of
    $T$ satisfies $GsU=sU$. In particular, for every tensor $A$,
    $F^\dagger A=B$ and $FF^\dagger A=A$; if $A$ is one-site injective,
    the $D^2$ letters of $B$ are linearly independent, and an exact action
    of $T$ on $A$ into copies of $A$ gives $FF^\dagger sF=sF$.
\end{lemma}
\begin{proof}
    \leanok
    Mixing by $G$ fixes the right side of the decomposition, hence also
    $T\cdot(UB)$. Comparing coefficients gives
    $\sum_k(GsU)_{xk}B^k=\sum_k(sU)_{xk}B^k$, and linear independence
    identifies the coefficients. For the polar factorization,
    $F^\dagger A=F^\dagger FB=\Pi B=B$. Injectivity of $B$ means that
    its $D^2$ letters span $\MN D$, which has dimension $D^2$.
\end{proof}

\begin{definition}[The common operator in arbitrary alphabets]
    \label{def:mpo_arbitrary_mixed_operator}
    \lean{MPSTensor.polarSupportMPO,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedMPO}
    \leanok
    \uses{def:mpo_mixed_action_tensor,def:mpo_arbitrary_physical_tensor}
    For an operator tensor $T_p$ with physical space $\C^{d_p}$, put
    $S_p=F_p^\dagger T_pF_p$, an operator tensor with physical space
    $\C^{D_p^2}$. Given action maps $V_{p,\mu},W_{p,\mu}$, let
    $T_{\mathrm{mix}}$ be the mixed operator of
    Definition~\ref{def:mpo_mixed_action_tensor} built from $S_0,S_1$,
    and define $\widetilde T=JT_{\mathrm{mix}}J^\dagger$.
\end{definition}

\begin{lemma}[Exact covariance of the arbitrary-alphabet path]
    \label{thm:mpo_arbitrary_mixed_action}
    \lean{MPSTensor.isBiorthogonalDecomposition_polarSupportMPO,
        MPSTensor.MPOSymmetry.isBiorthogonalDecomposition_arbitraryPhysicalMixedInterpolation,
        MPSTensor.MPOSymmetry.isBoundaryCompatible_arbitraryPhysicalMixedInterpolation,
        MPSTensor.MPOSymmetry.mpo_mulVec_arbitraryPhysicalMixedInterpolation,
        MPSTensor.MPOSymmetry.mpo_mulVec_mpv_commonPhysicalEmbeddingLeft,
        MPSTensor.MPOSymmetry.mpo_mulVec_mpv_commonPhysicalEmbeddingRight}
    \leanok
    \uses{def:mpo_arbitrary_mixed_operator,thm:mpo_mixed_action_exact,
        thm:mpo_support_invariance,thm:mpo_arbitrary_endpoint_vectors}
    Suppose that $(T_p\cdot A_p)^x=\sum_{\mu=0}^{m-1}W_{p,\mu}A_p^xV_{p,\mu}$
    and $V_{p,\mu}W_{p,\nu}=\delta_{\mu\nu}\Id$ for $p=0,1$. Then
    $S_p$ acts on $B_p$ by the same maps, and for every $\gamma\in\R$,
    \begin{align}
        (\widetilde T\cdot\widetilde A(\gamma))^x
         & =\sum_{\mu=0}^{m-1}W_\mu\widetilde A(\gamma)^xV_\mu,
        \label{eq:mpo_arbitrary_mixed_action}
    \end{align}
    with the mixed maps $V_\mu,W_\mu$ of
    Definition~\ref{def:mpo_mixed_action_maps}. Hence $\widetilde T$ is
    compatible with the arbitrary-boundary states of $\widetilde A(\gamma)$
    and, for every $N>0$,
    $O^{(N)}_{\widetilde T}\ket{V^{(N)}(\widetilde A(\gamma))}
        =m\ket{V^{(N)}(\widetilde A(\gamma))}$.
    If $A_p$ is injective, then also
    $O^{(N)}_{\widetilde T}\ket{V^{(N)}(E_pA_p)}=m\ket{V^{(N)}(E_pA_p)}$.
\end{lemma}
\begin{proof}
    \leanok
    Since $A_p=(F_p^\dagger)^\dagger B_p$ and $F_p^\dagger A_p=B_p$,
    Lemma~\ref{thm:mpo_physical_basis_action} transfers the action to
    $S_p$ and $B_p$ without any injectivity. Lemma~\ref{thm:mpo_mixed_action_exact}
    gives the action of $T_{\mathrm{mix}}$ on $\mathcal B(\gamma)$.
    Since $J^\dagger\widetilde A(\gamma)=\mathcal B(\gamma)$, the same
    lemma, applied with $G=J$, gives (\ref{eq:mpo_arbitrary_mixed_action}).
    Biorthogonality gives the boundary transport and, with identity
    boundaries, the eigenvalue $m$. At $\gamma=p$ the periodic vectors
    are those of $E_pA_p$ by Lemma~\ref{thm:mpo_arbitrary_endpoint_vectors}.
\end{proof}

\begin{lemma}[Fusion of the common operators]
    \label{thm:mpo_arbitrary_mixed_fusion}
    \lean{MPSTensor.IsBiorthogonalDecomposition.mulTensor_polarSupportMPO,
        MPSTensor.MPOSymmetry.isBiorthogonalDecomposition_arbitraryPhysicalMixedMPO,
        MPSTensor.MPOSymmetry.isMPOFusionAlgebra_arbitraryPhysicalMixedMPO,
        MPSTensor.MPOSymmetry.isMPOSymmetric_arbitraryPhysicalMixedInterpolation}
    \leanok
    \uses{thm:mpo_arbitrary_mixed_action,thm:mpo_mixed_fusion_exact,
        thm:mpo_mixed_fusion_periodic,thm:mpo_support_invariance,
        def:mpo_symmetric_family,thm:mpo_physical_basis_action}
    Let $(T_{p,a})_a$ be operator families with exact biorthogonal fusion
    into copies $T_{p,c}$, with multiplicities $N_{ab}^c$, and exact
    actions on $A_p$ with multiplicities $m_a$, for $p=0,1$. Suppose
    $D_0,D_1>0$, $A_0,A_1$ are one-site injective, and the chosen raw
    $L$ matrices agree for every pair $a,b$. Then the common operators
    $\widetilde T_a$ fuse exactly with the mixed fusion maps, and
    $O^{(N)}_{\widetilde T_a}O^{(N)}_{\widetilde T_b}
        =\sum_cN_{ab}^cO^{(N)}_{\widetilde T_c}$ for every $N>0$.
    Without the fusion and injectivity hypotheses, $\widetilde A(\gamma)$
    is symmetric under $(\widetilde T_a)_a$ with eigenvalues $m_a$ for
    every $\gamma\in\R$.
\end{lemma}
\begin{proof}
    \leanok
    By Lemma~\ref{thm:mpo_support_invariance}, the action of $T_{p,b}$
    on $A_p$ gives $F_pF_p^\dagger sF_p=sF_p$ on every physical slice
    of $T_{p,b}$. Lemma~\ref{thm:mpo_physical_basis_action} then
    compresses the fusion of $T_{p,a}T_{p,b}$ to $S_{p,a}S_{p,b}$.
    Lemma~\ref{thm:mpo_mixed_fusion_exact} gives exact fusion of the
    square-alphabet mixed operators, and $J^\dagger J=\Id$ transports it
    to $\widetilde T_a\widetilde T_b$. Taking identity boundaries gives
    the periodic fusion rule.
\end{proof}

The analysis and synthesis maps of
(\ref{eq:mpo_arbitrary_mixed_action}) are those of the square-alphabet
construction, so Lemma~\ref{thm:mpo_mixed_l_symbols} identifies their
$L$-symbols with the common endpoint value along the whole path. As in
Lemma~\ref{thm:mpo_mixed_fusion_periodic}, the comparison uses equal
chosen raw matrices rather than equivalent classes; see
\cite{gap:glm23_multiplicity_l_indices}.

\begin{lemma}[Canonical endpoint ground projections]
    \label{thm:mpo_arbitrary_ground_projection}
    \lean{MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_groundSpace_eq_range,
        MPSTensor.MPOSymmetry.continuous_arbitraryPhysicalMixed_groundSpace_starProjection,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixed_groundSpace_starProjection_endpoint,
        MPSTensor.MPOSymmetry.tendsto_arbitraryPhysicalMixed_groundSpace_starProjection,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_groundSpace_starProjection}
    \leanok
    \uses{thm:mpo_arbitrary_periodic_ground_line,
        thm:mpo_arbitrary_endpoint_attachment_gap,
        thm:mpo_mixed_periodic_line_continuity}
    Suppose $D_0,D_1>0$ and $A_0,A_1$ are one-site injective, and fix
    $N\geq2$. Let $P_\gamma$ be the orthogonal projection onto
    $\ker\widetilde H_{\gamma,N}$ and let $Q_p$ be the orthogonal
    projection onto the ground space of the canonical parent Hamiltonian
    $K_{p,N}$ of $E_pA_p$. Then $\gamma\mapsto P_\gamma$ is continuous on
    $[0,1]$, $P_p=Q_p$, and hence $P_\gamma\to Q_p$ as $\gamma\to p$
    within $[0,1]$. Along the affine attachments of
    Definition~\ref{def:mpo_arbitrary_endpoint_attachment}, the ground
    projection equals $Q_p$ for every $t\geq0$.
\end{lemma}
\begin{proof}
    \leanok
    The ground space of $\widetilde H_{\gamma,N}$ is the line of
    $\ket{V^{(N)}(\widetilde A(\gamma))}$, the range of a continuous
    family of injective maps from $\C$. Its orthogonal projection is
    therefore continuous. At $\gamma=p$ the kernels of $\widetilde
        H_{p,N}$ and $K_{p,N}$ agree by
    Theorem~\ref{thm:mpo_arbitrary_endpoint_attachment_gap}, and so do
    the kernels along the attachments.
\end{proof}

The source obtains the endpoint limit at the level of the local supports
by orthonormalizing a continuous basis. The periodic statement above
identifies the limit with the ground projection of the canonical parent
itself.

\cite[Section~5]{GarreRubio2023MPOSym} calls the interpolation ``a
continuous and symmetric embedding path that connects both MPSs and
therefore, their parent Hamiltonians'', and takes for $H(\gamma)$ the
canonical parent with local terms ``$h_\gamma = \Id - \Pi_{S_\gamma}$''.
The source does not show that these canonical parents commute with the
common operators, and they need not. The two-site support of an injective
tensor is invariant under an exactly acting symmetry. Its orthogonal
complement need not be invariant when the symmetry is not unitary. An
involutive non-unitary on-site symmetry of the tensor
$(E_{11},E_{12},E_{21},\Id_2)$ satisfies every action, fusion and
equal-$L$-matrix hypothesis above, yet its canonical parent does not
commute with it; see \cite{gap:glm23_mixed_path_parent_commutation}.
For unitary on-site endpoint symmetries the outcome depends on the
virtual gauge of the endpoint tensors, which leaves the endpoint states
and their parents unchanged but changes the interpolation and the common
operator. Let $\mathbb Z_2$ act at both endpoints by one unitary on-site
representation, on two different injective tensors with equal $L$-data.
If both tensors are in canonical form, the two-site boundary range of the
common operator is closed under the adjoint and the canonical path
commutes with it. If one tensor is in a gauge whose action gauge is not
proportional to a Hermitian matrix, the range is not closed under the
adjoint and the canonical parent does not commute with the periodic
common operator; see the same note.
The source's claim of a symmetric canonical path is therefore not
formalized. The following restricted statement assumes adjoint closure
of the two-site boundary range of the common operator.

\begin{lemma}[Commutation of the path under adjoint closure]
    \label{thm:mpo_arbitrary_parent_commutation_adjoint_closed}
    \lean{MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_commute_mpoWithBoundary_of_mem_Ioo,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_commute_of_forall_mem_Ioo,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_commute_mpoWithBoundary,
        MPSTensor.MPOSymmetry.hamiltonian_arbitraryPhysicalMixedInteraction_commute_mpoWithBoundary,
        MPSTensor.MPOSymmetry.hamiltonian_arbitraryPhysicalMixedInteraction_commute_mpo}
    \leanok
    \uses{thm:mpo_arbitrary_mixed_action,
        thm:mpo_arbitrary_interior_parent,
        thm:mpo_arbitrary_interaction_continuity,
        thm:glm_boundary_parent_local,thm:glm_boundary_parent_periodic,
        thm:glm_boundary_embed_commutation}
    Assume the hypotheses of Lemma~\ref{thm:mpo_arbitrary_mixed_action},
    that $D_0,D_1>0$, and that $A_0,A_1$ are one-site injective. Assume
    in addition, as a restriction not made by the source, that
    $\mathcal R_2(\widetilde T)^*=\mathcal R_2(\widetilde T)$. Then for
    every $\gamma\in[0,1]$ and every $X$,
    $[\widetilde h_\gamma,O^2_{\widetilde T,X}]=0$. Consequently, for
    every $N\geq2$ and every $X\in\mathcal C(\widetilde T)$,
    $[\widetilde H_{\gamma,N},O^N_{\widetilde T,X}]=0$, and in particular
    $[\widetilde H_{\gamma,N},O^{(N)}_{\widetilde T}]=0$.
\end{lemma}
\begin{proof}
    \leanok
    For $0<\gamma<1$, Lemma~\ref{thm:mpo_arbitrary_interior_parent}
    identifies $\widetilde h_\gamma$ with the canonical two-site parent
    of $\widetilde A(\gamma)$, which is compatible with $\widetilde T$ by
    Lemma~\ref{thm:mpo_arbitrary_mixed_action}. Under adjoint closure,
    Lemma~\ref{thm:glm_boundary_parent_local} gives the commutation.
    The set of $\gamma$ at which $\widetilde h_\gamma$ commutes with a
    fixed operator is closed, because $\widetilde h_\gamma$ is continuous
    by Lemma~\ref{thm:mpo_arbitrary_interaction_continuity}. It contains
    $(0,1)$, hence it contains $[0,1]$. Embedding each cyclic translate as
    in Lemma~\ref{thm:glm_boundary_embed_commutation} and summing gives
    the periodic statements; the identity boundary lies in
    $\mathcal C(\widetilde T)$.
\end{proof}

For on-site endpoint symmetries with Hermitian on-site matrices, which
include unitary representations of $\mathbb Z_2$, adjoint closure follows from
a condition on the action gauges. Let the operator bonds of $T_0,T_1$ be
one-dimensional, so that $T_p$ is given by a matrix $u_p$, let the action
multiplicity be $m=1$, and write $Y_p$ for the synthesis map $W_{p,0}$ read
as a square matrix and $V_p$ for $V_{p,0}$, so that $V_pY_p=\Id$.

    \leanok
    \uses{def:mpo_arbitrary_mixed_operator,def:mpo_mixed_action_tensor,

        thm:glm_boundary_similarity}
    Restrict the hypotheses of Lemma~\ref{thm:mpo_arbitrary_mixed_action}
    to on-site endpoint operators with $m=1$ as above. Suppose that $u_0$
    and $u_1$ are Hermitian, and that $c_pY_p$ is Hermitian for some
    $c_p\neq0$, for $p=0,1$. Then for every $N$ and every $X$ there is a
    $X'$ with $(O^N_{\widetilde T,X})^*=O^N_{\widetilde T,X'}$. In
    particular $\mathcal R_2(\widetilde T)^*=\mathcal R_2(\widetilde T)$.
\end{lemma}
\begin{proof}
    \leanok
    Write $K^\sharp$ for the tensor with letters
    $(K^\sharp)^{ij}_{\beta\alpha}=\overline{K^{ji}_{\beta\alpha}}$. Then
    $(O^N_{K,X})^*=O^N_{K^\sharp,\bar X}$, where $\bar X$ is the entrywise
    conjugate, and $K^\sharp$ commutes with every change of physical
    coordinates. Since $u_p$ is Hermitian, $S_p^\sharp=S_p$. Since
    $c_pY_p$ is Hermitian, $Y_p^*=\omega_pY_p$ with
    $\omega_p=c_p/\bar c_p$. From $V_pY_p=\Id$ also $Y_pV_p=\Id$, hence
    $V_p^*Y_p^*=\Id$ and $V_p^*=\omega_p^{-1}V_p$. The crossed blocks
    $W_0^{\mathsf T}\otimes V_1$ and $W_1^{\mathsf T}\otimes V_0$ of the
    mixed operator are therefore multiplied by $t=\omega_0\omega_1^{-1}$
    and $t^{-1}$ under ${}^\sharp$, and the diagonal blocks are fixed. With
    $g=\operatorname{diag}(1,t^{-1})$ on the bond $\C^2$, this says
    $(\widetilde T^\sharp)^{ij}=g\,\widetilde T^{ij}g^{-1}$, so
    $O^N_{\widetilde T^\sharp,Z}=O^N_{\widetilde T,g^{-1}Zg}$ for every $Z$.
    Take $X'=g^{-1}\bar Xg$.
\end{proof}

Unitary endpoint symmetries supply the gauge condition when the endpoint
tensors are in canonical form.

\begin{lemma}[Hermitian action gauges in canonical form]
    \label{thm:mpo_arbitrary_canonical_hermitian_gauge}
    \lean{Matrix.isHermitian_of_mem_unitaryGroup_of_mul_self_eq_one,
        MPSTensor.MPOSymmetry.sum_smul_eq_bondOneActionGauge_mul_mul,
        MPSTensor.MPOSymmetry.exists_isHermitian_smul_bondOneActionGauge}
    \leanok
    \uses{def:injective,thm:virtual_gauge_fixes_boundary_state,
        thm:mpo_arbitrary_hermitian_gauge_adjoint_closure}
    Let $A$ be injective with $\E_A(\Id)=\Id$ and with a positive definite
    $\Lambda$ satisfying $\E_A^\dagger(\Lambda)=\Lambda$, and let a unitary
    on-site matrix $u$ with $u^2=\Id$ act on $A$ as above, with action
    gauge $Y$. Then $u$ is Hermitian and $cY$ is Hermitian for some
    $c\neq0$.
\end{lemma}
\begin{proof}
    \leanok
    From $u^*u=\Id$ and $u^2=\Id$, $u^*=u^*u^2=u$. The action identity reads
    $\sum_ju_{ij}A^j=YA^iY^{-1}$. By
    Lemma~\ref{thm:virtual_gauge_fixes_boundary_state}, $Y=cU$ with $c\neq0$
    and $U$ unitary. Applying the action identity twice and using $u^2=\Id$
    gives $Y^2A^iY^{-2}=A^i$, so $Y^2$ commutes with the span of the
    letters, which is all of $\MN D$; hence $Y^2$ and $U^2$ are scalar,
    $U^2=\ell\Id$. Then $\ell U^*=U^*U^2=U$, and taking adjoints gives
    $\ell\bar\ell=1$. Choose $s\neq0$ with $\bar s=s\ell$: $s=1+\bar\ell$ if
    $\ell\neq-1$ and $s=i$ otherwise. Then $(sU)^*=\bar sU^*=s\ell U^*=sU$,
    so $(s/c)Y=sU$ is Hermitian.
\end{proof}

For a group the hypothesis of
Lemma~\ref{thm:mpo_arbitrary_parent_commutation_adjoint_closed}, imposed

of an operator of the label $g$ is instead an operator of the label
$g^{-1}$. Let a group $G$ act at both endpoints by unitary on-site
representations $g\mapsto u_{p,g}$, with on-site operator tensors
$T_{p,g}$, multiplicity $m=1$, action maps $V_{p,g},W_{p,g}$ and action
gauges $Y_{p,g}$ as above, and write $\widetilde T_g$ for the common
operator of the label $g$.

    This concerns the common operator $T_a$ of
    \cite[Section~5]{GarreRubio2023MPOSym}, with its crossed blocks
    $T^{0,1}_a$ and $T^{1,0}_a$. Suppose that $A_0,A_1$ are one-site
    injective and, as a restriction not made by the source
    \cite{gap:glm23_mixed_path_parent_commutation}, that every action
    gauge $Y_{p,g}$ is a nonzero multiple of a unitary matrix. Then for
    every $g\in G$, every $N$ and every $X$ there is an $X'$ with
    $(O^N_{\widetilde T_g,X})^*=O^N_{\widetilde T_{g^{-1}},X'}$. Hence the
    union $\bigcup_g\mathcal R_N(\widetilde T_g)$ is closed under the
    adjoint.
\end{lemma}
\begin{proof}
    \leanok
    Since $u_{p,g}$ is unitary and $g\mapsto u_{p,g}$ is a homomorphism,
    $u_{p,g}^*=u_{p,g}^{-1}=u_{p,g^{-1}}$, so $S_{p,g}^\sharp=S_{p,g^{-1}}$
    in the notation of
    Lemma~\ref{thm:mpo_arbitrary_hermitian_gauge_adjoint_closure}. Applying
    the action identities of $g^{-1}$ and $g$ in turn, conjugation by
    $Y_{p,g}Y_{p,g^{-1}}$ fixes every letter of $A_p$; since the letters
    span $\MN{D_p}$, $Y_{p,g}Y_{p,g^{-1}}=\kappa\Id$, and $\kappa\neq0$
    because both factors are invertible. Writing $Y_{p,g}=cU$ with $U$
    unitary, $Y_{p,g}^*=\bar c\,U^{-1}=|c|^2Y_{p,g}^{-1}=
        (|c|^2/\kappa)\,Y_{p,g^{-1}}$. So $Y_{p,g}^*=\omega_pY_{p,g^{-1}}$
    with $\omega_p\neq0$, and from $V_{p,g}Y_{p,g}=\Id$ also
    $V_{p,g}^*=\omega_p^{-1}V_{p,g^{-1}}$. As in the proof of
    Lemma~\ref{thm:mpo_arbitrary_hermitian_gauge_adjoint_closure}, the
    crossed blocks of $\widetilde T_g^\sharp$ are those of
    $\widetilde T_{g^{-1}}$ multiplied by $t=\omega_0\omega_1^{-1}$ and
    $t^{-1}$, and the diagonal blocks agree. With
    $\Delta=\operatorname{diag}(1,t^{-1})$ on the bond,
    $(\widetilde T_g^\sharp)^{ij}=\Delta\,\widetilde T_{g^{-1}}^{ij}\Delta^{-1}$,
    so $(O^N_{\widetilde T_g,X})^*=O^N_{\widetilde T_g^\sharp,\bar X}=
        O^N_{\widetilde T_{g^{-1}},\Delta^{-1}\bar X\Delta}$.
\end{proof}

        thm:glm_boundary_embed_commutation}
    Assume the hypotheses of Lemma~\ref{thm:mpo_arbitrary_mixed_action}
    for every label, restricted to unitary on-site endpoint representations

    $A_0,A_1$ are one-site injective. Assume, as a restriction not made by

    action gauge $Y_{p,g}$ is a nonzero multiple of a unitary matrix. The
    canonical parent $\widetilde h_\gamma=\Id-\Pi_{S_\gamma}$ is that of
    \cite[Section~5]{GarreRubio2023MPOSym}. Then for every $g\in G$,

    $X\in\mathcal C(\widetilde T_g)$,

    hypothesis holds when both endpoint tensors satisfy
    $\E_{A_p}(\Id)=\Id$ and have positive definite fixed points of
    $\E_{A_p}^\dagger$, by
    \cite[Lemma~1 and Theorem~2]{PerezGarcia2008StringOrder}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:mpo_arbitrary_parent_commutation_adjoint_closed,
        thm:mpo_arbitrary_canonical_hermitian_gauge,
        thm:virtual_gauge_fixes_boundary_state,def:mpdo_commuting_boundary}
    For $0<\gamma<1$, Lemma~\ref{thm:mpo_arbitrary_interior_parent}
    identifies $\widetilde h_\gamma$ with the canonical two-site parent of
    $\widetilde A(\gamma)$, which is compatible with every $\widetilde T_g$
    by Lemma~\ref{thm:mpo_arbitrary_mixed_action}. By
    Lemma~\ref{thm:mpo_arbitrary_unitary_gauge_union_closure} the union of
    the two-site boundary ranges is closed under the adjoint, so
    Lemma~\ref{thm:glm_boundary_parent_local_family} gives the commutation.

The source states its symmetrization of parent terms for operators that
represent a weak Hopf algebra:
\begin{quote}
    ``The previous symmetrization can be generalized to MPO representations
    of ${C}^*$-weak Hopf algebras''
\end{quote}
\cite[Section~5 and Appendix~B]{GarreRubio2023MPOSym}. Section~5 does not
construct such a representation for the common operator $\widetilde T$,
and the examples of \cite{gap:glm23_mixed_path_parent_commutation}, in
which the two-site boundary range is not closed under the adjoint, show
that none need exist. The following restricted statement assumes
one.

\begin{lemma}[Commutation of the path under a weak-Hopf representation]
    \label{thm:mpo_arbitrary_parent_commutation_weak_hopf}
    \lean{MPSTensor.MPOSymmetry.exists_conjTranspose_mpoWithBoundary_arbitraryPhysicalMixedMPO_eq_of_weakHopf,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_commute_mpoWithBoundary_of_weakHopf,
        MPSTensor.MPOSymmetry.hamiltonian_arbitraryPhysicalMixedInteraction_commute_mpoWithBoundary_of_weakHopf,
        MPSTensor.MPOSymmetry.hamiltonian_arbitraryPhysicalMixedInteraction_commute_mpo_of_weakHopf,
        MPSTensor.MPOSymmetry.toEuclideanLin_arbitraryPhysicalMixedInteraction_eq_one_sub_weakHopfAverage}
    \leanok
    \uses{thm:mpo_arbitrary_mixed_action,def:glm_wha_representation,
        def:glm_wha_average}
    Assume the hypotheses of Lemma~\ref{thm:mpo_arbitrary_mixed_action},
    that $D_0,D_1>0$, and that $A_0,A_1$ are one-site injective. Assume in
    addition, as a restriction not made by Section~5 of the source, that
    $\widetilde T$ is the operator tensor of a weak-Hopf representation with
    average $E_2$. Then
    $\mathcal R_2(\widetilde T)^*=\mathcal R_2(\widetilde T)$, and the
    conclusions of
    Lemma~\ref{thm:mpo_arbitrary_parent_commutation_adjoint_closed} hold.
    Moreover, for $0<\gamma<1$ the symmetrized parent term coincides with the
    canonical one: $\widetilde h_\gamma=\Id-E_2(P_\gamma)$, where $P_\gamma$
    is the orthogonal projection onto the two-site ground space of
    $\widetilde A(\gamma)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:glm_wha_integral_normalization,
        thm:mpo_arbitrary_parent_commutation_adjoint_closed,
        thm:mpo_arbitrary_tensor_continuity,
        lem:one_block_injective_of_injective,
        thm:glm_wha_parent_average,
        thm:spt_parent_hamiltonian_symmetry,
        thm:mpo_arbitrary_interior_parent}
    The star structure of the representation makes the boundary range
    closed under the adjoint at every length, by
    Lemma~\ref{thm:glm_wha_integral_normalization}, so
    Lemma~\ref{thm:mpo_arbitrary_parent_commutation_adjoint_closed}
    applies. For $0<\gamma<1$ the tensor $\widetilde A(\gamma)$ is
    one-site injective by Lemma~\ref{thm:mpo_arbitrary_tensor_continuity},
    hence one-block injective by
    Lemma~\ref{lem:one_block_injective_of_injective}, and it is compatible
    with $\widetilde T$ by Lemma~\ref{thm:mpo_arbitrary_mixed_action}.
    Lemma~\ref{thm:glm_wha_parent_average} gives $\Id-E_2(P_\gamma)=\Id-P_\gamma$,
    which is the matrix of the canonical parent term by
    Lemma~\ref{thm:spt_parent_hamiltonian_symmetry}, and this equals
    $\widetilde h_\gamma$ by Lemma~\ref{thm:mpo_arbitrary_interior_parent}.
\end{proof}

The weak-Hopf route therefore yields the canonical parent together with
adjoint closure derived from the star structure; it does not remove the
closure hypothesis, but moves it into the existence of the representation.

The fusion hypotheses and the equality of $L$ matrices are not needed for
this commutation. The construction for degenerate ground spaces is the
subject of Section~\ref{sec:mpo_block_mixed_path}.
