\section{Nearest-neighbor parents of injective PEPS}%
\label{sec:peps_injective_nearest_neighbor_parent}

\begin{definition}[Edge and isolated-site parent regions]%
    \label{def:peps_injective_nearest_neighbor_regions}
    \lean{TNLean.PEPS.edgePairRegion,TNLean.PEPS.edgeAndIsolatedRegion}
    \leanok
    On a finite graph $\Gamma=(V,E)$, define the edge-pair region family
    $\mathcal R_{\mathrm{edge}}=\{R_e:e\in E\}$ with $R_e=\{u,v\}$ for
    $e=\{u,v\}$, and the augmented family
    $\mathcal R_{\mathrm{aug}}=\mathcal R_{\mathrm{edge}}\cup\{\{v\}:v\in V\text{ isolated}\}$,
    each equipped with the open-boundary contraction range $\mathcal G_R(A)$.
    Isolated-site conditions are needed to exclude
    unused physical directions at vertices absent from every edge;
    see \cite{gap:cpgsv21_injective_parent_reconstruction}.
\end{definition}

\begin{lemma}[Coverage by the nearest-neighbor regions]%
    \label{lem:peps_injective_nearest_neighbor_cover}
    \lean{TNLean.PEPS.edgePairRegion_covers_edges,
        TNLean.PEPS.mem_edgePairRegion_of_adj,
        TNLean.PEPS.edgePairRegion_covers_vertices,
        TNLean.PEPS.edgeAndIsolatedRegion_covers_vertices,
        TNLean.PEPS.edgeAndIsolatedRegion_covers_edges}
    \leanok
    \uses{def:peps_injective_nearest_neighbor_regions}
    Every $e=\{u,v\}\in E$ satisfies $\{u,v\}\subseteq R_e$. If every $v\in V$
    has a neighbor, $\mathcal R_{\mathrm{edge}}$ covers both $V$ and $E$; on an
    arbitrary finite graph, $\mathcal R_{\mathrm{aug}}$ covers both $V$ and $E$.
\end{lemma}

\begin{proof}\leanok
    If $v$ has a neighbor $u$, then $v\in R_{\{u,v\}}\in\mathcal R_{\mathrm{edge}}$;
    if $v$ is isolated, then $v\in\{v\}\in\mathcal R_{\mathrm{aug}}$.
\end{proof}

\begin{theorem}[The nearest-neighbor parent space]%
    \label{thm:peps_injective_nearest_neighbor_parent_space}
    \lean{TNLean.PEPS.regionParentGroundSpace_edgePairs_eq_span,
        TNLean.PEPS.regionParentGroundSpace_edgeAndIsolated_eq_span,
        TNLean.PEPS.finrank_regionParentGroundSpace_edgePairs_eq_one,
        TNLean.PEPS.finrank_regionParentGroundSpace_edgeAndIsolated_eq_one}
    \leanok
    \uses{lem:peps_injective_nearest_neighbor_cover,
        thm:peps_injective_covering_parent_ground_space,
        def:peps_injective_tensor}
    Suppose every site tensor $A_v$ is injective. If every vertex has a
    neighbor, then $\mathcal P_{\mathcal R_{\mathrm{edge}}}(A)=\spn\{\Psi(A)\}$;
    on an arbitrary finite graph,
    $\mathcal P_{\mathcal R_{\mathrm{aug}}}(A)=\spn\{\Psi(A)\}$.
    If $A$ is an injective PEPS tensor (Definition~\ref{def:peps_injective_tensor},
    with all $D_e>0$), each space has dimension $1$; without $D_e>0$ the span
    equalities still hold and $\spn\{\Psi(A)\}$ may be $\{0\}$.
    These are the nearest-neighbor and isolated-site forms of the
    virtual-pair argument in
    \cite[Section~IV.C.1]{Cirac2021Matrix}.
\end{theorem}

\begin{proof}\leanok
    Combine Lemma~\ref{lem:peps_injective_nearest_neighbor_cover} with
    Theorem~\ref{thm:peps_injective_covering_parent_ground_space}.
\end{proof}

\begin{corollary}[Exact-kernel nearest-neighbor Hamiltonians]%
    \label{cor:peps_injective_nearest_neighbor_parent_kernel}
    \lean{TNLean.PEPS.ker_regionParentHamiltonian_edgePairs_eq_span,
        TNLean.PEPS.ker_regionParentHamiltonian_edgeAndIsolated_eq_span,
        TNLean.PEPS.finrank_ker_regionParentHamiltonian_edgePairs_eq_one,
        TNLean.PEPS.finrank_ker_regionParentHamiltonian_edgeAndIsolated_eq_one}
    \leanok
    \uses{thm:peps_injective_nearest_neighbor_parent_space}
    For interactions $h_R\ge 0$ with $\ker h_R=\mathcal G_R(A)$ and
    $H_{\mathcal R}=\sum_{R\in\mathcal R}h_R^{\mathrm{global}}$, the hypotheses
    of Theorem~\ref{thm:peps_injective_nearest_neighbor_parent_space} yield
    $\ker H_{\mathcal R_{\mathrm{edge}}}=\spn\{\Psi(A)\}$ (when every vertex
    has a neighbor) and $\ker H_{\mathcal R_{\mathrm{aug}}}=\spn\{\Psi(A)\}$
    (on any finite graph), both of dimension $1$ when $A$ is an injective PEPS
    tensor ($D_e>0$).
\end{corollary}

\begin{proof}\leanok
    Since $h_R\ge 0$, $\ker H_{\mathcal R}=\bigcap_{R\in\mathcal R}\ker h_R^{\mathrm{global}}=\mathcal P_{\mathcal R}(A)$,
    so Theorem~\ref{thm:peps_injective_nearest_neighbor_parent_space} applies.
\end{proof}
