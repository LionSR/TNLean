\begin{lemma}[Sections of a positive periodic interaction]
    \label{lem:periodic_interaction_sections}
    \lean{MPSTensor.cyclicRestrictES_periodicLocalInteractionES}
    \lean{MPSTensor.periodicLocalInteractionES_eq_zero_iff_sections}
    \lean{MPSTensor.periodicInteractionHamiltonianES_eq_zero_iff_sections}
    \leanok
    For $1\le R\le M$, let $h$ be an operator on $R$ consecutive sites,
    let $h_i$ be its cyclic translate extended by the identity, and write
    $R_{i,\tau}\psi$ for the coefficient section with the outside letters
    fixed to $\tau$. Then $R_{i,\tau}h_i\psi=hR_{i,\tau}\psi$.
    In particular, $h_i\psi=0$ precisely when $hR_{i,\tau}\psi=0$ for every
    $\tau$. If $h\ge0$, then
    $(\sum_i h_i)\psi=0$ precisely when every $hR_{i,\tau}\psi=0$.
\end{lemma}
\begin{proof}
    \leanok
    Extending $h$ by the identity applies it independently to each outside
    coefficient section. A sum of positive operators kills a vector precisely
    when every summand kills it.
\end{proof}

\begin{lemma}[Restriction of an interaction inside an interval]

\end{lemma}
\begin{proof}
    \leanok

    operator becomes the same $W$-site operator acting independently at
    each fixed outside configuration. Taking any such section preserves

\end{proof}

\section{Embeddings of physical coordinate spaces}

\begin{definition}[Euclidean extension by zero]
    \label{def:physical_euclidean_extension_zero}
    \lean{Function.Embedding.euclideanExtendByZero,
        Function.Embedding.euclideanExtendByZero_apply_image,
        Function.Embedding.euclideanExtendByZero_apply_not_mem_range}
    \leanok
    For an injection $e:I\hookrightarrow J$ of finite sets, let
    $E_e:\mathbb C^I\to\mathbb C^J$ extend each coefficient family by zero:
    \begin{align}
        (E_e\xi)(j) & =
        \begin{cases}
            \xi(i), & j=e(i),       \\
            0,      & j\notin e(I).
        \end{cases}
    \end{align}
    This applies in particular to an injection of dependent configuration
    sets induced by a family of physical embeddings.
\end{definition}

\begin{lemma}[Extension by zero is an isometry]
    \label{lem:physical_euclidean_extension_zero_isometry}
    \lean{Function.Embedding.norm_euclideanExtendByZero}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    For every injection $e:I\hookrightarrow J$ of finite sets, the map $E_e$
    is linear and $\lVert E_e\xi\rVert=\lVert\xi\rVert$ for every
    $\xi\in\mathbb C^I$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Since $(E_e\xi)(j)=0$ for $j\notin e(I)$ and $e$ is injective,
    \begin{align}
        \sum_{j\in J}\lvert(E_e\xi)(j)\rvert^2
         & =\sum_{i\in I}\lvert(E_e\xi)(e(i))\rvert^2
        =\sum_{i\in I}\lvert\xi(i)\rvert^2.
    \end{align}
\end{proof}

\begin{lemma}[Coefficients outside a single physical sector]
    \label{lem:physical_sector_embedded_coefficients}
    \lean{MPSTensor.embedded_physical_vector_apply_eq_zero_of_labels_ne}
    \lean{MPSTensor.coeff_eq_zero_of_mem_two_embedded_physical_sectors}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Let each physical letter have a sector label. Extend a vector from an
    alphabet on which this label is constant by zero to the larger alphabet,
    site by site. Its coefficient on any word containing two different
    labels is zero. The same assertion holds for a sum of two local subspaces
    embedded from alphabets on each of which the label is constant.
\end{lemma}
\begin{proof}
    \leanok
    A word in the image of either alphabet has constant label. A word with
    two different labels is outside that image, where extension by zero
    has zero coefficient. Addition preserves these zero equations.
\end{proof}

\begin{lemma}[Sector labels in a periodic projector kernel]
    \label{lem:physical_sector_projector_kernel_support}
    \lean{MPSTensor.physical_label_eq_of_projectorHamiltonian_coeff_ne_zero}
    \lean{MPSTensor.coeff_eq_zero_of_projectorHamiltonian_of_physical_labels_ne}
    \leanok
    \uses{lem:periodic_interaction_sections}
    Let $2\le R\le M$ and let $S$ be a subspace on $R$ sites. Suppose
    every vector in $S$ has zero coefficient on words whose first two
    letters have different sector labels. Form the periodic Hamiltonian
    on $M$ sites from the orthogonal projection onto $S^{\perp}$.
    Every nonzero coefficient of a kernel vector has the same label at
    every site. Equivalently, a word with two different labels has zero
    coefficient in every kernel vector.
\end{lemma}
\begin{proof}
    \leanok
    Let $\psi$ be in the kernel. Positivity and
    Lemma~\ref{lem:periodic_interaction_sections} give
    $R_{i,\tau}\psi\in S$ for every site $i$ and outside word $\tau$.
    Suppose $\psi(\sigma)\ne0$, and let $\tau$ be the outside part of
    $\sigma$ for the window beginning at $i$. That window is
    $\sigma_i\cdots\sigma_{i+R-1}$ and
    \begin{align}
        \psi(\sigma) & =(R_{i,\tau}\psi)(\sigma_i,\ldots,\sigma_{i+R-1}).
    \end{align}
    If $\mathrm{label}(\sigma_i)\ne\mathrm{label}(\sigma_{i+1})$, the
    hypothesis on $S$ makes the right side zero. Hence
    $\mathrm{label}(\sigma_i)=\mathrm{label}(\sigma_{i+1})$ for every $i$,
    and induction around the cycle gives
    $\mathrm{label}(\sigma_i)=\mathrm{label}(\sigma_0)$.
\end{proof}

\begin{lemma}[No mixed physical sectors in the periodic kernel]
    \label{lem:two_physical_sector_projector_kernel}
    \lean{MPSTensor.coeff_eq_zero_of_two_sector_projectorHamiltonian}
    \lean{MPSTensor.coeff_eq_zero_of_physical_sum_projectorHamiltonian}
    \leanok
    \uses{lem:physical_sector_embedded_coefficients,lem:physical_sector_projector_kernel_support}
    Let $2\le R\le M$. Embed two local spaces $S$ and $T$ from alphabets
    on which the sector label is constant, and let $U$ be the sum of their
    images on $R$ sites. The periodic Hamiltonian formed from the
    orthogonal projection onto $U^{\perp}$ has zero ground-state coefficient
    on every word containing different labels. In particular, split an
    alphabet of $a+b$ letters into its first $a$ and last $b$ letters:
    every word using letters from both parts has zero coefficient.
\end{lemma}
\begin{proof}
    \leanok
    The two embedded spaces each vanish on words with different labels,
    so their sum does also. Apply the periodic projector-kernel statement
    to this sum of embedded spaces.
\end{proof}

\begin{lemma}[Exact decomposition into physical sectors]
    \label{lem:physical_sector_state_decomposition}
    \lean{MPSTensor.physical_sum_state_decomposition}
    \lean{MPSTensor.physical_sum_state_decomposition_of_projectorHamiltonian_eq_zero}
    \leanok
    \uses{lem:two_physical_sector_projector_kernel}
    Split an alphabet into $a$ and $b$ letters, and let $M\ge1$.
    If a vector $\psi$ on $M$ sites has zero coefficient on every word
    using both alphabets, then $\psi=E_M\psi_a+F_M\psi_b$, where $E_M$
    and $F_M$ extend by zero from the two alphabets and $\psi_a,\psi_b$
    are the coefficient restrictions of $\psi$.
    In particular, this decomposition holds for every kernel vector of
    the periodic complementary projector of two embedded local spaces
    whenever $M\ge R\ge2$.
\end{lemma}
\begin{proof}
    \leanok
    The two word images are disjoint because the chain is nonempty.
    For a word $\sigma=e\circ\rho$ in the first image,
    \begin{align}
        (E_M\psi_a)(\sigma) & =\psi_a(\rho)=\psi(\sigma),
                            & (F_M\psi_b)(\sigma)         & =0,
    \end{align}
    and the second image is analogous. A word $\sigma$ in neither image
    uses both alphabets, so
    $\psi(\sigma)=(E_M\psi_a)(\sigma)=(F_M\psi_b)(\sigma)=0$.
    Lemma~\ref{lem:two_physical_sector_projector_kernel} supplies precisely
    this mixed-coefficient vanishing for kernel vectors.
\end{proof}

\section{Component Hamiltonians of disjoint physical sectors}

Write $E_m$ and $F_m$ for extension by zero on $m$-letter words along
one-letter injections $e$ and $f$ with disjoint images. For local subspaces
$S$ and $T$ on $R$ sites, put $U=E_R S+F_R T$. Let $H_U$, $H_S$, and $H_T$
be the periodic sums of the complementary projections of these spaces.

\begin{lemma}[Orthogonality of the physical sectors]
    \label{lem:physical_sector_word_orthogonality}
    \lean{Function.Embedding.inner_extendByZero_eq_zero_of_disjoint,
        MPSTensor.disjoint_physical_word_ranges}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Extension by zero from disjoint coordinate ranges gives orthogonal
    vectors. In particular, for $m\ge1$, the word embeddings of
    disjoint one-letter alphabets have disjoint ranges, and
    $\langle E_m v,F_m w\rangle=0$.
\end{lemma}
\begin{proof}
    \leanok
    In the coefficient sum defining the inner product, at least one
    coefficient vanishes at every coordinate. An embedded word in both
    ranges would have its first letter in both one-letter images.
\end{proof}

\begin{lemma}[The local projector restricts to its sector]
    \label{lem:physical_sector_parent_projection_transport}
    \lean{MPSTensor.physical_sector_parent_projection_embed}
    \leanok
    \uses{lem:physical_sector_word_orthogonality,
        lem:physical_euclidean_extension_zero_isometry}
    For $R\ge1$, the complementary projection of the sum
    $U=E_R S+F_R T$ satisfies
    \begin{align}
        P_{U^{\perp}}E_R v & =E_R P_{S^{\perp}}v.
    \end{align}
    Interchanging the two alphabets gives the analogous identity on the
    second sector. No range or projector identity is assumed.
\end{lemma}
\begin{proof}
    \leanok
    The candidate projection $E_R P_Sv$ belongs to $U$. Its residual is
    orthogonal to $E_R S$ by isometry and to $F_R T$ by sector orthogonality.
    Uniqueness of orthogonal projection gives $P_U E_Rv=E_RP_Sv$.
    Subtract from $E_Rv$ to obtain the complementary projection identity.
\end{proof}

\begin{lemma}[The periodic sum restricts to its component]
    \label{lem:physical_sector_periodic_hamiltonian_transport}
    \lean{MPSTensor.physical_sector_projectorHamiltonian_embed,
        MPSTensor.physical_sector_projectorHamiltonian_embed_eq_zero_iff}
    \leanok
    \uses{lem:physical_sector_parent_projection_transport,
        lem:physical_spectator_embedding_transport,
        lem:physical_euclidean_extension_zero_isometry}
    If $1\le R\le M$, then
    \begin{align}
        H_U E_Mv  & =E_MH_Sv,                             &
        H_UE_Mv=0 & \quad\Longleftrightarrow\quad H_Sv=0.
    \end{align}
    The second-sector assertions follow by interchanging the embeddings
    and local spaces. All operators are the periodic sums.
\end{lemma}
\begin{proof}
    \leanok
    Splitting a cyclic window and its complement commutes with sitewise
    physical embedding. The local complementary projection intertwines
    with $E_R$, and its extension by the identity intertwines with the
    corresponding outside embedding. Conjugate by the cyclic coordinate
    changes and sum over all starting sites. Injectivity of the physical
    isometry gives the zero equation in both directions.
\end{proof}

\begin{lemma}[Exact component kernel condition for a sector sum]
    \label{lem:physical_sector_sum_kernel_iff}
    \lean{MPSTensor.physical_sector_projectorHamiltonian_sum_eq_zero_iff}
    \leanok
    \uses{lem:physical_sector_periodic_hamiltonian_transport,
        lem:physical_sector_word_orthogonality}
    Under the same assumptions,
    \begin{align}
        H_U(E_Mv+F_Mw)=0
        \quad\Longleftrightarrow\quad H_Sv=0\ \text{and}\ H_Tw=0.
    \end{align}
    No whole-kernel decomposition or dynamics relation is a premise.
\end{lemma}
\begin{proof}
    \leanok
    The sum acts as $E_MH_Sv+F_MH_Tw$. Its two terms are orthogonal.
    If their sum vanishes, taking the inner product with the first term
    shows that term vanishes; the second then vanishes also. Injectivity
    gives the two component equations. The converse follows by linearity.
\end{proof}

\begin{lemma}[Full kernel decomposition for a split physical alphabet]
    \label{lem:physical_sum_full_kernel}
    \lean{MPSTensor.ker_physical_sum_projectorHamiltonian}
    \leanok
    \uses{lem:physical_sector_state_decomposition,lem:physical_sector_sum_kernel_iff}
    Let $M\ge R\ge2$, and split the physical alphabet into its first $a$
    and last $b$ letters. For local spaces $S$ and $T$, put
    $U=E_RS+F_RT$, where the two maps extend by zero on physical words.
    For the periodic sums of the complementary projections,
    $\ker H_U=E_M\ker H_S+F_M\ker H_T$.
\end{lemma}
\begin{proof}
    \leanok
    Every kernel vector has zero mixed-sector coefficients and therefore
    splits into its two coefficient restrictions. The component
    kernel equivalence puts each restriction in its respective kernel.
    Conversely, the component kernel equivalence and linearity show that
    the sum of any two embedded component kernel vectors is killed by
    the combined Hamiltonian.
\end{proof}

\begin{lemma}[Physical extension of a periodic tensor]
    \label{lem:mpv_state_extend_physical}
    \lean{MPSTensor.mpvState_extend_physical}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Let $e:\{0,\ldots,a-1\}\hookrightarrow\{0,\ldots,d-1\}$ be an embedding,
    and let $A_i$ be $D\times D$ matrices. Set $\widetilde A_{e(i)}=A_i$
    and $\widetilde A_j=0$ outside the image. At every length $N\ge0$,
    the periodic matrix product vectors satisfy
    \begin{align}
        \ket{V^{(N)}(\widetilde A)} & =E_N\ket{V^{(N)}(A)},
    \end{align}
    where $E_N$ is extension by zero along the sitewise configuration
    embedding. The bond dimension is unchanged.
\end{lemma}
\begin{proof}
    \leanok
    For $\sigma=e\circ\rho$ in the image, $\widetilde A_{\sigma_k}=A_{\rho_k}$
    at every site $k$, so
    \begin{align}
        V^{(N)}(\widetilde A)_\sigma
         & =\tr(\widetilde A_{\sigma_0}\cdots\widetilde A_{\sigma_{N-1}})
        =\tr(A_{\rho_0}\cdots A_{\rho_{N-1}})=V^{(N)}(A)_\rho.
    \end{align}
    If some $\sigma_k$ lies outside the image, then
    $\widetilde A_{\sigma_k}=0$, so the product and
    $V^{(N)}(\widetilde A)_\sigma$ vanish. At $N=0$ the empty
    configuration belongs to the image, and both coefficients equal
    $\tr(\mathbb 1_D)$.
\end{proof}

\section{Parent Hamiltonians with varying physical dimensions}

\begin{definition}[A common alphabet for differently sized physical sites]
    \label{def:varying_physical_common_alphabet}
    \lean{Matrix.commonPhysicalEmbedding,
        Matrix.commonPhysicalConfigurationIsometry,
        Matrix.commonPhysicalConfigurationIsometry_apply_image}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Let a finite family of sites have physical dimensions $d_i$, and set
    $D=\sum_i d_i$. Each alphabet $\{0,\ldots,d_i-1\}$ embeds in
    $\{0,\ldots,D-1\}$ by retaining its numerical label. The induced injection
    of dependent configuration sets gives a linear isometry from
    the physical chain space into the common-alphabet chain space.
    Each original coefficient is preserved and all coefficients outside
    the image vanish.
\end{definition}

\begin{lemma}[Physical extension preserves the canonical site equations]
    \label{lem:varying_physical_extension_span_row}
    \lean{Matrix.span_range_extend_physical,
        Matrix.sum_mul_conjTranspose_extend_physical}
    \leanok
    Let $A_i\in M_{a,b}(\mathbb C)$ and $e:I\hookrightarrow J$ be an injection
    of finite physical alphabets. Define $\widetilde A_{e(i)}=A_i$ and
    $\widetilde A_j=0$ outside the image. Then
    \begin{align}
        \operatorname{span}\{\widetilde A_j:j\in J\}
         & =\operatorname{span}\{A_i:i\in I\},
         & \sum_{j\in J}\widetilde A_j\widetilde A_j^\dagger
         & =\sum_{i\in I}A_iA_i^\dagger.
    \end{align}
    Thus both the spanning condition and canonical row normalization are
    unchanged by physical extension.
\end{lemma}
\begin{proof}\leanok
    Every original matrix occurs at its embedded label; every added matrix
    is zero. The same observation removes the zero terms from the row sum.
\end{proof}

\begin{lemma}[Pair boundary ranges and parent projectors intertwine]
    \label{lem:varying_physical_pair_projection_transport}
    \lean{Matrix.rectangularPairBoundary,
        Matrix.rectangularPairBoundary_apply,
        Matrix.range_rectangularPairBoundary_extend_physical,
        Matrix.pair_parent_projection_extend_physical}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Consider rectangular families $A_i\in M_{a,b}(\mathbb C)$ and
    $B_k\in M_{b,c}(\mathbb C)$, with independent physical embeddings
    $e:I\hookrightarrow J$ and $f:K\hookrightarrow L$. Their pair range is
    \begin{align}
        G & =\left\{\left(\operatorname{tr}(XA_iB_k)\right)_{i,k}:
        X\in M_{c,a}(\mathbb C)\right\}.
    \end{align}
    Let $\widetilde G$ be the range of the extended families. Then
    $\widetilde G=E_{e\times f}(G)$, and the complementary orthogonal
    projections satisfy
    \begin{align}
        P_{\widetilde G^\perp}E_{e\times f}
         & =E_{e\times f}P_{G^\perp}.
    \end{align}
    No spanning or normalization assumption is required.
\end{lemma}
\begin{proof}\leanok
    The pair coefficients extend by zero because a matrix at any added
    physical label is zero. This proves equality of the ranges.
    An isometry carries the orthogonal projection onto $G$ to the projection
    onto its image when evaluated on embedded vectors. Subtracting that
    identity from the identity operator gives the complementary projections.
\end{proof}

\begin{lemma}[Extended local parent terms intertwine]
    \label{lem:varying_physical_local_parent_transport}
    \lean{Matrix.pair_parent_localTerm_extend_physical}
    \leanok
    \uses{lem:varying_physical_pair_projection_transport,
        lem:physical_spectator_embedding_transport}
    Let $S$ be the finite configuration space outside a pair, with any
    injection $g:S\hookrightarrow T$. Extend the complementary pair
    projectors by the identity outside the pair. Then
    \begin{align}
        (P_{\widetilde G^\perp}\otimes I_T)E_{(e\times f)\times g}
         & =E_{(e\times f)\times g}(P_{G^\perp}\otimes I_S).
    \end{align}
    The local operators and their intertwining are derived from the
    rectangular families and their physical embeddings.
\end{lemma}
\begin{proof}\leanok
    At an outside configuration $g(s)$, the active-coordinate vector of an
    embedded vector is the embedding of its original active-coordinate vector
    at $s$. At outside configurations not in the image of $g$, that vector is
    zero. Apply the pair-projector intertwining on each such vector; linearity
    handles the zero vectors outside the image.
\end{proof}

\subsection{Dependent chain coordinates and the open sum}

\begin{definition}[Pair coordinates of a dependent physical chain]
    \label{def:dependent_physical_pair_coordinates}
    \lean{Matrix.pairConfigurationEquiv, Matrix.pairConfigurationEquiv_apply,
        Matrix.pairConfigurationReindex}
    \leanok
    Let the physical alphabet at site $k$ be a finite set $I_k$. For two
    distinct sites $i,j$, split the dependent configuration set as
    \begin{align}
        \prod_k I_k & \simeq (I_i\times I_j)\times
        \prod_{k\ne i,j}I_k.
    \end{align}
    The bijection reads both active letters in the displayed order and
    retains all other physical coordinates. Its Euclidean reindexing is
    a linear isometric equivalence.
\end{definition}

\begin{lemma}[Dependent pair coordinates respect physical embeddings]
    \label{lem:dependent_physical_pair_embedding}
    \lean{Matrix.pairConfigurationEquiv_embedding_apply,
        Matrix.pairConfigurationReindex_extendByZero}
    \leanok
    \uses{def:dependent_physical_pair_coordinates,
        def:physical_euclidean_extension_zero}
    For sitewise injections $e_k:I_k\hookrightarrow J_k$, the dependent
    pair/outside coordinate bijection commutes with the full configuration
    injection. The corresponding Euclidean coordinate changes $Q_I,Q_J$
    satisfy
    \begin{align}
        Q_J E_{\prod_k e_k}
         & =E_{(e_i\times e_j)\times\prod_{k\ne i,j}e_k} Q_I.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    The coordinate bijection reads the same sitewise embedded letters on
    both sides. At coordinates in the image, both Euclidean maps preserve
    the source coefficient. Outside that image, both coefficients vanish.
\end{proof}

\begin{definition}[The parent Hamiltonian on the true dependent chain space]
    \label{def:varying_physical_parent_hamiltonian}
    \lean{Matrix.physicalPairParentTerm, Matrix.physicalPairParentTerm_in_coordinates,
        Matrix.varyingPhysicalParentHamiltonian}
    \leanok
    \uses{def:dependent_physical_pair_coordinates,lem:varying_physical_pair_projection_transport}
    Let $A^{[j]}_i\in M_{D_j,D_{j+1}}(\mathbb C)$ have an arbitrary finite
    physical alphabet $I_j$ at each site. For the adjacent pair $j,j+1$,
    let $G_j$ be its pair range. On the
    chain space $\mathbb C^{\prod_k I_k}$, define
    \begin{align}
        h_j & =Q_j^{-1}(P_{G_j^\perp}\otimes I)Q_j,
            & H                                     & =\sum_{j=0}^{N-2}h_j.
    \end{align}
    These are the pair projections and open Hamiltonian of
    \cite[Theorem~9, lines~1195--1217]{PerezGarcia2007Matrix}, with no
    common-alphabet restriction on the sites.
\end{definition}

\begin{lemma}[Varying-physical parent transport and kernel equivalence]
    \label{lem:varying_physical_parent_embedding_transport}
    \lean{Matrix.physicalPairParentTerm_isPositive,
        Matrix.physicalPairParentTerm_extend_physical,
        Matrix.varyingPhysicalParentHamiltonian_isPositive,
        Matrix.varyingPhysicalParentHamiltonian_extend_physical,
        Matrix.varyingPhysicalParentHamiltonian_extend_eq_zero_iff}
    \leanok
    \uses{def:varying_physical_parent_hamiltonian,
        lem:dependent_physical_pair_embedding,
        lem:varying_physical_local_parent_transport}
    Every local term $h_j$ and their sum $H$ are positive. For
    arbitrary sitewise physical injections, extend each site matrix by
    zero to obtain the corresponding parent $\widetilde H$. Then
    \begin{align}
        \widetilde H E & =EH,
                       & \widetilde H(E\phi)=0 & \iff H\phi=0.
    \end{align}
    Here $E$ is the dependent configuration isometry. No canonical
    normalization, spanning assumption, or supplied Hamiltonian relation
    is required for these identities.
\end{lemma}
\begin{proof}\leanok
    A complementary pair projection is positive, and identity extension
    and isometric coordinate change preserve positivity. The pair/outside
    coordinate intertwining and local-projector transport give
    $\widetilde h_j E=E h_j$ at each bond. Summing these derived identities
    gives $\widetilde H E=EH$. Injectivity of the isometry gives the kernel
    equivalence.
\end{proof}

\begin{lemma}[Local equations are boundary sections]
    \label{lem:varying_physical_local_boundary_sections}
    \lean{Matrix.physicalPairParentTerm_boundary_of_apply_eq_zero,
        Matrix.physicalPairParentTerm_apply_eq_zero_iff_boundary}
    \leanok
    \uses{def:varying_physical_parent_hamiltonian}
    A local term annihilates a vector if and only if, at every
    fixed outside configuration $\eta$, its pair coefficients have a
    rectangular trace boundary:
    \begin{align}
        \phi\bigl(Q^{-1}((x,y),\eta)\bigr) & =\operatorname{tr}(XA_xB_y).
    \end{align}
    The matrix $X$ may depend on $\eta$.
\end{lemma}
\begin{proof}\leanok
    In pair/outside coordinates the local term applies the complementary
    pair projection independently on every outside vector. Its kernel is
    exactly the pair trace range. Membership in that range supplies
    $X$. Conversely the displayed representation places every outside
    vector in that kernel, so the identity-extended term vanishes.
\end{proof}

\begin{lemma}[Unused physical letters are excluded by the kernel]
    \label{lem:varying_physical_kernel_forces_support}
    \lean{Matrix.physicalPairParentTerm_coeff_eq_zero_of_not_mem_range,
        Matrix.varyingPhysicalParentHamiltonian_apply_eq_zero_iff_terms,
        Matrix.varyingPhysicalParentHamiltonian_pair_apply_eq_zero,
        Matrix.varyingPhysicalParentHamiltonian_coeff_eq_zero_of_not_mem_range}
    \leanok
    \uses{lem:varying_physical_local_boundary_sections,
        lem:varying_physical_parent_embedding_transport}
    Extend site matrices by zero along arbitrary sitewise physical
    injections $e_k:I_k\hookrightarrow J_k$. Suppose $N\ge2$ and the extended
    parent satisfies $\widetilde H\phi=0$. Then $\phi(\sigma)=0$
    whenever some physical letter $\sigma_k$ is outside $e_k(I_k)$.
    Every adjacent local term annihilates $\phi$.
\end{lemma}
\begin{proof}\leanok
    Positivity of the local terms derives each local equation from
    the equation for their sum. At a letter $\sigma_k$ outside the original
    alphabet, the extended site matrix $\widetilde A_{\sigma_k}$ is zero, so
    for every boundary matrix $X$ of a pair containing site $k$,
    \begin{align}
        \operatorname{tr}\bigl(X\widetilde A_{\sigma_k}\widetilde A_{\sigma_{k+1}}\bigr)=0,
        \qquad
        \operatorname{tr}\bigl(X\widetilde A_{\sigma_{k-1}}\widetilde A_{\sigma_k}\bigr)=0.
        \label{eq:varying_physical_unused_letter_trace}
    \end{align} Every site of a chain with $N\ge2$ belongs to an
    adjacent pair, including the final site, which belongs to the final pair.
    The local boundary sections therefore give the claimed vanishing at
    every configuration containing an unused letter.
\end{proof}

\begin{lemma}[The kernel lies in the original physical image]
    \label{lem:varying_physical_kernel_actual_pullback}
    \lean{Matrix.varyingPhysicalParentHamiltonian_eq_extendByZero_pullback_of_eq_zero}
    \leanok
    \uses{lem:varying_physical_kernel_forces_support}
    Under the same hypotheses, every extended-kernel vector is the
    extension of its literal coefficient pullback:
    \begin{align}
        \phi & =E\bigl(\tau\mapsto\phi((e_k(\tau_k))_k)\bigr).
    \end{align}
    Membership in this embedded state space is a conclusion, not a hypothesis.
\end{lemma}
\begin{proof}\leanok
    At a configuration in the embedding image, extension by zero recovers
    exactly its original coefficient. A configuration outside that image has
    a letter outside one of the site images, since sitewise preimages would
    assemble a full dependent preimage. Its coefficient vanishes by the
    kernel support statement, and the extended coefficient is zero too.
\end{proof}

\begin{lemma}[The two common-alphabet parent descriptions have the same kernel]
    \label{lem:uniform_physical_parent_kernel_comparison}
    \lean{Matrix.varyingPhysicalParentHamiltonian_uniform_apply_eq_zero_iff}
    \leanok
    \uses{lem:varying_physical_local_boundary_sections,lem:varying_obc_parent_kernel_sections,lem:varying_physical_kernel_forces_support}
    On a common nonempty physical alphabet, the dependent-coordinate
    open parent and the prefix/pair/suffix open parent have exactly
    the same kernel at every length.
\end{lemma}
\begin{proof}\leanok
    Both local equations say that every fixed-outside pair section has a
    rectangular trace boundary. To compare the coordinate descriptions,
    concatenate a prefix $u$ of length $j$, the active pair $s=(s_0,s_1)$,
    and a suffix $v$. With $Q_j$ the pair coordinates at $j$,
    \begin{align}
        Q_j(u\,s\,v) & =\bigl(s,\eta_{u,v}\bigr),
        \label{eq:uniform_physical_pair_coordinates}
    \end{align}
    where the outside coordinates $\eta_{u,v}$ do not depend on $s$; changing
    only the pair leaves all outside coordinates unchanged. Conversely, split a full
    configuration into its prefix, pair, and suffix; concatenating
    them recovers the configuration. A reference pair from the nonempty
    alphabet fixes the outside section. These two coordinate identifications
    transport the boundary representations in both directions. Positivity
    and the local-kernel characterizations give the full kernel
    equivalence. Canonical normalization derives alphabet nonemptiness in
    the subsequent uniqueness application.
\end{proof}
