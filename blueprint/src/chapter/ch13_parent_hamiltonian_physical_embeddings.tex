\section{Embeddings of physical coordinate spaces}

\begin{definition}[Euclidean extension by zero]
    \label{def:physical_euclidean_extension_zero}
    \lean{Function.Embedding.euclideanExtendByZero,
        Function.Embedding.euclideanExtendByZero_apply_image,
        Function.Embedding.euclideanExtendByZero_apply_not_mem_range}
    \leanok
    For an injection $e:I\hookrightarrow J$ of finite sets, let
    $E_e:\mathbb C^I\to\mathbb C^J$ extend each coefficient family by zero:
    \begin{align}
        (E_e\xi)(j) & =
        \begin{cases}
            \xi(i), & j=e(i),       \\
            0,      & j\notin e(I).
        \end{cases}
    \end{align}
    Then $E_e$ is a linear isometry. In particular, this applies to an
    injection of dependent configuration sets induced by a family of
    physical embeddings.
\end{definition}
\begin{proof}\leanok
    The sum of $|(E_e\xi)(j)|^2$ over $j\in J$ equals the sum of
    $|\xi(i)|^2$ over $i\in I$, since every coefficient outside $e(I)$ is
    zero and each coefficient in $e(I)$ occurs once.
\end{proof}

\begin{definition}[A common alphabet for differently sized physical sites]
    \label{def:varying_physical_common_alphabet}
    \lean{Matrix.commonPhysicalEmbedding,
        Matrix.commonPhysicalConfigurationIsometry,
        Matrix.commonPhysicalConfigurationIsometry_apply_image}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Let a finite family of sites have physical dimensions $d_i$, and set
    $D=\sum_i d_i$. Each alphabet $\{0,\ldots,d_i-1\}$ embeds in
    $\{0,\ldots,D-1\}$ by retaining its numerical label. The induced injection
    of dependent configuration sets gives a linear isometry from
    the physical chain space into the common-alphabet chain space.
    Each original coefficient is preserved and all coefficients outside
    the image vanish.
\end{definition}

\begin{lemma}[Physical extension preserves the canonical site equations]
    \label{lem:varying_physical_extension_span_row}
    \lean{Matrix.span_range_extend_physical,
        Matrix.sum_mul_conjTranspose_extend_physical}
    \leanok
    Let $A_i\in M_{a,b}(\mathbb C)$ and $e:I\hookrightarrow J$ be an injection
    of finite physical alphabets. Define $\widetilde A_{e(i)}=A_i$ and
    $\widetilde A_j=0$ outside the image. Then
    \begin{align}
        \operatorname{span}\{\widetilde A_j:j\in J\}
         & =\operatorname{span}\{A_i:i\in I\},
         & \sum_{j\in J}\widetilde A_j\widetilde A_j^\dagger
         & =\sum_{i\in I}A_iA_i^\dagger.
    \end{align}
    Thus both the spanning condition and canonical row normalization are
    unchanged by physical extension.
\end{lemma}
\begin{proof}\leanok
    Every original matrix occurs at its embedded label; every added matrix
    is zero. The same observation removes the zero terms from the row sum.
\end{proof}

\begin{lemma}[Pair boundary ranges and parent projectors intertwine]
    \label{lem:varying_physical_pair_projection_transport}
    \lean{Matrix.rectangularPairBoundary,
        Matrix.rectangularPairBoundary_apply,
        Matrix.range_rectangularPairBoundary_extend_physical,
        Matrix.pair_parent_projection_extend_physical}
    \leanok
    \uses{def:physical_euclidean_extension_zero}
    Consider rectangular families $A_i\in M_{a,b}(\mathbb C)$ and
    $B_k\in M_{b,c}(\mathbb C)$, with independent physical embeddings
    $e:I\hookrightarrow J$ and $f:K\hookrightarrow L$. Their pair range is
    \begin{align}
        G & =\left\{\left(\operatorname{tr}(XA_iB_k)\right)_{i,k}:
        X\in M_{c,a}(\mathbb C)\right\}.
    \end{align}
    Let $\widetilde G$ be the range of the extended families. Then
    $\widetilde G=E_{e\times f}(G)$, and the complementary orthogonal
    projections satisfy
    \begin{align}
        P_{\widetilde G^\perp}E_{e\times f}
         & =E_{e\times f}P_{G^\perp}.
    \end{align}
    No spanning or normalization assumption is required.
\end{lemma}
\begin{proof}\leanok
    The pair coefficients extend by zero because a matrix at any added
    physical label is zero. This proves equality of the ranges.
    An isometry carries the orthogonal projection onto $G$ to the projection
    onto its image when evaluated on embedded vectors. Subtracting that
    identity from the identity operator gives the complementary projections.
\end{proof}

\begin{lemma}[Extended local parent terms intertwine]
    \label{lem:varying_physical_local_parent_transport}
    \lean{Matrix.pair_parent_localTerm_extend_physical}
    \leanok
    \uses{lem:varying_physical_pair_projection_transport,
        lem:physical_spectator_embedding_transport}
    Let $S$ be the finite configuration space outside a pair, with any
    injection $g:S\hookrightarrow T$. Extend the complementary pair
    projectors by the identity outside the pair. Then
    \begin{align}
        (P_{\widetilde G^\perp}\otimes I_T)E_{(e\times f)\times g}
         & =E_{(e\times f)\times g}(P_{G^\perp}\otimes I_S).
    \end{align}
    The local operators and their intertwining are derived from the
    rectangular families and their physical embeddings.
\end{lemma}
\begin{proof}\leanok
    At an outside configuration $g(s)$, the active-coordinate vector of an
    embedded vector is the embedding of its original active-coordinate vector
    at $s$. At outside configurations not in the image of $g$, that vector is
    zero. Apply the pair-projector intertwining on each such vector; linearity
    handles the zero vectors outside the image.
\end{proof}

\subsection{Dependent chain coordinates and the open sum}

\begin{definition}[Pair coordinates of a dependent physical chain]
    \label{def:dependent_physical_pair_coordinates}
    \lean{Matrix.pairConfigurationEquiv, Matrix.pairConfigurationEquiv_apply,
        Matrix.pairConfigurationReindex}
    \leanok
    Let the physical alphabet at site $k$ be a finite set $I_k$. For two
    distinct sites $i,j$, split the dependent configuration set as
    \begin{align}
        \prod_k I_k & \simeq (I_i\times I_j)\times
        \prod_{k\ne i,j}I_k.
    \end{align}
    The bijection reads both active letters in the displayed order and
    retains all other physical coordinates. Its Euclidean reindexing is
    a linear isometric equivalence.
\end{definition}

\begin{lemma}[Dependent pair coordinates respect physical embeddings]
    \label{lem:dependent_physical_pair_embedding}
    \lean{Matrix.pairConfigurationEquiv_embedding_apply,
        Matrix.pairConfigurationReindex_extendByZero}
    \leanok
    \uses{def:dependent_physical_pair_coordinates,
        def:physical_euclidean_extension_zero}
    For sitewise injections $e_k:I_k\hookrightarrow J_k$, the dependent
    pair/outside coordinate bijection commutes with the full configuration
    injection. The corresponding Euclidean coordinate changes $Q_I,Q_J$
    satisfy
    \begin{align}
        Q_J E_{\prod_k e_k}
         & =E_{(e_i\times e_j)\times\prod_{k\ne i,j}e_k} Q_I.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    The coordinate bijection reads the same sitewise embedded letters on
    both sides. At coordinates in the image, both Euclidean maps preserve
    the source coefficient. Outside that image, both coefficients vanish.
\end{proof}

\begin{definition}[The parent Hamiltonian on the true dependent chain space]
    \label{def:varying_physical_parent_hamiltonian}
    \lean{Matrix.physicalPairParentTerm, Matrix.physicalPairParentTerm_in_coordinates,
        Matrix.varyingPhysicalParentHamiltonian}
    \leanok
    \uses{def:dependent_physical_pair_coordinates,lem:varying_physical_pair_projection_transport}
    Let $A^{[j]}_i\in M_{D_j,D_{j+1}}(\mathbb C)$ have an arbitrary finite
    physical alphabet $I_j$ at each site. For the adjacent pair $j,j+1$,
    let $G_j$ be its rectangular trace-boundary range. On the
    chain space $\mathbb C^{\prod_k I_k}$, define
    \begin{align}
        h_j & =Q_j^{-1}(P_{G_j^\perp}\otimes I)Q_j,
            & H                                     & =\sum_{j=0}^{N-2}h_j.
    \end{align}
    These are the pair projections and open Hamiltonian of
    \cite[Theorem~9, lines~1195--1217]{PerezGarcia2007Matrix}, with no
    common-alphabet restriction on the sites.
\end{definition}

\begin{lemma}[Varying-physical parent transport and kernel equivalence]
    \label{lem:varying_physical_parent_embedding_transport}
    \lean{Matrix.physicalPairParentTerm_isPositive,
        Matrix.physicalPairParentTerm_extend_physical,
        Matrix.varyingPhysicalParentHamiltonian_isPositive,
        Matrix.varyingPhysicalParentHamiltonian_extend_physical,
        Matrix.varyingPhysicalParentHamiltonian_extend_eq_zero_iff}
    \leanok
    \uses{def:varying_physical_parent_hamiltonian,
        lem:dependent_physical_pair_embedding,
        lem:varying_physical_local_parent_transport}
    Every local term $h_j$ and their sum $H$ are positive. For
    arbitrary sitewise physical injections, extend each site matrix by
    zero to obtain the corresponding parent $\widetilde H$. Then
    \begin{align}
        \widetilde H E & =EH,
                       & \widetilde H(E\phi)=0 & \iff H\phi=0.
    \end{align}
    Here $E$ is the dependent configuration isometry. No canonical
    normalization, spanning assumption, or supplied Hamiltonian relation
    is required for these identities.
\end{lemma}
\begin{proof}\leanok
    A complementary pair projection is positive, and identity extension
    and isometric coordinate change preserve positivity. The pair/outside
    coordinate intertwining and local-projector transport give
    $\widetilde h_j E=E h_j$ at each bond. Summing these derived identities
    gives $\widetilde H E=EH$. Injectivity of the isometry gives the kernel
    equivalence.
\end{proof}

\begin{lemma}[Local equations are boundary sections]
    \label{lem:varying_physical_local_boundary_sections}
    \lean{Matrix.physicalPairParentTerm_boundary_of_apply_eq_zero,
        Matrix.physicalPairParentTerm_apply_eq_zero_iff_boundary}
    \leanok
    \uses{def:varying_physical_parent_hamiltonian}
    A local term annihilates a vector if and only if, at every
    fixed outside configuration $\eta$, its pair coefficients have a
    rectangular trace boundary:
    \begin{align}
        \phi\bigl(Q^{-1}((x,y),\eta)\bigr) & =\operatorname{tr}(XA_xB_y).
    \end{align}
    The matrix $X$ may depend on $\eta$.
\end{lemma}
\begin{proof}\leanok
    In pair/outside coordinates the local term applies the complementary
    pair projection independently on every outside vector. Its kernel is
    exactly the pair trace range. Membership in that range supplies
    $X$. Conversely the displayed representation places every outside
    vector in that kernel, so the identity-extended term vanishes.
\end{proof}

\begin{lemma}[Unused physical letters are excluded by the kernel]
    \label{lem:varying_physical_kernel_forces_support}
    \lean{Matrix.physicalPairParentTerm_coeff_eq_zero_of_not_mem_range,
        Matrix.varyingPhysicalParentHamiltonian_apply_eq_zero_iff_terms,
        Matrix.varyingPhysicalParentHamiltonian_pair_apply_eq_zero,
        Matrix.varyingPhysicalParentHamiltonian_coeff_eq_zero_of_not_mem_range}
    \leanok
    \uses{lem:varying_physical_local_boundary_sections,
        lem:varying_physical_parent_embedding_transport}
    Extend site matrices by zero along arbitrary sitewise physical
    injections $e_k:I_k\hookrightarrow J_k$. Suppose $N\ge2$ and the extended
    parent satisfies $\widetilde H\phi=0$. Then $\phi(\sigma)=0$
    whenever some physical letter $\sigma_k$ is outside $e_k(I_k)$.
    Every adjacent local term annihilates $\phi$.
\end{lemma}
\begin{proof}\leanok
    Positivity of the local terms derives each local equation from
    the equation for their sum. At a letter outside the original alphabet,
    its extended site matrix is zero, so every pair trace coefficient
    at that letter is zero. Every site of a chain with $N\ge2$ belongs to an
    adjacent pair, including the final site, which belongs to the final pair.
    The local boundary sections therefore give the claimed vanishing at
    every configuration containing an unused letter.
\end{proof}

\begin{lemma}[The kernel lies in the original physical image]
    \label{lem:varying_physical_kernel_actual_pullback}
    \lean{Matrix.varyingPhysicalParentHamiltonian_eq_extendByZero_pullback_of_eq_zero}
    \leanok
    \uses{lem:varying_physical_kernel_forces_support}
    Under the same hypotheses, every extended-kernel vector is the
    extension of its literal coefficient pullback:
    \begin{align}
        \phi & =E\bigl(\tau\mapsto\phi((e_k(\tau_k))_k)\bigr).
    \end{align}
    Membership in this embedded state space is a conclusion, not a hypothesis.
\end{lemma}
\begin{proof}\leanok
    At a configuration in the embedding image, extension by zero recovers
    exactly its original coefficient. A configuration outside that image has
    a letter outside one of the site images, since sitewise preimages would
    assemble a full dependent preimage. Its coefficient vanishes by the
    kernel support statement, and the extended coefficient is zero too.
\end{proof}

\begin{lemma}[The two common-alphabet parent descriptions have the same kernel]
    \label{lem:uniform_physical_parent_kernel_comparison}
    \lean{Matrix.varyingPhysicalParentHamiltonian_uniform_apply_eq_zero_iff}
    \leanok
    \uses{lem:varying_physical_local_boundary_sections,lem:varying_obc_parent_kernel_sections,lem:varying_physical_kernel_forces_support}
    On a common nonempty physical alphabet, the dependent-coordinate
    open parent and the prefix/pair/suffix open parent have exactly
    the same kernel at every length.
\end{lemma}
\begin{proof}\leanok
    Both local equations say that every fixed-outside pair section has a
    rectangular trace boundary. To compare the coordinate descriptions,
    concatenate a prefix, the active pair, and a suffix. Changing only the
    pair leaves all outside coordinates unchanged. Conversely, split a full
    configuration into its prefix, pair, and suffix; concatenating
    them recovers the configuration. A reference pair from the nonempty
    alphabet fixes the outside section. These two coordinate identifications
    transport the boundary representations in both directions. Positivity
    and the local-kernel characterizations give the full kernel
    equivalence. Canonical normalization derives alphabet nonemptiness in
    the subsequent uniqueness application.
\end{proof}
