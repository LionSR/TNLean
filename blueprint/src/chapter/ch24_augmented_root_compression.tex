\section{The transport root in the original site coordinates}

Retain the augmented site bijection $a:I\to J$ and its replica
configuration bijection $e_k:\mathcal X_I^k\to\mathcal X_J^k$.
Write $\mathcal U_k:\C^{\mathcal X_I^k}\to\C^{\mathcal X_J^k}$
for the unitary $(\mathcal U_k f)(x)=f(e_k^{-1}x)$.
The physical dimension is $q$, and both auxiliary dimensions are $d$.
All replica metrics use the dimension of the same full one-copy system.

\begin{theorem}[Band matrices in the original coordinates]
    \label{thm:al_augmented_band_coordinates}
    \lean{TNLean.PEPS.AreaLaw.Scan.bandMetric_submatrix_augmentedReplicaConfigurationEquiv}
    \uses{def:al_scan_auxiliary_coordinates}
    Suppose $V$ is finite, $q,d,k\ge0$ are integers and $t\in\R$.
    For any three regions $P,Y,F\subseteq J$, put
    $P_I=a^{-1}(P)$, $Y_I=a^{-1}(Y)$ and $F_I=a^{-1}(F)$.
    Define
    \begin{align*}
        A_J(P,Y,F)       & =\bigl((W^J_{P,k})^{-1}(W^J_{F,k})^{-1}W^J_{Y,k}\bigr)^2,       \\
        A_I(P_I,Y_I,F_I) & =\bigl((W^I_{P_I,k})^{-1}(W^I_{F_I,k})^{-1}W^I_{Y_I,k}\bigr)^2.
    \end{align*}
    Then $A_J(P,Y,F)=\mathcal U_k A_I(P_I,Y_I,F_I)\mathcal U_k^*$.
    The regions need not form a partition. Empty coordinate spaces and
    zero replicas are included; inverses have the usual total matrix convention.
\end{theorem}
\begin{proof}
    \uses{thm:al_augmented_replica_metric_coordinates}
    Transport each of the three replica metrics by the same unitary.
    Matrix inverses and products are preserved by this conjugation,
    giving the identity after squaring.
\end{proof}

\begin{theorem}[The same compressed root and filtered vector]
    \label{thm:al_augmented_root_compression}
    \lean{TNLean.PEPS.AreaLaw.Scan.rootPath_augmented_compressed_bandProduct}
    Let $q,d>0$, $t\ge0$, and let $D$ be admissible transport data on $J$
    with $K$ bands. Thus its history and conditional weights are positive,
    its old regions partition $J$, and every move transfers a part of the
    old middle region to one side. Assume cross-band commutation on the
    symmetric replica space. Fix $k,N\ge0$, $p\in\R$, and a supplied
    matrix $Z:\C^N\to\C^{\mathcal X_J^k}$ satisfying
    \begin{align*}
        Z^*Z & =I,     \\
        ZZ^* & =\Pi_k,
    \end{align*}
    where $\Pi_k$ is the symmetric projection.
    Let $v\in\C^{\mathcal X_J^k}$ satisfy $\Pi_kv=v$.

    Let $T$ be the original transport tree at the parameter $p$ clamped
    to $[0,1]$. For each of its terminal labels $j$ and bands $g$, pull
    back the actual partition $(P_{j,g},Y_{j,g},F_{j,g})$ through $a$,
    and let $A^I_{j,g}$ be its band matrix in the original coordinates.
    Put $Z_I=\mathcal U_k^*Z$, $v_I=\mathcal U_k^*v$, and
    \begin{align}
        B_{j,g} & =Z_I^*A^I_{j,g}Z_I,\nonumber                         \\
        M       & =\prod_{g=0}^{K-1}T\bigl((B_{j,g})_j\bigr),\nonumber \\
        w       & =M^{-1/4}Z_I^*v_I.
        \label{eq:al_augmented_compressed_product}
    \end{align}
    The product in (\ref{eq:al_augmented_compressed_product}) has the
    original band order. Write $L(p)$ for the actual transport root.
    Then $M$ is positive definite, and for every $r\in\R$,
    \begin{align}
        L(p)Z              & =ZM,\nonumber      \\
        L(p)^rZ            & =ZM^r,\nonumber    \\
        \|L(p)^{-1/4}v\|^2 & =\|w\|^2,\nonumber \\
        \frac{L(p)^{-1/4}v}{\|L(p)^{-1/4}v\|}
                           & =Z\frac{w}{\|w\|}.
        \label{eq:al_augmented_filtered_vector}
    \end{align}
    A zero denominator gives the zero vector. No nonzero-vector premise
    is needed for these equalities.
\end{theorem}
\begin{proof}
    \uses{thm:al_augmented_band_coordinates}
    The coordinate identity gives
    $Z_I^*A^I_{j,g}Z_I=Z^*A^J_{j,g}Z$ and $Z_I^*v_I=Z^*v$.
    Apply the compressed-root identity to the same supplied $Z$ and
    the same clamped tree. It gives positivity of $M$ and $L(p)Z=ZM$.
    Positivity of the actual terminal inputs also gives positivity of
    $L(p)$. Functional calculus therefore yields $L(p)^rZ=ZM^r$.

    Since $ZZ^*v=\Pi_kv=v$, setting $r=-1/4$ gives
    $L(p)^{-1/4}v=Zw$. The isometry identity $Z^*Z=I$ preserves the
    norm. Dividing by that common norm proves the last identity in
    (\ref{eq:al_augmented_filtered_vector}), including the zero case.
    This is the same filtering normalization as in
    \cite[Section~6]{OpenAI2026AreaLaw}.
\end{proof}
