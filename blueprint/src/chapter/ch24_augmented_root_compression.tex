\section{The root operator in the original site coordinates}

Retain the augmented site bijection $a:I\to J$ and its replica
configuration bijection $e_k:\mathcal X_I^k\to\mathcal X_J^k$.
Write $\mathcal U_k:\C^{\mathcal X_I^k}\to\C^{\mathcal X_J^k}$
for the unitary $(\mathcal U_k f)(x)=f(e_k^{-1}x)$.
The physical dimension is $q$, and both auxiliary dimensions are $d$.
All replica metrics use the dimension of the full one-copy system.

\begin{lemma}[Band matrices in the original coordinates]
    \label{thm:al_augmented_band_coordinates}
    \lean{TNLean.PEPS.AreaLaw.Scan.bandMetric_submatrix_augmentedReplicaConfigurationEquiv}
    \leanok
    \uses{def:al_scan_auxiliary_coordinates}
    Let $V$ be finite, let $q,d,k\ge0$ be integers, and let $t\in\R$.
    For any three regions $P,Y,F\subseteq J$, put
    $P_I=a^{-1}(P)$, $Y_I=a^{-1}(Y)$, and $F_I=a^{-1}(F)$.
    Define
    \begin{align*}
        A_J(P,Y,F)       & =\bigl((W^J_{P,k})^{-1}(W^J_{F,k})^{-1}W^J_{Y,k}\bigr)^2,       \\
        A_I(P_I,Y_I,F_I) & =\bigl((W^I_{P_I,k})^{-1}(W^I_{F_I,k})^{-1}W^I_{Y_I,k}\bigr)^2.
    \end{align*}
    Then $A_J(P,Y,F)=\mathcal U_k A_I(P_I,Y_I,F_I)\mathcal U_k^*$.
    The regions $P,Y,F$ need not partition $J$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_replica_metric_coordinates}
    Conjugation by $\mathcal U_k$ maps each of the three replica metrics
    from $I$ to $J$. Since unitary conjugation commutes with matrix inversion
    and multiplication, the identity follows after squaring.
\end{proof}

\begin{lemma}[Compressed root and filtered vector in original coordinates]
    \label{thm:al_augmented_root_compression}
    \lean{TNLean.PEPS.AreaLaw.Scan.rootPath_augmented_compressed_bandProduct}
    \leanok
    Let $q,d>0$ and $t\ge0$, and let $D$ be admissible band-tree data on $J$
    with $K$ bands: its history and conditional weights are positive, its
    initial regions partition $J$, and every move transfers a subset of the
    middle region to one side. Assume that the band metrics commute on the
    symmetric replica space. Fix $k,N\ge0$, $p\in\R$, and a
    matrix $Z:\C^N\to\C^{\mathcal X_J^k}$ satisfying
    \begin{align*}
        Z^*Z & =I,     \\
        ZZ^* & =\Pi_k,
    \end{align*}
    where $\Pi_k$ is the symmetric projection.
    Let $v\in\C^{\mathcal X_J^k}$ satisfy $\Pi_kv=v$.

    Let $T$ be the evaluation tree at the parameter $p$ clamped
    to $[0,1]$. For each terminal label $j$ and band $g$, pull
    back the partition $(P_{j,g},Y_{j,g},F_{j,g})$ along $a$,
    and let $A^I_{j,g}$ be its band matrix in the coordinates $I$.
    Put $Z_I=\mathcal U_k^*Z$, $v_I=\mathcal U_k^*v$, and
    \begin{align}
        B_{j,g} & =Z_I^*A^I_{j,g}Z_I,\nonumber                         \\
        M       & =\prod_{g=0}^{K-1}T\bigl((B_{j,g})_j\bigr),\nonumber \\
        w       & =M^{-1/4}Z_I^*v_I.
        \label{eq:al_augmented_compressed_product}
    \end{align}
    The product in (\ref{eq:al_augmented_compressed_product}) is taken in the
    band order $g=0,\ldots,K-1$. Write $L(p)$ for the root operator on $J$.
    Then $M$ is positive definite, and for every $r\in\R$,
    \begin{align}
        L(p)Z              & =ZM,\nonumber      \\
        L(p)^rZ            & =ZM^r,\nonumber    \\
        \|L(p)^{-1/4}v\|^2 & =\|w\|^2,\nonumber \\
        \frac{L(p)^{-1/4}v}{\|L(p)^{-1/4}v\|}
                           & =Z\frac{w}{\|w\|}.
        \label{eq:al_augmented_filtered_vector}
    \end{align}
    Here $x/\|x\|$ denotes the zero vector when $x=0$; with this convention
    the last identity also holds for $v=0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_band_coordinates}
    The coordinate relation between $A^I_{j,g}$ and $A^J_{j,g}$ gives
    $Z_I^*A^I_{j,g}Z_I=Z^*A^J_{j,g}Z$ and $Z_I^*v_I=Z^*v$.
    The compressed-root identity for $Z$ and the clamped tree $T$
    implies that $M$ is positive definite and $L(p)Z=ZM$.
    Because the terminal band matrices are positive definite, $L(p)$ is
    positive definite as well, and functional calculus yields $L(p)^rZ=ZM^r$.

    Since $ZZ^*v=\Pi_kv=v$, setting $r=-1/4$ gives
    $L(p)^{-1/4}v=Zw$. The isometry condition $Z^*Z=I$ gives
    $\|L(p)^{-1/4}v\|=\|w\|$, and dividing by this common norm yields
    (\ref{eq:al_augmented_filtered_vector}).
    This matches the filtering normalization in
    \cite[Section~7]{OpenAI2026AreaLaw}.
\end{proof}
