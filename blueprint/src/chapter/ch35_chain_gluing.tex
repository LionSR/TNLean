\section{Gluing a left state at a cut of the chain}
\label{sec:chain_gluing}

Fix the cut after $k$ sites of a chain. Splitting off the last site left of
the cut, a configuration has coordinates $((x,y),z)$, with $x$ on the first
$k-1$ sites, $y$ on site $k-1$ and $z$ on the sites right of the cut. This is
the setting of the boundary contractions and the gluing lemma of Landau,
Vazirani and Vidick~\cite[lines 409--447]{Landau2015Polynomial}, proved in the
companion quantum-channel
volume~\cite[``Gluing'', ``The leading eigenvector of a near-optimal left state'',
    ``Three-way cut coordinates'']{QICLean2026Blueprint}.

\begin{definition}[Cut coordinates at a cut of the chain]
    \label{def:chain_gluing_cut}
    \lean{TNLean.MPS.AreaLaw.CutLeft,
        TNLean.MPS.AreaLaw.CutSite, TNLean.MPS.AreaLaw.CutRight,
        TNLean.MPS.AreaLaw.gluingCut, TNLean.MPS.AreaLaw.leftCutOperator,
        TNLean.MPS.AreaLaw.rightCutOperator}
    \leanok
    The coordinates $((x,y),z)$ of a configuration of the chain at the cut
    after $k$ sites. For an operator $H_L$ supported on the first $k$ sites,
    $H_L'$ is its restriction to them; for an operator $K$ supported off the
    first $k-1$ sites, $K'$ is its restriction to site $k-1$ and the sites right
    of the cut.
\end{definition}

\begin{definition}[Joining the left coordinates]
    \label{def:chain_gluing_left_coordinates}
    \lean{TNLean.MPS.AreaLaw.gluingLeftCoordinates}
    \leanok
    \uses{def:chain_gluing_cut}
    Let $L,q,k,m$ be natural numbers. Put $A=[0,k-1)$ and $B=[0,k)$,
    intersecting both intervals with the chain of $L$ sites, and write
    $\mathcal C_D=\{0,\ldots,q-1\}^{D}$ for the configurations of a region $D$.
    Restriction gives a bijection
    $r:\mathcal C_B\to\mathcal C_A\times\mathcal C_{B\setminus A}$.
    Define the linear isometric equivalence
    \begin{align}
        U_m:\mathbb C^{(\mathcal C_A\times\mathcal C_{B\setminus A})\times\{0,\ldots,m-1\}}
                     & \longrightarrow\mathbb C^{\mathcal C_B\times\{0,\ldots,m-1\}}, &
        (U_m u)(b,j) & =u(r(b),j). \notag
    \end{align}
    These are the left coordinates of the vectors used in the size-trimming
    program and Lemma~2.5 of~\cite[source lines 472--478 and 493--505]{Landau2015Polynomial}.
\end{definition}

\begin{theorem}[Dimension of the intervening configuration space]
    \label{thm:chain_gluing_site_dimension}
    \lean{TNLean.MPS.AreaLaw.card_cutSite_le}
    \leanok
    \uses{def:chain_gluing_cut}
    For every natural cut position $k$ and every $q\ge1$, the configuration
    space of $B\setminus A$ has at most $q$ elements, where
    $A=[0,k-1)$ and $B=[0,k)$ are intersected with the chain.
\end{theorem}
\begin{proof}
    \leanok
    If $a,b\in B\setminus A$, then $k-1\le a,b<k$, hence $a=b$.
    Thus $|B\setminus A|\le1$ and
    $|\mathcal C_{B\setminus A}|=q^{|B\setminus A|}\le q$.
\end{proof}

\begin{theorem}[Transport of the left support projection]
    \label{thm:chain_gluing_left_projection}
    \lean{TNLean.MPS.AreaLaw.gluingLeftCoordinates_spanProjection_eq_iff}
    \leanok
    \uses{def:chain_gluing_left_coordinates}
    Let $S\subseteq\mathbb C^{\mathcal C_B}$, let $P$ be the orthogonal
    projection onto $\operatorname{Span}(S)$, and define $P^r$ by
    $P^r(r(b),r(b'))=P(b,b')$. For every vector $u$ on the two left regions
    and a bond of dimension $m$,
    \begin{align}
        (P^r\otimes I_m)u=u
        \quad & \Longleftrightarrow\quad
        (P\otimes I_m)U_m u=U_m u. \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Changing the row and column coordinates of $P\otimes I_m$ by
    $r\times\mathrm{id}$ gives $P^r\otimes I_m$. Composing with this
    bijection gives the asserted equivalence of the fixed-vector conditions.
\end{proof}

\begin{theorem}[Extension after joining the left coordinates]
    \label{thm:chain_gluing_left_extension}
    \lean{TNLean.MPS.AreaLaw.gluingLeftCoordinates_rightApply}
    \leanok
    \uses{def:chain_gluing_left_coordinates}
    Let $W:\mathbb C^m\to\mathbb C^{\mathcal C_{B^c}}$ be any linear map
    and let $u$ be a vector on the two left regions and the bond.
    Extending $u$ by $I\otimes W$ in the three-way cut and then reading it
    on the chain gives the same vector as extending $U_m u$ in the cut at
    $B$. For a chain configuration $c$, their common value is
    \begin{align}
        v(c)=\sum_{j=0}^{m-1}(U_m u)(c|_B,j)\,W(c|_{B^c},j). \notag
    \end{align}
    No isometry assumption on $W$ is needed.
\end{theorem}
\begin{proof}
    \leanok
    The three-way cut sends $c$ to $(r(c|_B),c|_{B^c})$.
    If $C$ is the coefficient matrix of $U_m u$, its right extension has
    coefficient matrix $CW^{\mathsf T}$. Expanding this product gives
    the stated sum, which is also the value of $(I\otimes W)u$ in the
    three-way coordinates.
\end{proof}

\begin{theorem}[A trace-norm net on a chain boundary]
    \label{thm:chain_boundary_trace_norm_net}
    \lean{TNLean.MPS.AreaLaw.exists_finset_boundary_rectangularTraceNorm_net}
    \leanok
    \uses{def:chain_gluing_left_coordinates}
    Let $L,q,k,m,D$ be natural numbers with $q\ge1$ and $m\le D$, and let
    $\eta>0$. Use the clipped regions $A=\lbrack0,k-1)$ and $B=\lbrack0,k)$ of
    Definition~\ref{def:chain_gluing_left_coordinates}, and put
    $K=\mathcal C_{B\setminus A}\times\{0,\ldots,m-1\}$.
    There is a finite set $\mathcal N\subseteq\mathbb C^{K\times K}$ such that
    \begin{align}
        |\mathcal N|\le
        \bigl(2\lceil(qD)^2/\eta\rceil+1\bigr)^{2(qD)^2}. \notag
    \end{align}
    Every matrix $Y\in\mathbb C^{K\times K}$ with $\|Y\|_1\le1$ admits an
    $X\in\mathcal N$ with $\|Y-X\|_1\le\eta$.
    This is the boundary-space specialization of
    \cite[Lemma~2.4, source lines 451--460]{Landau2015Polynomial}, with the
    corrected grid and quadratic exponent recorded in~\cite{gap:lvv13_trace_norm_net}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:chain_gluing_site_dimension}
    Theorem~\ref{thm:chain_gluing_site_dimension} gives
    $c=|K|=|\mathcal C_{B\setminus A}|m\le qD$.
    Apply the trace-norm net theorem of the companion
    volume~\cite[``A trace-norm net'']{QICLean2026Blueprint} to matrices of size $c$.
    It gives the covering property and
    \begin{align}
        |\mathcal N|
        \le\bigl(2\lceil c^2/\eta\rceil+1\bigr)^{2c^2}
        \le\bigl(2\lceil(qD)^2/\eta\rceil+1\bigr)^{2(qD)^2}. \notag
    \end{align}
    The second inequality follows from $c^2\le(qD)^2$ and monotonicity of
    the ceiling and of natural powers with base at least one.
\end{proof}

\begin{theorem}[The Hamiltonian at a cut]
    \label{thm:chain_gluing_split}
    \lean{TNLean.MPS.AreaLaw.reindex_gluingCut_eq}
    \leanok
    \uses{def:chain_gluing_cut}
    If $H_L$ is supported on the first $k$ sites and $K$ off the first $k-1$
    sites, then in the cut coordinates $H_L+K=H_L'\otimes1+1\otimes K'$.
\end{theorem}

\begin{proof}
    \leanok
    An operator supported on a set of sites acts as the identity on the
    configuration of the other sites. Thus, in the cut coordinates, with
    the reassociation $((x,y),z)\leftrightarrow(x,(y,z))$ for $K$,
    \begin{align}
        H_L & =H_L'\otimes1, \\
        K   & =1\otimes K'.
    \end{align}
    Adding gives the asserted decomposition of $H_L+K$.
\end{proof}

\begin{theorem}[The leading eigenvector at a cut of the chain]
    \label{thm:chain_leading_eigenvector}
    \lean{TNLean.MPS.AreaLaw.exists_leading_eigenvector_chain}
    \leanok
    \uses{def:chain_gluing_cut}
    Let $H=H_L+K$ with $H_L$ supported on the first $k$ sites and $K$ off the
    first $k-1$ sites have a gapped ground state $\Gamma$ with energy $E_0$ and
    gap $\Delta$. Let $W$ be an isometry from a bond $\mathbb C^m$ into the
    sites right of the cut, and let $w=(1\otimes W)x$ in the cut coordinates
    have energy at most $E_0+e$. Let $\sigma$ be a density matrix on the first
    $k$ sites and the bond with
    $\operatorname{Re}\operatorname{tr}(\sigma(H_L'\otimes1))\le
        \operatorname{Re}\operatorname{tr}(\lvert x\rangle\langle x\rvert(H_L'\otimes1))$
    and
    $\lVert\operatorname{tr}_{[1,k-1]}\sigma-\operatorname{tr}_{[1,k-1]}
        \lvert x\rangle\langle x\rvert\rVert_1\le\tau$. If
    $4(e+\lVert K\rVert\tau)<\Delta$, then $\sigma$ has a unit eigenvector $u$
    with eigenvalue larger than $\tfrac12$ such that $(1\otimes W)u$, read on
    the chain, has energy at most $E_0+2(e+\lVert K\rVert\tau)$ and overlap at
    least $1-2(e+\lVert K\rVert\tau)/\Delta$ with $\Gamma$.

    This is Lemma~l:leading of~\cite[lines 493--505]{Landau2015Polynomial} on
    the chain, with $H_L=H_1+\cdots+H_{i-1}$ and $K=H_i+H_R$; the energy excess
    $e$ of the trimmed state of Lemma~l:within is corrected
    in~\cite{gap:lvv13_trimming_energy}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:chain_gluing_split}
    In the cut coordinates the Hamiltonian, the ground state and $w$ satisfy
    the hypotheses of the leading-eigenvector theorem of the companion volume,
    with $\lVert K'\rVert\le\lVert K\rVert$. This comparison follows from
    invariance of the operator norm under a change of coordinates and
    $\lVert1\otimes K'\rVert=\lVert K'\rVert$, since the first configuration
    factor is nonempty~\cite[``Operator norm after adjoining an identity factor'']{QICLean2026Blueprint}.
    Let $E$ be the cut-coordinate bijection and $U$ the
    unitary map $U\psi=\psi\circ E^{-1}$. For chain vectors $\phi,\psi$ and
    an operator $M$,
    \begin{align}
        \langle U\phi,U\psi\rangle
         & =\langle\phi,\psi\rangle,
        \label{eq:chain_gluing_inner} \\
        \langle U\phi,(UMU^*)U\psi\rangle
         & =\langle\phi,M\psi\rangle.
        \label{eq:chain_gluing_energy}
    \end{align}
    Applying~(\ref{eq:chain_gluing_inner}) and~(\ref{eq:chain_gluing_energy})
    to the ground state and the eigenvector
    extension carries the energy and overlap bounds back to the chain.
\end{proof}

\begin{theorem}[Bounds for every dominant eigenvector on the chain]
    \label{thm:chain_leading_eigenvector_uniform_bounds}
    \lean{TNLean.MPS.AreaLaw.leading_eigenvector_bounds_chain}
    \leanok
    \uses{thm:chain_leading_eigenvector}
    Under the hypotheses of Theorem~\ref{thm:chain_leading_eigenvector},
    let $u$ be any unit eigenvector of $\sigma$ with real eigenvalue
    $\mu>1/2$. Extend $u$ by $I\otimes W$ and read the resulting vector
    $v$ in the chain coordinates. Then
    \begin{align}
        \operatorname{Re}\langle v,(H_L+K)v\rangle
         & \le E_0+2(e+\|K\|\tau),\notag     \\
        |\langle\Gamma,v\rangle|
         & \ge1-2(e+\|K\|\tau)/\Delta.\notag
    \end{align}
    This is an auxiliary statement for the arbitrary choice of leading
    eigenvector in~\cite[Lemma~2.5, source lines 493--505]{Landau2015Polynomial}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_leading_eigenvector}
    Theorem~\ref{thm:chain_leading_eigenvector} supplies a unit eigenvector
    $u_0$ with eigenvalue greater than one half and the two bounds for its
    physical extension $v_0$.
    Uniqueness up to a unit phase in the companion
    volume~\cite[``Uniqueness of an eigenvector above one half'']{QICLean2026Blueprint}
    gives $u=zu_0$ with $|z|=1$. The physical extension is linear, so
    $v=zv_0$ and
    \begin{align}
        \langle v,(H_L+K)v\rangle & =\langle v_0,(H_L+K)v_0\rangle,\notag \\
        |\langle\Gamma,v\rangle|  & =|\langle\Gamma,v_0\rangle|.\notag
    \end{align}
\end{proof}

\begin{theorem}[The trimmed witness at a cut of the chain]
    \label{thm:chain_trimmed_witness}
    \lean{TNLean.MPS.AreaLaw.exists_trimmed_witness_chain}
    \leanok
    \uses{def:chain_gluing_cut}
    Let $H=H_L+H_R+K$ with $H_L$ supported on the first $k$ sites, $H_R$ on
    the others, $0\le K\le1$ and $H\ge E_0$. Let $\psi$ be a unit vector with
    $\operatorname{Re}\langle\psi,H\psi\rangle\le E_0+\delta$ whose cut matrix
    $C$ satisfies $P_VC=C$ for a matrix $P_V$ on the first $k$ sites, and let a
    unit vector of rank at most $D$ across the cut have overlap at least
    $1-\eta$ with $\psi$, where $2\eta<1$. Then there is a unit vector $w$ of
    rank at most $D$ across the cut, with
    $\operatorname{Re}\langle w,Hw\rangle\le E_0+(\delta+2\sqrt{2\eta})/(1-2\eta)$
    and $P_VC_w=C_w$, together with an isometry $W$ from $\mathbb C^m$, $m\le D$,
    into the sites right of the cut and a vector $x$ with $w=(1\otimes W)x$ in
    the cut coordinates of Definition~\ref{def:chain_gluing_cut}, whose left
    factor is again in the range of $P_V$.

    This is Lemma~l:within of~\cite[lines 509--515]{Landau2015Polynomial} on
    the chain, with $P_V$ the projection onto the span of a viable set. The
    energy bound is the corrected one; the energy
    $\varepsilon_0+6\sqrt{c_\varepsilon}$ printed there is not established by its
    argument~\cite{gap:lvv13_trimming_energy}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:chain_gluing_cut}
    In the cut coordinates after $k$ sites the Hamiltonian is
    $A\otimes1+1\otimes A'+K'$ with $0\le K'\le1$, and the trimmed witness of
    the companion volume~\cite[``A trimmed low-energy witness on a
        subspace'']{QICLean2026Blueprint} gives $w$, $W$ and $x$ there. The cut
    coordinates after $k$ sites, followed by the split of the first $k$ sites
    into the first $k-1$ and site $k-1$, are the coordinates of
    Definition~\ref{def:chain_gluing_cut}, which carries $w=(1\otimes W)x$ and the
    range condition on $x$ over.
\end{proof}

\begin{theorem}[A viable witness with a constant rank bound]
    \label{thm:lvv_viable_trimmed_witness_library_bond}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.exists_trimmed_witness_of_isViable}
    \leanok
    \uses{def:chain_model,def:lvv_viable_set,def:chain_constant_bond,def:chain_gluing_cut}
    Let $L\ge1$ and $q\ge2$, and let $H$ be a range-one Hamiltonian on the
    chain whose interaction terms are positive semidefinite and have
    operator norm at most one. Let $\Gamma$ be a gapped ground vector of
    energy $E_0$ and gap $\Delta>0$, and put $c=(\Delta/169)^2$.
    Fix a natural number $k$, and let $\mathcal H_R$ be the Hilbert space
    of the sites after the first $k$ sites.
    If $S$ is $(k,c/L)$-viable for $\Gamma$, there is a unit vector
    $w\in\operatorname{Span}(S)\otimes\mathcal H_R$ such that
    \begin{align}
        \operatorname{rank}M_k(w)            & \le B_c,                 &
        \operatorname{Re}\langle w,Hw\rangle & \le E_0+6\sqrt c. \notag
    \end{align}
    There are also $m\le B_c$, an isometry $W:\mathbb C^m\to\mathcal H_R$
    and a vector $x$ on the left sites and $\mathbb C^m$ such that, in the
    cut coordinates of Definition~\ref{def:chain_gluing_cut},
    \begin{align}
        w & =(I\otimes W)x, & (P_S\otimes I)x & =x, \notag
    \end{align}
    where $P_S$ is the orthogonal projection onto $\operatorname{Span}(S)$,
    expressed in the corresponding left coordinates.
    Here $B_c$ is the bound of Definition~\ref{def:chain_constant_bond} at
    $R=J=1$. The energy and support conclusions are those of
    \cite[Lemma~2.6, source lines 509--515]{Landau2015Polynomial}; the
    dependence of $B_c$ on $q$ and $\Delta$ differs from the bound in
    \cite[Lemma~1.7, source lines 342--345]{Landau2015Polynomial};
    see~\cite{gap:lvv13_constant_bond}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_range_one_gap_upper,thm:chain_range_one_overlap_energy,
        thm:chain_range_one_cut_decomposition,thm:chain_constant_bond_overlap,
        thm:lvv_viable_projection,thm:chain_trimmed_witness,
        thm:chain_numerical_trim_bound,thm:chain_numerical_trim_constant}
    The gap bound gives $\Delta\le4$, hence $0<c\le1/1600$.
    A viability witness $\psi$ has energy at most $E_0+8c$ and left support
    in $\operatorname{Span}(S)$. The normalized ground vector trimmed at
    $B_c$ gives a unit vector $u$ of rank at most $B_c$ across the cut and
    $|\langle u,\Gamma\rangle|\ge1-c$. Since
    $|\langle\psi,\Gamma\rangle|\ge1-c/L$, the overlap inequality gives
    $|\langle u,\psi\rangle|\ge1-2(c+c/L)\ge1-4c$.
    The Hamiltonian decomposition at the cut has $0\le K\le I$.
    Theorem~\ref{thm:chain_trimmed_witness}, with $P_V$ the projection onto
    $\operatorname{Span}(S)$, therefore gives a unit vector $w$ with the
    same left support, rank at most $B_c$, and energy at most
    \begin{align}
        E_0+\frac{8c+4\sqrt{2c}}{1-8c}\le E_0+6\sqrt c, \notag
    \end{align}
    by Theorem~\ref{thm:chain_numerical_trim_bound}.
    The same application of Theorem~\ref{thm:chain_trimmed_witness} gives
    $W^\dagger W=I$, $m\le B_c$, $w=(I\otimes W)x$ and the retained support
    condition $(P_S\otimes I)x=x$.
\end{proof}
