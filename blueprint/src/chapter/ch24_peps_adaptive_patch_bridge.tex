\section{Square patches from nested cylinders}
\label{sec:peps_adaptive_patch_bridge}
The statements below supply the last step of the proof of
Proposition~4.1 of \cite{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
lines 563--603: a vector in the sum of nested cylinder spaces is turned into
square cylinder data at the same radii.

\begin{theorem}[Square samples in integer coordinates]
    \label{thm:peps_square_sample_integer_coordinates}
    \lean{TNLean.PEPS.Approximation.squareSample_integerCoordinates,
        TNLean.PEPS.Approximation.integerCoordinates_add_one_eq_latticePos}
    \leanok
    For a finite set $\Lambda\subseteq\Z^2$ with its sites at their integer
    positions, the sample of the closed sup-norm square of center $c$ and
    radius $r$ is the set of sites $x\in\Lambda$ with
    $|x_1-c_1|\le r$ and $|x_2-c_2|\le r$, for every center and radius.
    Translating the integer positions of the open $L\times L$ square, which
    start at zero, by $(1,1)$ gives the lattice positions of
    \cite{OpenAI2026PolynomialPEPS}, \texttt{00-introduction.tex}, lines 21--25.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_adaptive_patch_constraint,
        def:peps_closed_square_sample, thm:area_law_domain_identifications}
    The sup-norm distance is the larger of the two coordinate distances.
\end{proof}

\begin{theorem}[Square patches from nested cylinders]
    \label{thm:peps_square_patch_of_nested_cylinders}
    \lean{TNLean.PEPS.EncodedFrame.exists_squarePatch_of_mem_iSup_dependentRegionCylinder}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_regional_cylinder}
    Let $0<u$ and let $u\le r_0\le\cdots\le r_{m-1}\le2u$. For each $j$ let $S_j$ be
    a subspace of the Hilbert space of the sample $D_j$ of the square of radius
    $r_j$ about $z$. If a vector $v$ lies in the sum of the cylinder spaces
    $S_j\otimes\mathcal H_{D_j^c}$, then for every vector $\Omega$ there are
    square cylinder data at $(z,u)$ with the radii $r_0,\ldots,r_{m-1}$, with
    $\sum_jd_j\le\sum_j\dim S_j$, and with
    \begin{align}
        \lVert(\id-\Pi)\Omega\rVert\le\lVert\Omega-v\rVert ,
    \end{align}
    where $\Pi$ is the sum of the cylinder projectors. No assumption is made on
    repeated radii, zero subspaces or an empty family.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_frame_square_patch,
        def:peps_adaptive_patch_constraint,
        def:peps_nested_cylinder_images,
        thm:peps_nested_cylinder_orthogonalization,
        thm:peps_nested_cylinder_projector_sum,
        thm:peps_cylinder_projector,
        thm:peps_regional_extension_algebra,
        def:peps_hole_encoder,
        thm:peps_coordinate_reset_gram,
        def:peps_coordinate_projector,
        def:peps_patch_canonical_marginal}
    Write $P_X$ for the orthogonal projection onto a subspace $X$. Let $W_j$ be
    the sum of the earlier spaces extended to $D_j$, that is, the span of the
    ranges of $P_{S_i}\otimes\id_{D_j\setminus D_i}$ for $i<j$, a subspace of
    $\mathcal H_{D_j}$. The $j$-th innovation is the projected image, formed
    inside $\mathcal H_{D_j}$,
    \begin{align}
        T_j=(\id-P_{W_j})S_j ,
    \end{align}
    which need not equal $S_j\cap W_j^{\perp}$. Being an image of $S_j$, it
    satisfies $\dim T_j\le\dim S_j$. Let $d_j=\dim T_j$, and let
    $v_{j0},\ldots,v_{j,d_j-1}$ be an orthonormal basis of $T_j$. The
    cylindrical extensions
    $\bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id$
    have mutually orthogonal ranges, and these ranges span the sum of the
    cylinder spaces $S_j\otimes\mathcal H_{D_j^c}$. Hence
    \begin{align}
        \Pi=\sum_{j=0}^{m-1}\sum_{\ell=0}^{d_j-1}
        \bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id
    \end{align}
    is the orthogonal projection onto that sum. Since $v$ lies in its range,
    \begin{align}
        \lVert(\id-\Pi)\Omega\rVert
        =\lVert(\id-\Pi)(\Omega-v)\rVert
        \le\lVert\Omega-v\rVert ,
        \qquad
        \sum_jd_j\le\sum_j\dim S_j
    \end{align}
    (\texttt{03-patches.tex}, lines 563--603).
\end{proof}

\begin{theorem}[Exact square patches at bounded radius]
    \label{thm:peps_bounded_radius_square_patch}
    \lean{TNLean.PEPS.Approximation.exists_squarePatch_integerCoordinates_of_radius_le_four}
    \leanok
    \uses{def:peps_frame_square_patch}
    Let $\Lambda\subseteq\Z^2$ be finite with its sites at their integer
    positions, let $q\ge1$, $z\in\R^2$ and $0<u\le4$. For every vector $\Omega$
    there are square cylinder data at $(z,u)$ with
    \begin{align}
        \sum_jd_j\le q^{81},
        \qquad
        (\id-\Pi)\Omega=0 .
    \end{align}
    The domain and the sample may be empty.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_square_patch_of_nested_cylinders,
        thm:peps_square_sample_integer_coordinates,
        thm:peps_closed_square_card,
        def:peps_regional_cylinder}
    Apply Theorem~\ref{thm:peps_square_patch_of_nested_cylinders} with one
    radius $r_0=u$, the whole space $S_0=\mathcal H_{D_0}$ and $v=\Omega$, which
    lies in the cylinder space $S_0\otimes\mathcal H_{D_0^c}$. The error is
    then at most $\lVert\Omega-\Omega\rVert=0$, and
    \begin{align}
        \sum_jd_j\le\dim\mathcal H_{D_0}=q^{\lvert D_0\rvert}\le q^{81},
    \end{align}
    since a closed square of radius at most four contains at most $81$ integer
    points (\cite{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex}, lines
    53--54).
\end{proof}

\begin{definition}[The constants of the adaptive patch constraint]
    \label{def:peps_adaptive_patch_constants}
    \lean{TNLean.PEPS.Approximation.adaptivePatchPrefactor,
        TNLean.PEPS.Approximation.adaptivePatchExponent}
    \leanok
    \uses{def:peps_nested_patch_radius_exponent}
    For an integer $q$ and reals $J,\Delta,B_0,p,C_A$, put
    \begin{align}
        C(q,B_0)=\max\{q^{81},1+B_0/2\},
        \qquad
        c(q,J,\Delta,B_0,p,C_A)=R_0(q,J,\Delta,B_0,p,C_A)+1,
        \label{eq:peps_adaptive_patch_constants}
    \end{align}
    where $R_0$ is the radius exponent of
    Definition~\ref{def:peps_nested_patch_radius_exponent}. These are the
    constants of
    \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
    lines 53--55 and 595--601.
\end{definition}

\begin{lemma}[Adaptive patches from nested cylinders]
    \label{lem:peps_adaptive_patch_supply_core}
    \lean{TNLean.PEPS.Approximation.adaptivePatchSupply_of_closedSquare_cylinders}
    \leanok
    \uses{def:peps_adaptive_patch_constants,def:peps_adaptive_patch_constraint,
        def:peps_closed_square_sample,def:peps_regional_cylinder}
    Let $q\ge1$, $L\ge2$, $B_0>0$ and $R_0\ge0$. Suppose that for each $u$
    there are $m(u)$ nondecreasing radii between $u$ and $2u$, with
    $m(u)\le\frac{B_0}{2}\log L+1$ when $0\le u\le B_0\log L$, and that for
    every centre and every $4\le u\le B_0\log L$ the samples of these radii
    carry subspaces of dimension at most $L^{R_0}$ whose cylinders contain a
    vector within $L^{-p}$ of $\Omega$. Then every centre and every
    $0<u\le B_0\log L$ admit an adaptive square patch with at most
    $C(q,B_0)L^{R_0+1}$ cylinder terms and error at most $L^{-p}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_square_patch_of_nested_cylinders,
        thm:peps_bounded_radius_square_patch,
        thm:peps_square_sample_integer_coordinates,
        thm:peps_square_patch_relabel,
        lem:peps_adaptive_patch_supply,
        thm:peps_patch_count_polynomial,
        def:square_area_law_state}
    Work on the integer domain and translate by $(1,1)$ at the end. If
    $u\ge4$, the square patch built from the cylinders has error at most
    $L^{-p}$ and at most
    $m(u)L^{R_0}\le(1+\frac{B_0}{2})L^{R_0+1}$ terms. If $u<4$, the exact
    patch at bounded radius has error zero and at most $q^{81}$ terms.
\end{proof}

\begin{theorem}[Adaptive patches at a given area-law constant]
    \label{thm:peps_adaptive_patch_supply_of_area_law_at}
    \lean{TNLean.PEPS.Approximation.adaptivePatchPrefactor_pos,
        TNLean.PEPS.Approximation.adaptivePatchExponent_pos,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.adaptivePatchSupply}
    \leanok
    \uses{def:square_entropy_area_law_at,def:peps_adaptive_patch_constants,
        def:peps_adaptive_patch_constraint}
    Let $q\ge2$ be an integer, $J,\Delta>0$ and $B_0,p>0$, and assume the
    square-grid entropy area law for the ground states of the model at $J$ and
    $\Delta$ with constant $C_A$.
    Then the constants $C=C(q,B_0)$ and $c=c(q,J,\Delta,B_0,p,C_A)$ of
    (\ref{eq:peps_adaptive_patch_constants}) are positive, and for every
    $L\ge2$, every original square Hamiltonian $H$ with interaction bound $J$,
    and every unit vector $\Omega$ and real $E_0$ with $H\Omega=E_0\Omega$ and
    $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$, every center and every
    $0<u\le B_0\log L$ admit an adaptive square patch with at most $CL^c$
    cylinder terms and error at most $L^{-p}$. This is
    \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
    lines 26--54 and 534--603, at a given area-law constant.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:peps_adaptive_patch_supply_core,
        thm:peps_nested_patch_cylinder_approximation_explicit,
        def:peps_nested_patches,thm:peps_nested_patch_sizes}
    Let $R_0=R_0(q,J,\Delta,B_0,p,C_A)$, which is positive by
    Theorem~\ref{thm:peps_nested_patch_cylinder_approximation_explicit}. The
    nested patches have radii between $u$ and $2u$, and
    $m=\lfloor u/2\rfloor+1\le\frac{B_0}{2}\log L+1$ for
    $0\le u\le B_0\log L$. The approximation theorem supplies the cylinders
    for $4\le u\le B_0\log L$, and
    Lemma~\ref{lem:peps_adaptive_patch_supply_core} gives the patches with
    $c=R_0+1$.
\end{proof}

\begin{theorem}[Adaptive patches from the square entropy area law]
    \label{thm:peps_adaptive_patch_constraint_of_area_law}
    \lean{TNLean.PEPS.Approximation.SquareEntropyAreaLaw.adaptivePatchConstraint}
    \leanok
    \uses{def:square_entropy_area_law, def:peps_adaptive_patch_constraint}
    If the square-grid entropy area law holds, then the adaptive patch
    constraint holds. This is the implication proved in
    \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
    lines 26--54 and 534--603, with the area law taken as an assumption.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_adaptive_patch_supply_of_area_law_at}
    The area law supplies a constant $C_A\ge0$ at $q$, $J$, $\Delta$. The
    constants $C(q,B_0)$ and $c(q,J,\Delta,B_0,p,C_A)$ of
    Theorem~\ref{thm:peps_adaptive_patch_supply_of_area_law_at} witness the
    adaptive patch constraint.
\end{proof}

\subsection{Adaptive patches at interaction range \texorpdfstring{$R$}{R}}
\label{sec:peps_range_adaptive_patches}

The following statements extend
\cite[Proposition~4.1]{OpenAI2026PolynomialPEPS} to the gapped ground states
of Hamiltonians whose terms have supports of graph diameter at most $R\ge1$.
They are not in the source. They use the patches of
Definition~\ref{def:peps_range_nested_patches} and the filter exponent
$\kappa_R$ of Definition~\ref{def:peps_range_patch_filter_exponent}.

\begin{definition}[The constants of the adaptive patches at range $R$]
    \label{def:peps_range_adaptive_patch_constants}
    \lean{TNLean.PEPS.Approximation.rangeNestedPatchRadiusExponent,
        TNLean.PEPS.Approximation.rangeAdaptivePatchExponent}
    \leanok
    \uses{def:peps_patch_radius_exponent,
        def:peps_range_patch_filter_exponent}
    For integers $q,R$ and reals $J,\Delta,B_0,p,C_A$, let $R_0^{(R)}$ be the
    radius exponent at $C_0=16\max\{1,16C_A\}+\log8$, $B_0$, $p$ and
    $\kappa_R(q,J,\Delta)$, and put $c_R=R_0^{(R)}+1$.
\end{definition}

\begin{theorem}[Adaptive patches at range $R$ at a given area-law constant]
    \label{thm:peps_range_adaptive_patch_supply}
    \lean{TNLean.PEPS.Approximation.rangeNestedPatchRadiusExponent_pos,
        TNLean.PEPS.Approximation.rangeAdaptivePatchExponent_pos,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.rangeNestedPatch_entropy_le,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.uniform_rangeNestedPatch_cylinders,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.rangeAdaptivePatchSupply}
    \leanok
    \uses{def:square_entropy_area_law_at,def:area_law_hamiltonian,
        def:peps_range_nested_patches,def:peps_range_adaptive_patch_constants,
        def:peps_adaptive_patch_constants,def:peps_adaptive_patch_constraint}
    Let $q,R\ge1$ be integers, $J,\Delta>0$ and $B_0,p>0$, and assume the
    square-grid entropy area law with constant $C_A$ for the gapped ground
    states of range-$R$ Hamiltonians on the square with term bound $J$ and gap
    $\Delta$. Then $R_0^{(R)}$ and $c_R$ are positive, and:
    \begin{enumerate}
        \item every range-$R$ patch $K_j$ about every centre at every $u\ge0$
              satisfies $S_{U\Omega}(K_j)\le\max\{1,16C_A\}(1+u)$;
        \item for every $L\ge2$, every such state $\Omega$, every centre and
              every $0\le u\le B_0\log L$, there are subspaces of dimension at
              most $L^{R_0^{(R)}}$ on the range-$R$ patches whose cylinders
              contain a vector within $L^{-p}$ of $U\Omega$;
        \item every centre and every $0<u\le B_0\log L$ admit an adaptive
              square patch with at most $C(q,B_0)L^{c_R}$ cylinder terms and
              error at most $L^{-p}$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_closed_square_entropy_explicit,
        lem:peps_closed_square_cylinder_core,
        lem:peps_adaptive_patch_supply_core,
        thm:peps_range_nested_patch_sizes,
        thm:peps_range_nested_patch_minimizer_marginal_closeness,
        lem:peps_range_patch_energy_parameter_choice}
    The radii $u+s_Rj$ are nondecreasing and lie between $u$ and $2u$, so
    Lemma~\ref{lem:peps_closed_square_entropy_explicit} gives the first item.
    Theorem~\ref{thm:peps_range_nested_patch_minimizer_marginal_closeness} at
    $\kappa_R$ and the first item give the second item by
    Lemma~\ref{lem:peps_closed_square_cylinder_core}. Since $s_R\ge2$,
    $m_R\le u/2+1\le\frac{B_0}{2}\log L+1$, and
    Lemma~\ref{lem:peps_adaptive_patch_supply_core} gives the third item.
\end{proof}
