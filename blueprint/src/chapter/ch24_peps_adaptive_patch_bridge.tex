\section{Square patches from nested cylinders}
\label{sec:peps_adaptive_patch_bridge}
The statements below supply the last step of the proof of
Proposition~4.1 of \cite{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
lines 563--603: a vector in the sum of nested cylinder spaces is turned into
square cylinder data at the same radii.

\begin{lemma}[Square samples in integer coordinates]
    \label{thm:peps_square_sample_integer_coordinates}
    \lean{TNLean.PEPS.Approximation.squareSample_integerCoordinates,
        TNLean.PEPS.Approximation.integerCoordinates_add_one_eq_latticePos}
    \leanok
    For finite $\Lambda\subseteq\Z^2$ at integer coordinates, the closed
    sup-norm square sample of center $c$ and radius $r$ is
    $\{x\in\Lambda:\max\{|x_1-c_1|,|x_2-c_2|\}\le r\}$. Translating
    $\{0,\ldots,L-1\}^2$ by $(1,1)$ recovers the lattice positions of
    \cite{OpenAI2026PolynomialPEPS}, \texttt{00-introduction.tex}, lines 21--25.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_adaptive_patch_constraint,
        def:peps_closed_square_sample, thm:area_law_domain_identifications}
    Immediate from $\|x-c\|_\infty=\max\{|x_1-c_1|,|x_2-c_2|\}$.
\end{proof}

\begin{lemma}[Square patches from nested cylinders]
    \label{thm:peps_square_patch_of_nested_cylinders}
    \lean{TNLean.PEPS.EncodedFrame.exists_squarePatch_of_mem_iSup_dependentRegionCylinder}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_regional_cylinder}
    Let $0<u\le r_0\le\cdots\le r_{m-1}\le2u$, and let $S_j\subseteq\mathcal H_{D_j}$
    be subspaces on the square samples $D_j$ of radii $r_j$ about $z$. If
    $v\in\sum_{j=0}^{m-1}S_j\otimes\mathcal H_{D_j^c}$, then every vector
    $\Omega$ admits square cylinder data at $(z,u)$ with radii
    $r_0,\ldots,r_{m-1}$ and cylinder-projector sum $\Pi$ satisfying
    (with no restriction on repeated radii, $S_j=0$, or $m=0$)
    \begin{align}
        \sum_jd_j                   & \le\sum_j\dim S_j,       &
        \lVert(\id-\Pi)\Omega\rVert & \le\lVert\Omega-v\rVert.
        \label{eq:peps_square_patch_nested_cylinders}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_frame_square_patch,
        def:peps_adaptive_patch_constraint,
        def:peps_nested_cylinder_images,
        thm:peps_nested_cylinder_orthogonalization,
        thm:peps_nested_cylinder_projector_sum,
        thm:peps_cylinder_projector,
        thm:peps_regional_extension_algebra,
        def:peps_hole_encoder,
        thm:peps_coordinate_reset_gram,
        def:peps_coordinate_projector,
        def:peps_patch_canonical_marginal}
    Write $P_X$ for the orthogonal projector onto $X$, and let
    $W_j=\sum_{i<j}\operatorname{ran}(P_{S_i}\otimes\id_{D_j\setminus D_i})\subseteq\mathcal H_{D_j}$.
    The innovation $T_j=(\id-P_{W_j})S_j\subseteq\mathcal H_{D_j}$ satisfies
    $d_j=\dim T_j\le\dim S_j$. Choosing an orthonormal basis
    $(v_{j\ell})_{\ell=0}^{d_j-1}$ of each $T_j$, Lemmas~\ref{thm:peps_nested_cylinder_orthogonalization}
    and \ref{thm:peps_nested_cylinder_projector_sum} show that
    \begin{align}
        \Pi=\sum_{j=0}^{m-1}\sum_{\ell=0}^{d_j-1}
        \bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id \notag
    \end{align}
    is the orthogonal projector onto $\sum_j S_j\otimes\mathcal H_{D_j^c}\ni v$,
    so $\lVert(\id-\Pi)\Omega\rVert=\lVert(\id-\Pi)(\Omega-v)\rVert\le\lVert\Omega-v\rVert$
    (\texttt{03-patches.tex}, lines 563--603).
\end{proof}

\begin{lemma}[Exact square patches at bounded radius]
    \label{thm:peps_bounded_radius_square_patch}
    \lean{TNLean.PEPS.Approximation.exists_squarePatch_integerCoordinates_of_radius_le_four}
    \leanok
    \uses{def:peps_frame_square_patch}
    For finite $\Lambda\subseteq\Z^2$ at integer coordinates, $q\ge1$,
    $z\in\R^2$, and $0<u\le4$, every $\Omega$ admits square cylinder data at
    $(z,u)$ satisfying
    \begin{align}
        \sum_jd_j & \le q^{81}, & (\id-\Pi)\Omega & =0. \notag
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_square_patch_of_nested_cylinders,
        thm:peps_square_sample_integer_coordinates,
        thm:peps_closed_square_card,
        def:peps_regional_cylinder}
    Apply Lemma~\ref{thm:peps_square_patch_of_nested_cylinders} with $m=1$,
    $r_0=u$, $S_0=\mathcal H_{D_0}$, and $v=\Omega$; Lemma~\ref{thm:peps_closed_square_card}
    gives $|D_0|\le81$ for $u\le4$, so $\sum_jd_j\le\dim\mathcal H_{D_0}=q^{|D_0|}\le q^{81}$
    (\cite{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex}, lines 53--54).
\end{proof}

\begin{definition}[The constants of the adaptive patch constraint]
    \label{def:peps_adaptive_patch_constants}
    \lean{TNLean.PEPS.Approximation.adaptivePatchPrefactor,
        TNLean.PEPS.Approximation.adaptivePatchExponent}
    \leanok
    \uses{def:peps_nested_patch_radius_exponent}
    With $R_0$ as in Definition~\ref{def:peps_nested_patch_radius_exponent},
    define the constants of \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}
    (\texttt{03-patches.tex}, lines 53--55 and 595--601):
    \begin{align}
        C(q,B_0)                & =\max\{q^{81},1+B_0/2\},      &
        c(q,J,\Delta,B_0,p,C_A) & =R_0(q,J,\Delta,B_0,p,C_A)+1.
        \label{eq:peps_adaptive_patch_constants}
    \end{align}
\end{definition}

\begin{lemma}[Adaptive patches from nested cylinders]
    \label{lem:peps_adaptive_patch_supply_core}
    \lean{TNLean.PEPS.Approximation.adaptivePatchSupply_of_closedSquare_cylinders}
    \leanok
    \uses{def:peps_adaptive_patch_constants,def:peps_adaptive_patch_constraint,
        def:peps_closed_square_sample,def:peps_regional_cylinder}
    Let $q\ge1$, $L\ge2$, $B_0>0$, and $R_0\ge0$. Suppose each $u\in[0,B_0\log L]$
    has $m(u)\le\frac{B_0}{2}\log L+1$ nondecreasing radii in $[u,2u]$ whose
    samples, for every centre and $u\in[4,B_0\log L]$, carry subspaces
    $S_j$ with $\dim S_j\le L^{R_0}$ and
    $\operatorname{dist}(\Omega,\sum_j S_j\otimes\mathcal H_{D_j^c})\le L^{-p}$.
    Then every centre and $u\in(0,B_0\log L]$ admit an adaptive square patch
    with at most $C(q,B_0)L^{R_0+1}$ cylinder terms and error at most $L^{-p}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_square_patch_of_nested_cylinders,
        thm:peps_bounded_radius_square_patch,
        thm:peps_square_sample_integer_coordinates,
        thm:peps_square_patch_relabel,
        lem:peps_adaptive_patch_supply,
        thm:peps_patch_count_polynomial,
        def:square_area_law_state}
    On the integer domain (translated by $(1,1)$ via
    Lemma~\ref{thm:peps_square_sample_integer_coordinates}),
    Lemma~\ref{thm:peps_square_patch_of_nested_cylinders} for $u\ge4$ gives
    error at most $L^{-p}$ and at most $m(u)L^{R_0}\le(1+B_0/2)L^{R_0+1}$ terms,
    while Lemma~\ref{thm:peps_bounded_radius_square_patch} for $u<4$ gives
    error $0$ and at most $q^{81}$ terms.
\end{proof}

\begin{lemma}[Adaptive patches at a given area-law constant]
    \label{thm:peps_adaptive_patch_supply_of_area_law_at}
    \lean{TNLean.PEPS.Approximation.adaptivePatchPrefactor_pos,
        TNLean.PEPS.Approximation.adaptivePatchExponent_pos,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.adaptivePatchSupply}
    \leanok
    \uses{def:square_entropy_area_law_at,def:peps_adaptive_patch_constants,
        def:peps_adaptive_patch_constraint}
    Let $q\ge2$ and $J,\Delta,B_0,p>0$, and assume the square-grid entropy area
    law at $(J,\Delta)$ with constant $C_A$. Then $C=C(q,B_0)>0$ and
    $c=c(q,J,\Delta,B_0,p,C_A)>0$ in (\ref{eq:peps_adaptive_patch_constants}),
    and for every $L\ge2$, square Hamiltonian $H$ with term bound $J$, and unit
    ground state $\Omega$ ($H\Omega=E_0\Omega$,
    $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$), every center and
    $0<u\le B_0\log L$ admit an adaptive square patch with at most $CL^c$
    cylinder terms and error at most $L^{-p}$
    (\cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex},
    lines 26--54 and 534--603).
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:peps_adaptive_patch_supply_core,
        thm:peps_nested_patch_cylinder_approximation_explicit,
        def:peps_nested_patches,thm:peps_nested_patch_sizes}
    By Lemma~\ref{thm:peps_nested_patch_sizes}, the $m=\lfloor u/2\rfloor+1\le\frac{B_0}{2}\log L+1$
    nested patches have radii in $[u,2u]$;
    Lemma~\ref{thm:peps_nested_patch_cylinder_approximation_explicit} gives
    $R_0>0$ and the cylinder approximation for $4\le u\le B_0\log L$, so
    Lemma~\ref{lem:peps_adaptive_patch_supply_core} yields the patches with
    $c=R_0+1$.
\end{proof}

\subsection{Adaptive patches at interaction range \texorpdfstring{$R$}{R}}
\label{sec:peps_range_adaptive_patches}

The following statements extend
\cite[Proposition~4.1]{OpenAI2026PolynomialPEPS} to gapped ground states of
range-$R$ Hamiltonians ($R\ge1$), using
Definitions~\ref{def:peps_range_nested_patches} and
\ref{def:peps_range_patch_filter_exponent}.

\begin{definition}[The constants of the adaptive patches at range $R$]
    \label{def:peps_range_adaptive_patch_constants}
    \lean{TNLean.PEPS.Approximation.rangeNestedPatchRadiusExponent,
        TNLean.PEPS.Approximation.rangeAdaptivePatchExponent}
    \leanok
    \uses{def:peps_patch_radius_exponent,
        def:peps_range_patch_filter_exponent}
    For integers $q,R$ and reals $J,\Delta,B_0,p,C_A$, let $R_0^{(R)}$ be the
    radius exponent of Definition~\ref{def:peps_patch_radius_exponent} at
    $C_0=16\max\{1,16C_A\}+\log8$, $B_0$, $p$, and $\kappa_R(q,J,\Delta)$, and
    set $c_R=R_0^{(R)}+1$.
\end{definition}

\begin{lemma}[Adaptive patches at range $R$ at a given area-law constant]
    \label{thm:peps_range_adaptive_patch_supply}
    \lean{TNLean.PEPS.Approximation.rangeNestedPatchRadiusExponent_pos,
        TNLean.PEPS.Approximation.rangeAdaptivePatchExponent_pos,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.rangeNestedPatch_entropy_le,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.uniform_rangeNestedPatch_cylinders,
        TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.rangeAdaptivePatchSupply}
    \leanok
    \uses{def:square_entropy_area_law_at,def:area_law_hamiltonian,
        def:peps_range_nested_patches,def:peps_range_adaptive_patch_constants,
        def:peps_adaptive_patch_constants,def:peps_adaptive_patch_constraint}
    Let $q,R\ge1$ and $J,\Delta,B_0,p>0$, and assume the square-grid entropy
    area law with constant $C_A$ for gapped ground states $\Omega$ of range-$R$
    square Hamiltonians with term bound $J$ and gap $\Delta$. Then
    $R_0^{(R)},c_R>0$, and for every $L\ge2$, such state $\Omega$, and centre:
    \begin{enumerate}
        \item every range-$R$ patch $K_j$ at $u\ge0$ satisfies
              $S_{U\Omega}(K_j)\le\max\{1,16C_A\}(1+u)$;
        \item for $0\le u\le B_0\log L$, the patches $K_j$ carry subspaces
              $S_j$ with $\dim S_j\le L^{R_0^{(R)}}$ and
              $\operatorname{dist}(U\Omega,\sum_j S_j\otimes\mathcal H_{K_j^c})\le L^{-p}$;
        \item for $0<u\le B_0\log L$, there is an adaptive square patch with at
              most $C(q,B_0)L^{c_R}$ cylinder terms and error at most $L^{-p}$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:peps_closed_square_entropy_explicit,
        lem:peps_closed_square_cylinder_core,
        lem:peps_adaptive_patch_supply_core,
        thm:peps_range_nested_patch_sizes,
        thm:peps_range_nested_patch_minimizer_marginal_closeness,
        lem:peps_range_patch_energy_parameter_choice}
    Since $u+s_Rj\in[u,2u]$ is nondecreasing,
    Lemma~\ref{lem:peps_closed_square_entropy_explicit} gives (1), which with
    Lemma~\ref{thm:peps_range_nested_patch_minimizer_marginal_closeness} at
    $\kappa_R$ and Lemma~\ref{lem:peps_closed_square_cylinder_core} gives (2);
    since $s_R\ge2$ implies $m_R\le\frac{B_0}{2}\log L+1$,
    Lemma~\ref{lem:peps_adaptive_patch_supply_core} gives (3).
\end{proof}
