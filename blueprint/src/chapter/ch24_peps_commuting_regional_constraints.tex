\subsection{Commuting constraints on overlapping regions}

Let $\Gamma=(V,E)$ be a finite simple graph with an auxiliary edge orientation and let $G$ be a finite group. A half-edge configuration assigns $\alpha_{v,e}\in G$ to each incident pair $(v,e)$; for $R\subseteq V$, retain all half-edges at vertices $v\in R$. Vertex and shared-edge translations act by
\begin{align}
    (L_k\alpha)_{v,e} & =k_v\alpha_{v,e}, &
    (T_r\alpha)_{v,e} & =\alpha_{v,e}r_e,
    \label{eq:peps_regional_left_right_actions}
\end{align}
satisfying $L_kT_r=T_rL_k$.

\begin{definition}[Translations and regional averages]%
    \label{def:peps_regional_translation_averages}
    \lean{TNLean.PEPS.regularRegionHalfEdgeRightEquiv,
        TNLean.PEPS.regularRegionHalfEdgeRightEquiv_apply,
        TNLean.PEPS.regularRegionEdgeAverage}
    \leanok
    Set $U_k\ket\alpha=\ket{L_k\alpha}$ and $W_r\ket\alpha=\ket{T_r\alpha}$ ($T_r^{-1}=T_{r^{-1}}$). With $r^{e,g}_{e'}=g^{\delta_{e,e'}}$, define
    \begin{align}
        B_{R,e} & =\frac{1}{|G|}\sum_{g\in G}W_{r^{e,g}},                       &
        P_R     & =\frac{1}{|G|^{|R|}}\sum_{k\in G^R}U_k=\bigotimes_{v\in R}P_v
        \label{eq:peps_regional_translation_averages}
    \end{align}
    (with $B_{R,e}=I$ when $e\cap R=\varnothing$).
\end{definition}

\begin{definition}[Flat regional half-edge configurations]%
    \label{def:peps_regional_half_edge_flat}
    \lean{TNLean.PEPS.IsRegularRegionHalfEdgeFlat,
        TNLean.PEPS.regularRegionFlatProjector}
    \leanok
    Following \cite[Section~6]{Schuch2010PEPS}, $\alpha$ is \emph{flat on $R$} if there exist $k_v\in G$ ($v\in R$) such that $k_v^{-1}\alpha_{v,e}=k_w^{-1}\alpha_{w,e}$ on every internal edge $e=(v,w)\in E_R$; write $D_R$ for the diagonal projector onto flat configurations on $R$.
\end{definition}

\begin{lemma}[Commutation of regional translations and flat support]%
    \label{thm:peps_regional_translation_commutation}
    \lean{TNLean.PEPS.regularRegionHalfEdgeRightEquiv_commute_gauge,
        TNLean.PEPS.regularRegionHalfEdgeRightMatrix_commute_vertexTranslation,
        TNLean.PEPS.regularRegionHalfEdgeRightMatrix_commute_localProjector,
        TNLean.PEPS.isRegularRegionHalfEdgeFlat_rightMul_iff,
        TNLean.PEPS.isRegularRegionHalfEdgeFlat_gauge_iff,
        TNLean.PEPS.regularRegionFlatProjector_commute_right,
        TNLean.PEPS.regularRegionFlatProjector_commute_vertexTranslation,
        TNLean.PEPS.regularRegionFlatProjector_commute_localProjector,
        TNLean.PEPS.regularRegionEdgeAverage_commute_flatProjector,
        TNLean.PEPS.regularRegionEdgeAverage_commute_vertexTranslation,
        TNLean.PEPS.regularRegionEdgeAverage_commute_localProjector,
        TNLean.PEPS.regularRegionEdgeAverage_commute}
    \leanok
    \uses{def:peps_regional_translation_averages,def:peps_regional_half_edge_flat}
    For all $k,r$, $L_kT_r=T_rL_k$, and flatness of $\alpha$, $L_k\alpha$, and $T_r\alpha$ on $R$ are equivalent. Consequently $W_r,U_k,P_R,D_R,B_{R,e},B_{R,f}$ commute pairwise as stated: $[W_r,U_k]=[W_r,P_R]=[D_R,W_r]=[D_R,U_k]=[D_R,P_R]=[B_{R,e},D_R]=[B_{R,e},U_k]=[B_{R,e},P_R]=[B_{R,e},B_{R,f}]=0$.
\end{lemma}
\begin{proof}\leanok
    By (\ref{eq:peps_regional_left_right_actions}), $(k_v\alpha_{v,e})r_e=k_v(\alpha_{v,e}r_e)$, and flatness is preserved by $T_r$ (right cancellation) and by $L_k$ (via $q\mapsto kq$). Thus the diagonal projector $D_R$ commutes with $W_r$ and $U_k$, and averaging in (\ref{eq:peps_regional_translation_averages}) yields the remaining commutators.
\end{proof}

\begin{definition}[Internal quotients and incident references]%
    \label{def:peps_regional_quotient_coordinates}
    \lean{TNLean.PEPS.RegularRegionInternalEdge,
        TNLean.PEPS.regularRegionInternalExtension,
        TNLean.PEPS.regularRegionHalfEdgesOfBonds,
        TNLean.PEPS.regularRegionHalfEdgeQuotients,
        TNLean.PEPS.regularRegionHalfEdgeReferences,
        TNLean.PEPS.regularRegionHalfEdgesOfBonds_tail,
        TNLean.PEPS.regularRegionHalfEdgesOfBonds_head,
        TNLean.PEPS.regularRegionHalfEdgeConfigEquiv,
        TNLean.PEPS.regularRegionCanonicalReferences,
        TNLean.PEPS.regularRegionReferenceBoundary}
    \leanok
    Let $E_R$ and $\partial E_R$ be the internal and boundary edges of $R$, and set $I_R=E_R\sqcup\partial E_R$. Extend $u\in G^{E_R}$ by $1$ outside $E_R$ to $\bar u\in G^E$. For $\eta\in G^{I_R}$, define $b(u,\eta)$ on $e=(v,w)\in E_R$ by
    \begin{align}
        b(u,\eta)_{v,e} & =\eta_e,    &
        b(u,\eta)_{w,e} & =u_e\eta_e,
        \label{eq:peps_regional_quotient_coordinates}
    \end{align}
    and set $b(u,\eta)_{x,e}=\eta_e$ at the unique endpoint $x\in R$ of $e\in\partial E_R$; this gives a bijection $(u,\eta)\mapsto b(u,\eta)$ with inverse $u_e=\alpha_{w,e}\alpha_{v,e}^{-1}$. For $\theta\in G^{\partial E_R}$, write $b(u,\theta)=b(u,\eta^\theta)$ where $\eta^\theta|_{E_R}=1$ and $\eta^\theta|_{\partial E_R}=\theta$.
\end{definition}

\begin{lemma}[Normalization of internal references]%
    \label{thm:peps_regional_reference_normalization}
    \lean{TNLean.PEPS.regularRegionHalfEdgesOfBonds_right_normalize,
        TNLean.PEPS.regularRegionHalfEdgesOfBonds_apply_of_rightInvariant}
    \leanok
    \uses{def:peps_regional_quotient_coordinates,def:peps_regional_translation_averages}
    With $r|_{E_R}=\eta|_{E_R}$ and $r|_{E\setminus E_R}=1$, one has $T_r b(u,\eta|_{\partial E_R})=b(u,\eta)$. Hence if $f(T_r\alpha)=f(\alpha)$ whenever $r|_{\partial E_R}=1$, then $f(b(u,\eta))=f(b(u,\eta|_{\partial E_R}))$.
\end{lemma}
\begin{proof}\leanok
    Apply $T_r$ directly to (\ref{eq:peps_regional_quotient_coordinates}).
\end{proof}

\begin{lemma}[Expansion in inserted regional columns]%
    \label{thm:peps_regional_invariant_expansion}
    \lean{TNLean.PEPS.sum_regularRegionProjectorProduct_eq,
        TNLean.PEPS.eq_sum_regularProjectorTwistedRegionMatrix_of_invariant}
    \leanok
    \uses{def:peps_regional_translation_averages,def:peps_regional_quotient_coordinates}
    Let $\Gamma_R(u)$ be the open contraction on $R$ with internal insertions $u\in G^{E_R}$. If $f(L_k\alpha)=f(\alpha)$ for all $k\in G^R$, then $P_Rf=f$:
    \begin{align}
        \sum_\beta f(\beta)\prod_{v\in R}P_v(\alpha_v,\beta_v) & =f(\alpha).
        \label{eq:peps_regional_left_average_fixed}
    \end{align}
    If in addition $f(T_r\alpha)=f(\alpha)$ whenever $r|_{\partial E_R}=1$, then
    \begin{align}
        f & =\sum_{u\in G^{E_R}}\sum_{\theta\in G^{\partial E_R}}
        f(b(u,\theta))\,\Gamma_R(u)\ket\theta.
        \label{eq:peps_regional_supported_orbit_expansion}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_regional_reference_normalization}
    Left invariance gives (\ref{eq:peps_regional_left_average_fixed}); reindexing $\beta=b(u,\eta)$, applying Lemma~\ref{thm:peps_regional_reference_normalization}, and summing over $\eta|_{E_R}$ at fixed $\theta=\eta|_{\partial E_R}$ yields (\ref{eq:peps_regional_supported_orbit_expansion}).
\end{proof}

\begin{lemma}[Removing flat internal insertions]%
    \label{thm:peps_regional_flat_column_transport}
    \lean{TNLean.PEPS.regularProjectorTwistedRegionMatrix_regionGauge,
        TNLean.PEPS.regularProjectorTwistedRegionMatrix_eq_open_of_flat,
        TNLean.PEPS.exists_regularRegionGaugeResidual_eq_one_of_flat}
    \leanok
    \uses{def:peps_regional_quotient_coordinates,def:peps_regional_half_edge_flat}
    For edge insertions $u\in G^E$ and $k\in G^R$, set $u^k_e=k_w^{-1}u_ek_v$ on $e=(v,w)\in E_R$, $u^k_e=1$ outside $E_R$, and on $e=(v,w)\in\partial E_R$,
    \begin{align}
        (\tau_{k,u}\theta)_e & =
        \begin{cases}
            k_v^{-1}\theta_e,    & v\in R,\ w\notin R, \\
            k_w^{-1}u_e\theta_e, & v\notin R,\ w\in R.
        \end{cases}
        \label{eq:peps_regional_column_boundary_transport}
    \end{align}
    Then the general inserted contraction satisfies $\Gamma_R[u]\ket\theta=\Gamma_R[u^k]\ket{\tau_{k,u}\theta}$; in particular, if $b(u,\theta)$ is flat on $R$, there exists $k\in G^R$ with $u^k=1$ and $\Gamma_R(u)\ket\theta=\Gamma_R\ket{\tau_{k,\bar u}\theta}$.
\end{lemma}
\begin{proof}\leanok
    Reindexing the bond variables by $\eta_e\mapsto k_v^{-1}\eta_e$ (if $v\in R$) or $k_w^{-1}u_e\eta_e$ (if $v\notin R,w\in R$) and using left invariance of $P_v$ gives $\Gamma_R[u]\ket\theta=\Gamma_R[u^k]\ket{\tau_{k,u}\theta}$. When $b(u,\theta)$ is flat on $R$, Definition~\ref{def:peps_regional_half_edge_flat} gives $k\in G^R$ with $k_v^{-1}=k_w^{-1}u_e$, i.e., $u^k_e=k_w^{-1}u_ek_v=1$ on $E_R$.
\end{proof}

\begin{lemma}[The regional range in regular coordinates]%
    \label{thm:peps_regional_range_invariant_flat}
    \lean{TNLean.PEPS.mem_regularProjectorOpenRegionRange_iff,
        TNLean.PEPS.mem_regularProjectorOpenRegionRange_of_invariant_flat,
        TNLean.PEPS.regularProjectorOpenRegionMatrix_eq_zero_of_not_flat,
        TNLean.PEPS.regularProjectorOpenRegionMatrix_gauge}
    \leanok
    \uses{def:peps_regional_half_edge_flat}
    A function $f$ on regional half-edge configurations lies in $\operatorname{range}\Gamma_R$ if and only if:
    \begin{enumerate}
        \item $\forall k\in G^R,\ f(L_k\alpha)=f(\alpha)$;
        \item $\forall r\text{ with }r|_{\partial E_R}=1,\ f(T_r\alpha)=f(\alpha)$;
        \item $f(\alpha)=0$ whenever $\alpha$ is not flat on $R$.
    \end{enumerate}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_regional_invariant_expansion,thm:peps_regional_flat_column_transport}
    If (1)--(3) hold, every nonzero summand in (\ref{eq:peps_regional_supported_orbit_expansion}) has $b(u,\theta)$ flat on $R$, hence equals $\Gamma_R\ket{\tau_{k,\bar u}\theta}\in\operatorname{range}\Gamma_R$ by Lemma~\ref{thm:peps_regional_flat_column_transport}. Conversely, every column $\Gamma_R\ket\theta$ satisfies (1)--(3).
\end{proof}

\begin{definition}[Ambient regional constraints]%
    \label{def:peps_ambient_regional_constraints}
    \lean{TNLean.PEPS.regularGlobalVertexAverage,
        TNLean.PEPS.regularGlobalVertexPermutation,
        TNLean.PEPS.restrictRegularRegionHalfEdges,
        TNLean.PEPS.regularGlobalConstraint,
        TNLean.PEPS.regularGlobalRegionFlatProjector,
        TNLean.PEPS.regularGlobalRegionConstraint}
    \leanok
    \uses{def:peps_regional_half_edge_flat}
    On the global half-edge space of $\Gamma=(V,E)$, let $A_v=|G|^{-1}\sum_{g\in G}\rho_v(g)$ average left multiplication $\rho_v(g)$ at $v\in V$, set $B_e=B_{V,e}$ for $e\in E$, and let $D_R$ project onto configurations whose restriction to $R\subseteq V$ is flat. Indexing $(C_v,C_e,C_R)=(A_v,B_e,D_R)$ by $i\in V\sqcup E\sqcup\mathcal P(V)$, define
    \begin{align}
        Q_R=\left(\prod_{v\in R} A_v\right)
        \left(\prod_{e\subseteq R} B_e\right)D_R.
        \label{eq:peps_ambient_regional_constraint}
    \end{align}
\end{definition}

\begin{lemma}[Orthogonal regional constraints]%
    \label{thm:peps_ambient_regional_constraint_projection}
    \lean{TNLean.PEPS.regularRegionEdgeAverage_isStarProjection,
        TNLean.PEPS.regularRegionEdgeAverage_mul_self,
        TNLean.PEPS.regularRegionEdgeAverage_isHermitian,
        TNLean.PEPS.regularRegionFlatProjector_isStarProjection,
        TNLean.PEPS.regularGlobalVertexAverage_isStarProjection,
        TNLean.PEPS.regularGlobalRegionFlatProjector_isStarProjection,
        TNLean.PEPS.regularGlobalConstraint_isStarProjection,
        TNLean.PEPS.regularGlobalConstraint_list_prod_isStarProjection,
        TNLean.PEPS.regularGlobalConstraint_list_prod_mulVec_eq_self_iff,
        TNLean.PEPS.regularGlobalRegionConstraint_isStarProjection,
        TNLean.PEPS.regularGlobalRegionConstraint_mulVec_eq_self_iff}
    \leanok
    \uses{def:peps_ambient_regional_constraints}
    The operators $B_{R,e}$, $D_R$, every $C_i$, every finite product $C_{i_1}\cdots C_{i_n}$, and $Q_R$ in (\ref{eq:peps_ambient_regional_constraint}) are orthogonal projectors, with
    \begin{align}
        \operatorname{ran}(C_{i_1}\cdots C_{i_n}) & =\bigcap_{j=1}^n\ker(C_{i_j}-I),                                                  &
        \operatorname{ran}Q_R                     & =\bigcap_{v\in R}\ker(A_v-I)\cap\bigcap_{e\subseteq R}\ker(B_e-I)\cap\ker(D_R-I).
        \label{eq:peps_ambient_regional_constraint_fixed}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_ambient_regional_constraint_commute}
    Group averaging and diagonal $\{0,1\}$-entries make each $A_v,B_{R,e},D_R$ a self-adjoint idempotent, and commuting orthogonal projectors (Lemma~\ref{thm:peps_ambient_regional_constraint_commute}) have orthogonal-projector products with common fixed space (\ref{eq:peps_ambient_regional_constraint_fixed}).
\end{proof}

\begin{lemma}[Commutation for arbitrary overlapping regions]%
    \label{thm:peps_ambient_regional_constraint_commute}
    \lean{TNLean.PEPS.regularGlobalConstraint_commute,
        TNLean.PEPS.regularGlobalRegionFlatProjector_commute_right,
        TNLean.PEPS.regularGlobalRegionFlatProjector_commute_vertexTranslation,
        TNLean.PEPS.regularGlobalRegionFlatProjector_commute,
        TNLean.PEPS.regularGlobalRegionFlatProjector_commute_vertexAverage,
        TNLean.PEPS.regularGlobalRegionFlatProjector_commute_edgeAverage,
        TNLean.PEPS.regularGlobalVertexAverage_commute_edgeAverage,
        TNLean.PEPS.regularGlobalVertexAverage_commute,
        TNLean.PEPS.regularGlobalConstraint_list_prod_commute,
        TNLean.PEPS.regularGlobalRegionConstraint_commute}
    \leanok
    \uses{def:peps_ambient_regional_constraints}
    For all $i,j\in V\sqcup E\sqcup\mathcal P(V)$, $C_iC_j=C_jC_i$, so any two finite products of the constraints $C_i$ commute; in particular, $Q_RQ_S=Q_SQ_R$ for all $R,S\subseteq V$ (\cite[Theorem~6.12]{Schuch2010PEPS}).
\end{lemma}
\begin{proof}\leanok
    By (\ref{eq:peps_regional_left_right_actions}), vertex and shared-edge translations commute and preserve flatness on every $R\subseteq V$, while diagonal projectors $D_R,D_S$ commute pairwise; hence $[C_i,C_j]=0$ for all $i,j$ and $[Q_R,Q_S]=0$.
\end{proof}
