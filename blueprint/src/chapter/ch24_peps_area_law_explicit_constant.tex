\section{The constant produced by the proof of the area law}
\label{sec:area_law_explicit_constant}

The proof of Theorem~1.1 of \cite{OpenAI2026AreaLaw} produces the constant of
the area law from the constants of Propositions~9.6, 10.2 and~11.2. This
section records that constant.

\begin{definition}[The constant in the two-dimensional area-law proof]
    \label{def:area_law_proof_constant}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLawConstantOf,
        TNLean.PEPS.AreaLaw.uniformAreaLawScale}
    \leanok
    For an integer $q\ge1$ and real numbers $C_2,C_3$ put $c_D=\max\{C_3,0\}$,
    $c_M=\max\{C_2,0\}$ and
    \begin{align}
        C(q,C_2,C_3) & =c_D\log q+\frac{c_Mc_D}{2}.
    \end{align}
    For integers $N_1,N_2$ and a real number $C_2$, let $N_3$ be the collar
    radius threshold at the coefficient $C_2$ and the factor $1$, and put
    $n_*(N_1,N_2,C_2)=\max\{N_1,N_2,N_3\}$.
\end{definition}

\begin{theorem}[The area law at given constants of the three propositions]
    \label{thm:area_law_conditional_assembly_constants}
    \lean{TNLean.PEPS.AreaLaw.regionalEntropy_le_of_collar_amplification_twoFamilies_constants,
        TNLean.PEPS.AreaLaw.uniformAreaLawConstantOf_nonneg,
        TNLean.PEPS.AreaLaw.two_le_uniformAreaLawScale,
        TNLean.PEPS.AreaLaw.exists_ambientSupDistance_le_of_mem_graphNeighborhood}
    \leanok
    \uses{def:area_law_proof_constant,def:area_law_collar_bound,
        def:area_law_amplification_bound,def:area_law_two_family_decomposition,
        def:area_law_ground_entropy}
    Fix $q\ge1$, $R\ge0$, $J$, $\Delta>0$ and template constants
    $D_0,C_{\rm tpl}$. Suppose that the assertions of the sublinear collar
    bound hold with the constant $C_1$ for all $n\ge N_1$; that the assertions
    of the amplification bound for the input constant $\max\{C_1,1\}$ and the
    order $M=100$ hold with the constant $C_2$ for all $n\ge\max\{2,N_2\}$;
    and that $C_3$ is a constant of the two-family decomposition at the lower
    scale $n_*(N_1,N_2,C_2)$. Then $C(q,C_2,C_3)\ge0$, $n_*(N_1,N_2,C_2)\ge2$,
    and for every finite induced domain $\Lambda$, every admissible
    Hamiltonian with a gapped ground vector $\Omega$ and every
    $A\subseteq\Lambda$,
    \begin{align}
        S_\Omega(A) & \le C(q,C_2,C_3)\,|\partial_\Lambda A|.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_zero_boundary,
        thm:area_law_boundary_assembly,thm:area_law_uniform_collar_threshold}
    If $|\partial_\Lambda A|=0$, the entropy vanishes by
    Theorem~\ref{thm:area_law_zero_boundary}. Otherwise apply the two-family
    decomposition with constant $C_3$ at the lower scale $n_*$. The collar
    radius threshold $N_3\ge2$ gives $L_n+2r<n^\beta$ for every $n\ge N_3$
    and every $r\le C_2n^{(1-\varepsilon)/\alpha}$. Each piece $X_i$ has scale
    $n_i\ge n_*\ge\max\{2,N_1,N_2,N_3\}$, so the collar bound gives the
    hypotheses of amplification for $T_i$ with input constant $\max\{C_1,1\}$.
    Since $L_{n_i}<n_i^\beta$, the cut clearance
    $\operatorname{dist}_\infty(T_i,Z)>2n_i^\beta$ gives
    $N_{2r_i}^\Lambda(S_i)\subseteq A$. A walk of length at most $2r_i$ from
    $S_i$ moves at most $2r_i$ in sup-norm, so every point of
    $N_{2r_i}^\Lambda(S_i)$ lies within $L_{n_i}+2r_i<n_i^\beta$ of $T_i$ and
    hence outside every earlier piece of the same family. Thus
    $J_i=(\Lambda\setminus A)\cup\bigcup_{j<i,\,f_j=f_i}X_j$ avoids
    $N_{2r_i}^\Lambda(S_i)$, and amplification applied to
    $X_i\subseteq A\cap T_i$ gives $I_\Omega(X_i:J_i)\le c_Mn_i^{-100}$. The
    boundary estimate from partition bounds, with $|D|\le c_Db$ and
    $\sum_ic_Mn_i^{-100}\le c_Mc_Db$, gives the constant
    $c_D\log q+c_Mc_D/2$.
\end{proof}

The constants of Theorem~\ref{thm:area_law_conditional_assembly_constants}
depend, through the sublinear collar bound, on truncation and transport
constants of the scanner. The next two entries pin these constants: the
truncation constants are explicit functions of $R,J,\Delta$, and the only
remaining input is one tuple of transport constants satisfying the
finite-family transport bound of
Definition~\ref{def:al_scan_finite_family_transport_bound}. That condition
involves only the five constants, independent of $q,R,J,\Delta$ and of every
domain, Hamiltonian, state and cut;
Theorem~\ref{thm:al_scan_actual_round_transport} specializes it to the rounds
of one scanner.

\begin{definition}[The constant of the area law from transport constants]
    \label{def:area_law_transport_constant}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLawConstantExplicit}
    \leanok
    \uses{def:area_law_proof_constant,def:area_law_two_family_explicit_constant}
    Fix $q\ge1$, $R\ge0$, $J$ and $\Delta$, and real numbers
    $c_0,C_{\rm ent},e_{\rm ent},C_{\rm en},e_{\rm en}$. Let $C_{\rm box}$ be
    the safe-box constant of the bootstrap of
    Theorem~\ref{thm:arealaw-physical-box-fixed-exponent} started from the
    explicit initial box estimate, with the truncation constant at the factor
    $2$ and these transport constants. Let $C_1$ and $N_1$ be the constant and
    the threshold of the sublinear collar bound at $C_{\rm box}$, the
    truncation constant at the factor $10$ and these transport constants. Put
    $C_0=\max\{C_1,1\}$, let $C_2$ and $N_2'$ be the constant and the
    threshold of the amplification bound at $C_0$ and the order $M=100$ for
    the filter-locality constants of the positive constraints, and let
    $N_2=\max\{N_2',N_3\}$ with $N_3$ the collar radius threshold at $C_2$
    and the factor $1$. With $n_*=n_*(N_1,N_2,C_2)$ and
    $C_3=C_{\rm expl}(2R+11,24,n_*,1/112)$, put
    \begin{align}
        C_{\rm AL}(q,R,J,\Delta;c_0,C_{\rm ent},e_{\rm ent},C_{\rm en},e_{\rm en})
         & =C(q,C_2,C_3).
    \end{align}
\end{definition}

\begin{theorem}[The area law with an explicit constant, assuming the transport bound]
    \label{thm:area_law_transport_constant}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw_explicit}
    \leanok
    \uses{def:area_law_transport_constant,def:area_law_ground_entropy,
        def:al_scan_finite_family_transport_bound}
    Fix $q\ge1$, $R\ge0$, $J\ge0$ and $\Delta>0$. Assume that the
    constants $c_0,C_{\rm ent},e_{\rm ent},C_{\rm en},e_{\rm en}$, with
    $c_0>0$, satisfy the finite-family transport bound of
    Definition~\ref{def:al_scan_finite_family_transport_bound}. Then the constant
    $C=C_{\rm AL}(q,R,J,\Delta;c_0,C_{\rm ent},e_{\rm ent},C_{\rm en},e_{\rm en})$
    of Definition~\ref{def:area_law_transport_constant} is nonnegative, and
    for every finite induced domain $\Lambda$, every admissible Hamiltonian
    with a gapped ground vector $\Omega$ and every $A\subseteq\Lambda$,
    \begin{align}
        S_\Omega(A) & \le C\,|\partial_\Lambda A|.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_conditional_assembly_constants,
        thm:area_law_sublinear_collar,thm:area_law_amplification,
        thm:area_law_two_family_explicit_constant,
        thm:arealaw-physical-box-fixed-exponent}
    The transport constants and the explicit initial box estimate give the
    safe-box bound with the constant $C_{\rm box}$ at safety parameter
    $2R+11$, and then the sublinear collar bound with the constant $C_1$ from
    the threshold $N_1$ on, at $D_0=2R+11>2R+10$ and $C_{\rm tpl}=24$. The
    filter-locality constants of the positive constraints give the
    amplification bound at the input constant $C_0$ and the order $100$ with
    the constant $C_2$ from the threshold $N_2$ on. Since $1/112$ is a
    birth-separation constant, $C_3$ is a constant of the two-family
    decomposition at $n_*$.
    Theorem~\ref{thm:area_law_conditional_assembly_constants} then gives the
    bound.
\end{proof}

\begin{theorem}[The transport bound at explicit constants]
    \label{thm:area_law_transport_bound_explicit}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.finiteFamilyTransportBound_explicit}
    \leanok
    \uses{def:al_scan_finite_family_transport_bound}
    The finite-family transport bound of
    Definition~\ref{def:al_scan_finite_family_transport_bound} holds for
    \begin{align}
        c_0         & =\tfrac18,                               \\
        C_{\rm ent} & =200000,                                 \\
        e_{\rm ent} & =2,                                      \\
        C_{\rm en}  & =3\bigl(18\pi^2+6144\,e^{\pi^2+1}\bigr), \\
        e_{\rm en}  & =4.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    The finite-family transport estimate of
    \cite[Proposition~7.4]{OpenAI2026AreaLaw} holds with the threshold $\tfrac18$, the entropy constant $200000$
    and exponent $2$, and the energy constant
    $3\bigl(18\pi^2+6144\,e^{\pi^2+1}\bigr)$ and exponent $4$. Its
    conclusion is the finite-family transport bound at these constants.
\end{proof}

\begin{definition}[The explicit constant of the area law]
    \label{def:area_law_closed_constant}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLawConstantClosed}
    \leanok
    \uses{def:area_law_transport_constant}
    For $q\ge1$, $R\ge0$, $J$ and $\Delta$, put
    \begin{align}
        C_{\rm AL}(q,R,J,\Delta)
         & =C_{\rm AL}\bigl(q,R,J,\Delta;\tfrac18,200000,2,
        3\bigl(18\pi^2+6144\,e^{\pi^2+1}\bigr),4\bigr),
    \end{align}
    the constant of Definition~\ref{def:area_law_transport_constant} at the
    transport constants of Theorem~\ref{thm:area_law_transport_bound_explicit}.
\end{definition}

\begin{theorem}[The area law with an explicit constant]
    \label{thm:area_law_closed_constant}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw_closed}
    \leanok
    \uses{def:area_law_closed_constant,def:area_law_ground_entropy}
    Fix $q\ge1$, $R\ge0$, $J\ge0$ and $\Delta>0$. Then
    $C=C_{\rm AL}(q,R,J,\Delta)$ is nonnegative, and for every finite induced
    domain $\Lambda$, every admissible Hamiltonian with a gapped ground vector
    $\Omega$ and every $A\subseteq\Lambda$,
    \begin{align}
        S_\Omega(A) & \le C\,|\partial_\Lambda A|.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_transport_constant,thm:area_law_transport_bound_explicit}
    Apply Theorem~\ref{thm:area_law_transport_constant} with the transport
    constants of Theorem~\ref{thm:area_law_transport_bound_explicit}, for which
    $c_0=\tfrac18>0$.
\end{proof}

\begin{theorem}[The square-grid area law at the closed-form constant]
    \label{thm:square_area_law_closed_constant}
    \lean{TNLean.PEPS.AreaLaw.squareEntropyAreaLawAt_closed}
    \leanok
    \uses{def:square_entropy_area_law_at,def:area_law_closed_constant}
    Fix $q\ge1$, $J\ge0$ and $\Delta>0$. Then the square-grid entropy area
    law of Definition~\ref{def:square_entropy_area_law_at} holds at $q$, $J$,
    $\Delta$ with the constant $C_{\rm AL}(q,1,J,\Delta)$ of
    Definition~\ref{def:area_law_closed_constant}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_closed_constant,thm:square_area_law_from_uniform}
    Theorem~\ref{thm:area_law_closed_constant} at range $R=1$ bounds the
    entropy of every cut of every finite induced domain by
    $C_{\rm AL}(q,1,J,\Delta)$ times the number of cut edges. Placing a
    square-grid Hamiltonian on the integer square, as in the proof of
    Theorem~\ref{thm:square_area_law_from_uniform}, transports this bound to
    every region of the square grid with the same constant.
\end{proof}
