\begin{lemma}[The area law at given constants of the three propositions]
    \label{thm:area_law_conditional_assembly_constants}
    \lean{TNLean.PEPS.AreaLaw.regionalEntropy_le_of_collar_amplification_twoFamilies_constants,
        TNLean.PEPS.AreaLaw.uniformAreaLawConstantOf_nonneg,
        TNLean.PEPS.AreaLaw.two_le_uniformAreaLawScale,
        TNLean.PEPS.AreaLaw.exists_ambientSupDistance_le_of_mem_graphNeighborhood}
    \leanok
    \uses{def:area_law_proof_constant,def:area_law_collar_bound,
        def:area_law_amplification_bound,def:area_law_two_family_decomposition,
        def:area_law_ground_entropy}
    Fix $q\ge1$, $R\ge0$, $J$, $\Delta>0$, and $D_0,C_{\rm tpl}$. Suppose the sublinear collar
    bound (Definition~\ref{def:area_law_collar_bound}) holds with $(C_1,N_1)$, the amplification bound (Definition~\ref{def:area_law_amplification_bound}) at $(\max\{C_1,1\},100)$ holds with $(C_2,\max\{2,N_2\})$,
    and $C_3$ is a constant of the two-family decomposition (Definition~\ref{def:area_law_two_family_decomposition}) at $n_*(N_1,N_2,C_2)$. Then $C(q,C_2,C_3)\ge0$, $n_*(N_1,N_2,C_2)\ge2$,
    and for every finite induced domain $\Lambda$, admissible Hamiltonian with gapped ground vector $\Omega$ satisfying (\ref{eq:area_law_projector_gap}), and $A\subseteq\Lambda$,
    \begin{align}
        S_\Omega(A) & \le C(q,C_2,C_3)\,|\partial_\Lambda A|.
        \label{eq:area_law_conditional_assembly}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:area_law_zero_boundary,
        thm:area_law_boundary_assembly,thm:area_law_uniform_collar_threshold}
    If $|\partial_\Lambda A|=0$, $S_\Omega(A)=0$ by Lemma~\ref{thm:area_law_zero_boundary}. Otherwise apply the two-family decomposition with $(C_3,n_*)$. By Lemma~\ref{thm:area_law_uniform_collar_threshold}, $N_3\ge2$ and $L_n+2r<n^\beta$ for $n\ge N_3$ and $r\le C_2n^{(1-\varepsilon)/\alpha}$. Since $n_i\ge n_*\ge\max\{2,N_1,N_2,N_3\}$, the collar bound applies to $T_i$, and $\operatorname{dist}_\infty(T_i,Z)>2n_i^\beta>L_{n_i}$ implies $N_{2r_i}^\Lambda(S_i)\subseteq A$ and $\operatorname{dist}_\infty(N_{2r_i}^\Lambda(S_i),T_i)\le L_{n_i}+2r_i<n_i^\beta$. Thus $J_i=(\Lambda\setminus A)\cup\bigcup_{j<i,\,f_j=f_i}X_j$ is disjoint from $N_{2r_i}^\Lambda(S_i)$, amplification yields $I_\Omega(X_i:J_i)\le c_Mn_i^{-100}$, and Lemma~\ref{thm:area_law_boundary_assembly} with $|D|\le c_Db$ and $\sum_i c_Mn_i^{-100}\le c_Mc_Db$ gives (\ref{eq:area_law_conditional_assembly}).
\end{proof}

\begin{lemma}[Iterates of a dominating map]
    \label{thm:area_law_bootstrap_iterate}
    \lean{TNLean.PEPS.AreaLaw.Scan.boxBootstrapConstant_le_iterate,
        TNLean.PEPS.AreaLaw.Scan.one_add_iterate_mul_one_add_pow_le,
        TNLean.PEPS.AreaLaw.Scan.boxBootstrapConstant_le_pow_pow}
    \leanok
    Let $V,C_0\ge0$, $G:\lbrack0,\infty)\to\lbrack0,\infty)$, $N:\mathbb R\to\mathbb N$, and
    $C_{j+1}=G(C_j)+V\,N(C_j)^2$.
    \begin{enumerate}
        \item If $F$ is nondecreasing on $[0,\infty)$ with $\max\{x,G(x)+VN(x)^2\}\le F(x)$ for $x\ge0$, then $C_j\le F^{M}(C_0)$ for all $j\le M$.
        \item If $A\ge0$, $K\in\mathbb Z_{\ge1}$, and $f(y)=A(1+y)^K$, then
              $1+f^{j}(x)\le\bigl((A+1)(1+x)\bigr)^{(2K)^j}$ for all $x\ge0$ and $j\in\mathbb N$.
        \item If $A,K\ge1$ and $G(C)+V\,N(C)^2\le A(1+C)^K$ for $C\ge0$, then
              $C_j\le\bigl((A+1)(1+C_0)\bigr)^{(2K)^M}$ for all $j\le M$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    (1) By induction, $F^k(C_0)$ is nondecreasing in $k$ and $C_{k+1}\le F(C_k)\le F^{k+1}(C_0)$.
    (2) With $b=(A+1)(1+x)\ge1$ and $u_j=1+f^j(x)\ge1$, $u_0\le b$ and $1+Km\le2Km$ ($m\ge1$) give
    $u_{j+1}=1+Au_j^K\le(A+1)u_j^K\le b^{1+K(2K)^j}\le b^{(2K)^{j+1}}$.
    (3) Apply (1) with $F(x)=A(1+x)^K$ and then (2).
\end{proof}

\begin{lemma}[One bootstrap round is at most a power]
    \label{thm:area_law_bootstrap_step_bound}
    \lean{TNLean.PEPS.AreaLaw.Scan.boxBootstrapStep_le_boxStepPrefactor,
        TNLean.PEPS.AreaLaw.Scan.boxImprovementThreshold_le_add,
        TNLean.PEPS.AreaLaw.Scan.max_two_mul_roughSelectedDensityCoefficient_le,
        TNLean.PEPS.AreaLaw.Scan.bootstrapRestoredBudgetCoefficient_weak_le,
        TNLean.PEPS.AreaLaw.Scan.natRpowThreshold_le_pow_add_one,
        TNLean.PEPS.AreaLaw.Scan.one_le_boxStepPrefactor}
    \leanok
    \uses{def:area_law_bootstrap_step_constants}
    In Definition~\ref{def:area_law_bootstrap_step_constants}, $A\ge1$, and for all $C\ge0$,
    \begin{align}
        G(C)+\log q & \le A\,(1+C)^{2k}.
        \label{eq:area_law_bootstrap_step_bound}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Since $C_{\rm out}$ and $C_\delta$ are affine with nonnegative coefficients, $C_{\rm out}(C)\le C_{\rm out}(1)(1+C)$ and $\max\{128,2C_\delta(C)\}\le\alpha(1+C)$. The only $C$-dependent entry in $N(C)$ is $\bigl\lceil\max\bigl\{1,\max\{128,2C_\delta(C)\}^{1/\nu}\bigr\}\bigr\rceil$, so
    $N(C)\le N(0)+\bigl(\alpha(1+C)\bigr)^k+1\le(N(0)+2)\bigl(\alpha(1+C)\bigr)^k$,
    giving (\ref{eq:area_law_bootstrap_step_bound}) via $1\le1+C\le(1+C)^{2k}$.
\end{proof}

\begin{corollary}[Polynomial PEPS approximation at range $R$ with explicit constants]
    \label{thm:polynomial_peps_range_explicit_constants}
    \lean{TNLean.PEPS.Approximation.rangePolynomialPEPSApproximation_explicit,

    \uses{def:polynomial_peps_range_constants,def:square_range_ground_states,

    every range-$R$ ground state $\Omega$ on $\Lambda_L$ admits a

    $D_e\le C_{{\rm PEPS},R}L^{c_{{\rm PEPS},R}}$ for all $e$ and
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
        \label{eq:polynomial_peps_range_error}
    \end{align}

\end{corollary}
\begin{proof}

    \uses{thm:square_area_law_range_constant,
        thm:peps_angular_information_range_closed,
        thm:peps_range_adaptive_patch_supply,
        thm:polynomial_peps_of_patch_supply,
        thm:square_ground_states_range_one,
        lem:peps_remove_small_sizes_named}
    Combine Corollary~\ref{thm:square_area_law_range_constant} (area-law constant $C_A$), Lemma~\ref{thm:peps_angular_information_range_closed} at $(a_0/2,1,4,60)$ (angular parameters $(D,L_{\rm ang})$), and Lemma~\ref{thm:peps_range_adaptive_patch_supply} at $(20D,60)$ (patch exponent $c_R$). Since $\|\Omega\|=1$ by Lemma~\ref{thm:square_ground_states_range_one}, Lemma~\ref{thm:polynomial_peps_of_patch_supply} gives the bound for $L\ge L_0$ with $C_{\rm large}$, and Lemma~\ref{lem:peps_remove_small_sizes_named} extends it to all $L\ge2$ with $C_{{\rm PEPS},R}$.
\end{proof}
