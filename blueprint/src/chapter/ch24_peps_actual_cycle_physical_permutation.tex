\begin{theorem}[Cycle operations from an arbitrary bond assignment]
    \label{thm:peps_actual_cycle_physical_permutation}
    \lean{
        TNLean.PEPS.exists_unitary_regularActualCyclePhysicalPermutation,
        TNLean.PEPS.regularGaugedTreeCycleAssignment_treeGauge_of_internal,
        TNLean.PEPS.regularRegionBondExtension_treeGauge,
        TNLean.PEPS.regularActualCyclePermutation
    }
    \leanok
    Fix a finite region $R$, a spanning tree $T\subset R$, and a root $v_0\in R$.
    Each bond assignment $u$ determines a normalized tree gauge $k_u$ and cycle residual tuple $\omega_u$, and reconstructing internal edges from $(k_u,\omega_u)$ while keeping exterior edges fixed recovers $u$.

    Let $\phi$ be a cycle-tuple permutation commuting with simultaneous conjugation, and suppose the site maps are regular $G$-isometric.
    As an auxiliary consequence of the accessible-coordinate and physical-movement constructions of SCP10 (local lines 1765--1920 and 2271--2305), there exists a regional unitary $W_\phi$ (chosen before $u$ and boundary labels $\theta$) such that for $u'=\mathrm{rec}(k_u,\phi(\omega_u))$ (agreeing with $u$ outside $R$),
    \begin{align}
        W_\phi\Psi_R(u,\theta)=\Psi_R(u',\theta),\qquad (W_\phi\otimes I)\Psi(u)=\Psi(u'). \notag
    \end{align}
    While $T$ and $\phi$ remain explicit, an occupied-to-vacant plaquette interpretation requires an explicit holonomy calculation, and the full four-endpoint braid of \cite{gap:rmp_peps_quantum_double_g_isometry} is not asserted here.
    \uses{thm:peps_regular_gauged_cycle_permutation}
\end{theorem}
\begin{proof}\leanok
    On internal edges $e=(t(e),h(e))$, tree edges satisfy $u_e=k_{u,h(e)}k_{u,t(e)}^{-1}$ and chords satisfy $\omega_{u,e}=k_{u,h(e)}^{-1}u_e k_{u,t(e)}$, so reconstruction from $(k_u,\omega_u)$ recovers $u$.
    Applying the common-background unitary $W_\phi$ of Theorem~\ref{thm:peps_regular_gauged_cycle_permutation} to $(k_u,\omega_u)$ while preserving exterior bonds yields the open and closed contraction identities.
\end{proof}
