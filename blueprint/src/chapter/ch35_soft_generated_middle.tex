\section{A native approximation of the soft middle term}
\label{sec:soft_generated_middle}

The middle term of the same regional soft Hamiltonian admits an approximation
in its native interval coordinates. Lemma~\ref{lem:native_soft_middle_mpo}
constructs its operator tensors, and
Lemma~\ref{lem:five_region_soft_middle_mpo} places them back on the original
chain while retaining every other term exactly. These are operator
constructions supporting \cite[Proposition~5.7 and Theorem~5.8]{Arad2016RigorousRG};
they assert neither a spectral gap nor an improving family.

\begin{lemma}[Native restriction of a supported approximation]
    \label{lem:native_supported_mpo_restriction}
    \lean{TNLean.MPS.AreaLaw.exists_native_interval_slice_of_mpo_approximation}
    \leanok
    Let $q\ge1$, $0\le l\le r\le N+1$, and let $P$ be supported on
    $[l,r)$. Suppose an actual Hermitian operator chain of virtual
    dimension $D$ approximates $P$ in norm with error at most $\xi$.
    Fix one exterior configuration and use the increasing coordinates
    on $[l,r)$. Restricting the site tensors to that interval and
    contracting the exterior tensors gives a native operator $Z$ of
    the same virtual dimension. Both $Z$ and its placement are Hermitian,
    and both the native principal restriction and the original $P$ are
    approximated with error at most $\xi$.
\end{lemma}
\begin{proof}\leanok
    The native closing matrix is the original cap multiplied by its
    fixed suffix on the left and fixed prefix on the right.
    The principal restriction is norm-contracting. Since $P$ is supported
    on the interval, its placement reconstructs $P$ exactly; contractivity
    of placement gives the original-chain error bound.
\end{proof}

\begin{lemma}[Native tensors for a regional soft term]
    \label{lem:native_soft_middle_mpo}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_native_soft_middle_mpo_approximation}
    \leanok
    Let $h_0,\ldots,h_{N-1}$ be positive-contraction nearest-neighbour terms on
    $N+1$ sites of dimension $q\geq1$. Let $0\leq l\leq r\leq N+1$, and put
    $R=\sum_{l\leq i<i+1<r}h_i$. Suppose $pI\leq R$, $m\geq1$, $t\geq3m$
    and $\xi>0$. Define $P=pI+h_{m,t}(R-pI)$ and
    \begin{align}
        W&=t2^{m+1},\qquad b=2(e^{1/3}-1),\notag\\
        a&=\max\left\{1,
          \left\lceil\frac{\log\max\{1,N^2\}
            -\log\log(1+\xi/W)}{-\log b}\right\rceil\right\},\notag\\
        B&=2\bigl(1+Nq^2+(m+1)a^2q^a\bigr).
        \label{eq:soft_middle_bond}
    \end{align}
    There are site-dependent operator tensors on $[l,r)$ with virtual
    dimension $B$ and a closing matrix $C$, representing a Hermitian matrix
    $Z$ on that interval. Write $\iota$ for inclusion of the interval sites,
    and let $P_{\mathrm{nat}}$ be the principal slice of $P$ obtained by fixing
    one exterior basis configuration. Then
    \begin{align}
        P&=\iota(P_{\mathrm{nat}}),\qquad
        \|P_{\mathrm{nat}}-Z\|\leq\xi,\qquad
        \|P-\iota(Z)\|\leq\xi.
        \label{eq:soft_middle_native_error}
    \end{align}
    The placement $\iota(Z)$ is Hermitian. Empty intervals are included;
    their scalar is represented by $\operatorname{Tr}C$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:soft_hamiltonian_mpo_accuracy,lem:hermitian_trace_chain_approximation,
        thm:actual_retained_interval_trace,lem:qc_placement_norm}
    Mask the original interaction family outside $[l,r)$, and apply the
    corrected heat-expansion construction to its soft replacement. Taking
    the Hermitian part doubles the virtual dimension without increasing
    the approximation error. Restrict these very same operator tensors at
    a fixed exterior configuration; the ordered exterior products are
    absorbed into $C$, so the virtual dimension is unchanged.

    Principal compression is contractive. Placement on idle sites is a
    star-algebra homomorphism and is contractive as well. Since $P$ acts
    trivially outside the interval, its slice reconstructs $P$ exactly.
    These two contractions give~(\ref{eq:soft_middle_native_error}).
\end{proof}

\begin{lemma}[Only the middle soft term is replaced]
    \label{lem:five_region_soft_middle_mpo}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_five_region_soft_middle_mpo_approximation}
    \leanok
    In the five-region notation, let $A_j$ be the actual internal sums,
    $Y$ the crossing-bond sum and $e_j,p_j\in\mathbb R$. For a selected set
    $S\subseteq\{0,\ldots,4\}$ containing $2$, put
    \begin{align}
        F_j&=p_jI+h_{m,t}(A_j-p_jI),\notag\\
        G&=\sum_{j=0}^4
          \left(\begin{cases}F_j,&j\in S,\\A_j,&j\notin S\end{cases}
          -e_jI\right)+Y.
        \label{eq:soft_middle_same_matrix}
    \end{align}
    Assume $p_2I\leq A_2$, $m\geq1$, $t\geq3m$ and $\xi>0$.
    There are native middle tensors as in
    Lemma~\ref{lem:native_soft_middle_mpo} such that the matrix obtained by
    replacing only $F_2$ by their placement satisfies
    \begin{align}
        \widehat G=G-F_2+\iota(Z),\qquad
        G=G^*,\qquad \widehat G=\widehat G^*,\qquad
        \|G-\widehat G\|\leq\xi.
        \label{eq:soft_middle_full_error}
    \end{align}
    The other soft terms, all scalar shifts and the original crossing
    bonds are retained exactly.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:native_soft_middle_mpo}
    Apply Lemma~\ref{lem:native_soft_middle_mpo} to the actual middle sum.
    All summands in~(\ref{eq:soft_middle_same_matrix}) are Hermitian.
    Since $2\in S$, cancellation of every unchanged term gives
    $G-\widehat G=F_2-\iota(Z)$, whose norm is at most $\xi$.
\end{proof}

The interaction-label cutoff and energy normalization are the corrected
ones in \cite{gap:alvv16_cluster_approximation}. The tensors may depend on
the supplied middle energy estimate. No independence of that estimate,
adaptive measurability or running-time conclusion is asserted.


\subsection{Fixed tensors before the estimated middle energy}

\begin{lemma}[A polynomial closing matrix for every lower estimate]
    \label{lem:uniform_native_soft_middle}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_uniform_native_soft_middle_mpo}
    \leanok
    Let $h_i$, $0\le i<N$, be positive-contraction nearest-neighbour
    terms on $N+1$ sites of dimension $q\ge1$. Fix $0\le l\le r\le N+1$
    and put $A=\sum_{l\le i<i+1<r}h_i$. Suppose $eI\le A$, $m\ge1$,
    $t\ge3m$, and $\xi>0$. Let $s$ be the explicit cluster order of
    (\ref{eq:soft_interval_cluster_order}), with $W=t2^{m+1}$, and set
    $B=2s^2q^s(m+1)$.

    There are native site tensors $A_*$ of virtual dimension $B$ and
    matrices $C_0,\ldots,C_m$, before every lower estimate $p\le e$,
    such that
    \begin{align}
        Z(z)=\mathcal T\!\left(A_*,\sum_{a=0}^m z^aC_a\right)
        \label{eq:uniform_soft_polynomial_cap}
    \end{align}
    and its placement are Hermitian for every real $z$.
    For every $p\le e$, put $z=e^{(p-e)/t}$ and
    $P=h_{m,t}(A-pI)$. Then $0<z\le1$, the native principal restriction
    of $P$ differs from $Z(z)$ by norm at most $\xi$, and $P$ differs
    from the placement of $Z(z)$ by norm at most $\xi$.
    The scalar $pI$ is represented separately and exactly.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:shifted_regional_heat_mpo,
      lem:trace_chain_linear_combination,lem:native_supported_mpo_restriction}
    Construct all heat atoms $E_a=e^{-a(A-eI)/t}$ at the fixed lower
    energy $e$. Replace each approximation by its Hermitian part and
    restrict it once to the native interval. The common-closing formula
    combines their fixed tensors without absorbing the coefficients.
    The exact heat expansion gives
    \begin{align}
        h_{m,t}(A-pI)=\sum_{a=0}^m c_a z^aE_a,
        \qquad \sum_{a=0}^m|c_a|\le t2^{m+1}.
    \end{align}
    Because $0<z\le1$, weighting by $z^a$ does not increase the absolute
    coefficient budget. The chosen cluster order makes the weighted
    approximation error at most $\xi$. Diagonal blocks give
    (\ref{eq:uniform_soft_polynomial_cap}) with fixed tensors and fixed
    coefficient matrices. Exact reconstruction of the supported heat
    atoms and contractivity of placement give the physical error bound.
\end{proof}
