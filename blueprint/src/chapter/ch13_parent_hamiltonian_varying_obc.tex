\section{Rectangular reconstruction for varying-bond open chains}
\label{sec:parent_varying_obc_reconstruction}

Consider three consecutive canonical open-chain sites with finite
physical alphabets $I,J,K$ and rectangular matrices
\begin{align}
    A_i & \in M_{a,b}(\mathbb C), &
    B_j & \in M_{b,c}(\mathbb C), &
    C_k & \in M_{c,e}(\mathbb C).
\end{align}
The virtual coordinate spaces may differ at every cut. The following
identities are the local reconstruction step of the open-boundary
uniqueness proof in \cite[Theorem~9, lines~1212--1237]{PerezGarcia2007Matrix}.
The first identities concern rectangular trace representations on three sites.
The subsequent induction treats the entire chain. The final section identifies
the adjacent-pair conditions with the kernel of the open parent Hamiltonian.

\begin{lemma}[Matching the boundary products on an overlap]
    \label{lem:varying_obc_overlap_boundary_products}
    \lean{Matrix.obc_overlap_boundary_mul_eq}
    \leanok
    Assume that $\operatorname{span}\{B_j\}=M_{b,c}(\mathbb C)$.
    Let $X_k\in M_{c,a}(\mathbb C)$ and $Y_i\in M_{e,b}(\mathbb C)$ satisfy
    \begin{align}
        \operatorname{tr}(X_kA_iB_j) & =\operatorname{tr}(Y_iB_jC_k)
    \end{align}
    for every $i,j,k$. Then $X_kA_i=C_kY_i$ for every $i,k$.
\end{lemma}
\begin{proof}\leanok
    Cyclicity gives
    $\operatorname{tr}(Y_iB_jC_k)=\operatorname{tr}(C_kY_iB_j)$.
    By the spanning assumption, equality of these trace functionals on
    all $B_j$ gives equality on every matrix in $M_{b,c}(\mathbb C)$.
    The rectangular trace pairing is nondegenerate, so $X_kA_i=C_kY_i$.
\end{proof}

\begin{lemma}[The canonical adjoint reconstruction]
    \label{lem:varying_obc_three_site_reconstruction}
    \lean{Matrix.obc_three_site_boundary_reconstruction,
        Matrix.obc_three_site_trace_reconstruction}
    \leanok
    \uses{lem:varying_obc_overlap_boundary_products}
    In addition assume the canonical row-isometry identity
    $\sum_iA_iA_i^\dagger=I_a$. Define the common boundary
    \begin{align}
        Z & =\sum_iY_iA_i^\dagger\in M_{e,a}(\mathbb C).
    \end{align}
    Then $X_k=C_kZ$ for every $k$, and
    \begin{align}
        \operatorname{tr}(X_kA_iB_j) & =\operatorname{tr}(ZA_iB_jC_k)
    \end{align}
    for every $i,j,k$.
\end{lemma}
\begin{proof}\leanok
    The preceding overlap identity and the canonical normalization give
    \begin{align}
        X_k & =X_k\sum_iA_iA_i^\dagger
        =\sum_iC_kY_iA_i^\dagger=C_kZ.
    \end{align}
    Substituting this in the original trace representation and using
    cyclicity of the trace gives
    \begin{align}
        \operatorname{tr}(X_kA_iB_j) & =\operatorname{tr}(C_kZA_iB_j)
        =\operatorname{tr}(ZA_iB_jC_k).
    \end{align}
    This is the adjoint boundary construction in the source proof.
\end{proof}

\begin{lemma}[Two adjacent section conditions give a three-site representation]
    \label{lem:varying_obc_pair_sections_reconstruct}
    \lean{Matrix.obc_three_site_reconstruction_of_pair_sections}
    \leanok
    \uses{lem:varying_obc_three_site_reconstruction}
    Assume the same row-isometry and middle-site spanning conditions.
    Let $\phi:I\times J\times K\to\mathbb C$. Suppose that for every
    fixed $k$ there is a matrix $X_k$ with
    $\phi(i,j,k)=\operatorname{tr}(X_kA_iB_j)$, and for every fixed $i$
    there is a matrix $Y_i$ with
    $\phi(i,j,k)=\operatorname{tr}(Y_iB_jC_k)$. Then there is one
    $Z\in M_{e,a}(\mathbb C)$ such that
    \begin{align}
        \phi(i,j,k) & =\operatorname{tr}(ZA_iB_jC_k)
    \end{align}
    simultaneously for every $i,j,k$.
\end{lemma}
\begin{proof}\leanok
    Choose boundary matrices from the two local section conditions. Their
    trace representations agree because they both give $\phi$.
    The canonical adjoint reconstruction supplies the common boundary.
\end{proof}

\begin{lemma}[Blocking preserves the rectangular row isometry]
    \label{lem:varying_obc_block_row_isometry}
    \lean{Matrix.obc_block_row_isometry}
    \leanok
    If $\sum_iA_iA_i^\dagger=I_a$ and $\sum_jB_jB_j^\dagger=I_b$, then the
    blocked family satisfies
    \begin{align}
        \sum_{i,j}(A_iB_j)(A_iB_j)^\dagger & =I_a.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Reverse the product under adjoints and first sum over the middle
    physical letter:
    \begin{align}
        \sum_{i,j}A_iB_jB_j^\dagger A_i^\dagger
         & =\sum_iA_i\Bigl(\sum_jB_jB_j^\dagger\Bigr)A_i^\dagger
        =\sum_iA_iA_i^\dagger=I_a.
    \end{align}
\end{proof}


\subsection{Reconstruction on the whole open chain}

\begin{lemma}[Products on the prefix bond spaces]
    \label{lem:varying_obc_prefix_product_row_isometry}
    \lean{Matrix.obc_rectangularEval_snoc,
        Matrix.obc_rectangularEval_row_isometry}
    \leanok
    \uses{lem:varying_obc_block_row_isometry}
    Let $B^{[j]}_i\in M_{D_j,D_{j+1}}(\mathbb C)$ be the site matrices,
    and put $P_m(i_0,\ldots,i_{m-1})=B^{[0]}_{i_0}\cdots B^{[m-1]}_{i_{m-1}}$,
    with $P_0=I_{D_0}$. Then
    \begin{align}
        P_{m+1}(u,i) & =P_m(u)B^{[m]}_i.
    \end{align}
    If every site is canonically normalized by
    $\sum_iB^{[j]}_i B^{[j]\dagger}_i=I_{D_j}$, then
    \begin{align}
        \sum_{u}P_m(u)P_m(u)^\dagger & =I_{D_0}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    The first equality follows from ordered multiplication. Induction on the
    prefix length applies the two-family row-isometry identity at each step.
\end{proof}

\begin{lemma}[Adjacent coefficient sections determine a full boundary]
    \label{lem:varying_obc_full_chain_reconstruction}
    \lean{Matrix.obc_trace_reconstruction_of_adjacent_pair_sections,
        Matrix.obc_scalar_reconstruction_of_adjacent_pair_sections}
    \leanok
    \uses{lem:varying_obc_pair_sections_reconstruct,
        lem:varying_obc_prefix_product_row_isometry}
    Consider an open chain of $N\ge2$ sites, canonically normalized as above,
    and assume that the matrices at each site span the corresponding
    rectangular matrix space. Let $\phi$ be a coefficient family on the
    physical configurations. Suppose that for every adjacent pair $j,j+1$
    and every fixed configuration outside that pair, there is a boundary
    matrix $X$ such that
    \begin{align}
        \phi(u,i,k,v) & =\operatorname{tr}
        \bigl(X B^{[j]}_i B^{[j+1]}_k\bigr).
    \end{align}
    Then one matrix $Z\in M_{D_N,D_0}(\mathbb C)$ represents the entire family:
    \begin{align}
        \phi(i_0,\ldots,i_{N-1}) & =\operatorname{tr}
        \bigl(Z B^{[0]}_{i_0}\cdots B^{[N-1]}_{i_{N-1}}\bigr).
    \end{align}
    If $D_0=D_N=1$, the coefficient family is a scalar multiple of the
    represented open-chain state.
\end{lemma}
\begin{proof}\leanok
    Inductively suppose that every fixed suffix gives a trace representation
    on a prefix of length $m\ge2$. Separate the first $m-1$ sites as the
    prefix product, followed by sites $m-1,m$. The induction hypothesis gives
    the left two-family sections, and the original adjacent-pair condition
    gives the right sections. The prefix product is row-isometric, and the
    middle site satisfies the spanning condition. The three-site adjoint
    reconstruction therefore extends the represented prefix by one site.
    Iterating gives one boundary for the full chain. Singleton endpoint
    spaces make this boundary a scalar. This proves the representation
    conclusion without identifying normalized vectors of different phases.
\end{proof}

\subsection{The open parent Hamiltonian}

\begin{definition}[Pair boundary ranges and open local terms]
    \label{def:varying_obc_parent_projectors}
    \lean{Matrix.obcPairBoundary, Matrix.obcPairBoundary_apply,
        Matrix.obcPairProjector,
        Matrix.obcPairSection, Matrix.obcParentHamiltonian}
    \leanok
    Let $B^{[j]}_i\in M_{D_j,D_{j+1}}(\mathbb C)$. The two-site
    boundary range at $j$ is
    \begin{align}
        G_j & =\left\{\left(\operatorname{tr}
        (XB^{[j]}_iB^{[j+1]}_k)\right)_{i,k}:
        X\in M_{D_{j+2},D_j}(\mathbb C)\right\}.
    \end{align}
    Write $P_j$ for the orthogonal projection onto $G_j^\perp$.
    If $u,v$ are fixed prefix and suffix configurations of lengths $j$ and
    $N-j-2$, let $(R_{j,u,v}\phi)(i,k)=\phi(u,i,k,v)$. The open parent
    Hamiltonian is
    \begin{align}
        H_N & =\sum_{j=0}^{N-2}\sum_{u,v}R_{j,u,v}^\dagger P_jR_{j,u,v}.
    \end{align}
    Summing over each outside configuration once extends $P_j$ by the identity
    outside its adjacent pair. There is no interaction across the endpoints.
    This is the local Hamiltonian of
    \cite[Theorem~9, lines~1201--1217]{PerezGarcia2007Matrix}.
\end{definition}

\begin{lemma}[Kernel of a pair projector]
    \label{lem:varying_obc_pair_projector_kernel}
    \lean{Matrix.obcPairProjector_isPositive,
        Matrix.obcPairProjector_apply_eq_zero_iff}
    \leanok
    \uses{def:varying_obc_parent_projectors}
    Each $P_j$ is positive. Moreover $P_j\xi=0$ if and only if there is a
    rectangular matrix $X$ with
    $\xi(i,k)=\operatorname{tr}(XB^{[j]}_iB^{[j+1]}_k)$ for every $i,k$.
\end{lemma}
\begin{proof}\leanok
    An orthogonal projection is positive, and the kernel of the projection
    onto $G_j^\perp$ is $G_j$. Membership in this range is exactly the
    displayed boundary representation.
\end{proof}

\begin{lemma}[Hamiltonian kernel and adjacent coefficient sections]
    \label{lem:varying_obc_parent_kernel_sections}
    \lean{Matrix.obcParentHamiltonian_isPositive,
        Matrix.obcParentHamiltonian_apply_eq_zero_iff,
        Matrix.obcParentHamiltonian_pair_sections_of_apply_eq_zero,
        Matrix.obcParentHamiltonian_apply_eq_zero_iff_pair_sections}
    \leanok
    \uses{def:varying_obc_parent_projectors,
        lem:varying_obc_pair_projector_kernel}
    The Hamiltonian $H_N$ is positive. A vector $\phi$ satisfies $H_N\phi=0$
    if and only if, for every adjacent
    pair and every fixed outside configuration, there is a
    rectangular boundary matrix $X$ such that
    \begin{align}
        \phi(u,i,k,v) & =\operatorname{tr}(XB^{[j]}_iB^{[j+1]}_k)
    \end{align}
    for every $i,k$. Neither a spanning condition nor canonical
    normalization is required for this equivalence.
\end{lemma}
\begin{proof}\leanok
    Every compression $R^\dagger P_jR$ is positive, so a vector lies in the kernel
    of their sum exactly when each compression annihilates it. The quadratic
    form identity
    $\langle R^\dagger P_jR\phi,\phi\rangle=\langle P_jR\phi,R\phi\rangle$
    and positivity identify this with $P_jR\phi=0$. The pair-projector kernel
    characterization gives the boundary matrix for each outside configuration.
    In the reverse direction these boundary representations make all the
    compressions vanish. Ordered prefix, pair and suffix concatenation
    identifies the same physical configurations in either description.
\end{proof}

\begin{lemma}[Boundary states lie in the open parent kernel]
    \label{lem:varying_obc_boundary_states_parent_kernel}
    \lean{Matrix.obc_rectangularEval_append, Matrix.obcBoundaryState,
        Matrix.obcBoundaryState_pair_section,
        Matrix.obcParentHamiltonian_apply_obcBoundaryState}
    \leanok
    \uses{def:varying_obc_parent_projectors,
        lem:varying_obc_parent_kernel_sections}
    Let $Z\in M_{D_N,D_0}(\mathbb C)$ and define the physical vector
    \begin{align}
        \phi_Z(i_0,\ldots,i_{N-1})
         & =\operatorname{tr}(ZB^{[0]}_{i_0}\cdots B^{[N-1]}_{i_{N-1}}).
    \end{align}
    Then $H_N\phi_Z=0$, with no spanning or canonical normalization assumption.
    More precisely, if $P_u$ and $P_v$ are the ordered prefix and suffix
    products outside the pair at $j$, its local boundary is
    $X=P_v ZP_u\in M_{D_{j+2},D_j}(\mathbb C)$.
\end{lemma}
\begin{proof}\leanok
    Concatenating physical words multiplies their rectangular products.
    This identity follows by induction on the suffix length, adjoining its
    last matrix. Hence cyclicity of the rectangular trace gives
    \begin{align}
        \phi_Z(u,i,k,v)
         & =\operatorname{tr}(ZP_uB^{[j]}_iB^{[j+1]}_kP_v)
        =\operatorname{tr}((P_vZP_u)B^{[j]}_iB^{[j+1]}_k).
    \end{align}
    Every pair section therefore belongs to its boundary range, so the
    Hamiltonian kernel characterization applies.
\end{proof}
