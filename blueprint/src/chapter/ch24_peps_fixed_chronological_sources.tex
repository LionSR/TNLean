\section{Fixed sources for the chronological partial expansion}

The two separated contractions in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS} (\texttt{04-compression.tex}, lines 342--434) act on common free inputs obtained by placing the sources of every partial branch into fixed ordered register spaces inherited from the original gate occurrences.

\begin{definition}[The retained source positions]
    \label{def:peps_partial_fixed_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialPositions}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlotEquiv}
    \leanok
    \uses{thm:peps_partial_source_order}
    For original ordered source positions $\mathcal E$ and affected parties $\mathcal A$, retain the ordered sublist
    \begin{align}
        \mathcal E_{\mathcal A}
        =\{e\in\mathcal E:
        p_e\in\mathcal A\text{ or }q_e\in\mathcal A\},
        \qquad r=|\mathcal E_{\mathcal A}|,
        \label{eq:peps_partial_fixed_positions}
    \end{align}
    enumerated as $i\mapsto e_i$ ($0\le i<r$), with grouped endpoints and fixed halfspaces $\C^{u_{e_i}},\C^{v_{e_i}}$.
\end{definition}

\begin{theorem}[The common ordered register spaces]
    \label{thm:peps_partial_fixed_layout}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.mem_partialPositions_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlotEquiv_apply}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.layout_partialSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlot_spec}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.layout_partialSlots_eq}
    \leanok
    \uses{def:peps_partial_fixed_positions,thm:peps_partial_source_order}
    At reference index $i$, the source has distinct grouped endpoints from $e_i$ and halfspaces $\C^{u_{e_i}},\C^{v_{e_i}}$. Every partial branch $\xi\in\Xi_{\mathcal A}$ has this exact ordered source-register layout, and $e\in\mathcal E_{\mathcal A}$ if and only if $p_e\in\mathcal A$ or $q_e\in\mathcal A$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_circuit_incidence,thm:peps_affected_owner_equality,
        thm:peps_partial_source_order}
    Combine Theorem~\ref{thm:peps_source_circuit_incidence}, Theorem~\ref{thm:peps_affected_owner_equality}, and Theorem~\ref{thm:peps_partial_source_order} on the filtered list (\ref{eq:peps_partial_fixed_positions}).
\end{proof}

\begin{theorem}[Preparation in the fixed chronological sources]
    \label{thm:peps_partial_fixed_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_partial_source_preparation}
    \leanok
    \uses{thm:peps_partial_fixed_layout,thm:peps_source_preparation}
    For an allowed chronological composition and every partial choice $\xi\in\Xi_{\mathcal A}$, there exist unit vectors
    \begin{align}
        \eta_{\xi,i}\in
        \C^{u_{e_i}}\otimes\C^{v_{e_i}},
        \qquad 0\leq i<r,
        \label{eq:peps_partial_fixed_unit_vectors}
    \end{align}
    and a source-free allowed composition $V_\xi$ (whose input and output register layouts are independent of $\xi$) such that, with $P_{\eta_\xi}$ preparing $(\eta_{\xi,i})_{i=0}^{r-1}$,
    \begin{align}
        W_\xi=V_\xi P_{\eta_\xi}.
        \label{eq:peps_partial_fixed_preparation}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_partial_source_inventory,thm:peps_partial_fixed_layout,
        thm:peps_source_preparation,thm:peps_common_source_lists}
    Transport the normalized source inventory of Theorem~\ref{thm:peps_partial_source_inventory} along the common layout of Theorem~\ref{thm:peps_partial_fixed_layout} and Theorem~\ref{thm:peps_common_source_lists}, then apply Theorem~\ref{thm:peps_source_preparation}.
\end{proof}

\begin{definition}[The original label and vector at a position]
    \label{def:peps_partial_original_source_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choiceAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceVectorAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlotChoice}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSlotVector}
    \leanok
    \uses{def:peps_partial_source_choices,def:peps_partial_fixed_positions}
    For a touched gate occurrence $g$, $\xi\in\Xi_{\mathcal A}$ determines $\xi_g\in\Xi_g$. Writing $\nu_{e,\alpha}\in\C^{u_e}\otimes\C^{v_e}$ for the source vector at $e=(g,j)$ and $\alpha\in\Xi_g$, set at reference index $i$ ($e_i=(g_i,j_i)$):
    \begin{align}
        \eta^{\mathrm{orig}}_{\xi,i}
        :=\nu_{e_i,\xi_{g_i}}\in\C^{u_{e_i}}\otimes\C^{v_{e_i}}.
        \label{eq:peps_partial_original_source_vector}
    \end{align}
\end{definition}

\begin{theorem}[Recovery of the locally selected source vectors]
    \label{thm:peps_partial_original_source_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceVectorAt_norm}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isTouched_of_source_endpoint}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceAt_eq_mapOwner_sourceVectorAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eq_partialSlotVector_of_inventory_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_partial_source_preparation_with_original_vectors}
    \leanok
    \uses{def:peps_partial_original_source_vector,thm:peps_partial_fixed_preparation}
    Every $e_i\in\mathcal E_{\mathcal A}$ belongs to a touched gate $g_i$, $\|\nu_{e,\alpha}\|=1$, and (\ref{eq:peps_partial_fixed_preparation}) of Theorem~\ref{thm:peps_partial_fixed_preparation} holds with $\eta_{\xi,i}=\eta^{\mathrm{orig}}_{\xi,i}$ from (\ref{eq:peps_partial_original_source_vector}), which depends only on $\xi_{g_i}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_circuit_incidence,thm:peps_partial_source_order,
        thm:peps_partial_fixed_layout,thm:peps_partial_fixed_preparation}
    By Theorem~\ref{thm:peps_source_circuit_incidence}, $e_i\in\mathcal E_{\mathcal A}$ implies $C_{g_i}\cap\mathcal A\ne\varnothing$, and comparing the ordered source inventory (Theorem~\ref{thm:peps_partial_source_order}) with the common layout (Theorem~\ref{thm:peps_partial_fixed_layout}) identifies the $i$th prepared vector in (\ref{eq:peps_partial_fixed_preparation}) with $\eta^{\mathrm{orig}}_{\xi,i}$.
\end{proof}

\begin{theorem}[Two contractions for the prepared sources]
    \label{thm:peps_actual_source_selective_partition}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_selective_partition_of_word}
    \leanok
    \uses{thm:peps_source_preparation,thm:peps_selective_partition}
    For an allowed composition $W$, a bipartition of its parties, and a set of free source positions containing all cut-crossing sources, let $P_{\mathrm{fix}}$ prepare the original vectors at the remaining (fixed) positions. Then there exist source-free allowed compositions $V,A_+,A_-$ (chosen independently of the free vectors) such that
    \begin{align}
        \|A_+\|\leq1,\qquad \|A_-\|\leq1,
        \qquad
        J_{\mathrm{out}}VP_{\mathrm{fix}}
        =(A_+\otimes A_-)J_{\mathrm{free}},
        \label{eq:peps_actual_source_selective_partition}
    \end{align}
    where $J_{\mathrm{free}},J_{\mathrm{out}}$ group registers by side. For any family $\zeta$ of free source vectors with preparation $P_\zeta^{\mathrm{free}}$ and full preparation $P_{\mathrm{fill}(\zeta)}=P_{\mathrm{fix}}P_\zeta^{\mathrm{free}}$,
    \begin{align}
        J_{\mathrm{out}}VP_{\mathrm{fill}(\zeta)}
        =(A_+\otimes A_-)J_{\mathrm{free}}
        P_\zeta^{\mathrm{free}},
        \notag
    \end{align}
    and taking the original free vectors recovers $VP_{\mathrm{fill}(\zeta)}=W$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_source_preparation,thm:peps_selective_preparation_recovery,
        thm:peps_selective_partition}
    Write $W=VP_\eta$ by Theorem~\ref{thm:peps_source_preparation} and apply Theorem~\ref{thm:peps_selective_partition} and Theorem~\ref{thm:peps_selective_preparation_recovery} to the internal unit vectors prepared by $P_{\mathrm{fix}}$.
\end{proof}

\begin{definition}[Choices as labels at touched occurrences]
    \label{def:peps_touched_labels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.projectAllChoices}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.TouchedLabels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesOfTouchedLabels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesTouchedLabelsEquiv}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.nonemptyGateLocations}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allChoicesNonemptyLabelsEquiv}
    \leanok
    \uses{def:peps_partial_original_source_vector}
    For affected parties $\mathcal A$, $\xi\mapsto(\xi_g)_{g:\,C_g\cap\mathcal A\neq\varnothing}$ defines a bijection $\Xi_{\mathcal A}\cong\prod_{g:\,C_g\cap\mathcal A\neq\varnothing}\Xi_g$; at $\mathcal A=P$, with $\mathcal G^+=\{g:C_g\neq\varnothing\}$, this is $\Xi_P\cong\prod_{g\in\mathcal G^+}\Xi_g$, and restriction gives the projection $\pi_{\mathcal A}\colon\Xi_P\to\Xi_{\mathcal A}$.
\end{definition}

\begin{theorem}[Projected choices keep the original labels]
    \label{thm:peps_touched_labels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choiceAt_projectAllChoices}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choiceAt_choicesOfTouchedLabels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesOfTouchedLabels_choiceAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isTouched_all_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allChoicesNonemptyLabelsEquiv_apply}
    \leanok
    \uses{def:peps_touched_labels}
    For $\xi\in\Xi_P$ and $g$ with $C_g\cap\mathcal A\ne\varnothing$, $(\pi_{\mathcal A}\xi)_g=\xi_g$; moreover, $C_g\cap P\ne\varnothing\iff C_g\ne\varnothing$.
\end{theorem}
\begin{proof}
    \leanok
    By structural induction on the chronological composition, touched gates retain $\xi_g$ while untouched gates have singleton choice sets.
\end{proof}

\begin{theorem}[Labels visible at one party]
    \label{thm:peps_incident_gate_labels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partySelector_affectedOwner_all_apply}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isTouched_groupedParty_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.OriginalGateLabelsAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesOfOriginalGateLabelsAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.originalGateLabelsAtOfAllChoices}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesOfOriginalGateLabelsAt_eq_projectAllChoices}
    \leanok
    \uses{thm:peps_touched_labels,def:compression_operator_gate_labels}
    For $p\in P$, the occurrences touched by $\mathcal A=\{p\}$ are $\{g\in\mathcal G^+:p\in C_g\}$, giving $\Xi_{\{p\}}\cong\prod_{g\in\mathcal G^+,\,p\in C_g}\Xi_g$ with $(\pi_{\{p\}}\xi)_g=(\xi_p)_g=\xi_g$ for all $\xi\in\Xi_P$.
\end{theorem}
\begin{proof}
    \leanok
    Apply Theorem~\ref{thm:peps_touched_labels} at $\mathcal A=\{p\}$.
\end{proof}

\begin{theorem}[Coefficients factor over participating occurrences]
    \label{thm:peps_chronological_coefficients}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient_eq_prod_dite}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient_all_eq_prod_nonemptyGateLocations}
    \leanok
    \uses{def:peps_touched_labels}
    For $\xi\in\Xi_{\mathcal A}$, $c_\xi=\prod_{g:\,C_g\cap\mathcal A\ne\varnothing} c_{g,\xi_g}$; in particular, for $\xi\in\Xi_P$,
    \begin{align}
        c_\xi & =\prod_{g\in\mathcal G^+}c_{g,\xi_g}
        \label{eq:peps_chronological_coefficients}
    \end{align}
    (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 351--381 and 565--573).
\end{theorem}
\begin{proof}
    \leanok
    Induction on the chronological composition multiplies coefficients over disjoint occurrence sets, with untouched occurrences contributing $1$.
\end{proof}

\begin{theorem}[Mixed coefficients at the original gate occurrences]
    \label{thm:peps_mixed_chronological_coefficients}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient_all_mul_conj_eq_prod_nonemptyGateLocations}
    \leanok
    \uses{def:peps_touched_labels}
    For any chronological source composition on $P$ and $\xi,\zeta\in\Xi_P$,
    \begin{align}
        c_\xi\overline{c_\zeta}
        =\prod_{g\in\mathcal G^+}
        c_{g,\xi_g}\overline{c_{g,\zeta_g}}
        \label{eq:peps_mixed_chronological_coefficients}
    \end{align}
    (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 351--381 and 565--573).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_chronological_coefficients}
    Multiply (\ref{eq:peps_chronological_coefficients}) for $\xi$ by the complex conjugate of (\ref{eq:peps_chronological_coefficients}) for $\zeta$.
\end{proof}

\begin{theorem}[Reindexing the mixed expansion by gate labels]
    \label{thm:peps_mixed_gate_label_sum}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_coefficient_all_mul_conj_smul_eq_sum_nonemptyGateLabels}
    \leanok
    \uses{def:peps_touched_labels}
    With $L=\prod_{g\in\mathcal G^+}\Xi_g$, the bijection $e\colon\Xi_P\to L$ of Definition~\ref{def:peps_touched_labels}, and any $K\colon\Xi_P\times\Xi_P\to\C^{I\times J}$,
    \begin{align}
        \sum_{\xi\in\Xi_P}\sum_{\zeta\in\Xi_P}
        c_\xi\overline{c_\zeta}\,K(\xi,\zeta)
         & =\sum_{\eta\in L}\sum_{\theta\in L}
        \left(\prod_{g\in\mathcal G^+}
        c_{g,\eta_g}\overline{c_{g,\theta_g}}\right)
        K(e^{-1}\eta,e^{-1}\theta)
        \label{eq:peps_mixed_gate_label_sum}
    \end{align}
    (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 351--381 and 565--573).
\end{theorem}
\begin{proof}\leanok
    \uses{def:peps_touched_labels,thm:peps_mixed_chronological_coefficients}
    Substitute (\ref{eq:peps_mixed_chronological_coefficients}) and reindex by $(\eta,\theta)=(e\xi,e\zeta)$.
\end{proof}
