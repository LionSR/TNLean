\section{The Ising anyon chain: exact arithmetic in $\Z[\sqrt2]$}%
\label{sec:asymex_ising_ring}

The topological-symmetry tensors of the Ising anyon chain have entries built
from the F-symbols of the Ising fusion category, and the quantum dimension of
the non-abelian object is $\sqrt2$: after a global rescaling, every entry lies
in the quadratic ring $\Z[\sqrt2]$. This section records the ring, its
embedding into the complex numbers, and the transport of the normality
criterion of Theorem~\ref{thm:asymex_normal_units} along that embedding; the
tensors themselves are introduced in
Section~\ref{sec:asymex_ising_tensors}.

\begin{definition}[The ring $\Z[\sqrt2]$ and its complex embedding]%
    \label{def:asymex_zsqrt2_ring}
    \lean{zsqrt2ToComplex, MPSTensor.complexOfZsqrt2, MPSTensor.mulZsqrt2Tensor,
        MPSTensor.actZsqrt2Tensor}
    \leanok
    The ring of \emph{$\sqrt2$-integers} is $\Z[\sqrt2]$, presented by pairs of
    integer coordinates $(a,b)\mapsto a+b\sqrt2$ with the multiplication
    obtained from $\sqrt2^2=2$. It embeds into $\C$ by sending $\sqrt2$ to the
    positive real square root of two; a matrix over $\Z[\sqrt2]$ becomes a
    complex matrix by applying that embedding entrywise. For matrix product
    operator tensors $M$ and $N$ over $\Z[\sqrt2]$, their bond-space product is
    $(M\mathbin{\cdot}N)^{ik}=\sum_jM^{ij}\otimes N^{jk}$, and for a matrix
    product operator tensor $M$ and a state tensor $A$ over $\Z[\sqrt2]$, the
    bond-space action is $(M\cdot A)^i=\sum_jM^{ij}\otimes A^j$.
\end{definition}

\begin{theorem}[Exact arithmetic in $\Z[\sqrt2]$]%
    \label{thm:asymex_zsqrt2_ring}
    % Principal declaration: MPSTensor.complexOfZsqrt2_mul.
    \lean{zsqrt2ToComplex_injective, zsqrt2ToComplex_ne_zero,
        MPSTensor.complexOfZsqrt2_mul, MPSTensor.mulTensor_complexOfZsqrt2,
        MPSTensor.actTensor_complexOfZsqrt2,
        MPSTensor.isNormal_of_single_eq_smul_zsqrt2}
    \leanok
    \uses{def:asymex_zsqrt2_ring}
    The embedding of $\Z[\sqrt2]$ into $\C$ is injective, because two is not
    the square of an integer, and it sends every nonzero element to a nonzero
    complex number. The entrywise embedding of matrices over $\Z[\sqrt2]$ is
    multiplicative, and it commutes with the bond-space product and with the
    bond-space action of tensors over $\Z[\sqrt2]$. A tensor that is the
    entrywise embedding of a tensor over $\Z[\sqrt2]$, and whose letters
    realize every matrix unit of its bond algebra as a nonzero
    $\Z[\sqrt2]$-multiple, is normal at blocking length one.
\end{theorem}

\begin{proof}
    \leanok
    Injectivity follows because a square root of two in $\Z[\sqrt2]$ would
    give an integer square root of two. Multiplicativity of the entrywise
    embedding is the fact that it is induced by a ring homomorphism, and the
    two compatibility statements follow by expanding the bond-space product and
    the bond-space action entrywise. Normality is the same normalization
    argument as Theorem~\ref{thm:asymex_normal_units}, transported along the
    embedding: dividing the designated letter by its nonzero image exhibits
    each matrix unit in the span of the one-site matrices.
\end{proof}

\section{The three topological-symmetry tensors of the Ising anyon chain}%
\label{sec:asymex_ising_tensors}

The Ising fusion category has three simple objects, the vacuum $1$, the
fermion $\psi$, and the non-abelian anyon $\sigma$, fused by
$\psi\otimes\psi=1$, $\sigma\otimes\psi=\sigma$ and
$\sigma\otimes\sigma=1\oplus\psi$. Each object $a$ gives a matrix product
operator tensor $A_a$ on the ten fusion-tree labels $h=(x',\rho,x)$ with
$x\in x'\otimes\rho$, whose letters are the F-symbols
$[F^{a\,x'\,\rho}_y]_{y'x}$ of the category; a letter of $A_a$ vanishes unless
its two labels carry the same middle label $\rho$.

\begin{definition}[The three tensors $A_1$, $A_\psi$, $A_\sigma$]%
    \label{def:asymex_ising_tensors}
    \lean{IsingTwist.isingRho, IsingTwist.isingOneZ, IsingTwist.isingPsiZ,
        IsingTwist.isingSigmaZ, IsingTwist.isingOne, IsingTwist.isingPsi,
        IsingTwist.isingSigma}
    \leanok
    \uses{def:asymex_zsqrt2_ring, def:mpo_tensor}
    The ten fusion-tree labels, in the order
    $(1,1,1)$, $(1,\psi,\psi)$, $(1,\sigma,\sigma)$, $(\psi,1,\psi)$,
    $(\psi,\psi,1)$, $(\psi,\sigma,\sigma)$, $(\sigma,1,\sigma)$,
    $(\sigma,\psi,\sigma)$, $(\sigma,\sigma,1)$, $(\sigma,\sigma,\psi)$, carry
    the middle labels $1,\psi,\sigma,1,\psi,\sigma,1,\psi,\sigma,\sigma$. The
    tensor $A_1$ is on the vertical space $P_1=[(1,1),(\psi,\psi),(\sigma,\sigma)]$
    and has a matrix unit at each of its ten diagonal letters; the tensor
    $A_\psi$ is on $P_\psi=[(1,\psi),(\psi,1),(\sigma,\sigma)]$ and has a
    signed matrix unit at each of ten off-diagonal or diagonal letters, the
    sign at the letter $\bigl((\sigma,\psi,\sigma),(\sigma,\psi,\sigma)\bigr)$
    being the F-symbol $[F^{\psi\sigma\psi}_\sigma]=-1$; the tensor
    $\sqrt2A_\sigma$, on $P_\sigma=[(1,\sigma),(\psi,\sigma),(\sigma,1),(\sigma,\psi)]$,
    has sixteen nonzero letters, each a matrix unit scaled by $\pm1$ or
    $\pm\sqrt2$, the rescaled F-symbols of the $\sigma\sigma\sigma$ fusion.
\end{definition}

\begin{theorem}[Sector vanishing of the three tensors]%
    \label{thm:asymex_ising_sector_vanishing}
    \lean{IsingTwist.isingOneZ_eq_zero_of_rho_ne,
        IsingTwist.isingPsiZ_eq_zero_of_rho_ne,
        IsingTwist.isingSigmaZ_eq_zero_of_rho_ne}
    \leanok
    \uses{def:asymex_ising_tensors}
    For each of the three tensors, a letter at a pair of labels with distinct
    middle labels vanishes.
\end{theorem}

\begin{proof}
    \leanok
    Each tensor has, by inspection of its ten or sixteen nonzero letters, a
    zero matrix at every pair of labels with distinct middle labels; a finite
    verification over the entries of Definition~\ref{def:asymex_ising_tensors}.
\end{proof}

\begin{theorem}[Normality of the three tensors]%
    \label{thm:asymex_ising_normal}
    \lean{IsingTwist.isingOne_isNormal, IsingTwist.isingPsi_isNormal,
        IsingTwist.isingSigma_isNormal}
    \leanok
    \uses{def:asymex_ising_tensors, def:normal}
    Each of the three tensors is normal at blocking length one: the ten
    letters of $A_1$ span $\MN3$, the ten letters of $A_\psi$ span $\MN3$, and
    the sixteen letters of $\sqrt2A_\sigma$ span $\MN4$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_zsqrt2_ring}
    Every matrix unit of the relevant bond algebra is realized, up to a
    nonzero factor in $\Z[\sqrt2]$, by exactly one letter of the tensor, a
    finite verification over integer or $\Z[\sqrt2]$-valued entries;
    Theorem~\ref{thm:asymex_zsqrt2_ring} then transports normality to the
    complex matrices.
\end{proof}

\section{The weighted bond object and its stacked product}%
\label{sec:asymex_ising_bond_object}

At the boundary of the Ising renormalization fixed point, a horizontal bond
carries the weighted bond object $\Theta=\lambda_1A_1\oplus\lambda_\psi A_\psi
    \oplus\lambda_\sigma A_\sigma$, and the vertical block of flag $\sigma$ is the
strand $A_\sigma$ with $\Theta$ inserted before it: the stacked product
$\Theta\star A_\sigma$. This section and the next formalize the two-object
part $\Theta_2=\lambda_1A_1\oplus\lambda_\psi A_\psi$ of that insertion, at
the generic point $(\lambda_1,\lambda_\psi)=(1/2,3/10)$ of the renormalization
family, scaled to the integers $5,3$, with the $\sigma$ strand scaled by
$\sqrt2$ so that every entry lies in $\Z[\sqrt2]$.

This stacked product is a product of two matrix product operators on the
pair alphabet of Section~\ref{sec:asym_mpo_product}, not an operator applied
to a state: the object acted on is the strand $A_\sigma$ itself, read as a
matrix product operator, and the theorem invoked below is
Theorem~\ref{thm:asym_mpo_product}. Section~\ref{sec:asymex_ising_action}
records a separate, genuine action-tensor instance, in the sense of
Corollary~\ref{cor:asym_action_tensors}, of the full weighted bond object
applied to a fusion-tree product state.

\begin{definition}[The weighted bond object and the stacked tensor]%
    \label{def:asymex_ising_bond_object}
    \lean{IsingTwist.thetaTwoZ, IsingTwist.thetaTwo, IsingTwist.isingBondObjectZ,
        IsingTwist.isingBondObject}
    \leanok
    \uses{def:asymex_ising_tensors, def:mpdo_operator_product}
    The weighted bond object is $\Theta_2=5A_1\oplus3A_\psi$ on
    $\C^6=P_1\oplus P_\psi$. The source is the stacked product
    $B=\Theta_2\star(\sqrt2A_\sigma)$ on the hundred-letter pair alphabet, of

    $4\,(\Theta_2\text{ index})+(P_\sigma\text{ index})$.
\end{definition}

\section{The gauge and the block decomposition}%
\label{sec:asymex_ising_gauge}

The change of bond coordinates of the two-object instance is a signed
permutation of the twenty-four bond coordinates of $B$: its first four rows
are the F-move isometry of the fusion $1\otimes\sigma=\sigma$, the next four
the F-move isometry of $\psi\otimes\sigma=\sigma$, and the remaining sixteen
rows a coordinate complement. The single sign occurring is the F-symbol
$[F^{\psi\sigma\psi}_\sigma]=-1$ of the $\psi$ line.

\begin{definition}[Signed permutation matrices and the Ising gauge]%
    \label{def:asymex_ising_gauge}
    \lean{MPSTensor.signedPermMatrix, IsingTwist.isingPerm, IsingTwist.isingSign,
        IsingTwist.isingGaugeZ, IsingTwist.isingGauge, IsingTwist.isingSlots,
        IsingTwist.isingWeights, IsingTwist.isingTargets, IsingTwist.isingBlockZ}
    \leanok
    \uses{def:asymex_ising_bond_object, def:asymex_explicit_gauge}
    For a map $\pi$ on an index set and a sign vector $s$, the associated
    \emph{signed permutation matrix} has the entry $s(x)$ in row $x$ and
    column $\pi(x)$. The Ising gauge is the signed permutation matrix of the
    F-move rows described above, an explicit gauge in the sense of
    Definition~\ref{def:asymex_explicit_gauge}. The two weighted slots are the
    channel $\sigma$ reached through the bond object $1$, with weight $5$, and
    through the bond object $\psi$, with weight $3$; both slots carry the
    target tensor $\sqrt2A_\sigma$, of bond dimension four, and there are
    $16=24-2\cdot4$ zero slots.
\end{definition}

\begin{theorem}[The Ising gauge is an involutive isometry]%
    \label{thm:asymex_ising_gauge_unitary}
    \lean{MPSTensor.signedPermMatrix_mul_mul_transpose,
        IsingTwist.isingGaugeZ_mul_transpose, IsingTwist.isingGaugeZ_transpose_mul}
    \leanok
    \uses{def:asymex_ising_gauge}
    Conjugating a matrix by a signed permutation matrix and its transpose
    reads off the entries of the conjugated matrix at the permuted
    coordinates, each carrying two signs. The Ising gauge $G$ satisfies
    $GG^{\mathsf T}=G^{\mathsf T}G=\id$, so its transpose is its inverse.
\end{theorem}

\begin{proof}
    \leanok
    The conjugation formula is a direct expansion of the matrix product
    against the two Kronecker deltas of the signed permutation matrices. For
    the Ising gauge, both products with its transpose are the identity, a
    finite verification over the twenty-four rows.
\end{proof}

\begin{lemma}[The letter identity of the Ising bond-object twist]%
    \label{lem:asymex_ising_letter_identity}
    % Principal declaration: IsingTwist.isingGaugeZ_conj.
    \lean{IsingTwist.isingGaugeZ_conj, IsingTwist.isingBondObject_conjMatrix}
    \leanok
    \uses{def:asymex_ising_gauge}
    In the block coordinates of Definition~\ref{def:asymex_ising_gauge}, every
    letter of the stacked tensor $B$ conjugated by the Ising gauge is block
    diagonal with the blocks $5(\sqrt2A_\sigma)^{(h,h')}$,
    $3(\sqrt2A_\sigma)^{(h,h')}$ and sixteen zeros.
\end{lemma}

\begin{proof}
    \leanok
    A letter at a pair of labels with distinct middle labels vanishes by
    Theorem~\ref{thm:asymex_ising_sector_vanishing}, matching the zero
    conjugated block. For a pair of labels sharing a middle label, the
    conjugated letter reduces, by Theorem~\ref{thm:asymex_ising_gauge_unitary},
    to a sum over the middle labels of that one sector, and this reduced
    identity is a finite verification over the entries of
    Definitions~\ref{def:asymex_ising_tensors} and~\ref{def:asymex_ising_gauge},
    split into three disjoint families of pair labels by the shared middle
    label so that no single verification carries more than a few of the
    conjugated matrices.
\end{proof}

\section{Compression onto the $\sigma$ strand}%
\label{sec:asymex_ising_compression}

\begin{theorem}[Compression of the stacked Ising bond object]%
    \label{thm:asymex_ising_compression}
    % Principal declaration: IsingTwist.isingCompression.

    the triangularity condition and the identification of the two matched
    diagonal blocks with the weighted target, by
    Theorem~\ref{thm:asymex_explicit_gauge}. The dimension count is clause
    (vii) of Theorem~\ref{thm:asym_multiblock}.
\end{proof}

\section{The length-dependent structure constant}%
\label{sec:asymex_ising_trace}

\begin{theorem}[The trace identity of the Ising bond-object twist]%
    \label{thm:asymex_ising_trace}
    \lean{IsingTwist.isingBondObject_trace_evalWord, IsingTwist.isingTwist_mpo}
    \leanok
    \uses{thm:asymex_ising_compression, prop:asym_compression_trace}
    For every nonempty word $w$ over the alphabet of pair letters,
    \begin{align}
        \tr(B^w) & =\bigl(5^{|w|}+3^{|w|}\bigr)\tr\bigl((\sqrt2A_\sigma)^w\bigr).
        \label{eq:asymex_ising_trace}
    \end{align}
    Equivalently, at every positive length $L$, the periodic operators
    satisfy $O_L(\Theta_2)O_L(\sqrt2A_\sigma)=(5^L+3^L)O_L(\sqrt2A_\sigma)$.
\end{theorem}

\begin{proof}
    \leanok
    Proposition~\ref{prop:asym_compression_trace} applied to the datum of
    Theorem~\ref{thm:asymex_ising_compression} gives $\tr(B^w)$ as the sum of
    the two weighted slot traces, and pulling each weight out of its word
    evaluation contributes the two displayed powers; the operator identity is
    the entrywise reading of the same identity at the letters of a periodic
    configuration.
\end{proof}

\begin{remark}[The hidden bond objects feed the same visible channel]%
    \label{rem:asymex_ising_hidden_channel}
    The two hidden bond objects $1$ and $\psi$ both feed the visible channel
    $\sigma$, through the fusions $1\otimes\sigma=\sigma$ and
    $\psi\otimes\sigma=\sigma$, with different weights; on a ring of length
    $L$ the periodic coefficient of the $\sigma$ strand is the power sum
    $5^L+3^L$ of Equation~(\ref{eq:asymex_ising_trace}). This is the sitewise
    origin of the length-dependent coefficient of the weighted fusion, the
    single fusion operator with two distinct eigenvalues that makes the
    length dependence non-removable.
\end{remark}

\section{A literal action-tensor instance}%
\label{sec:asymex_ising_action}

Unlike the stacked product of Section~\ref{sec:asymex_ising_bond_object},
this section is a genuine action-tensor instance: the full weighted bond
object $\Theta_3=5A_1\oplus3A_\psi\oplus2A_\sigma$ applied to a fusion-tree
product state, in the sense of Corollary~\ref{cor:asym_action_tensors}.

\begin{definition}[The full bond object and the fusion-tree product states]%
    \label{def:asymex_ising_action}
    \lean{IsingTwist.thetaThreeZ, IsingTwist.thetaThree, IsingTwist.productStateZ,
        IsingTwist.productState, IsingTwist.sectorState}
    \leanok
    \uses{def:asymex_ising_tensors, def:mps_tensor}
    The full weighted bond object is $\Theta_3=5A_1\oplus3A_\psi\oplus2A_\sigma$
    on $\C^{10}=P_1\oplus P_\psi\oplus P_\sigma$. For a fusion-tree label
    $h_0$, the associated product state is the bond-dimension-one tensor
    generating $\ket{h_0}^{\otimes L}$ at every length; $\ket{\sigma1\sigma}^{\otimes L}$,
    the fusion tree of the trivial anyon chain, is the product state at the
    label $(\sigma,1,\sigma)$.
\end{definition}

\begin{theorem}[Every fusion-tree product state is normal]%
    \label{thm:asymex_ising_action_normal}
    \lean{IsingTwist.productState_isNormal}
    \leanok

    bond algebra is one-dimensional and one letter is nonzero.
\end{theorem}

\begin{proof}
    \leanok

\begin{theorem}[The action tensor is diagonal]%
    \label{thm:asymex_ising_action_diagonal}

\begin{theorem}[Compression of the weighted Verlinde action]%
    \label{thm:asymex_ising_action_compression}
    % Principal declaration: IsingTwist.sectorCompression.
    \lean{IsingTwist.sectorSlots, IsingTwist.sectorLabel, IsingTwist.sectorWeights,
        IsingTwist.sectorTargets, IsingTwist.sectorGauge, IsingTwist.sectorCompression,
        IsingTwist.sectorAction_isReduction, IsingTwist.sectorAction_z_eq,
        IsingTwist.sectorAction_dim_eq}
    \leanok
    \uses{thm:asymex_ising_action_diagonal}
    The action tensor of Theorem~\ref{thm:asymex_ising_action_diagonal}
    carries the block triangular form of Theorem~\ref{thm:asym_multiblock} for
    four weighted bond-one product states, $\ket{\sigma1\sigma}^{\otimes L}$
    with weight $5$ (through the bond object $1$) and with weight $3$
    (through $\psi$), and $\ket{111}^{\otimes L}$, $\ket{\psi1\psi}^{\otimes L}$
    each with weight $2$ (through $\sigma$), with six zero slots,
    $10=1+1+1+1+6$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    The gauge is the permutation placing the four slot coordinates
    $e_2,e_5,e_8,e_9$ first, followed by the six zero-slot coordinates; in
    these coordinates the action tensor is block diagonal with the four
    weighted product-state letters and six zeros, a finite verification over
    integer entries. Block diagonality gives the triangularity condition and
    the identification of the matched blocks; the dimension count is clause
    (vii) of Theorem~\ref{thm:asym_multiblock}.
\end{proof}

\begin{theorem}[The weighted Verlinde action]%
    \label{thm:asymex_ising_action_trace}
    \lean{IsingTwist.sectorAction_trace_evalWord, IsingTwist.thetaThree_mpo_sectorState}
    \leanok
    \uses{thm:asymex_ising_action_split, prop:asym_compression_trace}
    For every nonempty word $w$,
    \begin{align}
        \tr\bigl((\Theta_3\cdot\psi_{\sigma1\sigma})^w\bigr)
         & =\bigl(5^{|w|}+3^{|w|}\bigr)\tr\bigl((\psi_{\sigma1\sigma})^w\bigr)
        +2^{|w|}\Bigl(\tr\bigl((\psi_{111})^w\bigr)+\tr\bigl((\psi_{\psi1\psi})^w\bigr)\Bigr),
        \notag
    \end{align}
    where $\psi_{h_0}$ denotes the product-state tensor at the label $h_0$ and
    $\Theta_3\cdot\psi_{\sigma1\sigma}$ is the action tensor of
    Theorem~\ref{thm:asymex_ising_action_diagonal}.
    Equivalently, at every positive length $L$,
    \begin{align}
        O_L(\Theta_3)\ket{\sigma1\sigma}^{\otimes L}
         & =\bigl(5^L+3^L\bigr)\ket{\sigma1\sigma}^{\otimes L}
        +2^L\Bigl(\ket{111}^{\otimes L}+\ket{\psi1\psi}^{\otimes L}\Bigr),
        \label{eq:asymex_ising_verlinde}
    \end{align}
    the weighted Verlinde action of the bond object on the sector labels: the
    state $\ket{\sigma1\sigma}^{\otimes L}$ is reached twice, through the
    bond objects $1$ and $\psi$, with the two different weights.
\end{theorem}

\begin{proof}
    \leanok
    Proposition~\ref{prop:asym_compression_trace} applied to the datum of
    Theorem~\ref{thm:asymex_ising_action_compression} gives the sum of the
    four weighted slot traces, and pulling each weight out of its word
    evaluation gives the displayed powers; the periodic-vector identity
    (\ref{eq:asymex_ising_verlinde}) is the same computation read at the
    letters of a periodic configuration.
\end{proof}
