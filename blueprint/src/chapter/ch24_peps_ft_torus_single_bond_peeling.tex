\subsection{Single-bond peeling and the window coefficient identity}
\label{subsec:peps_torus_single_bond_peeling_window_witness}

\begin{definition}[Reference edge of the horizontal staircase]
    \label{def:peps_horizontal_staircase_reference_edge}
    \lean{TNLean.PEPS.horizontalStaircaseReferenceEdge}
    \leanok
    \uses{def:peps_horizontal_staircase_patch}
    In staircase coordinates $s=(a,b)$, the reference edge $e$ is the horizontal
    edge $\{(a+L-1,b+K-1),(a+L,b+K-1)\}$ joining $W_{\mathrm{left}}$ and
    $W_{\mathrm{right}}$ (Definition~\ref{def:peps_horizontal_staircase_patch}).
\end{definition}

\begin{lemma}[The reference edge is the unique crossing of the two end windows]
    \label{thm:peps_horizontal_staircase_reference_edge_crossing}
    \lean{TNLean.PEPS.isCrossingEdge_horizontalStaircaseEndWindows}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_horizontalStaircaseLeftWindow_referenceEdge}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_horizontalStaircaseRightWindow_referenceEdge}
    \lean{TNLean.PEPS.not_isRegionBoundaryEdge_horizontalStaircaseEndPair_referenceEdge}
    \leanok
    \uses{def:peps_horizontal_staircase_reference_edge, def:peps_horizontal_staircase_patch}
    Assume $1<W$, $1<H$, $0<L$, $0<K$, $1\le a$, $a+2L\le W$, and
    $b+2K-1\le H$, and set $S=W_{\mathrm{left}}\cup W_{\mathrm{right}}$. Then
    \begin{align}
        f\in\partial W_{\mathrm{left}}\cap\partial W_{\mathrm{right}}
        \quad\Longleftrightarrow\quad
        f=e,
        \label{eq:peps_reference_edge_unique_crossing}
    \end{align}
    so $e\in\partial W_{\mathrm{left}}\cap\partial W_{\mathrm{right}}$ and
    $e\notin\partial S$.
\end{lemma}
\begin{proof}\leanok
    Direct coordinate inspection identifies $e$ (Definition~\ref{def:peps_horizontal_staircase_reference_edge})
    as the unique horizontal edge joining $W_{\mathrm{left}}$ and
    $W_{\mathrm{right}}$, giving~(\ref{eq:peps_reference_edge_unique_crossing});
    since both endpoints of $e$ lie in $S$, $e\notin\partial S$.
\end{proof}

\begin{lemma}[Boundary edges of the end pair come from the end windows]
    \label{lem:peps_horizontal_staircase_end_pair_boundary_from_windows}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_endPair_of_leftWindow}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_endPair_of_rightWindow}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_window_of_endPair}
    \lean{TNLean.PEPS.ne_referenceEdge_of_isRegionBoundaryEdge_endPair}
    \leanok
    \uses{thm:peps_horizontal_staircase_reference_edge_crossing}
    Under $1<W$, $1<H$, $0<L$, $0<K$, $1\le a$, $a+2L\le W$, and $b+2K-1\le H$:
    if $f\in\partial S$, then
    $f\in(\partial W_{\mathrm{left}}\cup\partial W_{\mathrm{right}})\setminus\{e\}$;
    conversely, any
    $f\in(\partial W_{\mathrm{left}}\cup\partial W_{\mathrm{right}})\setminus\{e\}$
    lies in $\partial S$.
\end{lemma}
\begin{proof}\leanok
    If $f\in\partial S$, its endpoint in $S=W_{\mathrm{left}}\cup W_{\mathrm{right}}$
    lies in one window while the other endpoint lies in $V\setminus S$, so
    $f\in\partial W_{\mathrm{left}}\cup\partial W_{\mathrm{right}}$ and $f\ne e$
    by Lemma~\ref{thm:peps_horizontal_staircase_reference_edge_crossing}.
    Conversely, any $f\in\partial W_{\mathrm{left}}\setminus\{e\}$ (or
    $\partial W_{\mathrm{right}}\setminus\{e\}$) cannot enter the opposite
    window by~(\ref{eq:peps_reference_edge_unique_crossing}), so $f\in\partial S$.
\end{proof}

\begin{lemma}[Boundary of the staircase end pair]
    \label{thm:peps_horizontal_staircase_end_pair_boundary}
    \lean{TNLean.PEPS.eq_referenceEdge_of_isRegionBoundaryEdge_both_endWindows}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_endPair_iff}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_endPair_exactlyOne_window}
    \leanok
    \uses{lem:peps_horizontal_staircase_end_pair_boundary_from_windows}
    Under $1<W$, $1<H$, $0<L$, $0<K$, $1\le a$, $a+2L\le W$, and $b+2K-1\le H$,
    \begin{align}
        f\in\partial S
        \quad\Longleftrightarrow\quad
        \bigl(f\in\partial W_{\mathrm{left}}
        \ \text{or}\
        f\in\partial W_{\mathrm{right}}\bigr)
        \ \text{and}\ f\ne e,
        \label{eq:peps_end_pair_boundary_iff}
    \end{align}
    and $\partial S=(\partial W_{\mathrm{left}}\setminus\{e\})\sqcup(\partial W_{\mathrm{right}}\setminus\{e\})$.
\end{lemma}
\begin{proof}\leanok
    Lemma~\ref{lem:peps_horizontal_staircase_end_pair_boundary_from_windows}
    gives~(\ref{eq:peps_end_pair_boundary_iff}), and~(\ref{eq:peps_reference_edge_unique_crossing})
    gives $\partial W_{\mathrm{left}}\cap\partial W_{\mathrm{right}}=\{e\}$.
\end{proof}

    set $\partial_{\mathrm{left}}S=\partial S\cap\partial W_{\mathrm{left}}$ and
    $\partial_{\mathrm{right}}S=\partial S\cap\partial W_{\mathrm{right}}$.
    Splitting at $e$ and reindexing along~(\ref{eq:peps_end_pair_boundary_iff})
    gives canonical label-preserving equivalences
    \begin{align}
        \operatorname{Conf}(\partial W_{\mathrm{left}})
         & \cong \operatorname{Conf}(e)
        \times\operatorname{Conf}(\partial_{\mathrm{left}}S),
        \notag                          \\

         & \cong \operatorname{Conf}(e)
        \times\operatorname{Conf}(\partial_{\mathrm{right}}S),
        \notag                          \\

         & \cong
        \operatorname{Conf}(\partial_{\mathrm{left}}S)
        \times\operatorname{Conf}(\partial_{\mathrm{right}}S),
        \label{eq:peps_end_window_conf_equiv}
    \end{align}
    isolating $e$ as the sole contracted virtual bond between
    $W_{\mathrm{left}}$ and $W_{\mathrm{right}}$.
\end{definition}

\begin{lemma}[Complement split for an end window]
    \label{lem:peps_horizontal_staircase_end_window_complement_split}
    \lean{TNLean.PEPS.horizontalStaircaseEndPair_sdiff_leftWindow}
    \lean{TNLean.PEPS.horizontalStaircaseEndPair_sdiff_rightWindow}
    \lean{TNLean.PEPS.compl_leftWindow_eq_rightWindow_union_complEndPair}
    \lean{TNLean.PEPS.compl_rightWindow_eq_leftWindow_union_complEndPair}
    \leanok
    \uses{def:peps_horizontal_staircase_patch, lem:peps_horizontal_staircase_end_pair_disjoint}
    If $2L\le W$, then
    \begin{align}
        V\setminus W_{\mathrm{left}}
         & =W_{\mathrm{right}}\cup (V\setminus S),
         &
        V\setminus W_{\mathrm{right}}
         & =W_{\mathrm{left}}\cup (V\setminus S).
        \label{eq:peps_end_window_compl_split}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{lem:peps_horizontal_staircase_end_pair_disjoint},
    $S=W_{\mathrm{left}}\sqcup W_{\mathrm{right}}$, so
    $S\setminus W_{\mathrm{left}}=W_{\mathrm{right}}$ and
    $S\setminus W_{\mathrm{right}}=W_{\mathrm{left}}$, yielding~(\ref{eq:peps_end_window_compl_split}).
\end{proof}

\begin{lemma}[The reference edge bounds the two end windows]
    \label{thm:peps_staircase_end_reference_edge_boundary}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_staircaseWindow_zero_referenceEdge}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_staircaseWindow_last_referenceEdge}
    \leanok
    On a $W\times H$ torus with $1<H$, $0<L$, $0<K$, $1\le a$, $a+2L\le W$, and
    $b+2K-1\le H$, the reference edge $e$ of the $(L,K)$-staircase at $(a,b)$
    satisfies
    \begin{align}
        e\in\partial W_0,\qquad e\in\partial W_{L+K-1}.
        \label{eq:peps_reference_edge_end_staircase_windows}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $W_0=W_{\mathrm{right}}$ and $W_{L+K-1}=W_{\mathrm{left}}$,~(\ref{eq:peps_reference_edge_end_staircase_windows})
    follows from Lemma~\ref{thm:peps_horizontal_staircase_reference_edge_crossing}.
\end{proof}

\begin{lemma}[Open-boundary equality on the staircase end pair]
    \label{thm:peps_staircase_pair_insert_eq_open}
    \lean{TNLean.PEPS.staircasePair_insert_eq_open}
    \leanok
    \uses{thm:peps_staircase_pair_cancel, thm:peps_corner_extension_linear_transitive}
    Let $S=W_0\cup W_{L+K-1}$. Suppose $B$ satisfies the one-orientation
    $L\times K$ window-injectivity hypotheses, union closure for injective
    regions, and positive bond dimensions, with $0<L$, $0<K$, $2L+1\le W$, and
    $2K+1\le H$. If inserts $C_j$ on $W_j$ all generate the same deformed state,
    then
    \begin{align}
        \Ext_{W_0\subseteq S}(C_0)
        =
        \Ext_{W_{L+K-1}\subseteq S}(C_{L+K-1}).
        \label{eq:peps_staircase_pair_insert_eq_open}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Composing consecutive-window equalities across the staircase patch,
    extending to $S\cup Q$ via Lemma~\ref{thm:peps_corner_extension_linear_transitive},
    and cancelling the injective $L\times K$ completed corner $Q$ via
    Lemma~\ref{thm:peps_staircase_pair_cancel} yields~(\ref{eq:peps_staircase_pair_insert_eq_open}).
\end{proof}

\begin{lemma}[Lemma 5 comparison of the two end blocks]
    \label{thm:peps_existsUnique_virtualOperation_of_endPair}
    \lean{TNLean.PEPS.gauge_eq1}
    \lean{TNLean.PEPS.gauge_eq2}
    \lean{TNLean.PEPS.existsUnique_virtualOperation_of_endPair}
    \leanok
    Let $\{A_{1,\mu}\}_{\mu\in K}$ and $\{A_{2,\mu}\}_{\mu\in K}$ be linearly
    independent families indexed by the bond basis $K$. If end operations
    $O_1,O_3$ produce families $C_1,C_2$ (without assuming injectivity of
    $C_1,C_2$) satisfying
    \begin{align}
        \sum_{\mu\in K} C_{1,\mu}\otimes A_{2,\mu}
        =
        \sum_{\nu\in K} A_{1,\nu}\otimes C_{2,\nu},
        \label{eq:peps_open_contraction_id}
    \end{align}
    then there is a unique matrix $X\in\MN{|K|}$ such that
    \cite[Lemma~5]{Molnar2018NormalPEPS}
    \begin{align}
        C_{1,\nu}
         & =\sum_{\mu\in K} A_{1,\mu}X_{\mu,\nu},
        \label{eq:peps_c1_expansion}              \\
        C_{2,\mu}
         & =\sum_{\nu\in K} X_{\mu,\nu}A_{2,\nu}.
        \label{eq:peps_c2_expansion}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Applying a dual basis of $\{A_{2,\mu}\}$ to~(\ref{eq:peps_open_contraction_id})
    yields~(\ref{eq:peps_c1_expansion}) for some $X$; substituting~(\ref{eq:peps_c1_expansion})
    into~(\ref{eq:peps_open_contraction_id}) and using linear independence of
    $\{A_{1,\mu}\}$ gives~(\ref{eq:peps_c2_expansion}) and uniqueness of $X$.
\end{proof}

\begin{definition}[End operations realizing a virtual matrix]
    \label{def:peps_end_operations_realizing_virtual_matrix}
    \lean{TNLean.PEPS.IsO1VirtualOperation}
    \lean{TNLean.PEPS.IsO3TransposeVirtualOperation}
    \leanok
    For end families $\{A_{1,\mu}\}_{\mu\in K}$ and $\{A_{3,\mu}\}_{\mu\in K}$,
    $O_1$ realizes $X\in\MN{|K|}$, and $O_3$ realizes $X$ in the transposed
    orientation $O_3^{\mathsf T}$ of \cite[Lemma~5]{Molnar2018NormalPEPS}, when
    \begin{align}
        O_1(A_{1,\nu})
        =\sum_{\mu\in K}X_{\mu\nu}A_{1,\mu},
        \qquad
        O_3(A_{3,\mu})
        =\sum_{\nu\in K}X_{\mu\nu}A_{3,\nu}.
        \label{eq:peps_end_operations_realize}
    \end{align}
\end{definition}

    \lean{TNLean.PEPS.horizontalStaircaseLeftWindow_threeBlockBlueCoeff_factorization}
    \lean{TNLean.PEPS.horizontalStaircaseRightWindow_threeBlockBlueCoeff_factorization}
    \lean{TNLean.PEPS.horizontalStaircaseLeftWindow_threeBlockBlueCoeff_eq_zero_of_external_ne}
    \lean{TNLean.PEPS.horizontalStaircaseRightWindow_threeBlockBlueCoeff_eq_zero_of_external_ne}
    \lean{TNLean.PEPS.horizontalStaircaseLeftWindow_normalized_threeBlockBlueCoeff_factorization}

    \leanok

    In the coordinates~(\ref{eq:peps_end_window_conf_equiv}), write
    $\mu_{\mathrm L}=(k,\eta_{\mathrm L})$,
    $\mu_{\mathrm R}=(k,\eta_{\mathrm R})$, and
    $\nu=(\eta'_{\mathrm L},\eta'_{\mathrm R})$ with $k\in\operatorname{Conf}(e)$.
    In extending a left-window insert to $S$, the normalized coupling to
    $W_{\mathrm{right}}$ equals
    \begin{align}
        \delta_{\eta_{\mathrm L},\eta'_{\mathrm L}}\Lambda_{\mathrm L}^{-1}
        A_{\mathrm R}(k,\eta'_{\mathrm R}),
        \label{eq:peps_normalized_blue_coupling}
    \end{align}
    where $A_{\mathrm R}$ is the blocked tensor on $W_{\mathrm{right}}$ and
    $\Lambda_{\mathrm L}$ is the product of the exterior interior-bond factor
    and external bond dimensions of $W_{\mathrm{left}}$ (and symmetrically for
    $\mathrm L\leftrightarrow\mathrm R$). If the tensor is translation invariant
    with positive bond dimensions, then $\Lambda_{\mathrm L}=\Lambda_{\mathrm R}$.
\end{lemma}
\begin{proof}\leanok
    By Definition~\ref{def:peps_horizontal_staircase_end_window_boundary_config}
    and Lemma~\ref{lem:peps_horizontal_staircase_end_window_complement_split},
    the external crossing edges of $W_{\mathrm{left}}$ are
    $\partial W_{\mathrm{left}}\setminus\{e\}$, so compatibility forces
    $\eta_{\mathrm L}=\eta'_{\mathrm L}$ and reduces the coupling to~(\ref{eq:peps_normalized_blue_coupling}).
    Since $W_{\mathrm{left}}$ and $W_{\mathrm{right}}$ are translates,
    translation invariance equates their full boundary products and cancelling
    $D(e)>0$ gives $\Lambda_{\mathrm L}=\Lambda_{\mathrm R}$.
\end{proof}

\begin{lemma}[The end-pair extensions contract over one bond]
    \label{thm:peps_horizontal_staircase_end_pair_single_bond_contraction}
    \lean{TNLean.PEPS.horizontalStaircaseLeftWindow_extendInsert_eq_singleBondContraction}
    \lean{TNLean.PEPS.horizontalStaircaseRightWindow_extendInsert_eq_singleBondContraction}
    \lean{TNLean.PEPS.singleBondContraction_eq_of_horizontalStaircaseEndPair_extendInsert_eq}
    \leanok
    \uses{thm:peps_horizontal_staircase_normalized_blue_coupling}
    For inserts $C_{\mathrm L}$ on $W_{\mathrm{left}}$ and $C_{\mathrm R}$ on
    $W_{\mathrm{right}}$, their extensions to $S$ in coordinates~(\ref{eq:peps_end_window_conf_equiv})
    are
    \begin{align}
        \Ext(C_{\mathrm L})
         & =\Lambda_{\mathrm L}^{-1}
        \sum_k C_{\mathrm L}(k,\eta_{\mathrm L})
        A_{\mathrm R}(k,\eta_{\mathrm R}),
        \qquad
        \Ext(C_{\mathrm R})
        =\Lambda_{\mathrm R}^{-1}
        \sum_k A_{\mathrm L}(k,\eta_{\mathrm L})
        C_{\mathrm R}(k,\eta_{\mathrm R}).
        \label{eq:peps_end_pair_single_bond_ext}
    \end{align}
    For a translation-invariant tensor with positive bond dimensions,
    $\Ext(C_{\mathrm L})=\Ext(C_{\mathrm R})$ implies
    \begin{align}
        \sum_k C_{\mathrm L}(k,\eta_{\mathrm L})
        A_{\mathrm R}(k,\eta_{\mathrm R})
        =
        \sum_k A_{\mathrm L}(k,\eta_{\mathrm L})
        C_{\mathrm R}(k,\eta_{\mathrm R}).
        \label{eq:peps_end_pair_single_bond_id}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Reindexing the extension sums via~(\ref{eq:peps_end_window_conf_equiv}) and
    substituting~(\ref{eq:peps_normalized_blue_coupling}) gives~(\ref{eq:peps_end_pair_single_bond_ext});
    cancelling $\Lambda_{\mathrm L}=\Lambda_{\mathrm R}>0$
    (Lemma~\ref{thm:peps_horizontal_staircase_normalized_blue_coupling})
    yields~(\ref{eq:peps_end_pair_single_bond_id}).
\end{proof}

\begin{lemma}[Lemma 5 operation on the staircase reference edge]
    \label{thm:peps_existsUnique_virtualOperation_of_horizontal_staircase_end_pair}
    \lean{TNLean.PEPS.existsUnique_virtualOperation_of_horizontalStaircaseEndPair}
    \lean{TNLean.PEPS.existsUnique_virtualOperation_of_horizontalStaircaseEndPhysicalOperations}

    and $A_{\mathrm L},A_{\mathrm R}$ are injective. If inserts $C_{\mathrm L},C_{\mathrm R}$

        C_{\mathrm L}(k,\eta_{\mathrm L})

    $A_{\mathrm L},A_{\mathrm R}$ makes the $k$-indexed families linearly
    independent, so Lemma~\ref{thm:peps_existsUnique_virtualOperation_of_endPair}
    yields~(\ref{eq:peps_staircase_ref_edge_op}) and uniqueness of $X$.
\end{proof}

\begin{lemma}[Single-bond peeling of the staircase end equality]
    \label{thm:peps_regionInsertedCoeff_endWindows_eq_of_staircase}
    \lean{TNLean.PEPS.regionInsertedCoeff_endWindows_eq_of_staircase}
    \leanok
    \uses{def:peps_bond_inserted_region_insert, thm:peps_regionInsertedCoeff_eq_extendInsert_bondInserted, thm:peps_staircase_end_reference_edge_boundary, thm:peps_staircase_pair_insert_eq_open}
    Suppose $B$ satisfies the one-orientation $L\times K$ window-injectivity
    hypotheses, union closure for injective regions, and positive bond
    dimensions on a $W\times H$ torus with $2\le L$, $2\le K$, $1\le a$,
    $a+2L\le W$, $b+2K-1\le H$, $2L+1\le W$, and $2K+1\le H$. If inserts $C_j$
    on $W_j$ all generate the same deformed state, with
    $C_0=I_{B,W_0,e,M}$ and $C_{L+K-1}=I_{B,W_{L+K-1},e,M'}$, then for every
    global physical configuration $\sigma$,
    \begin{align}
        C_B(W_0,e,M;\sigma|_{W_0},\sigma|_{V\setminus W_0})
         & =
        C_B\bigl(W_{L+K-1},e,M';
        \sigma|_{W_{L+K-1}},\sigma|_{V\setminus W_{L+K-1}}\bigr).
        \label{eq:peps_single_bond_peeling_coeff}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_regionInsertedCoeff_eq_extendInsert_bondInserted},
    \begin{align}
        C_B(W_0,e,M;\sigma|_{W_0},\sigma|_{V\setminus W_0})
         & =\Psi_{B,S,\Ext_{W_0\subseteq S}(C_0)}(\sigma),
        \notag                                                         \\
        C_B(W_{L+K-1},e,M';\sigma|_{W_{L+K-1}},\sigma|_{V\setminus W_{L+K-1}})
         & =\Psi_{B,S,\Ext_{W_{L+K-1}\subseteq S}(C_{L+K-1})}(\sigma),
        \notag
    \end{align}
    and~(\ref{eq:peps_staircase_pair_insert_eq_open}) of
    Lemma~\ref{thm:peps_staircase_pair_insert_eq_open} equates the two right-hand sides.
\end{proof}

\begin{definition}[Window-region coefficient-identity datum]
    \label{def:peps_window_edge_coeff_identity_witness}
    \lean{TNLean.PEPS.windowEdgeCoeffIdentityWitness}
    \leanok
    \uses{def:peps_regionInsertedCoeff, thm:peps_staircase_end_reference_edge_boundary}
    Let $W=W_{\mathrm{left}}$ and $e$ be the left end window and reference edge
    of an $(L,K)$-staircase at $(a,b)$ with $0<L$, $0<K$, $1\le a$,
    $a+2L\le W_{\mathrm{torus}}$, and $b+2K-1\le H_{\mathrm{torus}}$ (so
    $e\in\partial W$ by Lemma~\ref{thm:peps_staircase_end_reference_edge_boundary}).
    Given injectivity of $B$ on $W$ and $V\setminus W$, positive bond
    dimensions of $B$, $\iota_e:\MN{D_A(e)}\simeq\MN{D_B(e)}$, and
    $Z\in\GL(D_B(e))$ satisfying
    \begin{align}
        C_A(W,e,M;\sigma,\tau)
        =
        C_B(W,e,Z\,\iota_e(M)\,Z^{-1};\sigma,\tau)
        \qquad(\forall M,\sigma,\tau),
        \label{eq:peps_window_edge_coeff_id}
    \end{align}
    these data form the coefficient-identity datum at $(W,e)$ with both gauge
    entries equal to $Z$.
\end{definition}

\begin{definition}[Window-region coefficient-identity datum from window injectivity]
    \label{def:peps_window_edge_coeff_identity_witness_of_hypotheses}
    \lean{TNLean.PEPS.windowEdgeCoeffIdentityWitness_of_hypotheses}
    \leanok
    \uses{def:peps_window_edge_coeff_identity_witness}
    If every cyclic $L\times K$ window of $B$ is injective, unions of injective
    regions are injective, all bond dimensions of $B$ are positive, and
    $0<L$, $0<K$, $2L+1\le W_{\mathrm{torus}}$, $2K+1\le H_{\mathrm{torus}}$,
    $1\le a$, $a+2L\le W_{\mathrm{torus}}$, and $b+2K-1\le H_{\mathrm{torus}}$,
    then both $W=W_{\mathrm{left}}$ and $V\setminus W$ are injective, so any
    $\iota_e:\MN{D_A(e)}\simeq\MN{D_B(e)}$ and $Z\in\GL(D_B(e))$
    satisfying~(\ref{eq:peps_window_edge_coeff_id}) yield the datum of
    Definition~\ref{def:peps_window_edge_coeff_identity_witness}.
\end{definition}
