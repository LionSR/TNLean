\section{Nested square patches}
\label{sec:peps_nested_patch_geometry}

Fix a finite induced lattice domain $\Lambda\subset\mathbb Z^2$ and a centre
$c\in\mathbb R^2$. The square patches in the proof of
\cite[Proposition~4.1]{OpenAI2026PolynomialPEPS} are samples of closed ambient
squares; intersecting with $\Lambda$ clips at the physical boundary without
adding edges to the induced graph.
% This section covers the geometric part of 03-patches.tex only. It does not
% mark Proposition 4.1, its energy estimate, or its rank conclusion as proved.

\begin{definition}[Closed square sample]
    \label{def:peps_closed_square_sample}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.closedSquareSample,
        TNLean.PEPS.AreaLaw.Geometry.mem_closedSquareSample,
        TNLean.PEPS.AreaLaw.Geometry.mem_closedSquareSample_iff_int,
        TNLean.PEPS.AreaLaw.Geometry.closedSquareSample_mono
    }
    \leanok
    \uses{def:area_law_domain}
    For $r\in\mathbb R$, define the closed square sample
    \begin{align}
        K(c,r) & =\Lambda\cap
        \bigl([c_1-r,c_1+r]\times[c_2-r,c_2+r]\bigr)
        = \{x\in\Lambda:\lceil c_a-r\rceil\le x_a\le\lfloor c_a+r\rfloor\ (a=1,2)\},
        \label{eq:peps_closed_square_sample}
    \end{align}
    which is empty for $r<0$ and monotone in $r$.
\end{definition}

\begin{definition}[Distance from the square centre]
    \label{def:peps_square_radius}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.squareRadius,
        TNLean.PEPS.AreaLaw.Geometry.mem_closedSquareSample_iff_radius
    }
    \leanok
    \uses{def:peps_closed_square_sample}
    For $x\in\mathbb Z^2$, set $\rho(x):=\max\{|x_1-c_1|,|x_2-c_2|\}$, so that $K(c,r)=\{x\in\Lambda:\rho(x)\le r\}$.
\end{definition}

\begin{theorem}[Nearest neighbours and closed-square crossing]
    \label{thm:peps_square_crossing_interval}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.abs_coordinate_sub_le_one_of_adj,
        TNLean.PEPS.AreaLaw.Geometry.mem_closedSquareSample_of_adj,
        TNLean.PEPS.AreaLaw.Geometry.abs_squareRadius_sub_le_one_of_adj,
        TNLean.PEPS.AreaLaw.Geometry.mem_edgeBoundary_pair_iff,
        TNLean.PEPS.AreaLaw.Geometry.mem_edgeBoundary_closedSquareSample_iff
    }
    \leanok
    \uses{def:peps_square_radius,def:area_law_hamiltonian}
    Adjacent sites $x\sim y$ satisfy $|x_a-y_a|\le1$ ($a=1,2$), $|\rho(x)-\rho(y)|\le1$, and $x\in K(c,r)\implies y\in K(c,r+1)$. For any $A\subseteq\Lambda$, $\{x,y\}\in\partial_\Lambda A$ if and only if $x\sim y$ and $|\{x,y\}\cap A|=1$; in particular, for $x\sim y$,
    \begin{align}
        \{x,y\}\in\partial_\Lambda K(c,r)
         & \quad\Longleftrightarrow\quad
        \min\{\rho(x),\rho(y)\}\le r<\max\{\rho(x),\rho(y)\}.
        \label{eq:peps_square_crossing_interval}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Since $x\sim y$ implies $|x_1-y_1|+|x_2-y_2|=1$, the triangle inequality gives $|y_a-c_a|\le|x_a-c_a|+|x_a-y_a|\le\rho(x)+1$, so $|\rho(x)-\rho(y)|\le1$ by symmetry. Moreover, $\{x,y\}\in\partial_\Lambda K(c,r)$ holds if and only if $(\rho(x)\le r<\rho(y))\vee(\rho(y)\le r<\rho(x))$, which is (\ref{eq:peps_square_crossing_interval}).
\end{proof}

\begin{theorem}[Number of sites in a sampled square]
    \label{thm:peps_closed_square_card}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.card_closedSquareSample_le,
        TNLean.PEPS.AreaLaw.Geometry.card_closedSquareSample_le_eightyOne
    }
    \leanok
    \uses{def:peps_closed_square_sample}
    For $r\ge0$, $|K(c,r)|\le(2r+1)^2$; if $0\le r\le4$, then $|K(c,r)|\le81$.
\end{theorem}
\begin{proof}\leanok
    Each interval $[c_a-r,c_a+r]$ contains at most $2r+1$ integers, so $|K(c,r)|\le(2r+1)^2\le 9^2=81$ when $r\le4$.
\end{proof}

\begin{theorem}[Crossing edges of a sampled square]
    \label{thm:peps_closed_square_boundary}
    \lean{TNLean.PEPS.AreaLaw.Geometry.card_edgeBoundary_closedSquareSample_le}
    \leanok
    \uses{def:peps_closed_square_sample,def:area_law_hamiltonian}
    For every $r\ge0$, $|\partial_\Lambda K(c,r)|\le8r+4$.
\end{theorem}
\begin{proof}\leanok
    Each of the at most $2r+1$ integer rows (resp.\ columns) meeting $[c_2-r,c_2+r]$ (resp.\ $[c_1-r,c_1+r]$) contributes at most $2$ horizontal (resp.\ vertical) crossing edges, giving $|\partial_\Lambda K(c,r)|\le 2(2r+1)+2(2r+1)=8r+4$.
\end{proof}

\begin{definition}[Nested square patches]
    \label{def:peps_nested_patches}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.nestedPatchCount,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatchRadius,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch
    }
    \leanok
    \uses{def:peps_closed_square_sample}
    For $u\ge0$, set $m:=\lfloor u/2\rfloor+1$ and, for $0\le j<m$,
    \begin{align}
        r_j & =u+2j,\qquad K_j=K(c,r_j).
        \label{eq:peps_nested_patches}
    \end{align}
\end{definition}

\begin{theorem}[Number, radii, and boundary size of the patches]
    \label{thm:peps_nested_patch_sizes}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.nestedPatchCount_pos,
        TNLean.PEPS.AreaLaw.Geometry.half_lt_nestedPatchCount,
        TNLean.PEPS.AreaLaw.Geometry.le_nestedPatchRadius,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatchRadius_le_two_mul,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatchRadius_add_two_le,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch_mono,
        TNLean.PEPS.AreaLaw.Geometry.card_edgeBoundary_nestedPatch_le
    }
    \leanok
    \uses{def:peps_nested_patches,def:area_law_hamiltonian}
    For the patches of Definition~\ref{def:peps_nested_patches}, $m\ge1$, $m>u/2$, and for $0\le i,j<m$:
    \begin{align}
        u \le r_j              & \le 2u,               &
        |\partial_\Lambda K_j| & \le 16u+4,            &
        i<j \implies r_i+2     & \le r_j,              &
        i\le j \implies K_i    & \subseteq K_j. \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_closed_square_boundary}
    Since $0\le j\le\lfloor u/2\rfloor\le u/2<m$, we have $u\le r_j\le 2u$, so Theorem~\ref{thm:peps_closed_square_boundary} gives $|\partial_\Lambda K_j|\le 8r_j+4\le 16u+4$; $r_i+2\le r_j$ for $i<j$ and $K_i\subseteq K_j$ for $i\le j$ follow from (\ref{eq:peps_nested_patches}).
\end{proof}

\begin{theorem}[Supports of crossing edges under radius separation]
    \label{thm:peps_square_crossing_support}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.edgeBoundary_support_disjoint_of_add_one_le,
        TNLean.PEPS.AreaLaw.Geometry.edgeBoundary_support_subset_of_add_one_le
    }
    \leanok
    \uses{def:peps_closed_square_sample,def:area_law_hamiltonian}
    If $e=\{x,y\}\in\partial_\Lambda K(c,r)$, then $e\cap K(c,s)=\varnothing$ whenever $s+1\le r$, and $e\subseteq K(c,t)$ whenever $r+1\le t$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_square_crossing_interval}
    By (\ref{eq:peps_square_crossing_interval}) and $|\rho(x)-\rho(y)|\le1$ (Theorem~\ref{thm:peps_square_crossing_interval}), $\max\{\rho(x),\rho(y)\}>r\ge s+1$ implies $\min\{\rho(x),\rho(y)\}>s$, while $\min\{\rho(x),\rho(y)\}\le r$ implies $\max\{\rho(x),\rho(y)\}\le r+1\le t$.
\end{proof}

\begin{theorem}[A single crossing among nested patches]
    \label{thm:peps_nested_patch_crossing}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch_crossing_disjoint_earlier,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch_crossing_subset_later,
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch_crossing_unique,
        TNLean.PEPS.AreaLaw.Geometry.disjoint_edgeBoundary_nestedPatch
    }
    \leanok
    \uses{def:peps_nested_patches,def:area_law_hamiltonian}
    If $e=\{x,y\}\in\partial_\Lambda K_j$, then $e\cap K_i=\varnothing$ for all $i<j$ and $e\subseteq K_k$ for all $k>j$; consequently, $\partial_\Lambda K_i\cap\partial_\Lambda K_k=\varnothing$ for $i\ne k$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_nested_patch_sizes,thm:peps_square_crossing_support}
    Since $r_i+2\le r_j$ for $i<j$ and $r_j+2\le r_k$ for $j<k$ by Theorem~\ref{thm:peps_nested_patch_sizes}, Theorem~\ref{thm:peps_square_crossing_support} gives $e\cap K_i=\varnothing$ and $e\subseteq K_k$ (so $e\notin\partial_\Lambda K_k$).
\end{proof}

$c=(5/4,3/2)$ and $u=4$ has clipped patches
$K_0=\{0,\ldots,5\}\times\{0,1,2\}$,
$K_1=\{0,\ldots,7\}\times\{0,1,2\}$, and
$K_2=\{0,\ldots,9\}\times\{0,1,2\}$;
the edge $e=\{(7,1),(8,1)\}\in\partial_\Lambda K_1$ satisfies $e\cap K_0=\varnothing$

\begin{theorem}[Uniform quadratic boundary sum]
    \label{thm:peps_nested_patch_quadratic_sum}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.nestedPatch_quadratic_budget,
        TNLean.PEPS.AreaLaw.Geometry.sum_card_edgeBoundary_nestedPatch_mul_sq_le
    }
    \leanok
    \uses{def:peps_nested_patches,def:area_law_hamiltonian}
    For $u\ge4$, $\kappa\in\mathbb R$, and $t_j=\kappa/m$ ($0\le j<m$),
    \begin{align}
        (16u+4)\sum_{j=0}^{m-1}t_j^2                & \le36\kappa^2,
        \label{eq:peps_nested_patch_uniform_budget}                  \\
        \sum_{j=0}^{m-1}|\partial_\Lambda K_j|t_j^2 & \le36\kappa^2.
        \label{eq:peps_nested_patch_quadratic_sum}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_nested_patch_sizes}
    By Theorem~\ref{thm:peps_nested_patch_sizes}, $m>u/2$ and $u\ge4$ give $16u+4\le 18u\le 36m$, so $(16u+4)\sum_{j=0}^{m-1}t_j^2=(16u+4)\kappa^2/m\le 36\kappa^2$, and $|\partial_\Lambda K_j|\le 16u+4$ yields (\ref{eq:peps_nested_patch_quadratic_sum}) from (\ref{eq:peps_nested_patch_uniform_budget}).
\end{proof}

\subsection{Patches for interactions of range \texorpdfstring{$R$}{R}}
\label{sec:peps_range_nested_patches}

The following statements extend the patch geometry of \cite{OpenAI2026PolynomialPEPS} from $R=1$ (Definition~\ref{def:peps_nested_patches}) to interactions of graph diameter at most $R\in\mathbb N$.

\begin{definition}[Nested square patches at range $R$]
    \label{def:peps_range_nested_patches}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.rangePatchSpacing,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchCount,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchRadius,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatch
    }
    \leanok
    \uses{def:peps_closed_square_sample}
    For $R\in\mathbb N$, set $s_R:=2\lceil R/2\rceil$. For $R\ge1$, $u\ge0$, and $m_R:=\lfloor u/s_R\rfloor+1$, define for $0\le j<m_R$:
    \begin{align}
        r_j & =u+s_Rj,\qquad K_j=K(c,r_j).
        \label{eq:peps_range_nested_patches}
    \end{align}
\end{definition}

\begin{theorem}[Spacing, number and radii of the range-$R$ patches]
    \label{thm:peps_range_nested_patch_sizes}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.le_rangePatchSpacing,
        TNLean.PEPS.AreaLaw.Geometry.rangePatchSpacing_le,
        TNLean.PEPS.AreaLaw.Geometry.two_le_rangePatchSpacing,
        TNLean.PEPS.AreaLaw.Geometry.rangePatchSpacing_one,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchCount_one,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchRadius_one,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatch_one,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchCount_pos,
        TNLean.PEPS.AreaLaw.Geometry.div_lt_rangeNestedPatchCount,
        TNLean.PEPS.AreaLaw.Geometry.le_rangeNestedPatchRadius,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchRadius_le_two_mul,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchRadius_add_spacing_le,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatchRadius_mono,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatch_mono
    }
    \leanok
    \uses{def:peps_range_nested_patches,def:peps_nested_patches}
    In Definition~\ref{def:peps_range_nested_patches}, $R\le s_R\le R+1$, $s_R\ge2$ for $R\ge1$, and $R=1$ recovers Definition~\ref{def:peps_nested_patches}. Moreover, $m_R\ge1$, $m_R>u/s_R$, and for $u\ge0$ and $0\le i,j<m_R$:
    \begin{align}
        u\le r_j             & \le 2u,               &
        i<j \implies r_i+s_R & \le r_j,              &
        i\le j \implies K_i  & \subseteq K_j. \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    From $s_R=2\lceil R/2\rceil$ and $m_R=\lfloor u/s_R\rfloor+1$, one has $u/s_R<m_R$ and $0\le s_Rj\le s_R\lfloor u/s_R\rfloor\le u$ for $0\le j<m_R$.
\end{proof}

\begin{theorem}[A range-$R$ support is split by at most one patch]
    \label{thm:peps_range_nested_patch_split}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.abs_squareRadius_sub_le_length,
        TNLean.PEPS.AreaLaw.Geometry.abs_squareRadius_sub_le_of_isAdmissibleSupport,
        TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatch_split_unique
    }
    \leanok
    \uses{def:peps_range_nested_patches,def:peps_square_radius,
        def:area_law_hamiltonian}
    Endpoints $x,y$ of a graph walk of length $\ell$ satisfy $|\rho(x)-\rho(y)|\le\ell$, so $|\rho(x)-\rho(y)|\le R$ on any support $X$ of graph diameter at most $R$. For any $X\subseteq\Lambda$ with $\sup_{x,y\in X}|\rho(x)-\rho(y)|\le R$, at most one patch $K_j$ of Definition~\ref{def:peps_range_nested_patches} satisfies both $X\cap K_j\ne\varnothing$ and $X\setminus K_j\ne\varnothing$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_square_crossing_interval,thm:peps_range_nested_patch_sizes}
    By Theorem~\ref{thm:peps_square_crossing_interval}, $|\rho(x)-\rho(y)|\le1$ along each edge, hence $|\rho(x)-\rho(y)|\le\ell$ along a walk of length $\ell$. If $i<j$ and $z\in X\cap K_i$, $y\in X\setminus K_j$, then Theorem~\ref{thm:peps_range_nested_patch_sizes} gives $\rho(y)>r_j\ge r_i+s_R\ge\rho(z)+R$, contradicting $|\rho(y)-\rho(z)|\le R$.
\end{proof}

\begin{definition}[Inner layer of a square]
    \label{def:peps_square_layer}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.squareLayer,
        TNLean.PEPS.AreaLaw.Geometry.mem_squareLayer
    }
    \leanok
    \uses{def:peps_square_radius}
    For $r\in\mathbb R$ and $R\in\mathbb N$, define the inner layer $\mathcal N(c,r,R):=\{x\in\Lambda:r-R<\rho(x)\le r\}$.
\end{definition}

\begin{theorem}[Size of the inner layer]
    \label{thm:peps_square_layer}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.mem_squareLayer_of_split,
        TNLean.PEPS.AreaLaw.Geometry.card_integerInterval_le,
        TNLean.PEPS.AreaLaw.Geometry.card_squareLayer_le,
        TNLean.PEPS.AreaLaw.Geometry.card_squareLayer_rangeNestedPatch_le
    }
    \leanok
    \uses{def:peps_closed_square_sample,def:peps_square_layer,
        def:peps_range_nested_patches}
    If $\sup_{x,y\in X}|\rho(x)-\rho(y)|\le R$ and $X\setminus K(c,r)\ne\varnothing$, then $X\cap K(c,r)\subseteq\mathcal N(c,r,R)$. Moreover, for $r\ge0$ and $u\ge0$ ($0\le j<m_R$),
    \begin{align}
        |\mathcal N(c,r,R)|   & \le R(8r+4),         &
        |\mathcal N(c,r_j,R)| & \le R(16u+4). \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_range_nested_patch_sizes}
    If $y\in X\setminus K(c,r)$ and $x\in X\cap K(c,r)$, then $r\ge\rho(x)\ge\rho(y)-R>r-R$, so $x\in\mathcal N(c,r,R)$. Covering $\mathcal N(c,r,R)$ by four strips of width $R$ (containing $R$ integers) and length $2r$ (containing at most $2r+1$ integers) gives $|\mathcal N(c,r,R)|\le 4R(2r+1)=R(8r+4)$, and $r_j\le 2u$ (Theorem~\ref{thm:peps_range_nested_patch_sizes}) gives $R(8r_j+4)\le R(16u+4)$.
\end{proof}

\begin{theorem}[Quadratic coefficient at range $R$]
    \label{thm:peps_range_nested_patch_quadratic_sum}
    \lean{TNLean.PEPS.AreaLaw.Geometry.rangeNestedPatch_quadratic_budget}
    \leanok
    \uses{def:peps_range_nested_patches}
    For $R\ge1$, $u\ge0$, $\kappa\in\mathbb R$, and $t_j=\kappa/m_R$ ($0\le j<m_R$),
    \begin{align}
        (16u+4)\sum_{j=0}^{m_R-1}t_j^2 & \le18s_R\kappa^2.
        \label{eq:peps_range_nested_patch_budget}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_range_nested_patch_sizes}
    By Theorem~\ref{thm:peps_range_nested_patch_sizes}, $u<s_Rm_R$ and $s_Rm_R\ge2$ give $16u+4\le 18s_Rm_R$, while $\sum_{j=0}^{m_R-1}t_j^2=\kappa^2/m_R$.
\end{proof}

\begin{lemma}[Entropy of a sampled square at a given area-law constant]
    \label{lem:peps_closed_square_entropy_explicit}
    \lean{TNLean.PEPS.Approximation.SquareEntropyAreaLawAt.closedSquareSample_entropy_le}
    \leanok
    \uses{def:square_entropy_area_law_at,def:area_law_ground_entropy,
        def:square_area_law_state,def:peps_closed_square_sample}
    Under the square-grid entropy area law with constant $C_A$ (Definition~\ref{def:square_entropy_area_law_at}), put $C_S:=\max\{1,16C_A\}$. For every $L$, family state $\Omega$, centre $c$, and $0\le r\le2u$,
    \begin{align}
        S_{U\Omega}(K(c,r)) & \le C_S(1+u). \notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:finite_product_entropy_relabel,thm:square_edge_boundary_card,
        thm:peps_closed_square_boundary}
    By Theorems~\ref{thm:finite_product_entropy_relabel}, \ref{thm:square_edge_boundary_card}, and~\ref{thm:peps_closed_square_boundary}, $S_{U\Omega}(K(c,r))\le C_A|\partial_\Lambda K(c,r)|\le C_A(8r+4)\le C_A(16u+4)\le C_S(1+u)$.
\end{proof}
