\begin{definition}[Incident form of the region resonance]
    \label{def:peps_region_resonate_reconcile}
    \lean{TNLean.PEPS.RegionResonateReconcile}
    \leanok
    \uses{thm:peps_region_resonate_identity_readoffs}
    For a boundary edge $f\in\partial R$, the incident-form condition requires that for every $X\in\MN{D_A(f)}$, the virtual pullback $W_X$ of the in-region endpoint operator satisfies
    \begin{align}
        \exists Y\in\MN{D_B(f)}\colon\quad W_X=\mc{I}_f(Y).
        \label{eq:peps_region_resonate_reconcile}
    \end{align}
\end{definition}

\begin{definition}[Region insertion transfer]
    \label{def:peps_region_insertion_transfer}
    \lean{TNLean.PEPS.RegionInsertionTransfer}
    \leanok
    \uses{def:peps_regionInsertedCoeff}
    A region insertion transfer along $f\in\partial R$ consists of maps $T_f:\MN{D_A(f)}\to\MN{D_B(f)}$ and $S_f:\MN{D_B(f)}\to\MN{D_A(f)}$ such that for all $X\in\MN{D_A(f)}$, $Y\in\MN{D_B(f)}$, and physical configurations $\sigma,\tau$,
    \begin{align}
        C_A(R,f,X;\sigma,\tau)
                 & =C_B(R,f,T_f(X);\sigma,\tau), &
        C_B(R,f,Y;\sigma,\tau)
                 & =C_A(R,f,S_f(Y);\sigma,\tau),
        \label{eq:peps_region_insertion_transfer}  \\
        T_f(XY)  & =T_f(X)T_f(Y),                &
        T_f(\Id) & =\Id.
        \notag
    \end{align}
\end{definition}

\begin{definition}[Region physical-to-virtual realization]
    \label{def:peps_region_transfer_realizes}
    \lean{TNLean.PEPS.RegionTransferRealizes}
    \leanok
    \uses{thm:peps_region_insertion_virtual_pullback_realizes}
    Let $v\in R$ be the in-region endpoint of $f\in\partial R$. A map $T_f:\MN{D_A(f)}\to\MN{D_B(f)}$ is realized on $R$ if for every $X\in\MN{D_A(f)}$ and every local virtual coefficient $c$ for $B$ at $v$,
    \begin{align}
        O_{A,R,f}(X^T) \mc{T}_{B,v}(c)
         & =
        \mc{T}_{B,v}\!\left(\mc{I}_f(T_f(X)^T)c\right)
        \label{eq:peps_region_transfer_realizes}
    \end{align}
    (\cite[Section~3, lines~254--582]{Molnar2018NormalPEPS}).
\end{definition}

\begin{lemma}[Realized region transfers are multiplicative and unital]
    \label{thm:peps_region_transfer_realized_maps}
    \lean{TNLean.PEPS.regionTransferMatrix_mul_of_realizes,
        TNLean.PEPS.regionTransferMatrix_one_of_realizes}
    \leanok
    \uses{def:peps_region_transfer_realizes, def:peps_region_insertion_transfer}
    If $T_f$ satisfies (\ref{eq:peps_region_transfer_realizes}), then $T_f(XY)=T_f(X)T_f(Y)$. If in addition $A$ and $B$ define the same state with $D_A=D_B>0$ and the blocked maps of $B$ on $R$ and $R^c$ are injective, then $T_f(\Id)=\Id$.
\end{lemma}
\begin{proof}\leanok
    Since $O_{A,R,f}((XY)^T)=O_{A,R,f}(Y^T)O_{A,R,f}(X^T)$, (\ref{eq:peps_region_transfer_realizes}) gives $\mc{I}_f(T_f(XY)^T)=\mc{I}_f((T_f(X)T_f(Y))^T)$ on the image of $\mc{T}_{B,v}$, whence $T_f(XY)=T_f(X)T_f(Y)$. For $X=\Id$, state and bond-dimension equality yield $C_A(R,f,\Id;\sigma,\tau)=C_B(R,f,\Id;\sigma,\tau)$, and injectivity of $C_B(R,f,-;\sigma,\tau)$ gives $T_f(\Id)=\Id$.
\end{proof}

\begin{definition}[Region insertion transfer from realized maps]
    \label{def:peps_region_insertion_transfer_of_realizes}
    \lean{TNLean.PEPS.regionInsertionTransfer_of_realizes}
    \leanok
    \uses{def:peps_region_transfer_realizes, thm:peps_region_transfer_realized_maps}
    If $T_f$ and $S_f$ satisfy (\ref{eq:peps_region_transfer_realizes}), $A$ and $B$ define the same state with $D_A=D_B>0$, and the blocked maps of $B$ on $R$ and $R^c$ are injective, then $(T_f,S_f)$ forms a region insertion transfer in the sense of Definition~\ref{def:peps_region_insertion_transfer}.
\end{definition}

\begin{lemma}[Incident form gives the region insertion transfer and gauge]
    \label{thm:peps_region_reconcile_to_transfer_gauge}
    \lean{TNLean.PEPS.regionTransferRealizes_of_isIncidentMatrixForm,
        TNLean.PEPS.regionTransferRealizes_of_reconcile,
        TNLean.PEPS.regionInsertionTransfer_of_reconcile,
        TNLean.PEPS.exists_regionEdgeGauge_of_reconcile}
    \leanok
    \uses{def:peps_region_resonate_reconcile, thm:peps_hform_of_region_incident_form, def:peps_same_state, def:IsVertexInjective, def:peps_regionInjectivityData}
    Assume $A$ and $B$ define the same PEPS state, are vertex-injective with positive bond dimensions, and have injective blocked maps on $R$ and $R^c$. If (\ref{eq:peps_region_resonate_reconcile}) holds in both directions, then the induced maps $T_f:\MN{D_A(f)}\to\MN{D_B(f)}$ and $S_f:\MN{D_B(f)}\to\MN{D_A(f)}$ satisfy (\ref{eq:peps_region_insertion_transfer}). If in addition $D_A(e)=D_B(e)$ for every edge $e$, with induced identification $\iota_f:\MN{D_A(f)}\simeq\MN{D_B(f)}$, then there exists $Z_f\in\GL(D_B(f))$ such that $T_f(X)=Z_f\,\iota_f(X)\,Z_f^{-1}$.
\end{lemma}
\begin{proof}\leanok
    Condition (\ref{eq:peps_region_resonate_reconcile}) and Lemma~\ref{thm:peps_hform_of_region_incident_form} yield (\ref{eq:peps_region_transfer_realizes}) in both directions, hence a region insertion transfer by Definition~\ref{def:peps_region_insertion_transfer_of_realizes}; the one-bond algebra theorem then gives $Z_f\in\GL(D_B(f))$ with $T_f(X)=Z_f\,\iota_f(X)\,Z_f^{-1}$.
\end{proof}

\begin{lemma}[Coefficient transfer gives incident form]
    \label{thm:peps_region_coeff_transfer_to_reconcile}
    \lean{TNLean.PEPS.regionInsertionOp_regionStateVec_pin_leg,
        TNLean.PEPS.regionInsertionOp_realizes_of_realizes_on_stateVec,
        TNLean.PEPS.regionInteriorBondProd_pos,
        TNLean.PEPS.regionInsertionOp_regionStateVec_eq_of_coeff_eq,
        TNLean.PEPS.isIncidentMatrixForm_of_coeff_eq,
        TNLean.PEPS.regionResonateReconcile_of_coeff_transfer}
    \leanok
    \uses{def:peps_region_resonate_reconcile}
    Assume $A$ and $B$ define the same PEPS state, $B$ is vertex-injective, $D_A(e)=D_B(e)>0$ for every edge $e$, and $A_v,B_v$ are linearly independent at the in-region endpoint $v$ of $f$. If for every $X\in\MN{D_A(f)}$ there exists $Y\in\MN{D_B(f)}$ such that $C_A(R,f,X;\sigma,\tau)=C_B(R,f,Y;\sigma,\tau)$ for all $\sigma,\tau$, then (\ref{eq:peps_region_resonate_reconcile}) holds.
\end{lemma}
\begin{proof}\leanok
    Matching $C_A(R,f,X;\sigma,\tau)=C_B(R,f,Y;\sigma,\tau)$ pins the closed state vectors one physical leg at a time. Since the state-vector coefficients span the local virtual space at $v$, the identity extends to all local virtual coefficients, giving $W_X=\mc{I}_f(Y^T)$.
\end{proof}

\begin{lemma}[First coefficient through the second complement block]
    \label{thm:peps_region_first_coeff_second_complement_block}
    \lean{TNLean.PEPS.regionInteriorBondProd_smul_regionStateVec,
        TNLean.PEPS.regionBlockedWeight_update_eq_regionWeightVec,
        TNLean.PEPS.regionInteriorBondProd_smul_regionStateVec_eq_sum,
        TNLean.PEPS.regionInsertedCoeff_eq_complement_blockedMap_B,
        TNLean.PEPS.regionBlockedLeftInverse_complement_regionInsertedCoeff_B}
    \leanok
    \uses{def:peps_region_out_endpoint_block_inverse}
    Assume $B$'s complement blocked map is injective, $A$ and $B$ define the same PEPS state with $D_A=D_B$, and $A_{v_f}$ is linearly independent at the in-region endpoint $v_f\in R$ of $f$. Then $\tau\mapsto C_A(R,f,X;\sigma,\tau)$ lies in the range of $B$'s complement blocked map, and applying $B$'s complement blocked left inverse recovers $\nu\mapsto[O_{A,R,f}(X^T)T_{B,R}^{\widetilde{\nu}}(\sigma)]_{\sigma(v_f)}$.
\end{lemma}
\begin{proof}\leanok
    Contracting $R$ and $R^c$ at $X=\Id$ and applying $O_{A,R,f}(X^T)$ factors $\tau\mapsto C_A(R,f,X;\sigma,\tau)$ through $B$'s complement blocked map; applying its left inverse reads off the row $\nu\mapsto[O_{A,R,f}(X^T)T_{B,R}^{\widetilde{\nu}}(\sigma)]_{\sigma(v_f)}$.
\end{proof}

\begin{lemma}[Transfer coefficient and double factorization]
    \label{thm:peps_region_transfer_coeff_double_factorization}
    \lean{TNLean.PEPS.vSideRow,
        TNLean.PEPS.regionInsertedCoeff_eq_complement_blockedMap_vSideRow,
        TNLean.PEPS.regionInsertedCoeff_eq_of_complementRow_eq,
        TNLean.PEPS.regionDoubleComplBoundaryConfigEquiv,
        TNLean.PEPS.regionBlockedTensorMap_doubleCompl,
        TNLean.PEPS.complSideRow,
        TNLean.PEPS.regionInsertedCoeff_eq_region_blockedMap_B,
        TNLean.PEPS.regionRowB,
        TNLean.PEPS.regionRowB_eq_complSideRow,
        TNLean.PEPS.regionRowB_mem_range,
        TNLean.PEPS.transferCoeff,
        TNLean.PEPS.regionRowB_eq_complement_blockedMap,
        TNLean.PEPS.regionInsertedCoeff_eq_doubleSum_transferCoeff,
        TNLean.PEPS.regionInsertedCoeff_eq_of_transferCoeff_form,
        TNLean.PEPS.vSideRow_eq_region_blockedMap_transferCoeff}
    \leanok
    \uses{thm:peps_region_first_coeff_second_complement_block}
    Assume $B$'s blocked maps on $R$ and $R^c$ are injective, $A$'s tensors at both endpoints of $f$ are linearly independent, and $A$ and $B$ define the same PEPS state with $D_A=D_B$. Set
    \begin{align}
        v_X^\sigma(\nu)
         & =
        \left[O_{A,R,f}(X^T)
            T_{B,R}^{\widetilde{\nu}}(\sigma)\right]_{\sigma(v_f)},
        \label{eq:peps_v_side_row}
    \end{align}
    where $\widetilde{\nu}$ is the $R$-boundary configuration corresponding to $\nu$ on $\partial R^c$ and $v_f\in R$ is the endpoint of $f$. Then there exist transfer coefficients $K_X(\mu,\nu)$ such that
    \begin{align}
        C_A(R,f,X;\sigma,\tau)
         & =
        \sum_{\mu,\nu}
        K_X(\mu,\nu)\,T_{B,R}^{\mu}(\sigma)
        T_{B,R^c}^{\nu}(\tau).
        \label{eq:peps_transfer_coeff_double_sum}
    \end{align}
    If $K_X$ has the incident form
    \begin{align}
        K_X(\mu,\nu)
         & =
        \mathbf 1_{
                \mu|_{\partial R\setminus\{f\}}
                =
                \nu|_{\partial R\setminus\{f\}}
            }\,Y_{\mu_f,\nu_f},
        \label{eq:peps_transfer_coeff_form}
    \end{align}
    then $C_A(R,f,X;\sigma,\tau)=C_B(R,f,Y;\sigma,\tau)$.
\end{lemma}
\begin{proof}\leanok
    Factoring $C_A$ successively through $B$'s complement and region blocks via Lemma~\ref{thm:peps_region_first_coeff_second_complement_block} and their left inverses produces $K_X$ satisfying (\ref{eq:peps_transfer_coeff_double_sum}). Substituting (\ref{eq:peps_transfer_coeff_form}) into (\ref{eq:peps_transfer_coeff_double_sum}) yields $C_B(R,f,Y;\sigma,\tau)$.
\end{proof}

\begin{lemma}[Incident form of the transfer coefficient]
    \label{thm:peps_region_transfer_coeff_incident_form}
    \lean{TNLean.PEPS.transferCoeff_column_eq_regionBlockedLeftInverse_vSideRow,
        TNLean.PEPS.vSideRow_eq_localTensorMap_virtualPullback,
        TNLean.PEPS.vSideRow_eq_incidentSum_of_virtualPullback_incidentForm,
        TNLean.PEPS.transferCoeff_eq_incidentForm_of_virtualPullback_incidentForm}
    \leanok
    \uses{thm:peps_region_transfer_coeff_double_factorization, thm:peps_hform_of_region_incident_form, thm:peps_region_insertion_virtual_pullback_realizes}
    Under the hypotheses of Lemma~\ref{thm:peps_region_transfer_coeff_double_factorization}, assume further that $A$ and $B$ are vertex-injective with $D_A,D_B>0$ and $B_{v_f}$ is linearly independent. If $W_X=\mc{I}_f(Y^T)$, then $K_X$ satisfies (\ref{eq:peps_transfer_coeff_form}).
\end{lemma}
\begin{proof}\leanok
    For fixed $\nu$, $\mu\mapsto K_X(\mu,\nu)$ is the $R$-blocked left inverse of $\sigma\mapsto v_X^\sigma(\nu)$ from (\ref{eq:peps_v_side_row}). When $W_X=\mc{I}_f(Y^T)$, $v_X^\sigma(\nu)$ expands as the incident-matrix sum of $Y$ against $T_{B,R}^\mu(\sigma)$, and applying the left inverse yields (\ref{eq:peps_transfer_coeff_form}).
\end{proof}

\begin{definition}[Coefficient transfer map]
    \label{def:peps_coeff_transfer_map}
    \lean{TNLean.PEPS.coeffTransferMap}
    \leanok
    \uses{def:peps_regionInsertedCoeff}
    If for every $X\in\MN{D_A(f)}$ there exists $Y\in\MN{D_B(f)}$ with $C_A(R,f,X;\sigma,\tau)=C_B(R,f,Y;\sigma,\tau)$ for all $\sigma,\tau$, choosing one such $Y=T_f(X)$ defines the coefficient transfer map $T_f:\MN{D_A(f)}\to\MN{D_B(f)}$.
\end{definition}

\begin{lemma}[Coefficient transfer matches coefficients and preserves the identity]
    \label{thm:peps_coeff_transfer_map_properties}
    \lean{TNLean.PEPS.coeffTransferMap_coeff,
        TNLean.PEPS.coeffTransferMap_one}
    \leanok
    \uses{def:peps_coeff_transfer_map}
    The map $T_f$ of Definition~\ref{def:peps_coeff_transfer_map} satisfies $C_A(R,f,X;\sigma,\tau)=C_B(R,f,T_f(X);\sigma,\tau)$ for all $X,\sigma,\tau$. If $A$ and $B$ define the same state with $D_A=D_B>0$ and $B$'s blocked maps on $R$ and $R^c$ are injective, then $T_f(\Id)=\Id$.
\end{lemma}
\begin{proof}\leanok
    Coefficient matching holds by Definition~\ref{def:peps_coeff_transfer_map}. At $X=\Id$, state and bond-dimension equality give $C_A(R,f,\Id;\sigma,\tau)=C_B(R,f,\Id;\sigma,\tau)$, and injectivity of $C_B(R,f,-;\sigma,\tau)$ forces $T_f(\Id)=\Id$.
\end{proof}

\begin{definition}[Region insertion transfer from coefficient transfers]
    \label{def:peps_region_insertion_transfer_of_coeff_transfer}
    \lean{TNLean.PEPS.regionInsertionTransfer_of_coeffTransfer}
    \leanok
    \uses{def:peps_region_insertion_transfer, def:peps_coeff_transfer_map, thm:peps_coeff_transfer_map_properties}
    If coefficient transfers $T_f$ and $S_f$ exist in both directions with $T_f(XY)=T_f(X)T_f(Y)$, $A$ and $B$ define the same state with $D_A=D_B>0$, and $B$'s blocked maps on $R$ and $R^c$ are injective, then $(T_f,S_f)$ forms a region insertion transfer in the sense of Definition~\ref{def:peps_region_insertion_transfer}.
\end{definition}

\begin{lemma}[Per-edge gauge from coefficient transfers]
    \label{thm:peps_region_edge_gauge_of_coeff_transfer}
    \lean{TNLean.PEPS.exists_regionEdgeGauge_of_coeffTransfer}
    \leanok
    \uses{def:peps_region_insertion_transfer_of_coeff_transfer}
    Under the hypotheses of Definition~\ref{def:peps_region_insertion_transfer_of_coeff_transfer}, if the blocked maps of $A$ on $R$ and $R^c$ are also injective, then there exists $Z_f\in\GL(D_B(f))$ such that $T_f(X)=Z_f\,\iota_f(X)\,Z_f^{-1}$ for all $X\in\MN{D_A(f)}$, where $\iota_f:\MN{D_A(f)}\simeq\MN{D_B(f)}$ is induced by $D_A(f)=D_B(f)$.
\end{lemma}
\begin{proof}\leanok
    By Definition~\ref{def:peps_region_insertion_transfer_of_coeff_transfer}, $(T_f,S_f)$ is a region insertion transfer, inducing an algebra isomorphism $\MN{D_A(f)}\simeq\MN{D_B(f)}$; the Skolem--Noether theorem yields $Z_f\in\GL(D_B(f))$ with $T_f(X)=Z_f\,\iota_f(X)\,Z_f^{-1}$.
\end{proof}

\begin{lemma}[Determinacy of a region insertion transfer]
    \label{thm:peps_regionInsertedCoeff_transferMap_unique}
    \lean{TNLean.PEPS.regionInsertedCoeff_transferMap_unique}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_region_out_endpoint_block_inverse}
    Assume $B$'s blocked maps on $R$ and $R^c$ are injective and $D_B>0$. If $T_1,T_2:\MN{D_A(f)}\to\MN{D_B(f)}$ satisfy $C_A(R,f,M;\sigma,\tau)=C_B(R,f,T_i(M);\sigma,\tau)$ for $i=1,2$ and all $M,\sigma,\tau$, then $T_1=T_2$.
\end{lemma}
\begin{proof}\leanok
    For all $M,\sigma,\tau$, $C_B(R,f,T_1(M);\sigma,\tau)=C_A(R,f,M;\sigma,\tau)=C_B(R,f,T_2(M);\sigma,\tau)$, and injectivity of $M'\mapsto C_B(R,f,M';-,-)$ gives $T_1(M)=T_2(M)$.
\end{proof}

\begin{lemma}[Coefficient identities determine the boundary conjugation]
    \label{thm:peps_gaugeConj_eq_of_coeffIdentities}
    \lean{TNLean.PEPS.gaugeConj_eq_of_coeffIdentities}
    \leanok
    \uses{thm:peps_regionInsertedCoeff_transferMap_unique}
    Under the hypotheses of Lemma~\ref{thm:peps_regionInsertedCoeff_transferMap_unique}, let $\iota:\MN{D_A(f)}\simeq\MN{D_B(f)}$ be induced by $D_A(f)=D_B(f)$. If $Z_1,Z_2\in\GL(D_B(f))$ satisfy $C_A(R,f,M;\sigma,\tau)=C_B(R,f,Z_i\,\iota(M)\,Z_i^{-1};\sigma,\tau)$ for $i=1,2$ and all $M,\sigma,\tau$, then $Z_1\,\iota(M)\,Z_1^{-1}=Z_2\,\iota(M)\,Z_2^{-1}$ for all $M$.
\end{lemma}
\begin{proof}\leanok
    Apply Lemma~\ref{thm:peps_regionInsertedCoeff_transferMap_unique} to $T_i(M):=Z_i\,\iota(M)\,Z_i^{-1}$.
\end{proof}
