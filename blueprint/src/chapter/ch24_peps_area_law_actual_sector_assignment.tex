\section{Initial regions in eight sectors}
\label{sec:area_law_actual_sector_assignment}

Fix an origin $o\in\mathbb R^2$, a finite endpoint set
$Z\subset\mathbb Z^2$, an integer width $C\ge2$, and an initial layer
$k_0\ge50{,}000{,}000$. Choose valid pitch residues at every
layer, and let $\mathcal I$ be the resulting set of initial region identifiers.
Write $X_i^0$ for their closed initial regions and
$U_i=\operatorname{int}(X_i^0)$ for their open interiors.
Let $k\ge k_0$, let $z\in F_{k,\ell_k}$ be a fine-cell index,
and let $v$ be any of the nine marks of its cell. Set
\begin{align}
    \ell & =\ell_k-5,                        \\
    r    & =\frac{2^\ell}{2}=\frac{t_k}{64}.
\end{align}
The bound $k\ge k_0$ implies $\ell_k\ge5$. Endow $\mathbb R^2$ with the
maximum norm. Center two fans at $v$, with origins
$o_{\rm S}=v-r(1,1)$ and $o_{\rm L}=v-2r(1,1)$, exponents $\ell$ and
$\ell+1$, and cell index $(0,0)$, subdividing every side at its midpoint.
Let $\mathcal J$ index the eight resulting elementary perimeter segments, and
write $P_s^{\rm S}$ and $P_s^{\rm L}$ for the corresponding closed
triangles, whose squares have half-sides $r=t_k/64$ and $2r=t_k/32$.
This local construction appears in
\cite[Section~11]{OpenAI2026AreaLaw}.

\begin{lemma}[Closed square centered at a point]
    \label{thm:area_law_centered_dyadic_cell_ball}
    \lean{TNLean.PEPS.AreaLaw.Geometry.closure_centered_dyadicCell_eq_closedBall}
    \leanok
    \uses{def:area_law_dyadic_cells}
    Let $c\in\mathbb R^2$ and $m\in\mathbb N$, and set $\rho=2^m/2$.
    Let $Q=c-\rho(1,1)+2^m[0,1)^2$ be the $(0,0)$ half-open
    dyadic cell with origin $c-\rho(1,1)$. In the maximum norm,
    \begin{align}
        \overline Q
         & =c+[-\rho,\rho]^2
        =\overline B_\infty(c,\rho).
    \end{align}
    This is the centered square used in the local fan construction of
    \cite[Section~11]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dyadic_closures}
    The closure of $Q$ is the product of the closed coordinate intervals
    with lower endpoints $c_1-\rho$ and $c_2-\rho$ and length $2^m=2\rho$.
    Their upper endpoints are $c_1+\rho$ and $c_2+\rho$, so their product is
    $\overline B_\infty(c,\rho)$.
\end{proof}

\begin{lemma}[Closed half-radius scale]
    \label{thm:area_law_initial_closed_half_radius}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineScaleIndex_closed_half_radius_eq}
    \leanok
    \uses{def:area_law_fine_belt_scales}
    Let $k\ge50{,}000{,}000$, write $\ell_k$ for its fine-cell exponent,
    and set
    \begin{align}
        t_k & =2^{\ell_k},           \\
        r   & =\frac{2^{\ell_k-5}}2.
    \end{align}
    Then
    \begin{align}
        \frac r2 & =\frac{t_k}{128}.
        \label{eq:area_law_closed_half_radius}
    \end{align}
    This is the scale relation for the half-radius square in the initial-star
    geometry of \cite[Section~11]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{def:area_law_fine_belt_scales,thm:area_law_adjacent_fine_meshes}
    Since $k\ge50{,}000{,}000$, we have $\ell_k\ge5$ and
    $2^{\ell_k}=32\cdot2^{\ell_k-5}$.
    Dividing by $128$ proves~(\ref{eq:area_law_closed_half_radius}).
\end{proof}

\begin{lemma}[Unique sector assignment and closed restriction]
    \label{thm:area_law_actual_sector_assignment}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_unique_initialRegion_sector_assignment}
    \leanok
    \uses{def:area_law_initial_region_indices,
        def:area_law_initial_birth_regions,def:area_law_initial_open_regions,
        def:area_law_fine_layer_indices,def:area_law_fine_belt_scales,
        def:area_law_belt_cell_marks,def:area_law_cell_fan}
    There is a unique map $\sigma:\mathcal J\to\mathcal I$
    such that every $s\in\mathcal J$ satisfies
    \begin{align}
        \operatorname{int}(P_s^{\rm L}) & \subseteq U_{\sigma(s)},
    \end{align}
    and every $i\in\mathcal I$ satisfies
    \begin{align}
        X_i^0\cap\overline B_\infty(v,r)
         & =\bigcup_{\substack{s\in\mathcal J \\\sigma(s)=i}}P_s^{\rm S}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_belt_marks_closed_cell,
        thm:area_law_fine_cell_containment,thm:area_law_fine_belt_scales,
        thm:area_law_dyadic_layer_distance,
        thm:area_law_initial_closed_cover,thm:area_law_initial_birth_regularity,
        thm:area_law_initial_open_disjoint,
        thm:area_law_initial_frontier_near_mark,
        thm:area_law_cell_fan_regularity,thm:area_law_cell_fan_cover,
        thm:area_law_cell_fan_sector_directions,
        thm:area_law_concentric_fan_restriction,
        thm:area_law_dyadic_closures,thm:area_law_adjacent_fine_meshes,
        def:area_law_cell_fan}
    Since $v$ belongs to the closed fine cell, it belongs to the closed layer
    $\overline{D_k}$. The distance condition defining $D_k$ shows that $Z$ is
    nonempty, so the closed initial regions $X_i^0$ cover $\mathbb R^2$. From
    $k\ge k_0$ we obtain $\ell_k\ge5$ and $2r=t_k/32$.

    Set $S_s=\operatorname{int}(P_s^{\rm L})$ and $W=B_\infty(v,2r)$. Because
    $\bigcup_sP_s^{\rm L}=\overline B_\infty(v,2r)$, taking interiors gives
    $S_s\subseteq W$. Each $S_s$ is nonempty and connected.

    No $S_s$ meets the boundary $\partial X_i^0$ of any initial region: a point
    $x\in S_s\cap\partial X_i^0$ would satisfy $\|x-v\|_\infty<t_k/32$, forcing
    $x-v$ to lie along one of the four allowed directions through $v$, whereas
    the open triangle $S_s$ is disjoint from those four lines. Since the sets
    $X_i^0=\overline{U_i}$ cover $S_s$ and $\partial X_i^0\cap S_s=\varnothing$,
    connectedness of $S_s$ and pairwise disjointness of the open sets $U_i$
    imply that $S_s\subseteq U_{\sigma(s)}$ for a unique $\sigma(s)\in\mathcal I$.

    Put $D=\bigcup_sS_s$, so $\overline D=\bigcup_sP_s^{\rm L}=\overline B_\infty(v,2r)$.
    For $i\in\mathcal I$, the finite union $E_i=\bigcup_{\sigma(s)=i}P_s^{\rm L}$
    is closed and satisfies $U_i\cap D\subseteq E_i$. Since $U_i$ and $W$ are
    open and $W\subseteq\overline D$,
    \begin{align}
        U_i\cap W   & \subseteq\overline{U_i\cap D}\subseteq E_i, \\
        X_i^0\cap W & =\overline{U_i}\cap W
        \subseteq\overline{U_i\cap W}\subseteq E_i.
    \end{align}
    Conversely, taking closures of $S_s\subseteq U_i$ gives $E_i\subseteq X_i^0$.
    Since $\overline B_\infty(v,r)\subseteq W$, intersecting with
    $\overline B_\infty(v,r)$ and applying
    $P_s^{\rm L}\cap\overline B_\infty(v,r)=P_s^{\rm S}$ yields the stated
    decomposition.
\end{proof}

\begin{definition}[Initial sector assignment]
    \label{def:area_law_initial_sector_assignment}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialSectorAssignment}
    \leanok
    \uses{thm:area_law_actual_sector_assignment}
    The initial sector assignment is the unique map
    $\sigma:\mathcal J\to\mathcal I$ from
    Lemma~\ref{thm:area_law_actual_sector_assignment}.
\end{definition}
