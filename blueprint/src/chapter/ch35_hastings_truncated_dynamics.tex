\section{The truncated approximation of a filtered term}
\label{sec:hastings_truncated_dynamics}

% Hastings, arXiv:cond-mat/0508554v3, proj3new.tex, eqs. mid, odef, hpdef, apb
% (lines 285--320), proved at lines 1458--1533.

For a term $H_i$ of $H=\sum_jH_j$, \cite{Hastings2006Solving} approximates the
filtered term $\tilde H_i^0$ by filtering $H_i$ under the truncated Hamiltonian
$H_{\rm loc}=\sum_{j:\,d(i,j)\le l_{proj}-R}H_j$:
\begin{align}
    M_i=\frac{\Delta E}{\sqrt{2\pi q}}\int dt\,e^{-(t\Delta E)^2/(2q)}\,
    e^{iH_{\rm loc}t}H_ie^{-iH_{\rm loc}t}
    \label{eq:hastings_truncated_mi}
\end{align}
\cite[eqs.~(mid), (odef), (hpdef), lines 285--303]{Hastings2006Solving}. More
generally, for $W$ supported on $S\subseteq K$ and a subset $s$ of terms
supported inside $K$, the shell consists of omitted terms ($k\notin s$) whose
supports meet $K$ \cite{gap:hastings05_truncated_dynamics}.

\begin{theorem}[Support of the truncated filter]
    \label{thm:hastings_truncated_support}
    \lean{TNLean.Lattice.hermitianUnitaryPath_conj_mem_supportedOperators,
        TNLean.Lattice.gaussianFilteredTerm_mem_supportedOperators}
    \leanok
    \uses{def:hastings_gaussian_filtered_term}
    If $H$ and $W$ act on $K$, then $e^{iHt}We^{-iHt}$ ($t\in\mathbb R$) and
    $\widetilde W$ act on $K$; in particular, $M_i$ in
    (\ref{eq:hastings_truncated_mi}) acts within distance $l_{proj}$ of $i$
    \cite[lines 302--304]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    Operators supported on $K$ form a closed subalgebra, hence contain
    $e^{iHt}We^{-iHt}$ and its Gaussian integral $\widetilde W$.
\end{proof}

\begin{theorem}[Comparison of two evolutions]
    \label{thm:hastings_truncated_evolution_comparison}
    \lean{TNLean.Lattice.norm_conj_sub_conj_le}
    \leanok
    Let $H=H^\dagger$, $H'$ be any matrix, and $\tau'_u(W)=e^{iH'u}We^{-iH'u}$.
    If $\|[H-H',\tau'_u(W)]\|\le\varepsilon$ for all $u$ between $0$ and $t$,
    then
    \begin{align}
        \|e^{iHt}We^{-iHt}-e^{iH't}We^{-iH't}\|\le\varepsilon\,|t|
        \label{eq:hastings_truncated_evolution_comparison}
    \end{align}
    \cite[lines 1469--1482]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    The map $f(s)=e^{iHs}\tau'_{t-s}(W)e^{-iHs}$ satisfies $f(0)=\tau'_t(W)$,
    $f(t)=e^{iHt}We^{-iHt}$, and
    $\|f'(s)\|=\|[H-H',\tau'_{t-s}(W)]\|\le\varepsilon$.
\end{proof}

\begin{theorem}[Filtering a comparison of evolutions]
    \label{thm:hastings_truncated_filter_comparison}
    \lean{TNLean.Lattice.norm_gaussianFilteredTerm_sub_le}
    \leanok
    \uses{def:hastings_gaussian_filtered_term}
    If $H,H'$ are Hermitian, $q,T,\varepsilon\ge0$, and
    $\|e^{iHt}We^{-iHt}-e^{iH't}We^{-iH't}\|\le\varepsilon$ for all $|t|\le T$,
    then the Gaussian filters $\widetilde W_H$ and $\widetilde W_{H'}$ satisfy
    $\|\widetilde W_H-\widetilde W_{H'}\|\le\varepsilon+2\|W\|e^{-(T\Delta E)^2/(2q)}$
    \cite[eq.~(dmi), lines 1524--1533]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_filtered_locality}
    Split the Gaussian integral at $|t|=T$ as in
    Theorem~\ref{thm:hastings_filtered_locality}, using integrand bound
    $\varepsilon$ on $[-T,T]$ and $2\|W\|$ on $\mathbb R\setminus[-T,T]$.
\end{proof}

\begin{theorem}[Lieb--Robinson bound against an operator on several sites]
    \label{thm:hastings_truncated_multisite_lieb_robinson}
    \lean{TNLean.Lattice.norm_commutator_heisenbergEvolution_le_graphDistance_finset}
    \leanok
    \uses{thm:qc_graph_lieb_robinson_finset}
    Under the assumptions of Theorem~\ref{thm:qc_graph_lieb_robinson_finset}, if
    $A$ acts on $S$, $B$ acts on $Y$, and $d_G(x,y)\ge m$ for all $x\in S$,
    $y\in Y$, then for all $\mu\ge0$, with $C=2+4|b_0|e^{2\mu R}$ and
    $v=2v_R|b_0|e^{2\mu R}+1$,
    \begin{align}
        \|[B,\tau_t(A)]\|\le2|Y|\,C\,|S|\,\|A\|\,\|B\|\,e^{v|t|-\mu m}.
        \label{eq:hastings_truncated_multisite_lieb_robinson}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:qc_graph_lieb_robinson_finset, lem:qc_site_expectation_commutator}
    With $E$ the normalized partial trace over $Y$,
    $\|[B,\tau_t(A)]\|\le2\|B\|\,\|\tau_t(A)-E(\tau_t(A))\|$; bound the latter
    by summing the single-site commutators over $y\in Y$ via
    Lemma~\ref{lem:qc_site_expectation_commutator} and
    Theorem~\ref{thm:qc_graph_lieb_robinson_finset}.
\end{proof}

\begin{theorem}[The omitted terms against the truncated evolution]
    \label{thm:hastings_truncated_omitted_commutator}
    \lean{TNLean.Lattice.norm_commutator_sub_truncated_le}
    \leanok
    \uses{thm:hastings_truncated_multisite_lieb_robinson}
    Let $W$ act on $S\subseteq K$, $s$ be a family of terms supported in $K$,
    $H_s=\sum_{k\in s}h_k$, and $d_G(S,X_k)\ge m$ for every omitted term
    $k\notin s$ with $X_k\cap K\ne\emptyset$. Then
    \begin{align}
        \|[H-H_s,e^{iH_su}We^{-iH_su}]\|
        \le2v_RC\,|S|\,\|W\|\Bigl(\sum_{\substack{k\notin s \\ X_k\cap K\ne\emptyset}}\|h_k\|\Bigr)e^{v|u|-\mu m}
        \label{eq:hastings_truncated_omitted_commutator}
    \end{align}
    \cite[lines 1469--1505]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_support,
        thm:hastings_truncated_multisite_lieb_robinson}
    By Theorem~\ref{thm:hastings_truncated_support}, $e^{iH_su}We^{-iH_su}$ acts
    on $K$, so omitted terms with $X_k\cap K=\emptyset$ commute with it; for
    $X_k\cap K\ne\emptyset$, apply
    (\ref{eq:hastings_truncated_multisite_lieb_robinson}) with $|X_k|\le v_R$.
\end{proof}

\begin{theorem}[Hastings' truncated approximation]
    \label{thm:hastings_truncated_approximation}
    \lean{TNLean.Lattice.norm_gaussianFilteredTerm_sub_truncated_le_sum,
        TNLean.Lattice.norm_gaussianFilteredTerm_sub_truncated_le,
        TNLean.Lattice.truncated_approx_gaussianFilteredTerm}
    \leanok
    \uses{def:hastings_gaussian_filtered_term}
    In the setting of Theorem~\ref{thm:hastings_truncated_omitted_commutator},
    if $\|h_k\|\le J$ and $N=|\{k\notin s:X_k\cap K\ne\emptyset\}|$, then the
    filter $M$ of $W$ under $H_s$ acts on $K$, satisfies $\|M\|\le\|W\|$ (and
    $M=M^\dagger$ if $W=W^\dagger$), and for all $q,T,\mu\ge0$,
    \begin{align}
        \|\widetilde W-M\|\le2v_RC\,|S|\,\|W\|\,J\,N\,T\,e^{vT-\mu m}
        +2\|W\|e^{-(T\Delta E)^2/(2q)}
        \label{eq:hastings_truncated_approximation}
    \end{align}
    \cite[eq.~(apb), lines 300--320]{Hastings2006Solving},
    \cite{gap:hastings05_truncated_dynamics}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_support,
        thm:hastings_truncated_evolution_comparison,
        thm:hastings_truncated_filter_comparison,
        thm:hastings_truncated_omitted_commutator,
        thm:hastings_gaussian_filtered_spectral}
    Combine Theorems~\ref{thm:hastings_truncated_omitted_commutator},
    \ref{thm:hastings_truncated_evolution_comparison}, and
    \ref{thm:hastings_truncated_filter_comparison} with
    Theorems~\ref{thm:hastings_truncated_support}
    and~\ref{thm:hastings_gaussian_filtered_spectral}.
\end{proof}

\begin{theorem}[The truncated approximation at the time window $T=q/\Delta E$]
    \label{thm:hastings_truncated_approximation_time}
    \lean{TNLean.Lattice.exp_neg_sq_div_eq_of_time_eq,
        TNLean.Lattice.norm_gaussianFilteredTerm_sub_truncated_le_of_time}
    \leanok
    \uses{thm:hastings_truncated_approximation}
    In the setting of Theorem~\ref{thm:hastings_truncated_approximation} with
    $q\ge0$ and $\Delta E>0$, setting $T=q/\Delta E$ gives
    \begin{align}
        \|\widetilde W-M\|\le2v_RC\,|S|\,\|W\|\,J\,N\,\frac{q}{\Delta E}\,
        e^{vq/\Delta E-\mu m}+2\|W\|e^{-q/2}
        \label{eq:hastings_truncated_approximation_time}
    \end{align}
    \cite[lines 337--349]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_approximation}
    Substitute $T=q/\Delta E$ into (\ref{eq:hastings_truncated_approximation}).
\end{proof}

\begin{theorem}[Ground vectors under the truncated approximation]
    \label{thm:hastings_truncated_facld}
    \lean{TNLean.Lattice.norm_sub_starProjection_truncated_apply_le,
        TNLean.Lattice.norm_truncated_apply_sub_inner_smul_le}
    \leanok
    \uses{thm:hastings_truncated_approximation_time, thm:hastings05_facld}
    In the setting of Theorem~\ref{thm:hastings_truncated_approximation} with
    $q\ge0$, ground energy $E_0$, gap $\Delta E>0$, ground projection $P$, and
    \begin{align}
        \delta=3\|W\|e^{-q/2}+2v_RC\,|S|\,\|W\|\,J\,N\,\frac{q}{\Delta E}\,
        e^{vq/\Delta E-\mu m},
        \label{eq:hastings_truncated_facld_delta}
    \end{align}
    every ground vector $\psi$ satisfies $\|(1-P)M\psi\|\le\delta\|\psi\|$, and
    a unique unit ground state $\Omega$ satisfies
    $\|M\Omega-\langle\Omega,M\Omega\rangle\Omega\|\le\delta$
    \cite[eq.~(facld), lines 322--348 and 375--387]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_approximation_time, thm:hastings05_facld}
    Combine Theorem~\ref{thm:hastings05_facld} with
    (\ref{eq:hastings_truncated_approximation_time}).
\end{proof}

\begin{theorem}[The sum of the truncated approximations]
    \label{thm:hastings_truncated_sum}
    \lean{TNLean.Lattice.norm_sum_truncated_sub_le,
        TNLean.Lattice.norm_sum_truncated_sub_le_card}
    \leanok
    \uses{thm:hastings_truncated_approximation_time,
        thm:hastings05_sum_approximation}
    In the setting of Theorem~\ref{thm:hastings_truncated_approximation} with
    $q,\mu\ge0$, $\Delta E>0$, and $H=\sum_ih_i=H^\dagger$, let each $h_i$
    (supported on $X_i\subseteq K_i$) be filtered under $\sum_{k\in s_i}h_k$ to
    $M_i$, where the $N_i$ omitted terms meeting $K_i$ satisfy
    $d_G(X_i,X_k)\ge m$. Then
    \begin{align}
        \Bigl\|\sum_iM_i-H\Bigr\|\le\sum_i\Bigl(2v_RC\,|X_i|\,\|h_i\|\,J\,N_i\,
        \frac{q}{\Delta E}\,e^{vq/\Delta E-\mu m}+2\|h_i\|e^{-q/2}\Bigr),
        \label{eq:hastings_truncated_sum}
    \end{align}
    which is at most $|\{i\}|\bigl(2v_R^2CJ^2N_0(q/\Delta E)e^{vq/\Delta E-\mu m}+2Je^{-q/2}\bigr)$
    when $N_i\le N_0$ \cite[lines 343--374]{Hastings2006Solving}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_approximation_time,
        thm:hastings05_sum_approximation}
    Combine Theorem~\ref{thm:hastings05_sum_approximation} with
    (\ref{eq:hastings_truncated_approximation_time}) for each $W=h_i$, and use
    $|X_i|\le v_R$, $\|h_i\|\le J$, $N_i\le N_0$.
\end{proof}

\begin{remark}
    For $\mu>0$, $m=l_{proj}-3R$, $q=c_1l_{proj}\Delta E$ with $0<c_1<\mu/v$,
    and $N$ subexponential in $l_{proj}$, $\delta$ in
    (\ref{eq:hastings_truncated_facld_delta}) decays as
    $JO(e^{-l_{proj}/l_{fac}})$ \cite{gap:hastings05_truncated_dynamics}.
\end{remark}

\begin{theorem}[The exponential decay of eq.~(facld)]
    \label{thm:hastings_truncated_facld_decay}
    \lean{TNLean.Lattice.exists_norm_sub_starProjection_truncated_apply_le_exp,
        TNLean.Lattice.exists_norm_truncated_apply_sub_inner_smul_le_exp,
        Real.add_one_pow_mul_le_mul_exp}
    \leanok
    \uses{thm:hastings_truncated_facld}
    In the setting of Theorem~\ref{thm:hastings_truncated_facld}, let $J\ge0$,
    $0<c_1v<\mu$, and $l_{fac}=2/\min(c_1\Delta E,\mu-c_1v)$. For each
    $l\in\mathbb N$, let $M_l$ be the truncated filter of $W$ at
    $q=c_1l\Delta E$ with separation $m=l-3R$ and shell count
    $N_l\le K_g(l+1)^k$. Then there exist $C_1',C_2'\ge0$ independent of $l$
    such that $\|(1-P)M_l\psi\|\le\|W\|(3+C_1'J)e^{-l/l_{fac}}\|\psi\|$ for
    every ground vector $\psi$, and
    $\|M_l\Omega-\langle\Omega,M_l\Omega\rangle\Omega\|\le\|W\|(3+C_2'J)e^{-l/l_{fac}}$
    for a unique ground state $\Omega$
    \cite[eq.~(facld), lines 322--349]{Hastings2006Solving},
    \cite{gap:hastings05_truncated_dynamics}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings_truncated_facld}
    With $\delta_0=\min(c_1\Delta E,\mu-c_1v)=2/l_{fac}>0$, $q=c_1l\Delta E$, and
    $\mu m\ge\mu l-3\mu R$, one has $e^{-q/2}\le e^{-\delta_0l/2}$,
    $e^{vq/\Delta E-\mu m}\le e^{3\mu R}e^{-\delta_0l}$, and
    $N_ll\le K_g(l+1)^kl\le K_g(k+1)!e^{\delta_0/2}(\delta_0/2)^{-(k+1)}e^{\delta_0l/2}$;
    substitute into (\ref{eq:hastings_truncated_facld_delta}).
\end{proof}
