\section{Square-grid norms, regional states and boundaries}%
\label{sec:peps_square_grid_conversion}

Let $V_L=\{0,\ldots,L-1\}^2$ be an open square grid. Its forward edges are
$((x,y),(x+1,y))$ and $((x,y),(x,y+1))$ whenever both vertices lie in $V_L$.
The PEPS definition preceding Theorem~1.1 of
\emph{Polynomial PEPS approximation of gapped square-grid ground states}
(OpenAI, September 24, 2026) assigns a dimension $D_e$ to each edge and an
arbitrary linear map at every vertex. We compare its coefficients with those
of the same finite simple graph whose edges are oriented lexicographically.

For a $2\times2$ grid, the contraction has four independent virtual bonds
and four open physical indices. Each bond retains its own dimension.

\begin{theorem}[Exact Euclidean vector equality]%
    \label{thm:peps_square_vectors_exact}
    \lean{TNLean.PEPS.Approximation.stateVector_pinnedTensorToGraphTensor,
        TNLean.PEPS.Approximation.stateVector_vectorTensorToGraphTensor,
        TNLean.PEPS.Approximation.isometry_squareGridState,
        TNLean.PEPS.Approximation.norm_stateVector_pinnedTensorToGraphTensor}
    \leanok
    \uses{thm:peps_square_grid_contraction}
    The two coordinate descriptions give the identical vector
    $\Psi_{\widetilde A}=\Psi_A$ in the physical Hilbert space.
    Their physical configuration indices coincide, so this identification is an isometry.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_square_grid_contraction}
    The contracted coefficients agree for every physical configuration.
\end{proof}

\begin{definition}[Minimizing phase error]%
    \label{def:peps_square_phase_error}
    \lean{TNLean.PEPS.Approximation.Vector.PhaseErrorAtMost}
    \leanok
    Put $\widehat x=\|x\|^{-1}x$, with inverse zero at zero. A phase-error
    bound $\varepsilon$ means there is a phase $\theta$ minimizing
    $\|\widehat x-e^{i\theta}\Omega\|$ over all real phases and attaining
    a value at most $\varepsilon$. Nonvanishing of $x$ is a separate condition.
\end{definition}

\begin{theorem}[Unchanged bond bounds and normalized phase error]%
    \label{thm:peps_square_bounds_exact}
    \lean{TNLean.PEPS.Approximation.maxBondDim_vectorTensorToGraphTensor,
        TNLean.PEPS.Approximation.Vector.Tensor.maxBondDim_le_iff,
        TNLean.PEPS.Approximation.vectorTensorToGraphTensor_bondDim,
        TNLean.PEPS.Approximation.pinnedTensorToGraphTensor_bond_bounds,
        TNLean.PEPS.Approximation.pinnedTensorToGraphTensor_normalized,
        TNLean.PEPS.Approximation.pinnedTensorToGraphTensor_approximation,
        TNLean.PEPS.Approximation.vectorTensorToGraphTensor_approximation,
        TNLean.PEPS.Approximation.phaseErrorAtMost_vectorTensorToGraphTensor}
    \leanok
    \uses{thm:peps_square_vectors_exact}
    Every bond dimension, positivity condition and per-edge upper bound
    is preserved, as is the maximum bond dimension, with maximum zero for
    an empty edge set. If $\Psi_A\ne0$, its normalized vector has norm one
    and, for every $\Omega$ and phase $\theta$,
    \begin{align}
        \left\|\frac{\Psi_{\widetilde A}}{\|\Psi_{\widetilde A}\|}
        -e^{i\theta}\Omega\right\|
         & =\left\|\frac{\Psi_A}{\|\Psi_A\|}-e^{i\theta}\Omega\right\|.
    \end{align}
    A minimizing phase and any prescribed error bound are unchanged.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_square_edge_correspondence, thm:peps_square_vectors_exact}
    Dimensions are pulled back along a bijection. The vector equality
    gives equality of norms, nonvanishing, normalized vectors, and all
    phase-dependent distances.
\end{proof}

\begin{definition}[Regional coefficients and crossing edges]%
    \label{def:peps_square_regional_coefficients}
    \lean{TNLean.PEPS.Approximation.Pinned.RegionConfiguration,
        TNLean.PEPS.Approximation.Pinned.joinConfigurations,
        TNLean.PEPS.Approximation.Pinned.coefficientMatrix,
        TNLean.PEPS.Approximation.Pinned.reducedDensity,
        TNLean.PEPS.Approximation.Pinned.vonNeumannEntropy,
        TNLean.PEPS.Approximation.Pinned.boundaryCard,
        TNLean.PEPS.Approximation.joinConfigurations_eq_assembleRegion}
    \leanok
    A pair of configurations $x,z$ on $R,R^c$ gives one global configuration
    $x\cup z$. For a vector $\Psi$, let $M_R(x,z)=\Psi(x\cup z)$ and
    $\rho_R=M_RM_R^\dagger$. The spectral entropy is
    $\sum_j-\lambda_j\log\lambda_j$, where the $\lambda_j$ are the eigenvalues
    of $\rho_R$. The boundary consists of edges with exactly one endpoint in $R$.
\end{definition}

\begin{theorem}[Equal regional densities and crossing boundaries]%
    \label{thm:peps_square_regions_exact}
    \lean{TNLean.PEPS.Approximation.regionReducedDensity_pinnedTensorToGraphTensor,
        TNLean.PEPS.Approximation.regionReducedDensity_vectorTensorToGraphTensor,
        TNLean.PEPS.Approximation.entropy_pinnedTensorToGraphTensor,
        TNLean.PEPS.Approximation.forwardSquareBoundaryEquiv,
        TNLean.PEPS.Approximation.boundaryCard_eq_graph}
    \leanok
    \uses{thm:peps_square_vectors_exact, def:peps_region_reduced_density}
    For every region $R\subseteq V_L$, the two unnormalized reduced
    matrices agree: $\rho_R(\Psi_{\widetilde A})=\rho_R(\Psi_A)$.
    Their spectral entropy functions $\sum_j-\lambda_j\log\lambda_j$
    agree, using natural logarithms and $0\log0=0$. This is the physical
    entropy when the global vector is a unit vector. The edge bijection
    also restricts to a bijection of the crossing boundaries, so their
    cardinalities agree for every cut, including empty and full regions.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_square_vectors_exact, def:peps_region_reduced_density,
        def:peps_square_edge_correspondence}
    The region/complement coefficient matrices coincide entrywise, hence
    so do their Gram matrices and spectral entropies. The edge bijection
    preserves both ordered endpoints and thus preserves whether exactly
    one endpoint belongs to $R$.
\end{proof}
