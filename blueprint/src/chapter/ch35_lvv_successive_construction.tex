\section{The successive viable-set construction}
\label{sec:lvv_successive_construction}

Fix a local dimension $q\ge2$ and a gap parameter $\Delta>0$.
The construction below follows the induction of \cite{Landau2015Polynomial},
with the independent trimming radius corrected as
in~\cite{gap:lvv13_trimming_radius}. The cardinality constants use the
stated constant-cut estimate and corrected net exponent
of~\cite{gap:lvv13_constant_bond}. Error reduction uses the finite Gaussian
construction of \cite[Section~5.4, Lemmas~18--19]{Huang2015GroundState}.
This permits arbitrary ground energy, unlike the uniform polynomial
estimate for the original power sampler; the distinction is explained
in~\cite{gap:lvv13_runtime_ground_energy}.
All polynomial bounds have the quantitative scope specified in those constructions.

\begin{definition}[Successive size and bond budgets]
    \label{def:lvv_successive_budgets}
    \lean{TNLean.MPS.AreaLaw.lvvSuccessiveError,
        TNLean.MPS.AreaLaw.lvvSizeTrimmingDegree,
        TNLean.MPS.AreaLaw.lvvSizeTrimmingCoefficient,
        TNLean.MPS.AreaLaw.lvvSuccessiveCardBound,
        TNLean.MPS.AreaLaw.lvvSuccessiveBondBound,
        TNLean.MPS.AreaLaw.lvvSuccessiveError_pos}
    \leanok
    \uses{thm:lvv_size_bond_trimmed_family, def:huang_filter_parameters}
    Put $c_\Delta=(\Delta/169)^2>0$. Let $D$ be the constant-cut bond bound
    used in size trimming at precision $c_\Delta$, and define
    \begin{align}
        p   & =2(qD)^2,                                           &
        A   & =\left\lceil\frac{2(qD)^2}{c_\Delta}\right\rceil_+,   \\
        C_s & =(D+1)D(2A+1)^p,                                    &
        m_L & =(48L)^2r_{\rm ref}(L).
    \end{align}
    Here $r_{\rm ref}(L)$ is the reference bond from the existing constant-accuracy
    MPS approximation at length $48L$. Write $E_L$ and $R_L$ for the energy count
    and finite-filter rank budget with precision parameter $c_\Delta$. Set
    \begin{align}
        s_L=C_sL^pE_LR_L,\qquad b_L=R_L\max\{1,m_L\}.
    \end{align}
\end{definition}

\begin{lemma}[Uniform polynomial budgets]
    \label{thm:lvv_successive_polynomial_budgets}
    \lean{TNLean.MPS.AreaLaw.exists_lvvSuccessive_uniform_polynomial}
    \leanok
    \uses{def:lvv_successive_budgets, thm:huang_filter_uniform_polynomial,
        lem:lvv_bond_trimming_radius_polynomial}
    There are natural numbers $P,d$, with $P\ge1$, depending only on $q,\Delta$
    and chosen before the length, such that for every $L\ge1$,
    \begin{align}
        \max\{1,s_L\}    & \le P(L+1)^d, &
        \max\{1,b_L\}    & \le P(L+1)^d, &
        s_L\max\{1,b_L\} & \le P(L+1)^d,
    \end{align}
    and the factor budget $E_LR_L$ and the rank budget $R_L$ of one step are
    at most $P(L+1)^d$ as well.
\end{lemma}
\begin{proof}
    \leanok
    The energy count and rank product have a uniform natural-coefficient
    polynomial bound. So does the corrected reference trimming radius, after
    taking a natural ceiling of its coefficient. Multiply these bounds by the
    fixed size coefficient and the power $L^p$. Products of the resulting two
    polynomial bounds control both budgets and their final-span product. Taking
    maxima with one also covers the scalar initial family.
\end{proof}

\begin{definition}[One step of the sweep with its selections named]
    \label{def:lvv_successive_step}
    \lean{TNLean.MPS.AreaLaw.successiveStep,
        TNLean.MPS.AreaLaw.card_successiveStep_le,
        TNLean.MPS.AreaLaw.isLeftOpenMPS_of_mem_successiveStep}
    \leanok
    \uses{def:lvv_bond_trimmed_family, def:lvv_error_reduction_family}
    Let $i\le L$, let $m$ be a trimming radius, let $(A_j)_{j\in J}$ be a
    finite family of left factors at the cut after $i$ sites, and let $T_0$
    be a finite family of left vectors at that cut. The step applied to
    $T_0$ is the family
    \begin{align}
        \{A_j\,\mathcal T_m(u) : j\in J,\ u\in T_0\}
    \end{align}
    of the factors applied to the independent trimmings of
    Definition~\ref{def:lvv_bond_trimmed_family}. It has at most $|J||T_0|$
    members, and its members have bond bound $D\max\{1,m\}$ when every
    factor has operator-Schmidt rank at most $D$ at every internal left cut.
    In the algorithm, $T_0$ is the size-trimming selection of
    Definition~\ref{def:lvv_size_trimming_selection} applied to the
    extension $S\otimes\{|1\rangle,\ldots,|q\rangle\}$ of the family
    $S$ at the previous cut, and the factors are those of the Gaussian
    filters, indexed by themselves, or those of the sampled projector.
\end{definition}

\begin{definition}[One step of the sweep]
    \label{def:lvv_sweep_step}
    \lean{TNLean.MPS.AreaLaw.lvvSweepStep,
        TNLean.MPS.AreaLaw.card_lvvSweepStep_le,
        TNLean.MPS.AreaLaw.isLeftOpenMPS_of_mem_lvvSweepStep}
    \leanok
    \uses{def:lvv_successive_budgets, def:lvv_successive_step,
        def:lvv_size_trimming_selection, def:huang_gaussian_filter_factors,
        thm:huang_positive_chain_operator}
    Let $L\ge1$ and $i+1\le L$. Let $H=\sum_{j=0}^{L-2}h_j$, where $h_j$ is
    supported on sites $j,j+1$ and $0\le h_j\le I$. The step of the sweep
    sends a family $S$ on the first $i$ sites to the step of
    Definition~\ref{def:lvv_successive_step}, with the Gaussian filter factors
    of Definition~\ref{def:huang_gaussian_filter_factors} at precision
    $c_\Delta$ and the radius $m_L$, applied to the size-trimming selection
    (Definition~\ref{def:lvv_size_trimming_selection}, for the exact
    adjacent-pair embedding of $H$) of $S\otimes\{|1\rangle,\ldots,|q\rangle\}$.
    Its output has at most $s_L$ members, each with effective MPS bond bound
    $b_L$, and is a function of $S$, the interaction and the gap parameter.
\end{definition}

\begin{lemma}[One step of the sweep preserves viability]
    \label{thm:lvv_complete_successive_step_factors}
    \lean{TNLean.MPS.AreaLaw.isViable_lvvSweepStep}
    \leanok
    \uses{def:lvv_sweep_step, thm:lvv_size_trimmed_family_bond_trim_set,
        thm:huang_positive_chain_error_reduction_factors, thm:lvv_viable_extension}
    In the setting of Definition~\ref{def:lvv_sweep_step}, for every unit
    ground vector $\Gamma$ with projector gap $\Delta$,
    $(i,c_\Delta/L)$-viability of $S$ implies $(i+1,c_\Delta/L)$-viability
    of the step of the sweep applied to $S$.
\end{lemma}
\begin{proof}
    \leanok
    Extension preserves viability. By
    Lemma~\ref{thm:lvv_size_trimmed_family_bond_trim_set}, applied with the
    exact adjacent-pair embedding of $H$, the trimmed size-trimming selection
    of the extended family is half-viable. The Gaussian filter factors then
    restore viability precision $c_\Delta/L$ on the family of factors applied
    to the trimmed family, which is the step.
\end{proof}

\begin{lemma}[A complete successive step]
    \label{thm:lvv_complete_successive_step}
    \lean{TNLean.MPS.AreaLaw.exists_positive_chain_successive_step}
    \leanok
    \uses{thm:lvv_complete_successive_step_factors}
    Let $L\ge1$ and $1\le i\le L$. Let
    $H=\sum_{j=0}^{L-2}h_j$, where $h_j$ is supported on sites $j,j+1$ and
    $0\le h_j\le I$. Given a finite family $S$ on the first $i-1$ sites whose
    members have a common finite MPS bond bound, one can choose a family $T$
    on the first $i$ sites with $|T|\le s_L$ and individual effective bond
    bound $b_L$, before any ground energy or ground vector is supplied.
    For every unit ground vector $\Gamma$ with projector gap $\Delta$,
    $(i-1,c_\Delta/L)$-viability of $S$ implies $(i,c_\Delta/L)$-viability of $T$.
    This is the existence form of
    Lemma~\ref{thm:lvv_complete_successive_step_factors}, with the step
    left unnamed.
\end{lemma}
\begin{proof}
    \leanok
    Take $T$ to be the step of
    Lemma~\ref{thm:lvv_complete_successive_step_factors}. Its cardinality
    is at most $E_LR_L$ times the polynomial size of the selection, and its
    individual bond multiplier is $R_L$, by
    Definition~\ref{def:lvv_successive_step}.
\end{proof}

\begin{lemma}[One recursively chosen viable sequence]
    \label{thm:lvv_coherent_successive_sequence}
    \lean{TNLean.MPS.AreaLaw.exists_positive_chain_successive_viable_sets}
    \leanok
    \uses{def:lvv_sweep, thm:lvv_sweep_spec}
    Under the physical hypotheses of Definition~\ref{def:lvv_sweep_step},
    there is one sequence $S_0,\ldots,S_L$, chosen before every ground pair,
    with $S_0$ the scalar singleton, in which each $S_{i+1}$ is the step of
    the sweep applied to $S_i$. For each positive cut, $|S_i|\le s_L$ and
    every member has effective MPS bond bound $b_L$. For every unit ground
    vector $\Gamma$ with projector gap $\Delta$, all these same families are
    $(i,c_\Delta/L)$-viable. This is the existence form of
    Definition~\ref{def:lvv_sweep} and Lemma~\ref{thm:lvv_sweep_spec}.
\end{lemma}
\begin{proof}
    \leanok
    Take the sweep.
\end{proof}

\begin{theorem}[Uniform polynomial successive families and final-span MPS]
    \label{thm:lvv_uniform_successive_approximation}
    \lean{TNLean.MPS.AreaLaw.exists_uniform_positive_chain_successive_approximation}
    \leanok
    \uses{thm:lvv_successive_polynomial_budgets, thm:lvv_sweep_spec,
        def:huang_gaussian_filter_factors, thm:lvv_left_span_bond}
    Fix $q\ge2$ and $\Delta>0$. There exist natural numbers $P,d$, with $P\ge1$,
    chosen before $L$ and the interaction, such that the following holds.
    For every $L\ge1$ and every adjacent-pair positive-contraction interaction,
    the Gaussian filter factors of
    Definition~\ref{def:huang_gaussian_filter_factors} at every cut have
    cardinality and operator-Schmidt rank at most $P(L+1)^d$, and the sweep
    $S_0,\ldots,S_L$ of Definition~\ref{def:lvv_sweep} has cardinalities and
    individual effective bond bounds at most $P(L+1)^d$; both are functions
    of the interaction, chosen before every ground pair.

    Suppose $\|\Gamma\|=1$, $H\Gamma=E_0\Gamma$ and
    $H-E_0I\ge\Delta(I-|\Gamma\rangle\langle\Gamma|)$. Then every $S_i$ is
    $(i,c_\Delta/L)$-viable, and there is a vector $\psi$ in the span of the
    actual final family, read on the full chain, such that
    \begin{align}
        \|\psi\|=1,\qquad
        |\langle\psi,\Gamma\rangle|\ge1-c_\Delta/L.
    \end{align}
    This $\psi$ is an open-boundary MPS with bond dimension at most $P(L+1)^d$.
\end{theorem}
\begin{proof}
    \leanok
    Choose the polynomial budgets first. Enlarge the explicit budgets of the
    sweep and of the filter factors to this polynomial at each cut.
    At the final cut, the right configuration space is one-dimensional. Thus
    a normalized viable witness belongs to the span of the actual $S_L$ in
    full-left coordinates. Pull it back by the coordinate isometry, preserving
    norm and overlap. The finite-span rank estimate bounds its MPS bond by
    $s_L\max\{1,b_L\}$, already controlled by the same polynomial.
\end{proof}

The conclusion states the overlap precision $1-c_\Delta/L$ and the displayed
polynomial existence bounds. It makes no computational running-time assertion.
