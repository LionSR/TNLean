\section{Cell and angular information from the original spectral gap}
\label{sec:peps_angular_information_from_gap}

\begin{theorem}[Cell information from the original spectral gap]
    \label{thm:peps_cell_conditional_information_from_gap}
    \lean{TNLean.PEPS.Approximation.exists_cell_conditionalMutualInformation_le_square}
    \leanok
    \uses{def:polynomial_peps_model,def:area_law_ground_entropy}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$, and $c>0$.
    In \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}, there exist constants $C_1>0$ and $K\ge0$ such that for every $s\ge1$, $o\in\mathbb R^2$, $Q_i=o+si+[0,s)^2$ ($i\in\mathbb Z^2$), $i_0\in\mathbb Z^2$, $A\subseteq\mathbb Z^2\setminus\{i_0\}$, square-grid Hamiltonian $H$ on $\Lambda_L=\{1,\ldots,L\}^2$ with interaction bound $J$, unit ground vector $\Omega$ with full-system gap $\Delta$, $G=\Lambda_L\cap Q_{i_0}$, $B=\Lambda_L\cap\bigcup_{i\in A}Q_i$, and $E\subseteq\Lambda_L\setminus(G\cup B)$ with $\operatorname{dist}_\infty(\overline{Q_{i_0}},E)\ge cs$,
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_1(1+\log s)^K.
        \label{eq:peps_cell_cmi_from_gap}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,
        thm:peps_square_cell_conditional_information_of_collar}
    Taking $R=1$, $D_0=13>2R+10$, and $C_{\rm tpl}=24$ in Theorem~\ref{thm:area_law_sublinear_collar} and applying Theorem~\ref{thm:peps_square_cell_conditional_information_of_collar} yields (\ref{eq:peps_cell_cmi_from_gap}).
\end{proof}

\begin{theorem}[Information under angular separation]
    \label{thm:peps_angular_information_from_gap}
    \lean{TNLean.PEPS.Approximation.angularInformationBound}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_angular_separation}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, and $\Delta>0$.
    In~\cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, for every $a>0$, $A_0\ge1$, $m_0\in\mathbb N$, and $p>0$, there exist $D,L_0\in\mathbb R$ such that for every integer $L\ge L_0$, square-grid Hamiltonian $H$ on $\Lambda_L=\{1,\ldots,L\}^2$ with interaction bound $J$, unit ground vector $\Omega$ with full-system gap $\Delta$, set $\mathcal V\subseteq\mathbb R^2$ with $|\mathcal V|\le m_0$, disjoint $X,E\subseteq\Lambda_L$, and $n\in[D\log L,A_0L]$ with $\operatorname{diam}_\infty X\le A_0n$, setting $d_{\mathcal V}(x)=\min_{v\in\mathcal V}|x-v|_\infty$ ($d_{\varnothing}=+\infty$), if
    \begin{align}
        \forall x\in X,\ \forall y\in E,\qquad
        |x-y|_\infty & \ge a\max\{D\log L,\min(n,d_{\mathcal V}(x))\},
        \label{eq:peps_angular_sep_cond}
    \end{align}
    then $I_\Omega(X:E)\le L^{-p}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,thm:peps_angular_information_of_collar}
    Combine Theorem~\ref{thm:area_law_sublinear_collar} at $(R,D_0,C_{\rm tpl})=(1,13,24)$ with Theorem~\ref{thm:peps_angular_information_of_collar}.
\end{proof}

\begin{theorem}[Cell and angular information at closed-form constants]
    \label{thm:peps_information_from_gap_closed}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_closed,
        TNLean.PEPS.Approximation.isAngularScale_closed_le}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_angular_scale,
        def:peps_closed_information_constants}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, and $\Delta>0$. For every $c>0$, Theorem~\ref{thm:peps_cell_conditional_information_from_gap} holds with $(C_1,K)=(C_{\rm cell}(q,1,J,\Delta,c),K_{\rm cell}(q,1,J,\Delta,c))$. For $a>0$, $A_0\in\mathbb R$, $m_0\in\mathbb N$, and integer $p>0$, every $D\ge D_{\rm ang}(J,\Delta,a,p)$ and $L_0\ge L_{\rm ang}(q,J,\Delta,a,A_0,m_0,p)$ form an angular scale at accuracy $L^{-p}$ in the sense of Definition~\ref{def:peps_angular_scale}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_collar_closed,
        thm:peps_information_of_collar_named_amplification}
    Combine Theorem~\ref{thm:peps_collar_closed} at $(R,D_0,C_{\rm tpl})=(1,13,24)$ with Theorem~\ref{thm:peps_information_of_collar_named_amplification}.
\end{proof}

    \label{thm:peps_cell_information_range_closed}
    \lean{TNLean.PEPS.Approximation.cellInformationConstantClosed_pos,

\end{proof}

\subsection{Angular information at interaction range $R$}

This subsection extends \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS} from $R=1$ to general interaction range $R$ by running the multi-scale argument for any state family satisfying the cell bound (\ref{eq:peps_cell_cmi_from_gap}) and the information reset of Theorem~\ref{thm:peps_information_reset_explicit}.

\begin{definition}[The cell and reset estimates for a family of states]
    \label{def:peps_family_cell_reset_bounds}
    \lean{TNLean.PEPS.Approximation.SquareCellInformationBound,
        TNLean.PEPS.Approximation.SquareResetBound}
    \leanok
    \uses{def:peps_angular_grid,thm:peps_cell_conditional_information_from_gap,
        thm:peps_information_reset_explicit}
    Fix $q\in\mathbb N$. A family $\mathcal G$ of states on $\Lambda_L=\{1,\ldots,L\}^2$ with local dimension $q$ satisfies:
    \begin{enumerate}
        \item the \emph{cell bound} with $(c,C_1,K)$ if every $\Omega\in\mathcal G$ satisfies (\ref{eq:peps_cell_cmi_from_gap}) of Theorem~\ref{thm:peps_cell_conditional_information_from_gap};
        \item the \emph{reset bound} with $(C_p,L_0)$ at accuracy $L^{-p}$ if for every $L\ge L_0$, every $\Omega\in\mathcal G$ satisfies Theorem~\ref{thm:peps_information_reset_explicit} with width $C_p$.
    \end{enumerate}
\end{definition}

\begin{theorem}[Angular information for a family with the two estimates]
    \label{thm:peps_angular_information_of_bounds}
    \lean{TNLean.PEPS.Approximation.angular_information_reset_of_squareResetBound,
        TNLean.PEPS.Approximation.angular_single_scale_reset_of_bounds,
        TNLean.PEPS.Approximation.angularInformation_of_bounds,
        TNLean.PEPS.Approximation.isAngularScale_of_bounds}
    \leanok
    \uses{def:peps_family_cell_reset_bounds,def:peps_angular_scale,
        def:peps_angular_information_constants,
        def:peps_angular_cell_reset_constants,
        def:peps_angular_single_scale_constants}
    Let $q\in\mathbb N$, $a>0$, $A_0\in\mathbb R$, $m_0\in\mathbb N$, and integer $p>0$. With $a'=a'(a)$, $d=d(a)$, and $N_{\rm cell}(a,A_0,m_0)$ from Definition~\ref{def:peps_angular_information_constants}, $c_{\rm ang}(a,a',d)$ from Definition~\ref{def:peps_angular_single_scale_constants}, and $(D_{\rm cell},L_{\rm cell})$ from Definition~\ref{def:peps_angular_cell_reset_constants}, if a family $\mathcal G$ of unit vectors satisfies the cell bound with $(c_{\rm ang}(a,a',d),C_1,K)$ ($C_1>0,K\ge0$) and the reset bound with $(C_p,L_0)$ ($C_p>0$), then every $D\ge D_{\rm cell}(C_p,a',d)$ and
    \begin{align}
        L_1 & \ge \max\bigl\{L_{\rm cell}\bigl(a',d,C_p,C_1,N_{\rm cell}(a,A_0,m_0),K\bigr),
        L_0\bigr\}
        \label{eq:peps_angular_family_threshold}
    \end{align}
    form an angular scale for $\mathcal G$ (Definition~\ref{def:peps_angular_scale}), extending \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_scaled_cell_clearance,thm:peps_annular_information_induction,
        thm:peps_annular_cutoff,thm:peps_annular_separation,
        thm:peps_annular_cell_clearance,
        thm:peps_selected_region_information_of_cell_bounds,
        thm:peps_angular_reset_width_of_cell_information_explicit,
        lem:peps_angular_scale_of_site_position,
        def:peps_angular_single_scale_constants}
    Set $M=N_{\rm cell}(a,A_0,m_0)$; for $D\ge D_{\rm cell}(C_p,a',d)\ge1$ and $L\ge L_1\ge e$, $t=D\log L\in[1,n]$ gives dyadic scales $s_j=2^{-j}n\ge t$ ($0\le j\le m$) with $t\le s_m<2t$ (Theorem~\ref{thm:peps_annular_cutoff}). At scale $s_j$, the cumulative processed set $A_j$ adds at most $M$ cells with $\operatorname{dist}_\infty(A_j,E)\ge as_j/2$ (Theorem~\ref{thm:peps_annular_separation}), and combining the cell bound at $c_{\rm ang}(a,a',d)$ (Theorem~\ref{thm:peps_selected_region_information_of_cell_bounds}) with the reset bound of width $C_p$ (Theorem~\ref{thm:peps_angular_reset_width_of_cell_information_explicit}) restores
    \begin{align}
        I_\Omega\bigl(\Lambda_L\cap A_j^{+a's_j}:
        \Lambda_L\cap E^{+a's_j}\bigr) & \le L^{-p}
        \notag
    \end{align}
    inductively by Theorem~\ref{thm:peps_annular_information_induction}. At $j=m$, $A_m=X$, and monotonicity of mutual information gives $I_\Omega(X:E)\le L^{-p}$; Lemma~\ref{lem:peps_angular_scale_of_site_position} extends this to arbitrary common translations of $\Lambda_L$.
\end{proof}

\begin{theorem}[Information of a state of the square on the integer square]
    \label{thm:peps_square_state_relabel_information}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_of_rectangle,
        TNLean.PEPS.Approximation.mutualInformation_squareAreaLawState}
    \leanok
    \uses{def:square_area_law_state,def:peps_angular_grid}
    For a state $\Omega$ on $\{0,\ldots,L-1\}^2$ and its placement $\tilde\Omega$ on $\Lambda_L=\{1,\ldots,L\}^2$ (Definition~\ref{def:square_area_law_state}), mutual and conditional mutual informations match under the shift $x\mapsto x+(1,1)$, so a cell bound for $\tilde\Omega$ at origin $o-(1,1)$ transfers to $\Omega$ at origin $o$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:finite_product_entropy_relabel}
    The translation $x\mapsto x+(1,1)$ is a site bijection preserving $\ell_\infty$-distances, cells, and regional entropies by Theorem~\ref{thm:finite_product_entropy_relabel}.
\end{proof}

\begin{definition}[The angular constants at range $R$]
    \label{def:peps_angular_constants_range}
    \lean{TNLean.PEPS.Approximation.rangeAngularScale,
        TNLean.PEPS.Approximation.rangeAngularThreshold,
        TNLean.PEPS.Approximation.rangeAngularScale_eq}
    \leanok
    \uses{def:peps_reset_width_range,def:peps_closed_information_constants,
        def:peps_angular_information_constants,
        def:peps_angular_cell_reset_constants,
        def:peps_angular_single_scale_constants}
    For $q,R,m_0\in\mathbb N$ and $J,\Delta,a,A_0,p\in\mathbb R$, with $a'=a'(a)$, $d=d(a)$, $N_{\rm cell}(a,A_0,m_0)$ from Definition~\ref{def:peps_angular_information_constants}, $c=c_{\rm ang}(a,a',d)$ from Definition~\ref{def:peps_angular_single_scale_constants}, $C_{p,R}=C_{p,R}(J,\Delta,p)$ from Definition~\ref{def:peps_reset_width_range}, and $(D_{\rm cell},L_{\rm cell})$ from Definition~\ref{def:peps_angular_cell_reset_constants}, set
    \begin{align}
        D_R(J,\Delta,a,p)           & =D_{\rm cell}(C_{p,R},a',d), \notag         \\
        L_R(q,J,\Delta,a,A_0,m_0,p) & =\max\bigl\{L_{\rm cell}\bigl(a',d,C_{p,R},
        C_{\rm cell}(q,R,J,\Delta,c),N_{\rm cell}(a,A_0,m_0),
        K_{\rm cell}(q,R,J,\Delta,c)\bigr),L_{0,R}(q,J,\Delta,p)\bigr\}.
        \label{eq:peps_angular_constants_range}
    \end{align}
    For $J\ge0$, $\Delta>0$, $a>0$, and $p>0$, since $C_{p,R}\ge1$,
    \begin{align}
        D_R(J,\Delta,a,p) & =\frac{2400\,C_{p,R}(J,\Delta,p)}{\min\{a,1\}}.
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Angular information at range $R$, at closed-form constants]
    \label{thm:peps_angular_information_range_closed}
    \lean{TNLean.PEPS.Approximation.isAngularScale_range_closed_le}

\end{theorem}
\begin{proof}
    \leanok

        thm:peps_range_cell_reset_bounds,thm:peps_cell_information_range_closed,

\end{proof}
