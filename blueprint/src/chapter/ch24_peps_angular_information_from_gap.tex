\section{Cell and angular information from the original spectral gap}
\label{sec:peps_angular_information_from_gap}

\begin{theorem}[Cell information from the original spectral gap]
    \label{thm:peps_cell_conditional_information_from_gap}
    \lean{TNLean.PEPS.Approximation.exists_cell_conditionalMutualInformation_le_square}
    \leanok
    \uses{def:polynomial_peps_model,def:area_law_ground_entropy}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$ and $c>0$.
    In the cell estimate of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS},
    there are constants $C_1>0$ and $K\ge0$, chosen before the scale, grid origin,
    conditioning cells, system size, Hamiltonian, ground vector and exterior region,
    with the following property.
    Let $s\ge1$, let $o\in\mathbb R^2$, and put
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Fix $i_0\in\mathbb Z^2$ and a set $A\subseteq\mathbb Z^2$ with $i_0\notin A$.
    Let $H$ be a Hamiltonian of the square-grid model on
    $\Lambda_L=\{1,\ldots,L\}^2$, with interaction bound $J$, and let $\Omega$
    be its unit ground vector with full-system gap $\Delta$.
    Set $G=\Lambda_L\cap Q_{i_0}$ and
    $B=\Lambda_L\cap\bigcup_{i\in A}Q_i$.
    For every $E\subseteq\Lambda_L\setminus(G\cup B)$ satisfying
    $\operatorname{dist}_\infty(\overline{Q_{i_0}},E)\ge cs$, one has
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_1(1+\log s)^K.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,
        thm:peps_square_cell_conditional_information_of_collar}
    With range $R=1$, clearance $D_0=13$ and template constant $C_{\rm tpl}=24$,
    the inequalities $2R+10=12<13$ and $24\le C_{\rm tpl}$ satisfy the hypotheses
    of the sublinear collar theorem, so the conditional square-cell estimate
    applies to the same ground vector, cell, conditioning region and exterior.
\end{proof}

\begin{theorem}[Information under angular separation]
    \label{thm:peps_angular_information_from_gap}
    \lean{TNLean.PEPS.Approximation.angularInformationBound}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_angular_separation}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$ and $\Delta>0$.
    In~\cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, for every $a>0$,
    $A_0\ge1$, $m_0\in\mathbb N$ and $p>0$, there are constants
    $D,L_0\in\mathbb R$, chosen before the system size, Hamiltonian, ground
    vector, marked points, physical regions and scale, with the following property.
    Let $L\ge L_0$ be a natural number, let $H$ be a Hamiltonian of the
    square-grid model on $\Lambda_L=\{1,\ldots,L\}^2$ with interaction bound $J$,
    and let $\Omega$ be its unit ground vector with full-system gap $\Delta$.
    Let $\mathcal V\subseteq\mathbb R^2$ consist of at most $m_0$ marked points,
    let $X,E\subseteq\Lambda_L$ be disjoint, and let $n\in\mathbb R$ satisfy
    $D\log L\le n\le A_0L$ and $\operatorname{diam}_\infty X\le A_0n$.
    Put $d_{\mathcal V}(x)=\min_{v\in\mathcal V}|x-v|_\infty$, with
    $d_{\varnothing}=+\infty$.
    If every $x\in X$ and $y\in E$ satisfy
    \begin{align}
        |x-y|_\infty & \ge a\max\{D\log L,\min(n,d_{\mathcal V}(x))\},
        \notag
    \end{align}
    then $I_\Omega(X:E)\le L^{-p}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,thm:peps_angular_information_of_collar}
    The sublinear collar theorem with $R=1$, $D_0=13$ and $C_{\rm tpl}=24$
    follows from the original model and gap assumptions, and applying the conditional
    angular information theorem with these fixed parameters yields constants
    with the stated order of quantifiers.
\end{proof}

\begin{definition}[Closed-form cell and angular constants]
    \label{def:peps_closed_information_constants}
    \lean{TNLean.PEPS.Approximation.cellInformationConstantClosed,
        TNLean.PEPS.Approximation.cellInformationExponentClosed,
        TNLean.PEPS.Approximation.angularThresholdClosed}
    \leanok
    \uses{def:peps_closed_collar_constants,def:peps_cell_information_constants,
        def:peps_angular_information_constants,
        def:peps_collar_amplification_constants}
    For $q,R\in\mathbb N$ and $J,\Delta\in\mathbb R$, let $C=C(q,R,J,\Delta)$
    and $N=N(q,R,J,\Delta)$ be the closed-form collar constant and threshold of
    Definition~\ref{def:peps_closed_collar_constants}. With
    $C_a=C_a(q,R,J,\Delta,C)$ and $N_a=N_a(q,R,J,\Delta,C)$, let
    $C_{\rm cell}(q,R,J,\Delta,c)$ and $K_{\rm cell}(q,R,J,\Delta,c)$ be the cell
    constants at the template constant $24$, the threshold $N$ and $(C_a,N_a)$.
    Let $L_{\rm ang}(q,J,\Delta,a,A_0,m_0,p)$ be the angular size threshold at
    the same data for $R=1$. The source has only site and nearest-neighbour
    terms, which is range one; ranges $R>1$ extend it.
\end{definition}

\begin{theorem}[Cell and angular information at closed-form constants]
    \label{thm:peps_information_from_gap_closed}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_closed,
        TNLean.PEPS.Approximation.isAngularScale_closed_le}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_angular_scale,
        def:peps_closed_information_constants}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$ and $\Delta>0$. For every
    $c>0$, the conclusion of
    Theorem~\ref{thm:peps_cell_conditional_information_from_gap} holds with
    $C_1=C_{\rm cell}(q,1,J,\Delta,c)$ and $K=K_{\rm cell}(q,1,J,\Delta,c)$. For
    $a>0$, $A_0\in\mathbb R$, $m_0\in\mathbb N$ and an integer $p>0$, every
    $D\ge D_{\rm ang}(J,\Delta,a,p)$ and every
    $L_0\ge L_{\rm ang}(q,J,\Delta,a,A_0,m_0,p)$ form a slope and a threshold at
    accuracy $L^{-p}$ in the sense of Definition~\ref{def:peps_angular_scale}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_collar_closed,
        thm:peps_information_of_collar_named_amplification}
    Theorem~\ref{thm:peps_collar_closed} with $R=1$ and $D_0=13$ gives the
    collar bound with template constant $24$, constant $C$ and threshold $N$. The constants $C_{\rm cell}$ and $K_{\rm cell}$ are positive, and
    Theorem~\ref{thm:peps_information_of_collar_named_amplification} gives the
    remaining conclusions.
\end{proof}

\begin{theorem}[Cell information at range $R$ and closed-form constants]
    \label{thm:peps_cell_information_range_closed}
    \lean{TNLean.PEPS.Approximation.cellInformationConstantClosed_pos,
        TNLean.PEPS.Approximation.cellInformationExponentClosed_pos,
        TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_closed}
    \leanok
    \uses{def:area_law_hamiltonian,def:area_law_ground_entropy,
        def:peps_closed_information_constants,
        thm:peps_cell_conditional_information_of_collar_explicit}
    Fix $q,R\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$ and $c>0$. The
    constants $C_{\rm cell}(q,R,J,\Delta,c)$ and $K_{\rm cell}(q,R,J,\Delta,c)$
    are positive, and the conclusion of
    Theorem~\ref{thm:peps_cell_conditional_information_of_collar_explicit} holds
    with these constants for every Hamiltonian of range $R$ with interaction bound
    $J$ on a finite integer domain and every unit ground vector with full-system
    gap $\Delta$, without a collar or amplification hypothesis. The source has
    only site and nearest-neighbour terms, which is range one; ranges $R>1$ extend
    it.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_collar_closed,thm:peps_collar_amplification_bound,
        thm:peps_cell_information_constants_pos,
        thm:peps_cell_conditional_information_of_collar_explicit}
    The cell constants are positive for all values of the parameters.
    Theorem~\ref{thm:peps_collar_closed} with clearance $D_0=2R+11>2R+10$ gives
    the collar bound at range $R$ with template constant $24$, constant $C$ and
    threshold $N$, and Theorem~\ref{thm:peps_collar_amplification_bound} gives
    the amplification bound at $(\max\{C,1\},1)$ with constant $C_a$ and
    threshold $N_a$. Theorem~\ref{thm:peps_cell_conditional_information_of_collar_explicit}
    then gives the bound.
\end{proof}

\subsection{Angular information at interaction range $R$}

The source states Proposition~3.4 for site and nearest-neighbour terms only,
which is interaction range one. The proof uses the Hamiltonian only through
the cell bound of Lemma~3.2 and the information reset of Lemma~3.3 for the
states concerned. This subsection runs the proof for any family of states
with these two estimates, supplies them at range $R$, and so extends the
proposition to Hamiltonians of range $R$ on the square; the range-one
statements above are unchanged.

\begin{definition}[The cell and reset estimates for a family of states]
    \label{def:peps_family_cell_reset_bounds}
    \lean{TNLean.PEPS.Approximation.SquareCellInformationBound,
        TNLean.PEPS.Approximation.SquareResetBound}
    \leanok
    \uses{def:peps_angular_scale}
    Let $\mathcal G$ assign to each $L$ a set of states of the $L\times L$
    square. It satisfies the \emph{cell bound} with clearance coefficient $c$,
    prefactor $C_1$ and exponent $K$ if, for every state of $\mathcal G$, the
    conclusion of Theorem~\ref{thm:peps_cell_conditional_information_from_gap}
    holds with these constants. It satisfies the \emph{reset bound} with width
    $C_p$ and threshold $L_0$ at accuracy $L^{-p}$ if, for every $L\ge L_0$ and
    every state of $\mathcal G$, the conclusion of
    Theorem~\ref{thm:peps_information_reset_explicit} holds with the width
    $C_p$.
\end{definition}

\begin{theorem}[Proposition~3.4 for a family with the two estimates]
    \label{thm:peps_angular_information_of_bounds}
    \lean{TNLean.PEPS.Approximation.angular_information_reset_of_squareResetBound,
        TNLean.PEPS.Approximation.angular_single_scale_reset_of_bounds,
        TNLean.PEPS.Approximation.angularInformation_of_bounds,
        TNLean.PEPS.Approximation.isAngularScale_of_bounds,
        TNLean.PEPS.Approximation.IsSquareGroundState.squareResetBound}
    \leanok
    \uses{def:peps_family_cell_reset_bounds,def:peps_angular_scale,
        def:peps_angular_information_constants}
    Let $a>0$ and let $p>0$ be an integer. Let $\mathcal G$ consist of unit
    vectors and satisfy the cell bound with
    $C_1>0$ and $K\ge0$ at the clearance coefficient of $a'$ and $d$, and the
    reset bound with $C_p>0$ and $L_0$, where $a'$ and $d$ are the margin and
    the cell side of Definition~\ref{def:peps_angular_information_constants}.
    Then every slope $D\ge\max\{1,12C_p/a',1/d\}$ and every threshold at least
    the larger of the cell-reset threshold at $(C_p,C_1,K)$ and the cell count,
    and of $L_0$, form an angular scale for $\mathcal G$ in the sense of
    Definition~\ref{def:peps_angular_scale}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_scaled_cell_clearance,thm:peps_annular_information_induction,
        thm:peps_annular_cutoff,thm:peps_annular_separation,
        lem:peps_angular_scale_of_site_position,
        def:peps_angular_single_scale_constants}
    Run the proof of
    Theorem~\ref{thm:peps_angular_information_of_collar_explicit}. At one
    scale, the cell clearance and the cell bound control the information of
    the selected cells, and the reset bound with width $C_p$ restores the
    accuracy $L^{-p}$ on the cores. The dyadic cutoff, the annular separation
    and the induction over the layers then give the bound at every scale, and
    translating the marked points with the sites gives the slope and the
    threshold. The Hamiltonian enters only through the two estimates.
\end{proof}

\begin{theorem}[The two estimates at range $R$]
    \label{thm:peps_range_cell_reset_bounds}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_of_rectangle,
        TNLean.PEPS.Approximation.IsRangeSquareGroundState.squareCellInformationBound,
        TNLean.PEPS.Approximation.mutualInformation_squareAreaLawState,
        TNLean.PEPS.Approximation.IsRangeSquareGroundState.squareResetBound}
    \leanok
    \uses{def:peps_family_cell_reset_bounds,def:square_range_ground_states,
        def:peps_closed_information_constants,def:peps_reset_width_range}
    Fix $q\ge1$, $R\ge0$, $J\ge0$ and $\Delta>0$. The unit gapped ground vectors
    of Hamiltonians of range $R$ with term bound $J$ on the square satisfy the
    cell bound at every $c>0$ with $C_1=C_{\rm cell}(q,R,J,\Delta,c)$ and
    $K=K_{\rm cell}(q,R,J,\Delta,c)$, and for every $p>0$ the reset bound with
    width $C_{p,R}(J,\Delta)$ and threshold $L_{0,R}(q,J,\Delta,p)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_cell_information_range_closed,
        thm:peps_information_reset_range}
    Place the state on the integer square. A common translation by $(1,1)$
    identifies the two placements of the sites, so
    Theorem~\ref{thm:peps_cell_information_range_closed} gives the cell bound,
    and Theorem~\ref{thm:peps_information_reset_range} gives the reset; the
    entropies are unchanged by the relabelling.
\end{proof}

\begin{definition}[The angular constants at range $R$]
    \label{def:peps_angular_constants_range}
    \lean{TNLean.PEPS.Approximation.rangeAngularScale,
        TNLean.PEPS.Approximation.rangeAngularThreshold,
        TNLean.PEPS.Approximation.rangeAngularScale_eq}
    \leanok
    \uses{def:peps_reset_width_range,def:peps_closed_information_constants,
        def:peps_angular_information_constants}
    With $a'$, $d$, the clearance coefficient $c$ and the cell count of
    Definition~\ref{def:peps_angular_information_constants}, let
    $D_R(J,\Delta,a,p)=\max\{1,12C_{p,R}/a',1/d\}$, and let
    $L_R(q,J,\Delta,a,A_0,m_0,p)$ be the larger of the cell-reset threshold at
    $C_{p,R}$, $C_{\rm cell}(q,R,J,\Delta,c)$, $K_{\rm cell}(q,R,J,\Delta,c)$ and
    the cell count, and of $L_{0,R}(q,J,\Delta,p)$. For $J\ge0$, $\Delta>0$,
    $a>0$ and $p>0$,
    \begin{align*}
        D_R(J,\Delta,a,p) & =\frac{2400\,C_{p,R}(J,\Delta)}{\min\{a,1\}}.
    \end{align*}
\end{definition}

\begin{theorem}[Proposition~3.4 at range $R$, at closed-form constants]
    \label{thm:peps_angular_information_range_closed}
    \lean{TNLean.PEPS.Approximation.isAngularScale_range_closed_le}
    \leanok
    \uses{def:peps_angular_constants_range,def:peps_angular_scale,
        def:square_range_ground_states}
    Fix $q\ge1$, $R\ge0$, $J\ge0$, $\Delta>0$, $a>0$, $m_0\in\mathbb N$ and an
    integer $p>0$. For the unit gapped ground vectors of Hamiltonians of range
    $R$ with term bound $J$ on the $L\times L$ square, every slope
    $D\ge D_R(J,\Delta,a,p)$ and every threshold
    $L_0\ge L_R(q,J,\Delta,a,A_0,m_0,p)$ form an angular scale at accuracy
    $L^{-p}$. This extends
    Theorem~\ref{thm:peps_information_from_gap_closed} from range one to range
    $R$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_angular_information_of_bounds,
        thm:peps_range_cell_reset_bounds}
    Apply Theorem~\ref{thm:peps_angular_information_of_bounds} with the
    estimates of Theorem~\ref{thm:peps_range_cell_reset_bounds}.
\end{proof}
