\section{Cell and angular information from the original spectral gap}
\label{sec:peps_angular_information_from_gap}

\begin{theorem}[Cell information from the original spectral gap]
    \label{thm:peps_cell_conditional_information_from_gap}
    \lean{TNLean.PEPS.Approximation.exists_cell_conditionalMutualInformation_le_square}
    \leanok
    \uses{def:polynomial_peps_model,def:area_law_ground_entropy}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$ and $c>0$.
    In the cell estimate of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS},
    there are constants $C_1>0$ and $K\ge0$, chosen before the scale, grid origin,
    conditioning cells, system size, Hamiltonian, ground vector and exterior region,
    with the following property.
    Let $s\ge1$, let $o\in\mathbb R^2$, and put
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Fix $i_0\in\mathbb Z^2$ and a set $A\subseteq\mathbb Z^2$ with $i_0\notin A$.
    Let $H$ be a Hamiltonian of the square-grid model on
    $\Lambda_L=\{1,\ldots,L\}^2$, with interaction bound $J$, and let $\Omega$
    be its unit ground vector with full-system gap $\Delta$.
    Set $G=\Lambda_L\cap Q_{i_0}$ and
    $B=\Lambda_L\cap\bigcup_{i\in A}Q_i$.
    For every $E\subseteq\Lambda_L\setminus(G\cup B)$ satisfying
    $\operatorname{dist}_\infty(\overline{Q_{i_0}},E)\ge cs$, one has
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_1(1+\log s)^K.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,
        thm:peps_square_cell_conditional_information_of_collar}
    With range $R=1$, clearance $D_0=13$ and template constant $C_{\rm tpl}=24$,
    the inequalities $2R+10=12<13$ and $24\le C_{\rm tpl}$ satisfy the hypotheses
    of the sublinear collar theorem, so the conditional square-cell estimate
    applies to the same ground vector, cell, conditioning region and exterior.
\end{proof}

\begin{theorem}[Information under angular separation]
    \label{thm:peps_angular_information_from_gap}
    \lean{TNLean.PEPS.Approximation.angularInformationBound}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_angular_separation}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$ and $\Delta>0$.
    In~\cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, for every $a>0$,
    $A_0\ge1$, $m_0\in\mathbb N$ and $p>0$, there are constants
    $D,L_0\in\mathbb R$, chosen before the system size, Hamiltonian, ground
    vector, marked points, physical regions and scale, with the following property.
    Let $L\ge L_0$ be a natural number, let $H$ be a Hamiltonian of the
    square-grid model on $\Lambda_L=\{1,\ldots,L\}^2$ with interaction bound $J$,
    and let $\Omega$ be its unit ground vector with full-system gap $\Delta$.
    Let $\mathcal V\subseteq\mathbb R^2$ consist of at most $m_0$ marked points,
    let $X,E\subseteq\Lambda_L$ be disjoint, and let $n\in\mathbb R$ satisfy
    $D\log L\le n\le A_0L$ and $\operatorname{diam}_\infty X\le A_0n$.
    Put $d_{\mathcal V}(x)=\min_{v\in\mathcal V}|x-v|_\infty$, with
    $d_{\varnothing}=+\infty$.
    If every $x\in X$ and $y\in E$ satisfy
    \begin{align}
        |x-y|_\infty & \ge a\max\{D\log L,\min(n,d_{\mathcal V}(x))\},
        \notag
    \end{align}
    then $I_\Omega(X:E)\le L^{-p}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar,thm:peps_angular_information_of_collar}
    The sublinear collar theorem with $R=1$, $D_0=13$ and $C_{\rm tpl}=24$
    follows from the original model and gap assumptions, and applying the conditional
    angular information theorem with these fixed parameters yields constants
    with the stated order of quantifiers.
\end{proof}

\begin{definition}[Closed-form cell and angular constants]
    \label{def:peps_closed_information_constants}
    \lean{TNLean.PEPS.Approximation.collarBoxConstantClosed,
        TNLean.PEPS.Approximation.collarConstantClosed,
        TNLean.PEPS.Approximation.collarThresholdClosed,
        TNLean.PEPS.Approximation.cellInformationConstantClosed,
        TNLean.PEPS.Approximation.cellInformationExponentClosed,
        TNLean.PEPS.Approximation.angularThresholdClosed}
    \leanok
    \uses{def:area_law_collar_bound,def:peps_cell_information_constants,
        def:peps_angular_information_constants,
        def:peps_collar_amplification_constants}
    For $q\in\mathbb N$ and $J,\Delta\in\mathbb R$, let $C_{\rm box}(q,J,\Delta)$
    be the safe-box constant of \cite[Proposition~9.5]{OpenAI2026AreaLaw} at
    range one, at the transport constants with threshold $1/8$, entropy constant
    $200000$ with exponent $2$, and energy constant
    $3(18\pi^2+6144e^{\pi^2+1})$ with exponent $4$. Let $C(q,J,\Delta)$ and
    $N(q,J,\Delta)$ be the constant and threshold of the range-one sublinear
    collar bound of \cite[Proposition~9.6]{OpenAI2026AreaLaw} at this safe-box
    constant, the truncation constant at the factor $10$ and these transport
    constants. With $C_a=C_a(q,1,J,\Delta,C)$ and
    $N_a=N_a(q,1,J,\Delta,C)$, let $C_{\rm cell}(q,J,\Delta,c)$ and
    $K_{\rm cell}(q,J,\Delta,c)$ be the cell constants at the template constant
    $24$, the threshold $N$ and $(C_a,N_a)$, and let
    $L_{\rm ang}(q,J,\Delta,a,A_0,m_0,p)$ be the angular size threshold at the
    same data.
\end{definition}

\begin{theorem}[Cell and angular information at closed-form constants]
    \label{thm:peps_information_from_gap_closed}
    \lean{TNLean.PEPS.Approximation.sublinearCollarBoundAt_closed,
        TNLean.PEPS.Approximation.cellInformationConstantClosed_pos,
        TNLean.PEPS.Approximation.cellInformationExponentClosed_pos,
        TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_closed,
        TNLean.PEPS.Approximation.isAngularScale_closed_le}
    \leanok
    \uses{def:area_law_collar_bound,def:polynomial_peps_model,def:peps_angular_scale,
        def:peps_closed_information_constants}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$ and $\Delta>0$. The range-one
    sublinear collar bound with clearance $13$ and template constant $24$ holds
    with constant $C(q,J,\Delta)$ and threshold $N(q,J,\Delta)$. For every
    $c>0$, the conclusion of
    Theorem~\ref{thm:peps_cell_conditional_information_from_gap} holds with
    $C_1=C_{\rm cell}(q,J,\Delta,c)$ and $K=K_{\rm cell}(q,J,\Delta,c)$. For
    $a>0$, $A_0\in\mathbb R$, $m_0\in\mathbb N$ and an integer $p>0$, every
    $D\ge D_{\rm ang}(J,\Delta,a,p)$ and every
    $L_0\ge L_{\rm ang}(q,J,\Delta,a,A_0,m_0,p)$ form a slope and a threshold at
    accuracy $L^{-p}$ in the sense of Definition~\ref{def:peps_angular_scale}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_sublinear_collar_explicit,
        thm:area_law_transport_bound_explicit,
        thm:peps_information_of_collar_named_amplification}
    The stated transport constants satisfy the common step-error bound by
    Theorem~\ref{thm:area_law_transport_bound_explicit}, so
    Theorem~\ref{thm:area_law_sublinear_collar_explicit} with $R=1$, $D_0=13$
    and $C_{\rm tpl}=24$ gives the collar bound with constant $C$ and threshold
    $N$. The constants $C_{\rm cell}$ and $K_{\rm cell}$ are positive, and
    Theorem~\ref{thm:peps_information_of_collar_named_amplification} gives the
    remaining conclusions.
\end{proof}
