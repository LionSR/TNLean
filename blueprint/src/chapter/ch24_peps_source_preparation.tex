\begin{theorem}[Exchanging lists of registers]
    \label{thm:peps_register_exchange}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_exchangeBlocks_appendIso_symm}
    \lean{TNLean.Lattice.PartyCircuit.Word.exchangeBlocks}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_exchangeBlocks}
    \lean{TNLean.Lattice.PartyCircuit.Word.moveHead}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_moveHead}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_moveHead_appendIso_symm}
    \lean{TNLean.Lattice.PartyCircuit.appendIso_symm_cons_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_moveHead}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_exchangeBlocks}
    \leanok
    \uses{def:peps_party_layout}
    For ordered register lists $A,B,C$ with memories $\mathcal H_A,\mathcal H_B,\mathcal H_C$, adjacent exchanges form an allowed source-free composition sending $a\otimes(b\otimes c)\mapsto b\otimes(a\otimes c)$ while preserving the internal order of each list (and moving a single register $B=(r)$ to the front).
\end{theorem}

\begin{proof}\leanok
    Let $J_{A,C}:\mathcal H_{(A,C)}\xrightarrow{\cong}\mathcal H_A\otimes\mathcal H_C$. Inducting on $A$ constructs an allowed source-free move $M_{r;A,C}$ of a single register $r$ past $A$ satisfying
    \begin{align}
        M_{r;A,C}J_{A,(r,C)}^{-1}(a\otimes(x\otimes c))
         & =x\otimes J_{A,C}^{-1}(a\otimes c),
        \label{eq:peps_register_head_move}
    \end{align}
    and inducting on $B$ by composing $M_{r;A,C}$ on the head of $B$ with the exchange of the tail yields an allowed source-free composition $X_{A,B;C}$ satisfying
    \begin{align}
        X_{A,B;C}J_{A,(B,C)}^{-1}
        \bigl(a\otimes J_{B,C}^{-1}(b\otimes c)\bigr)
         & =J_{B,(A,C)}^{-1}
        \bigl(b\otimes J_{A,C}^{-1}(a\otimes c)\bigr).
        \label{eq:peps_register_block_exchange}
    \end{align}
\end{proof}

    \leanok
    \uses{def:peps_party_layout}
    A pair source consists of distinct parties $p,q$, spaces $\mathcal U,\mathcal V$ owned by $p,q$, and a vector $\eta\in\mathcal U\otimes\mathcal V$. A finite list $S=(s_1,\ldots,s_m)$ of pair sources (recording each occurrence in a composition separately, as in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 233--251) has memory $\mathcal H_S$, is normalized if $\|\eta_j\|=1$ for all $j$, and defines the preparation map
    \begin{align}
        P_{S,\mathcal H}:\mathcal H & \longrightarrow\mathcal H_S\otimes\mathcal H,
                                    & x                                             & \longmapsto(\eta_1\otimes\cdots\otimes\eta_m)\otimes x.
        \label{eq:peps_source_preparation}
    \end{align}
\end{definition}

\begin{theorem}[Preparation before the local operations]
    \label{thm:peps_source_preparation}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_eq_localPart_comp_prepare}
    \lean{TNLean.Lattice.PartyCircuit.Word.localPart}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_localPart}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_localPart}
    \lean{TNLean.Lattice.PartyCircuit.Word.isNormalized_sources}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_localPart_prepare}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_frameList_prepare}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.eval_moveHead_prepare}
    \lean{TNLean.Lattice.PartyCircuit.Word.castInput}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_castInput}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_castInput}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_castInput_of_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_frameList}
    \leanok
    \uses{def:peps_source_inventory}
    For any allowed composition $F:\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ with source list $S$, every source in $S$ is normalized and there exists an allowed source-free composition $B:\mathcal H_S\otimes\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ such that
    \begin{align}
        F=B\circ P_{S,\mathcal H_{\rm in}}.
        \label{eq:peps_prepared_composition}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange}
    Induct on $F$, commuting later source preparations past earlier operations $A:\mathcal H\to\mathcal H'$ via $P_{S,\mathcal H'}\circ A=(\operatorname{id}_{\mathcal H_S}\otimes A)\circ P_{S,\mathcal H}$ and using Theorem~\ref{thm:peps_register_exchange} to move untouched spectator registers past newly prepared source registers.
\end{proof}

\begin{theorem}[Parties of the remaining operations]
    \label{thm:peps_source_word_parties}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties_localPart}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_localPart_mapOwner_eval_heq_id_of_disjoint}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    For any composition $W$ with party set $\mathcal P(W)$ and source-free local part $B_W$, one has $\mathcal P(B_W)=\mathcal P(W)$. Under an owner relabeling $f:P\to Q$ (converting same-owner pairs into local preparations), $\mathcal P(W^f)=f(\mathcal P(W))$, and for $q\notin f(P)$ the restriction $B_{W^f}|_{\{q\}}$ acts as $\operatorname{id}_{\C}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    By structural induction on $W$, moving pair preparations to the front preserves register owners and named local parties, so $\mathcal P(B_{W_2W_1})=\mathcal P(B_{W_1})\cup\mathcal P(B_{W_2})=\mathcal P(W_1)\cup\mathcal P(W_2)=\mathcal P(W_2W_1)$. Since $f(S_1\cup S_2)=f(S_1)\cup f(S_2)$ and $f(\{p,p'\})=\{f(p)\}$ when $f(p)=f(p')$, $\mathcal P(W^f)=f(\mathcal P(W))$. For $q\notin f(P)$, Theorem~\ref{thm:peps_party_restriction} gives $\mathcal P(B_{W^f}|_{\{q\}})=f(\mathcal P(W))\cap\{q\}=\varnothing$, so $B_{W^f}|_{\{q\}}=\operatorname{id}_{\C}$ by Theorem~\ref{thm:peps_party_factorization}.
\end{proof}

\begin{definition}[Expansion of a preparation]
    \label{def:peps_source_expansion}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.partyPair}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.reverse}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.partyPair_reverse}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.norm_reverse_vector}
    \leanok
    \uses{def:peps_source_inventory}
    A source list $G$ expands to $S$ if for every spectator memory $\mathcal H$ there exists an allowed source-free composition $E$ with $E\circ P_{G,\mathcal H}=P_{S,\mathcal H}$. The unordered pair of a source on $(p,q)$ is $\{p,q\}$, and its reversal swaps $(p,q)\mapsto(q,p)$ via the swap unitary $\mathcal U\otimes\mathcal V\to\mathcal V\otimes\mathcal U$, preserving $\{p,q\}$ and the vector norm.
\end{definition}

\begin{theorem}[Composition and reordering of expansions]
    \label{thm:peps_source_expansion_operations}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.refl}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.trans}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.cons}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.reverse}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.swap}
    \lean{TNLean.Lattice.PartyCircuit.eval_swap_reversedSource}
    \leanok
    \uses{def:peps_source_expansion}
    Source-list expansion is reflexive, transitive, preserved by prepending a common source $s$, and satisfies $(t,s,G)\to(s,t,G)$ and $(\operatorname{rev}(s),G)\to(s,G)$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange,thm:peps_source_preparation}
    Identities and composition give reflexivity and transitivity; tensoring with $\operatorname{id}_{\mathcal H_s}$ via Theorem~\ref{thm:peps_source_preparation} gives $(s,G)\to(s,S)$ from $G\to S$; and (\ref{eq:peps_register_block_exchange}) swaps the two registers of $\operatorname{rev}(s)$ or the register blocks of $t$ and $s$.
\end{proof}

\begin{theorem}[One source for each participating pair]
    \label{thm:peps_grouped_sources}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_grouped}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_combined}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_grouped_source_preparation}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.combine}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.partyPair_combine}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.norm_combine_vector}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.combine}
    \lean{TNLean.Lattice.PartyCircuit.expandCombinedPair}
    \lean{TNLean.Lattice.PartyCircuit.isAllowed_expandCombinedPair}
    \lean{TNLean.Lattice.PartyCircuit.eval_expandCombinedPair}
    \leanok
    \uses{def:peps_source_expansion}
    For any finite list $S$ of normalized pair sources, there exist a list $G$ of normalized pair sources with exactly one source per unordered pair occurring in $S$ and an allowed source-free composition $E:\mathcal H_G\otimes\mathcal H\to\mathcal H_S\otimes\mathcal H$ such that
    \begin{align}
        P_{S,\mathcal H}=E\circ P_{G,\mathcal H}.
        \label{eq:peps_grouped_preparation}
    \end{align}
    Consequently every allowed composition admits (\ref{eq:peps_prepared_composition}) with at most one normalized source per unordered party pair.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,thm:peps_source_expansion_operations,thm:peps_register_exchange}
    By Theorem~\ref{thm:peps_source_expansion_operations}, sources sharing an unordered pair $\{p,q\}$ can be made adjacent with matching orientation $(p,q)$ and combined via the unitary $(U\otimes V)\otimes(U'\otimes V')\to(U\otimes U')\otimes(V\otimes V')$ (which preserves unit norm and is inverted by Theorem~\ref{thm:peps_register_exchange}), yielding (\ref{eq:peps_grouped_preparation}); composing with (\ref{eq:peps_prepared_composition}) completes the proof.
\end{proof}
