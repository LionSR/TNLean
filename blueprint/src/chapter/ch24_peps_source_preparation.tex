\section{Preparing and combining pair sources}
\label{sec:peps_source_preparation}

\begin{lemma}[Exchanging lists of registers]
    \label{thm:peps_register_exchange}
    \lean{QuantumCircuit.Party.PartyCircuit.relabel,
        QuantumCircuit.Party.PartyCircuit.op_relabel,
        QuantumCircuit.Party.PartyCircuit.gates_relabel,
        QuantumCircuit.Party.Word.isAllowed_relabel,
        QuantumCircuit.Party.Word.sources_relabel}
    \leanok
    \uses{def:peps_party_layout}
    For two systems of registers and a bijection between the basis
    configurations of the two memories of each party, relabelling along these
    bijections is an allowed source-free composition without gates, whose
    operator is the permutation matrix of the induced bijection of
    configurations. In particular, for ordered register lists $A,B,C$ with
    memories $\mathcal H_A,\mathcal H_B,\mathcal H_C$, exchanging $A$ and $B$
    while preserving the internal order of each list (for instance moving a
    single register $B=(r)$ to the front) is such a composition $X_{A,B;C}$:
    \begin{align}
        X_{A,B;C}\bigl(a\otimes(b\otimes c)\bigr)
         & =b\otimes(a\otimes c).
        \label{eq:peps_register_block_exchange}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    At each party the relabelling applies the permutation matrix of a
    bijection of basis configurations, which has norm at most one, and it
    prepares no source. Exchanging $A$ and $B$ permutes the configurations
    of the registers of each party, which gives~(\ref{eq:peps_register_block_exchange}).
\end{proof}

\begin{definition}[The sources of a composition]
    \label{def:peps_source_inventory}
    \lean{QuantumCircuit.Party.PairSource}
    \lean{QuantumCircuit.Party.pairFactor}
    \lean{QuantumCircuit.Party.SourceInventory}
    \lean{QuantumCircuit.Party.SourceInventory.system}
    \lean{QuantumCircuit.Party.SourceInventory.IsNormalized}
    \lean{QuantumCircuit.Party.SourceInventory.prepare}
    \lean{QuantumCircuit.Party.SourceInventory.vector}
    \lean{QuantumCircuit.Party.SourceInventory.system_nil}
    \lean{QuantumCircuit.Party.SourceInventory.system_cons}
    \lean{QuantumCircuit.Party.SourceInventory.system_append}
    \lean{QuantumCircuit.Party.SourceInventory.isAllowed_prepare_iff}
    \lean{QuantumCircuit.Party.SourceInventory.isNormalized_cons}
    \lean{QuantumCircuit.Party.SourceInventory.op_prepare_apply}
    \lean{QuantumCircuit.Party.SourceInventory.op_prepare_append}
    \lean{QuantumCircuit.Party.Word.sources}
    \lean{QuantumCircuit.Party.SourceInventory.sources_prepare}
    \leanok
    \uses{def:peps_party_layout}
    A pair source consists of distinct parties $p,q$, spaces $\mathcal U,\mathcal V$ owned by $p,q$, and a vector $\eta\in\mathcal U\otimes\mathcal V$. A finite list $S=(s_1,\ldots,s_m)$ of pair sources (recording each occurrence in a composition separately, as in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 233--251) has memory $\mathcal H_S$, is normalized if $\|\eta_j\|=1$ for all $j$, and defines the preparation map
    \begin{align}
        P_{S,\mathcal H}:\mathcal H & \longrightarrow\mathcal H_S\otimes\mathcal H,
                                    & x                                             & \longmapsto(\eta_1\otimes\cdots\otimes\eta_m)\otimes x.
        \label{eq:peps_source_preparation}
    \end{align}
\end{definition}

\begin{lemma}[Preparation before the local operations]
    \label{thm:peps_source_preparation}
    \lean{QuantumCircuit.Party.Word.op_eq_localPart_comp_prepare}
    \lean{QuantumCircuit.Party.Word.localPart}
    \lean{QuantumCircuit.Party.Word.sources_localPart}
    \lean{QuantumCircuit.Party.Word.isAllowed_localPart}
    \lean{QuantumCircuit.Party.Word.isNormalized_sources}
    \lean{QuantumCircuit.Party.Word.op_localPart_mul_prepare}
    \lean{QuantumCircuit.Party.Word.sources_frame}
    \leanok
    \uses{def:peps_source_inventory}
    For any allowed composition $F:\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ with source list $S$, every source in $S$ is normalized and there exists an allowed source-free composition $B:\mathcal H_S\otimes\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ such that
    \begin{align}
        F=B\circ P_{S,\mathcal H_{\rm in}}.
        \label{eq:peps_prepared_composition}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange}
    Induct on $F$, commuting later source preparations past earlier operations $A:\mathcal H\to\mathcal H'$ via $P_{S,\mathcal H'}\circ A=(\operatorname{id}_{\mathcal H_S}\otimes A)\circ P_{S,\mathcal H}$ and using Lemma~\ref{thm:peps_register_exchange} to move untouched spectator registers past newly prepared source registers.
\end{proof}

\begin{lemma}[Parties of the remaining operations]
    \label{thm:peps_source_word_parties}
    \lean{QuantumCircuit.Party.Word.localPartOp_of_notMem_parties,
        QuantumCircuit.Party.Word.parties_coarsen_subset}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    For any composition $W$ with party set $\mathcal P(W)$ and source-free local part $B_W$, one has $\mathcal P(B_W)=\mathcal P(W)$. Under an owner relabeling $f:P\to Q$ (converting same-owner pairs into local preparations), $\mathcal P(W^f)=f(\mathcal P(W))$, and for $q\notin f(P)$ the restriction $B_{W^f}|_{\{q\}}$ acts as $\operatorname{id}_{\C}$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    By structural induction on $W$, moving pair preparations to the front preserves register owners and named local parties, so $\mathcal P(B_{W_2W_1})=\mathcal P(B_{W_1})\cup\mathcal P(B_{W_2})=\mathcal P(W_1)\cup\mathcal P(W_2)=\mathcal P(W_2W_1)$. Since $f(S_1\cup S_2)=f(S_1)\cup f(S_2)$ and $f(\{p,p'\})=\{f(p)\}$ when $f(p)=f(p')$, $\mathcal P(W^f)=f(\mathcal P(W))$. For $q\notin f(P)$, Lemma~\ref{thm:peps_party_restriction} gives $\mathcal P(B_{W^f}|_{\{q\}})=f(\mathcal P(W))\cap\{q\}=\varnothing$, so $B_{W^f}|_{\{q\}}=\operatorname{id}_{\C}$ by Lemma~\ref{thm:peps_party_factorization}.
\end{proof}

\begin{definition}[Expansion of a preparation]
    \label{def:peps_source_expansion}
    \lean{QuantumCircuit.Party.SourceInventory.Expands}
    \lean{QuantumCircuit.Party.PairSource.partyPair}
    \lean{QuantumCircuit.Party.PairSource.reverse}
    \lean{QuantumCircuit.Party.PairSource.partyPair_reverse}
    \lean{QuantumCircuit.Party.PairSource.norm_reverse_vector}
    \leanok
    \uses{def:peps_source_inventory}
    A source list $G$ expands to $S$ if for every spectator memory $\mathcal H$ there exists an allowed source-free composition $E$ with $E\circ P_{G,\mathcal H}=P_{S,\mathcal H}$. The unordered pair of a source on $(p,q)$ is $\{p,q\}$, and its reversal swaps $(p,q)\mapsto(q,p)$ via the swap unitary $\mathcal U\otimes\mathcal V\to\mathcal V\otimes\mathcal U$, preserving $\{p,q\}$ and the vector norm.
\end{definition}

\begin{lemma}[Composition and reordering of expansions]
    \label{thm:peps_source_expansion_operations}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.refl}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.trans}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.cons}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.reverse}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.swap}
    \lean{QuantumCircuit.Party.Word.op_relabel_sourceAt_reverse}
    \leanok
    \uses{def:peps_source_expansion}
    Source-list expansion is reflexive, transitive, preserved by prepending a common source $s$, and satisfies $(t,s,G)\to(s,t,G)$ and $(\operatorname{rev}(s),G)\to(s,G)$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange,thm:peps_source_preparation}
    Identities and composition give reflexivity and transitivity; tensoring with $\operatorname{id}_{\mathcal H_s}$ via Lemma~\ref{thm:peps_source_preparation} gives $(s,G)\to(s,S)$ from $G\to S$; and (\ref{eq:peps_register_block_exchange}) swaps the two registers of $\operatorname{rev}(s)$ or the register blocks of $t$ and $s$.
\end{proof}

\begin{lemma}[One source for each participating pair]
    \label{thm:peps_grouped_sources}
    \lean{QuantumCircuit.Party.SourceInventory.exists_grouped}
    \lean{QuantumCircuit.Party.SourceInventory.exists_combined}
    \lean{QuantumCircuit.Party.Word.exists_grouped_source_preparation}
    \lean{QuantumCircuit.Party.PairSource.combine}
    \lean{QuantumCircuit.Party.PairSource.partyPair_combine}
    \lean{QuantumCircuit.Party.PairSource.norm_combine_vector}
    \lean{QuantumCircuit.Party.SourceInventory.Expands.combine}
    \lean{QuantumCircuit.Party.Word.expandCombinedPair}
    \lean{QuantumCircuit.Party.Word.isAllowed_expandCombinedPair}
    \lean{QuantumCircuit.Party.Word.op_expandCombinedPair}
    \leanok
    \uses{def:peps_source_expansion}
    For any finite list $S$ of normalized pair sources, there exist a list $G$ of normalized pair sources with exactly one source per unordered pair occurring in $S$ and an allowed source-free composition $E:\mathcal H_G\otimes\mathcal H\to\mathcal H_S\otimes\mathcal H$ such that
    \begin{align}
        P_{S,\mathcal H}=E\circ P_{G,\mathcal H}.
        \label{eq:peps_grouped_preparation}
    \end{align}
    Consequently every allowed composition admits (\ref{eq:peps_prepared_composition}) with at most one normalized source per unordered party pair.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,thm:peps_source_expansion_operations,thm:peps_register_exchange}
    By Lemma~\ref{thm:peps_source_expansion_operations}, sources sharing an unordered pair $\{p,q\}$ can be made adjacent with matching orientation $(p,q)$ and combined via the unitary $(U\otimes V)\otimes(U'\otimes V')\to(U\otimes U')\otimes(V\otimes V')$ (which preserves unit norm and is inverted by Lemma~\ref{thm:peps_register_exchange}), yielding (\ref{eq:peps_grouped_preparation}); composing with (\ref{eq:peps_prepared_composition}) completes the proof.
\end{proof}
