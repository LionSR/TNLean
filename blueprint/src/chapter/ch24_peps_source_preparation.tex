\begin{theorem}[Exchanging lists of registers]
    \label{thm:peps_register_exchange}
    \lean{TNLean.PEPS.PairEffect.Word.eval_exchangeBlocks_appendIso_symm}
    \lean{TNLean.PEPS.PairEffect.Word.exchangeBlocks}
    \lean{TNLean.PEPS.PairEffect.Word.isAllowed_exchangeBlocks}
    \lean{TNLean.PEPS.PairEffect.Word.moveHead}
    \lean{TNLean.PEPS.PairEffect.Word.isAllowed_moveHead}
    \lean{TNLean.PEPS.PairEffect.Word.eval_moveHead_appendIso_symm}
    \lean{TNLean.PEPS.PairEffect.appendIso_symm_cons_tmul}
    \lean{TNLean.PEPS.PairEffect.Word.sources_moveHead}
    \lean{TNLean.PEPS.PairEffect.Word.sources_exchangeBlocks}
    \leanok
    \uses{def:peps_party_layout}
    Let $\mathcal H_A$, $\mathcal H_B$, and $\mathcal H_C$ be the memories
    of three ordered lists of registers. Adjacent exchanges give an allowed
    composition that sends $a\otimes(b\otimes c)$ to
    $b\otimes(a\otimes c)$. The order within each list is preserved.
    This composition contains no pair-source preparations. Taking the
    second list to consist of one register gives the corresponding move
    of that register to the front.
\end{theorem}

\begin{proof}\leanok
    Writing $(A,C)$ for the list $A$ followed by $C$, let $J_{A,C}$
    identify its memory with $\mathcal H_A\otimes\mathcal H_C$.
    For a register $r$ with vector
    $x$ and vectors $a\in\mathcal H_A$, $c\in\mathcal H_C$, the move
    $M_{r;A,C}$ of $r$ past $A$ satisfies
    \begin{align}
        M_{r;A,C}J_{A,(r,C)}^{-1}(a\otimes(x\otimes c))
         & =x\otimes J_{A,C}^{-1}(a\otimes c).
        \label{eq:peps_register_head_move}
    \end{align}
    By induction on $A$, moving $r$ through the tail and exchanging it with
    the first register establishes the move identity, with the empty-list case
    reducing to the scalar unit identity.
    Inducting next on $B$ and composing this move for its first register with
    the exchange of the remaining lists, while leaving that register
    untouched, produces an exchange $X_{A,B;C}$ that satisfies, for
    $b\in\mathcal H_B$,
    \begin{align}
        X_{A,B;C}J_{A,(B,C)}^{-1}
        \bigl(a\otimes J_{B,C}^{-1}(b\otimes c)\bigr)
         & =J_{B,(A,C)}^{-1}
        \bigl(b\otimes J_{A,C}^{-1}(a\otimes c)\bigr),
        \label{eq:peps_register_block_exchange}
    \end{align}
    Extending from elementary tensors by linearity completes the identity, and
    because only adjacent exchanges and identities appear, the composition is
    allowed and has no sources.
\end{proof}

\begin{definition}[The sources of a composition]
    \label{def:peps_source_inventory}
    \lean{TNLean.PEPS.PairEffect.PairSource}
    \lean{TNLean.PEPS.PairEffect.PairSource.layout}
    \lean{TNLean.PEPS.PairEffect.SourceInventory}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.layout}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.IsNormalized}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.prepare}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.vector}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.layout_nil}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.layout_cons}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.layout_append}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.isAllowed_prepare_iff}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.isNormalized_cons}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.eval_prepare_eq_appendIso_symm}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.eval_prepare_append}
    \lean{TNLean.PEPS.PairEffect.Word.sources}
    \lean{TNLean.PEPS.PairEffect.Word.sources_prepare}
    \leanok
    \uses{def:peps_party_layout}
    A pair source consists of distinct parties $p,q$, their Hilbert spaces
    $\mathcal U,\mathcal V$, and a vector
    $\eta\in\mathcal U\otimes\mathcal V$.
    Its two registers remain assigned to $p$ and $q$, respectively.
    For a finite list $S=(s_1,\ldots,s_m)$ of pair sources, let
    $\mathcal H_S$ be the tensor product of their registers in list order.
    The list is normalized when $\|\eta_j\|=1$ for every source in $S$;
    the empty list is normalized.
    Its preparation on a memory $\mathcal H$ is the map
    \begin{align}
        P_{S,\mathcal H}:\mathcal H & \longrightarrow\mathcal H_S\otimes\mathcal H,
                                    & x                                             & \longmapsto(\eta_1\otimes\cdots\otimes\eta_m)\otimes x.
        \label{eq:peps_source_preparation}
    \end{align}
    Tensor products are identified by the prescribed associativity maps.
    As in the source data of
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 233--251, the source list of a
    composition records each occurrence separately, including repetitions of
    the same vector on the same pair of parties.
\end{definition}

\begin{theorem}[Preparation before the local operations]
    \label{thm:peps_source_preparation}
    \lean{TNLean.PEPS.PairEffect.Word.eval_eq_localPart_comp_prepare}
    \lean{TNLean.PEPS.PairEffect.Word.localPart}
    \lean{TNLean.PEPS.PairEffect.Word.sources_localPart}
    \lean{TNLean.PEPS.PairEffect.Word.isAllowed_localPart}
    \lean{TNLean.PEPS.PairEffect.Word.isNormalized_sources}
    \lean{TNLean.PEPS.PairEffect.Word.eval_localPart_prepare}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.eval_frameList_prepare}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.eval_moveHead_prepare}
    \lean{TNLean.PEPS.PairEffect.Word.castInput}
    \lean{TNLean.PEPS.PairEffect.Word.sources_castInput}
    \lean{TNLean.PEPS.PairEffect.Word.isAllowed_castInput}
    \lean{TNLean.PEPS.PairEffect.Word.eval_castInput_of_heq}
    \lean{TNLean.PEPS.PairEffect.Word.sources_frameList}
    \leanok
    \uses{def:peps_source_inventory}
    Let $F:\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ be an allowed
    composition of local contractions, pair-source preparations, and
    changes of tensor-factor order. Let $S$ be its source list.
    There is an allowed composition
    $B:\mathcal H_S\otimes\mathcal H_{\rm in}\to\mathcal H_{\rm out}$
    consisting of local contractions and changes of tensor-factor order,
    with no source preparations, such that
    \begin{align}
        F=B\circ P_{S,\mathcal H_{\rm in}}.
        \label{eq:peps_prepared_composition}
    \end{align}
    Every vector in $S$ is normalized. The source occurrences and their
    endpoint parties are those of the original composition.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange}
    We proceed by induction on the composition. A local operation, an
    exchange, or the identity contributes no source, whereas a pair-source
    preparation has that source as its entire list and the identity as the
    remaining operation. In the composition case, moving a later preparation
    before an earlier operation $A:\mathcal H\to\mathcal H'$ by adjoining its
    untouched registers uses the identity
    \begin{align*}
        P_{S,\mathcal H'}\circ A
        =(\operatorname{id}_{\mathcal H_S}\otimes A)
        \circ P_{S,\mathcal H},
    \end{align*}
    which holds first on elementary tensors and then by linearity.
    When an untouched register precedes the composition, isometric exchanges
    move it in front of the newly prepared source registers before the
    remaining local operations act, and tensoring each original local map
    with an identity preserves its contraction bound.
\end{proof}

\begin{theorem}[Parties of the remaining operations]
    \label{thm:peps_source_word_parties}
    \lean{TNLean.PEPS.PairEffect.Word.parties_localPart}
    \lean{TNLean.PEPS.PairEffect.Word.parties_mapOwner}
    \lean{TNLean.PEPS.PairEffect.Word.restrict_localPart_mapOwner_eval_heq_id_of_disjoint}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    For any composition $W$, write $\mathcal P(W)$ for the parties of its
    registers and named local operations, and let $B_W$ be the remaining
    composition after its pair preparations are moved to the beginning.
    Then $\mathcal P(B_W)=\mathcal P(W)$.
    If $f:P\to Q$ relabels the owners, retaining a pair whose new owners
    coincide as a local preparation, then
    $\mathcal P(W^f)=f(\mathcal P(W))$.
    In particular, for $q\notin f(P)$, the restriction of $B_{W^f}$ to
    $q$ acts as the identity on the empty memory $\C$.
    These statements require neither normalization nor nonzero source
    vectors. A scalar local operation retains its named party.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,thm:peps_party_factorization,thm:peps_party_restriction,def:peps_single_party_selector}
    By induction on the composition, moving a pair preparation to the
    beginning replaces it by the identity on its two endpoint registers and
    the original spectators, with the same owners. In a concatenation, the
    extra spectators are source registers of the later composition, so
    their owners already occur there, and
    \begin{align*}
        \mathcal P(B_{W_2W_1})
         & =\mathcal P(B_{W_1})\cup\mathcal P(B_{W_2})
        =\mathcal P(W_1)\cup\mathcal P(W_2)=\mathcal P(W_2W_1).
    \end{align*}
    Because the exchanges in the framed case involve only the framed owner and
    owners of the original registers and sources, no party is added
    or removed. Relabelling preserves register owners and named local
    maps, and the party sets transform through
    \begin{align*}
        f(S_1\cup S_2) & =f(S_1)\cup f(S_2),
                       &
        f(\{p\}\cup S) & =\{f(p)\}\cup f(S).
    \end{align*}
    When two endpoints acquire the same owner, $\{f(p),f(q)\}=\{f(p)\}$
    and that owner is still recorded in the local preparation. Finally,
    for $q\notin f(P)$,
    \begin{align*}
        \mathcal P(B_{W^f}|_q)=\mathcal P(B_{W^f})\cap\{q\}
        =f(\mathcal P(W))\cap\{q\}=\varnothing,
    \end{align*}
    and the no-party clause of
    Theorem~\ref{thm:peps_party_factorization} gives the identity on $\C$.
\end{proof}

\begin{definition}[Expansion of a preparation]
    \label{def:peps_source_expansion}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands}
    \lean{TNLean.PEPS.PairEffect.PairSource.partyPair}
    \lean{TNLean.PEPS.PairEffect.PairSource.reverse}
    \lean{TNLean.PEPS.PairEffect.PairSource.partyPair_reverse}
    \lean{TNLean.PEPS.PairEffect.PairSource.norm_reverse_vector}
    \leanok
    \uses{def:peps_source_inventory}
    A source list $G$ expands to $S$ when, beside every spectator memory
    $\mathcal H$, there is an allowed composition $E$ with no source
    preparations such that $E\circ P_{G,\mathcal H}=P_{S,\mathcal H}$.
    The unordered pair of a source with endpoints $p,q$ is $\{p,q\}$.
    Reversing that source exchanges its endpoints and sends its vector
    through the isometry $\mathcal U\otimes\mathcal V\to
        \mathcal V\otimes\mathcal U$; its unordered pair and norm are unchanged.
\end{definition}

\begin{theorem}[Composition and reordering of expansions]
    \label{thm:peps_source_expansion_operations}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.refl}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.trans}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.cons}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.reverse}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.swap}
    \lean{TNLean.PEPS.PairEffect.eval_swap_reversedSource}
    \leanok
    \uses{def:peps_source_expansion}
    Expansion is reflexive and transitive, and remains valid upon placing
    a common source first in both lists. A list beginning with two sources
    $t,s$ expands to the list beginning with $s,t$ and the same remaining
    sources. A list beginning with the reversal of $s$ expands to the list
    beginning with $s$ and the same remaining sources.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange,thm:peps_source_preparation}
    Identity maps and composition prove reflexivity and transitivity.
    Tensoring an expansion $E$ with the identity on a common first source's
    registers, via the prescribed associativity maps, produces $E_s$.
    Writing $(s,G)$ for $s$ followed by $G$, the preparation identity in
    Theorem~\ref{thm:peps_source_preparation} gives
    $E_sP_{(s,G),\mathcal H}=P_{(s,S),\mathcal H}$ whenever
    $EP_{G,\mathcal H}=P_{S,\mathcal H}$.
    Exchanging the entire register lists of successive sources via
    (\ref{eq:peps_register_block_exchange}) handles the swap, while
    exchanging the two registers of a reversed source uses the fact that
    applying the tensor exchange twice is the identity. Each construction is
    allowed and contains no sources.
\end{proof}

\begin{theorem}[One source for each participating pair]
    \label{thm:peps_grouped_sources}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.exists_grouped}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.exists_combined}
    \lean{TNLean.PEPS.PairEffect.Word.exists_grouped_source_preparation}
    \lean{TNLean.PEPS.PairEffect.PairSource.combine}
    \lean{TNLean.PEPS.PairEffect.PairSource.partyPair_combine}
    \lean{TNLean.PEPS.PairEffect.PairSource.norm_combine_vector}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.Expands.combine}
    \lean{TNLean.PEPS.PairEffect.expandCombinedPair}
    \lean{TNLean.PEPS.PairEffect.isAllowed_expandCombinedPair}
    \lean{TNLean.PEPS.PairEffect.eval_expandCombinedPair}
    \leanok
    \uses{def:peps_source_expansion}
    Let $S$ be a finite list of normalized pair sources. There is a list
    $G$ of normalized pair sources containing exactly one source for each
    unordered pair of parties occurring in $S$, and an allowed composition
    $E:\mathcal H_G\otimes\mathcal H\to\mathcal H_S\otimes\mathcal H$
    with no source preparations, such that
    \begin{align}
        P_{S,\mathcal H}=E\circ P_{G,\mathcal H}.
        \label{eq:peps_grouped_preparation}
    \end{align}
    Consequently every allowed composition admits
    (\ref{eq:peps_prepared_composition}) with at most one normalized source
    on each participating pair.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_preparation,thm:peps_source_expansion_operations,thm:peps_register_exchange}
    Inserting the sources successively and exchanging tensor factors brings
    together the sources with a prescribed unordered pair of owners,
    reversing the order of the two half-spaces when necessary. Two
    such vectors, on $U\otimes V$ and $U'\otimes V'$, can then be replaced by
    the image of their tensor product under the factor-swapping isometry
    $(U\otimes V)\otimes(U'\otimes V')
        \to(U\otimes U')\otimes(V\otimes V')$,
    whose norm is the product of the two original norms and is therefore
    one. Splitting the two local tensor factors and exchanging the
    middle registers recovers the original preparations by isometries.
    Continuing through the finite list proves
    (\ref{eq:peps_grouped_preparation}) without introducing any other pair of
    owners, and composing this expansion with the local operations in
    Theorem~\ref{thm:peps_source_preparation} completes the factorization.
\end{proof}
