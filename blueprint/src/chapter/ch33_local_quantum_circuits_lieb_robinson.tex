\section{Lieb--Robinson propagation on finite graphs}
\label{sec:qc_lieb_robinson}

\begin{theorem}[Extended distance under graph maps]
    \label{thm:qc_graph_edist_transport}
    \lean{SimpleGraph.Hom.edist_le, SimpleGraph.Iso.edist_eq,
        SimpleGraph.ConnectedComponent.edist_eq}
    \leanok
    Let $G$ and $H$ be simple graphs on arbitrary vertex sets. A graph
    homomorphism $f:G\to H$ satisfies
    $d_H(f(x),f(y))\le d_G(x,y)$ for all vertices $x,y$.
    Equality holds when $f$ is a graph isomorphism. If $C$ is a connected
    component of $G$, the distance in the graph induced on $C$ is the
    ambient distance: $d_{G[C]}(x,y)=d_G(x,y)$ for $x,y\in C$.
    All distances take values in $\mathbb N\cup\{\infty\}$; no finiteness
    hypothesis on either vertex set is imposed.
\end{theorem}
\begin{proof}
    \leanok
    A homomorphism maps each walk to a walk of the same length. Taking
    infima over walk lengths proves the inequality, with the infimum of
    the empty set equal to infinity. Applying this twice, to an isomorphism
    and its inverse, gives equality. Every vertex on an ambient walk from
    $x\in C$ is connected to $x$, so a walk with endpoints in $C$ stays in
    $C$. It therefore gives a walk of the same length in the induced graph.
    The two infimum comparisons give the component assertion.
\end{proof}

Let $I$ be a finite set of sites of local dimension $q$, and write
$\mathcal A_X$ for the operators on $(\C^q)^{\otimes I}$ acting on
$X\subseteq I$. All norms are Hilbert-space operator norms. For an operator
$H$ put $\tau_t(A)=e^{itH}Ae^{-itH}$. The dynamics generated by a sum of
local Hermitian terms moves support at a bounded speed, up to exponentially
small tails \cite{Hastings2006Spectral,Nachtergaele2006LiebRobinson}. The
estimate below measures distance by the path metric of an arbitrary simple
graph on $I$, so that holes and separate connected components cost nothing:
sites in different components are at infinite distance, and there the
commutator vanishes exactly.

\begin{definition}[Restricted commutator norm]
    \label{aux:restricted-commutator-norm}
    \lean{QuantumCircuit.heisenbergCommutatorNorm,
        QuantumCircuit.heisenbergEvolution}\leanok
    For operators $H,B$ and a set of sites $X$, define
    \[
        C_B(X,t)=\sup\{\|[\tau_t(A),B]\|: A\in\mathcal A_X,\ \|A\|\leq1\}.
    \]
\end{definition}

\begin{lemma}[Evaluation of the restricted commutator norm]
    \label{aux:restricted-commutator-evaluation}
    \lean{QuantumCircuit.norm_heisenberg_commutator_le_heisenbergCommutatorNorm}\leanok
    \uses{aux:restricted-commutator-norm}
    For $A\in\mathcal A_X$,
    $\|[\tau_t(A),B]\|\leq C_B(X,t)\|A\|$.
\end{lemma}
\begin{proof}\leanok
    The bound follows directly from the defining operator-norm supremum
    applied to $A$.
\end{proof}

\begin{lemma}[Reversal of time]
    \label{aux:restricted-commutator-negative-time}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_neg_time}\leanok
    \uses{aux:restricted-commutator-norm}
    The restricted commutator norm for $H$ at time $-t$ equals that for
    $-H$ at time $t$.
\end{lemma}
\begin{proof}\leanok
    Both Heisenberg evolutions are $A\mapsto e^{-itH}Ae^{itH}$.
\end{proof}

\begin{lemma}[Continuity in time]
    \label{aux:restricted-commutator-continuity}
    \lean{QuantumCircuit.continuous_heisenbergCommutatorNorm}\leanok
    \uses{aux:restricted-commutator-norm}
    For fixed $H,B,X$, the function $t\mapsto C_B(X,t)$ is continuous.
\end{lemma}
\begin{proof}\leanok
    Matrix exponentiation and multiplication are continuous, so the operator
    norm of the resulting linear map on $\mathcal A_X$ varies continuously
    with $t$.
\end{proof}

\begin{lemma}[Initial commutator bound]
    \label{aux:restricted-initial-commutator-bound}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_zero_le}\leanok
    \uses{aux:restricted-commutator-norm}
    For arbitrary $H,B,X$, $C_B(X,0)\leq2\|B\|$.
\end{lemma}
\begin{proof}\leanok
    Every $A\in\mathcal A_X$ with $\|A\|\leq1$ satisfies
    $\|AB-BA\|\leq2\|A\|\|B\|\leq2\|B\|$, and taking the supremum gives
    $C_B(X,0)\leq2\|B\|$.
\end{proof}

\begin{lemma}[Vanishing initial commutator]
    \label{aux:restricted-initial-commutator-vanishing}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_zero_of_disjoint}\leanok
    \uses{aux:restricted-commutator-norm}
    If $B\in\mathcal A_Y$ and $X\cap Y=\varnothing$, then $C_B(X,0)=0$.
\end{lemma}
\begin{proof}\leanok
    Operators on disjoint tensor factors commute.
\end{proof}

\begin{lemma}[Local commutator recursion]
    \label{aux:restricted-commutator-recursion}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_integral}\leanok
    \uses{aux:restricted-commutator-norm}
    Following the recursion of Hastings--Koma, Appendix A, (A.12)--(A.14),
    let $H=\sum_j h_j$ be a finite sum, where each $h_j$ is Hermitian and
    belongs to $\mathcal A_{T_j}$. For arbitrary $B,X$ and $t\geq0$,
    \[
        C_B(X,t)\leq C_B(X,0)
        +2\sum_{j:X\cap T_j\neq\varnothing}\|h_j\|
        \int_0^t C_B(T_j,s)\,ds,
    \]
    with no bound on the ground energy or positivity assumption on $h_j$
    required, and with several terms allowed to share a support.
\end{lemma}
\begin{proof}\leanok
    Separating the terms disjoint from $X$ yields a sum that commutes with
    every operator in $\mathcal A_X$. Differentiating in the corresponding
    interaction picture, applying the fundamental theorem of calculus together
    with unitary invariance of the operator norm, and taking the restricted
    operator norm then gives the integral bound.
\end{proof}

\begin{lemma}[Weighted integral comparison]
    \label{aux:weighted-volterra-comparison}
    \lean{finite_volterra_le_weight_mul_exp,
        finite_volterra_zero_le_weight_mul_exp_sub_one,
        finite_rectangular_volterra_zero_le_weight_mul_exp_sub_one,
        le_mul_exp_of_le_const_add_integral}\leanok
    Let $f_i\geq0$, for $i$ in a finite set, be continuous on $[0,T]$ and satisfy
    $f_i(t)\leq b_i+\sum_j a_{ij}\int_0^tf_j(s)\,ds$ with $a_{ij}\geq0$.
    If positive weights $w_i$ satisfy $\sum_ja_{ij}w_j\leq vw_i$ and
    $b_i\leq cw_i$, then $f_i(t)\leq cw_ie^{vt}$ for $0\leq t\leq T$. If in addition
    $b_i=0$ and $v>0$, then $f_i(t)\leq cw_i(e^{vt}-1)$.
\end{lemma}
\begin{proof}\leanok
    The weighted maximum $g(t)=\max_if_i(t)/w_i$ satisfies the scalar
    inequality $g(t)\leq c+v\int_0^tg$, so Gr\"onwall's inequality gives
    $g(t)\leq ce^{vt}$, and substituting this bound back into the integral
    once more yields the $b_i=0$ refinement.
\end{proof}

\begin{lemma}[Exact locality on interaction-closed regions]
    \label{lem:qc_lr_closed_region}
    \lean{QuantumCircuit.commute_heisenbergEvolution_of_closed}\leanok
    Let $H=\sum_jh_j$ with $h_j\in\mathcal A_{T_j}$, and let $C$ be a set of sites
    such that each $T_j$ is contained in $C$ or disjoint from $C$. If
    $A\in\mathcal A_C$ and $B\in\mathcal A_D$ with $C\cap D=\varnothing$, then
    $[\tau_t(A),B]=0$ for every $t$.
\end{lemma}
\begin{proof}\leanok
    Splitting $H=H_C+H'$ with $H_C$ the sum of the terms supported inside $C$
    makes the two parts commute and $H'$ commute with $\mathcal A_C$, so
    $\tau_t(A)=e^{itH_C}Ae^{-itH_C}$, where every factor commutes with $B$.
\end{proof}

\paragraph{Finite-range interactions on a graph.}
Let $G$ be a simple graph on $I$ with path metric $d_G$, taking the value
$\infty$ between different connected components. Let $H=\sum_kh_k$ be a
finite sum of Hermitian terms with $h_k\in\mathcal A_{X_k}$, and choose an
anchor $a_k\in X_k$. Assume that every support has $d_G$-diameter at most
$R$ and at most $v_R$ sites, and that the norms of the terms whose support
contains any given site sum to at most $b_0$.

\begin{theorem}[Graph-distance propagation]
    \label{thm:qc_graph_lieb_robinson}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_graphDistance,
        QuantumCircuit.norm_heisenberg_commutator_le_graphDistance,
        QuantumCircuit.exists_graph_lieb_robinson}\leanok
    \uses{aux:restricted-commutator-recursion,aux:weighted-volterra-comparison,
        aux:restricted-commutator-evaluation,aux:restricted-commutator-negative-time,
        aux:restricted-initial-commutator-bound,
        aux:restricted-initial-commutator-vanishing,
        aux:restricted-commutator-continuity,thm:qc_graph_lieb_robinson_zero}
    In this setting let $A\in\mathcal A_{X_i}$, let $y$ be a site, and let
    $B\in\mathcal A_{\{y\}}$. For every $\mu\geq0$, every integer
    $0\leq n\leq d_G(a_i,y)$ and every real $t$,
    \[
        \|[\tau_t(A),B]\|\leq 2e^{\mu R}\|A\|\|B\|
        \exp\bigl(2v_Rb_0e^{2\mu R}|t|-\mu n\bigr).
    \]
    The restricted norm satisfies
    \begin{align}
        C_B(X_i,t) & \leq 2e^{\mu R}\|B\|
        \exp\bigl(2v_Rb_0e^{2\mu R}|t|-\mu n\bigr).\notag
    \end{align}
    Consequently there are positive constants $C,v,c$, depending only on
    $R,v_R,b_0$, such that
    $\|[\tau_t(A),B]\|\leq C\|A\|\|B\|e^{v|t|-cd_G(a_i,y)}$ whenever
    $d_G(a_i,y)<\infty$, and $[\tau_t(A),B]=0$ otherwise. The constants do not
    depend on the number of sites, on $q$, or on the graph.
\end{theorem}
\begin{proof}\leanok
    Let $F_k(t)=C_B(X_k,t)$ for $t\geq0$. The local commutator recursion gives
    $F_k(t)\leq F_k(0)+\sum_ja_{kj}\int_0^tF_j$, with $a_{kj}=2\|h_j\|$ when
    $X_j$ meets $X_k$ and $a_{kj}=0$ otherwise. Put
    $D(x)=\min\{n,d_G(x,y)\}$ and $w_k=e^{-\mu D(a_k)}$. If $X_j$ meets
    $X_k$, then $d_G(a_k,a_j)\leq2R$, so $w_j\leq e^{2\mu R}w_k$; the norms of the
    terms meeting $X_k$ sum to at most $|X_k|b_0\leq v_Rb_0$. Hence
    $\sum_ja_{kj}w_j\leq2v_Rb_0e^{2\mu R}w_k$. The initial value $F_k(0)$
    vanishes unless $y\in X_k$, and then $D(a_k)\leq R$ and
    $F_k(0)\leq2\|B\|\leq2\|B\|e^{\mu R}w_k$. The weighted integral comparison
    gives $F_i(t)\leq2\|B\|e^{\mu R}w_ie^{2v_Rb_0e^{2\mu R}t}$ with
    $w_i=e^{-\mu n}$. Replacing $H$ by $-H$ extends the bound to negative
    times and multiplying by $\|A\|$ bounds $\|[\tau_t(A),B]\|$; setting
    $\mu=1$ gives the constants $C,v,c$ for $d_G(a_i,y)<\infty$, while
    Lemma~\ref{thm:qc_graph_lieb_robinson_zero} handles $d_G(a_i,y)=\infty$.
\end{proof}

\begin{theorem}[Graph-distance propagation from an arbitrary finite support]
    \label{thm:qc_graph_lieb_robinson_finset}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_graphDistance_finset,
        QuantumCircuit.norm_heisenberg_commutator_le_graphDistance_finset}\leanok
    \uses{aux:restricted-commutator-norm}
    Under the preceding finite-range interaction assumptions, let $S$ be any
    finite set of sites, let $A\in\mathcal A_S$, and let
    $B\in\mathcal A_{\{y\}}$. No diameter bound on $S$ is assumed, and
    $S=\varnothing$ is allowed. Fix $\mu\geq0$ and an integer $n\geq0$ such
    that $n\leq d_G(x,y)$ for every $x\in S$. Put
    \begin{align}
        C & =2+4|b_0|e^{2\mu R},\notag    \\
        v & =2v_R|b_0|e^{2\mu R}+1.\notag
    \end{align}
    For every real $t$,
    \begin{align}
        C_B(S,t)          & \leq C|S|\|B\|e^{v|t|-\mu n},
        \label{eq:qc_lr_finset_norm}                           \\
        \|[\tau_t(A),B]\| & \leq C|S|\|A\|\|B\|e^{v|t|-\mu n}.
        \label{eq:qc_lr_finset_commutator}
    \end{align}
    The constants depend only on $R,v_R,b_0,\mu$, not on the graph, the
    number of sites, the local dimension, or $S$.
    The second estimate has the form used in the lemma
    ``Lieb--Robinson bound on a chain'' of
    \cite{TNLean2026OneDimensional}.
\end{theorem}
\begin{proof}\leanok
    \uses{aux:restricted-commutator-recursion,thm:qc_graph_lieb_robinson,
        aux:restricted-initial-commutator-bound,
        aux:restricted-initial-commutator-vanishing,
        aux:restricted-commutator-continuity,
        aux:restricted-commutator-evaluation,
        aux:restricted-commutator-negative-time}
    Write $b=|b_0|$ and $\lambda=2v_Rbe^{2\mu R}$, and first take $t\geq0$.
    The per-site budget gives
    $\sum_{k:X_k\cap S\ne\varnothing}\|h_k\|\leq |S|b$.
    For a term meeting $S$, choose $x\in S\cap X_k$.
    Since $d_G(x,a_k)\leq R$, the triangle inequality gives
    $n-R\leq d_G(a_k,y)$, with subtraction in the nonnegative integers.
    Graph-distance propagation applied at $n-R$ therefore bounds
    $C_B(X_k,s)$ by $2\|B\|e^{2\mu R-\mu n}e^{\lambda s}$ for $s\geq0$;
    here $n\leq(n-R)+R$ also covers $n<R$.
    The initial norm vanishes unless $y\in S$, in which case $n=0$ and
    $|S|\geq1$. Thus $C_B(S,0)\leq2|S|\|B\|e^{-\mu n}$, including
    $S=\varnothing$.
    Applying the local recursion once and bounding $e^{\lambda s}$ by
    $e^{\lambda t}$ for $0\leq s\leq t$ gives
    \begin{align}
        C_B(S,t)
         & \leq |S|\|B\|e^{-\mu n}
        \bigl(2+4be^{2\mu R}t e^{\lambda t}\bigr)\notag \\
         & \leq (2+4be^{2\mu R})|S|\|B\|
        e^{(\lambda+1)t-\mu n}.\notag
    \end{align}
    The last inequality uses $1\leq e^{\lambda t}$ and $1,t\leq e^t$.
    Replacing $H$ by $-H$ gives the same bound for negative times and proves
    (\ref{eq:qc_lr_finset_norm}); evaluation on $A$ gives
    (\ref{eq:qc_lr_finset_commutator}).
\end{proof}

\begin{theorem}[Exact commutation across disconnected components]
    \label{thm:qc_graph_lieb_robinson_zero_finset}
    \lean{QuantumCircuit.heisenberg_commutator_eq_zero_of_forall_edist_eq_top}
    \leanok
    \uses{aux:restricted-commutator-norm}
    Let $G$ be a simple graph on a finite site set $I$, with local dimension
    $q\in\mathbb N$. Let $H=\sum_k h_k$ be a finite sum of operators supported
    on sets $X_k$ of graph diameter at most $R\in\mathbb N$.
    Let $A$ be supported on an arbitrary set $S\subseteq I$, and let $B$ be
    supported at $y$. If $d_G(x,y)=\infty$ for every $x\in S$, then
    $[\tau_t(A),B]=0$ for every real $t$.
    This is the exact-vanishing argument of
    \cite[Lemma~4.1]{OpenAI2026AreaLaw}, with arbitrary initial support.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_lr_closed_region}
    Let $C$ consist of the sites not connected to $y$. Any two sites of
    $X_k$ are connected, so $X_k$ lies in $C$ or is disjoint from $C$.
    Since $S\subseteq C$ and $y\notin C$, exact locality on the closed
    region $C$ gives the result. This also covers $S=\varnothing$.
\end{proof}

\begin{theorem}[Vanishing between connected components]
    \label{thm:qc_graph_lieb_robinson_zero}
    \lean{QuantumCircuit.heisenberg_commutator_eq_zero_of_edist_eq_top}\leanok
    \uses{thm:qc_graph_lieb_robinson_zero_finset}
    Let $H=\sum_kh_k$ be a finite sum, with each $h_k$ supported on a set
    $X_k$ of graph diameter at most $R$, and choose anchors $a_k\in X_k$.
    Let $A$ be supported on $X_i$ and $B$ at $y$.
    If $d_G(a_i,y)=\infty$, then $[\tau_t(A),B]=0$ for every real $t$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:qc_graph_lieb_robinson_zero_finset}
    Every site of $X_i$ is connected to its anchor $a_i$, by the finite
    diameter bound. As $d_G(a_i,y)=\infty$, every site of $X_i$ has infinite
    distance to $y$. Apply
    Theorem~\ref{thm:qc_graph_lieb_robinson_zero_finset} with $S=X_i$.
\end{proof}

\begin{lemma}[Exponential kernel outside a graph ball]
    \label{lem:qc_graph_kernel_tail}
    \lean{QuantumCircuit.sum_exp_neg_edist_compl_graphBall_le}
    \leanok
    Let $G$ be a simple graph on a finite set $I$, let $x\in I$, let
    $K_g\in\mathbb R$ and $k\in\mathbb N$, and suppose that for every
    $d\in\mathbb N$ at most $K_g(d+1)^k$ vertices have graph distance
    exactly $d$ from $x$. For $\ell\in\mathbb N$ put
    $N_\ell(x)=\{y:d_G(x,y)\leq\ell\}$ and
    $T_k=2^kk!e^{1/2}/(1-e^{-1/2})$. Then
    \begin{align}
        \sum_{\substack{y\notin N_\ell(x) \\d_G(x,y)<\infty}}
        e^{-d_G(x,y)}
         & \leq K_gT_ke^{-\ell/2}.
        \label{eq:qc_kernel_tail}
    \end{align}
    This is the polynomial-growth form of the distance sum used in
    \cite[Lemma~4.1]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}
    \leanok
    The growth inequality at $d=0$ gives $K_g\geq0$.
    Grouping the vertices by their finite distance from $x$ bounds the sum
    by $K_g\sum_{d\in D}(d+1)^ke^{-d}$ for a finite set of integers
    $D\subseteq\{d:d>\ell\}$. The power-series bound for the exponential
    gives
    \begin{align}
        (d+1)^ke^{-d}
         & \leq 2^kk!e^{1/2}e^{-d/2}.\notag
    \end{align}
    Summing the geometric series from $d=\ell+1$ and using
    $e^{-(\ell+1)/2}\leq e^{-\ell/2}$ proves
    (\ref{eq:qc_kernel_tail}). Vertices at infinite distance do not
    contribute.
\end{proof}

\paragraph{Normalized partial traces.}
For $K\subseteq I$ let
$E_K(B)=q^{-|I\setminus K|}\operatorname{Tr}_{I\setminus K}(B)\otimes\mathbb 1_{I\setminus K}$,
with every tensor factor at its original site.

\begin{definition}[Normalized partial-trace expectation]
    \label{def:qc_site_expectation}
    \lean{QuantumCircuit.siteExpectation, QuantumCircuit.traceOutside,
        QuantumCircuit.siteExpectationLM}\leanok
    For $K\subseteq I$, the normalized partial-trace expectation on the
    operators of $(\C^q)^{\otimes I}$ is the map $E_K$ defined above.
\end{definition}

\begin{lemma}[Weyl averaging formula]
    \label{lem:qc_site_expectation_average}
    \lean{QuantumCircuit.siteExpectation_eq_average,
        QuantumCircuit.sum_localWeyl_mul_star, QuantumCircuit.outsideWeyl}\leanok
    \uses{def:qc_site_expectation}
    Let $q\geq1$, and for $p=(a,b)$ let $W(p)=X^aZ^b$ be the Weyl operators on
    $\C^q$. Then $E_K(B)$ is the uniform average of $W_pBW_p^\dagger$ over all
    products $W_p=\bigotimes_{x\notin K}W(p_x)$ of Weyl operators outside $K$.
\end{lemma}
\begin{proof}\leanok
    The Weyl operators satisfy
    $\sum_pW(p)_{ii'}\overline{W(p)_{jj'}}=q\,\delta_{ij}\delta_{i'j'}$. The
    corresponding sum over the products factorizes over the sites, and the
    resulting Kronecker deltas reproduce the partial-trace formula.
\end{proof}

\begin{lemma}[Properties of the expectation]
    \label{lem:qc_site_expectation_properties}
    \lean{QuantumCircuit.siteExpectation_mem_supportedOperators,
        QuantumCircuit.siteExpectation_of_mem_supportedOperators,
        QuantumCircuit.siteExpectation_one, QuantumCircuit.norm_siteExpectation_le,
        QuantumCircuit.siteExpectationLM_isKrausCPTP}\leanok
    \uses{lem:qc_site_expectation_average}
    For $q\geq1$, the map $E_K$ takes values in $\mathcal A_K$, fixes every
    operator in $\mathcal A_K$, is unital, completely positive and trace
    preserving, and is a contraction in operator norm.
\end{lemma}
\begin{proof}\leanok
    By the partial-trace formula, $E_K(B)$ acts on $K$, while the Weyl
    averaging formula expresses $E_K$ as a Kraus decomposition by rescaled
    unitaries; since operators in $\mathcal A_K$ commute with every $W_p$ and
    a uniform average of unitary conjugations is contractive, all stated
    properties follow.
\end{proof}

\begin{lemma}[Localization error through on-site commutators]
    \label{lem:qc_site_expectation_commutator}
    \lean{QuantumCircuit.norm_sub_siteExpectation_le}\leanok
    \uses{lem:qc_site_expectation_average}
    If $\|[B,U]\|\leq\varepsilon_y$ for every site $y\notin K$ and every unitary
    $U\in\mathcal A_{\{y\}}$, then $\|B-E_K(B)\|\leq\sum_{y\notin K}\varepsilon_y$.
\end{lemma}
\begin{proof}\leanok
    By the averaging formula, $B-E_K(B)$ is the uniform average of
    $(BW_p-W_pB)W_p^\dagger$, and factoring $W_p$ into on-site unitaries
    together with $\|[B,UV]\|\leq\|[B,U]\|+\|[B,V]\|$ for unitaries $U,V$
    bounds each summand by $\sum_{y\notin K}\varepsilon_y$.
\end{proof}

\begin{theorem}[Localization near an arbitrary finite support]
    \label{thm:qc_graph_neighborhood_localization}
    \lean{QuantumCircuit.norm_heisenberg_sub_siteExpectation_graphNeighborhood_le}
    \leanok
    \uses{def:qc_site_expectation}
    Under the finite-range interaction assumptions above, with the chosen
    anchors $a_k\in X_k$, let $q\geq1$,
    let $K_g\in\mathbb R$ and $k\in\mathbb N$, and assume that for every
    vertex $x$ and integer $d\geq0$ at most $K_g(d+1)^k$ vertices have
    graph distance exactly $d$ from $x$. Let $S$ be any finite set of sites
    and $A\in\mathcal A_S$. For $\ell\in\mathbb N$ set
    $N_\ell(S)=\bigcup_{x\in S}N_\ell(x)$, and put
    \begin{align}
        C   & =2+4|b_0|e^{2R},\notag                  \\
        v   & =2v_R|b_0|e^{2R}+1,\notag               \\
        T_k & =\frac{2^kk!e^{1/2}}{1-e^{-1/2}}.\notag
    \end{align}
    Then for every real $t$,
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
         & \leq C|S|^2\|A\|K_gT_k e^{v|t|-\ell/2}.
        \label{eq:qc_neighborhood_localization}
    \end{align}
    No diameter bound or nonemptiness assumption on $S$ is imposed.
    The constants depend only on the interaction bounds and the sphere-growth
    bounds, not on the number of sites, the local dimension, or $S$.
    This is the natural-radius graph form of the localization estimate
    in the lemma ``Lieb--Robinson bound on a chain'' of
    \cite{TNLean2026OneDimensional}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:qc_graph_lieb_robinson_finset,lem:qc_graph_kernel_tail,
        thm:qc_graph_lieb_robinson_zero_finset,lem:qc_site_expectation_commutator}
    For each vertex $y$ define
    $w(y)=\sum_{x\in S,\ d_G(x,y)<\infty}e^{-d_G(x,y)}$.
    If some $x\in S$ has finite distance from $y$, let $n$ be the least
    such distance. The arbitrary-support propagation estimate with $\mu=1$
    and $e^{-n}\leq w(y)$ gives, for every on-site unitary $U_y$,
    \begin{align}
        \|[\tau_t(A),U_y]\|
         & \leq C|S|\|A\|e^{v|t|}w(y).\notag
    \end{align}
    If all these distances are infinite,
    Theorem~\ref{thm:qc_graph_lieb_robinson_zero_finset} makes the commutator
    zero, so the same inequality holds with $w(y)=0$.
    For $S=\varnothing$, the zero cardinality in the propagation bound gives
    the same conclusion without choosing a vertex of $S$.

    The on-site commutator estimate for the expectation therefore gives
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
         & \leq C|S|\|A\|e^{v|t|}
        \sum_{y\notin N_\ell(S)}w(y).\notag
    \end{align}
    Interchange the two finite sums. For each $x\in S$, every vertex outside
    $N_\ell(S)$ is outside $N_\ell(x)$, so the kernel-tail lemma gives
    \begin{align}
        \sum_{y\notin N_\ell(S)}w(y)
         & \leq |S|K_gT_ke^{-\ell/2}.\notag
    \end{align}
    Substitution proves (\ref{eq:qc_neighborhood_localization}).
    The sums are empty when $S=\varnothing$; otherwise a vertex of $S$
    and its zero-distance sphere give $K_g\geq0$.
\end{proof}

\paragraph{Comparison with real radii.}
For $r\geq0$, finite graph distances are integers, so
$N_r(S)=N_{\lfloor r\rfloor}(S)$, and
\begin{align}
    e^{-\lfloor r\rfloor/2} & \leq e^{1/2}e^{-r/2}.\notag
\end{align}
Thus passing from the natural-radius estimate to a real-radius estimate costs
an additional factor $e^{1/2}$ in the constant. This radius comparison is
separate from identifying a physical chain Hamiltonian and its doubled local
space with the graph interaction hypotheses.

\begin{theorem}[Localization onto graph balls]
    \label{thm:qc_graph_ball_localization}
    \lean{QuantumCircuit.norm_heisenberg_sub_siteExpectation_graphBall_le,
        QuantumCircuit.graphBall, QuantumCircuit.sum_pow_mul_exp_neg_le}\leanok
    \uses{thm:qc_graph_lieb_robinson,thm:qc_graph_lieb_robinson_zero,
        lem:qc_site_expectation_commutator,lem:qc_site_expectation_properties}
    In the setting of Theorem~\ref{thm:qc_graph_lieb_robinson}, with $q\geq1$,
    assume that at most $K_g(d+1)^k$ sites lie at distance exactly $d$ from any
    site. Let $N_l(a)=\{x:d_G(a,x)\leq l\}$. Then
    \[
        \|\tau_t(A)-E_{N_l(a_i)}(\tau_t(A))\|
        \leq\|A\|\min\bigl\{2,\;Ce^{v|t|-l/2}\bigr\},
    \]
    with $v=2v_Rb_0e^{2R}$ and $C=2e^RK_g\,2^kk!\,e^{1/2}/(1-e^{-1/2})$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_graph_kernel_tail}
    Contractivity of $E_{N_l(a_i)}$ gives the bound $2\|A\|$. For the
    exponential bound, applying the commutator localization lemma with
    $\varepsilon_y=0$ when $d_G(a_i,y)=\infty$ and
    $\varepsilon_y=2e^R\|A\|e^{v|t|-d_G(a_i,y)}$ otherwise (via graph-distance
    propagation with $\mu=1$), grouping the sites outside the ball by their
    distance $d>l$, and using $(d+1)^k\leq2^kk!\,e^{(d+1)/2}$ together with a
    geometric series yields $Ce^{v|t|-l/2}\|A\|$.
\end{proof}
