\section{Lieb--Robinson propagation on finite graphs}
\label{sec:qc_lieb_robinson}

\begin{theorem}[Extended distance under graph maps]
    \label{thm:qc_graph_edist_transport}
    \lean{SimpleGraph.Hom.edist_le, SimpleGraph.Iso.edist_eq,
        SimpleGraph.ConnectedComponent.edist_eq}
    \leanok
    Let $G,H$ be simple graphs on arbitrary vertex sets with extended path metrics $d_G,d_H\in\mathbb N\cup\{\infty\}$. Every graph homomorphism $f\colon G\to H$ satisfies $d_H(f(x),f(y))\le d_G(x,y)$, with equality when $f$ is an isomorphism. For any connected component $C$ of $G$ and $x,y\in C$, $d_{G[C]}(x,y)=d_G(x,y)$.
\end{theorem}
\begin{proof}
    \leanok
    A homomorphism maps walks of length $\ell$ to walks of length $\ell$, so taking infima (with $\inf\varnothing=\infty$) gives $d_H(f(x),f(y))\le d_G(x,y)$, and applying this to $f$ and $f^{-1}$ gives equality for isomorphisms. Every walk in $G$ starting at $x\in C$ stays in $C$, so walks between $x,y\in C$ in $G$ and $G[C]$ coincide.
\end{proof}

Let $I$ be a finite set of sites of local dimension $q$, and write
$\mathcal A_X$ for the operators on $(\C^q)^{\otimes I}$ acting on
$X\subseteq I$. All norms are Hilbert-space operator norms. For an operator
$H$ put $\tau_t(A)=e^{itH}Ae^{-itH}$. The dynamics generated by a sum of
local Hermitian terms moves support at a bounded speed, up to exponentially
small tails \cite{Hastings2006Spectral,Nachtergaele2006LiebRobinson}. The
estimate below measures distance by the path metric of an arbitrary simple
graph on $I$, so that holes and separate connected components cost nothing:
sites in different components are at infinite distance, and there the
commutator vanishes exactly.

\begin{definition}[Restricted commutator norm]
    \label{aux:restricted-commutator-norm}
    \lean{QuantumCircuit.heisenbergCommutatorNorm,
        QuantumCircuit.heisenbergEvolution}\leanok
    For operators $H,B$ and $X\subseteq I$, define
    \begin{align}
        C_B(X,t) & =\sup\{\|[\tau_t(A),B]\|: A\in\mathcal A_X,\ \|A\|\leq1\}.\notag
    \end{align}
\end{definition}

\begin{lemma}[Evaluation of the restricted commutator norm]
    \label{aux:restricted-commutator-evaluation}
    \lean{QuantumCircuit.norm_heisenberg_commutator_le_heisenbergCommutatorNorm}\leanok
    \uses{aux:restricted-commutator-norm}
    For $A\in\mathcal A_X$, $\|[\tau_t(A),B]\|\leq C_B(X,t)\|A\|$.
\end{lemma}
\begin{proof}\leanok
    Immediate from Definition~\ref{aux:restricted-commutator-norm} by homogeneity of $A\mapsto[\tau_t(A),B]$.
\end{proof}

\begin{lemma}[Reversal of time]
    \label{aux:restricted-commutator-negative-time}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_neg_time}\leanok
    \uses{aux:restricted-commutator-norm}
    $C_B^{(H)}(X,-t)=C_B^{(-H)}(X,t)$.
\end{lemma}
\begin{proof}\leanok
    Both Heisenberg evolutions equal $A\mapsto e^{-itH}Ae^{itH}$.
\end{proof}

\begin{lemma}[Continuity in time]
    \label{aux:restricted-commutator-continuity}
    \lean{QuantumCircuit.continuous_heisenbergCommutatorNorm}\leanok
    \uses{aux:restricted-commutator-norm}
    For fixed $H,B,X$, $t\mapsto C_B(X,t)$ is continuous.
\end{lemma}
\begin{proof}\leanok
    Continuity of $t\mapsto e^{\pm itH}$ implies continuity of the operator norm of $A\mapsto[\tau_t(A),B]$ on $\mathcal A_X$.
\end{proof}

\begin{lemma}[Initial commutator bound]
    \label{aux:restricted-initial-commutator-bound}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_zero_le}\leanok
    \uses{aux:restricted-commutator-norm}
    For arbitrary $H,B,X$, $C_B(X,0)\leq2\|B\|$.
\end{lemma}
\begin{proof}\leanok
    For $\|A\|\le1$, $\|[A,B]\|\le2\|A\|\|B\|\le2\|B\|$.
\end{proof}

\begin{lemma}[Vanishing initial commutator]
    \label{aux:restricted-initial-commutator-vanishing}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_zero_of_disjoint}\leanok
    \uses{aux:restricted-commutator-norm}
    If $B\in\mathcal A_Y$ and $X\cap Y=\varnothing$, then $C_B(X,0)=0$.
\end{lemma}
\begin{proof}\leanok
    Operators on disjoint supports commute.
\end{proof}

\begin{lemma}[Local commutator recursion]
    \label{aux:restricted-commutator-recursion}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_integral}\leanok
    \uses{aux:restricted-commutator-norm}
    Following Hastings--Koma, Appendix A, (A.12)--(A.14), let $H=\sum_j h_j$ with $h_j=h_j^\dagger\in\mathcal A_{T_j}$. For arbitrary $B,X$ and $t\geq0$,
    \begin{align}
        C_B(X,t) & \leq C_B(X,0)
        +2\sum_{j:X\cap T_j\neq\varnothing}\|h_j\|
        \int_0^t C_B(T_j,s)\,ds.\notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Let $H_{\rm dis}=\sum_{T_j\cap X=\varnothing}h_j$, which commutes with $\mathcal A_X$. Differentiating $[\tau_t(A),B]$ in the interaction picture of $H-H_{\rm dis}$, integrating over $[0,t]$, and applying unitary invariance of the operator norm and Lemma~\ref{aux:restricted-commutator-evaluation} yields the bound.
\end{proof}

\begin{lemma}[Weighted integral comparison]
    \label{aux:weighted-volterra-comparison}
    \lean{finite_volterra_le_weight_mul_exp,
        finite_volterra_zero_le_weight_mul_exp_sub_one,
        finite_rectangular_volterra_zero_le_weight_mul_exp_sub_one,
        le_mul_exp_of_le_const_add_integral}\leanok
    Let $f_i\geq0$ ($i\in J$ finite) be continuous on $[0,T]$ and satisfy $f_i(t)\leq b_i+\sum_j a_{ij}\int_0^tf_j(s)\,ds$ with $a_{ij}\geq0$. If $w_i>0$ satisfy $\sum_ja_{ij}w_j\leq vw_i$ and $b_i\leq cw_i$, then $f_i(t)\leq cw_ie^{vt}$ for $t\in[0,T]$; if $b_i=0$ and $v>0$, then $f_i(t)\leq cw_i(e^{vt}-1)$.
\end{lemma}
\begin{proof}\leanok
    The function $g(t)=\max_i f_i(t)/w_i$ satisfies $g(t)\le c+v\int_0^t g(s)\,ds$, so Gr\"onwall's inequality gives $g(t)\le ce^{vt}$; for $b_i=0$, integrating $f_j(s)\le cw_j e^{vs}$ once gives $cw_i(e^{vt}-1)$.
\end{proof}

\begin{lemma}[Exact locality on interaction-closed regions]
    \label{lem:qc_lr_closed_region}
    \lean{QuantumCircuit.commute_heisenbergEvolution_of_closed}\leanok
    Let $H=\sum_jh_j$ with $h_j\in\mathcal A_{T_j}$, and let $C\subseteq I$ satisfy $T_j\subseteq C$ or $T_j\cap C=\varnothing$ for all $j$. If $A\in\mathcal A_C$, $B\in\mathcal A_D$, and $C\cap D=\varnothing$, then $[\tau_t(A),B]=0$ for all $t\in\mathbb R$.
\end{lemma}
\begin{proof}\leanok
    Splitting $H=H_C+H_{C^c}$ with $H_C=\sum_{T_j\subseteq C}h_j\in\mathcal A_C$ and $H_{C^c}\in\mathcal A_{C^c}$ gives $[H_C,H_{C^c}]=0$ and $[H_{C^c},A]=0$, so $\tau_t(A)=e^{itH_C}Ae^{-itH_C}\in\mathcal A_C$ commutes with $B\in\mathcal A_D$.
\end{proof}

\paragraph{Finite-range interactions on a graph.}
Let $G$ be a simple graph on $I$ with path metric $d_G\in\mathbb N\cup\{\infty\}$. Let $H=\sum_kh_k$ with $h_k=h_k^\dagger\in\mathcal A_{X_k}$ and anchors $a_k\in X_k$ such that $\operatorname{diam}_{d_G}(X_k)\le R$, $|X_k|\le v_R$, and $\sum_{k:x\in X_k}\|h_k\|\le b_0$ for all $x\in I$.

\begin{theorem}[Graph-distance propagation]
    \label{thm:qc_graph_lieb_robinson}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_graphDistance,
        QuantumCircuit.norm_heisenberg_commutator_le_graphDistance,
        QuantumCircuit.exists_graph_lieb_robinson}\leanok
    \uses{aux:restricted-commutator-recursion,aux:weighted-volterra-comparison,
        aux:restricted-commutator-evaluation,aux:restricted-commutator-negative-time,
        aux:restricted-initial-commutator-bound,
        aux:restricted-initial-commutator-vanishing,
        aux:restricted-commutator-continuity,thm:qc_graph_lieb_robinson_zero}
    For $A\in\mathcal A_{X_i}$, $B\in\mathcal A_{\{y\}}$, $\mu\geq0$, $0\leq n\leq d_G(a_i,y)$, and $t\in\mathbb R$,
    \begin{align}
        C_B(X_i,t)        & \leq 2e^{\mu R}\|B\|
        \exp\bigl(2v_Rb_0e^{2\mu R}|t|-\mu n\bigr),\notag \\
        \|[\tau_t(A),B]\| & \leq 2e^{\mu R}\|A\|\|B\|
        \exp\bigl(2v_Rb_0e^{2\mu R}|t|-\mu n\bigr).\notag
    \end{align}
    Consequently, with $C=2e^R$, $v=2v_Rb_0e^{2R}$, and $c=1$, $\|[\tau_t(A),B]\|\leq C\|A\|\|B\|e^{v|t|-cd_G(a_i,y)}$ when $d_G(a_i,y)<\infty$, and $[\tau_t(A),B]=0$ when $d_G(a_i,y)=\infty$.
\end{theorem}
\begin{proof}\leanok
    For $t\ge0$, $F_k(t)=C_B(X_k,t)$ satisfies Lemma~\ref{aux:restricted-commutator-recursion} with $a_{kj}=2\|h_j\|\mathbf 1_{X_j\cap X_k\ne\varnothing}$. With $D(x)=\min\{n,d_G(x,y)\}$ and $w_k=e^{-\mu D(a_k)}$, $X_j\cap X_k\ne\varnothing\implies d_G(a_k,a_j)\le2R\implies w_j\le e^{2\mu R}w_k$, and $\sum_{X_j\cap X_k\ne\varnothing}\|h_j\|\le|X_k|b_0\le v_Rb_0$, so $\sum_j a_{kj}w_j\le 2v_Rb_0e^{2\mu R}w_k$. By Lemmas~\ref{aux:restricted-initial-commutator-bound} and~\ref{aux:restricted-initial-commutator-vanishing}, $F_k(0)\le 2\|B\|e^{\mu R}w_k$, so Lemma~\ref{aux:weighted-volterra-comparison} gives the bound at $k=i$ ($w_i=e^{-\mu n}$). Lemma~\ref{aux:restricted-commutator-negative-time} covers $t<0$, Lemma~\ref{aux:restricted-commutator-evaluation} bounds $\|[\tau_t(A),B]\|$, $\mu=1$ gives $(C,v,c)$, and Theorem~\ref{thm:qc_graph_lieb_robinson_zero} handles $d_G(a_i,y)=\infty$.
\end{proof}

\begin{theorem}[Graph-distance propagation from an arbitrary finite support]
    \label{thm:qc_graph_lieb_robinson_finset}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_graphDistance_finset,
        QuantumCircuit.norm_heisenberg_commutator_le_graphDistance_finset}\leanok
    \uses{aux:restricted-commutator-norm}
    Under the preceding finite-range interaction assumptions, let $S\subseteq I$, $A\in\mathcal A_S$, $B\in\mathcal A_{\{y\}}$, $\mu\geq0$, and $0\le n\leq\min_{x\in S}d_G(x,y)$. With $C=2+4|b_0|e^{2\mu R}$ and $v=2v_R|b_0|e^{2\mu R}+1$, for all $t\in\mathbb R$,
    \begin{align}
        C_B(S,t)          & \leq C|S|\|B\|e^{v|t|-\mu n},
        \label{eq:qc_lr_finset_norm}                           \\
        \|[\tau_t(A),B]\| & \leq C|S|\|A\|\|B\|e^{v|t|-\mu n}.
        \label{eq:qc_lr_finset_commutator}
    \end{align}
    Equation~(\ref{eq:qc_lr_finset_commutator}) has the form used in the lemma ``Lieb--Robinson bound on a chain'' of~\cite{TNLean2026OneDimensional}.
\end{theorem}
\begin{proof}\leanok
    \uses{aux:restricted-commutator-recursion,thm:qc_graph_lieb_robinson,
        aux:restricted-initial-commutator-bound,
        aux:restricted-initial-commutator-vanishing,
        aux:restricted-commutator-continuity,
        aux:restricted-commutator-evaluation,
        aux:restricted-commutator-negative-time}
    Write $b=|b_0|$ and $\lambda=2v_Rbe^{2\mu R}$, and take $t\geq0$. We have $\sum_{X_k\cap S\ne\varnothing}\|h_k\|\leq |S|b$, and for $x\in S\cap X_k$, $d_G(x,a_k)\le R\implies\max(0,n-R)\le d_G(a_k,y)$, so Theorem~\ref{thm:qc_graph_lieb_robinson} gives $C_B(X_k,s)\le 2\|B\|e^{2\mu R-\mu n}e^{\lambda s}$ for $s\ge0$. Since $C_B(S,0)\le 2|S|\|B\|e^{-\mu n}$, Lemma~\ref{aux:restricted-commutator-recursion} and $1,t\le e^t$ give
    \begin{align}
        C_B(S,t)
         & \leq |S|\|B\|e^{-\mu n}
        \bigl(2+4be^{2\mu R}t e^{\lambda t}\bigr)
        \leq (2+4be^{2\mu R})|S|\|B\|
        e^{(\lambda+1)t-\mu n}.\notag
    \end{align}
    Lemmas~\ref{aux:restricted-commutator-negative-time} and~\ref{aux:restricted-commutator-evaluation} yield~(\ref{eq:qc_lr_finset_norm})--(\ref{eq:qc_lr_finset_commutator}).
\end{proof}

\begin{theorem}[Propagation between finite supports]
    \label{thm:qc_lr_finsets_numeric}
    \lean{QuantumCircuit.heisenbergCommutatorNorm_le_graphDistance_finsets_numeric,
        QuantumCircuit.norm_heisenberg_commutator_le_graphDistance_finsets_numeric}\leanok
    \uses{aux:restricted-commutator-norm}
    Under the assumptions of Theorem~\ref{thm:qc_graph_lieb_robinson_finset} with $v_R\geq1$, let $S,Y\subseteq I$, $A\in\mathcal A_S$, $B\in\mathcal A_Y$, $\mu\geq0$, and $0\le n\leq d_G(S,Y)$. With $u=4v_R|b_0|e^{2\mu R}$, for all $t\in\mathbb R$,
    \begin{align}
        C_B(S,t)          & \leq4|S|\|B\|e^{u|t|-\mu n},\notag      \\
        \|[\tau_t(A),B]\| & \leq4|S|\|A\|\|B\|e^{u|t|-\mu n}.\notag
    \end{align}
    This captures the $|S|$-linear, $|Y|$-independent support dependence of~\cite[eq.~(1)]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{aux:weighted-volterra-comparison,aux:restricted-commutator-recursion,
        aux:restricted-initial-commutator-bound,
        aux:restricted-initial-commutator-vanishing,
        aux:restricted-commutator-continuity,
        aux:restricted-commutator-evaluation,
        aux:restricted-commutator-negative-time}
    Put $\delta_Y(x)=\min_{y\in Y}d_G(x,y)$ ($\infty$ if $Y=\varnothing$) and $w_k=e^{-\mu\min(n,\delta_Y(a_k))}$. By the triangle inequality, $X_j\cap X_k\ne\varnothing\implies w_j\le e^{2\mu R}w_k$, and $X_k\cap Y=\varnothing\implies C_B(X_k,0)=0$, so Lemma~\ref{aux:weighted-volterra-comparison} gives $C_B(X_k,s)\le 2\|B\|e^{\mu R}w_k e^{\lambda s}$ ($\lambda=2v_R|b_0|e^{2\mu R}$). Applying Lemma~\ref{aux:restricted-commutator-recursion} on $S$ and $4|b_0|e^{2\mu R}t\le 2\lambda t\le 2e^{\lambda t}$ (since $v_R\ge1$) gives $C_B(S,t)\le |S|\|B\|e^{-\mu n}(2+2e^{2\lambda t})\le 4|S|\|B\|e^{ut-\mu n}$ for $t\ge0$; Lemmas~\ref{aux:restricted-commutator-negative-time} and~\ref{aux:restricted-commutator-evaluation} complete the proof.
\end{proof}

\begin{theorem}[Exact commutation across disconnected components]
    \label{thm:qc_graph_lieb_robinson_zero_finset}
    \lean{QuantumCircuit.heisenberg_commutator_eq_zero_of_forall_edist_eq_top}
    \leanok
    \uses{aux:restricted-commutator-norm}
    Let $H=\sum_k h_k$ with $h_k\in\mathcal A_{X_k}$ and $\operatorname{diam}_{d_G}(X_k)\le R<\infty$. If $A\in\mathcal A_S$, $B\in\mathcal A_{\{y\}}$, and $d_G(x,y)=\infty$ for all $x\in S$, then $[\tau_t(A),B]=0$ for all $t\in\mathbb R$~\cite[Lemma~4.1]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_lr_closed_region}
    The set $C=\{x\in I:d_G(x,y)=\infty\}$ contains $S$, excludes $y$, and satisfies $X_k\subseteq C$ or $X_k\cap C=\varnothing$ since $\operatorname{diam}_{d_G}(X_k)<\infty$, so Lemma~\ref{lem:qc_lr_closed_region} gives $[\tau_t(A),B]=0$.
\end{proof}

\begin{theorem}[Vanishing between connected components]
    \label{thm:qc_graph_lieb_robinson_zero}
    \lean{QuantumCircuit.heisenberg_commutator_eq_zero_of_edist_eq_top}\leanok
    \uses{thm:qc_graph_lieb_robinson_zero_finset}
    If $H=\sum_kh_k$ with $\operatorname{diam}_{d_G}(X_k)\le R$, $a_k\in X_k$, $A\in\mathcal A_{X_i}$, $B\in\mathcal A_{\{y\}}$, and $d_G(a_i,y)=\infty$, then $[\tau_t(A),B]=0$ for all $t\in\mathbb R$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:qc_graph_lieb_robinson_zero_finset}
    Since $\operatorname{diam}_{d_G}(X_i)\le R$, $d_G(a_i,y)=\infty\implies d_G(x,y)=\infty$ for all $x\in X_i$, so Theorem~\ref{thm:qc_graph_lieb_robinson_zero_finset} applies with $S=X_i$.
\end{proof}

\begin{lemma}[Exponential kernel outside a graph ball]
    \label{lem:qc_graph_kernel_tail}
    \lean{QuantumCircuit.sum_exp_neg_edist_compl_graphBall_le}
    \leanok
    Let $G$ be a simple graph on a finite set $I$, $x\in I$, $K_g\in\mathbb R$, and $k\in\mathbb N$ such that $|\{y:d_G(x,y)=d\}|\le K_g(d+1)^k$ for all $d\in\mathbb N$. With $N_\ell(x)=\{y:d_G(x,y)\leq\ell\}$ and $T_k=2^kk!e^{1/2}/(1-e^{-1/2})$,
    \begin{align}
        \sum_{\substack{y\notin N_\ell(x) \\d_G(x,y)<\infty}}
        e^{-d_G(x,y)}
         & \leq K_gT_ke^{-\ell/2},
        \label{eq:qc_kernel_tail}
    \end{align}
    as in~\cite[Lemma~4.1]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}
    \leanok
    Evaluating at $d=0$ gives $K_g\ge0$. Grouping by $d=d_G(x,y)>\ell$ and using $(d+1)^ke^{-(d+1)/2}\le 2^kk!$ gives $(d+1)^ke^{-d}\le 2^kk!e^{1/2}e^{-d/2}$, so summing the geometric series over $d\ge\ell+1$ proves~(\ref{eq:qc_kernel_tail}).
\end{proof}

\paragraph{Normalized partial traces.}
For $K\subseteq I$ let
$E_K(B)=q^{-|I\setminus K|}\operatorname{Tr}_{I\setminus K}(B)\otimes\mathbb 1_{I\setminus K}$,
with every tensor factor at its original site.

\begin{definition}[Normalized partial-trace expectation]
    \label{def:qc_site_expectation}
    \lean{QuantumCircuit.siteExpectation, QuantumCircuit.traceOutside,
        QuantumCircuit.siteExpectationLM}\leanok
    For $K\subseteq I$, the normalized partial-trace expectation on
    $(\C^q)^{\otimes I}$ is $E_K(B)=q^{-|I\setminus K|}\operatorname{Tr}_{I\setminus K}(B)\otimes\mathbb 1_{I\setminus K}$.
\end{definition}

\begin{lemma}[Weyl averaging formula]
    \label{lem:qc_site_expectation_average}
    \lean{QuantumCircuit.siteExpectation_eq_average,
        QuantumCircuit.sum_localWeyl_mul_star, QuantumCircuit.outsideWeyl}\leanok
    \uses{def:qc_site_expectation}
    For $q\geq1$ and Weyl operators $W(a,b)=X^aZ^b$ on $\C^q$,
    \begin{align}
        E_K(B) & =q^{-2|I\setminus K|}\sum_{p}W_pBW_p^\dagger, &
        W_p    & =\bigotimes_{x\notin K}W(p_x).\notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $\sum_p W(p)_{ii'}\overline{W(p)_{jj'}}=q\,\delta_{ij}\delta_{i'j'}$, the product over $x\notin K$ reproduces $q^{-|I\setminus K|}\operatorname{Tr}_{I\setminus K}(B)\otimes\mathbb 1_{I\setminus K}$.
\end{proof}

\begin{lemma}[Properties of the expectation]
    \label{lem:qc_site_expectation_properties}
    \lean{QuantumCircuit.siteExpectation_mem_supportedOperators,
        QuantumCircuit.siteExpectation_of_mem_supportedOperators,
        QuantumCircuit.siteExpectation_one, QuantumCircuit.norm_siteExpectation_le,
        QuantumCircuit.siteExpectationLM_isKrausCPTP}\leanok
    \uses{lem:qc_site_expectation_average}
    For $q\geq1$, $E_K$ is a unital CPTP map onto $\mathcal A_K$ satisfying $E_K|_{\mathcal A_K}=\operatorname{id}$ and $\|E_K(B)\|\le\|B\|$.
\end{lemma}
\begin{proof}\leanok
    By Definition~\ref{def:qc_site_expectation}, $\operatorname{im}E_K\subseteq\mathcal A_K$. By Lemma~\ref{lem:qc_site_expectation_average}, $E_K$ is a uniform average of unitary conjugations $W_p\cdot W_p^\dagger$ with $W_p\in\mathcal A_{K^c}$, so it is unital, CPTP, contractive, and fixes $\mathcal A_K$.
\end{proof}

\begin{theorem}[Comparison with a supported approximant]
    \label{thm:qc_site_expectation_supported_approximant}
    \lean{QuantumCircuit.norm_sub_siteExpectation_le_two_mul_norm_sub_of_mem_supportedOperators}
    \leanok
    \uses{def:qc_site_expectation}
    For $q\ge1$, any operator $A$, and $A_0\in\mathcal A_B$,
    \begin{align}
        \|A-E_B(A)\| & \le2\|A-A_0\|.\notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_site_expectation_properties}
    By Lemma~\ref{lem:qc_site_expectation_properties}, $\|A-E_B(A)\|=\|(A-A_0)-E_B(A-A_0)\|\le\|A-A_0\|+\|E_B(A-A_0)\|\le2\|A-A_0\|$.
\end{proof}

\begin{lemma}[Localization error through on-site commutators]
    \label{lem:qc_site_expectation_commutator}
    \lean{QuantumCircuit.norm_sub_siteExpectation_le}\leanok
    \uses{lem:qc_site_expectation_average}
    If $\|[B,U]\|\leq\varepsilon_y$ for every $y\notin K$ and unitary $U\in\mathcal A_{\{y\}}$, then $\|B-E_K(B)\|\leq\sum_{y\notin K}\varepsilon_y$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{lem:qc_site_expectation_average}, $B-E_K(B)=q^{-2|I\setminus K|}\sum_p[B,W_p]W_p^\dagger$, and $\|[B,\bigotimes_{y\notin K}W(p_y)]\|\le\sum_{y\notin K}\|[B,W(p_y)]\|\le\sum_{y\notin K}\varepsilon_y$.
\end{proof}

\begin{theorem}[Localization near an arbitrary finite support]
    \label{thm:qc_graph_neighborhood_localization}
    \lean{QuantumCircuit.norm_heisenberg_sub_siteExpectation_graphNeighborhood_le}
    \leanok
    \uses{def:qc_site_expectation}
    Under the finite-range interaction and sphere-growth bounds $|\{y:d_G(x,y)=d\}|\le K_g(d+1)^k$ above, let $q\geq1$, $S\subseteq I$, $A\in\mathcal A_S$, $\ell\in\mathbb N$, and $N_\ell(S)=\bigcup_{x\in S}N_\ell(x)$. With $C=2+4|b_0|e^{2R}$, $v=2v_R|b_0|e^{2R}+1$, and $T_k=2^kk!e^{1/2}/(1-e^{-1/2})$, for all $t\in\mathbb R$,
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
         & \leq C|S|^2\|A\|K_gT_k e^{v|t|-\ell/2}.
        \label{eq:qc_neighborhood_localization}
    \end{align}
    This is the natural-radius graph form of the localization estimate in ``Lieb--Robinson bound on a chain'' of~\cite{TNLean2026OneDimensional}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:qc_graph_lieb_robinson_finset,lem:qc_graph_kernel_tail,
        thm:qc_graph_lieb_robinson_zero_finset,lem:qc_site_expectation_commutator}
    Put $w(y)=\sum_{x\in S,\,d_G(x,y)<\infty}e^{-d_G(x,y)}$. By Theorems~\ref{thm:qc_graph_lieb_robinson_finset} (with $\mu=1$ and $n=\min_{x\in S}d_G(x,y)$) and~\ref{thm:qc_graph_lieb_robinson_zero_finset}, every unitary $U_y\in\mathcal A_{\{y\}}$ satisfies $\|[\tau_t(A),U_y]\|\le C|S|\|A\|e^{v|t|}w(y)$. Applying Lemmas~\ref{lem:qc_site_expectation_commutator} and~\ref{lem:qc_graph_kernel_tail} gives
    \begin{align}
        \|\tau_t(A)-E_{N_\ell(S)}(\tau_t(A))\|
         & \leq C|S|\|A\|e^{v|t|}\sum_{x\in S}\sum_{\substack{y\notin N_\ell(S) \\d_G(x,y)<\infty}}e^{-d_G(x,y)}
        \leq C|S|^2\|A\|K_gT_ke^{v|t|-\ell/2}.\notag
    \end{align}
\end{proof}

\paragraph{Comparison with real radii.}
For $r\geq0$, finite graph distances are integers, so
$N_r(S)=N_{\lfloor r\rfloor}(S)$, and
\begin{align}
    e^{-\lfloor r\rfloor/2} & \leq e^{1/2}e^{-r/2}.\notag
\end{align}
Thus passing from the natural-radius estimate to a real-radius estimate costs
an additional factor $e^{1/2}$ in the constant. This radius comparison is
separate from identifying a physical chain Hamiltonian and its doubled local
space with the graph interaction hypotheses.

\begin{theorem}[Localization onto graph balls]
    \label{thm:qc_graph_ball_localization}
    \lean{QuantumCircuit.norm_heisenberg_sub_siteExpectation_graphBall_le,
        QuantumCircuit.graphBall, QuantumCircuit.sum_pow_mul_exp_neg_le}\leanok
    \uses{thm:qc_graph_lieb_robinson,thm:qc_graph_lieb_robinson_zero,
        lem:qc_site_expectation_commutator,lem:qc_site_expectation_properties}
    In the setting of Theorem~\ref{thm:qc_graph_lieb_robinson} with $q\geq1$ and $|\{y:d_G(x,y)=d\}|\le K_g(d+1)^k$, for $N_l(a)=\{x:d_G(a,x)\leq l\}$,
    \begin{align}
        \|\tau_t(A)-E_{N_l(a_i)}(\tau_t(A))\|
         & \leq\|A\|\min\bigl\{2,\;Ce^{v|t|-l/2}\bigr\},\notag
    \end{align}
    with $v=2v_Rb_0e^{2R}$ and $C=2e^RK_g\,2^kk!\,e^{1/2}/(1-e^{-1/2})$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_graph_kernel_tail}
    Contractivity of $E_{N_l(a_i)}$ (Lemma~\ref{lem:qc_site_expectation_properties}) gives $2\|A\|$, while Theorems~\ref{thm:qc_graph_lieb_robinson} ($\mu=1$) and~\ref{thm:qc_graph_lieb_robinson_zero} together with Lemmas~\ref{lem:qc_site_expectation_commutator} and~\ref{lem:qc_graph_kernel_tail} give $Ce^{v|t|-l/2}\|A\|$.
\end{proof}

\begin{lemma}[Strict margins contain closed graph balls]
    \label{lem:qc_graph_ball_strict_margin}
    \lean{QuantumCircuit.graphBall_subset_of_finite_distance_margin}\leanok
    If $m\in\mathbb N$ and $w\in\mathbb R$ satisfy $m\le w$, and $w<d_G(x,y)$ whenever $y\notin D$ and $d_G(x,y)<\infty$, then $N_m(x)=\{y:d_G(x,y)\le m\}\subseteq D$.
\end{lemma}
\begin{proof}\leanok
    If $d_G(x,y)\le m$ and $y\notin D$, then $d_G(x,y)<\infty$, giving the contradiction $w<d_G(x,y)\le m\le w$.
\end{proof}

\begin{theorem}[Crossing supports avoid interior cores]
    \label{thm:qc_graph_support_separation}
    \lean{QuantumCircuit.reset_graphNeighborhood_disjoint_cores}\leanok
    Let $(X_j)_{j\in J}$ be subsets of $I$ with $\operatorname{diam}_{d_G}(X_j)\le R$, let $D\cap C=\varnothing$, and suppose $N_{\ell+R}(R_0)\subseteq D$ and $N_{\ell+R}(C_0)\subseteq C$. With $Q=(D\cup C)^c$, $\mathcal C=\{j\in J:X_j\cap D\ne\varnothing,\ X_j\setminus D\ne\varnothing\}$, and $Z=\bigcup_{j\in\mathcal C}X_j$, one has
    \begin{align}
        N_\ell(Q\cup Z)\cap(R_0\cup C_0) & =\varnothing,\notag
    \end{align}
    realizing the support separation in the information-reset lemma of~\cite[Part~II]{TNLean2026OneDimensional}.
\end{theorem}
\begin{proof}\leanok
    Suppose $a\in R_0\cup C_0$ and $d_G(a,z)\le\ell$ for $z\in Q\cup Z$. If $z\in Q$, then $z\in N_{\ell+R}(\{a\})\subseteq D\cup C$, a contradiction. If $z\in X_j$ ($j\in\mathcal C$), pick $x\in X_j\cap D$ and $y\in X_j\setminus D$; then $d_G(a,x),d_G(a,y)\le\ell+R$, forcing $y\in D$ if $a\in R_0$ and $x\in C$ if $a\in C_0$, contradicting $y\notin D$ and $D\cap C=\varnothing$.
\end{proof}
