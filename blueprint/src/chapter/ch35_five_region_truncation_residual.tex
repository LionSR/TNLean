\section{A physical low cluster with the required parameter bounds}

\begin{theorem}[One scale for the original ground space and clipped cluster]
    \label{thm:five_region_dg_cluster_parameters}
    \lean{TNLean.MPS.AreaLaw.exists_five_region_dg_cluster_parameters}
    \uses{def:chain_model,def:five_region_bond_partition}
    There is a real constant $A\ge21$ with the following property.
    For every $q\ge2$ and $0<g\le1$, put
    \begin{align}
        u          & =1+\log q+\log(1/g), & \ell & =\lceil Au^2/g\rceil,\notag \\
        \mathcal D & =(q\ell^2)^{22\ell}, & c_A  & =6+2\log(A+1).
    \end{align}
    Then $\ell\ge21$, $\mathcal D\ge2$ and
    $\log\mathcal D\le22(A+1)c_Ag^{-1}u^3$.
    Let $a+2\ell\le b+1$ and $b+2\ell\le L$, and let
    $H=\sum_{i\in\mathcal B_L}h_i$ be an original nearest-neighbour
    Hamiltonian with supported terms $0\le h_i\le\Id$.
    Suppose $\mu\in\operatorname{spec}(H)$ and
    $\operatorname{spec}(H)\subseteq\{\mu\}\cup[\mu+g,\infty)$.
    Write $G=\ker(H-\mu\Id)$, and use the five cuts
    $(0,a+1,a+2\ell,b+1,b+2\ell,L)$.
    There are original regional minima $e_j$ with a common unit ground vector
    and a number $0\le E_0\le4$ with $E_0+\sum_j e_j=\mu$.
    Put $K=\{0,2,4\}$ and
    \begin{align}
        B      & =\sum_{j\in K}\bigl(\min(A_j,e_j+\ell)-e_j\Id\bigr)
        +\sum_{j\notin K}(A_j-e_j\Id)+Y,\notag                                                      \\
        \eta_0 & =E_0+g/10,                                            & \eta_1 & =E_0+9g/10,\notag \\
        R      & =7\ell-\eta_0,                                        &
        \Delta & =\bigl(2e^{-2\ell^2\sqrt{(\eta_1-\eta_0)/R}}\bigr)^2.
    \end{align}
    One has $\|B\|\le7\ell$, $\eta_0<\eta_1\le7\ell$, $0<R\le8\ell$,
    $\Delta>0$ and $\mathcal D^{12}\Delta\le10^{-5}$.
    There is an actual orthonormal eigenbasis of $B$ for which
    \begin{align}
        T=\mathbf1_{\{x\le\eta_0\}}(B)\mathcal H_{L,q}
        =\mathbf1_{\{x\le\eta_1\}}(B)\mathcal H_{L,q}.
    \end{align}
    Every closed low cutoff between $E_0+g/20$ and $E_0+19g/20$ gives $T$.
    Moreover $\dim T=\dim G$, $T,G$ are mutually $1/200$-close, and every
    eigenvalue of $B$ is at most $\eta_0$ or at least $\eta_1$.
\end{theorem}
\begin{proof}
    \uses{thm:disjoint_regional_clipping,thm:five_region_original_bond_bounds,
        thm:five_region_seven_ell_budget,thm:consecutive_regional_truncation_low,
        thm:physical_dg_scalar_parameters,thm:physical_chebyshev_endpoints}
    The original regional minima and the common vector give the centered
    ground energy $E_0\in[0,4]$. The two uncut buffers have at most
    $2\ell-2$ internal bonds each, and at most four bonds cross the cuts;
    the total clipped budget is therefore $7\ell$.
    Apply the regional restricted-error estimate at each of the three
    clipped regions with $s=1/8$ and $u=19+\varepsilon$.
    The resulting total is at most
    $288e^{11/8}e^{-(\ell-g)/8}\le10^{-6}g$.
    The two actual restricted residuals, at the centered original cutoff $E_0+g/40$
    and the clipped cutoff $E_0+19g/20$, give the equal-rank cluster and
    mutual closeness by the low-cluster theorem of
    \cite[``A low cluster from two restricted residuals'']{QICLean2026Blueprint}.
    Apply the same scalar choice to its Chebyshev endpoints to obtain
    $\mathcal D^{12}\Delta\le10^{-5}$.
    The comparison uses the full original ground space and the closed
    cutoffs described in~\cite{gap:alvv16_spectral_projector_boundary}.
\end{proof}

\begin{theorem}[The physical cluster at a prescribed projection accuracy]
    \label{thm:five_region_scaled_low_cluster}
    \lean{TNLean.MPS.AreaLaw.exists_five_region_physical_scaled_low_cluster}
    \uses{def:chain_model,def:five_region_bond_partition}
    Let $q\ge1$, $\ell\ge21$, $a+2\ell\le b+1$ and $b+2\ell\le L$.
    Let $H=\sum_{i\in\mathcal B_L}h_i$ be the original nearest-neighbour
    Hamiltonian with supported terms $0\le h_i\le\Id$.
    Suppose $\mu\in\operatorname{spec}(H)$, $0<g\le1$, and
    $\operatorname{spec}(H)\subseteq\{\mu\}\cup[\mu+g,\infty)$.
    Put $G=\ker(H-\mu\Id)$ and assume
    \begin{align}
        0<\tau                     & \le1,\notag                   \\
        288e^{11/8}e^{-(\ell-g)/8} & \le\xi\le\frac{\tau g}{10^6}.
    \end{align}
    Use the five cuts $(0,a+1,a+2\ell,b+1,b+2\ell,L)$ and their original
    regional sums $A_j$ and crossing sum $Y$.
    There are original regional minima $e_j$, a common unit ground vector
    for all $A_j$, and $E_0\in[0,4]$ with $E_0+\sum_j e_j=\mu$.
    Define
    \begin{align}
        B=\sum_{j\in\{0,2,4\}}(\min(A_j,e_j+\ell)-e_j\Id)
        +\sum_{j\in\{1,3\}}(A_j-e_j\Id)+Y.
    \end{align}
    Then $\|B\|\le7\ell$, and there is an actual orthonormal spectral basis
    in which the closed low space $T$ at $E_0+g/10$ equals the low space
    at $E_0+9g/10$. Every cutoff in
    $[E_0+g/20,E_0+19g/20]$ also gives $T$.
    Moreover $\dim T=\dim G$, the spaces $T,G$ are mutually
    $\tau/200$-close, and every eigenvalue of $B$ is at most $E_0+g/10$
    or at least $E_0+9g/10$.
\end{theorem}
\begin{proof}
    \uses{thm:disjoint_regional_clipping,thm:five_region_original_bond_bounds,
        thm:five_region_seven_ell_budget,thm:consecutive_regional_truncation_low}
    The disjoint regional minima and the original crossing budget give
    $E_0\in[0,4]$ and $\|B\|\le7\ell$.
    The original-bond restricted estimates bound both actual residuals,
    at the centered original cutoff $E_0+g/40$ and the clipped cutoff
    $E_0+19g/20$, by $288e^{11/8}e^{-(\ell-g)/8}\le\xi$.
    Apply the scaled restricted-residual cluster theorem of
    \cite[``A low cluster with a scaled projection error'']{QICLean2026Blueprint}.
    All cutoffs use the original ground-energy shift, as in
    \cite{gap:alvv16_spectral_projector_boundary}.
\end{proof}
