\section{Elimination of normalized pair effects}
\label{sec:peps_pair_effects}
Eliminating normalized pair effects
(\cite[Lemma~5.1]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{04-compression.tex}, lines 53--127), together with the stack-length
selection and rescaling in the proof of
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} (lines 201--212), replaces each
pair effect by a cyclic insertion into a fresh register stack.

\begin{lemma}[Elimination of normalized pair effects]
    \label{lem:peps_pair_effects_source}
    \lean{TNLean.Lattice.PartyCircuit.partyPairEffectElimination_with_grouped_sources}
    \leanok
    \uses{def:peps_party_monomial}
    Let $G=\sum_\xi c_\xi M_\xi$ be a contraction expanded into $K$ allowed
    monomials on a fixed set of parties, each with at most $r$ pair effects.
    For every integer $m\ge1$ there exist a gate $\widetilde G_m$ with
    additional garbage registers owned by the original participating parties
    and a unit garbage vector $\Gamma_m$ such that
    \begin{align}
        \bigl\|\widetilde G_m-G\otimes|\Gamma_m\rangle\bigr\|
        \le\frac r{\sqrt m}\sum_\xi|c_\xi|,
        \label{eq:peps_pair_effects_source_error}
    \end{align}
    where $\widetilde G_m$ expands into at most $Km^r$ monomials of local
    contractions and normalized pair sources (at most one per unordered party
    pair) with absolute coefficient sum at most $\sum_\xi|c_\xi|$, as in
    \cite[Lemma~5.1]{OpenAI2026PolynomialPEPS}.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_pair_effects_party,thm:peps_grouped_sources}
    Theorem~\ref{thm:peps_pair_effects_party} gives $\widetilde G_m$ and
    $\Gamma_m$ satisfying (\ref{eq:peps_pair_effects_source_error}), the
    $Km^r$ monomial bound, coefficient sum $\sum_\xi|c_\xi|$, and register
    ownership. Applying Lemma~\ref{thm:peps_grouped_sources} to each monomial
    combines all sources on each unordered party pair into a single normalized
    pair source while preserving the monomial operators, count, coefficients,
    and output space.
\end{proof}

\begin{theorem}[Elimination of pair effects on parties]
    \label{thm:peps_pair_effects_party}
    \lean{TNLean.Lattice.PartyCircuit.partyPairEffectElimination}
    \leanok
    \uses{def:peps_party_monomial}
    For an expansion $G=\sum_\xi c_\xi M_\xi$ into $K$ allowed monomials on
    parties, each with at most $r$ pair effects, and $m\ge1$, the vector
    $\Gamma_m$ is a unit vector prepared by normalized pair sources,
    \begin{align}
        \bigl\|\widetilde G_m-G\otimes|\Gamma_m\rangle\bigr\|
        \le\frac r{\sqrt m}\sum_\xi|c_\xi|,
        \label{eq:peps_pair_effects_party_error}
    \end{align}
    and $\widetilde G_m$ is a weighted sum of at most $Km^r$ allowed
    source-only compositions with absolute coefficient sum $\sum_\xi|c_\xi|$
    and additional registers consisting of $m$ registers of $p$ and $m$
    registers of $q$ for each effect on $(p,q)$, establishing
    \cite[Lemma~5.1]{OpenAI2026PolynomialPEPS} prior to grouping sources by
    party pairs.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_pair_effects_operator,thm:peps_party_terms_source_only,thm:peps_party_ownership}
    Lemma~\ref{thm:peps_pair_effects_operator} gives
    (\ref{eq:peps_pair_effects_party_error}), the monomial count $\le Km^r$, and
    coefficient sum $\sum_\xi|c_\xi|$; Lemma~\ref{thm:peps_party_terms_source_only}
    shows that each term is an allowed source-only composition; and
    Lemma~\ref{thm:peps_party_ownership} gives the stack register ownership.
\end{proof}

Throughout, $\eta\in\mathcal W$ is a unit vector in a finite-dimensional
Hilbert space (typically $\mathcal W=\mathcal U\otimes\mathcal V$) and $m\ge1$.

\begin{definition}[Cyclic insertion map]
    \label{def:peps_cyclic_insertion}
    \lean{CyclicInsertion.insertAt}
    \lean{CyclicInsertion.cyclicInsertion}
    \lean{CyclicInsertion.idealInsertion}
    \leanok
    For $k\in\{0,\dots,m-1\}$, let $I_k:\mathcal W\to\mathcal W^{\otimes m}$
    place its input in factor $k$ and $\eta$ in the remaining $m-1$ factors.
    Define the cyclic and ideal insertion maps by
    \begin{align}
        T_m & =\frac1m\sum_{k=0}^{m-1} I_k,          &
        P_m & =|\eta^{\otimes m}\rangle\langle\eta|.
        \label{eq:peps_cyclic_insertion}
    \end{align}
\end{definition}

\begin{lemma}[Insertions as register permutations]
    \label{thm:peps_insertion_permutation}
    \lean{CyclicInsertion.insertAt_perm}
    \lean{CyclicInsertion.cyclicInsertion_eq_sum_addRight}
    \lean{CyclicInsertion.registerPerm_pair_index}
    \leanok
    \uses{def:peps_cyclic_insertion}
    For any permutation $\sigma$ of $\{0,\dots,m-1\}$ with induced factor
    permutation $U_\sigma$ on $\mathcal W^{\otimes m}$, $I_{\sigma(k)}=U_\sigma I_k$,
    so $T_m=m^{-1}\sum_{k=0}^{m-1}U_{(+k)}I_0$. On
    $\mathcal W=\mathcal U\otimes\mathcal V$, $U_\sigma$ permutes the
    $\mathcal U$ and $\mathcal V$ coordinates by $\sigma$.
\end{lemma}

\begin{proof}\leanok
    Verify $I_{\sigma(k)}=U_\sigma I_k$ and the pair-coordinate action of
    $U_\sigma$ on product basis vectors.
\end{proof}

\begin{lemma}[Contractivity and the fixed vector]
    \label{thm:peps_cyclic_insertion_basic}
    \lean{CyclicInsertion.norm_cyclicInsertion_le_one}
    \lean{CyclicInsertion.cyclicInsertion_self}
    \leanok
    \uses{def:peps_cyclic_insertion}
    The map $T_m$ satisfies $\|T_m\|\le 1$ and $T_m\eta=\eta^{\otimes m}$.
\end{lemma}

\begin{proof}\leanok
    Since $\|\eta\|=1$, each $I_k$ is an isometry with $I_k\eta=\eta^{\otimes m}$,
    so (\ref{eq:peps_cyclic_insertion}) gives $\|T_m\|\le m^{-1}\sum_{k=0}^{m-1}\|I_k\|=1$
    and $T_m\eta=\eta^{\otimes m}$.
\end{proof}

\begin{lemma}[One-effect estimate]
    \label{thm:peps_one_effect_estimate}
    \lean{CyclicInsertion.norm_cyclicInsertion_of_inner_eq_zero}
    \lean{CyclicInsertion.norm_cyclicInsertion_sub_ideal_le}
    \lean{CyclicInsertion.norm_cyclicInsertion_sub_ideal_eq}
    \lean{CyclicInsertion.cyclicInsertion_sub_ideal_eq_zero}
    \lean{CyclicInsertion.norm_rTensor_cyclicInsertion_sub_ideal_le}
    \leanok
    \uses{def:peps_cyclic_insertion}
    For every $v\perp\eta$, $\|T_mv\|=\|v\|/\sqrt m$. Hence
    $\|T_m-P_m\|\le m^{-1/2}$ (with equality if $\{\eta\}^\perp\neq\{0\}$ and
    $T_m=P_m$ if $\{\eta\}^\perp=\{0\}$), and
    $\|(T_m-P_m)\otimes\mathbf 1_{\mathcal S}\|\le m^{-1/2}$ for any spectator
    space $\mathcal S$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_cyclic_insertion_basic}
    For $v\perp\eta$ and $j\neq k$, $\langle I_jv,I_kv\rangle=|\langle v,\eta\rangle|^2\|\eta\|^{2(m-2)}=0$
    and $\|I_kv\|=\|v\|$, so $\|\sum_{k=0}^{m-1}I_kv\|^2=m\|v\|^2$ and
    $\|T_mv\|=\|v\|/\sqrt m$. For any $v\in\mathcal W$, set
    $v_\perp=v-\langle\eta,v\rangle\eta\perp\eta$; by
    Lemma~\ref{thm:peps_cyclic_insertion_basic},
    $\|(T_m-P_m)v\|=\|T_mv_\perp\|=\|v_\perp\|/\sqrt m\le\|v\|/\sqrt m$, with
    equality on any nonzero $v\in\{\eta\}^\perp$. Tensoring with
    $\mathbf 1_{\mathcal S}$ preserves the norm bound.
\end{proof}

\begin{definition}[Monomials with recorded pair effects]
    \label{def:peps_effect_chain}
    \lean{TNLean.Lattice.PartyCircuit.effectMap}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.eval}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.effectCount}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.IsAllowed}
    \leanok
    A monomial with $n$ recorded pair effects is a composition
    \begin{align}
        M=F_n\circ E_n\circ F_{n-1}\circ\cdots\circ E_1\circ F_0,
        \label{eq:peps_effect_chain}
    \end{align}
    where $F_i$ outputs $\mathcal W_{i+1}\otimes\mathcal S_{i+1}$ and
    $E_i=\langle\eta_i|\otimes\mathbf 1_{\mathcal S_i}$ contracts $\mathcal W_i$.
    It is allowed if $\|F_i\|\le 1$ and $\|\eta_i\|=1$ for all $i$.
\end{definition}

\begin{definition}[Replaced monomial and replaced gate]
    \label{def:peps_replaced_gate}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.stackVector}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.replace}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.replaceTerm}
    \lean{TNLean.Lattice.PartyCircuit.GateExpansion}
    \lean{TNLean.Lattice.PartyCircuit.gate}
    \lean{TNLean.Lattice.PartyCircuit.inventory}
    \lean{TNLean.Lattice.PartyCircuit.inventoryVector}
    \lean{TNLean.Lattice.PartyCircuit.replaceGate}
    \lean{TNLean.Lattice.PartyCircuit.termList}
    \leanok
    \uses{def:peps_cyclic_insertion,def:peps_effect_chain}
    Assign each effect $E_i$ of $M$ its own output stack $\mathcal W_i^{\otimes m}$
    with ideal content $\eta_{(M)}=\bigotimes_i\eta_i^{\otimes m}$. The
    replaced monomial $\widetilde M_m$ substitutes $T_m$ for each $E_i$, with
    individual insertion terms $\widetilde M_{m,\kappa}$ weighted by $m^{-n}$
    for $\kappa\in\{0,\dots,m-1\}^n$. For $G=\sum_\xi c_\xi M_\xi$, let
    $\Gamma_m=\bigotimes_\xi\eta_{(M_\xi)}$ on the joint stack space, and define
    $\widetilde G_m$ by summing $c_\xi\widetilde M_{\xi,m}$ tensored with
    $\bigotimes_{\zeta\neq\xi}|\eta_{(M_\zeta)}\rangle$.
\end{definition}

\begin{lemma}[Error of one replaced monomial]
    \label{thm:peps_replaced_monomial_error}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.norm_replace_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.norm_eval_le_one}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.norm_stackVector}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.norm_replace_le_one}
    \leanok
    \uses{def:peps_replaced_gate}
    For an allowed monomial $M$ with $n$ recorded pair effects, $\|M\|\le 1$,
    $\|\widetilde M_m\|\le 1$, $\|\bigotimes_i\eta_i^{\otimes m}\|=1$, and
    \begin{align}
        \Bigl\|\widetilde M_m-M\otimes\Bigl|\bigotimes_i\eta_i^{\otimes m}\Bigr\rangle\Bigr\|
        \le\frac n{\sqrt m}.
        \label{eq:peps_replaced_monomial_error}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_cyclic_insertion_basic,thm:peps_one_effect_estimate}
    By induction on $n$, write $M=M'\circ E_1\circ F_0$ and
    $I_{M'}=M'\otimes|\bigotimes_{i\ge2}\eta_i^{\otimes m}\rangle$. Up to stack
    rearrangement,
    \begin{align}
        \widetilde M_m-M\otimes\Bigl|\bigotimes_i\eta_i^{\otimes m}\Bigr\rangle
        =\bigl(\mathbf 1\otimes(\widetilde{M'}_m-I_{M'})\bigr)\circ(T_m\otimes\mathbf 1)\circ F_0
        +(\mathbf 1\otimes I_{M'})\circ\bigl((T_m-P_m)\otimes\mathbf 1\bigr)\circ F_0.\notag
    \end{align}
    By Lemmas~\ref{thm:peps_cyclic_insertion_basic}
    and~\ref{thm:peps_one_effect_estimate} and the induction hypothesis, the
    two terms have norms at most $(n-1)/\sqrt m$ and $1/\sqrt m$.
\end{proof}

\begin{lemma}[Expansion of a replaced monomial]
    \label{thm:peps_replaced_monomial_expansion}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.replace_eq_sum}
    \lean{TNLean.Lattice.PartyCircuit.EffectChain.norm_replaceTerm_le_one}
    \leanok
    \uses{def:peps_replaced_gate}
    For an allowed monomial $M$ with $n$ pair effects,
    $\widetilde M_m=m^{-n}\sum_{\kappa\in\{0,\dots,m-1\}^n}\widetilde M_{m,\kappa}$
    with $\|\widetilde M_{m,\kappa}\|\le 1$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_cyclic_insertion}
    Expanding each $T_m=m^{-1}\sum_{k=0}^{m-1}I_k$ from (\ref{eq:peps_cyclic_insertion})
    by multilinearity gives $m^n$ compositions $\widetilde M_{m,\kappa}$ of
    contractions $F_i$ and isometries $I_{\kappa_i}$.
\end{proof}

\begin{lemma}[Pair-effect elimination, operator estimates]
    \label{thm:peps_pair_effects_operator}
    \lean{TNLean.Lattice.PartyCircuit.pairEffectElimination}
    \lean{TNLean.Lattice.PartyCircuit.norm_inventoryVector}
    \lean{TNLean.Lattice.PartyCircuit.norm_replaceGate_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.replaceGate_eq_sum_termList}
    \lean{TNLean.Lattice.PartyCircuit.length_termList_le}
    \lean{TNLean.Lattice.PartyCircuit.sum_norm_coeff_termList}
    \lean{TNLean.Lattice.PartyCircuit.norm_term_le_one}
    \leanok
    \uses{def:peps_replaced_gate}
    For $G=\sum_\xi c_\xi M_\xi$ with $K$ allowed monomials, each having
    $n_\xi\le r$ recorded pair effects, and $m\ge1$, $\Gamma_m$ is a unit vector,
    \begin{align}
        \bigl\|\widetilde G_m-G\otimes|\Gamma_m\rangle\bigr\|
        \le\frac r{\sqrt m}\sum_\xi|c_\xi|,
        \label{eq:peps_pair_effects_operator}
    \end{align}
    and $\widetilde G_m$ is a weighted sum of at most $Km^r$ contractions with
    absolute coefficient sum $\sum_\xi|c_\xi|$, establishing the operator
    estimates of \cite[Lemma~5.1]{OpenAI2026PolynomialPEPS}.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_replaced_monomial_error,thm:peps_replaced_monomial_expansion}
    By induction on the monomial list, splitting off the first monomial $M_\xi$
    with stack vector $\eta_{(\xi)}$ and remaining gate $G_L$ gives, up to
    tensor rearrangement,
    \begin{align}
        \widetilde G_m-G\otimes|\Gamma_m\rangle
        =c_\xi\bigl(\widetilde M_{\xi,m}-M_\xi\otimes|\eta_{(\xi)}\rangle\bigr)
        \otimes|\Gamma_{L,m}\rangle
        +\bigl(\widetilde G_{L,m}-G_L\otimes|\Gamma_{L,m}\rangle\bigr)
        \otimes|\eta_{(\xi)}\rangle.\notag
    \end{align}
    Lemma~\ref{thm:peps_replaced_monomial_error} bounds the first term by
    $|c_\xi|n_\xi/\sqrt m\le|c_\xi|r/\sqrt m$, yielding (\ref{eq:peps_pair_effects_operator}).
    Lemma~\ref{thm:peps_replaced_monomial_expansion} expands each
    $c_\xi\widetilde M_{\xi,m}$ into $m^{n_\xi}\le m^r$ contractions with
    coefficients $c_\xi m^{-n_\xi}$ summing in modulus to $|c_\xi|$.
\end{proof}

\begin{lemma}[Stack length and rescaling]
    \label{cor:peps_pair_effects_rescaling}
    \lean{TNLean.Lattice.PartyCircuit.stackLength}
    \lean{TNLean.Lattice.PartyCircuit.stackLength_spec}
    \lean{TNLean.Lattice.PartyCircuit.replaceGate_rescaled}
    \lean{TNLean.Lattice.PartyCircuit.replaceGate_rescaled_stackLength}
    \leanok
    \uses{thm:peps_pair_effects_operator}
    In the setting of Lemma~\ref{thm:peps_pair_effects_operator}, if
    $\|G\|\le 1$, $\delta>0$, $S\ge\sum_\xi|c_\xi|$, and
    $m=\lceil(rS/\delta)^2\rceil+1$, then
    $\|\widetilde G_m-G\otimes|\Gamma_m\rangle\|\le\delta$,
    $\|\widetilde G_m\|\le1+\delta$, and $\widetilde G_m/(1+\delta)$ is a
    contraction satisfying
    \begin{align}
        \Bigl\|\frac{\widetilde G_m}{1+\delta}-G\otimes|\Gamma_m\rangle\Bigr\|
        \le 2\delta.
        \label{eq:peps_pair_effects_rescaling}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $\sqrt m\ge rS/\delta$, (\ref{eq:peps_pair_effects_operator}) gives
    $\|\widetilde G_m-G\otimes|\Gamma_m\rangle\|\le\delta$. Since
    $\|G\otimes|\Gamma_m\rangle\|\le 1$, $\|\widetilde G_m\|\le 1+\delta$, so
    $\|\widetilde G_m/(1+\delta)\|\le 1$ and
    $\|\widetilde G_m/(1+\delta)-G\otimes|\Gamma_m\rangle\|\le\delta/(1+\delta)+\delta\le 2\delta$.
\end{proof}
