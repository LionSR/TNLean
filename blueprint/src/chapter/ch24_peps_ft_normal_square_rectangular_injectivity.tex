\begin{definition}[Normal rectangular injectivity hypotheses]%
    \label{def:peps_normalRectangleInjectivityHypotheses}
    \lean{TNLean.PEPS.NormalRectangleInjectivityHypotheses}
    \lean{TNLean.PEPS.NormalSquareLatticeRectangleInjectivityHypotheses}
    \leanok
    \uses{def:peps_regionInjectivityData, def:peps_squareLatticeCoordinateRegions}
    For a region-injectivity assignment $\inj$
    (Definition~\ref{def:peps_regionInjectivityData}), the abstract
    \emph{normal rectangular injectivity hypotheses} specify two families of
    finite regions $\mathcal R_{2\times 3},\mathcal R_{3\times 2}$ and assert
    $\inj(R)$ for every $R\in\mathcal R_{2\times 3}\cup\mathcal R_{3\times 2}$.
    On the $n\times m$ coordinate square lattice
    (Definition~\ref{def:peps_squareLatticeCoordinateRegions}),
    $\mathcal R_{2\times 3}$ and $\mathcal R_{3\times 2}$ are the families of
    contiguous coordinate $2\times 3$ and $3\times 2$ rectangles.
\end{definition}

\begin{definition}[Normal square-lattice tensor with injective rectangles]%
    \label{def:peps_normalSquareLatticeRectanglePEPS}
    \lean{TNLean.PEPS.IsNormalSquareLatticeRectanglePEPS}
    \leanok
    \uses{def:peps_normalRectangleInjectivityHypotheses, def:peps_regionInjectivityDataOf}
    A PEPS tensor $A$ on the open $n\times m$ square lattice is
    \emph{normal with injective rectangles} if $\inj_A$
    (Definition~\ref{def:peps_regionInjectivityDataOf}) satisfies the coordinate
    rectangular injectivity hypotheses of
    Definition~\ref{def:peps_normalRectangleInjectivityHypotheses} and
    $D_A(e)>0$ for every edge $e$. This is the open-lattice form of the
    hypothesis of \cite[Theorem~3]{Molnar2018NormalPEPS}, with positivity
    recorded explicitly \cite{gap:peps_injective_ft_section3_route}.
\end{definition}

    Immediate by specializing $\mathcal R_{2\times 3}$ and
    $\mathcal R_{3\times 2}$ in

\end{proof}

\begin{lemma}[A large rectangle is injective]%
    \label{lem:peps_normalSquareWideRectangle_injective}
    \lean{TNLean.PEPS.NormalSquareLatticeRectangleInjectivityHypotheses.wideRectangle_injective}
    \lean{TNLean.PEPS.NormalSquareLatticeRectangleInjectivityHypotheses.shortRectangle_injective}
    \leanok
    \uses{lem:peps_injectiveRegion_union, lem:peps_squareLatticeRectangleInjectivity_rectangles}
    Under rectangular injectivity and union closure, any contiguous coordinate
    rectangle $R=I(x_0,x_0+w)\times I(y_0,y_0+h)$ with $x_0+w\le n$,
    $y_0+h\le m$, and either ($w\ge 2$, $h\ge 3$) or ($w\ge 3$, $h\ge 2$)
    satisfies $\inj(R)$.
\end{lemma}
% Current source-comparison status: docs/paper-gaps/peps_normal_ft_section3_route.tex.
\begin{proof}\leanok
    For $w\ge 2$ and $h\ge 3$,
    \begin{align}
        R
         & =\bigcup_{i=0}^{w-2}\bigcup_{j=0}^{h-3}
        I(x_0+i,x_0+i+2)\times I(y_0+j,y_0+j+3),
        \notag
    \end{align}
    so Lemma~\ref{lem:peps_squareLatticeRectangleInjectivity_rectangles} and
    finite union closure give $\inj(R)$ (and similarly using $3\times 2$
    windows when $w\ge 3$, $h\ge 2$).
\end{proof}

\begin{lemma}[Interior edge-complement injectivity]%
    \label{lem:peps_normalSquareInteriorEdgeComplement_injective}
    \lean{TNLean.PEPS.NormalSquareLatticeRectangleInjectivityHypotheses.interiorEdgeComplement_injective}

    For a frame origin $(x_0,y_0)$ with interior margins $2\le x_0$,
    $2\le y_0$, $x_0+8\le n$, and $y_0+8\le m$, rectangular injectivity and
    union closure imply $\inj(A_3)$ for the horizontal complementary block and
    $\inj(A_3^{\mathrm v})$ for the vertical complementary block, without any
    rectangular-cover hypothesis.

    \textbf{Scope restriction (open-boundary interior edges):}
    These edge-blocking constructions apply only to edges with interior margins
    in the open rectangular model, whereas \cite[Theorem~3]{Molnar2018NormalPEPS}
    obtains every edge via periodic translations on the torus; documented in
    \cite{gap:peps_normal_ft_section3_route}.

% Current source-comparison status: docs/paper-gaps/peps_normal_ft_section3_route.tex.
\begin{proof}\leanok

    is the union of four full-length rectangular bands (below, above, left, and
    right of the bounding box of the two removed blocks) and two $2\times 3$ /
    $3\times 2$ filler rectangles inside that box. Each band is injective by
    Lemma~\ref{lem:peps_normalSquareWideRectangle_injective} and each filler by
    Lemma~\ref{lem:peps_squareLatticeRectangleInjectivity_rectangles}, so union
    closure gives $\inj(A_3)$ (and transposed for $A_3^{\mathrm v}$).

% to local and rotated T-covers, respectively. The translated every-edge

\begin{proof}\leanok
    Endpoint memberships are direct coordinate checks; disjointness and
    $R_0\cup B_0\cup A_3=V$ follow from
    Lemma~\ref{lem:peps_normalSquareEdgeComplementRegion}; $\inj(R_0)$ and

    Lemma~\ref{lem:peps_normalSquareRegionT_edgeBlocks_injective}; and
    $\inj(A_3),\inj(A_3^{\mathrm v})$ follow from
    Lemma~\ref{lem:peps_normalSquareEdgeComplementRegion_topCollar}.
\end{proof}

\begin{theorem}[Bond-dimension equality at interior edges from rectangular
        injectivity]%
    \label{thm:peps_bondDim_interiorEdge_eq_of_normalSquareRectangularInjectivity}
    \lean{TNLean.PEPS.bondDim_apply_eq_of_normalSquareInteriorEdge}
    \leanok
    \uses{def:peps_normalRectangleInjectivityHypotheses, def:peps_regionInjectivityData, def:peps_regionInjectivityDataOf, def:peps_same_state, lem:peps_normalSquareInteriorEdgeBlocking}
    Let $A$ and $B$ be PEPS tensors on the $n\times m$ square lattice with
    $d>0$ and $D_A,D_B>0$, generating the same state
    (Definition~\ref{def:peps_same_state}). Suppose $\inj_A$ and $\inj_B$
    (Definition~\ref{def:peps_regionInjectivityDataOf}) satisfy the coordinate
    rectangular injectivity hypotheses
    (Definition~\ref{def:peps_normalRectangleInjectivityHypotheses}) and finite
    union closure (Definition~\ref{def:peps_regionInjectivityData}). Then for
    every edge $e$ with interior margins (at least $1$ row/column below and
    left, and $3$ rows/columns above and right), $D_A(e)=D_B(e)$
    \cite[Section~3, Lemma \emph{inj\_isomorph}, lines~560--583 and
        1407--1500]{Molnar2018NormalPEPS}.
    \begin{proof}
        \leanok
        \uses{lem:peps_normalSquareHorizontalTranslatedEdge_blockingData, lem:peps_normalSquareVerticalTranslatedEdge_blockingData, thm:peps_bondDim_eq_of_normalBlocking}
        For horizontal $e$ at $(x_e,y_e)$ (the vertical case is rotated via
        Lemma~\ref{lem:peps_normalSquareVerticalTranslatedEdge_blockingData}),
        set $(x_0,y_0)=(x_e-1,y_e-2)$, so $2\le x_0,y_0$, $x_0+8\le n$, and
        $y_0+8\le m$. By
        Lemma~\ref{lem:peps_normalSquareHorizontalTranslatedEdge_blockingData},
        both $A$ and $B$ share the injective blocking partition
        $(R_{x_0,y_0},B_{x_0,y_0},C_{x_0,y_0})$ with unique crossing edge
        $\{g\in E:\text{$g$ joins $R_{x_0,y_0}$ and $B_{x_0,y_0}$}\}=\{e'\}$,
        where $e'=e_{x_0,y_0}=e$. Applying the one-edge form of
        Theorem~\ref{thm:peps_bondDim_eq_of_normalBlocking} yields
        $D_A(e)=D_A(e')=D_B(e')=D_B(e)$.
    \end{proof}
\end{theorem}
