\section{Regularity of initial birth regions}
\label{sec:area_law_initial_region_regularity}

Retain the initial family of \cite[Section~11]{OpenAI2026AreaLaw},
with origin $o$, finite endpoint set $Z$, valid residues,
$C\ge 2$, and $k_0\ge 50{,}000{,}000$.

\begin{theorem}[Birth regions are closures of their interiors]
    \label{thm:area_law_initial_birth_regularity}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialBirthRegion_eq_closure_initialOpenRegion}
    \leanok
    \uses{def:area_law_initial_birth_regions,def:area_law_initial_open_regions}
    For every initial identifier $i$,
    \begin{align}
        X_i^0 & =\overline{\operatorname{int}(X_i^0)}.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_cell_fan_cover,
        thm:area_law_primary_fine_cell_cover,def:area_law_fan_runs,
        def:area_law_initial_birth_regions}
    Each fan triangle is the closed convex hull of vertices $a,b,c$ with
    \begin{align}
        (b_1-a_1)(c_2-a_2)-(b_2-a_2)(c_1-a_1) & \neq 0.
    \end{align}
    Because the edge vectors $b-a$ and $c-a$ have nonzero determinant, they
    span the plane, so the triangle has nonempty interior. Every closed convex
    set with nonempty interior is the closure of its interior.

    Each closed dyadic square is convex and contains an unsplit fan triangle,
    hence also has nonempty interior and equals the closure of its interior.

    A finite union of regular closed sets is regular closed: each member's
    interior lies in the interior of the union, so the union lies in the
    closure of its interior, and closedness gives the reverse inclusion.
    Since the dummy region is a finite union of closed coarse squares, each
    primary region is a finite union of closed nonbelt fine cells, and each
    sector run is a finite union of fan triangles, the identity holds in all
    three cases.
\end{proof}
