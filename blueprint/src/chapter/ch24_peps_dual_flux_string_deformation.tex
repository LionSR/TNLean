\begin{definition}[Oriented dual paths and signed crossings]%
    \label{def:peps_dual_flux_path}
    \lean{TNLean.PEPS.TorusDualStep,
        TNLean.PEPS.torusDualQuiver,
        TNLean.PEPS.TorusDualPath,
        TNLean.PEPS.TorusDualStep.reverse,
        TNLean.PEPS.TorusDualStep.reverse_reverse,
        TNLean.PEPS.torusDualInvolutiveReverse,
        TNLean.PEPS.torusPointMass,
        TNLean.PEPS.TorusDualStep.crossings,
        TNLean.PEPS.TorusDualStep.crossings_reverse,
        TNLean.PEPS.torusDualPathCrossings,
        TNLean.PEPS.TorusDualPath.crossings_nil,
        TNLean.PEPS.TorusDualPath.crossings_cons,
        TNLean.PEPS.TorusDualPath.crossings_comp,
        TNLean.PEPS.TorusDualPath.crossings_reverse}
    \leanok
    On $X=\mathbb Z_w\times\mathbb Z_h$ ($w,h>0$), label each plaquette by its
    lower-left primal vertex. A finite dual path $p$ is a sequence of east,
    west, north, and south unit steps (retaining step labels at periodic seams
    and parallel bonds). Write $n(p)=(n_h(p),n_v(p)):X\to\mathbb Z^2$ for its
    signed crossing counts on rightward and downward native bonds: with
    $\delta_s(v)=\mathbf1_{\{v=s\}}$,
    \begin{align}
        n\bigl((x,y)\to(x+1,y)\bigr) & =(0,\delta_{(x+1,y)}), &
        n\bigl((x,y)\to(x,y+1)\bigr) & =(\delta_{(x,y+1)},0),
        \label{eq:peps_dual_step_counts}
    \end{align}
    extended by $n(p^{-1})=-n(p)$ and $n(pq)=n(p)+n(q)$ (so east steps insert
    $U_g$ on downward bonds and south steps insert $U_{g^{-1}}$ on rightward
    bonds \cite[Definition~6.13]{Schuch2010PEPS}).
\end{definition}

\begin{definition}[Finite deformations with prescribed swept vertices]%
    \label{def:peps_dual_flux_homotopy}
    \lean{TNLean.PEPS.torusFluxGradient,
        TNLean.PEPS.torusFluxGradient_zero,
        TNLean.PEPS.torusFluxGradient_add,
        TNLean.PEPS.torusFluxGradient_neg,
        TNLean.PEPS.torusDualEastNorth,
        TNLean.PEPS.torusDualNorthEast,
        TNLean.PEPS.TorusDualHomotopy,
        TNLean.PEPS.torusDualSquareLoop}
    \leanok
    \uses{def:peps_dual_flux_path}
    For $R\subseteq X$, let $\sim_R$ be the fixed-endpoint equivalence relation
    on dual paths (closed under reversal and concatenation) generated by
    backtrack cancellation $ee^{-1}\leftrightarrow\mathrm{nil}$ and the square
    move \cite[proof of Lemma~6.14]{Schuch2010PEPS}
    \begin{align}
        (x,y)\to(x+1,y)\to(x+1,y+1)
        \quad\longleftrightarrow\quad
        (x,y)\to(x,y+1)\to(x+1,y+1)
        \qquad\bigl((x+1,y+1)\in R\bigr).
        \label{eq:peps_dual_square_move}
    \end{align}
    For $f:X\to\mathbb Z$, define $\nabla f=((\nabla f)_h,(\nabla f)_v)$ by
    \begin{align}
        (\nabla f)_h(v) & =f(v+(1,0))-f(v), &
        (\nabla f)_v(v) & =f(v)-f(v+(0,1)).
        \label{eq:peps_dual_gradient}
    \end{align}
\end{definition}

\begin{lemma}[A potential derived from the deformation]%
    \label{thm:peps_dual_flux_potential}
    \lean{TNLean.PEPS.torusDualSquare_crossings,
        TNLean.PEPS.TorusDualHomotopy.exists_potential,
        TNLean.PEPS.TorusDualHomotopy.cancel_reverse,
        TNLean.PEPS.TorusDualHomotopy.squareLoop}
    \leanok
    \uses{def:peps_dual_flux_homotopy}
    If $p\sim_R q$, there is $f:X\to\mathbb Z$ with $\operatorname{supp}(f)\subseteq R$
    such that $n(q)=n(p)+\nabla f$ (with $f=\delta_{(x+1,y+1)}$ for a square
    move~(\ref{eq:peps_dual_square_move})). Moreover, $pp^{-1}\sim_\varnothing\mathrm{nil}$
    and the four-sided square loop around $s=(x+1,y+1)\in R$ satisfies
    $\ell_s\sim_R\mathrm{nil}$.
\end{lemma}
\begin{proof}\leanok
    Substituting~(\ref{eq:peps_dual_step_counts}) into~(\ref{eq:peps_dual_gradient})
    gives $n(\mathrm{NE})-n(\mathrm{EN})=\nabla\delta_{(x+1,y+1)}$. Inducting on
    $p\sim_R q$, backtracks have $f=0$, symmetry/reversal send $f\mapsto-f$, and
    concatenation/transitivity add potentials, preserving
    $\operatorname{supp}(f)\subseteq R$.
\end{proof}

\begin{definition}[Lifted displacement]%

\begin{lemma}[Winding is preserved]%
    \label{thm:peps_dual_flux_displacement}
    \lean{TNLean.PEPS.TorusDualHomotopy.displacement_eq}
    \leanok
    \uses{def:peps_dual_flux_homotopy,def:peps_dual_flux_displacement}
    If $p\sim_R q$, then $d(p)=d(q)$ (so equal endpoints in $X$ do not imply
    $p\sim_R q$).
\end{lemma}
\begin{proof}\leanok
    Since $d(ee^{-1})=(0,0)$ and both sides of~(\ref{eq:peps_dual_square_move})
    have $d=(1,1)$, induction on $p\sim_R q$ using $d(pq)=d(p)+d(q)$ and
    $d(p^{-1})=-d(p)$ gives $d(p)=d(q)$.
\end{proof}

\begin{lemma}[Absorption of a supported vertex gauge]%
    \label{thm:peps_dual_flux_gauge_absorption}
    \lean{TNLean.PEPS.torusBondNetwork_groupGauge_of_vecMul_eq,
        TNLean.PEPS.torusBondNetwork_groupGauge_of_invariant_on,
        TNLean.PEPS.torusBondNetwork_groupGauge,
        TNLean.PEPS.torusBondNetwork_torusBondGauge,
        TNLean.PEPS.torusBondNetwork_torusBondGauge_of_invariant_on,
        TNLean.PEPS.torusBondGauge_fst_eq_of_eq_one,
        TNLean.PEPS.torusBondGauge_snd_eq_of_eq_one,
        TNLean.PEPS.torusBondGauge_fst_eq_of_not_mem,
        TNLean.PEPS.torusBondGauge_snd_eq_of_not_mem}
    \leanok
    For native bond matrices $(O_h,O_v)$ and $k:X\to G$, define
    \begin{align}
        O'_h(v) & =U_{k(v+(1,0))}O_h(v)U_{k(v)^{-1}}, &
        O'_v(v) & =U_{k(v)}O_v(v)U_{k(v+(0,1))^{-1}}.
        \label{eq:peps_dual_matrix_gauge}
    \end{align}
    If every $A_v$ with $k(v)\ne1$ is virtually $G$-invariant, the closed
    contraction is unchanged under $(O_h,O_v)\mapsto(O'_h,O'_v)$; if $k=1$ on
    $R^c$, all bonds and site tensors in $R^c$ are unchanged.
\end{lemma}
\begin{proof}\leanok
    Expanding~(\ref{eq:peps_dual_matrix_gauge}) in the bond sum and grouping the
    four adjacent factors at each vertex $v$ absorbs them into $A_v$ when
    $k(v)\ne1$, while $U_{k(v)}=I$ when $k(v)=1$.
\end{proof}

\begin{lemma}[Physical invariance under finite dual-path deformation]%
    \label{thm:peps_dual_flux_deformation}

    In the finite generated-deformation form of
    \cite[Lemma~6.14]{Schuch2010PEPS}, if $p\sim_R q$ and every $A_v$ ($v\in R$)
    is virtually $G$-invariant (for arbitrary $w,h\ge1$, nonabelian $G$, and
    site-dependent $A_v$), then $\Psi_p^E(\sigma)=\Psi_q^E(\sigma)$ for all

\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dual_flux_potential,thm:peps_dual_flux_gauge_absorption}

    single-generator exponentiation gives
    \begin{align}
        u_h^q(v) & =k(v+(1,0))u_h^p(v)k(v)^{-1}, &
        u_v^q(v) & =k(v)u_v^p(v)k(v+(0,1))^{-1},
        \label{eq:peps_dual_label_gauge}
    \end{align}
    with $k=1$ on $R^c$, so Lemma~\ref{thm:peps_dual_flux_gauge_absorption}
    gives $\Psi_p^E(\sigma)=\Psi_q^E(\sigma)$.
\end{proof}

\begin{definition}[Clockwise native holonomy and signed curl]%
    \label{def:peps_dual_flux_curl}
    \lean{TNLean.PEPS.torusFluxCurl,

    \leanok
    \uses{def:peps_dual_flux_path}
    For native labels $u=(u_h,u_v)$ and $n=(n_h,n_v)$ at $z=(x,y)$, define
    \begin{align}
        H_u(z)                    & =u_h(z)^{-1}u_v(x+1,y)u_h(x,y+1)u_v(z)^{-1},
        \label{eq:peps_dual_native_holonomy}                                     \\
        (\operatorname{curl}n)(z) & =n_h(x,y+1)+n_v(x+1,y)-n_h(z)-n_v(z),
        \label{eq:peps_dual_native_curl}
    \end{align}
    so $\operatorname{curl}$ is additive and
    $H_{(g^{n_h},g^{n_v})}(z)=g^{(\operatorname{curl}n)(z)}$.
\end{definition}

\begin{lemma}[Flux is supported at the two path endpoints]%
    \label{thm:peps_dual_flux_endpoints}
    \lean{TNLean.PEPS.TorusDualStep.fluxCurl,
        TNLean.PEPS.TorusDualPath.fluxCurl,
        TNLean.PEPS.torusBondPlaquetteHolonomy_torusDualFluxLabels,
        TNLean.PEPS.torusBondPlaquetteHolonomy_torusDualFluxLabels_source,
        TNLean.PEPS.torusBondPlaquetteHolonomy_torusDualFluxLabels_target,
        TNLean.PEPS.torusBondPlaquetteHolonomy_torusDualFluxLabels_of_ne,
        TNLean.PEPS.torusBondPlaquetteHolonomy_torusDualFluxLabels_loop}
    \leanok
    \uses{def:peps_dual_flux_curl,def:peps_dual_flux_matrices}
    Every finite dual path $p:a\to b$ satisfies
    \cite[Definition~6.13]{Schuch2010PEPS}
    \begin{align}
        \operatorname{curl}n(p) & =\delta_a-\delta_b,           &
        H_{u^p}(z)              & =g^{\delta_a(z)-\delta_b(z)}.
        \label{eq:peps_dual_endpoint_flux}
    \end{align}
    In particular, $H_{u^p}\equiv1$ whenever $a=b$.
\end{lemma}
\begin{proof}\leanok
    By~(\ref{eq:peps_dual_step_counts}) and~(\ref{eq:peps_dual_native_curl}), each
    step $s\to t$ has $\operatorname{curl}n=\delta_s-\delta_t$; telescoping
    along $p$ and exponentiating gives~(\ref{eq:peps_dual_endpoint_flux}).
\end{proof}

\begin{lemma}[Four-spin detection of native flux labels]%

    $S_z$ there is a complete orthogonal projective measurement

        Q_C^{\rm ccw}M_{u,S_z}                   & =
        \mathbf1_{\{C=[H_u(z)^{-1}]\}}M_{u,S_z}, &
        Q_\perp^{\rm ccw}M_{u,S_z}               & =0.

        \mathbf1_{\{C=[g^{\delta_a(z)-\delta_b(z)}]\}}M_{u^p,S_z}, &
        Q_\perp^{\rm cw}M_{u^p,S_z}                                & =0,
        \label{eq:peps_dual_cw_measurement}
    \end{align}
    detecting $[g]$ at $a$ and $[g^{-1}]$ at $b$ when $a\ne b$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_torus_plaquette_holonomy_measurement,
        def:peps_dual_flux_cut,thm:peps_dual_flux_orientation,

    into~(\ref{eq:peps_dual_ccw_measurement}) gives~(\ref{eq:peps_dual_cw_measurement}).
\end{proof}

    For $w,h\ge3$ and homogeneous virtually $G$-invariant $A$, if $H_u(z)=1$ (in
    particular if $(\operatorname{curl}n)(z)=0$ for $u=(g^{n_h},g^{n_v})$),
    then every $S_z^c$-slice of $\Psi_u$ lies in the plaquette ground space
    $\mathcal G_{S_z}$, so every $P_z\ge0$ with $\ker P_z=\mathcal G_{S_z}$
    (including the orthogonal projector onto $\mathcal G_{S_z}^\perp$) satisfies
    \begin{align}
        (P_z\otimes I_{S_z^c})\Psi_u & =0.
        \label{eq:peps_dual_parent_constraint}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dual_flux_gauge_absorption}
    For $z=(x,y)$, setting $k(z)=1$, $k(x+1,y)=u_h(z)^{-1}$,
    $k(x,y+1)=u_v(z)$, $k(x+1,y+1)=u_v(z)u_h(x,y+1)^{-1}$, and $k=1$ elsewhere
    trivializes all four internal bonds of $S_z$ when $H_u(z)=1$;
    Lemma~\ref{thm:peps_dual_flux_gauge_absorption} then places every
    $S_z^c$-slice of $\Psi_u$ in $\mathcal G_{S_z}=\ker P_z$.
\end{proof}

\begin{lemma}[Parent interactions away from dual-string endpoints]%
    \label{thm:peps_dual_flux_endpoint_parent}
    \lean{TNLean.PEPS.torusDualFluxState_parent_annihilates_of_ne,
        TNLean.PEPS.torusDualFluxState_canonicalParent_annihilates_of_ne,
        TNLean.PEPS.torusDualFluxState_canonicalParent_annihilates_loop}
    \leanok
    \uses{thm:peps_dual_flux_endpoints,thm:peps_dual_flux_local_parent}
    Under the hypotheses of Lemma~\ref{thm:peps_dual_flux_local_parent}, for
    any finite dual path $p:a\to b$ and $z\notin\{a,b\}$ (or every $z\in X$
    when $a=b$), every $P_z\ge0$ with $\ker P_z=\mathcal G_{S_z}$ satisfies
    $(P_z\otimes I_{S_z^c})\Psi_p=0$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dual_flux_endpoints,thm:peps_dual_flux_local_parent}
    By~(\ref{eq:peps_dual_endpoint_flux}), $H_{u^p}(z)=1$, so
    Lemma~\ref{thm:peps_dual_flux_local_parent} gives~(\ref{eq:peps_dual_parent_constraint}).
\end{proof}
