\chapter{Sequential Generation of Matrix Product States}\label{ch:seqanc}

A chain of $N$ sites with local dimension $d$ starts in $\ket{0}^{\otimes N}$,
and an ancilla $\C^D$ interacts sequentially with the sites, producing
open-boundary matrix product states of bond dimension at most
$D$~\cite[Section~5.1]{PerezGarcia2007Matrix}.

Sites are labelled $k=0,\dots,N-1$ in ket order, with configuration
$\sigma=(\sigma_0,\dots,\sigma_{N-1})$ and $A_k$ the operation acting at site
$k$. For square matrices $Q_0,\dots,Q_{n-1}$ indexed by the physical index,
write $Q^\sigma=Q_0^{\sigma_0}\cdots Q_{n-1}^{\sigma_{n-1}}$.

\section{Sequential schemes}

\begin{definition}[Sequentially generated state]
    \label{def:seqanc_state}
    \lean{MPSPreparation.seqAmplitude}
    \leanok
    Let $\ket{\varphi_I},\ket{\varphi_F}\in\C^D$ and let
    $A_k^i\in\MN{D}$ for $k=0,\dots,N-1$ and $i=0,\dots,d-1$. Following
    \cite[Section~5.1]{PerezGarcia2007Matrix}, the state produced by the
    operations
    $\sum_{i,\alpha,\beta}(A_k^i)_{\alpha\beta}\ketbra{\alpha,i}{\beta,0}$
    followed by projecting the ancilla onto $\ket{\varphi_F}$ is
    \begin{align}
        \psi(\sigma)
         & =\bra{\varphi_F}A_0^{\sigma_0}A_1^{\sigma_1}\cdots
        A_{N-1}^{\sigma_{N-1}}\ket{\varphi_I}.
        \label{eq:seqanc_state}
    \end{align}
\end{definition}

\begin{definition}[Probabilistic sequential scheme]
    \label{def:seqanc_probabilistic}
    \lean{MPSPreparation.IsProbabilisticallyGenerated}
    \leanok
    \uses{def:seqanc_state}
    A vector $\psi\in(\C^d)^{\otimes N}$ is \emph{probabilistically generated
        with a $D$-dimensional ancilla} if it satisfies~(\ref{eq:seqanc_state})
    for some matrices $A_k^i\in\MN{D}$ and vectors
    $\ket{\varphi_I},\ket{\varphi_F}\in\C^D$. The vector is the
    unnormalized post-measurement vector, as in the source's ``up to
    normalization''.
\end{definition}

\begin{definition}[Operations induced by unitaries]
    \label{def:seqanc_step}
    \lean{MPSPreparation.stepMatrix, MPSPreparation.jointState}
    \leanok
    Let $d\ge 1$. A unitary $U$ on $\C^D\otimes\C^d$ acting on the site state
    $\ket{0}$ induces the matrices
    $(A^i)_{\alpha\beta}=\bra{\alpha,i}U\ket{\beta,0}$. For unitaries
    $U_0,\dots,U_{N-1}$, with $U_k$ applied at step $k+1$ and acting on site
    $N-1-k$, and an initial ancilla state $\ket{\varphi_I}$, the ancilla
    component of the joint state at configuration $\sigma$ is
    $A_0^{\sigma_0}\cdots A_{N-1}^{\sigma_{N-1}}\ket{\varphi_I}\in\C^D$, where
    $A_k$ is induced by $U_{N-1-k}$~\cite[Section~5.1]{PerezGarcia2007Matrix}.
\end{definition}

\begin{definition}[Deterministic sequential scheme]
    \label{def:seqanc_deterministic}
    \lean{MPSPreparation.IsDeterministicallyGenerated}
    \leanok
    \uses{def:seqanc_step}
    Let $d\ge 1$. A vector $\psi$ is \emph{deterministically generated with a
        $D$-dimensional ancilla} if there are unitaries $U_0,\dots,U_{N-1}$ on
    $\C^D\otimes\C^d$ and unit vectors $\ket{\varphi_I},\ket{\varphi_F}\in\C^D$
    such that the ancilla component of the joint state at every
    configuration $\sigma$ equals $\psi(\sigma)\ket{\varphi_F}$; that is, the
    final joint state is $\ket{\varphi_F}\otimes\ket{\psi}$ and the ancilla
    decouples in the last step without measurement.
\end{definition}

\begin{definition}[Open-boundary representation with bounded bond dimension]
    \label{def:seqanc_obc_rep}
    \lean{MPSPreparation.HasOBCRep}
    \leanok
    \uses{def:varying_bond_obc_coeff}
    A vector $\psi$ has an \emph{open-boundary representation with bond
        dimension at most $D$} if $\psi=c_A$ for some varying-bond open chain
    $A$ with common bound $D$, with $c_A$ as in
    (\ref{eq:varying_bond_obc_coeff}).
\end{definition}

\begin{definition}[Scaling one site]
    \label{def:seqanc_smul_site}
    \lean{OBCChainTensor.smulSite}
    \leanok
    \uses{def:varying_bond_obc_chain}
    For a varying-bond open chain $A$ of length $N$, a site $q$ and $c\in\C$,
    the chain obtained by scaling the site $q$ by $c$ has the bond dimensions of
    $A$, the site matrices $c\,A_q^i$ at the site $q$, and the site matrices
    $A_p^i$ at every other site $p$.
\end{definition}

\begin{theorem}[Open-boundary states form a cone]
    \label{thm:seqanc_obc_cone}
    \lean{OBCChainTensor.coeff_smulSite, MPSPreparation.HasOBCRep.smul,
        MPSPreparation.hasOBCRep_zero}
    \leanok
    \uses{def:seqanc_smul_site, def:seqanc_obc_rep, def:varying_bond_obc_coeff}
    Scaling one site of a varying-bond open chain by $c$ multiplies its
    coefficient by $c$. Consequently, for $N\ge 1$, if $\psi$ has an
    open-boundary representation with bond dimension at most $D$, then so has
    $c\,\psi$ for every $c\in\C$, and for $N\ge 1$ and $D\ge 1$ the zero vector
    has an open-boundary representation with bond dimension at most $D$.
\end{theorem}
\begin{proof}
    \leanok
    The coefficient is a sum over virtual paths of products over the sites, and
    exactly one factor of each product is multiplied by $c$. For $c\,\psi$
    scale the first site. For the zero vector take all bond dimensions equal to
    one and all site matrices zero: every product has a zero factor, since
    $N\ge 1$.
\end{proof}

\section{Isometric decompositions}

\begin{definition}[Isometry on a bond block]
    \label{def:seqanc_isometry_on}
    \lean{MPSPreparation.IsIsometryOn}
    \leanok
    A family $Q^i\in\MN{D}$, $i=0,\dots,d-1$, is \emph{isometric on its first
        $b$ columns} if for all $\beta,\beta'<b$
    \begin{align}
        \sum_{i=0}^{d-1}\sum_{\alpha=0}^{D-1}
        \overline{Q^i_{\alpha\beta}}\,Q^i_{\alpha\beta'}
         & =\delta_{\beta\beta'}.
        \notag
    \end{align}
    For $b=D$ this is $\sum_i(Q^i)^\dagger Q^i=\Id_D$.
\end{definition}

\begin{definition}[Supports below a level]
    \label{def:seqanc_support}
    \lean{MPSPreparation.IsSupportedBelow, MPSPreparation.IsRowSupportedBelow}
    \leanok
    A vector $v\in\C^D$ is \emph{supported below $b$} if $v_\beta=0$ for
    $\beta\ge b$. A matrix $M\in\MN{D}$ is \emph{row-supported below $a$} if
    $M_{\alpha\beta}=0$ for $\alpha\ge a$.
\end{definition}

\begin{theorem}[Propagation of supports]
    \label{thm:seqanc_support_propagation}
    \lean{MPSPreparation.IsRowSupportedBelow.mul,
        MPSPreparation.IsRowSupportedBelow.mulVec,
        MPSPreparation.isSupportedBelow_eval_mulVec}
    \leanok
    \uses{def:seqanc_support}
    If $M$ is row-supported below $a$, then so is $MN$ for every $N$, and
    $Mv$ is supported below $a$ for every $v$. Consequently, if $Q_p^i$ is
    row-supported below $b_p$ for $p=0,\dots,n-1$ and $v$ is supported
    below $b_n$, then $Q^\sigma v$ is supported below $b_0$.
\end{theorem}
\begin{proof}
    \leanok
    The rows $\alpha\ge a$ of $MN$ and of $Mv$ are sums of multiples of
    vanishing entries. The chain statement follows by induction on $n$,
    applying the vector statement to the first factor.
\end{proof}

\begin{definition}[Boundary vector and boundary matrices]
    \label{def:seqanc_boundary}
    \lean{MPSPreparation.basisVecZero, MPSPreparation.rowMat,
        MPSPreparation.colMat}
    \leanok
    Let $\ket{0}\in\C^D$ be the first standard basis vector. For
    $v,r\in\C^D$ let $\ket{0}v^{\mathsf T}$ be the matrix with first row $v$
    and all other rows zero, and $r\bra{0}$ the matrix with first column $r$
    and all other columns zero.
\end{definition}

\begin{theorem}[Coordinates of the boundary vector and matrices]
    \label{thm:seqanc_boundary_coords}
    \lean{MPSPreparation.isRowSupportedBelow_rowMat,
        MPSPreparation.basisVecZero_dotProduct, MPSPreparation.star_basisVecZero,
        MPSPreparation.mulVec_basisVecZero_apply,
        MPSPreparation.rowMat_mul_mulVec_zero, MPSPreparation.mul_colMat_apply,
        MPSPreparation.mul_colMat_apply_of_ne,
        MPSPreparation.eq_smul_basisVecZero_of_isSupportedBelow_one,
        MPSPreparation.star_smul_basisVecZero_dotProduct_self,
        MPSPreparation.star_basisVecZero_dotProduct_self}
    \leanok
    \uses{def:seqanc_boundary, def:seqanc_support}
    Let $D>0$. The matrix $\ket{0}v^{\mathsf T}$ is row-supported below one;
    $\ket{0}$ is real and of norm one; $\bra{0}w=w_0$ and
    $(M\ket{0})_\alpha=M_{\alpha 0}$; the first coordinate of
    $\ket{0}v^{\mathsf T}Mw$ is $v^{\mathsf T}Mw$; the first column of
    $M\,r\bra{0}$ is $Mr$ and its other columns vanish; a vector supported
    below one equals $w_0\ket{0}$; and $\|x\ket{0}\|^2=|x|^2$.
\end{theorem}
\begin{proof}
    \leanok
    Direct computation of coordinates.
\end{proof}

\begin{theorem}[Peeling off the last site]
    \label{thm:seqanc_eval_last}
    \lean{MPSChainTensor.eval_succ'}
    \leanok
    For square matrices $Q_0,\dots,Q_n$ and a configuration $\sigma$ of $n+1$
    sites, $Q^\sigma=(Q_0^{\sigma_0}\cdots
        Q_{n-1}^{\sigma_{n-1}})\,Q_n^{\sigma_n}$.
\end{theorem}
\begin{proof}
    \leanok
    Associativity of the ordered product.
\end{proof}

\begin{theorem}[One decomposition step]
    \label{thm:seqanc_one_step}
    \lean{MPSPreparation.exists_isIsometryOn_mul}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    Let $Y^i\in\MN{D}$ be row-supported below $a$ for every $i$. Then there
    are $b\le D$, matrices $Q^i$ and a matrix $R$ with $Y^i=Q^iR$ for every
    $i$, such that each $Q^i$ is row-supported below $a$ and vanishes on the
    columns $\beta\ge b$, the family $Q$ is isometric on its first $b$
    columns, and $R$ is row-supported below $b$. Moreover $b\le c$ whenever
    every $Y^i$ vanishes on the columns $\gamma\ge c$.
\end{theorem}
\begin{proof}
    \leanok
    Stack the matrices $Y^i$ into one matrix with rows indexed by
    $(\alpha,i)$, choose an orthonormal basis $q_0,\dots,q_{b-1}$ of its
    column space, and let $Q^i$ have columns $q_\beta$ for $\beta<b$ and
    zero columns otherwise. The coefficients of the columns of the stacked
    matrix in this basis form $R$. The column space is spanned by the at most
    $c$ nonzero columns, so $b\le c$.
\end{proof}

\begin{theorem}[Successive decompositions with a matrix remainder]
    \label{thm:seqanc_sweep_matrix_remainder}
    \lean{MPSPreparation.exists_isometric_chain_mul,
        MPSPreparation.sum_star_mulVec_dotProduct_of_isIsometryOn,
        MPSPreparation.sum_star_eval_mulVec_dotProduct}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    Let $a\le D$, let $R_{-1}\in\MN{D}$ be row-supported below $a$, and let
    $P_0,\dots,P_{n-1}$ be families of $D\times D$ matrices. Then there are
    $b_0=a$ and $b_1,\dots,b_n\le D$, families $Q_0,\dots,Q_{n-1}$ such that
    $Q_p^i$ is row-supported below $b_p$, vanishes on the columns
    $\beta\ge b_{p+1}$, and $Q_p$ is isometric on its first $b_{p+1}$ columns,
    and a matrix $R$ row-supported below $b_n$, such that for every
    configuration $\sigma$
    \begin{align}
        R_{-1}\,P^\sigma
         & =Q^\sigma\,R.
        \label{eq:seqanc_sweep_matrix_remainder}
    \end{align}
    Moreover $b_{p+1}\le c$ whenever every $P_p^i$ vanishes on the columns
    $\gamma\ge c$. If $Q$ is isometric on its first $b$ columns and $v,w$ are
    supported below $b$, then
    $\sum_i\langle Q^iv,Q^iw\rangle=\langle v,w\rangle$. Consequently, if
    $Q_p^i$ is row-supported below $b_p$, $Q_p$ is isometric on its first
    $b_{p+1}$ columns, and $v,w$ are supported below $b_n$, then
    \begin{align}
        \sum_\sigma\langle Q^\sigma v,Q^\sigma w\rangle
         & =\langle v,w\rangle.
        \label{eq:seqanc_sweep_inner_product}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_one_step, thm:seqanc_support_propagation}
    By induction on $n$. For $n=0$ take $R=R_{-1}$. Otherwise apply
    Theorem~\ref{thm:seqanc_one_step} to $Y^i=R_{-1}P_0^i$, which gives
    $Y^i=Q_0^iR_0$ with $R_0$ row-supported below $b_1$, and apply the
    induction hypothesis to $R_0$ and $P_1,\dots,P_{n-1}$:
    \begin{align}
        R_{-1}P_0^{\sigma_0}P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
         & =Q_0^{\sigma_0}R_0P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
        =Q_0^{\sigma_0}Q_1^{\sigma_1}\cdots Q_{n-1}^{\sigma_{n-1}}R.
        \notag
    \end{align}
    For one site, expand
    \begin{align}
        \sum_i\langle Q^iv,Q^iw\rangle
         & =\sum_{\beta,\beta'<b}\overline{v_\beta}w_{\beta'}
        \sum_{i,\alpha}\overline{Q^i_{\alpha\beta}}Q^i_{\alpha\beta'}
        =\sum_{\beta<b}\overline{v_\beta}w_\beta
        \notag
    \end{align}
    by the isometry condition. For the chain, write
    $Q^{\sigma}=Q_0^{\sigma_0}Q'^{\sigma'}$, where $Q'$ is the chain
    $Q_1,\dots,Q_{n-1}$ and $\sigma'=(\sigma_1,\dots,\sigma_{n-1})$. The
    vectors $Q'^{\sigma'}v$ and $Q'^{\sigma'}w$ are supported below $b_1$ by
    Theorem~\ref{thm:seqanc_support_propagation}, so the one-site identity for
    $Q_0$ and the induction hypothesis for $Q'$ give
    \begin{align}
        \sum_{\sigma_0,\sigma'}
        \langle Q_0^{\sigma_0}Q'^{\sigma'}v,Q_0^{\sigma_0}Q'^{\sigma'}w\rangle
         & =\sum_{\sigma'}\langle Q'^{\sigma'}v,Q'^{\sigma'}w\rangle
        =\langle v,w\rangle.
        \notag
    \end{align}
\end{proof}

\begin{theorem}[Successive decompositions]
    \label{thm:seqanc_sweep}
    \lean{MPSPreparation.exists_isometric_chain}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    Let $a\le D$, let $R_{-1}\in\MN{D}$ be row-supported below $a$, let
    $P_0,\dots,P_{n-1}$ be families of $D\times D$ matrices, and let
    $r\in\C^D$. Then there are $b_0=a$ and $b_1,\dots,b_n\le D$, families
    $Q_0,\dots,Q_{n-1}$ such that $Q_p^i$ is row-supported below $b_p$,
    vanishes on the columns $\beta\ge b_{p+1}$, and $Q_p$ is isometric on its
    first $b_{p+1}$ columns, and a vector $r'$ supported below $b_n$, such
    that for every configuration $\sigma$
    \begin{align}
        R_{-1}\,P^\sigma\,r
         & =Q^\sigma\,r'.
        \notag
    \end{align}
    Moreover $b_{p+1}\le c$ whenever every $P_p^i$ vanishes on the columns
    $\gamma\ge c$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_matrix_remainder, thm:seqanc_support_propagation}
    Theorem~\ref{thm:seqanc_sweep_matrix_remainder} gives the bonds $b_p$, the
    families $Q_p$ and a matrix $R$ row-supported below $b_n$ with
    $R_{-1}P^\sigma=Q^\sigma R$ for every $\sigma$. Setting $r'=Rr$, which is
    supported below $b_n$ by Theorem~\ref{thm:seqanc_support_propagation},
    yields $R_{-1}P^\sigma r=Q^\sigma Rr=Q^\sigma r'$.
\end{proof}

\begin{theorem}[Isometric chains preserve the norm]
    \label{thm:seqanc_norm}
    \lean{MPSPreparation.sum_normSq_mulVec_of_isIsometryOn,
        MPSPreparation.sum_normSq_eval_mulVec}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    If $Q$ is isometric on its first $b$ columns and $v$ is supported below
    $b$, then $\sum_i\|Q^iv\|^2=\|v\|^2$. Consequently, if $Q_p^i$ is
    row-supported below $b_p$, $Q_p$ is isometric on its first $b_{p+1}$
    columns, and $v$ is supported below $b_n$, then
    $\sum_\sigma\|Q^\sigma v\|^2=\|v\|^2$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_matrix_remainder}
    Take $w=v$ in the inner-product identities of
    Theorem~\ref{thm:seqanc_sweep_matrix_remainder}: $\sum_i\langle Q^iv,Q^iv\rangle
        =\langle v,v\rangle$ for one site, and
    (\ref{eq:seqanc_sweep_inner_product}) with $w=v$ for the chain.
\end{proof}

\begin{theorem}[Changing sites outside the used block]
    \label{thm:seqanc_block_congr}
    \lean{MPSPreparation.mulVec_eq_of_isSupportedBelow,
        MPSPreparation.eval_mulVec_congr}
    \leanok
    \uses{def:seqanc_support}
    If $M'_{\alpha\beta}=M_{\alpha\beta}$ for all $\beta<b$ and $w$ is
    supported below $b$, then $M'w=Mw$. Consequently, if $Q_p^i$ is
    row-supported below $b_p$, the families $Q'_p$ agree with $Q_p$ on the
    columns $\beta<b_{p+1}$, and $v$ is supported below $b_n$, then
    $Q'^\sigma v=Q^\sigma v$ for every $\sigma$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_support_propagation}
    The first claim holds entrywise. The second follows by induction on the
    chain from the right, since each partial product applied to $v$ is
    supported below the next bond level.
\end{proof}

\begin{theorem}[Unitary extension of a block isometry]
    \label{thm:seqanc_unitary_extension}
    \lean{MPSPreparation.exists_unitary_extension}
    \leanok
    \uses{def:seqanc_isometry_on}
    Let $d\ge 1$ and let $Q$ be isometric on its first $b$ columns. Then
    there is a unitary $U$ on $\C^D\otimes\C^d$ with
    $\bra{\alpha,i}U\ket{\beta,0}=Q^i_{\alpha\beta}$ for all $i$, $\alpha$
    and $\beta<b$.
\end{theorem}
\begin{proof}
    \leanok
    The vectors $\sum_{\alpha,i}Q^i_{\alpha\beta}\ket{\alpha,i}$, $\beta<b$,
    are orthonormal. Extend them to an orthonormal basis of
    $\C^D\otimes\C^d$ and let $U$ send the standard basis, with the vectors
    $\ket{\beta,0}$, $\beta<b$, first, to this basis.
\end{proof}

\begin{theorem}[Positive bond bound]
    \label{thm:seqanc_bond_bound_pos}
    \lean{OBCChainTensor.bondBound_pos}
    \leanok
    \uses{def:varying_bond_obc_chain}
    The common bound $D$ of a varying-bond open chain is positive.
\end{theorem}
\begin{proof}
    \leanok
    $1=D_0\le D$.
\end{proof}

\begin{theorem}[Open-chain coefficient as a corner entry]
    \label{thm:seqanc_coeff_corner}
    \lean{OBCChainTensor.coeff_eq_eval_zeroPad}
    \leanok
    \uses{def:varying_bond_obc_coeff, def:varying_bond_obc_zero_pad}
    For a varying-bond open chain with square padding $\widetilde A$, and
    every configuration $\sigma$ of any length $N\ge 0$,
    $c_A(\sigma)=(\widetilde A^\sigma)_{00}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:varying_bond_obc_coeff_zero, thm:varying_bond_obc_zero_pad_coeff}
    For $N=0$ both sides are one. Otherwise $c_A(\sigma)$ is the trace of
    $\widetilde A^\sigma$, and all diagonal entries except the first vanish
    because the first padded site matrix is row-supported below $D_0=1$.
\end{proof}

\begin{definition}[Cutting a square chain down to its bond blocks]
    \label{def:seqanc_cut_down}
    \lean{OBCChainTensor.ofSupported}
    \leanok
    \uses{def:varying_bond_obc_chain}
    Given $b_0,\dots,b_N\le D$ with $b_0=b_N=1$ and square matrices $Q_k^i$,
    the varying-bond open chain with dimensions $b_k$ has as site matrices
    the upper-left $b_k\times b_{k+1}$ corners of $Q_k^i$.
\end{definition}

\begin{theorem}[Cutting down and padding]
    \label{thm:seqanc_cut_down}
    \lean{OBCChainTensor.zeroPad_ofSupported,
        OBCChainTensor.sum_conjTranspose_mul_ofSupported}
    \leanok
    \uses{def:seqanc_cut_down, def:varying_bond_obc_zero_pad,
        def:seqanc_isometry_on}
    If every $Q_k^i$ is row-supported below $b_k$ and vanishes on the
    columns $\beta\ge b_{k+1}$, then padding the cut-down chain returns
    $Q$. If $Q_k^i$ is row-supported below $b_k$ and $Q_k$ is isometric on
    its first $b_{k+1}$ columns, then the cut-down site matrices $A_k^i$
    satisfy $\sum_i(A_k^i)^\dagger A_k^i=\Id_{b_{k+1}}$.
\end{theorem}
\begin{proof}
    \leanok
    Entrywise comparison; for the second claim the sum over rows $\alpha$
    may be restricted to $\alpha<b_k$.
\end{proof}

\begin{theorem}[Assembling an open chain]
    \label{thm:seqanc_assemble}
    \lean{OBCChainTensor.exists_of_isometric_chain}
    \leanok
    \uses{def:varying_bond_obc_coeff, def:seqanc_isometry_on, def:seqanc_support}
    Let $b_0,\dots,b_{n+1}\le D$ with $b_0=1$, and let $Q_0,\dots,Q_n$ be
    families with $Q_p^i$ row-supported below $b_p$, vanishing on the columns
    $\beta\ge b_{p+1}$, and isometric on the first $b_{p+1}$ columns. For
    every $r'\in\C^D$ there is a varying-bond open chain $A$ with common
    bound $D$ such that $c_A(\sigma)=(Q^\sigma r')_0$ for every $\sigma$,
    $D_k=b_k$ for $k\le n$, and $\sum_i(A_p^i)^\dagger A_p^i=\Id$ for
    $p<n$. If moreover $r'$ is supported below $b_{n+1}$ and $\|r'\|=1$,
    then also $\sum_i(A_n^i)^\dagger A_n^i=1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:seqanc_cut_down, thm:seqanc_cut_down, thm:seqanc_coeff_corner,
        thm:seqanc_eval_last, thm:seqanc_norm, thm:seqanc_boundary_coords}
    Replace the last family by $Q_n^i\,r'\bra{0}$, set the last bond
    dimension to one, and cut down. By Theorems~\ref{thm:seqanc_coeff_corner}
    and~\ref{thm:seqanc_eval_last}, the coefficient at $\sigma$ is the first
    coordinate of $Q^\sigma r'$. If $r'$ is a unit vector supported below $b_{n+1}$, the isometry of
    $Q_n$ on its first $b_{n+1}$ columns gives
    $\sum_i\|Q_n^ir'\|^2=\|r'\|^2=1$.
\end{proof}

\begin{theorem}[Successive decompositions of an open chain]
    \label{thm:seqanc_sweep_obc}
    \lean{OBCChainTensor.exists_isometric_chain_coeff}
    \leanok
    \uses{def:varying_bond_obc_coeff, def:seqanc_isometry_on}
    Let $A$ be a varying-bond open chain of length $n+1$ with dimensions
    $D_k$ and common bound $D$, and let $c\in\C$. There are $b_0=1$ and
    $b_k\le D_k$, families $Q_0,\dots,Q_n$ as in
    Theorem~\ref{thm:seqanc_sweep}, and a vector $r'$ supported below
    $b_{n+1}$, such that $Q^\sigma r'=c\,c_A(\sigma)\ket{0}$ for every
    $\sigma$ and $\|r'\|^2=\sum_\sigma|c\,c_A(\sigma)|^2$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep, thm:seqanc_coeff_corner, thm:seqanc_norm,
        thm:seqanc_support_propagation, thm:seqanc_boundary_coords}
    Apply Theorem~\ref{thm:seqanc_sweep} with $a=1$, $R_{-1}=c\ketbra{0}{0}$,
    the square padding of $A$, and $r=\ket{0}$. The padded site matrix
    $\widetilde A_p$ vanishes on the columns $\gamma\ge D_{p+1}$, so
    $b_{p+1}\le D_{p+1}$. The vector $Q^\sigma r'$ is supported below
    $b_0=1$, so it is its first coordinate times $\ket{0}$, and that
    coordinate is $c\,c_A(\sigma)$ by Theorem~\ref{thm:seqanc_coeff_corner}.
    The norm identity is Theorem~\ref{thm:seqanc_norm}.
\end{proof}

\begin{theorem}[Left-canonical open-boundary representation]
    \label{thm:seqanc_left_canonical}
    \lean{OBCChainTensor.exists_isometric_coeff_eq}
    \leanok
    \uses{def:varying_bond_obc_coeff}
    Every varying-bond open chain $A$ of length $n+1$ has a varying-bond
    open chain $A'$ with the same common bound, $c_{A'}=c_A$, and
    $D'_k\le D_k$ for every $k$, such that
    $\sum_i(A'^i_p)^\dagger A'^i_p=\Id$ for every site $p<n$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_obc, thm:seqanc_assemble}
    Apply Theorem~\ref{thm:seqanc_sweep_obc} with $c=1$ and assemble the
    result by Theorem~\ref{thm:seqanc_assemble}. The last bond has
    dimension one on both sides, and the others are bounded by
    Theorem~\ref{thm:seqanc_sweep_obc}.
\end{proof}

\begin{theorem}[Left-canonical representation of a normalized state]
    \label{thm:seqanc_left_canonical_norm}
    \lean{OBCChainTensor.exists_isometric_coeff_eq_of_norm}
    \leanok
    \uses{def:varying_bond_obc_coeff}
    If moreover $\sum_\sigma|c_A(\sigma)|^2=1$, the chain $A'$ of
    Theorem~\ref{thm:seqanc_left_canonical} can be chosen with
    $\sum_i(A'^i_p)^\dagger A'^i_p=\Id$ at every site $p\le n$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_obc, thm:seqanc_assemble}
    In the construction of Theorem~\ref{thm:seqanc_left_canonical}, the
    final vector $r'$ has norm one by Theorem~\ref{thm:seqanc_sweep_obc}, so
    Theorem~\ref{thm:seqanc_assemble} makes the last site isometric as well.
\end{proof}

\section{Resources of the successive decompositions}

After the successive decompositions, \cite{Schoen2005Sequential} state that
``simple rank considerations'' bound the size of the isometries: for
qubits, the isometry $V'_{[n-k]}$, which is produced after $k$ earlier
decompositions counted from the end at which they start, has dimension
$2\min[D,2^k]\times\min[D,2^{k+1}]$. According to \cite[Section~5.1]{PerezGarcia2007Matrix},
the proof of the characterization by open-boundary representations also
provides a recipe for the generation of any given state with minimal resources.
The entries of this section give upper bounds on the bond dimensions of the
representation produced by the decompositions of a given representation. Read
as an exact dimension, the printed size fails for low-rank states, since a
product state has a representation with all bond dimensions one; the minimal
bond dimensions are the cut ranks of Section~\ref{sec:seqanc_cut_ranks}
\cite{gap:ssvcw05_sequential_bond_upper_bound_scope}. In the present labelling
the decompositions start at site $0$, and the bound concerns the bond $k$
between the sites $k-1$ and $k$.

\begin{theorem}[Rank bound for one isometric site]
    \label{thm:seqanc_rank_bound}
    \lean{Matrix.card_le_mul_of_sum_conjTranspose_mul_eq_one,
        OBCChainTensor.bondDim_succ_le_mul}
    \leanok
    \uses{def:varying_bond_obc_chain}
    If $A^0,\dots,A^{d-1}$ are $a\times b$ matrices with
    $\sum_i(A^i)^\dagger A^i=\Id_b$, then $b\le ad$. In particular, a site $p$
    of a varying-bond open chain with $\sum_i(A_p^i)^\dagger A_p^i=\Id$
    satisfies $D_{p+1}\le d\,D_p$.
\end{theorem}
\begin{proof}
    \leanok
    The $(ad)\times b$ matrix $S$ obtained by stacking the $A^i$ satisfies
    $S^\dagger S=\Id_b$, so $b=\operatorname{rank}(S^\dagger S)
        =\operatorname{rank}S\le ad$.
\end{proof}

\begin{theorem}[Bond growth from the start of the decompositions]
    \label{thm:seqanc_bond_pow}
    \lean{OBCChainTensor.bondDim_le_pow}
    \leanok
    \uses{def:varying_bond_obc_chain, thm:seqanc_rank_bound}
    Let $d\ge 1$ and let $A$ be a varying-bond open chain of length $n+1$ with
    $\sum_i(A_p^i)^\dagger A_p^i=\Id$ for every site $p<n$. Then
    $D_k\le d^k$ for every $k\le n+1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_rank_bound}
    Induction on $k$: $D_0=1$, $D_{p+1}\le d\,D_p$ for $p<n$ by
    Theorem~\ref{thm:seqanc_rank_bound}, and $D_{n+1}=1\le d^{n+1}$.
\end{proof}

\begin{theorem}[Resources of the successive decompositions]
    \label{thm:seqanc_resources}
    \lean{OBCChainTensor.exists_isometric_coeff_eq_le_pow}
    \leanok
    \uses{def:varying_bond_obc_coeff}
    Let $d\ge 1$. The chain $A'$ of Theorem~\ref{thm:seqanc_left_canonical}
    can be chosen with $D'_k\le\min(D_k,d^k)$ for every $k$; in particular
    $D'_k\le\min(D,d^k)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_left_canonical, thm:seqanc_bond_pow}
    Theorem~\ref{thm:seqanc_bond_pow} applies to the chain of
    Theorem~\ref{thm:seqanc_left_canonical}.
\end{proof}

\begin{definition}[Reversed open chain]
    \label{def:seqanc_reverse}
    \lean{OBCChainTensor.reverse}
    \leanok
    \uses{def:varying_bond_obc_chain}
    The reverse of a varying-bond open chain $A$ of length $N$ has dimensions
    $D_{N-k}$ and site matrices $(A_{N-1-p}^i)^T$.
\end{definition}

\begin{theorem}[Coefficients of the reversed chain]
    \label{thm:seqanc_coeff_reverse}
    \lean{OBCChainTensor.bondDim_reverse, OBCChainTensor.coeff_reverse}
    \leanok
    \uses{def:seqanc_reverse, def:varying_bond_obc_coeff}
    The reverse of $A$ has coefficient
    $\sigma\mapsto c_A(\sigma_{N-1},\dots,\sigma_0)$.
\end{theorem}
\begin{proof}
    \leanok
    Relabel the virtual paths $k\mapsto N-k$ and the factors of the product
    $p\mapsto N-1-p$.
\end{proof}

\begin{theorem}[Successive decompositions from both ends]
    \label{thm:seqanc_resources_two_sided}
    \lean{OBCChainTensor.exists_isometric_coeff_eq_le_min}
    \leanok
    \uses{def:varying_bond_obc_coeff}
    Let $d\ge 1$ and let $A$ be a varying-bond open chain of length $N=n+1$.
    There is a varying-bond open chain $A'$ with the same common bound,
    $c_{A'}=c_A$, $\sum_i(A'^i_p)^\dagger A'^i_p=\Id$ for every site $p<n$, and
    \begin{align}
        D'_k & \le\min(D_k,d^k,d^{N-k})
        \notag
    \end{align}
    for every $k\le N$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_resources, thm:seqanc_coeff_reverse}
    Apply Theorem~\ref{thm:seqanc_resources} to the reverse of $A$ and reverse
    the result: by Theorem~\ref{thm:seqanc_coeff_reverse} this represents
    $c_A$ with dimensions at most $\min(D_k,d^{N-k})$. Apply
    Theorem~\ref{thm:seqanc_resources} once more. The bound $d^{N-k}$ is not
    stated in the source, which bounds only the side at which the
    decompositions start.
\end{proof}

\section{Cut ranks}
\label{sec:seqanc_cut_ranks}

For a configuration $\sigma$ of the first $k$ sites and a configuration $\tau$
of the remaining $N-k$ sites, write $\sigma\tau$ for the configuration of $N$
sites that agrees with $\sigma$ on the sites $p<k$ and with $\tau$ on the sites
$p\ge k$, the site $p$ carrying $\tau_{p-k}$. Following
\cite[Section~3.1]{PerezGarcia2007Matrix}, the bond dimensions of an
open-boundary representation are bounded below by the ranks of the reduced
density operators of the first $k$ sites, and successive Schmidt decompositions
of the state attain these ranks.

\begin{definition}[Cut matrix and cut rank]
    \label{def:seqanc_cut_matrix}
    \lean{MPSPreparation.cutConfig, MPSPreparation.cutMatrix,
        MPSPreparation.cutRank}
    \leanok
    For $\psi\in(\C^d)^{\otimes N}$ and $k\ge 0$, the \emph{cut matrix} of
    $\psi$ at $k$ is the matrix with rows indexed by the configurations
    $\sigma$ of $k$ sites, columns indexed by the configurations $\tau$ of
    $N-k$ sites, and entries $\psi(\sigma\tau)$; for $k>N$ only the first $N$
    letters of $\sigma$ enter. The \emph{cut rank} $\operatorname{rank}_k\psi$
    is its rank. For $k\le N$ it is the rank of the reduced density operator
    of the first $k$ sites.
\end{definition}

\begin{theorem}[Splitting a configuration at a cut]
    \label{thm:seqanc_cut_split}
    \lean{MPSPreparation.eval_cutConfig, MPSPreparation.cutConfig_restrict}
    \leanok
    \uses{def:seqanc_cut_matrix}
    Let $k\le N$. For square matrices $M_0,\dots,M_{N-1}$ indexed by the
    physical index, $M^{\sigma\tau}=(M_0^{\sigma_0}\cdots
        M_{k-1}^{\sigma_{k-1}})(M_k^{\tau_0}\cdots M_{N-1}^{\tau_{N-k-1}})$.
    Every configuration $c$ of $N$ sites is $\sigma\tau$ with
    $\sigma=(c_0,\dots,c_{k-1})$ and $\tau=(c_k,\dots,c_{N-1})$.
\end{theorem}
\begin{proof}
    \leanok
    Split the ordered product after its first $k$ factors; the second claim
    holds site by site.
\end{proof}

\begin{theorem}[Elementary bounds on cut ranks]
    \label{thm:seqanc_cut_rank_bounds}
    \lean{MPSPreparation.cutRank_le_pow, MPSPreparation.cutRank_le_pow_sub,
        MPSPreparation.cutRank_le_pow_half, MPSPreparation.cutRank_pos,
        MPSPreparation.cutRank_zero_eq_one, MPSPreparation.cutRank_self_eq_one}
    \leanok
    \uses{def:seqanc_cut_matrix}
    For every $k$, $\operatorname{rank}_k\psi\le d^k$ and
    $\operatorname{rank}_k\psi\le d^{N-k}$; if $d\ge 1$, then
    $\operatorname{rank}_k\psi\le d^{\lfloor N/2\rfloor}$. If $\psi\ne 0$, then
    $\operatorname{rank}_k\psi\ge 1$ for every $k\le N$, and
    $\operatorname{rank}_0\psi=\operatorname{rank}_N\psi=1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_cut_split}
    The rank is at most the number of rows, $d^k$, and of columns, $d^{N-k}$.
    If $k\le\lfloor N/2\rfloor$ use $d^k$, and otherwise $N-k\le\lfloor
        N/2\rfloor$. If $\psi(c)\ne 0$, the entry of the cut matrix at the
    restrictions of $c$ is nonzero by Theorem~\ref{thm:seqanc_cut_split}, so
    the rank is positive. The cut matrices at $0$ and $N$ have one row and one
    column, respectively.
\end{proof}

\begin{theorem}[Cut ranks bound the bond dimensions]
    \label{thm:seqanc_cut_rank_lower}
    \lean{OBCChainTensor.cutRank_coeff_le}
    \leanok
    \uses{def:seqanc_cut_matrix, def:varying_bond_obc_coeff}
    Every varying-bond open chain $A$ of length $N$ satisfies
    $\operatorname{rank}_k c_A\le D_k$ for every $k\le N$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_cut_split, thm:seqanc_coeff_corner}
    By Theorem~\ref{thm:seqanc_coeff_corner} and
    Theorem~\ref{thm:seqanc_cut_split},
    $c_A(\sigma\tau)=\sum_{\gamma}(L^\sigma)_{0\gamma}(R^\tau)_{\gamma 0}$, where
    $L^\sigma$ and $R^\tau$ are the products of the square paddings of the
    sites before and after the cut. The first factor of $R^\tau$ vanishes on
    the rows $\gamma\ge D_k$, and for $k=N$ the empty product $R^\tau=\Id$
    vanishes off the diagonal while $D_N=1$. So the cut matrix is the product
    of a matrix with $D_k$ columns and a matrix with $D_k$ rows.
\end{proof}

\begin{theorem}[Representation with the cut ranks as bond dimensions]
    \label{thm:seqanc_cut_rank_attained}
    \lean{OBCChainTensor.exists_coeff_eq_of_cutRank_le}
    \leanok
    \uses{def:seqanc_cut_matrix, def:varying_bond_obc_coeff}
    Let $N\ge 1$, let $\psi\ne 0$, and let $\operatorname{rank}_k\psi\le D$ for
    every $k\le N$. Then $\psi=c_A$ for a varying-bond open chain $A$ with
    common bound $D$ and $D_k=\operatorname{rank}_k\psi$ for every $k\le N$,
    such that $\sum_i(A_p^i)^\dagger A_p^i=\Id$ for every site $p<N-1$. If
    moreover $\sum_\sigma|\psi(\sigma)|^2=1$, this holds at the site $N-1$ as
    well.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_cut_rank_bounds, def:seqanc_smul_site, thm:seqanc_obc_cone,
        thm:seqanc_coeff_corner}
    This is the construction by successive Schmidt decompositions
    \cite[Section~3.1]{PerezGarcia2007Matrix}. Let $W_k$ be the column space
    of the cut matrix at $k$, a subspace of $(\C^d)^{\otimes k}$ of dimension
    $\operatorname{rank}_k\psi$, and let $e^{(k)}_\beta$ be an orthonormal
    basis of $W_k$. A column of the cut matrix at $k+1$, with the last of its
    $k+1$ sites fixed to $i$, is the column of the cut matrix at $k$ at the
    configuration $i\tau$; hence $w\in W_{k+1}$ implies
    $w(\,\cdot\,i)\in W_k$. Put
    $(A_k^i)_{\beta\beta'}=\langle e^{(k)}_\beta,e^{(k+1)}_{\beta'}(\,\cdot\,i)\rangle$,
    so that
    \begin{align}
        e^{(k+1)}_{\beta'}(\sigma i)
         & =\sum_\beta e^{(k)}_\beta(\sigma)\,(A_k^i)_{\beta\beta'}.
        \notag
    \end{align}
    Parseval's identity in $W_k$ and the orthonormality of $e^{(k+1)}$ give
    $\sum_i(A_k^i)^\dagger A_k^i=\Id$. By induction on $k$, the first row of
    the product of the padded sites before the cut $k$ is
    $\overline{u}\,e^{(k)}_\beta(\sigma)$, where $u$ is the value of the unit
    vector $e^{(0)}_0$ of the one-dimensional space $W_0$, so $|u|=1$. Since
    $W_N$ is spanned by $\psi$, $\psi=c_0\,e^{(N)}_0$ with
    $c_0=\langle e^{(N)}_0,\psi\rangle$; scaling the last site by $u\,c_0$
    gives $\psi$ by Theorem~\ref{thm:seqanc_obc_cone}. If $\psi$ is
    normalized, $|u\,c_0|=\|\psi\|=1$, and the last site remains isometric.
\end{proof}

\section{Sequential generation with an ancilla}

\begin{theorem}[Generated states are open-boundary states]
    \label{thm:seqanc_generated_obc}
    \lean{MPSPreparation.exists_hasOBCRep_of_isProbabilisticallyGenerated}
    \leanok
    \uses{def:seqanc_probabilistic, def:seqanc_obc_rep}
    If a nonzero vector $\psi$ is probabilistically generated with a
    $D$-dimensional ancilla, then $c\,\psi$ has an open-boundary
    representation with bond dimension at most $D$ for some $c\ne 0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep, thm:seqanc_assemble, thm:seqanc_boundary_coords}
    Since $\psi\ne 0$, $D>0$. For $N=0$ take $c=\psi()^{-1}$ and the empty
    chain. For $N\ge 1$ apply Theorem~\ref{thm:seqanc_sweep} with
    $R_{-1}=\ket{0}\bra{\varphi_F}$, the matrices $A_k$, and
    $r=\ket{\varphi_I}$, assemble the result by
    Theorem~\ref{thm:seqanc_assemble}, and take $c=1$.
\end{proof}

\begin{theorem}[Open-boundary states are generated]
    \label{thm:seqanc_obc_generated}
    \lean{MPSPreparation.isProbabilisticallyGenerated_of_hasOBCRep}
    \leanok
    \uses{def:seqanc_probabilistic, def:seqanc_obc_rep}
    If $c\ne 0$ and $c\,\psi$ has an open-boundary representation with bond
    dimension at most $D$, then $\psi$ is probabilistically generated with a
    $D$-dimensional ancilla.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_coeff_corner, thm:seqanc_bond_bound_pos,
        thm:seqanc_boundary_coords}
    Take the square padding $\widetilde A$ of the representation,
    $\ket{\varphi_F}=\ket{0}$ and $\ket{\varphi_I}=c^{-1}\ket{0}$; then
    $\bra{0}\widetilde A^\sigma c^{-1}\ket{0}=c^{-1}c\,\psi(\sigma)$ by
    Theorem~\ref{thm:seqanc_coeff_corner}.
\end{proof}

\begin{theorem}[Probabilistic sequential generation]
    \label{thm:seqanc_probabilistic}
    \lean{MPSPreparation.isProbabilisticallyGenerated_iff}
    \leanok
    \uses{def:seqanc_probabilistic, def:seqanc_obc_rep}
    Let $N\ge 1$ and $D\ge 1$. A vector $\psi$ is probabilistically generated
    with a $D$-dimensional ancilla if and only if it has an open-boundary
    representation with bond dimension at most $D$~\cite[Section~5,
        Theorem ``Sequential generation with ancilla'']{PerezGarcia2007Matrix}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_generated_obc, thm:seqanc_obc_generated,
        thm:seqanc_obc_cone}
    The zero vector has a representation by Theorem~\ref{thm:seqanc_obc_cone}.
    If $\psi\ne 0$, Theorem~\ref{thm:seqanc_generated_obc} gives a
    representation of $c\,\psi$ for some $c\ne 0$, and
    Theorem~\ref{thm:seqanc_obc_cone} one of $\psi=c^{-1}(c\,\psi)$. The
    converse is Theorem~\ref{thm:seqanc_obc_generated} with $c=1$.
\end{proof}

\begin{remark}
    For $N=0$ the only coefficient of a varying-bond open chain is the empty
    product $1$, and Theorems~\ref{thm:seqanc_generated_obc}
    and~\ref{thm:seqanc_obc_generated} hold up to the scalar $c$. An ancilla
    of dimension $D=0$ generates only the zero vector, which has no
    open-boundary representation with bond dimension at most $0$.
\end{remark}

\begin{theorem}[Deterministically generated states are normalized]
    \label{thm:seqanc_deterministic_norm}
    \lean{MPSPreparation.star_dotProduct_self_of_isDeterministicallyGenerated}
    \leanok
    \uses{def:seqanc_deterministic}
    Every deterministically generated vector $\psi$ satisfies
    $\sum_\sigma|\psi(\sigma)|^2=1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_step_isometry, thm:seqanc_norm}
    By Theorems~\ref{thm:seqanc_step_isometry} and~\ref{thm:seqanc_norm},
    $\sum_\sigma\|A^\sigma\varphi_I\|^2=\|\varphi_I\|^2=1$, and
    $\|A^\sigma\varphi_I\|^2=|\psi(\sigma)|^2\|\varphi_F\|^2=|\psi(\sigma)|^2$.
\end{proof}

\begin{theorem}[Deterministic schemes are probabilistic]
    \label{thm:seqanc_deterministic_probabilistic}
    \lean{MPSPreparation.isProbabilisticallyGenerated_of_isDeterministicallyGenerated}
    \leanok
    \uses{def:seqanc_deterministic, def:seqanc_probabilistic}
    Every deterministically generated vector is probabilistically generated
    with an ancilla of the same dimension.
\end{theorem}
\begin{proof}
    \leanok
    Use the induced matrices and project onto $\ket{\varphi_F}$:
    $\bra{\varphi_F}A^\sigma\ket{\varphi_I}
        =\psi(\sigma)\braket{\varphi_F}{\varphi_F}=\psi(\sigma)$.
\end{proof}

\begin{theorem}[Normalized open-boundary states are deterministically generated]
    \label{thm:seqanc_obc_deterministic}
    \lean{MPSPreparation.isDeterministicallyGenerated_of_hasOBCRep}
    \leanok
    \uses{def:seqanc_deterministic, def:seqanc_obc_rep}
    Let $d\ge 1$. If $\sum_\sigma|\psi(\sigma)|^2=1$, $c\ne 0$, and
    $c\,\psi$ has an open-boundary representation with bond dimension at
    most $D$, then $\psi$ is deterministically generated with a
    $D$-dimensional ancilla.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_obc, thm:seqanc_unitary_extension,
        thm:seqanc_block_congr, thm:seqanc_boundary_coords,
        thm:seqanc_bond_bound_pos}
    For $N=0$ take $\ket{\varphi_I}=\psi()\ket{0}$ and
    $\ket{\varphi_F}=\ket{0}$. Otherwise apply
    Theorem~\ref{thm:seqanc_sweep_obc} with the scalar $c^{-1}$: this gives
    isometric families $Q_p$ and a vector $r'$ with
    $Q^\sigma r'=\psi(\sigma)\ket{0}$ and $\|r'\|^2=\sum_\sigma|\psi(\sigma)|^2=1$.
    Extend each $Q_p$ to a unitary $U_p$ by
    Theorem~\ref{thm:seqanc_unitary_extension}, and take
    $\ket{\varphi_I}=r'$ and $\ket{\varphi_F}=\ket{0}$. The induced matrices
    agree with $Q_p$ on the used columns, so by
    Theorem~\ref{thm:seqanc_block_congr} the ancilla component at $\sigma$
    is $Q^\sigma r'=\psi(\sigma)\ket{0}$. The ancilla decouples because the
    first bond is one-dimensional.
\end{proof}

\begin{theorem}[Deterministic sequential generation]
    \label{thm:seqanc_deterministic}
    \lean{MPSPreparation.isDeterministicallyGenerated_iff}
    \leanok
    \uses{def:seqanc_deterministic, def:seqanc_obc_rep}
    Let $d\ge 1$ and $N\ge 1$. A vector $\psi$ is deterministically generated
    with a $D$-dimensional ancilla if and only if $\sum_\sigma|\psi(\sigma)|^2=1$
    and $\psi$ has an open-boundary representation with bond dimension at
    most $D$~\cite[Section~5,
        Theorem ``Sequential generation with ancilla'']{PerezGarcia2007Matrix}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_deterministic_norm,
        thm:seqanc_deterministic_probabilistic, thm:seqanc_generated_obc,
        thm:seqanc_obc_deterministic, thm:seqanc_obc_cone}
    For the forward direction, $\psi$ is normalized by
    Theorem~\ref{thm:seqanc_deterministic_norm}, hence nonzero, and
    probabilistically generated by
    Theorem~\ref{thm:seqanc_deterministic_probabilistic};
    Theorem~\ref{thm:seqanc_generated_obc} gives a representation of
    $c\,\psi$ for some $c\ne 0$, and Theorem~\ref{thm:seqanc_obc_cone} one of
    $\psi$. The converse is Theorem~\ref{thm:seqanc_obc_deterministic} with
    $c=1$.
\end{proof}

\begin{theorem}[Both schemes agree on normalized states]
    \label{thm:seqanc_det_iff_prob}
    \lean{MPSPreparation.isDeterministicallyGenerated_iff_isProbabilisticallyGenerated}
    \leanok
    \uses{def:seqanc_deterministic, def:seqanc_probabilistic}
    Let $d\ge 1$. A vector $\psi$ with $\sum_\sigma|\psi(\sigma)|^2=1$ is
    deterministically generated with a $D$-dimensional ancilla if and only if
    it is probabilistically generated with a $D$-dimensional ancilla.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_deterministic_probabilistic, thm:seqanc_generated_obc,
        thm:seqanc_obc_deterministic}
    One direction is Theorem~\ref{thm:seqanc_deterministic_probabilistic}.
    Conversely, $\psi$ is normalized, hence nonzero, so
    Theorem~\ref{thm:seqanc_generated_obc} gives a representation of
    $c\,\psi$ for some $c\ne 0$, and
    Theorem~\ref{thm:seqanc_obc_deterministic} applies.
\end{proof}

\section{The deterministic transition scheme}

\begin{definition}[Deterministic transition scheme]
    \label{def:seqanc_transition}
    \lean{MPSPreparation.IsTransitionInteraction,
        MPSPreparation.transitionStep, MPSPreparation.IsTransitionGenerated}
    \leanok
    \uses{def:seqanc_deterministic}
    Let $d\ge 1$ and let the ancilla be $\C^D\otimes\C^d$, the second factor
    being a tag, started in a unit vector $\ket{\varphi_I}$. In every step an
    arbitrary unitary $W^{[k]}$ acts on the ancilla, followed by the fixed
    interaction $T$ of the ancilla with the current site, which interchanges
    the tag with the site on site inputs $\ket{0}$:
    \begin{align}
        \ket{\varphi}\ket{t}\ket{0}
         & \mapsto\ket{\varphi}\ket{0}\ket{t},
         & t                                   & =0,\dots,d-1,
        \label{eq:seqanc_transition}
    \end{align}
    and the ancilla decouples after the last step, that is, the final joint
    state is $\ket{\varphi_F}\otimes\ket{\psi}$ for a unit vector
    $\ket{\varphi_F}\in\C^D\otimes\C^d$. For $d=2$ this is the third scheme
    of~\cite[Section~5.1]{PerezGarcia2007Matrix}; the extension to sites
    $\C^d$ is the remark of~\cite{Schoen2005Sequential} that the ideas and
    proofs apply to any multi-qudit state. The step with ancilla unitary $W$
    leaving the site in $\ket{i}$ is the matrix $S_i(W)$ on $\C^D\otimes\C^d$
    with entries
    \begin{align}
        S_i(W)_{x,y}=\delta_{x_2,0}\,W_{(x_1,i),y}.
        \label{eq:seqanc_transition_step}
    \end{align}
\end{definition}

\begin{theorem}[The interaction for qubits]
    \label{thm:seqanc_transition_qubit}
    \lean{MPSPreparation.isTransitionInteraction_two_iff}
    \leanok
    \uses{def:seqanc_transition}
    For $d=2$, an interaction $T$ satisfies (\ref{eq:seqanc_transition}) if
    and only if
    $T\ket{\varphi}\ket{1}\ket{0}=\ket{\varphi}\ket{0}\ket{1}$ and
    $T\ket{\varphi}\ket{0}\ket{0}=\ket{\varphi}\ket{0}\ket{0}$ for every
    $\ket{\varphi}\in\C^D$, the interaction
    of~\cite[Section~5.1]{PerezGarcia2007Matrix}.
\end{theorem}
\begin{proof}
    \leanok
    The two relations are the cases $t=1$ and $t=0$.
\end{proof}

\begin{theorem}[A transition step on a site in $\ket{0}$]
    \label{thm:seqanc_transition_step}
    \lean{MPSPreparation.transitionStep_eq_of_isTransitionInteraction}
    \leanok
    \uses{def:seqanc_transition}
    Let $T$ be an interaction on $\C^D\otimes\C^d\otimes\C^d$ satisfying
    (\ref{eq:seqanc_transition}), and let $W$ be a matrix on $\C^D\otimes\C^d$.
    Then for all $x,y$ and $i$,
    \begin{align*}
        \bra{x,i}T(W\otimes\Id)\ket{y,0}=S_i(W)_{x,y}.
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    For $z=(z_1,z_2)$, the relations (\ref{eq:seqanc_transition}) applied to
    $\ket{\varphi}=\ket{z_1}$ and $t=z_2$ give
    $T\ket{z_1,z_2,0}=\ket{z_1,0,z_2}$, so
    \begin{align*}
        \bra{x,i}T\ket{z,0}=\delta_{x_2,0}\,\delta_{x_1,z_1}\,\delta_{i,z_2}.
    \end{align*}
    Since $(W\otimes\Id)_{(z,j),(y,0)}=\delta_{j,0}\,W_{z,y}$,
    \begin{align*}
        \bra{x,i}T(W\otimes\Id)\ket{y,0}
        =\sum_z\bra{x,i}T\ket{z,0}\,W_{z,y}
        =\delta_{x_2,0}\,W_{(x_1,i),y}
        =S_i(W)_{x,y}.
    \end{align*}
\end{proof}

\begin{theorem}[Transition and deterministic schemes agree]
    \label{thm:seqanc_transition_deterministic}
    \lean{MPSPreparation.isTransitionGenerated_iff_isDeterministicallyGenerated,
        MPSPreparation.isDeterministicallyGenerated_of_isTransitionGenerated,
        MPSPreparation.isTransitionGenerated_of_isDeterministicallyGenerated}
    \leanok
    \uses{def:seqanc_transition, def:seqanc_deterministic}
    Let $d\ge 1$. A vector is generated by a deterministic transition scheme
    with ancilla $\C^D\otimes\C^d$ if and only if it is deterministically
    generated with a $D$-dimensional ancilla.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_unitary_extension, thm:seqanc_step_isometry,
        thm:seqanc_norm}
    A step of the transition scheme factors as $E R_i$, where
    $E\ket{\varphi}=\ket{\varphi}\ket{0}$ and $(R_i)_{\alpha,y}=W_{(\alpha,i),y}$,
    and $R_iE$ is the operation $\bra{\alpha,i}W\ket{\beta,0}$ of the unitary
    $W$ on ancilla $\otimes$ site. Given a deterministic scheme with unitaries
    $U^{[k]}$, take $W^{[k]}=U^{[k]}$ and the initial and final ancilla states
    $E\ket{\varphi_I}$ and $E\ket{\varphi_F}$; the ancilla components are then
    $E$ applied to those of the deterministic scheme.

    Conversely, after the first step the tag is $\ket{0}$, so the ancilla
    component at $\sigma$ is $E$ applied to the product of the matrices $R_iE$
    acting on $v_i=R_i\ket{\varphi_I}$, where $R_i$ comes from the first
    ancilla unitary. Since $\sum_i\|v_i\|^2=\|\varphi_I\|^2=1$, the vectors
    $v_i$ are the first column of a unitary on $\C^D\otimes\C^d$ by
    Theorem~\ref{thm:seqanc_unitary_extension}, and the scheme becomes a
    deterministic scheme with ancilla $\C^D$ started in $\ket{0}$. The final
    state is $\psi(\sigma)$ times the tag-zero part of $\varphi_F$. The total
    squared norm is $1$ by Theorem~\ref{thm:seqanc_norm}, so some
    $\psi(\sigma)$ is nonzero, the parts of $\varphi_F$ with a nonzero tag
    vanish, and the tag-zero part is normalized.
\end{proof}

\begin{theorem}[Transition schemes generate the open-boundary states]
    \label{thm:seqanc_transition}
    \lean{MPSPreparation.isTransitionGenerated_iff}
    \leanok
    \uses{def:seqanc_transition, def:seqanc_obc_rep}
    Let $d\ge 1$ and $N\ge 1$. A vector $\psi$ is generated by a deterministic
    transition scheme with ancilla $\C^D\otimes\C^d$ if and only if
    $\sum_\sigma|\psi(\sigma)|^2=1$ and $\psi$ has an open-boundary
    representation with bond dimension at most $D$. For $d=2$ this is the
    statement of~\cite[Section~5, Theorem ``Sequential generation with
        ancilla'']{PerezGarcia2007Matrix} for the third scheme.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_transition_deterministic, thm:seqanc_deterministic}
    Theorems~\ref{thm:seqanc_transition_deterministic}
    and~\ref{thm:seqanc_deterministic}.
\end{proof}

\section{Minimal resources}

By \cite[Section~5.1]{PerezGarcia2007Matrix}, the proof of the
characterization by open-boundary representations provides a recipe for the
generation of any given state with minimal resources, and
by~\cite{Schoen2005Sequential}, since every state has a matrix product
representation, it is a general recipe for its sequential generation.

\begin{theorem}[Open-boundary representations from cut ranks]
    \label{thm:seqanc_obc_cut_rank}
    \lean{MPSPreparation.hasOBCRep_iff_cutRank_le}
    \leanok
    \uses{def:seqanc_obc_rep, def:seqanc_cut_matrix}
    Let $N\ge 1$ and $D\ge 1$. A vector $\psi$ has an open-boundary
    representation with bond dimension at most $D$ if and only if
    $\operatorname{rank}_k\psi\le D$ for every $k\le N$. The backward
    direction is the statement that any state for which
    $\max_m\operatorname{rank}(\rho_m)\le D$ can be written as a matrix
    product state of bond dimension $D$~\cite[Section~3.1]{PerezGarcia2007Matrix}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_cut_rank_lower, thm:seqanc_cut_rank_attained,
        thm:seqanc_obc_cone}
    The forward direction is Theorem~\ref{thm:seqanc_cut_rank_lower}. For the
    backward direction, the zero vector is covered by
    Theorem~\ref{thm:seqanc_obc_cone} and every other vector by
    Theorem~\ref{thm:seqanc_cut_rank_attained}.
\end{proof}

\begin{theorem}[Sequential generation from cut ranks]
    \label{thm:seqanc_generated_cut_rank}
    \lean{MPSPreparation.isProbabilisticallyGenerated_iff_cutRank_le,
        MPSPreparation.isDeterministicallyGenerated_iff_cutRank_le}
    \leanok
    \uses{def:seqanc_probabilistic, def:seqanc_deterministic,
        def:seqanc_cut_matrix}
    Let $N\ge 1$. A vector $\psi$ is probabilistically generated with a
    $D$-dimensional ancilla if and only if $\operatorname{rank}_k\psi\le D$
    for every $k\le N$. If $d\ge 1$, it is deterministically generated with a
    $D$-dimensional ancilla if and only if moreover
    $\sum_\sigma|\psi(\sigma)|^2=1$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_probabilistic, thm:seqanc_obc_cut_rank,
        thm:seqanc_cut_rank_bounds, thm:seqanc_det_iff_prob,
        thm:seqanc_deterministic_norm, thm:seqanc_deterministic_probabilistic}
    For $D\ge 1$ combine Theorems~\ref{thm:seqanc_probabilistic}
    and~\ref{thm:seqanc_obc_cut_rank}. For $D=0$ both sides say $\psi=0$: the
    generated vector is a pairing in $\C^0$, and a nonzero vector has
    $\operatorname{rank}_0\psi=1$ by Theorem~\ref{thm:seqanc_cut_rank_bounds}.
    The deterministic case follows by
    Theorems~\ref{thm:seqanc_det_iff_prob},
    \ref{thm:seqanc_deterministic_norm}
    and~\ref{thm:seqanc_deterministic_probabilistic}.
\end{proof}

\begin{theorem}[Minimal ancilla dimension]
    \label{thm:seqanc_minimal_resources}
    \lean{MPSPreparation.isLeast_isProbabilisticallyGenerated,
        MPSPreparation.isLeast_isDeterministicallyGenerated}
    \leanok
    \uses{def:seqanc_probabilistic, def:seqanc_deterministic,
        def:seqanc_cut_matrix}
    Let $N\ge 1$. The least dimension of an ancilla that probabilistically
    generates $\psi$ is $\max_{k\le N}\operatorname{rank}_k\psi$. If $d\ge 1$
    and $\sum_\sigma|\psi(\sigma)|^2=1$, the least dimension of an ancilla
    that deterministically generates $\psi$ is the same number.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_generated_cut_rank}
    By Theorem~\ref{thm:seqanc_generated_cut_rank} the dimensions that work
    are exactly those at least every cut rank.
\end{proof}

\begin{theorem}[Every state is sequentially generated]
    \label{thm:seqanc_every_state}
    \lean{MPSPreparation.hasOBCRep_pow_half,
        MPSPreparation.isProbabilisticallyGenerated_pow_half,
        MPSPreparation.isDeterministicallyGenerated_pow_half,
        MPSPreparation.isTransitionGenerated_pow_half}
    \leanok
    \uses{def:seqanc_obc_rep, def:seqanc_probabilistic,
        def:seqanc_deterministic, def:seqanc_transition}
    Let $d\ge 1$ and $N\ge 1$. Every $\psi\in(\C^d)^{\otimes N}$ has an
    open-boundary representation with bond dimension at most
    $d^{\lfloor N/2\rfloor}$, and is probabilistically generated with an
    ancilla of dimension $d^{\lfloor N/2\rfloor}$. If moreover
    $\sum_\sigma|\psi(\sigma)|^2=1$, then $\psi$ is deterministically generated
    with an ancilla of dimension $d^{\lfloor N/2\rfloor}$ and generated by a
    deterministic transition scheme with ancilla
    $\C^{d^{\lfloor N/2\rfloor}}\otimes\C^d$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_cut_rank_bounds, thm:seqanc_obc_cut_rank,
        thm:seqanc_generated_cut_rank, thm:seqanc_transition_deterministic}
    Every cut rank is at most $d^{\lfloor N/2\rfloor}$ by
    Theorem~\ref{thm:seqanc_cut_rank_bounds}; apply
    Theorems~\ref{thm:seqanc_obc_cut_rank},
    \ref{thm:seqanc_generated_cut_rank}
    and~\ref{thm:seqanc_transition_deterministic}.
\end{proof}

\begin{remark}
    The existence statement of Theorem~\ref{thm:seqanc_every_state} is the
    existence part of the completeness theorem
    of~\cite[Section~3.1]{PerezGarcia2007Matrix}; its gauge conditions on the
    site matrices are not part of the statement
    \cite{gap:ssvcw05_sequential_bond_upper_bound_scope}.
\end{remark}

\section{Sequential generation without an ancilla}

\begin{definition}[Sequential scheme without an ancilla]
    \label{def:seqanc_no_ancilla}
    \lean{MPSPreparation.noAncillaState,
        MPSPreparation.IsProbabilisticallyGeneratedWithoutAncilla,
        MPSPreparation.IsDeterministicallyGeneratedWithoutAncilla}
    \leanok
    Let $d\ge 1$ and $N\ge 2$. Starting from $\ket{0}^{\otimes N}$, an operation $U^{[k]}$
    acts on sites $k$ and $k+1$, for $k=1,\dots,N-1$ in turn. The scheme is
    deterministic if every $U^{[k]}$ is unitary and probabilistic if the
    operations are arbitrary \cite[Section~5.2]{PerezGarcia2007Matrix}. Here
    sites carry the source's labels $1,\dots,N$; source site $j$ is ket
    position $N-j$ in the labelling of this chapter. After
    $U^{[k]}$ the scheme is described by the vector $\chi_k$ on the first
    $k+1$ sites, defined by $\chi_1=U^{[1]}\ket{0,0}$ and
    \begin{align*}
        \chi_k(\dots,i,\beta)=\sum_\alpha\bra{i,\beta}U^{[k]}\ket{\alpha,0}
        \chi_{k-1}(\dots,\alpha),
    \end{align*}
    and the generated state is $\chi_{N-1}$.
\end{definition}

\begin{theorem}[The scheme without an ancilla uses the next site as ancilla]
    \label{thm:seqanc_no_ancilla_joint}
    \lean{MPSPreparation.swapRows, MPSPreparation.swapRows_swapRows,
        MPSPreparation.swapRows_mem_unitaryGroup,
        MPSPreparation.noAncillaState_eq_jointState}
    \leanok

    Let $d\ge 1$ and $N\ge 2$, and let $\widehat U$ be the operation $U$ on
    $\C^d\otimes\C^d$ followed by the interchange of its two output factors;
    $\widehat{\widehat U}=U$, and $\widehat U$ is unitary if $U$ is. The state
    generated by $U^{[1]},\dots,U^{[N-1]}$ without an ancilla is the joint
    state of the scheme with an ancilla $\C^d$ for the unitaries
    $\widehat U^{[k]}$, started in $\ket{0}$, read out as the site at ket
    position $0$:
    \begin{align*}
        \psi(\sigma)=\bigl(A_0^{\sigma_1}\cdots A_{N-2}^{\sigma_{N-1}}\ket{0}
        \bigr)_{\sigma_0},
    \end{align*}
    with $A_p$ induced by $\widehat U^{[N-1-p]}$, so
    $(A_p^i)_{\beta\alpha}=\bra{i,\beta}U^{[N-1-p]}\ket{\alpha,0}$
    \cite[Section~5.2]{PerezGarcia2007Matrix}.
\end{theorem}
\begin{proof}
    \leanok
    Interchanging the outputs twice is the identity and permutes the rows of a
    unitary. The identity of states follows by induction on $N$ from the
    recursion for $\chi_k$, since $\bra{\beta,i}\widehat U\ket{\alpha,0}
        =\bra{i,\beta}U\ket{\alpha,0}$.
\end{proof}

\begin{theorem}[An ancilla read out as a site]
    \label{thm:seqanc_readout}
    \lean{MPSPreparation.hasOBCRep_of_eq_eval,
        MPSPreparation.star_dotProduct_self_of_eq,
        MPSPreparation.exists_isometric_chain_of_hasOBCRep,
        MPSPreparation.hasOBCRep_iff_exists_eq_jointState,
        MPSPreparation.hasOBCRep_and_norm_iff_exists_unitary_eq_jointState}
    \leanok
    \uses{def:seqanc_step, def:seqanc_obc_rep}
    Let $d\ge 1$ and $N=n+2$. The vectors
    $\psi(\sigma)=\bigl(A_0^{\sigma_1}\cdots A_n^{\sigma_{n+1}}
        \ket{\varphi_I}\bigr)_{\sigma_0}$, where $A_p$ is induced by an operation
    $U_{n-p}$ on $\C^d\otimes\C^d$ and $\ket{\varphi_I}\in\C^d$, are exactly
    the vectors with an open-boundary representation of bond dimension at most
    $d$. With unitary operations and a unit vector $\ket{\varphi_I}$ they are
    exactly the normalized such vectors.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep, thm:seqanc_assemble, thm:seqanc_coeff_corner,
        thm:seqanc_norm, thm:seqanc_block_congr, thm:seqanc_unitary_extension,
        thm:seqanc_step_isometry, thm:seqanc_bond_bound_pos,
        thm:seqanc_boundary_coords}
    Such a vector is $\bigl(\ket{0}\bra{e_{\sigma_0}}\bigr)A^{\sigma'}
        \ket{\varphi_I}$ read at the first coordinate, an open-boundary state by
    Theorems~\ref{thm:seqanc_sweep} and~\ref{thm:seqanc_assemble}; for unitary
    operations it is normalized by Theorems~\ref{thm:seqanc_step_isometry}
    and~\ref{thm:seqanc_norm}. Conversely, write an open-boundary state as
    $\psi(\sigma)=\bigl(R\,P_1^{\sigma_1}\cdots P_{N-1}^{\sigma_{N-1}}
        \ket{0}\bigr)_{\sigma_0}$ with $R_{j\gamma}$ the entries of the first
    site (Theorem~\ref{thm:seqanc_coeff_corner}), and run the successive
    decompositions of Theorem~\ref{thm:seqanc_sweep} starting from $R$: this
    gives $\psi(\sigma)=\bigl(Q_1^{\sigma_1}\cdots Q_{N-1}^{\sigma_{N-1}}
        r'\bigr)_{\sigma_0}$ with every $Q_p$ isometric on its bond block, and
    $\|r'\|=1$ when $\psi$ is normalized. Take $\ket{\varphi_I}=r'$ and
    operations inducing $Q_p$, which in the unitary case are unitary
    extensions by Theorem~\ref{thm:seqanc_unitary_extension}; then
    Theorem~\ref{thm:seqanc_block_congr} identifies the state.
\end{proof}

\begin{theorem}[Starting the ancilla in $\ket{0}$]
    \label{thm:seqanc_absorb_initial}
    \lean{MPSPreparation.stepMatrix_mul_kronecker_one,
        MPSPreparation.jointState_update_zero}
    \leanok
    \uses{def:seqanc_step}
    If $U$ is an operation on $\C^D\otimes\C^d$ and $W$ a matrix on $\C^D$, the
    matrices induced by $U(W\otimes\Id)$ are $A^iW$, where $A^i$ are those
    induced by $U$. Consequently, replacing the first operation $U_0$ by
    $U_0(W\otimes\Id)$ turns the ancilla component of the joint state started
    in $\ket{\varphi}$ into the one started in $W\ket{\varphi}$.
\end{theorem}
\begin{proof}
    \leanok
    $\bra{\alpha,i}U(W\otimes\Id)\ket{\beta,0}
        =\sum_\gamma\bra{\alpha,i}U\ket{\gamma,0}W_{\gamma\beta}$, and the
    first operation acts last in the ordered product.
\end{proof}

\begin{theorem}[Sequential generation without an ancilla]
    \label{thm:seqanc_no_ancilla}
    \lean{MPSPreparation.isProbabilisticallyGeneratedWithoutAncilla_iff,
        MPSPreparation.isDeterministicallyGeneratedWithoutAncilla_iff}
    \leanok
    \uses{def:seqanc_no_ancilla, def:seqanc_obc_rep}
    Let $d\ge 1$ and $N\ge 2$. The states generated by sequential schemes without an
    ancilla, either deterministically or probabilistically (the latter up to
    normalization), are exactly the states with an open-boundary
    representation of bond dimension at most $d$, normalized in the
    deterministic case~\cite[Section~5,
        Theorem ``Sequential generation without ancilla'']{PerezGarcia2007Matrix}.
    The condition $N\ge 2$ is necessary because the two-site gates require at
    least two sites~\cite{gap:pgvwc07_sequential_no_ancilla_two_sites}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_no_ancilla_joint, thm:seqanc_readout,
        thm:seqanc_absorb_initial}
    As in the source, the site $k+1$ serves as the ancilla of the $k$-th step.
    By Theorem~\ref{thm:seqanc_no_ancilla_joint}, a generated state is a
    vector of Theorem~\ref{thm:seqanc_readout} with $\ket{\varphi_I}=\ket{0}$,
    hence an open-boundary state of bond dimension at most $d$, normalized in
    the deterministic case. Conversely, Theorem~\ref{thm:seqanc_readout}
    writes an open-boundary state as such a vector for some
    $\ket{\varphi_I}$. Choose $W$ with $W\ket{0}=\ket{\varphi_I}$, the matrix
    with first column $\ket{\varphi_I}$ in the probabilistic case and a
    unitary in the deterministic case, absorb it into the first operation by
    Theorem~\ref{thm:seqanc_absorb_initial}, and interchange the outputs of
    every operation by Theorem~\ref{thm:seqanc_no_ancilla_joint}.
\end{proof}
