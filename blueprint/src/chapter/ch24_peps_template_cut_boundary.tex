\subsection{Physical boundaries of separated templates}
\label{sec:area_law_template_cut_boundary}

Let $\Lambda\subset\mathbb Z^2$ be any finite induced domain, $A\subseteq\Lambda$,
and $Z$ the endpoints of its unordered crossing edges. No connectedness
assumption is imposed on $\Lambda$ or $A$.

\begin{theorem}[Comparison with the ambient boundary]
    \label{thm:area_law_physical_ambient_boundary}
    \leanok
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.mem_boundaryEndpoints_of_mem_edgeBoundary,
        TNLean.PEPS.AreaLaw.Geometry.edgeBoundary_filter_image_subset_ambientBoundary,
        TNLean.PEPS.AreaLaw.Geometry.card_edgeBoundary_filter_le_ambientBoundary
    }
    \uses{def:area_law_ambient_boundary,def:area_law_cell_counts}
    For any finite ambient set $U$ disjoint from $Z$, inclusion into the
    lattice sends $\partial_\Lambda(A\cap U)$ injectively into $\partial U$.
    In particular, $|\partial_\Lambda(A\cap U)|\le|\partial U|$.
\end{theorem}
\begin{proof}
    \leanok
    Orient a physical crossing edge from $p\in A\cap U$ to
    $q\notin A\cap U$. If $q\notin A$, then $p\in Z$, contrary to
    $U\cap Z=\varnothing$. Thus $q\in A$ and $q\notin U$, and the edge
    crosses $U$. Inclusion preserves unordered pairs injectively.
\end{proof}

\begin{theorem}[Separation of template dilations]
    \label{thm:area_law_template_dilation_separation}
    \leanok
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.disjoint_ambientDilation}
    \uses{def:area_law_template}
    Suppose $D_0\ge1$, $s_0\ge1$, and
    $\operatorname{dist}_\infty(T,Z)>4D_0s_0$.
    For $0\le j\le s_0$, the ambient dilation $T_j$ is disjoint from $Z$.
\end{theorem}
\begin{proof}
    \leanok
    A point $z\in T_j\cap Z$ would be within distance $j\le s_0$ of a
    point of $T$, contradicting $4D_0s_0<\operatorname{dist}_\infty(T,Z)$.
\end{proof}

\begin{theorem}[Physical template boundary bounds]
    \label{thm:area_law_template_physical_boundary}
    \leanok
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.template_cut_boundary_card_le,
        TNLean.PEPS.AreaLaw.Geometry.template_core_boundary_card_le,
        TNLean.PEPS.AreaLaw.Geometry.template_shell_cut_boundary_card_le
    }
    \uses{thm:area_law_physical_ambient_boundary,
        thm:area_law_template_dilation_separation,lem:area_law_template_ambient_boundary}
    Let $T$ be a template separated from the cut, with $C_{\rm tpl}\ge24$
    and $D_0\ge1$. For every $0\le j\le s_0$,
    \begin{align}
        |\partial_\Lambda(A\cap T_j)|             & \le4n, \\
        |\partial_\Lambda(A\cap T)|               & \le4n, \\
        |\partial_\Lambda(A\cap(T_j\setminus T))| & \le8n.
    \end{align}
    These are the physical edge estimates in
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    The preceding separation estimate excludes cut endpoints from $T_j$
    and its shell. By Theorem~\ref{thm:area_law_physical_ambient_boundary}
    and Lemma~\ref{lem:area_law_template_ambient_boundary},
    \begin{align}
        |\partial_\Lambda(A\cap T_j)|             & \le|\partial T_j|\le4n,             \\
        |\partial_\Lambda(A\cap(T_j\setminus T))| & \le|\partial(T_j\setminus T)|\le8n.
    \end{align}
    The core is the case $j=0$.
    The manuscript fixes $D_0>2R+10$, which implies $D_0\ge1$ for $R\ge0$.
\end{proof}

\begin{lemma}[Strict clearance of contained dyadic squares]
    \label{lem:area_law_template_dyadic_clearance}
    \leanok
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.latticeDyadicCell_eq_Icc_product,
        TNLean.PEPS.AreaLaw.Geometry.latticeDyadicCell_interval_length,
        TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.lt_dist_of_mem_ambientDilation,
        TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.side_lt_dist_of_mem_ambientDilation,
        TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.dyadicCell_clearance
    }
    \uses{def:area_law_template,def:area_law_lattice_dyadic_square,
        def:area_law_cell_counts}
    Suppose $\operatorname{dist}_\infty(T,Z)>4D_0s_0$.
    For every $j\ge0$, $p\in T_j$, and $z\in Z$,
    \begin{align}
        \|p-z\|_\infty>4D_0s_0-j.
    \end{align}
    If $D_0\ge1$, $j\le s_0$, and $u\le s_0$, this distance exceeds $D_0u$.
    In particular, if $C_{k,c}\subseteq T_j$ and $2^k\le s_0$, every site
    of $C_{k,c}$ has distance greater than $D_0 2^k$ from every endpoint
    of a physical crossing edge. The square is exactly the product of
    integer intervals
    \begin{align}
        [2^kc_1,2^kc_1+2^k-1]\times[2^kc_2,2^kc_2+2^k-1],
    \end{align}
    each of which has $2^k$ sites. This is the strict-clearance step in
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}
    \leanok
    Choose $q\in T$ with $\|p-q\|_\infty\le j$. The triangle inequality gives
    $\|p-z\|_\infty\ge\|q-z\|_\infty-j>4D_0s_0-j$.
    Since $s_0>0$, $D_0\ge1$, and $j\le s_0$, we have
    $4D_0s_0-j>D_0s_0\ge D_0u$.
    Both endpoints of every crossing edge belong to $Z$.
    The integer interval description follows directly from the half-open
    dyadic square inequalities.
\end{proof}

\begin{theorem}[Graph neighborhoods separated from a cut]
    \label{thm:area_law_graph_neighborhood_cut_clearance}
    \lean{TNLean.PEPS.AreaLaw.graphNeighborhood_subset_of_cut_clearance}
    \leanok
    \uses{def:area_law_domain,def:area_law_cell_counts}
    Let $\Lambda\subset\mathbb Z^2$ be finite, $A\subseteq\Lambda$, and
    $T\subset\mathbb Z^2$ finite. For integers $L,R\ge0$, write
    $T_L=T+[-L,L]^2_{\mathbb Z}$. Suppose
    $\|t-z\|_\infty>L+R$ for every $t\in T$ and every endpoint
    $z$ of a physical crossing edge of $A$. Then
    \begin{align}
        N_R^\Lambda(A\cap T_L)\subseteq A\cap T_{L+R}.
    \end{align}
    The statement includes empty targets and disconnected domains. This is
    the cut-exclusion step in the proof of
    \cite[Theorem~1.1]{OpenAI2026AreaLaw}; separation from the earlier
    members of a family is a further geometric condition.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:al_scan_ambient_rows}
    Let $x$ be joined to $y\in A\cap T_L$ by a path of length at most $R$.
    Then $T$ is nonempty. The depth
    $d(v)=\min_{t\in T}\|v-t\|_\infty$ satisfies $d(y)\le L$ and
    changes by at most one along each graph edge. Every endpoint of a
    prefix of the path therefore has depth at most $L+R$.
    If the path leaves $A$, its first crossing gives a physical cut
    endpoint of depth at most $L+R$, contrary to the hypothesis.
    Thus $x\in A$; the depth bound at $x$ gives $x\in T_{L+R}$.
\end{proof}

\begin{theorem}[Exclusion of the exterior and a remote union]
    \label{thm:area_law_graph_neighborhood_remote_exclusion}
    \lean{TNLean.PEPS.AreaLaw.exterior_union_subset_graphNeighborhood_compl_of_clearance}
    \uses{def:area_law_domain,def:area_law_cell_counts}
    Let $\Lambda,T\subset\mathbb Z^2$ be finite, let $A,J\subseteq\Lambda$,
    and let $L,R\in\mathbb N$. Let $Z_A$ be the set of endpoints of
    the physical cut edges. Suppose $\|t-z\|_\infty>L+R$ for every
    $t\in T$ and $z\in Z_A$, and $\|t-x\|_\infty>L+R$ for every
    $t\in T$ and $x\in J$.
    Put $S=A\cap(T+[-L,L]^2_{\mathbb Z})$. Then
    \begin{align*}
        (\Lambda\setminus A)\cup J
        &\subseteq\Lambda\setminus N_R^\Lambda(S).
    \end{align*}
    Here $N_R^\Lambda$ is the closed graph neighbourhood in the induced
    domain. The assertion permits disconnected domains and empty targets.
    In the application to \cite[Theorem~1.1, lines~825--838 of
    Section~11]{OpenAI2026AreaLaw}, $J$ is the whole union of earlier
    actual pieces of the same family.
\end{theorem}
\begin{proof}
    \uses{thm:area_law_graph_neighborhood_cut_clearance}
    By Theorem~\ref{thm:area_law_graph_neighborhood_cut_clearance},
    every $x\in N_R^\Lambda(S)$ belongs to $A$ and to
    $T+[-(L+R),L+R]^2_{\mathbb Z}$. Thus it lies outside
    $\Lambda\setminus A$. The dilation supplies a point $t\in T$
    with $\|t-x\|_\infty\le L+R$, so the strict clearance from $J$
    implies $x\notin J$. Both parts of the union are therefore excluded.
\end{proof}
