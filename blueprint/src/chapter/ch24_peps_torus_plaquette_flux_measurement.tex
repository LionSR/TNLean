\begin{definition}[The native plaquette walk and its physical sites]%
    \label{def:peps_torus_plaquette_flux_geometry}
    \lean{TNLean.PEPS.torusPlaquetteWalk,
        TNLean.PEPS.torusPlaquetteRegion,
        TNLean.PEPS.torusPlaquetteRegion_card}
    \leanok
    Following \cite[proof of Theorem~6.15]{Schuch2010PEPS}, on a torus with periods $L_x,L_y\ge3$, for $v=(x,y)$ (including across either seam) define the plaquette walk $p_v$ and four-site region $R_v$ by
    \begin{align}
        p_v & :\ (x,y)\longrightarrow(x+1,y)
        \longrightarrow(x+1,y+1)\longrightarrow(x,y+1)
        \longrightarrow(x,y), \notag                              \\
        R_v & =\{(x,y),(x+1,y),(x+1,y+1),(x,y+1)\}, & |R_v| & =4.
        \label{eq:peps_plaquette_flux_walk}
    \end{align}
\end{definition}

\begin{definition}[A one-bond regular flux string]%
    \label{def:peps_torus_plaquette_flux_insertion}
    \lean{TNLean.PEPS.torusPlaquetteFluxInsertion,
        TNLean.PEPS.regularWalkHolonomy_torusPlaquetteFluxInsertion}
    \leanok
    \uses{def:peps_torus_plaquette_flux_geometry,
        thm:peps_regular_walk_holonomy}
    For $g\in G$, let $u^{v,g}$ (\cite[Section~6.5 and proof of Theorem~6.15]{Schuch2010PEPS}) assign directed bond value $g$ to the first side $(x,y)\to(x+1,y)$ of $p_v$ (i.e., $g$ or $g^{-1}$ on the ordered edge) and $1$ to all other bonds; its holonomy satisfies
    \begin{align}
        \operatorname{Hol}_{u^{v,g}}(p_v) & =g.
        \label{eq:peps_plaquette_flux_holonomy}
    \end{align}
\end{definition}
\begin{proof}\leanok
    Since $L_x,L_y\ge3$, the three remaining sides of $p_v$ contain vertices at height $y+1\neq y$, so their ordered edges are distinct from $(x,y)\to(x+1,y)$ and carry $1$; the reverse-ordered product along $p_v$ (Theorem~\ref{thm:peps_regular_walk_holonomy}) is $1\cdot1\cdot1\cdot g=g$.
\end{proof}

\begin{theorem}[Four-site measurement of native plaquette holonomy]%
    \label{thm:peps_torus_plaquette_holonomy_measurement}
    \lean{TNLean.PEPS.IsGIsometric.exists_torusPlaquette_physicalMeasurement,
        TNLean.PEPS.IsGIsometric.exists_torusPlaquette_holonomy_cutMeasurement}
    \leanok
    \uses{def:peps_torus_plaquette_flux_geometry,
        thm:peps_closed_walk_class_physical_measurement,
        thm:peps_regular_boundary_route,
        thm:peps_regular_torus_incident_isometry,
        thm:peps_regular_open_region_projector}
    On a torus with periods $L_x,L_y\ge3$ and a regular $G$-isometric local tensor for a finite group $G$ (\cite[proof of Theorem~6.15]{Schuch2010PEPS}; see \cite{gap:rmp_peps_quantum_double_g_isometry}), at every $v$ there exists a complete orthogonal projective measurement $\{Q_C\}_C\cup\{Q_\perp\}$ ($Q_i=Q_i^\dagger\ge0$, $Q_iQ_j=\delta_{ij}Q_i$, $\sum_CQ_C+Q_\perp=I$) on the four spins of $R_v$ such that for every bond assignment $u$, crossing configuration $\theta$, twisted open column $\psi_{u,\theta}$, and globally contracted cut matrix $M_u$ across $R_v$,
    \begin{align}
        Q_C\psi_{u,\theta}
                               & =\mathbf1_{\{[\operatorname{Hol}_u(p_v)]=C\}}\psi_{u,\theta}, &
        Q_\perp\psi_{u,\theta} & =0,
        \label{eq:peps_plaquette_holonomy_measurement}                                           \\
        Q_CM_u                 & =\mathbf1_{\{[\operatorname{Hol}_u(p_v)]=C\}}M_u,             &
        Q_\perp M_u            & =0.
        \label{eq:peps_plaquette_arbitrary_closed_measurement}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Passing regular $G$-isometry to incident-edge coordinates via Theorem~\ref{thm:peps_regular_torus_incident_isometry} and applying Theorem~\ref{thm:peps_closed_walk_class_physical_measurement} to the connected walk $p_v$ on $R_v$ gives (\ref{eq:peps_plaquette_holonomy_measurement}); factoring $M_u$ across $\partial R_v$ via Theorem~\ref{thm:peps_regular_boundary_route} and Theorem~\ref{thm:peps_regular_open_region_projector} gives (\ref{eq:peps_plaquette_arbitrary_closed_measurement}).
\end{proof}

\begin{theorem}[Detection of a native one-bond flux string]%
    \label{thm:peps_torus_plaquette_flux_class_measurement}
    \lean{TNLean.PEPS.IsGIsometric.exists_torusPlaquette_fluxClass_measurement,
        TNLean.PEPS.IsGIsometric.exists_torusPlaquette_fluxClass_cutMeasurement}
    \leanok
    \uses{thm:peps_torus_plaquette_holonomy_measurement,
        def:peps_torus_plaquette_flux_insertion,
        thm:peps_regular_open_region_projector}
    Under the hypotheses of Theorem~\ref{thm:peps_torus_plaquette_holonomy_measurement} (\cite[Theorem~6.15]{Schuch2010PEPS}), the measurement $\{Q_C\}_C\cup\{Q_\perp\}$ on $R_v$ satisfies, for the cut matrix $M_{v,g}=M_{u^{v,g}}$ of every one-bond flux string $u^{v,g}$,
    \begin{align}
        Q_CM_{v,g}      & =\mathbf1_{\{[g]=C\}}M_{v,g}, &
        Q_\perp M_{v,g} & =0.
        \label{eq:peps_plaquette_flux_closed_measurement}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Substitute (\ref{eq:peps_plaquette_flux_holonomy}) into (\ref{eq:peps_plaquette_arbitrary_closed_measurement}).
\end{proof}

\begin{definition}[Flux strings attached to each plaquette side]%
    \label{def:peps_torus_plaquette_all_sides_insertion}
    \lean{TNLean.PEPS.torusPlaquetteSide,
        TNLean.PEPS.torusPlaquetteSideInsertion,
        TNLean.PEPS.regularWalkHolonomy_torusPlaquetteSideInsertion}
    \leanok
    \uses{def:peps_torus_plaquette_flux_geometry,
        thm:peps_regular_walk_holonomy}
    Indexing the four oriented sides of $p_v$ by $i\in\{0,1,2,3\}$ (\cite[proof of Theorem~6.15]{Schuch2010PEPS}), let $u^{v,i,g}$ assign directed bond value $g$ to side $i$ and $1$ to all other bonds; then
    \begin{align}
        \operatorname{Hol}_{u^{v,i,g}}(p_v) & =g.
        \label{eq:peps_plaquette_all_sides_holonomy}
    \end{align}
\end{definition}
\begin{proof}\leanok
    Since $L_x,L_y\ge3$, the four sides of $p_v$ have distinct ordered endpoint pairs, so side $i$ contributes $g$ and the other three contribute $1$ in Theorem~\ref{thm:peps_regular_walk_holonomy}.
\end{proof}

\begin{theorem}[A single flux measurement for all four attachment directions]%
    \label{thm:peps_torus_plaquette_all_sides_measurement}
    \lean{TNLean.PEPS.IsGIsometric.exists_torusPlaquette_allSides_cutMeasurement}
    \leanok
    \uses{thm:peps_torus_plaquette_holonomy_measurement,
        def:peps_torus_plaquette_all_sides_insertion}
    Under the hypotheses of Theorem~\ref{thm:peps_torus_plaquette_holonomy_measurement} (\cite[proof of Theorem~6.15]{Schuch2010PEPS}, without asserting parent-Hamiltonian excitation membership), the single measurement $\{Q_C\}_C\cup\{Q_\perp\}$ on $R_v$ (independent of $i,g$) satisfies, for all $i\in\{0,1,2,3\}$, $g\in G$, and $M_{v,i,g}=M_{u^{v,i,g}}$,
    \begin{align}
        Q_CM_{v,i,g}      & =\mathbf1_{\{[g]=C\}}M_{v,i,g}, &
        Q_\perp M_{v,i,g} & =0.
        \label{eq:peps_plaquette_all_sides_measurement}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Substitute (\ref{eq:peps_plaquette_all_sides_holonomy}) into (\ref{eq:peps_plaquette_arbitrary_closed_measurement}).
\end{proof}

\begin{corollary}[Orthogonality of distinct measured plaquette classes]%
    \label{cor:peps_torus_plaquette_class_orthogonality}
    \lean{TNLean.PEPS.IsGIsometric.torusPlaquette_cutMatrix_orthogonal_of_class_ne}
    \leanok
    \uses{thm:peps_torus_plaquette_holonomy_measurement}
    Under Theorem~\ref{thm:peps_torus_plaquette_holonomy_measurement} (\cite[Theorem~6.15]{Schuch2010PEPS}), if bond assignments $u,w$ satisfy $[\operatorname{Hol}_u(p_v)]\neq[\operatorname{Hol}_w(p_v)]$, then
    \begin{align}
        M_u^\dagger M_w & =0.
        \label{eq:peps_plaquette_class_orthogonality}
    \end{align}
\end{corollary}
\begin{proof}\leanok
    For $C=[\operatorname{Hol}_u(p_v)]$, (\ref{eq:peps_plaquette_arbitrary_closed_measurement}) gives $Q_CM_u=M_u$ and $Q_CM_w=0$, so $M_u^\dagger M_w=(Q_CM_u)^\dagger M_w=M_u^\dagger Q_CM_w=0$.
\end{proof}
