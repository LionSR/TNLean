\begin{lemma}[Right invariance of a shared internal regular bond]%
    \label{lem:peps_regular_region_bond_right_invariance}
    \lean{TNLean.PEPS.regularRegionHalfEdgeRightMul,
        TNLean.PEPS.regularProjectorTwistedRegionMatrix_halfEdgeRightMul,
        TNLean.PEPS.regularProjectorOpenRegionMatrix_halfEdgeRightMul,
        TNLean.PEPS.regularProjectorOpenRegionMatrix_internalEdgeRightMul,
        TNLean.PEPS.regionPhysicalMap_halfEdgeRightMul_of_mem_regularGroundSpace,
        TNLean.PEPS.regionPhysicalMap_slice_halfEdgeRightMul_of_parent_annihilates}
    \leanok
    \uses{def:peps_region_ground_space, def:peps_region_parent_interaction}
    On a finite graph region $R$, choose $r_e\in G$ with $r_e=1$ for every
    crossing bond $e\in\partial E_R$, and define the half-edge right
    translation $(R_r\alpha)_{(v,e)}=\alpha_{(v,e)}r_e$. For the canonical
    regular averaging contraction $\Pi_R$ (and its left-twisted variant
    $\Pi_R^u$), every boundary column $\theta$ satisfies
    $(\Pi_R^u\ket{\theta})\circ R_r=\Pi_R^u\ket{\theta}$, including when $r$
    is supported on a single internal bond $e\in E_R$.

    Let the physical site coefficients $a_v$ and local inverse coefficients
    $F_v$ satisfy
    \begin{align}
        \sum_s F_v(\alpha,s)a_v(\eta,s) &= \Pi_v(\alpha,\eta),
        \label{eq:peps_regular_bond_local_inverse}
    \end{align}
    where $\Pi_v$ is the regular averaging projector, and put
    $F_R=\bigotimes_{v\in R}F_v$. Then every $\phi\in\mathcal G_R$
    (Definition~\ref{def:peps_region_ground_space}) satisfies
    $(F_R\phi)\circ R_r=F_R\phi$. Consequently, if
    $h_R^{\mathrm{global}}\Psi=0$
    (Definition~\ref{def:peps_region_parent_interaction}), every exterior
    slice $\Psi(\cdot,\tau)$ satisfies
    $(F_R\Psi(\cdot,\tau))\circ R_r=F_R\Psi(\cdot,\tau)$. Local inverses
    satisfying~(\ref{eq:peps_regular_bond_local_inverse}) exist whenever each
    site tensor is regular $G$-injective.

    No $G$-isometry, flat-connection reconstruction, or global
    closure-span representation is assumed: this local
    identity of the contractions in the accessible
    coordinate argument of \cite{Schuch2010PEPS} is an auxiliary
    regular bond relation and does not establish global closure spanning
    \cite{gap:rmp_peps_quantum_double_g_isometry}.
\end{lemma}

\begin{proof}\leanok
    Because left group translation commutes with legwise right multiplication,
    $\Pi_v(\alpha\cdot r,\eta\cdot r)=\Pi_v(\alpha,\eta)$. Reindexing each
    summed internal bond variable $\eta_e\mapsto\eta_e r_e$ while keeping
    $r_e=1$ on $\partial E_R$ preserves the contraction sum, and associativity
    $(u_e\eta_e)r_e=u_e(\eta_e r_e)$ gives
    $(\Pi_R^u\ket{\theta})\circ R_r=\Pi_R^u\ket{\theta}$.

    By~(\ref{eq:peps_regular_bond_local_inverse}), $F_R$ maps each physical
    open column $T_R(\theta;\cdot)$ to $\Pi_R\ket{\theta}$, so linearity over
    $\mathcal G_R=\spn\{T_R(\theta;\cdot):\theta\}$ yields
    $(F_R\phi)\circ R_r=F_R\phi$. Finally, $h_R^{\mathrm{global}}\Psi=0$
    implies $\Psi(\cdot,\tau)\in\mathcal G_R$ for every $\tau$.
\end{proof}
