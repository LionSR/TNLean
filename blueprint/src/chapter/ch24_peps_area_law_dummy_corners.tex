\section{Dummy-neighborhood corners on fine-cell sides}
\label{sec:area_law_dummy_corners}

Fix an origin $o\in\mathbb R^2$, a finite endpoint set $Z$, an integer width
$C_0$, and a lower layer index $k_0$. Let $P_{k_0}$ be the set of indices of
the coarse cells forming $N_{k_0}$. The results below apply to each coarse
cell individually; see \cite[Section~11]{OpenAI2026AreaLaw}.

% Source: 10-geometry.tex, lines 160--191, 200--207 and 299--310.
% Constancy of the opposing dummy region and its label remains separate.

\begin{lemma}[Dummy-cell corner on a full fine-cell side]
    \label{thm:area_law_dummy_corner_side}
    \lean{TNLean.PEPS.AreaLaw.Geometry.dyadicNeighborhood_corner_on_side}
    \leanok
    \uses{def:area_law_dyadic_layers,def:area_law_cell_corner,
        def:area_law_dyadic_cell_side,def:area_law_fine_layer_indices,
        def:area_law_fine_belt_scales}
    Suppose $C_0\ge2$, $k\ge k_0$, $z\in F_{k,\ell_k}$, and
    $w\in P_{k_0}$. For $\varepsilon\in\{0,1\}^2$, set
    $x=o+2^{k_0}(w+\varepsilon)$. If $x$ lies on the side $S_{\ell_k,z,s}$,
    then $x=a_{\ell_k,z,s}$ or $x=b_{\ell_k,z,s}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dummy_contact_index,thm:area_law_fine_cell_containment,
        thm:area_law_fine_belt_scales,thm:area_law_cell_fan_mark_vertices,
        thm:area_law_belt_marks_closed_cell,thm:area_law_dyadic_closures,
        thm:area_law_dyadic_cell_distance,
        def:area_law_affine_mesh,thm:area_law_affine_mesh_point_separation}
    The corner $x$ belongs to $\overline{N_{k_0}}$, while the side
    $S_{\ell_k,z,s}=[a,b]$ lies in $\overline{Q_{k,z}}\subseteq\overline{D_k}$,
    so $k=k_0$. Set $q=2^{\ell_{k_0}}$. Since $\ell_{k_0}\le k_0$, the point
    $x$ and both side endpoints $a,b$ lie in the mesh $o+q\mathbb Z^2$. If
    $x\ne a$ and $x\ne b$, mesh separation and $x\in[a,b]$ give
    \begin{align}
        2q\le\|a-x\|_\infty+\|x-b\|_\infty
        =\|a-b\|_\infty\le q,
    \end{align}
    contradicting $q>0$.
\end{proof}

\begin{lemma}[Dummy-cell corner on an elementary fine-cell segment]
    \label{thm:area_law_dummy_corner_elementary_side}
    \lean{TNLean.PEPS.AreaLaw.Geometry.dyadicNeighborhood_corner_on_elementarySide}
    \leanok
    \uses{def:area_law_dyadic_layers,def:area_law_cell_corner,
        def:area_law_cell_fan,def:area_law_fine_layer_indices,
        def:area_law_fine_belt_scales}
    Suppose $C_0\ge2$, $k\ge k_0$, $z\in F_{k,\ell_k}$, and
    $w\in P_{k_0}$. Choose any midpoint subdivision of the sides of
    $\overline{Q_{k,z}}$ and let $[u,v]$ be one of its elementary segments.
    For $\varepsilon\in\{0,1\}^2$, if the corner
    $x=o+2^{k_0}(w+\varepsilon)$ belongs to $[u,v]$, then $x=u$ or $x=v$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dummy_corner_side,thm:area_law_elementary_side_geometry}
    Since $[u,v]$ lies on a full side $[a,b]$ of $\overline{Q_{k,z}}$, the
    preceding theorem gives $x=a$ or $x=b$. Whether the side is unsplit
    ($[u,v]=[a,b]$) or split at $(a+b)/2$, the only endpoints of $[a,b]$
    contained in $[u,v]$ are endpoints of $[u,v]$, so $x=u$ or $x=v$.
\end{proof}
