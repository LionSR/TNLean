\section{A common middle family after a change of interval}

For a finite set of sites $S$, write $\mathcal C(S)=\{0,\ldots,q-1\}^S$.
Restriction of configurations identifies the full chain with the tensor
product of a region and its complement. In the identities below the
complement factor is listed first. Open matrix product representations
use the paired row and column alphabet, in increasing site order.


\begin{lemma}[Natural coordinates of one transferred interval factor]
    \label{thm:interval_transfer_entry_coordinates}
    \lean{TNLean.MPS.AreaLaw.exists_intervalSlice_changingRegionalMiddleFactor}
    \leanok
    \uses{def:regional_middle_transferred_family,def:physical_interval_increasing_coordinates}
    Let $q\ge1$ and let $E=\lbrack c,d)$ and $J=\lbrack u,v)$ in a $T$-site chain satisfy
    $c\le d\le T$, $u\le v\le T$ and $\max(c,u)\le\min(d,v)$.
    Supply one matrix $V$ on $\mathcal C(E)$ and fixed configurations on
    $E\setminus J$ and $J\setminus E$ defining the transferred matrix $W$.
    In increasing coordinates, there are nonnegative integers $a,b,l$ with
    $a+l\le d-c$ and $b+l\le v-u$, old full configurations $f,g$, and new
    outside configurations $\eta,\theta$ such that
    \begin{align}
        W(\sigma,\tau)
         & =\mathbf1_{\sigma|_{[b,b+l)^c}=\eta}
        \mathbf1_{\tau|_{[b,b+l)^c}=\theta}
        V\bigl(\iota_a(f,\sigma|_{[b,b+l)}),
        \iota_a(g,\tau|_{[b,b+l)})\bigr).
        \label{eq:interval_transfer_entry_coordinates}
    \end{align}
    Here $V,W$ are read in their natural coordinates, and $\iota_a$ has
    the meaning in Lemma~\ref{thm:trace_interval_slice_configuration_extend}.
    Complements in this formula are in the new natural interval.
    The overlap length $l$ may be zero. No matrix product representation
    is assumed.
\end{lemma}
\begin{proof}\leanok
    Take the increasing coordinates of $E\cap J$. Extend the supplied old
    and new outside configurations to full natural configurations, using
    $q\ge1$ for unused entries. Evaluation of the prescribed regional
    matrix gives (\ref{eq:interval_transfer_entry_coordinates}).
\end{proof}

\begin{lemma}[Transfer of a common middle family between intervals]
    \label{thm:interval_mpo_common_family_transfer}
    \lean{TNLean.MPS.AreaLaw.changingRegionalMiddleFamily_mpo_and_global_decomposition}
    \leanok
    \uses{def:regional_middle_transferred_family,
        thm:regional_middle_exterior_transfer,
        def:physical_interval_increasing_coordinates,
        thm:physical_interval_subinterval_coordinates}
    Let $q$ be a positive integer, and let $c,d,u,v,T$ be nonnegative
    integers satisfying
    \begin{align}
        c\le d\le T,\qquad u\le v\le T,\qquad
        \max(c,u)\le\min(d,v).
        \label{eq:interval_mpo_transfer_endpoints}
    \end{align}
    Put $E={[c,d)}$,
    $J={[u,v)}$, $A=E\setminus J$, $B=J\setminus E$, and
    $C=E\cap J$. Let $\iota$ be a finite index set and $D:\iota\to\N$.
    Suppose that for each
    $t\in\iota$, the matrix $V_t$ on $\mathcal C(E)$ has an open matrix
    product representation with internal bond dimensions at most $D_t$.

    Define the prescribed new family, indexed by
    \begin{align}
        I=\iota\times\mathcal C(A)^2\times\mathcal C(B)^2,
        \label{eq:interval_mpo_transfer_index}
    \end{align}
    by the entries
    \begin{align}
         & W_{t,\alpha_r,\alpha_c,\eta_r,\eta_c}(\sigma,\tau)\notag   \\
         & \quad=\mathbf1_{\sigma|_B=\eta_r}\mathbf1_{\tau|_B=\eta_c}
        V_t\bigl(\alpha_r\sqcup\sigma|_C,
        \alpha_c\sqcup\tau|_C\bigr).
        \label{eq:interval_mpo_transfer_factor}
    \end{align}
    Here $\sigma,\tau\in\mathcal C(J)$, and the disjoint union combines
    configurations on $A$ and $C$ into a configuration on $E$.
    Then
    \begin{align}
        |I|=|\iota|q^{2|A|+2|B|},
        \label{eq:interval_mpo_transfer_count}
    \end{align}
    and the member indexed by $(t,\alpha_r,\alpha_c,\eta_r,\eta_c)$ has an
    open matrix product representation on $J$, in increasing site order,
    with internal bond dimensions at most $\max(1,D_t)$.

    The family $W$ is fixed before any exterior coefficient matrices are
    supplied. For every family $M_t$ on $\mathcal C(E^c)$ there are
    matrices $M'_r$ on $\mathcal C(J^c)$ such that the following two
    expressions define the same whole-chain operator:
    \begin{align}
        X & =\sum_{t\in\iota}M_t\otimes V_t
        =\sum_{r\in I}M'_r\otimes W_r.
        \label{eq:interval_mpo_transfer_decomposition}
    \end{align}
    The two expressions use their respective canonical regional
    configuration identifications. An empty index set and an overlap of
    length zero are allowed.
    This is supporting interval-transport algebra for
    \cite[Proposition~5.7]{Arad2016RigorousRG}.
\end{lemma}
\begin{proof}\leanok
    In natural coordinates, restrict each old matrix to the interval
    $E\cap J$ by fixing its row and column configurations on $E\setminus J$.
    Extend the result to $J$ by the indicated fixed row and column
    configurations on $J\setminus E$. Interval restriction does not
    increase the open bond bound, and extension gives the bound
    $\max(1,D_t)$. Positive $q$ supplies a fixed configuration on the
    common interval when identifying these operations with
    (\ref{eq:interval_mpo_transfer_factor}). The endpoint inequalities
    give precisely the increasing coordinates of the overlap, including
    an overlap of length zero.

    The configuration dimensions are $q^{|A|}$ and $q^{|B|}$, giving
    (\ref{eq:interval_mpo_transfer_count}). Finally expand the transferred
    old-only and new-only coordinates in matrix units. Lemma~\ref{thm:regional_middle_exterior_transfer} then gives
    (\ref{eq:interval_mpo_transfer_decomposition}), with the prescribed
    middle factors and the exterior coefficients chosen afterwards.
\end{proof}


\begin{lemma}[A numbered collar family before every whole operator]
    \label{thm:common_collar_mpo_tensor_family}
    \lean{TNLean.MPS.AreaLaw.exists_common_collar_mpo_tensor_family}
    \leanok
    Let $q\ge1$ and $0\le p\le c\le d\le r\le L$ be integers.
    Put $E=[c,d)$ and $J=[p,r)$. Supply a family
    $W^0_1,\ldots,W^0_n$ on $\mathcal C(E)$, each with open matrix
    product bond bound $B$ in increasing site order. There is one
    numbered family $(W_j)_{j=1}^{n'}$ on $\mathcal C(J)$, chosen
    before every whole-chain operator and its exterior coefficients,
    such that
    \begin{align}
        n'&=nq^{2|J\setminus E|}.
    \end{align}
    Each $W_j$ has open matrix product bond bound $\max(1,B)$ in
    increasing site order.

    For a region $S$, let $\Gamma_S$ denote the canonical unitary
    separating exterior and retained configurations, in that order.
    Every whole-chain matrix $Z$ that admits a decomposition
    \begin{align}
        Z&=\Gamma_E^*\left(\sum_{j=1}^nM^0_j\otimes W^0_j\right)\Gamma_E
    \end{align}
    admits exterior matrices $M_j$ on $\mathcal C(J^c)$ satisfying
    \begin{align}
        Z&=\Gamma_J^*\left(\sum_{j=1}^{n'}M_j\otimes W_j\right)\Gamma_J,\notag\\
        \Psi_J Z\Psi_J^*&=\sum_{j=1}^{n'}W_j\otimes M_j,
    \end{align}
    where $\Psi_J$ is the same canonical configuration identification
    followed by tensor exchange. No Hermiticity or spectral assumption
    on $Z$ is required. In particular the family is unchanged when a
    later estimated energy center changes $Z$ or its old exterior
    coefficients.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:interval_mpo_common_family_transfer}
    Apply the prescribed changing-region construction to $E\subseteq J$.
    The configuration space on $E\setminus J$ consists of the one
    empty configuration; the remaining labels are pairs of configurations
    on $J\setminus E$.
    Choose a finite enumeration of this one family before supplying
    any exterior coefficients. The count is $nq^{2|J\setminus E|}$,
    and each member retains the preceding bond bound. Transport the
    existing matrix identity by the canonical tensor-coordinate
    isometry to obtain the middle-first identity for the same $Z$.
\end{proof}

\begin{lemma}[Collar count within the fixed soft-family parameter]
    \label{thm:precenter_collar_count}
    \lean{TNLean.MPS.AreaLaw.precenter_count_ge_collar_factor,
        TNLean.MPS.AreaLaw.common_collar_count_le_precenter_square}
    \leanok
    Let $q,k\ge1$ and let $m,\ell,n,c$ be nonnegative integers. Put
    \begin{align}
        D=(mk+1)(qk)^{36\ell+4\lfloor k/\ell\rfloor},
    \end{align}
    with natural-number division when $\ell=0$. Then $q^{8\ell}\le D$.
    If $n\le D$ and $c\le4\ell$, then $nq^{2c}\le D^2$.
    Consequently the enlarged common family may still use the same
    parameter $D$ in any estimate requiring family cardinality at most
    $D^2$; its suppression budget is unchanged.
\end{lemma}
\begin{proof}\leanok
    Since $q\le qk$, $qk\ge1$ and
    $8\ell\le36\ell+4\lfloor k/\ell\rfloor$, monotonicity of powers
    gives $q^{8\ell}\le(qk)^{36\ell+4\lfloor k/\ell\rfloor}\le D$.
    Hence $nq^{2c}\le Dq^{8\ell}\le D^2$.
\end{proof}

\begin{lemma}[The same filter through one common collar family]
    \label{lem:common_collar_filter_words}
    \lean{TNLean.MPS.AreaLaw.exists_common_collar_filter_family}
    \leanok
    Let $q\ge1$ and $0\le p\le c\le d\le r\le L$, and put
    $E=[c,d)$ and $J=[p,r)$. Supply $n_0$ matrices $W_i^0$ on
    $\mathcal C(E)$, each with open matrix product bond bound $B$.
    There is one family $(W_i)_{i=1}^n$ on $\mathcal C(J)$, chosen
    before every global operator and target, with
    \begin{align}
        n=n_0q^{2|J\setminus E|},
        \qquad \text{member bond bound }\max(1,B).
    \end{align}
    Suppose a supplied global operator $K_*$ has a decomposition through
    $W^0$ in the original configuration split. Let $T$ be any global
    subspace and let $\Delta\ge0$. Assume $K_*$ preserves $T,T^\perp$,
    has minimum gain one on $T$, and satisfies
    $\|K_*v\|^2\le\Delta\|v\|^2$ on $T^\perp$.

    With $\Psi_J$ placing the retained factor first, the same operator
    $K=\Psi_JK_*\Psi_J^{-1}$ has an actual decomposition
    \begin{align}
        K=\sum_{i=1}^n W_i\otimes M_i.
    \end{align}
    Its whole-chain matrix is unchanged, and its four operator bounds
    hold on $T_J=\Psi_JT$ and $T_J^\perp$. For every $a\ge0$ and
    every $\delta$-viable subspace $S$ for $T_J$, where $\delta<1$,
    \begin{align}
        S_a=\sum_{w\in\{1,\ldots,n\}^a}W_{w_1}\cdots W_{w_a}S
    \end{align}
    has dimension at most $n^a\dim S$ and is
    $\Delta^a/(1-\delta)^2$-viable for $T_J$.
    The original decomposition is an intermediate hypothesis of this
    algebraic lemma. No decomposition on $J$ is assumed, and no
    viability statement is transferred between distinct tensor splits.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:common_collar_mpo_tensor_family}
    Choose the collar family before the whole operator. Apply its
    matrix identity to the actual matrix of $K_*$, and conjugate that
    identity by $\Psi_J$. Unitary conjugation preserves the target,
    its orthogonal complement, and the two norm inequalities. The
    tensor decomposition then gives the ordered-word error estimate;
    its $n^a$ products give the dimension bound. This is supporting
    algebra for \cite[Proposition~5.7]{Arad2016RigorousRG}.
\end{proof}
