\subsection{Partial averaging and unrestricted exterior bonds}
\label{sec:peps_dependent_open_support}

Let $\mathcal V$ and $\mathcal E$ be finite sets of vertices and labelled
oriented bonds, with tail and head maps $t,h:\mathcal E\to\mathcal V$.
Each bond has its own finite virtual alphabet $X_e$, and each vertex its
own physical alphabet $Y_v$. Parallel bonds and self-bonds are allowed;
the two incidences of a self-bond remain distinct. Write
$v(e,-)=t(e)$, $v(e,+)=h(e)$,
$I_v=\{(e,\epsilon):v(e,\epsilon)=v\}$, and
$\mathcal X_v=\prod_{(e,\epsilon)\in I_v}X_e$.
An endpoint configuration $\beta$ specifies all $\beta_{e,\epsilon}$.
The bijection $R$ groups these endpoints by vertex,
$(R\beta)_v=\beta|_{I_v}$, while $S$ groups them into head--tail pairs,
$(S\beta)_e=(\beta_{e,+},\beta_{e,-})$.

The tensor $A_v(\eta,s)$ defines
$T_vx(s)=\sum_{\eta\in\mathcal X_v}A_v(\eta,s)x(\eta)$.
Its network contraction with matrices $B_e$ is
\begin{align}
    \mathcal N_A(B)(\sigma)
    =\sum_\beta\prod_e B_e(\beta_{e,+},\beta_{e,-})
    \prod_v A_v((R\beta)_v,\sigma_v).
    \label{eq:peps_open_network_notation}
\end{align}
For a cut $C\subseteq\mathcal E$, put
$\mathcal X_{\partial C}=\prod_{e\in C,\epsilon\in\{-,+\}}X_e$,
$r_C\beta=\beta|_{C\times\{-,+\}}$, and
$w_C(\beta)=\prod_{e\notin C}\delta_{\beta_{e,+},\beta_{e,-}}$.
The cut map and its full range are
\begin{align}
    \mathcal C_{A,C}(M)(\sigma)
                     & =\sum_\beta M(r_C\beta)w_C(\beta)
    \prod_v A_v((R\beta)_v,\sigma_v),
    \label{eq:peps_open_cut_notation}                      \\
    \mathcal S_{A,C} & =\operatorname{im}\mathcal C_{A,C}.
    \nonumber
\end{align}
The boundary $M:\mathcal X_{\partial C}\to\C$ is arbitrary.
Expanding it in coordinate vectors shows that $\mathcal S_{A,C}$ is
spanned by networks with matrix units on $C$ and identities elsewhere.

When the physical alphabets are $\mathcal X_v$, put $J=SR^{-1}$ and
$(\mathcal R\psi)(\gamma)=\psi(J^{-1}\gamma)$. Define
$b_B(\sigma)=\prod_eB_e((J\sigma)_e^+,(J\sigma)_e^-)$ and the bond slice
$S_{e,\tau}\psi(s)=(\mathcal R\psi)(\tau[e\leftarrow s])$.
Let $G$ be a finite group and let
$U_e:G\to\operatorname{GL}(\C^{X_e})$ be a representation on each bond,
matched at its two ends. Write
$F_e((a,b),g)=U_e(g)_{a,b}$, so
$\operatorname{im}F_e=\spn_{\C}\{U_e(g):g\in G\}$ in matrix coordinates.
The incident representation $\rho_v$ acts by
\begin{align}
    W_v(g)_{s,\eta}
    =\prod_{(e,+)\in I_v}U_e(g)_{s_{e,+},\eta_{e,+}}
    \prod_{(e,-)\in I_v}U_e(g^{-1})_{\eta_{e,-},s_{e,-}}.
    \label{eq:peps_open_incident_notation}
\end{align}
For any representations on these local spaces, use
$P_v=|G|^{-1}\sum_g\rho_v(g)$, $A_v^0(\eta,s)=(P_v)_{s,\eta}$,
and $\mathcal P=\bigotimes_vP_v$.

Products of finite-coordinate maps act in coefficient coordinates by
$(\bigotimes_vF_v)x(\sigma)=\sum_\eta\prod_vF_v(\sigma_v,\eta_v)x(\eta)$
and compose coordinatewise. In finite dimensions, $\psi\in\operatorname{im}(\bigotimes_vF_v)$
iff every one-coordinate slice at $v$ lies in $\operatorname{im}F_v$: choosing
$L_v$ with $F_vL_v|_{\operatorname{im}F_v}=\Id$, each single-coordinate map
$F_vL_v$ fixes $\psi$, so $(\bigotimes_vF_v)(\bigotimes_vL_v)\psi=\psi$.
By the same coordinatewise argument, $(\bigotimes_vM_v)\psi=\psi$ whenever
$M_v$ fixes every slice of $\psi$ at $v$.

These support and reconstruction statements provide the open-boundary
steps of \cite[Theorem~5.4]{Schuch2010PEPS}.

\begin{definition}[Incident actions selected by vertex endomorphisms]%
    \label{def:peps_partial_incident_representation}
    \lean{TNLean.PEPS.DependentBondNetwork.partialIncidentRepresentation,
        TNLean.PEPS.DependentBondNetwork.partialIncidentRepresentation_apply,
        TNLean.PEPS.DependentBondNetwork.partialIncidentRepresentation_of_eq_id,
        TNLean.PEPS.DependentBondNetwork.partialIncidentRepresentation_of_eq_one}
    \leanok
    For group endomorphisms $f_v:G\to G$, define
    $\rho_v^f(g)=\rho_v(f_v(g))$, acting by
    $\rho_v^f(g)x=W_v(f_v(g))x$. In particular, $\rho_v^{\Id}=\rho_v$ and
    $\rho_v^1(g)=I$.
\end{definition}

\begin{definition}[Partially averaged tensors]%
    \label{def:peps_partial_averaging_site}
    \lean{TNLean.PEPS.DependentBondNetwork.partialAveragingSite,
        TNLean.PEPS.DependentBondNetwork.localSiteMap_partialAveragingSite,
        TNLean.PEPS.DependentBondNetwork.partialAveragingSite_of_eq_id,
        TNLean.PEPS.DependentBondNetwork.localSiteMap_partialAveragingSite_of_eq_one,
        TNLean.PEPS.DependentBondNetwork.partialAveragingSite_of_eq_one,
        TNLean.PEPS.DependentBondNetwork.partialAveragingSite_apply}
    \leanok
    \uses{def:peps_partial_incident_representation}
    The partially averaged tensor $A_v^f$ and its local map $T_v=P_v^f$ are
    \begin{align}
        A_v^f(\eta,s)
              & =\frac{1}{|G|}\sum_{g\in G}W_v(f_v(g))_{s,\eta},
        \label{eq:peps_partial_average}                          \\
        P_v^f & =\frac{1}{|G|}\sum_{g\in G}\rho_v^f(g).
        \nonumber
    \end{align}
    For $f_v=\Id$, $A_v^f=A_v^0$ and $P_v^f=P_v$; for $f_v=1$, $P_v^f=I$ and
    $A_v^f(\eta,s)=\delta_{s,\eta}$.
\end{definition}

\begin{lemma}[Partial averages are injective on invariants]%
    \label{thm:peps_partial_averaging_injective}
    \lean{TNLean.PEPS.DependentBondNetwork.isGInjective_partialAveragingSite}
    \leanok
    \uses{def:peps_partial_averaging_site,
        def:peps_g_injective}
    At every vertex $v$, $P_v^f$ is $G$-injective for $\rho_v^f$.
\end{lemma}
\begin{proof}\leanok
    Right translation in (\ref{eq:peps_partial_average}) gives
    $P_v^f\rho_v^f(g)=P_v^f$, and $P_v^f|_{(\C^{\mathcal X_v})^{\rho_v^f}}=\Id$.
\end{proof}

\begin{lemma}[Coherent expansion of a partially averaged network]%
    \label{thm:peps_partial_network_expansion}
    \lean{TNLean.PEPS.DependentBondNetwork.network_partialAveragingSite,
        TNLean.PEPS.DependentBondNetwork.bondRegrouping_network_partialAveragingSite,
        TNLean.PEPS.DependentBondNetwork.network_partialAveragingSite_eq_sum_matrixBondProduct}
    \leanok
    \uses{def:peps_partial_averaging_site}
    For any matrix family $B_e$, put
    $B_e^f(q)=U_e(f_{h(e)}(q_{h(e)}))B_e
        U_e(f_{t(e)}(q_{t(e)})^{-1})$.
    With $n=|\mathcal V|$, one has
    \begin{align}
        \mathcal N_{A^f}(B)
         & =|G|^{-n}\sum_{q:\mathcal V\to G}b_{B^f(q)},
        \label{eq:peps_partial_network_sum}                         \\
        \mathcal N_{A^f}(B)(R\beta)
         & =|G|^{-n}\sum_q\prod_e
        B_e^f(q)_{\beta_{e,+},\beta_{e,-}},
        \nonumber                                                   \\
        (\mathcal R\mathcal N_{A^f}(B))(\gamma)
         & =|G|^{-n}\sum_q\prod_e B_e^f(q)_{\gamma_e^+,\gamma_e^-}.
        \nonumber
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_partial_averaging_site}
    Substitute (\ref{eq:peps_partial_average}) into
    (\ref{eq:peps_open_network_notation}), expand over
    $q\in G^{\mathcal V}$, and contract (\ref{eq:peps_open_incident_notation})
    along each $e\in\mathcal E$ to obtain $B_e^f(q)$; applying $\mathcal R$ and
    the definition of $b_B$ gives (\ref{eq:peps_partial_network_sum}).
\end{proof}

\begin{lemma}[Selected vertex gauges preserve the contraction]%
    \label{thm:peps_partial_vertex_gauge}
    \lean{TNLean.PEPS.DependentBondNetwork.network_partialAveragingSite_bondGauge,
        TNLean.PEPS.DependentBondNetwork.network_partialAveragingSite_vertexGauge}
    \leanok
    \uses{def:peps_partial_averaging_site}
    For every $q:\mathcal V\to G$ and every family $B_e$,
    $\mathcal N_{A^f}(B^f(q))=\mathcal N_{A^f}(B)$.
    In particular, for $B_e=U_e(p_e)$, replacing
    $p_e\mapsto f_{h(e)}(q_{h(e)})p_ef_{t(e)}(q_{t(e)})^{-1}$ preserves
    $\mathcal N_{A^f}(B)$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_partial_network_expansion}
    Reindexing (\ref{eq:peps_partial_network_sum}) by $r\mapsto rq$ and using
    $(B^f(q))_e^f(r)=B_e^f(rq)$ gives $\mathcal N_{A^f}(B^f(q))=\mathcal N_{A^f}(B)$;
    multiplicativity of $U_e$ specializes this to $B_e=U_e(p_e)$.
\end{proof}

\begin{lemma}[Uncut bonds retain representation support under partial averaging]%
    \label{thm:peps_partial_uncut_support}
    \lean{TNLean.PEPS.DependentBondNetwork.physicalBondSlice_network_partialAveragingSite_mem_range,
        TNLean.PEPS.DependentBondNetwork.physicalBondSlice_mem_range_of_mem_partialCutSpace}
    \leanok
    \uses{def:peps_partial_averaging_site}
    If $B_e=I$, then
    $S_{e,\tau}\mathcal N_{A^f}(B)\in\operatorname{im}F_e=\spn_{\C}\{U_e(g):g\in G\}$
    for every $\tau$. Consequently, if $e\notin C$ and
    $\psi\in\mathcal S_{A^f,C}$, then $S_{e,\tau}\psi\in\operatorname{im}F_e$
    for every $\tau$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_partial_network_expansion}
    When $B_e=I$, $B_e^f(q)=U_e(f_{h(e)}(q_{h(e)})f_{t(e)}(q_{t(e)})^{-1})\in\operatorname{im}F_e$,
    so each term of $S_{e,\tau}\mathcal N_{A^f}(B)$ in
    (\ref{eq:peps_partial_network_sum}) lies in $\operatorname{im}F_e$.
    Expanding $M$ in (\ref{eq:peps_open_cut_notation}) in coordinate vectors
    expresses $\psi\in\mathcal S_{A^f,C}$ as a linear combination of networks
    $\mathcal N_{A^f}(B)$ with $B_e=I$ for $e\notin C$.
\end{proof}

\begin{definition}[Bond columns with unrestricted exterior matrices]%
    \label{def:peps_open_bond_generators}
    \lean{TNLean.PEPS.DependentBondNetwork.OpenBondLabel,
        TNLean.PEPS.DependentBondNetwork.openBondGenerator,
        TNLean.PEPS.DependentBondNetwork.openBondMatrix}
    \leanok
    For an exterior cut $C\subseteq\mathcal E$, set
    $\Lambda_e=G\sqcup(X_e\times X_e)$ with labels $[g]$ and $[a,b]$, and define
    \begin{align}
        Q_e^C([g])   & =
        \begin{cases}0,&e\in C,\\U_e(g),&e\notin C,\end{cases}
        \label{eq:peps_open_bond_columns} \\
        Q_e^C([a,b]) & =
        \begin{cases}E_{a,b},&e\in C,\\U_e(1),&e\notin C.\end{cases}
        \nonumber
    \end{align}
    Let $\widetilde F_e^C:\C^{\Lambda_e}\to\C^{X_e\times X_e}$ have column
    $Q_e^C(p)$ at $p\in\Lambda_e$.
\end{definition}

\begin{lemma}[Interior representation columns and exterior full ranges]%
    \label{thm:peps_open_bond_ranges}
    \lean{TNLean.PEPS.DependentBondNetwork.bondMatrix_range_le_openBondMatrix_range,
        TNLean.PEPS.DependentBondNetwork.openBondMatrix_range_eq_top}
    \leanok
    \uses{def:peps_open_bond_generators}
    If $e\notin C$, then
    $\operatorname{im}F_e\subseteq\operatorname{im}\widetilde F_e^C$; if
    $e\in C$, then $\operatorname{im}\widetilde F_e^C=\C^{X_e\times X_e}$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_open_bond_generators}
    By (\ref{eq:peps_open_bond_columns}), extending $c\in\C^G$ by $0$ on
    $X_e\times X_e$ gives $\widetilde F_e^Cc=F_ec$ when $e\notin C$, while for
    $e\in C$ setting $c([g])=0$ and $c([a,b])=x_{a,b}$ gives
    $\widetilde F_e^Cc=\sum_{a,b}x_{a,b}E_{a,b}=x$.
\end{proof}

\begin{lemma}[A coherent expansion with a free exterior]%
    \label{thm:peps_open_bond_expansion}
    \lean{TNLean.PEPS.DependentBondNetwork.exists_openBondCoefficients_of_slices}
    \leanok
    \uses{def:peps_open_bond_generators}
    If $S_{e,\tau}\psi\in\operatorname{im}F_e$ for every $e\notin C$ and every
    $\tau$, then there exists $c:\prod_e\Lambda_e\to\C$ such that
    \begin{align}
        \psi=\sum_{p\in\prod_e\Lambda_e}
        c(p)\,b_{(Q_e^C(p_e))_e}.
        \label{eq:peps_open_bond_expansion}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_open_bond_ranges}
    By Lemma~\ref{thm:peps_open_bond_ranges}, every bond slice of
    $\mathcal R\psi$ lies in $\operatorname{im}\widetilde F_e^C$, so the
    product-range criterion gives $\mathcal R\psi=(\bigotimes_e\widetilde F_e^C)c$;
    applying $\mathcal R^{-1}$ yields (\ref{eq:peps_open_bond_expansion}).
\end{proof}

\subsection{Open cut intersections from removable internal labels}
\label{sec:peps_dependent_open_intersection}

\begin{lemma}[Identity insertions away from a cut]%
    \label{thm:peps_network_in_open_cut}
    \lean{TNLean.PEPS.DependentBondNetwork.network_mem_cutSpace_of_eq_one}
    \leanok
    If $B_e=I$ for all $e\notin C$, then $\mathcal N_A(B)\in\mathcal S_{A,C}$.
\end{lemma}
\begin{proof}\leanok
    For $M_B(\eta)=\prod_{e\in C}B_e(\eta_{e,+},\eta_{e,-})$,
    (\ref{eq:peps_open_network_notation}) and (\ref{eq:peps_open_cut_notation})
    give $\mathcal C_{A,C}(M_B)=\mathcal N_A(B)$.
\end{proof}

\begin{lemma}[Monotonicity under additional cuts]%
    \label{thm:peps_cut_space_mono}
    \lean{TNLean.PEPS.DependentBondNetwork.cutSpace_mono}
    \leanok
    If $C\subseteq C'$, then
    $\mathcal S_{A,C}\subseteq\mathcal S_{A,C'}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_network_in_open_cut}
    Each coordinate-basis column of $\mathcal C_{A,C}$ is a network
    $\mathcal N_A(B)$ with $B_e=I$ for $e\notin C$ (hence for $e\notin C'$), so
    it lies in $\mathcal S_{A,C'}$ by Lemma~\ref{thm:peps_network_in_open_cut}.
\end{proof}

\begin{lemma}[Physical images of cut ranges]%
    \label{thm:peps_cut_space_physical_image}
    \lean{TNLean.PEPS.DependentBondNetwork.map_cutSpace}
    \leanok
    For finite $Y_v$ and linear maps $F_v:\C^{Y_v}\to\C^{Z_v}$,
    $(\bigotimes_v F_v)(\mathcal S_{A,C})=\mathcal S_{FA,C}$.
\end{lemma}
\begin{proof}\leanok
    By (\ref{eq:peps_open_cut_notation}),
    $(\bigotimes_vF_v)\mathcal C_{A,C}(M)=\mathcal C_{FA,C}(M)$ for all $M$;
    taking images gives the claim.
\end{proof}

\begin{lemma}[Averaging bare bond products]%
    \label{thm:peps_average_bond_product}
    \lean{TNLean.PEPS.DependentBondNetwork.averagingProjector_matrixBondProduct}
    \leanok
    For representations $\rho_v$ on $\C^{\mathcal X_v}$ with averaging tensors
    $A^0$ and $\mathcal P=\bigotimes_v P_v$, every matrix family $B$ satisfies
    $\mathcal P b_B=\mathcal N_{A^0}(B)$.
\end{lemma}
\begin{proof}\leanok
    Since $b_B=\mathcal N_I(B)$ and $P_vI=A_v^0$, applying
    $\mathcal P=\bigotimes_vP_v$ to (\ref{eq:peps_open_network_notation}) gives
    $\mathcal P b_B=\mathcal N_{A^0}(B)$.
\end{proof}

\begin{lemma}[Removing internal labels after averaging]%
    \label{thm:peps_open_product_in_cut}
    \lean{TNLean.PEPS.DependentBondNetwork.averagingProjector_openBondProduct_mem_cutSpace}
    \leanok
    \uses{def:peps_partial_averaging_site,
        def:peps_open_bond_generators}
    Suppose that every $p:\mathcal E\to G$ admits $q:\mathcal V\to G$
    such that
    \begin{align}
        f_{h(e)}(q_{h(e)})p_e f_{t(e)}(q_{t(e)})^{-1}=1
        \quad(e\notin C).
        \label{eq:peps_open_remove_labels}
    \end{align}
    With $\mathcal P^f=\bigotimes_vP_v^f$, every
    $a\in\prod_e\Lambda_e$ satisfies
    $\mathcal P^f b_{(Q_e^C(a_e))_e}\in\mathcal S_{A^f,C}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_average_bond_product,
        thm:peps_partial_vertex_gauge,
        thm:peps_network_in_open_cut}
    Set $p_e=g$ if $a_e=[g]$ and $p_e=1$ if $a_e=[x,y]$, and choose $q$
    satisfying (\ref{eq:peps_open_remove_labels}). By
    Lemmas~\ref{thm:peps_average_bond_product}
    and~\ref{thm:peps_partial_vertex_gauge},
    $\mathcal P^f b_{(Q_e^C(a_e))_e}=\mathcal N_{A^f}(Q^C(a))=\mathcal N_{A^f}((Q^C(a))^f(q))$,
    whose interior bond matrices are
    $U_e(f_{h(e)}(q_{h(e)})p_ef_{t(e)}(q_{t(e)})^{-1})=I$ for $e\notin C$, so
    Lemma~\ref{thm:peps_network_in_open_cut} places it in $\mathcal S_{A^f,C}$.
\end{proof}

\begin{lemma}[Intersection of partially averaged open cut ranges]%
    \label{thm:peps_partial_open_cut_intersection}
    \lean{TNLean.PEPS.DependentBondNetwork.iInf_partialCutSpace_eq_of_remove_labels}
    \leanok
    \uses{def:peps_partial_averaging_site}
    Let $(C_i)_{i\in I}$ be a nonempty family of cuts with $C_0\subseteq C_i$
    for all $i\in I$ such that every $e\notin C_0$ satisfies $e\notin C_i$ for
    some $i\in I$, and suppose (\ref{eq:peps_open_remove_labels}) holds for
    $C=C_0$. Then
    \begin{align}
        \bigcap_{i\in I}\mathcal S_{A^f,C_i}
        =\mathcal S_{A^f,C_0}.
        \label{eq:peps_partial_open_intersection}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_partial_uncut_support,
        thm:peps_open_bond_expansion,
        thm:peps_open_product_in_cut,
        thm:peps_cut_space_mono}
    If $\psi\in\bigcap_{i\in I}\mathcal S_{A^f,C_i}$, then for each $e\notin C_0$
    choosing $i$ with $e\notin C_i$ gives $S_{e,\tau}\psi\in\operatorname{im}F_e$
    by Lemma~\ref{thm:peps_partial_uncut_support}, so
    Lemma~\ref{thm:peps_open_bond_expansion} yields
    $\psi=\sum_a c(a)b_{Q^{C_0}(a)}$. Since $I\ne\varnothing$ and
    $(P_v^f)^2=P_v^f$, $\mathcal P^f\psi=\psi$, and
    Lemma~\ref{thm:peps_open_product_in_cut} gives
    \begin{align}
        \psi=\sum_a c(a)\mathcal P^f b_{Q^{C_0}(a)}
        \in\mathcal S_{A^f,C_0}.
        \label{eq:peps_partial_projected_sum}
    \end{align}
    Conversely, $C_0\subseteq C_i$ implies
    $\mathcal S_{A^f,C_0}\subseteq\mathcal S_{A^f,C_i}$ by
    Lemma~\ref{thm:peps_cut_space_mono}.
\end{proof}

\begin{lemma}[Open cut intersection for invariant injective tensors]%
    \label{thm:peps_open_cut_intersection}
    \lean{TNLean.PEPS.DependentBondNetwork.iInf_cutSpace_eq_of_remove_labels}
    \leanok
    \uses{def:peps_partial_incident_representation,
        def:peps_g_injective}
    Suppose every $Y_v$ is finite and $T_v$ is $G$-injective for $\rho_v^f$.
    Under the hypotheses of Lemma~\ref{thm:peps_partial_open_cut_intersection}
    on $(C_i)_{i\in I}$ and $C_0$,
    \begin{align}
        \bigcap_{i\in I}\mathcal S_{A,C_i}=\mathcal S_{A,C_0}.
        \label{eq:peps_open_cut_intersection}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_g_injective_left_inverse,
        thm:peps_partial_open_cut_intersection,
        thm:peps_cut_space_physical_image}
    Set $\mathcal T=\bigotimes_vT_v$, choose $L_v$ with $L_vT_v=P_v^f$ by
    Lemma~\ref{thm:peps_g_injective_left_inverse}, and put
    $\mathcal L=\bigotimes_vL_v$. Since $L_vT_v=P_v^f$ and $T_vP_v^f=T_v$,
    Lemma~\ref{thm:peps_cut_space_physical_image} gives
    $\mathcal L(\mathcal S_{A,C})=\mathcal S_{A^f,C}$,
    $\mathcal T(\mathcal S_{A^f,C})=\mathcal S_{A,C}$, and
    $\mathcal T\mathcal L|_{\mathcal S_{A,C}}=\Id$. Applying
    (\ref{eq:peps_partial_open_intersection}) gives
    \begin{align}
        \bigcap_i\mathcal S_{A,C_i}
        =\mathcal T\left(\bigcap_i\mathcal S_{A^f,C_i}\right)
        =\mathcal T(\mathcal S_{A^f,C_0})
        =\mathcal S_{A,C_0}.
        \label{eq:peps_open_inverse_transport}
    \end{align}
\end{proof}

\subsection{Lifted regional contractions and physical support}
\label{sec:peps_lifted_cut_support}

The constructions in this subsection and the reconstruction below require
no group action. Retain finite incidence and virtual alphabets, but allow
arbitrary physical alphabets unless finiteness is explicitly assumed.
A region $\Omega\subseteq\mathcal V$ retains all physical coordinates
outside $\Omega$ as part of one joint boundary.

\begin{definition}[Outside physical coordinates]%
    \label{def:peps_outside_physical_config}
    \lean{TNLean.PEPS.DependentBondNetwork.OutsidePhysicalConfig,
        TNLean.PEPS.DependentBondNetwork.outsidePhysicalRestriction,
        TNLean.PEPS.DependentBondNetwork.outsidePhysicalRestriction_update}
    \leanok
    Put $Y_{\Omega^c}=\prod_{v\notin\Omega}Y_v$ and write
    $\sigma_{\Omega^c}=(\sigma_v)_{v\notin\Omega}$. For $v\in\Omega$,
    $(\sigma[v\leftarrow s])_{\Omega^c}=\sigma_{\Omega^c}$.
\end{definition}

\begin{definition}[Lifted regional cut map]%
    \label{def:peps_lifted_cut}
    \lean{TNLean.PEPS.DependentBondNetwork.liftedCutCoeff,
        TNLean.PEPS.DependentBondNetwork.liftedCutMap,
        TNLean.PEPS.DependentBondNetwork.liftedCutMap_apply,
        TNLean.PEPS.DependentBondNetwork.liftedCutSpace}
    \leanok
    \uses{def:peps_outside_physical_config}
    For $M:\mathcal X_{\partial C}\times Y_{\Omega^c}\to\C$, define the linear map
    \begin{align}
        \widehat\Gamma_{A,\Omega,C}(M)(\sigma)
        =\sum_\beta M(r_C\beta,\sigma_{\Omega^c})w_C(\beta)
        \prod_{v\in\Omega}A_v((R\beta)_v,\sigma_v)
        \label{eq:peps_lifted_cut}
    \end{align}
    with range $\widehat{\mathcal S}_{A,\Omega,C}=\operatorname{im}\widehat\Gamma_{A,\Omega,C}$.
\end{definition}

\begin{lemma}[Regional slices lie in local tensor ranges]%
    \label{thm:peps_lifted_cut_local_slice}
    \lean{TNLean.PEPS.DependentBondNetwork.liftedCutMap_slice_mem_range,
        TNLean.PEPS.DependentBondNetwork.slice_mem_range_of_mem_liftedCutSpace}
    \leanok
    \uses{def:peps_lifted_cut}
    If $v\in\Omega$ and $\psi\in\widehat{\mathcal S}_{A,\Omega,C}$, then every
    physical slice $s\mapsto\psi(\sigma[v\leftarrow s])$ lies in
    $\operatorname{im}T_v$; in particular this holds for
    $\psi=\widehat\Gamma_{A,\Omega,C}(M)$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_lifted_cut,
        def:peps_outside_physical_config}
    Since $(\sigma[v\leftarrow s])_{\Omega^c}=\sigma_{\Omega^c}$ by
    Definition~\ref{def:peps_outside_physical_config}, separating the factor at
    $v\in\Omega$ in (\ref{eq:peps_lifted_cut}) gives
    \begin{align}
        \psi(\sigma[v\leftarrow s])
         & =\sum_\beta a_\beta A_v((R\beta)_v,s)\in\operatorname{im}T_v,
        \label{eq:peps_lifted_local_slice}                               \\
        a_\beta
         & =M(r_C\beta,\sigma_{\Omega^c})w_C(\beta)
        \prod_{w\in\Omega\setminus\{v\}}
        A_w((R\beta)_w,\sigma_w).
        \nonumber
    \end{align}
\end{proof}

\begin{lemma}[Covering regions force product physical support]%
    \label{thm:peps_lifted_cover_support}
    \lean{TNLean.PEPS.DependentBondNetwork.iInf_liftedCutSpace_le_range_product}
    \leanok
    \uses{def:peps_lifted_cut}
    For finite $Y_v$, put $\mathcal T=\bigotimes_vT_v$ and
    $\mathcal H_A=\operatorname{im}\mathcal T$. If $(\Omega_i,C_i)_{i\in I}$
    satisfies $v\in\bigcup_{i\in I}\Omega_i$ or $\operatorname{im}T_v=\C^{Y_v}$
    for every $v\in\mathcal V$, then
    \begin{align}
        \bigcap_{i\in I}\widehat{\mathcal S}_{A,\Omega_i,C_i}
        \subseteq\mathcal H_A.
        \label{eq:peps_lifted_cover_support}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_lifted_cut_local_slice}
    By Lemma~\ref{thm:peps_lifted_cut_local_slice} (when $v\in\Omega_i$) or
    surjectivity of $T_v$, every slice at $v$ lies in $\operatorname{im}T_v$,
    so the product-range criterion yields (\ref{eq:peps_lifted_cover_support}).
\end{proof}

\begin{definition}[Outside virtual labels read from a cut]%
    \label{def:peps_outside_cut_config}
    \lean{TNLean.PEPS.DependentBondNetwork.outsideCutLocalConfig,
        TNLean.PEPS.DependentBondNetwork.outsideCutLocalConfig_restriction}
    \leanok
    A cut $C$ exposes $\Omega^c$ if $e\in C$ for every $(e,\epsilon)\in I_v$
    with $v\notin\Omega$. Then $\kappa\in\mathcal X_{\partial C}$ determines
    $o_v(\kappa)\in\mathcal X_v$ for $v\notin\Omega$ by
    $o_v(\kappa)(e,\epsilon)=\kappa_{e,\epsilon}$, satisfying
    $o_v(r_C\beta)=(R\beta)_v$.
\end{definition}

\begin{lemma}[Full cut ranges lie in the lifted range and local support]%
    \label{thm:peps_full_cut_in_lifted_support}
    \lean{TNLean.PEPS.DependentBondNetwork.cutSpace_le_liftedCutSpace,
        TNLean.PEPS.DependentBondNetwork.cutSpace_le_range_product}
    \leanok
    \uses{def:peps_lifted_cut,
        def:peps_outside_cut_config}
    If $C$ exposes $\Omega^c$, then
    $\mathcal S_{A,C}\subseteq\widehat{\mathcal S}_{A,\Omega,C}$.
    For every cut $C$,
    $\mathcal S_{A,C}\subseteq\operatorname{im}(\bigotimes_vT_v)$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_outside_cut_config,
        def:peps_lifted_cut}
    Setting $\widetilde M(\kappa,\tau)=M(\kappa)\prod_{v\notin\Omega}A_v(o_v(\kappa),\tau_v)$
    and using $o_v(r_C\beta)=(R\beta)_v$ in (\ref{eq:peps_lifted_cut}) gives
    $\widehat\Gamma_{A,\Omega,C}(\widetilde M)=\mathcal C_{A,C}(M)$.
    Setting $x(\eta)=M(r_CR^{-1}\eta)w_C(R^{-1}\eta)$ gives
    $(\bigotimes_vT_v)x=\mathcal C_{A,C}(M)$ by (\ref{eq:peps_open_cut_notation}).
\end{proof}

\subsection{Reconstructing full cuts from lifted vectors}
\label{sec:peps_lifted_cut_reconstruction}

\begin{definition}[Replacing the outside part of a cut configuration]%
    \label{def:peps_cut_replace_outside}
    \lean{TNLean.PEPS.DependentBondNetwork.cutReplaceOutside}
    \leanok
    For $\kappa\in\mathcal X_{\partial C}$ and
    $\xi\in\prod_{v\notin\Omega}\mathcal X_v$, define
    $\kappa[\Omega^c\leftarrow\xi]\in\mathcal X_{\partial C}$ by
    $(\kappa[\Omega^c\leftarrow\xi])_{e,\epsilon}=\kappa_{e,\epsilon}$ if
    $v(e,\epsilon)\in\Omega$ and $\xi_{v(e,\epsilon)}(e,\epsilon)$ if
    $v(e,\epsilon)\notin\Omega$.
\end{definition}

\begin{definition}[Replacing outside physical coordinates]%
    \label{def:peps_replace_outside_physical}
    \lean{TNLean.PEPS.DependentBondNetwork.replaceOutsidePhysical,
        TNLean.PEPS.DependentBondNetwork.outsidePhysicalRestriction_replaceOutsidePhysical,
        TNLean.PEPS.DependentBondNetwork.replaceOutsidePhysical_restriction}
    \leanok
    \uses{def:peps_outside_physical_config}
    For $\tau\in Y_{\Omega^c}$, define $\sigma[\Omega^c\leftarrow\tau]$ by
    $(\sigma[\Omega^c\leftarrow\tau])_v=\sigma_v$ for $v\in\Omega$ and
    $\tau_v$ for $v\notin\Omega$. Then
    $(\sigma[\Omega^c\leftarrow\tau])_{\Omega^c}=\tau$ and
    $\sigma[\Omega^c\leftarrow\sigma_{\Omega^c}]=\sigma$.
\end{definition}

\begin{definition}[Boundary reconstructed through outside local maps]%
    \label{def:peps_reconstructed_cut_boundary}
    \lean{TNLean.PEPS.DependentBondNetwork.reconstructedCutBoundary}
    \leanok
    \uses{def:peps_cut_replace_outside,
        def:peps_outside_cut_config,
        def:peps_lifted_cut}
    Suppose every $Y_v$ is finite and $C$ exposes $\Omega^c$. For
    $L_v:\C^{Y_v}\to\C^{\mathcal X_v}$ ($v\notin\Omega$) and a lifted boundary
    $M$, define
    \begin{align}
        K_{L,M}(\kappa)
        =\sum_{\xi\in\prod_{v\notin\Omega}\mathcal X_v}
        \sum_{\tau\in Y_{\Omega^c}}
        M(\kappa[\Omega^c\leftarrow\xi],\tau)
        \prod_{v\notin\Omega}L_v(o_v(\kappa),\tau_v).
        \label{eq:peps_reconstructed_boundary}
    \end{align}
\end{definition}

\begin{lemma}[Contraction of the reconstructed boundary]%
    \label{thm:peps_reconstructed_boundary_contraction}
    \lean{TNLean.PEPS.DependentBondNetwork.cutMap_reconstructedCutBoundary}
    \leanok
    \uses{def:peps_reconstructed_cut_boundary,
        def:peps_replace_outside_physical}
    Under the hypotheses of
    Definition~\ref{def:peps_reconstructed_cut_boundary}, put
    $\psi=\widehat\Gamma_{A,\Omega,C}(M)$ and
    $\psi_\sigma(\tau)=\psi(\sigma[\Omega^c\leftarrow\tau])$.
    Then
    \begin{align}
        \mathcal C_{A,C}(K_{L,M})(\sigma)
        =\left[\left(\bigotimes_{v\notin\Omega}T_vL_v\right)
            \psi_\sigma\right](\sigma_{\Omega^c}).
        \label{eq:peps_reconstructed_contraction}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_reconstructed_cut_boundary,
        def:peps_lifted_cut,
        def:peps_replace_outside_physical}
    Expand $\mathcal C_{A,C}(K_{L,M})(\sigma)$ via
    (\ref{eq:peps_open_cut_notation}) and (\ref{eq:peps_reconstructed_boundary})
    over $(\beta,\xi,\tau)$ and swap $((R\beta)_v)_{v\notin\Omega}\leftrightarrow\xi$.
    Since $C$ exposes $\Omega^c$, every $e\notin C$ has both endpoints in
    $\Omega$, so $w_C(\beta)$ is invariant and
    $(r_C\beta)[\Omega^c\leftarrow\xi]$ becomes $r_C\beta$. Summing the new
    outside virtual indices yields
    \begin{align}
        (T_vL_v)_{\sigma_v,\tau_v}
        =\sum_{\eta\in\mathcal X_v}A_v(\eta,\sigma_v)L_v(\eta,\tau_v),
        \label{eq:peps_outside_local_composition}
    \end{align}
    and interchanging the remaining sums gives (\ref{eq:peps_reconstructed_contraction}).
\end{proof}

\begin{lemma}[Full cut reconstruction inside product physical support]%
    \label{thm:peps_lifted_cut_reconstruction}
    \lean{TNLean.PEPS.DependentBondNetwork.cutSpace_eq_liftedCutSpace_inf_range_product,
        TNLean.PEPS.DependentBondNetwork.liftedCutSpace_inf_range_product_le_cutSpace}
    \leanok
    \uses{def:peps_lifted_cut,
        def:peps_outside_cut_config}
    Suppose every $Y_v$ is finite and $C$ exposes $\Omega^c$. With
    $\mathcal H_A=\operatorname{im}(\bigotimes_vT_v)$,
    \begin{align}
        \mathcal S_{A,C}
        =\widehat{\mathcal S}_{A,\Omega,C}\cap\mathcal H_A.
        \label{eq:peps_lifted_reconstruction}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_full_cut_in_lifted_support,
        thm:peps_reconstructed_boundary_contraction}
    Lemma~\ref{thm:peps_full_cut_in_lifted_support} gives
    $\mathcal S_{A,C}\subseteq\widehat{\mathcal S}_{A,\Omega,C}\cap\mathcal H_A$.
    Conversely, for $\psi=\widehat\Gamma_{A,\Omega,C}(M)\in\mathcal H_A$,
    choose $L_v:\C^{Y_v}\to\C^{\mathcal X_v}$ ($v\notin\Omega$) with
    $T_vL_v|_{\operatorname{im}T_v}=\Id$. Since $\psi\in\mathcal H_A$, every
    slice of $\psi_\sigma$ at $v\notin\Omega$ lies in $\operatorname{im}T_v$, so
    $(\bigotimes_{v\notin\Omega}T_vL_v)\psi_\sigma=\psi_\sigma$, and
    (\ref{eq:peps_reconstructed_contraction}) gives
    $\mathcal C_{A,C}(K_{L,M})(\sigma)=\psi_\sigma(\sigma_{\Omega^c})=\psi(\sigma)$.
\end{proof}

\subsection{Intersections of lifted regional ranges}
\label{sec:peps_lifted_cut_intersection}

\begin{lemma}[Covering lifted intersections equal full cut intersections]%
    \label{thm:peps_lifted_cut_intersection}
    \lean{TNLean.PEPS.DependentBondNetwork.iInf_liftedCutSpace_eq_iInf_cutSpace,
        TNLean.PEPS.DependentBondNetwork.liftedCutSpace_eq_cutSpace_of_cover_or_surjective,
        TNLean.PEPS.DependentBondNetwork.liftedCutSpace_inf_eq_cutSpace_inf}
    \leanok
    \uses{def:peps_lifted_cut,
        def:peps_outside_cut_config}
    Suppose every $Y_v$ is finite, each $C_i$ exposes $\Omega_i^c$, and every
    $v\in\mathcal V$ satisfies $v\in\bigcup_{i\in I}\Omega_i$ or
    $\operatorname{im}T_v=\C^{Y_v}$. Then
    \begin{align}
        \bigcap_{i\in I}\widehat{\mathcal S}_{A,\Omega_i,C_i}
        =\bigcap_{i\in I}\mathcal S_{A,C_i}.
        \label{eq:peps_lifted_intersection}
    \end{align}
    In particular, $\widehat{\mathcal S}_{A,\Omega,C}=\mathcal S_{A,C}$ if
    $T_v$ is surjective for all $v\notin\Omega$, and for two pairs
    $(\Omega,C),(\Xi,E)$ with $v\in\Omega\cup\Xi$ or $T_v$ surjective for all $v$,
    \begin{align}
        \widehat{\mathcal S}_{A,\Omega,C}
        \cap\widehat{\mathcal S}_{A,\Xi,E}
        =\mathcal S_{A,C}\cap\mathcal S_{A,E}.
        \label{eq:peps_two_lifted_intersection}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_lifted_cover_support,
        thm:peps_lifted_cut_reconstruction,
        thm:peps_full_cut_in_lifted_support}
    Combine (\ref{eq:peps_lifted_cover_support}) and
    (\ref{eq:peps_lifted_reconstruction}) with
    Lemma~\ref{thm:peps_full_cut_in_lifted_support}; the one- and two-region
    identities are the $|I|=1,2$ cases of (\ref{eq:peps_lifted_intersection}).
\end{proof}

\begin{lemma}[Canonical reduction of a lifted intersection]%
    \label{thm:peps_lifted_canonical_intersection}
    \lean{TNLean.PEPS.DependentBondNetwork.iInf_liftedCutSpace_eq_map_representationAveragingSite}
    \leanok
    \uses{def:peps_lifted_cut,
        def:peps_g_injective}
    Let $G$ be finite, suppose every $Y_v$ is finite, and let $T_v$ be
    $G$-injective for $\rho_v$ with averaging tensors $A^0$. For a nonempty
    family $(\Omega_i,C_i)_{i\in I}$ satisfying the hypotheses of
    Lemma~\ref{thm:peps_lifted_cut_intersection} and
    $\mathcal T=\bigotimes_vT_v$,
    \begin{align}
        \bigcap_{i\in I}\widehat{\mathcal S}_{A,\Omega_i,C_i}
        =\mathcal T\left(\bigcap_{i\in I}\mathcal S_{A^0,C_i}\right).
        \label{eq:peps_lifted_canonical_intersection}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_g_injective_left_inverse,
        thm:peps_lifted_cut_intersection}
    Choose $L_v$ with $L_vT_v=P_v$ by
    Lemma~\ref{thm:peps_g_injective_left_inverse} and set
    $\mathcal L=\bigotimes_vL_v$. Since $L_vT_v=P_v$ and $T_vP_v=T_v$,
    $\mathcal L\mathcal C_{A,C}(M)=\mathcal C_{A^0,C}(M)$ and
    $\mathcal T\mathcal C_{A^0,C}(M)=\mathcal C_{A,C}(M)$ for all $C,M$, so
    $\bigcap_i\mathcal S_{A,C_i}=\mathcal T(\bigcap_i\mathcal S_{A^0,C_i})$;
    combining with (\ref{eq:peps_lifted_intersection}) gives
    (\ref{eq:peps_lifted_canonical_intersection}).
\end{proof}

\begin{lemma}[A common inverse realizes canonical membership and recovery]%
    \label{thm:peps_lifted_common_inverse}
    \lean{TNLean.PEPS.DependentBondNetwork.exists_liftedCutSpace_common_localInverse}
    \leanok
    \uses{def:peps_lifted_cut,
        def:peps_g_injective}
    Under the hypotheses of
    Lemma~\ref{thm:peps_lifted_canonical_intersection}, there exists
    $L_v:\C^{Y_v}\to\C^{\mathcal X_v}$ with $\mathcal L=\bigotimes_vL_v$ such
    that for every cut $C$, boundary $M$, and
    $\psi\in\bigcap_i\widehat{\mathcal S}_{A,\Omega_i,C_i}$,
    \begin{align}
        \mathcal L\mathcal C_{A,C}(M) & =\mathcal C_{A^0,C}(M),
        \label{eq:peps_lifted_inverse_boundary}                           \\
        \mathcal L\psi                & \in\bigcap_i\mathcal S_{A^0,C_i},
        \label{eq:peps_lifted_inverse_membership}                         \\
        \mathcal T\mathcal L\psi      & =\psi.
        \nonumber
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_g_injective_left_inverse,
        thm:peps_lifted_cut_intersection}
    Choosing $L_v$ with $L_vT_v=P_v$ by
    Lemma~\ref{thm:peps_g_injective_left_inverse} gives
    (\ref{eq:peps_lifted_inverse_boundary}). For
    $\psi\in\bigcap_i\widehat{\mathcal S}_{A,\Omega_i,C_i}=\bigcap_i\mathcal S_{A,C_i}$
    by (\ref{eq:peps_lifted_intersection}), writing $\psi=\mathcal C_{A,C_i}(M_i)$
    and applying (\ref{eq:peps_lifted_inverse_boundary}) with $T_vP_v=T_v$ gives
    (\ref{eq:peps_lifted_inverse_membership}) and $\mathcal T\mathcal L\psi=\psi$.
\end{proof}
