\section{Encoded frames and small-patch rewrites}
\label{sec:peps_encoded_frames}
The statements below formalize Definition~6.1 and the first assertion of
Lemma~6.3 of \cite{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 11--97 and 182--252, the registers of a frame
with one register per site and per tag, together with the steps of the proof
of the second assertion listed below. The expansion of the branches into
allowed monomials on these registers is in
Section~\ref{sec:peps_small_rewrite_monomials}, and the second assertion with
its polynomial count is
Theorem~\ref{thm:peps_frame_small_rewrite_polynomial}.

Throughout, $\Lambda$ is a finite set of sites with positions in the plane,
each site carries a site register $\C^q$ with $q\ge1$, and
$\mathcal H_\Lambda=\bigotimes_{x\in\Lambda}\C^q$. For $D\subseteq\Lambda$ and
an operator $X$ on $\mathcal H_D$, write $X_D\otimes\id$ for the operator
acting as $X$ on the sites of $D$ and as the identity elsewhere. Operators
supported on disjoint sets of sites commute. A vector $\Omega$ is fixed, and
$\lVert\cdot\rVert$ denotes the Euclidean norm of vectors and the operator
norm of matrices, rectangular ones included.

\begin{definition}[Square cylinder data]
    \label{def:peps_frame_square_patch}
    \lean{TNLean.PEPS.EncodedFrame.squareSample,
        TNLean.PEPS.EncodedFrame.SquarePatch,
        TNLean.PEPS.EncodedFrame.SquarePatch.inner,
        TNLean.PEPS.EncodedFrame.SquarePatch.outer,
        TNLean.PEPS.EncodedFrame.SquarePatch.Tag,
        TNLean.PEPS.EncodedFrame.SquarePatch.cyl,
        TNLean.PEPS.EncodedFrame.SquarePatch.proj,
        TNLean.PEPS.EncodedFrame.SquarePatch.proj_of_outer_eq_empty}
    \leanok
    For a center $c\in\R^2$ and a scale $h>0$, the inner and outer squares
    $Q^-(c,h)$ and $Q^+(c,h)$ are the closed sup-norm balls of radii $h$ and
    $2h$; their physical samples are the sites whose positions lie in them.
    Square cylinder data at $(c,h)$ consist of radii
    $h\le r_1\le\cdots\le r_n\le 2h$, the samples $D_j$ of the squares of
    radius $r_j$, and unit vectors $v_{j\ell}\in\mathcal H_{D_j}$ for
    $\ell\le d_j$, such that the cylinder projectors
    $\bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id$ have mutually
    orthogonal ranges. Their sum is
    \begin{align}
        P_{c,h}=\sum_{j=1}^{n}\sum_{\ell=1}^{d_j}
        \bigl(\ket{v_{j\ell}}\bra{v_{j\ell}}\bigr)_{D_j}\otimes\id .
        \label{eq:peps_frame_patch_proj}
    \end{align}
    Every $D_j$ satisfies $Q^-(c,h)\cap\Lambda\subseteq D_j\subseteq Q^+(c,h)\cap\Lambda$. If $Q^+(c,h)\cap\Lambda=\emptyset$, then $P_{c,h}=\id$, whatever the
    cylinder data. In \texttt{05-frames.tex}, lines 26--48,
    with the cylinder data of \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS},
    the accuracy $\lVert(\id-P_{c,h})\Omega\rVert\le\varepsilon$ enters as
    a hypothesis where it is used, while the count of branches below takes a uniform bound on the number
    of cylinder terms of each patch rather than recording a polynomial bound on $\sum_jd_j$.
\end{definition}

\begin{definition}[Hole encoder]
    \label{def:peps_frame_hole_encoder}
    \lean{TNLean.PEPS.EncodedFrame.SquarePatch.branch,
        TNLean.PEPS.EncodedFrame.SquarePatch.encoder,
        TNLean.Lattice.Sheet.stack,
        TNLean.PEPS.EncodedFrame.SquarePatch.card_tag_of_outer_eq_empty,
        TNLean.PEPS.EncodedFrame.SquarePatch.encoder_submatrix_of_outer_eq_empty}
    \leanok
    \uses{def:peps_frame_square_patch}
    Let $\mathcal J_{c,h}$ be a tag register with orthonormal basis
    $\{\ket{j,\ell}\}$ and let $\ket0$ be a fixed one-site unit vector. The
    hole encoder is
    \begin{align}
        K_{c,h}=\sum_{j,\ell}\ket{j,\ell}\otimes
        \bigl[\bigl(\ket0^{\otimes D_j}\bra{v_{j\ell}}\bigr)_{D_j}
            \otimes\id\bigr]
        \colon\mathcal H_\Lambda\to\mathcal J_{c,h}\otimes\mathcal H_\Lambda .
        \label{eq:peps_frame_hole_encoder}
    \end{align}
    It keeps a site register at every site, inside the hole as well. If the
    outer sample is empty, $\dim\mathcal J_{c,h}=1$ and $K_{c,h}$
    is the identity encoding, as in \texttt{05-frames.tex}, lines 67--69.
    On a square grid of unit spacing with $c$ a site and $h=3/2$, the inner
    sample, one sample $D_j$ of radius $r_j=2$, and the outer sample are
    nested squares of sites, and $K_{c,h}$ maps the site registers to a tag
    together with the site registers.

\end{definition}

\begin{lemma}[Encoders are contractions]
    \label{lem:peps_frame_encoder_contraction}
    \lean{TNLean.PEPS.EncodedFrame.SquarePatch.proj_isStarProjection,
        TNLean.PEPS.EncodedFrame.SquarePatch.encoder_conjTranspose_mul_self,
        TNLean.PEPS.EncodedFrame.SquarePatch.norm_encoder_le_one}
    \leanok
    \uses{def:peps_frame_hole_encoder}
    As in \texttt{05-frames.tex}, equation (encoder-contraction),
    lines 62--64, the operator $P_{c,h}$ in~(\ref{eq:peps_frame_patch_proj}) is an orthogonal projection,
    $K_{c,h}^\dagger K_{c,h}=P_{c,h}$, and $\lVert K_{c,h}\rVert\le1$.
\end{lemma}

\begin{proof}\leanok
    If $Q^+(c,h)\cap\Lambda=\emptyset$, $K_{c,h}$ is the identity encoding and all
    three claims hold. Otherwise, each $(\ket{v_{j\ell}}\bra{v_{j\ell}})_{D_j}\otimes\id$ is Hermitian
    and idempotent since $\lVert v_{j\ell}\rVert=1$, and products of distinct cylinder projectors vanish by
    orthogonality, so $P_{c,h}^2=P_{c,h}=P_{c,h}^\dagger$. Orthonormality
    of $\{\ket{j,\ell}\}$ in~(\ref{eq:peps_frame_hole_encoder}) gives
    $K_{c,h}^\dagger K_{c,h}=\sum_{j,\ell}
        (\ket{v_{j\ell}}\braket{0^{\otimes D_j}}{0^{\otimes D_j}}\bra{v_{j\ell}})_{D_j}
        \otimes\id=P_{c,h}$, whence
    $\lVert K_{c,h}\rVert^2=\lVert K_{c,h}^\dagger K_{c,h}\rVert
        =\lVert P_{c,h}\rVert\le1$.
\end{proof}

\begin{definition}[Encoded frame]
    \label{def:peps_frame}
    \lean{TNLean.PEPS.EncodedFrame.Hole,
        TNLean.PEPS.EncodedFrame.Frame,
        TNLean.PEPS.EncodedFrame.Frame.raw,
        TNLean.PEPS.EncodedFrame.TagSpace,
        TNLean.PEPS.EncodedFrame.rawProd,
        TNLean.PEPS.EncodedFrame.frameEncoder,
        TNLean.PEPS.EncodedFrame.Frame.encoder,
        TNLean.PEPS.EncodedFrame.Frame.refVec}
    \leanok
    \uses{def:peps_frame_hole_encoder}
    Following \cite[Definition~6.1]{OpenAI2026PolynomialPEPS}
    (\texttt{05-frames.tex}, lines 71--82), an encoded frame $\mathcal F$ consists of a site ownership assignment on
    one sheet, a list of holes $(c_a,h_a)_{a=1}^r$ with square cylinder data and pairwise
    disjoint outer samples, and a specified party holding each hole's tag.
    The order of the list is the fixed ordering of the appended tags. The
    encoding $K_{\mathcal F}$ is the product of the hole encoders $K_{c_a,h_a}$, with the
    tags appended in that order, and the reference vector is
    $\Omega_{\mathcal F}=K_{\mathcal F}\Omega$; a frame with $r=0$ is an unencoded
    sheet. While the source requires the outer
    squares $Q^+(c_a,h_a)$ to be disjoint in $\R^2$, only the resulting disjointness of their
    physical samples is used.
\end{definition}

\begin{lemma}[Disjoint encodings commute]
    \label{lem:peps_frame_commute}
    \lean{TNLean.PEPS.EncodedFrame.rawProd_commute,
        TNLean.PEPS.EncodedFrame.rawProd_append,
        TNLean.PEPS.EncodedFrame.tagAppendEquiv}
    \leanok
    \uses{def:peps_frame}
    For two lists of holes with disjoint outer footprints, the site parts of
    their encodings commute on every pair $(s,t)$ of tag configurations. The tags of
    a concatenated list are the pairs $(s,t)$ of tags of the two parts, on which the site part of the concatenated encoding is the product of
    the site parts of the two encodings; hence the encodings commute under
    the natural identification of tag orderings
    (\texttt{05-frames.tex}, lines 84--85).
\end{lemma}

\begin{proof}\leanok
    The site part on a tag configuration is a product of operators supported
    in the outer footprints of the holes in the list, so the two products
    have disjoint supports and commute; the factorization on $(s,t)$ follows by
    induction on the first list.
\end{proof}

\begin{lemma}[Frame Gram identity]
    \label{lem:peps_frame_gram}
    \lean{TNLean.PEPS.EncodedFrame.sum_rawProd_conjTranspose_mul_self,
        TNLean.PEPS.EncodedFrame.frameEncoder_conjTranspose_mul_self,
        TNLean.PEPS.EncodedFrame.projProd_isStarProjection,
        TNLean.PEPS.EncodedFrame.norm_frameEncoder_le_one}
    \leanok
    \uses{lem:peps_frame_encoder_contraction,lem:peps_frame_commute}
    For a frame $\mathcal F$ with hole projectors $P_1,\dots,P_r$,
    $K_{\mathcal F}^\dagger K_{\mathcal F}=P_1\cdots P_r$, this product is an
    orthogonal projection, and $\lVert K_{\mathcal F}\rVert\le1$.
\end{lemma}

\begin{proof}\leanok
    Proceeding by induction on $r$, write the site part on the tags $(s,t)$ as
    $B_sR_t$, with $B_s$ from the first hole. By Lemma~\ref{lem:peps_frame_encoder_contraction} and Lemma~\ref{lem:peps_frame_commute},
    $\sum_s R_t^\dagger B_s^\dagger B_sR_t=R_t^\dagger P_1R_t=P_1 R_t^\dagger R_t$; summing over $t$
    then gives $P_1\cdots P_r$ by the induction hypothesis. Because $P_1,\dots,P_r$
    commute pairwise, $P_1\cdots P_r$ is an orthogonal projection and $\lVert K_{\mathcal F}\rVert\le1$.
\end{proof}

\begin{lemma}[Telescoping on a vector]
    \label{lem:peps_frame_telescoping}
    \lean{TNLean.Lattice.Sheet.norm_act_list_prod_sub_le,
        TNLean.PEPS.EncodedFrame.norm_act_projProd_sub_le}
    \leanok
    For contractions $A_1,\dots,A_s$ and every vector $\psi$,
    \begin{align}
        \lVert(A_1\cdots A_s-\id)\psi\rVert
        \le\sum_{j=1}^s\lVert(A_j-\id)\psi\rVert .
        \label{eq:peps_frame_telescoping}
    \end{align}
    In particular, if $\lVert(P_a-\id)\psi\rVert\le\varepsilon$ for $1\le a\le r$, then $\lVert(P_1\cdots P_r-\id)\psi\rVert\le r\varepsilon$ without assuming commutation
    (\texttt{05-frames.tex}, lines 237--248).
\end{lemma}

\begin{proof}\leanok
    Induction on $s$ using
    $(A_1R-\id)\psi=A_1(R-\id)\psi+(A_1-\id)\psi$ and $\lVert A_1\rVert\le1$.
\end{proof}

\begin{theorem}[Norm of the frame reference vector]
    \label{thm:peps_frame_norm}
    \lean{TNLean.PEPS.EncodedFrame.Frame.norm_refVec_eq,
        TNLean.PEPS.EncodedFrame.Frame.norm_refVec_le,
        TNLean.PEPS.EncodedFrame.Frame.one_sub_mul_le_norm_refVec}
    \leanok
    \uses{lem:peps_frame_gram,lem:peps_frame_telescoping}
    As in \texttt{05-frames.tex}, equation (frame-norm), lines 86--94, a frame $\mathcal F$ with $r$ holes satisfies $\lVert\Omega_{\mathcal F}\rVert
        =\lVert P_1\cdots P_r\Omega\rVert$. If $\lVert\Omega\rVert=1$ and
    $\lVert(\id-P_a)\Omega\rVert\le\varepsilon$ for $1\le a\le r$, then
    \begin{align}
        1-r\varepsilon\le\lVert\Omega_{\mathcal F}\rVert\le1 .
        \label{eq:peps_frame_norm}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_frame_gram}, $P_1\cdots P_r$ is an orthogonal projection and
    $\lVert\Omega_{\mathcal F}\rVert^2=\braket{\Omega}{P_1\cdots P_r\Omega}=\lVert P_1\cdots P_r\Omega\rVert^2$. Contractivity gives $\lVert\Omega_{\mathcal F}\rVert\le1$, and the reverse triangle inequality
    with~(\ref{eq:peps_frame_telescoping}) of Lemma~\ref{lem:peps_frame_telescoping} gives $1-r\varepsilon\le\lVert\Omega_{\mathcal F}\rVert$.
\end{proof}

\begin{definition}[Site registers]
    \label{def:lattice_site_registers}
    \lean{TNLean.Lattice.Sheet.siteRegs,
        TNLean.Lattice.Sheet.sites,
        TNLean.Lattice.Sheet.nodup_sites,
        TNLean.Lattice.Sheet.mem_sites,
        TNLean.Lattice.Sheet.sitesEquiv,
        TNLean.Lattice.Sheet.owner_siteRegs}
    \leanok
    \uses{def:peps_party_layout}
    For an owner $o$ of each site of a finite set $\Lambda$ and a list $\ell$ of
    sites, the site registers of $\ell$ are one register $\C^q$ for each entry
    of $\ell$, held by the owner of that site, in the order of $\ell$. A fixed
    list of the sites of $\Lambda$ contains every site exactly once, so taking
    the site at a position gives a bijection from its positions to $\Lambda$.
    If every site of a list has owner $p$, every site register of the list is
    held by $p$.
\end{definition}

\begin{proof}\leanok
    Immediate from the definitions.
\end{proof}

\begin{definition}[Registers of a frame]
    \label{def:peps_frame_registers}
    \lean{TNLean.PEPS.EncodedFrame.tagRegs,
        TNLean.PEPS.EncodedFrame.layoutRegs,
        TNLean.PEPS.EncodedFrame.Frame.regs,
        TNLean.PEPS.EncodedFrame.Frame.map_owner_regs}
    \leanok
    \uses{def:peps_frame,def:peps_party_layout,def:lattice_site_registers}
    The registers of a frame $\mathcal F$ are one tag register $\C^{\mathcal
            J_a}$ for each hole $a$, held by its tag owner and listed in
    the hole order, followed by one site register $\C^q$ for each
    $x\in\Lambda$, held by the site owner of $x$ and listed in a fixed site order. Since each $x\in\Lambda$ appears once, the site map is a bijection from site-register positions to $\Lambda$; if all sites in a list $\ell$ have site owner $p$, every site
    register of $\ell$ has owner $p$. Matching the
    sheet of \texttt{05-frames.tex}, lines 13--20, and Definition~6.1, lines 71--82, the tensor product of the registers of $\mathcal F$
    is its memory in the sense of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}.
\end{definition}

\begin{lemma}[Canonical coordinates of site registers]
    \label{lem:lattice_site_register_coordinates}
    \lean{TNLean.Lattice.Sheet.siteVec,
        TNLean.Lattice.Sheet.siteVec_append,
        TNLean.Lattice.Sheet.siteVec_congr,
        TNLean.Lattice.Sheet.siteIsoList,
        TNLean.Lattice.Sheet.siteIsoList_siteVec,
        TNLean.Lattice.Sheet.groupIso,
        TNLean.Lattice.Sheet.groupIso_siteVec}
    \leanok
    \uses{def:lattice_site_registers}
    For a list $\ell$ of sites, including lists with repetitions, write
    $\ket c_\ell$ for the product basis vector of a configuration $c$ on the
    site registers of $\ell$. If $c(x)=c'(x)$ for every site in $\ell$, then
    $\ket c_\ell=\ket{c'}_\ell$. If $\ell$ has length $n$, a unitary identifies
    its site registers with $\C^{[q]^n}$ and sends $\ket c_\ell$ to the basis
    vector of the configuration $k\mapsto c(\ell_k)$ read along the list. More
    generally, for a bijection $e$ from the positions of $\ell$ to a finite set $K$
    and a map $f:K\to\Lambda$ with $\ell_k=f(e(k))$ for every position $k$, the
    grouping along $e$ is a unitary onto $\C^{[q]^K}$ sending $\ket c_\ell$ to
    $\ket{c\circ f}$. Under the canonical unitary for
    concatenating register lists, $\ket c_{\ell_1\ell_2}$ is identified with
    $\ket c_{\ell_1}\otimes\ket c_{\ell_2}$; the empty list contributes the
    scalar one.
\end{lemma}

\begin{proof}\leanok
    By induction on the list of registers, identifying $\C^\alpha\otimes\C^\beta$
    with $\C^{\alpha\times\beta}$ at each step.
\end{proof}

\begin{lemma}[Canonical coordinates of the registers]
    \label{lem:peps_frame_registers_coordinates}
    \lean{TNLean.PEPS.EncodedFrame.tagVec,
        TNLean.PEPS.EncodedFrame.tagIso,
        TNLean.PEPS.EncodedFrame.tagIso_tagVec,
        TNLean.PEPS.EncodedFrame.layoutVec,
        TNLean.PEPS.EncodedFrame.layoutIso,
        TNLean.PEPS.EncodedFrame.layoutIso_layoutVec,
        TNLean.PEPS.EncodedFrame.layoutIso_symm_single,
        TNLean.PEPS.EncodedFrame.Frame.regIso}
    \leanok
    \uses{def:peps_frame_registers,lem:lattice_site_register_coordinates}
    The tensor product of the registers of a frame is identified, by a
    unitary, with $\C^{\mathcal J\times[q]^\Lambda}$, the space of the tags
    and the sheet on which the encoding $K_{\mathcal F}$ and its operators
    act: the product basis vector $\ket\tau\otimes\bigotimes_x\ket{c(x)}$ is
    sent to $\ket{\tau,c}$. More generally, the site registers of a list of
    distinct sites enumerating a set $A$ are identified with $\C^{[q]^A}$ in
    the same way. These are the canonical site and sheet coordinates of
    \texttt{05-frames.tex}, lines 94--96, in which vectors are compared while
    the owners of their registers are specified separately.
    For any list $\ell$ of sites, including lists with repetitions, write
    $\ket c_\ell$ for its product basis vector. If $c(x)=c'(x)$ for every
    site in $\ell$, then $\ket c_\ell=\ket{c'}_\ell$.
    Under the canonical unitary for concatenating register lists, the basis
    vector $\ket c_{\ell_1\ell_2}$ is identified with
    $\ket c_{\ell_1}\otimes\ket c_{\ell_2}$. The empty list contributes
    the scalar one.
\end{lemma}

\begin{proof}\leanok
    By induction on the list of registers, identifying $\C^\alpha\otimes
        \C^\beta$ with $\C^{\alpha\times\beta}$ at each step; product basis
    vectors are sent to standard basis vectors, and a list of distinct sites
    enumerating $A$ gives a bijection between its positions and $A$.
    Configurations agreeing on a list give the same basis vector at every
    position, hence the same product vector, even if sites repeat.
    The same induction splits the vector at any prescribed list boundary;
    its empty-list case is $1\otimes v\mapsto v$.
\end{proof}

\begin{definition}[Data and conditions of a small-patch rewrite]
    \label{def:peps_frame_rewrite_data}
    \lean{TNLean.PEPS.EncodedFrame.SmallPatchRewrite,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.oldFrame,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.newFrame,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.Conditions,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.AvoidsUntouched,
        TNLean.PEPS.EncodedFrame.Frame.innerHoles,
        TNLean.PEPS.EncodedFrame.Frame.outerHoles}
    \leanok
    \uses{def:peps_frame}
    Old and new one-sheet frames $\mathcal F_{\rm old},\mathcal F_{\rm new}$ share their untouched holes (with identical encodings and tag owners), alongside $r_{\rm old}$ affected old holes and $r_{\rm new}$ affected new holes, $m$ additional square patches
    $(c_i,u_i)$ with cylinder data, and a specified finite list of parties (entering the polynomial count of
    Theorem~\ref{thm:peps_frame_small_rewrite_polynomial}).
    Write $H^\mp(\mathcal F)$ for the unions of the physical samples of the
    inner and outer hole squares of $\mathcal F$. The four conditions of
    \cite[Lemma~6.3]{OpenAI2026PolynomialPEPS}
    (\texttt{05-frames.tex}, lines 193--207) are:
    \begin{enumerate}
        \item every patch outer sample $Q^+(c_i,u_i)\cap\Lambda$ and every affected old or new hole outer sample is disjoint from the outer footprints of the untouched holes;
        \item $\bigcup_{i=1}^m (Q^-(c_i,u_i)\cap\Lambda)$ contains every site whose
              site owner changes and the outer footprint of every
              affected old or new hole;
        \item in each $Q^+(c_i,u_i)\cap\Lambda$, the sites outside
              $H^-(\mathcal F_{\rm old})$ have at most two old owners, and those outside $H^-(\mathcal F_{\rm new})$ have at most two new
              owners;
        \item all old and new site owners on the patch and affected-hole outer footprints, and all affected tag owners, belong to
              the specified list.
    \end{enumerate}
    Patches may overlap, and affected old holes may overlap affected new
    holes. In the canonical coordinates below, the tags of the untouched
    holes come first in both frames; Theorem~\ref{thm:peps_frame_small_rewrite}
    also covers arbitrary tag orderings.
\end{definition}

\begin{definition}[Canonical rewrite]
    \label{def:peps_frame_rewrite}
    \lean{TNLean.PEPS.EncodedFrame.SmallPatchRewrite.patchProj,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.affectedRewrite,
        TNLean.PEPS.EncodedFrame.SmallPatchRewrite.layoutEquiv,

    \leanok
    \uses{def:peps_frame_rewrite_data}
    With $K_{\rm old}$ and $K_{\rm new}$ the encodings of the affected old
    and new holes alone and $P_1,\dots,P_m$ the patch projectors in a fixed
    order, define
    \begin{align}
        M=K_{\rm new}P_m\cdots P_1K_{\rm old}^\dagger ,
        \label{eq:peps_frame_rewrite_map}
    \end{align}
    extended by the identity on the tags of the untouched holes, from the register layout of $\mathcal F_{\rm old}$ to that of
    $\mathcal F_{\rm new}$ (\texttt{05-frames.tex}, lines 222--232).
\end{definition}

\begin{theorem}[Tensoring with an identity is a contraction]
    \label{thm:peps_frame_identity_tensor_norm}
    \lean{Matrix.l2_opNorm_kronecker_one_le,
        Matrix.l2_opNorm_one_kronecker_rect_le}
    \leanok
    For complex matrices $A$ and $B$ (not necessarily square) and identity
    matrices $I$ of any size, $\lVert A\otimes I\rVert\le\lVert A\rVert$ and
    $\lVert I\otimes B\rVert\le\lVert B\rVert$.
\end{theorem}

\begin{proof}\leanok
    For $x=\sum_k x_k\otimes e_k$,
    $\lVert(A\otimes I)x\rVert^2=\sum_k\lVert Ax_k\rVert^2
        \le\lVert A\rVert^2\lVert x\rVert^2$; the second bound follows by
    permuting the two tensor factors.
\end{proof}

\begin{lemma}[The rewrite is a contraction]
    \label{lem:peps_frame_rewrite_contraction}
    \lean{TNLean.PEPS.EncodedFrame.SmallPatchRewrite.norm_rewrite_le_one,
        Matrix.l2_opNorm_reindex_le}
    \leanok
    \uses{def:peps_frame_rewrite,lem:peps_frame_gram,thm:peps_frame_identity_tensor_norm}
    The canonical rewrite $M$ in~(\ref{eq:peps_frame_rewrite_map}) satisfies $\lVert M\rVert\le1$.
\end{lemma}

\begin{proof}\leanok
    By Lemma~\ref{lem:peps_frame_gram}, $\lVert K_{\rm new}\rVert,\lVert P_i\rVert,\lVert K_{\rm old}^\dagger\rVert\le1$, and by Theorem~\ref{thm:peps_frame_identity_tensor_norm} tensoring with $\id$ on the
    untouched tags and relabelling coordinates preserve $\lVert M\rVert\le1$.
\end{proof}

\begin{lemma}[Exact intertwining]
    \label{lem:peps_frame_rewrite_intertwining}
    \lean{TNLean.PEPS.EncodedFrame.SmallPatchRewrite.rewrite_mul_encoder}
    \leanok
    \uses{def:peps_frame_rewrite,lem:peps_frame_commute,lem:peps_frame_gram}
    Under condition~(1) of Definition~\ref{def:peps_frame_rewrite_data},
    $MK_{\mathcal F_{\rm old}}=K_{\mathcal F_{\rm new}}P_m\cdots P_1
        Q_1\cdots Q_{r_{\rm old}}$, where $Q_1,\dots,Q_{r_{\rm old}}$ are the
    projectors of the affected old holes.
\end{lemma}

\begin{proof}\leanok
    On untouched tags $u$, affected old tags $b$ and affected new tags $a$,
    let $U_u$, $A_b$ and $B_a$ be the site parts of the encodings. By
    condition~(1) and Lemma~\ref{lem:peps_frame_commute}, $U_u$ commutes with
    $A_b^\dagger$, $P_i$, $B_a$ and $Q_t$. By Lemma~\ref{lem:peps_frame_gram},
    $\sum_bA_b^\dagger U_uA_b=U_u\sum_bA_b^\dagger A_b
        =U_uQ_1\cdots Q_{r_{\rm old}}$, and commuting $U_u$ past $B_aP_m\cdots P_1$ yields the identity.
\end{proof}

\begin{theorem}[Small-patch rewrite: canonical contraction]
    \label{thm:peps_frame_small_rewrite}

\begin{lemma}[Rescaled branchwise approximation]
    \label{lem:peps_frame_rescale}
    \lean{TNLean.PEPS.EncodedFrame.norm_rescale_le,
        TNLean.PEPS.EncodedFrame.norm_rescale_sum_le_of_branches,
        TNLean.PEPS.EncodedFrame.truncationRank,

    \leanok
    Let $M=\sum_\beta M_\beta$ be a contraction with $N$ branches and
    $\lVert M_\beta-\widetilde M_\beta\rVert\le\eta$ for all $\beta$. If
    $N\eta\le\tfrac12\delta$, then
    $M_\delta=(1+\tfrac12\delta)^{-1}\sum_\beta\widetilde M_\beta$ satisfies
    $\lVert M_\delta\rVert\le1$ and $\lVert M_\delta-M\rVert\le\delta$. For any
    $N,g\ge0$ and $\delta>0$, $k=\lceil(2Ng/\delta)^2\rceil+1\ge1$ satisfies
    $Ng/\sqrt k\le\tfrac12\delta$ (\texttt{05-frames.tex}, lines 331--341).
\end{lemma}

\begin{proof}\leanok

    $\lVert\widetilde M\rVert\le1+\tfrac12\delta$ and $\lVert M_\delta\rVert\le1$. Writing
    $M_\delta-M=(1+\tfrac12\delta)^{-1}(\widetilde M-M)
        -\tfrac{\delta/2}{1+\delta/2}M$ gives $\lVert M_\delta-M\rVert\le\delta/(1+\tfrac12\delta)\le\delta$.
\end{proof}

\begin{lemma}[Chains of rank-one site operators as networks]
    \label{lem:peps_frame_chain_network}
    \lean{TNLean.PEPS.SiteChain.Step,
        TNLean.PEPS.SiteChain.chainOp,
        TNLean.PEPS.SiteChain.tensor,
        Matrix.ofFn_prod_apply,
        TNLean.PEPS.SiteChain.chainOp_apply}
    \leanok
    Let $L_j=\ket{k_j}\bra{b_j}_{S_j}\otimes\id$ ($0\le j<n$) be operators on $\mathcal H_\Lambda$, where each
    $k_j,b_j\in\mathcal H_{S_j}$ is a vector on the finite set $S_j\subseteq\Lambda$
    or the product zero vector $\ket0^{\otimes S_j}$. Form the network with distinct vertices
    $\bra{b_j}$ and $\ket{k_j}$ whose internal
    edges join the output of $\ket{k_j}$ at $x\in S_j$ to the input of the next bra
    $\bra{b_{j'}}$ ($j'<j$) with $x\in S_{j'}$, in the application order $L_{n-1},\dots,L_0$;
    the input of $\bra{b_j}$ at $x\notin\bigcup_{j'>j}S_{j'}$ is an open
    input leg, and the output of $\ket{k_j}$ at $x\notin\bigcup_{j'<j}S_{j'}$
    is an open output leg, except that legs of $\ket0^{\otimes S_j}$ are
    fixed to $0$. Then $\braket{\sigma}{L_0\cdots L_{n-1}|\tau}=0$ unless
    $\sigma$ and $\tau$ agree outside $\bigcup_j S_j$ and equal
    $0$ at the fixed legs, in which case $\braket{\sigma}{L_0\cdots L_{n-1}|\tau}$ equals the network contraction $Z$
    evaluated at the open legs of $(\sigma,\tau)$. No edge
    joins a vertex to itself (\texttt{05-frames.tex}, lines 274--281).
\end{lemma}

\begin{proof}\leanok
    Expanding $L_0\cdots L_{n-1}$ over intermediate site configurations,
    any path of nonzero weight preserves the configuration at each site across factors $L_j$ with $x\notin S_j$; thus nonzero paths are in bijection with labellings of the internal edges, and each path weight is the product of the vertex tensors.
\end{proof}

\begin{lemma}[Whole-group truncation of a chain]

    (\texttt{05-frames.tex}, lines 306--329).
\end{lemma}

\end{proof}

\subsection{Pair sources of the rewrite gates}

\end{lemma}

\begin{proof}\leanok
    When read operation by operation, each nonprivate gate is a single

\end{proof}

\begin{lemma}[A sum of monomials as one gate keeps its pair sources]
    \label{lem:peps_party_gate_source_bound}
    \lean{TNLean.Lattice.PartyCircuit.Word.sourceCount_selectedPart,
        TNLean.Lattice.PartyCircuit.PartyChain.sourceCount_selectedPart,
        TNLean.Lattice.PartyCircuit.Word.sourceCount_mapOwner_le,
        TNLean.Lattice.PartyCircuit.PartyChain.sourceCount_mapOwner_le,
        TNLean.PEPS.PairEffect.isSourceBounded_gateCircuit,
        TNLean.Lattice.PartyCircuit.PartyGate.exists_originalCircuit_isSourceBounded}
    \leanok
    \uses{lem:peps_circuit_source_bound_monomials}
    Let $\sum_\xi c_\xi M_\xi$ be a contraction expanded into allowed
    monomials that use only the parties of a nonempty finite set $S$ and act
    as $\id$ on all other registers. Then $\sum_\xi c_\xi M_\xi$ is the operator of a
    circuit with at most one gate, whose parties lie in $S$ as in
    Theorem~\ref{thm:peps_frame_small_rewrite_gate}, and which satisfies the pair-source bound with constant $r$ whenever every
    $M_\xi$ has at most $r$ pair sources.
\end{lemma}

\begin{proof}\leanok
    Restricting each $M_\xi$ to the registers of $S$ and relabelling owners into $S$ preserves or reduces the number of pair sources (turning a pair source with coincided relabelled owners into a local preparation), while the factor exchanges bringing the registers of $S$ to the front and back contain no nonprivate gates.
\end{proof}

    \lean{TNLean.PEPS.EncodedFrame.SmallPatchRewrite.exists_originalCircuit_refVec_isSourceBounded,
        TNLean.PEPS.Approximation.GuideFrameRewrite.PlaneConditions.exists_originalCircuit_isSourceBounded,

        TNLean.PEPS.Approximation.resize_step_isSourceBounded,

        thm:peps_guide_frame_rewrite_events,
        thm:peps_guide_frame_resize_treatment_gates,
        thm:peps_guide_frame_reading_gates}
    In the setting of Theorem~\ref{thm:peps_frame_small_rewrite_gate}, the
    circuit can be chosen to satisfy the pair-source bound with constant $2b$. The same holds for the
    guide-frame rewrites of Theorem~\ref{thm:peps_guide_frame_rewrite} and, in the dyadic protocol of
    \cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}, for the direct
    repainting, hole resize, and point treatment gates of
    Theorems~\ref{thm:peps_guide_frame_rewrite_events}, \ref{thm:peps_guide_frame_resize_treatment_gates},
    and~\ref{thm:peps_guide_frame_reading_gates}; hence every monomial of
    these gates has at most $2b$ pair sources and at most $2b$ pair effects.
\end{theorem}

        thm:peps_frame_small_rewrite_polynomial, lem:peps_party_gate_source_bound,
        thm:peps_original_finite_expansion_bound, def:peps_party_monomial,

        thm:peps_guide_frame_resize_treatment_gates,
        thm:peps_dyadic_sampled_ratio, lem:peps_dyadic_floor,
        thm:peps_guide_frame_reading_gates, def:peps_dyadic_block_guide}
    Each monomial in Theorem~\ref{thm:peps_frame_small_rewrite_polynomial} has at most
    $2m\le 2b$ normalized pair sources, and Lemma~\ref{lem:peps_party_gate_source_bound} preserves this bound on
    the gate, while the tag-register exchanges in Theorem~\ref{thm:peps_guide_frame_rewrite} add no nonprivate gates.
\end{proof}
