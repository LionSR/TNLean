\section{Restriction of concentric dyadic fans}
\label{sec:area_law_concentric_fans}

Endow $\mathbb R^2$ with the maximum norm. Fix a center
$c\in\mathbb R^2$ and an exponent $\ell\in\mathbb N$, and set $r=2^\ell/2$.
The smaller square has origin $o_{\rm S}=c-r(1,1)$, exponent $\ell$,
and cell index $(0,0)$; the larger square has origin
$o_{\rm L}=c-2r(1,1)$, exponent $\ell+1$, and cell index $(0,0)$.
Subdivide the same sides at their midpoints in both squares, and denote
corresponding fan triangles by $P_e^{\rm S}$ and $P_e^{\rm L}$.
This restriction gives the margin in the local star construction of
\cite[Section~11]{OpenAI2026AreaLaw}.

\begin{lemma}[Restriction of corresponding fan triangles]
    \label{thm:area_law_concentric_fan_restriction}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_succ_inter_closedBall}
    \leanok
    \uses{def:area_law_cell_fan}
    For every pair of corresponding elementary perimeter segments $e$,
    \begin{align}
        P_e^{\rm L}\cap\overline B_\infty(c,r) & =P_e^{\rm S}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:area_law_cell_fan,thm:area_law_cell_fan_base_radius}
    The homothety $H(y)=c+(y-c)/2$ fixes $c$ and maps the two perimeter
    vertices of $P_e^{\rm L}$ to those of $P_e^{\rm S}$, so
    $H(P_e^{\rm L})=P_e^{\rm S}$.

    Any $x\in P_e^{\rm L}$ has the form $x=c+\lambda(y-c)$ with
    $0\le\lambda\le1$ and $y$ on the base of $P_e^{\rm L}$. Since
    $\|y-c\|_\infty=2r$, we have $\|x-c\|_\infty=2r\lambda$. Thus
    $x\in\overline B_\infty(c,r)$ if and only if $\lambda\le1/2$, which is
    equivalent to $x=c+2\lambda(H(y)-c)\in P_e^{\rm S}$.
\end{proof}
