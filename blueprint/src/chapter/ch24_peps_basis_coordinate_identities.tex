% Auxiliary coordinate identities for polynomial PEPS compression.

\subsection{Physical configurations and tensor coordinates}

For $d\in\N$, write $[d]=\{0,\ldots,d-1\}$ (with $[0]=\varnothing$). With local dimensions $d_p\in\N$, the physical space $\bigotimes_{p\in P}\C^{d_p}$ of Definition~\ref{def:peps_family_physical_output} has the standard orthonormal basis indexed by the physical configurations $\mathcal X=\prod_{p\in P}[d_p]$.

\begin{definition}[Physical configurations of a region]
    \label{def:peps_basis_region_coordinates}
    \lean{ManyBody.RegionConfig}
    \leanok
    \uses{def:peps_family_physical_output}
    For finite $A\subseteq P$, the physical configurations of $A$ are $\mathcal X_A:=\prod_{p\in A}[d_p]$ (with $d_p=d$ in the constant-dimension case). They index the standard orthonormal basis of the physical space $\bigotimes_{p\in A}\C^{d_p}$ of $A$.
\end{definition}

\begin{lemma}[Physical dimensions of regions]
    \label{thm:peps_basis_physical_dimensions}
    \lean{QuantumCircuit.Party.card_regionConfig_fin,
        QuantumCircuit.Party.card_regionConfig_fin_le}
    \leanok
    For $d_p\in\N$ and finite $A\subseteq P$, $\#\prod_{p\in A}[d_p]=\prod_{p\in A}d_p$, so the physical space of $A$ has dimension $\prod_{p\in A}d_p$.
    In particular, this dimension is $d^{|A|}$ when $d_p=d$, and at most $D^{|A|}$ when $d_p\le D$ on $A$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_family_physical_output,def:peps_basis_region_coordinates}
    The basis of Definition~\ref{def:peps_basis_region_coordinates} has cardinality $\#\prod_{p\in A}[d_p]=\prod_{p\in A}d_p\le D^{|A|}$.
\end{proof}

\begin{definition}[Physical and regional coordinate isometries]
    \label{def:peps_basis_physical_readouts}
    \lean{QuantumCircuit.Party.physicalReadout,
        ManyBody.cutEquiv,
        QuantumCircuit.Party.regionalReadout}
    \leanok
    At each party $p$, let the output consist of the physical register $\C^{d_p}$ followed by a private register with basis labels $V_p$, so that the output configurations are $\prod_{p\in P}([d_p]\times V_p)$, and write $\mathcal V=\prod_{p\in P}V_p$. Define the coordinate isometry $R:\C^{\prod_p([d_p]\times V_p)}\to\C^{\mathcal X\times\mathcal V}$ and, for finite $A\subseteq P$ under the restriction bijection $\mathcal X\simeq\mathcal X_A\times\mathcal X_{P\setminus A}$ ($s\sqcup t\leftrightarrow(s,t)$), the regional coordinate isometry $R_A:\C^{\prod_p([d_p]\times V_p)}\to\C^{\mathcal X_A\times(\mathcal X_{P\setminus A}\times\mathcal V)}$ by
    \begin{align}
        R(v)(x,i)       & = v\bigl((x_p,i_p)_{p\in P}\bigr), &
        R_A(v)(s,(t,i)) & = R(v)(s\sqcup t,i).
    \end{align}
\end{definition}

\begin{lemma}[Regional coordinate formulas and contraction bounds]
    \label{thm:peps_basis_regional_readouts}
    \lean{QuantumCircuit.Party.norm_regionalReadout_le_one,
        QuantumCircuit.Party.toEuclideanLin_regionalReadout_apply}
    \leanok
    In the setting of Definition~\ref{def:peps_basis_physical_readouts}, for every $v\in\C^{\prod_p([d_p]\times V_p)}$, $s\in\mathcal X_A$, $t\in\mathcal X_{P\setminus A}$, and $i\in\mathcal V$, both general and constant local dimensions satisfy
    \begin{align}
        R_A(v)(s,(t,i))  & = R(v)(s\sqcup t,i), &
        \|R_A\|_{\rm op} & \le 1. \notag
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_basis_physical_readouts}
    Both properties follow directly from Definition~\ref{def:peps_basis_physical_readouts} since $R_A$ is a reindexed coordinate isometry.
\end{proof}

\subsubsection{Party tensor coordinates}

\begin{lemma}[Matrix entries of party tensor products]
    \label{thm:peps_basis_party_map_entries}
    \lean{Matrix.piKronecker_apply}
    \leanok
    For finite $P$, finite sets $I_p,J_p$, and matrices $A_p\in\C^{J_p\times I_p}$, the Kronecker product $A_P:=\bigotimes_{p\in P}A_p\in\C^{\prod_pJ_p\times\prod_pI_p}$ has matrix entries
    \begin{align}
        (A_P)_{x,y}
         & =\prod_{p\in P}(A_p)_{x_p,y_p}.
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    This is the definition of the Kronecker product of a finite family of matrices.
\end{proof}

These are the tensor-coordinate identities used in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 137--151, 233--267, 383--450 and
565--588.


\begin{lemma}[Matrix entries of party restrictions]
    \label{thm:peps_actual_party_word_entries}
    \lean{QuantumCircuit.Party.PartyCircuit.op_eq_piKronecker_partyOp}
    \leanok
    \uses{thm:peps_party_restriction}
    Let $P$ be finite, and let $W:\mathcal H_a\to\mathcal H_z$ be a source-free composition from a system with configurations $y\in\prod_{p\in P}I_p$ to a system with configurations $x\in\prod_{p\in P}J_p$. In configuration coordinates,
    \begin{align}
        W_{x,y}
         & =\prod_{p\in P}
        \bigl(W|_{\{p\}}\bigr)_{x_p,y_p}.
        \label{eq:peps_actual_party_word_entries}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_actual_singleton_factorization,
        thm:peps_basis_party_map_entries}
    By~(\ref{eq:peps_actual_singleton_factorization}), $W$ is the Kronecker product of its party restrictions $W|_{\{p\}}$, and Lemma~\ref{thm:peps_basis_party_map_entries} yields~(\ref{eq:peps_actual_party_word_entries}).
\end{proof}

\begin{lemma}[Mixed matrices of three source-free compositions]
    \label{thm:peps_actual_party_mixed_matrix}
    \lean{QuantumCircuit.Party.PartyCircuit.op_mul_op_mul_conjTranspose_op_eq_piKronecker}
    \leanok
    \uses{def:peps_dependent_physical_product,
        thm:peps_party_restriction}
    In the setting of Lemma~\ref{thm:peps_actual_party_word_entries}, let $W,V:\mathcal H_a\to\mathcal H_z$ and $U:\mathcal H_a\to\mathcal H_a$ be source-free compositions with matrices $M_W,M_V,M_U$ in configuration coordinates, and let $A_p,B_p,X_p$ be the matrices of $W|_{\{p\}},V|_{\{p\}},U|_{\{p\}}$. Then
    \begin{align}
        M_W M_U M_V^\dagger
         & =\bigotimes_{p\in P}(A_pX_pB_p^\dagger).
        \label{eq:peps_actual_party_mixed_matrix}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_actual_party_word_entries,thm:peps_party_product_conjugation}
    Combining~(\ref{eq:peps_actual_party_word_entries}) for $W,U,V$ with Lemma~\ref{thm:peps_party_product_conjugation} gives~(\ref{eq:peps_actual_party_mixed_matrix}).
\end{proof}

These are auxiliary identities for
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 409--434 and 565--588.
