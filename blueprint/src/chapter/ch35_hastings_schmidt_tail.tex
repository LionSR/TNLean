\begin{lemma}[The Schmidt-tail constraint]
    \label{thm:hastings_schmidt_tail_constraint}
    \lean{TNLean.MPS.AreaLaw.Hastings.one_sub_kyFanNorm_le_of_factorization}
    \leanok
    Let $\|\Omega\|=1$ on an $L$-site chain of local dimension $q\ge1$, and let
    $\rho$ be the marginal of $\Omega$ on $\{0,\dots,j-1\}$ with eigenvalues
    $p_1\ge p_2\ge\dots$. Suppose that for every $m$ with $j+m\le L$ there
    exist $O_B,O_L,O_R$ supported on $\{0,\dots,j+m-1\}$, $\{0,\dots,j-1\}$, and
    $\{j,\dots,L-1\}$ such that
    $\|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|\le C_1e^{-m/\xi'}$. If $k\ge1$
    and $\sum_{a=1}^kp_a\ge1/2$, then for every $m$ and every $n\ge kq^{2m}$,
    \begin{align}
        \sum_{a>n}p_a & \le 4C_1^2e^{-2m/\xi'}.
        \label{eq:hastings_schmidt_tail_constraint}
    \end{align}
    For $k\ge\exp(2S(\rho))/2$, Lemma~\ref{thm:hastings_leading_weight}
    supplies $\sum_{a=1}^kp_a\ge1/2$
    (Lemma~\ref{thm:hastings_schmidt_tail_constraint_entropy},
    \cite[lines 540--552]{Hastings2007AreaLaw}).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_bootstrap_scale, thm:hastings_mixture_tail,
        thm:hastings_cut_coordinates, lem:chain_sequential_truncation}
    Projecting $\{0,\dots,j-1\}$ onto the top $k$ eigenvectors of $\rho$ gives
    $\psi=E\Omega$ of Schmidt rank at most $k$ with
    $P=\langle\psi,\Omega\rangle=\|\psi\|^2=\sum_{a=1}^kp_a\ge1/2$. Applying
    Lemmas~\ref{thm:hastings_bootstrap_scale}
    and~\ref{thm:hastings_mixture_tail} to $\psi/\|\psi\|$ yields
    $\sum_{a>kq^{2m}}p_a\le(2C_1^2/P)e^{-2m/\xi'}\le4C_1^2e^{-2m/\xi'}$ whenever
    the bound is below $1$ (and trivially otherwise); monotonicity in $n$ gives
    (\ref{eq:hastings_schmidt_tail_constraint}).
\end{proof}

\begin{lemma}[Leading weight from the entropy]
    \label{thm:hastings_leading_weight}
    \lean{TNLean.MPS.AreaLaw.Hastings.one_half_le_kyFanNorm_of_exp_two_mul_vonNeumannEntropy_le}
    \leanok
    For a density matrix $\rho$ with eigenvalues $p_1\ge p_2\ge\cdots$, if
    $k\ge\exp(2S(\rho))/2$, then $\sum_{a=1}^kp_a\ge1/2$
    \cite[line 540]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    If $\sum_{a=1}^kp_a<1/2$, then $p_a\le p_k<1/(2k)$ for all $a>k$ and
    $\sum_{a>k}p_a>1/2$, so
    $S(\rho)\ge-\sum_{a>k}p_a\log p_a>\tfrac12\log(2k)$, contradicting
    $k\ge\exp(2S(\rho))/2$.
\end{proof}

\begin{lemma}[The Schmidt-tail constraint from the entropy]
    \label{thm:hastings_schmidt_tail_constraint_entropy}
    \lean{TNLean.MPS.AreaLaw.Hastings.one_sub_kyFanNorm_le_of_factorization_of_entropy}
    \leanok
    \uses{thm:hastings_schmidt_tail_constraint}
    In the setting of Lemma~\ref{thm:hastings_schmidt_tail_constraint}, if
    $k\ge\exp(2S(\rho))/2$, then (\ref{eq:hastings_schmidt_tail_constraint})
    holds for every $m$ and $n\ge kq^{2m}$
    \cite[lines 540--546]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_leading_weight}
    Combine Lemma~\ref{thm:hastings_leading_weight} ($k\ge1$ since
    $\exp(2S(\rho))/2>0$) with
    Lemma~\ref{thm:hastings_schmidt_tail_constraint}.
\end{proof}

\begin{lemma}[Polynomial decay of the Schmidt tail from geometric bounds]
    \label{thm:hastings_schmidt_tail_rpow}
    \lean{TNLean.MPS.AreaLaw.Hastings.schmidtTail_le_rpow_of_geometric}
    \leanok
    Let $D>1$, $\xi,k_0>0$, $c\ge0$, and $\tau\colon\mathbb N\to\mathbb R$. If
    for all $m,k\in\mathbb N$ with $k_0D^{2m}\le k$,
    \begin{align}
        \tau(k) & \le c\,e^{-2m/\xi},
        \label{eq:hastings_schmidt_tail_geometric}
    \end{align}
    then for every $k\ge k_0$,
    \begin{align}
        \tau(k) & \le c\,e^{2/\xi}\Bigl(\frac{k}{k_0}\Bigr)^{-1/(\xi\log D)}.
        \label{eq:hastings_schmidt_tail_rpow}
    \end{align}
    With $\tau(k)=\sum_{a>k}p_a$, $k_0=\exp(2S)/2$, $\xi=\xi'$, and $c=4C_1^2$,
    (\ref{eq:hastings_schmidt_tail_rpow}) is the Schmidt-tail estimate of
    \cite[lines 547--550]{Hastings2007AreaLaw} with $e^{2/\xi}$ and the negative
    exponent sign restored~\cite{gap:hastings07_schmidt_tail_constant}.
\end{lemma}
\begin{proof}\leanok
    Set $x=\log(k/k_0)/\log D\ge0$, $n=\lfloor x\rfloor$, and
    $m=\lfloor n/2\rfloor$. Then $D^{2m}\le D^x=k/k_0$ and $2m\ge n-1>x-2$, so
    (\ref{eq:hastings_schmidt_tail_geometric}) gives
    \begin{align}
        \tau(k) & \le c\,e^{-2m/\xi}\le c\,e^{2/\xi}e^{-x/\xi}
        =c\,e^{2/\xi}\Bigl(\frac{k}{k_0}\Bigr)^{-1/(\xi\log D)}.
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Matrix product state from the decay of the Schmidt tail]
    \label{thm:hastings_schmidt_tail_mps}
    \lean{TNLean.MPS.AreaLaw.Hastings.isOpenMPS_norm_sub_sq_le_of_schmidtTail_geometric}
    \leanok
    Let $\|\psi\|=1$ on an $L$-site chain, $D>1$, $\xi,k_0>0$, and $c\ge0$. If
    at every internal cut $j$ the tail $\sum_{a>n}p_{j,a}$ beyond the $n$
    largest left-marginal eigenvalues satisfies
    $\sum_{a>n}p_{j,a}\le ce^{-2m/\xi}$ whenever $k_0D^{2m}\le n$, then for
    every $k'\ge k_0$, sequential truncation onto the $k'$ leading eigenvectors
    at each cut produces an (unnormalized) open-boundary MPS $\Phi$ of bond
    dimension at most $k'$ with $\Phi=\psi$ for $L<2$ and, for $L\ge2$,
    \begin{align}
        \lVert\Phi-\psi\rVert^2
         & \le(L-1)\,c\,e^{2/\xi}\Bigl(\frac{k'}{k_0}\Bigr)^{-1/(\xi\log D)}
        \label{eq:hastings_schmidt_tail_mps}
    \end{align}
    \cite[lines 547--552]{Hastings2007AreaLaw},
    \cite{gap:hastings07_mps_error_measure,gap:hastings07_schmidt_tail_constant}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_sequential_truncation_eigen, thm:chain_sequential_truncation_sq,
        thm:hastings_schmidt_tail_rpow}
    By Lemma~\ref{thm:chain_sequential_truncation_eigen}
    and Theorem~\ref{thm:chain_sequential_truncation_sq},
    $\|\Phi-\psi\|^2\le\sum_{j=1}^{L-1}\sum_{a>k'}p_{j,a}$; apply
    (\ref{eq:hastings_schmidt_tail_rpow}) to each of the $L-1$ internal cuts.
\end{proof}

\begin{lemma}[Matrix product state from the entropy of the cuts]
    \label{thm:hastings_mps_from_entropy}
    \lean{TNLean.MPS.AreaLaw.Hastings.isOpenMPS_norm_sub_sq_le_of_factorization_of_entropy}
    \leanok
    \uses{thm:hastings_schmidt_tail_constraint}
    Let $\|\Omega\|=1$ on an $L$-site chain of local dimension $D\ge1$, and let
    $\xi'>0$. If the factorization of
    Lemma~\ref{thm:hastings_schmidt_tail_constraint} and
    $k_0\ge\exp(2S_j)/2$ hold at every internal cut $j$ ($0<j<L$), then for
    every $k'\ge k_0$ sequential truncation gives an open-boundary MPS $\Phi$ of
    bond dimension at most $k'$ with $\Phi=\Omega$ when $L<2$ or $D=1$, and for
    $L\ge2$ (reading the exponent as $0$ at $D=1$),
    \begin{align}
        \lVert\Phi-\Omega\rVert^2
         & \le(L-1)\,4C_1^2e^{2/\xi'}\Bigl(\frac{k'}{k_0}\Bigr)^{-1/(\xi'\log D)}
        \label{eq:hastings_mps_from_entropy}
    \end{align}
    \cite{gap:hastings07_mps_entropy_stage,gap:hastings07_mps_error_measure,
        gap:hastings07_schmidt_tail_constant}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_schmidt_tail_constraint_entropy, thm:hastings_schmidt_tail_mps,
        thm:chain_sequential_truncation_eigen, thm:chain_sequential_truncation_sq}
    For $L\ge2$ and $D\ge2$, combine
    Lemma~\ref{thm:hastings_schmidt_tail_constraint_entropy} ($k=k_0$) with
    Lemma~\ref{thm:hastings_schmidt_tail_mps} ($c=4C_1^2$). For $L<2$ or
    $D=1$, Lemma~\ref{thm:chain_sequential_truncation_eigen}
    and Theorem~\ref{thm:chain_sequential_truncation_sq} give $\Phi=\Omega$.
\end{proof}

\begin{lemma}[Matrix product state of the gapped ground state from Lemma~1]
    \label{thm:hastings_mps_ground_state}
    \lean{TNLean.MPS.AreaLaw.Hastings.isOpenMPS_norm_sub_sq_le_of_gappedGroundState}
    \leanok
    \uses{thm:hastings_area_law_from_factorization}
    Let $\Omega$ be a gapped ground vector of a finite-range chain Hamiltonian
    on $L$ sites of local dimension $D\ge1$, and assume the approximate
    factorization of Lemma~\ref{thm:hastings_area_law_from_factorization} with
    $\xi'>0$ and $C_1\ge1$. With $S_{\max}$ from
    Lemma~\ref{thm:hastings_area_law_from_factorization} and
    $k_0=\lceil\exp(2S_{\max})/2\rceil$, for every $k'\ge k_0$ sequential
    truncation gives an open-boundary MPS $\Phi$ of bond dimension at most $k'$
    satisfying (\ref{eq:hastings_mps_from_entropy})
    \cite{gap:hastings07_lemma1_decay_length,
        gap:hastings07_gaussian_average_exponent,
        gap:hastings07_mps_error_measure,
        gap:hastings07_schmidt_tail_constant}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_mps_from_entropy_lemma1, thm:hastings_area_law_from_factorization}
    By Lemma~\ref{thm:hastings_area_law_from_factorization}, $S_j\le S_{\max}$
    for every internal cut $j$, so $\exp(2S_j)/2\le k_0$; apply
    Lemma~\ref{thm:hastings_mps_from_entropy_lemma1}
    \cite{gap:hastings07_mps_entropy_stage}.
\end{proof}

\begin{theorem}[Matrix product state approximation of the gapped ground state]
    \label{thm:hastings_mps_gapped_chain}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_isOpenMPS_norm_sub_sq_le_of_isGappedGroundState_endpoint}
    \leanok
    \uses{thm:hastings_lemma1_endpoint, thm:hastings_area_law}
    For every range $R$ there exists $K=K(R)\ge0$ such that, for any term norm
    bound $J\ge0$, gap $\Delta E>0$, raised-velocity length $\zeta=\tilde\xi$ of
    Lemma~\ref{thm:hastings_lemma1_endpoint}, $\xi'=6\zeta$,
    $C_1=\max(K\zeta^3,1)$, and $k_0=\lceil\exp(2S_{\max})/2\rceil$ from
    Theorem~\ref{thm:hastings_area_law}, every gapped ground vector $\Omega$ on
    $L$ sites of local dimension $D\ge1$ and every $k'\ge k_0$ admit an
    open-boundary MPS $\Phi$ of bond dimension at most $k'$ satisfying
    (\ref{eq:hastings_mps_from_entropy}), with constants depending only on
    $(R,J/\Delta E,D)$
    \cite{gap:hastings07_lemma1_decay_length,
        gap:hastings07_gaussian_average_exponent,
        gap:hastings07_mps_error_measure,
        gap:hastings07_schmidt_tail_constant}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint, thm:hastings_mps_ground_state,
        thm:hastings_factorization_mono}
    Combine Lemma~\ref{thm:hastings_lemma1_endpoint} (with
    $C_1=\max(K\zeta^3,1)$ via Lemma~\ref{thm:hastings_factorization_mono})
    and Lemma~\ref{thm:hastings_mps_ground_state}
    \cite{gap:hastings07_mps_entropy_stage}.
\end{proof}
