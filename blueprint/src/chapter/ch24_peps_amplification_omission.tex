\section{Omitting distant events from the constraint product}
\label{sec:al_channel_word_omission}

The local constraint products lemma of \cite[Section~10]{OpenAI2026AreaLaw}
has two estimates. This section proves the second one: removing the events
whose anchors are far from the support of the observable costs little in
expectation, on the same realization of the clocks. As in the source, the
argument runs through the unital channels
$\mathcal E_i(B)=G_iBG_i+K_iBK_i$, with $G_i=(\mathbb 1-k_i)^{1/2}$ and
$K_i=k_i^{1/2}$, and passes to the products of the $G_i$ only at the end.

Throughout, $I$ and $J$ are finite sets, $q>0$, the physical space is
$(\C^q)^{\otimes I}$ and the spectator space $\C^J$ may have dimension zero.
For a word $w=(i_0,\ldots,i_{m-1})$ over a label set $F$, write
$\mathcal E_w=\mathcal E_{i_{m-1}}\circ\cdots\circ\mathcal E_{i_0}$, as in
Definition~\ref{def:al_channel_word}. For a predicate $P$ on $F$, let
$w|_P$ be the subword of letters satisfying $P$, in their original order and
with their multiplicities. All norms are operator norms.

\subsection{Telescoping the omitted events}

\begin{definition}[Omission defect]
    \label{def:al_channel_word_omission_defect}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.channelWordOmissionDefect}
    \leanok
    For an operator $B$ and a word $w$, put
    $D_P(w;B)=\|\mathcal E_w(B)-\mathcal E_{w|_P}(B)\|$.
\end{definition}

\begin{lemma}[Elementary bounds for the defect]
    \label{thm:al_channel_word_omission_bounds}
    \uses{def:al_channel_word_omission_defect}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_sub,
        TNLean.PEPS.AreaLaw.norm_spectatorRootChannelWord_sub_le,
        TNLean.PEPS.AreaLaw.channelWordOmissionDefect_bounds,
        TNLean.PEPS.AreaLaw.channelWordOmissionDefect_append_sub_le}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ for every $i\in F$. Every $\mathcal E_w$ is
    linear with $\|\mathcal E_w(A-R)\|\le\|A-R\|$ and
    $0\le D_P(w;B)\le2\|B\|$. For any $i\in F$, $D_P(wi;B)\le D_P(w;B)$ when
    $P(i)$ holds, and
    $D_P(wi;B)-D_P(w;B)\le\|(\mathcal E_i-\mathrm{id})(\mathcal E_{w|_P}(B))\|$
    when $\neg P(i)$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_channel_word_norm,thm:al_root_channel_contraction,
        thm:al_channel_word_append}
    Linearity holds letter by letter, and contractivity follows from
    Lemma~\ref{thm:al_channel_word_norm}. With $A=\mathcal E_w(B)$ and
    $R=\mathcal E_{w|_P}(B)$, if $P(i)$ holds then
    $D_P(wi;B)=\|\mathcal E_i(A-R)\|\le\|A-R\|=D_P(w;B)$; if $\neg P(i)$, then
    $D_P(wi;B)=\|\mathcal E_i(A)-R\|\le\|\mathcal E_i(A-R)\|+\|\mathcal E_i(R)-R\|
        \le D_P(w;B)+\|(\mathcal E_i-\mathrm{id})(R)\|$.
\end{proof}

\begin{lemma}[Finite prefix increments]
    \label{thm:al_poisson_prefix_increment}
    \lean{PoissonWord.sum_prefix_increment}
    \leanok
    Let $f$ be a real function on finite words over any alphabet. For a word
    $w=(i_0,\ldots,i_{m-1})$, let $w_{<j}=(i_0,\ldots,i_{j-1})$. Then
    \begin{align}
        \sum_{j=0}^{m-1}\bigl(f(w_{<j}i_j)-f(w_{<j})\bigr)
        =f(w)-f(\varnothing).
        \label{eq:al_poisson_prefix_increment}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $w_{<j}i_j=w_{<j+1}$, the sum telescopes from $w_{<0}=\varnothing$ to
    $w_{<m}=w$.
\end{proof}

\begin{lemma}[Omitted-prefix telescope]
    \label{thm:al_channel_word_omission_telescope}
    \uses{def:al_channel_word_omission_defect}
    \lean{TNLean.PEPS.AreaLaw.channelWordOmissionDefect_le_sum_prefix}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ for every $i\in F$. With $w_{<j}$ the prefix
    of length $j$,
    \begin{align}
        D_P(w;B)\le\sum_{0\le j<m,\ \neg P(i_j)}
        \|(\mathcal E_{i_j}-\mathrm{id})(\mathcal E_{w_{<j}|_P}(B))\|.
        \label{eq:al_channel_word_omission_telescope}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_channel_word_omission_bounds,thm:al_poisson_prefix_increment}
    Since $D_P(\varnothing;B)=0$, applying (\ref{eq:al_poisson_prefix_increment})
    to $f(u)=D_P(u;B)$ and bounding each increment by
    Lemma~\ref{thm:al_channel_word_omission_bounds} gives
    (\ref{eq:al_channel_word_omission_telescope}).
\end{proof}

For a finite alphabet $A$ and $t\ge0$, let $\nu_{A,t}$ be the law of the
word recorded by independent rate-one clocks on $[0,t]$, one for each label.

\begin{lemma}[Occupation at omitted letters]
    \label{thm:al_poisson_omitted_prefix_occupation}
    \lean{PoissonWord.integrable_sum_omitted_partition_prefix,
        PoissonWord.integral_sum_omitted_partition_prefix}
    \leanok
    Let $F$ be finite, let $P$ be a predicate on $F$, and put
    $F_P=\{i\in F:P(i)\}$. If $0\le H(u,i)\le M$ for words $u$ over $F_P$ and
    $i\in F$, then for $t\ge0$ the omitted-position sum is $\nu_{F,t}$-integrable
    and
    \begin{align}
        \int\sum_{0\le j<|w|,\ \neg P(i_j)}H(w_{<j}|_P,i_j)\,d\nu_{F,t}(w)
        =\int_0^t\sum_{i\in F,\ \neg P(i)}\int H(u,i)\,d\nu_{F_P,s}(u)\,ds.
        \label{eq:al_poisson_omitted_prefix_occupation}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Following
    \cite[``Real occupation at omitted labels'']{QICLean2026Blueprint},
    bounding the sum by $M|w|$ and using $\int|w|\,d\nu_{F,t}(w)<\infty$ gives
    integrability, while setting the kernel to zero on $F_P$ in the retained-prefix
    occupation identity gives (\ref{eq:al_poisson_omitted_prefix_occupation}).
\end{proof}

\begin{lemma}[Expected defect as a time integral]
    \label{thm:al_channel_word_expected_omission}
    \uses{def:al_channel_word_omission_defect}
    \lean{TNLean.PEPS.AreaLaw.integrable_and_integral_channelWordOmissionDefect_le}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ for every $i$ in a finite set $F$, and put
    $F_P=\{i\in F:P(i)\}$. Then $w\mapsto D_P(w;B)$ is $\nu_{F,t}$-integrable
    and
    \begin{align}
        \int D_P(w;B)\,d\nu_{F,t}(w)
        \le\int_0^t\sum_{i\in F,\ \neg P(i)}\int
        \|(\mathcal E_i-\mathrm{id})(\mathcal E_u(B))\|\,d\nu_{F_P,s}(u)\,ds.
        \label{eq:al_channel_word_expected_omission}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_channel_word_omission_bounds,thm:al_channel_word_omission_telescope,
        thm:al_channel_word_filter,thm:al_poisson_omitted_prefix_occupation}
    Since $0\le D_P(w;B)\le 2\|B\|$ by Lemma~\ref{thm:al_channel_word_omission_bounds},
    $D_P(\cdot;B)$ is integrable. Integrating (\ref{eq:al_channel_word_omission_telescope}),
    identifying $\mathcal E_{w_{<j}|_P}$ with the word over $F_P$ via
    Lemma~\ref{thm:al_channel_word_filter}, and applying
    (\ref{eq:al_poisson_omitted_prefix_occupation}) to
    $H(u,i)=\|(\mathcal E_i-\mathrm{id})(\mathcal E_u(B))\|\in[0,2\|B\|]$ yields
    (\ref{eq:al_channel_word_expected_omission}).
\end{proof}

\subsection{One omitted event}

\begin{lemma}[One-event error from the channel shells]
    \label{thm:al_channel_event_error}
    \uses{def:al_local_root_channel,def:qc_site_oscillation}
    \lean{TNLean.PEPS.AreaLaw.norm_localRootChannel_sub_self_le_exp_sum_siteOscillation,
        TNLean.PEPS.AreaLaw.norm_spectatorRootChannel_sub_self_le_exp_of_component_support}
    \leanok
    Let $0\le k\le\mathbb 1$, let $(K_l)_{l\ge0}$ be increasing site sets, let
    $C\ge0$, $c>0$, and $0<\alpha\le1$, and suppose
    $\|k-E_{K_l}k\|\le Ce^{-cl^\alpha}$ for $0\le l\le n$. With
    $A=2+8\sqrt Ce^{c/2}$, every $B$ satisfies
    \begin{align}
        \|\mathcal E_{K_n}(B)-B\|\le\sum_{l=0}^nAe^{-cl^\alpha/2}
        \sum_{z\in K_l}d_z(B).
        \label{eq:al_channel_event_error}
    \end{align}
    For $K_l=N_l(a)$, if $k$ is supported on the component of $a$ and $n$ is at
    least the maximum distance from $a$ in that component, then
    $\mathcal E_{K_n}(B)=\mathcal E(B)$ on the left-hand side of
    (\ref{eq:al_channel_event_error}).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_channel_shell_exp_bound,thm:qc_component_channel_stabilization}
    Summing the shell bounds of Lemma~\ref{thm:al_channel_shell_exp_bound}
    over $0\le l\le n$ gives (\ref{eq:al_channel_event_error}), and
    Lemma~\ref{thm:qc_component_channel_stabilization} gives
    $\mathcal E_{N_n(a)}=\mathcal E$.
\end{proof}

\begin{lemma}[Component support from the tails]
    \label{thm:al_component_support_of_tail}
    \lean{TNLean.PEPS.AreaLaw.mem_supportedOperators_reachable_of_forall_tail}
    \leanok
    If $\|k-E_{N_l(a)}k\|\le Ce^{-cl^\alpha}$ for every $l\ge0$, with
    $c,\alpha>0$, then $k$ is supported on the connected component of $a$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_graph_ball_stabilization}
    For $l$ beyond the maximum distance from $a$ in its component $C_a$,
    $N_l(a)=C_a$ by Lemma~\ref{thm:qc_graph_ball_stabilization}, while
    $Ce^{-cl^\alpha}\to 0$ as $l\to\infty$, so $k=E_{C_a}k$.
\end{proof}

\begin{lemma}[Anchors whose balls contain a site]
    \label{thm:al_graph_ball_anchor_count}
    \lean{TNLean.PEPS.AreaLaw.card_graphBall_anchors_le}
    \leanok
    Let labels on a graph be anchored at $a_i$ with $|\{i:a_i=x\}|\le\mu$ for
    every site $x$. For every site $y$ and $l\ge0$,
    $|\{i:y\in N_l(a_i)\}|\le\mu|N_l(y)|$.
\end{lemma}
\begin{proof}\leanok
    Since $y\in N_l(a_i)\iff a_i\in N_l(y)$ and each site in $N_l(y)$ anchors
    at most $\mu$ labels, $|\{i:y\in N_l(a_i)\}|\le\mu|N_l(y)|$.
\end{proof}

\begin{lemma}[Distance tail of the omitted anchors]
    \label{thm:al_omitted_anchor_tail}
    \lean{TNLean.PEPS.AreaLaw.sum_graph_singleton_exponentialDistanceProfile_le,
        TNLean.PEPS.AreaLaw.sum_graph_exponentialDistanceProfile_le,
        TNLean.PEPS.AreaLaw.exp_mul_graph_exponentialDistanceProfile_le_of_anchor_far,
        TNLean.PEPS.AreaLaw.sum_graphBall_site_function_le,
        TNLean.PEPS.AreaLaw.sum_graphBall_exponentialDistanceProfile_le,
        TNLean.PEPS.AreaLaw.exists_sum_omitted_graphBall_exponentialDistanceProfile_le}
    \leanok
    Let $K,\mu\ge0$, $b,\beta>0$, and $0<\alpha\le1$, and put
    $\gamma=\min(b,\beta)/2$. There exists $L\ge0$ such that on any finite graph
    with $|N_l(x)|\le K(l+1)^2$ and any labels anchored at $a_i$ with at most
    $\mu$ labels per site, setting $p(z)=e^{-\beta d(z,S)^\alpha}$ on the
    components meeting $S$ and $p(z)=0$ elsewhere, every set $E$ of labels with
    $d(a_i,S)>r\ge0$ for all $i\in E$ and every $N\ge0$ satisfy
    \begin{align}
        \sum_{i\in E}\sum_{l=0}^Ne^{-bl^\alpha}\sum_{z\in N_l(a_i)}p(z)
        \le L|S|e^{-\gamma r^\alpha}.
        \label{eq:al_omitted_anchor_tail}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_graph_ball_anchor_count}
    For $i\in E$ and $z\in N_l(a_i)$, $r^\alpha\le l^\alpha+d(z,S)^\alpha$ since
    $0<\alpha\le1$, so $e^{-bl^\alpha}p(z)\le e^{-\gamma r^\alpha}e^{-bl^\alpha/2}e^{-\beta d(z,S)^\alpha/2}$.
    By Lemma~\ref{thm:al_graph_ball_anchor_count}, each $z$ lies in at most
    $\mu K(l+1)^2$ balls $N_l(a_i)$, and
    $\sum_z e^{-\beta d(z,S)^\alpha/2}\le|S|K\sum_{n=0}^\infty(1+n)^2e^{-\beta n^\alpha/2}<\infty$,
    giving (\ref{eq:al_omitted_anchor_tail}).
\end{proof}

\subsection{The coupled omission estimates}

\begin{lemma}[Expected channel omission on a graph]
    \label{thm:al_graph_channel_omission}
    \uses{def:al_channel_word_omission_defect}
    \lean{TNLean.PEPS.AreaLaw.exists_integrable_and_integral_channelWordOmissionDefect_le}
    \leanok
    Let $C,K,\mu\ge0$, $c>0$, and $0<\alpha\le1$. There exist $D\ge0$ and
    $v,\gamma>0$ such that on any finite graph with $|N_l(x)|\le K(l+1)^2$, if
    $0\le k_i\le\mathbb 1$ are anchored at $a_i$ with at most $\mu$ labels per
    site, act on the component of $a_i$, and satisfy
    $\|k_i-E_{N_l(a_i)}k_i\|\le Ce^{-cl^\alpha}$ up to a cutoff beyond all
    component distances, then for any $B$ supported on $S$ and a spectator and
    any $P$ with $d(a_i,S)>r\ge0$ whenever $\neg P(i)$,
    \begin{align}
        \int D_P(w;B)\,d\nu_{F,t}(w)\le Dt|S|\|B\|e^{vt-\gamma r^\alpha}.
        \label{eq:al_graph_channel_omission}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_channel_word_expected_omission,thm:al_channel_event_error,
        thm:al_omitted_anchor_tail,thm:al_actual_channel_uniform_bounds,
        thm:al_channel_poisson_graph_growth,thm:al_retained_anchor_fibers}
    By Lemma~\ref{thm:al_channel_poisson_graph_growth} (applied via
    Lemma~\ref{thm:al_retained_anchor_fibers} and
    Lemma~\ref{thm:al_actual_channel_uniform_bounds} to $F_P$),
    $\int d_z(\mathcal E_u(B))\,d\nu_{F_P,s}(u)\le 2\|B\|e^{vs}p(z)$ with
    $\beta=c/(4\cdot2^\alpha)$. Substituting this into
    (\ref{eq:al_channel_event_error}), summing over $\{i\in F:\neg P(i)\}$ via
    (\ref{eq:al_omitted_anchor_tail}), and integrating over $s\in[0,t]$ in
    (\ref{eq:al_channel_word_expected_omission}) yields
    (\ref{eq:al_graph_channel_omission}).
\end{proof}

satisfy $|N_l(x)|\le2(l+1)^2$ by Lemma~\ref{thm:al_lattice_counts}.

\begin{lemma}[Omission of distant events from the constraint product]
    \label{thm:al_domain_constraint_omission}
    \uses{def:al_local_root_channel}
    \lean{TNLean.PEPS.AreaLaw.exists_domainConstraintWord_omission}
    \leanok
    Under the hypotheses of Lemma~\ref{thm:al_domain_channel_omission}, with
    the same constants $A,v,\gamma$, suppose $k_i\Omega=0$ for a unit vector
    $\Omega$. Fix an integer $r\ge0$ and retain the labels with
    $d_\Lambda(a_i,S)\le r$. On each realization, let $D$ and $D_{\rm ret}$ be
    the chronological products of $G_i$ over all events and over the retained
    events, respectively. For every unit spectator vector $\xi$,
    \begin{align}
        \mathbb E\|(D-D_{\rm ret})B(\Omega\otimes\xi)\|
        \le At|S|\|B\|e^{vt-\gamma r^\alpha}.
        \label{eq:al_domain_constraint_omission}
    \end{align}
    This establishes the omission estimate of \cite[Lemma~10.1]{OpenAI2026AreaLaw}
    without requiring $S\ne\varnothing$ or $r\ge1$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_domain_channel_omission,thm:al_channel_word_vector_difference}
    By Lemma~\ref{thm:al_channel_word_vector_difference}, $k_i\Omega=0$
    implies $\|(D-D_{\rm ret})B(\Omega\otimes\xi)\|\le D_P(w;B)$ pointwise for
    $P(i)\iff d_\Lambda(a_i,S)\le r$; integrating via
    (\ref{eq:al_domain_channel_omission}) gives
    (\ref{eq:al_domain_constraint_omission}).
\end{proof}

\subsection{The local constraint products lemma}

\begin{lemma}[Local constraint products]
    \label{thm:al_local_constraint_products}
    \lean{TNLean.PEPS.AreaLaw.exists_domainLocalConstraintProducts}
    \leanok
    Let $C,\mu\ge0$, $c>0$, and $0<\alpha\le1$. There exist $A\ge0$ and
    $\gamma,v>0$ with the following property. Let $0\le k_i\le\mathbb 1$ on a
    finite domain $\Lambda\subseteq\Z^2$ be anchored at $a_i$ with at most $\mu$
    labels per site and satisfy $\|k_i-E_{i,l}k_i\|\le Ce^{-cl^\alpha}$ for all
    $l\ge0$, where $E_{i,l}=E_{N_l(a_i)}$. Put $G_i=(\mathbb 1-k_i)^{1/2}$ and
    $G_{i,r}=(\mathbb 1-E_{i,r}k_i)^{1/2}$. Fix $S\subseteq\Lambda$, an integer
    $r\ge0$, and $t\ge0$, and run independent rate-one Poisson clocks on
    $[0,t]$. On each realization, in chronological order (later factors on the
    left), let $D$ be the product of all $G_i$, let $D_{\rm ret}$ retain the
    events with $d_\Lambda(a_i,S)\le r$, and let $L$ replace each $G_i$ in
    $D_{\rm ret}$ by $G_{i,r}$. Then every $L$ is a contraction supported on
    $N_{2r}(S)$,
    \begin{align}
        \mathbb E\|D_{\rm ret}-L\|\le At|S|(1+r)^2e^{-\gamma r^\alpha},
        \label{eq:al_local_constraint_products_trunc}
    \end{align}
    and if $k_i\Omega=0$ for a unit vector $\Omega$ and $B$ acts on $S$ and a
    finite spectator, then for every unit spectator vector $\xi$,
    \begin{align}
        \mathbb E\|(D-D_{\rm ret})B(\Omega\otimes\xi)\|
        \le At|S|\|B\|e^{vt-\gamma r^\alpha}.
        \label{eq:al_local_constraint_products_omit}
    \end{align}
    This proves \cite[Lemma~10.1]{OpenAI2026AreaLaw} on a single clock
    realization without requiring $S\ne\varnothing$ or $r\ge1$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:al_domain_product_truncation,thm:al_domain_constraint_omission}
    Combine Lemma~\ref{thm:al_domain_product_truncation} for
    (\ref{eq:al_local_constraint_products_trunc}) and the support of $L$ with
    Lemma~\ref{thm:al_domain_constraint_omission} for
    (\ref{eq:al_local_constraint_products_omit}), taking the maximum of the two
    prefactors $A$ and the minimum of the two decay rates $\gamma$.
\end{proof}
