\begin{theorem}[Injective parent uniqueness from vertex and edge coverage]%
    \label{thm:peps_injective_covering_parent_ground_space}
    \lean{TNLean.PEPS.globalVertexTensorMap_virtualBondProduct,
        TNLean.PEPS.regionParentGroundSpace_eq_span_of_isVertexInjective_of_cover,
        TNLean.PEPS.ker_regionParentHamiltonian_eq_span_of_isVertexInjective_of_cover,
        TNLean.PEPS.finrank_regionParentGroundSpace_eq_one_of_isInjective_of_cover}
    \leanok
    \uses{thm:peps_covering_parent_virtual_image,
        thm:peps_regional_virtual_conditions_bell,
        thm:peps_bell_product_physical_image,
        def:peps_injective_tensor}
    Let the site tensors $A_v$ on a finite graph $\Gamma=(V,E)$ be injective,
    and let $\mathcal R=(R_i)_i$ be a region family such that
    $\forall v\in V,\ \exists i,\ v\in R_i$ and
    $\forall e=\{u,v\}\in E,\ \exists i,\ \{u,v\}\subseteq R_i$.
    Then the common parent space and the kernel of any finite parent
    Hamiltonian $H=\sum_i h_{R_i}^{\mathrm{global}}$ with $h_{R_i}\ge 0$ and
    $\ker h_{R_i}=\mathcal G_{R_i}(A)$ satisfy
    \begin{align}
        \mathcal P_{\mathcal R}(A) &= \ker H = \spn\{\Psi(A)\}.
        \label{eq:peps_injective_covering_parent_span}
    \end{align}
    If $A$ is an injective PEPS tensor (Definition~\ref{def:peps_injective_tensor},
    with all bond dimensions $D_e>0$), then $\Psi(A)\ne 0$ and
    $\dim\mathcal P_{\mathcal R}(A)=1$;~(\ref{eq:peps_injective_covering_parent_span})
    also holds when some $D_e=0$, in which case $\spn\{\Psi(A)\}=\{0\}$.

    This is the covering-region form of the virtual-pair argument in
    \cite[Section IV.C.1]{Cirac2021Matrix}. Vertex coverage is explicit,
    including for isolated vertices. The physical reconstruction
    correction is recorded in
    \cite{gap:cpgsv21_injective_parent_reconstruction}.
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_covering_parent_virtual_image}, vertex coverage
    gives $\mathcal P_{\mathcal R}(A)=(\bigotimes_{v\in V}A_v)(\mathcal V_{\mathcal R})$,
    where $\mathcal V_{\mathcal R}$ is the common virtual regional subspace.
    By Theorem~\ref{thm:peps_regional_virtual_conditions_bell}, edge coverage
    reduces $\mathcal V_{\mathcal R}$ to $\spn\{\bigotimes_{e\in E}\ket{\Phi_e}\}$,
    and Theorem~\ref{thm:peps_bell_product_physical_image} gives
    $(\bigotimes_{v\in V}A_v)(\bigotimes_{e\in E}\ket{\Phi_e})=\Psi(A)$, proving~(\ref{eq:peps_injective_covering_parent_span})
    since $\Psi(A)\in\mathcal P_{\mathcal R}(A)$ and $h_{R_i}\ge 0\implies\ker H=\mathcal P_{\mathcal R}(A)$.
    When all $D_e>0$, $\bigotimes_{e\in E}\ket{\Phi_e}\ne 0$, so injectivity
    of $\bigotimes_{v\in V}A_v$ gives $\Psi(A)\ne 0$ and $\dim\mathcal P_{\mathcal R}(A)=1$.
\end{proof}
