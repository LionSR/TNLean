\section{Conditional two-family entropy estimate}

\begin{theorem}[Labelled conditional information bound]
    \label{thm:peps_labelled_conditional_entropy}
    \lean{TNLean.PEPS.AreaLaw.regionalEntropy_conditional_le_residual_add_sum}
    \leanok
    \uses{def:area_law_ground_entropy, def:area_law_two_family_partition}
    Let $\Omega$ be a normalized pure state on a finite domain, and let $G,B,C$
    be pairwise-disjoint regions. Suppose an ordered two-family partition of $G$
    has remainder $D$, pieces $X_i$ and family labels $f_i\in\{0,1\}$.
    Put $P_i=\bigcup_{j<i,\,f_j=f_i}X_j$.
    Assume $I_\Omega(X_i:C\cup B\cup P_i)\le\varepsilon_i$ when $f_i=0$ and
    $I_\Omega(X_i:(G\cup B)^c\cup P_i)\le\varepsilon_i$ when $f_i=1$.
    Then
    \begin{align}
        S_\Omega(G\cup B)+S_\Omega(C\cup B)-S_\Omega(B)-S_\Omega(G\cup C\cup B)
        \le 2S_\Omega(D)+\sum_i\varepsilon_i.
        \notag
    \end{align}
    This is the entropy estimate used before the geometric counting argument in
    \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}\leanok
    Filter the common order by the two labels. In each family, its preceding union
    is exactly $P_i$. Since $C$ is disjoint from $G\cup B$,
    $C\cup(G\cup B\cup C)^c=(G\cup B)^c$, so the second-family hypothesis has
    the required complementary-exterior form. Apply the generic pure-state
    conditional two-family bound and combine the two disjoint index sums.
    The finite-domain regional entropies agree with the corresponding generic
    finite-product reduced-state entropies.
\end{proof}
