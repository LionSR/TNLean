\section{Inhomogeneous short-range correlated states}
\label{sec:ldp_inhomogeneous}

The paragraph ``Inhomogeneous short-range correlated MPS''
of~\cite{Malz2024Preparation} extends the preparation to matrix product states
that are not translation invariant. After blocking, the positive parts of the
polar decompositions of the blocked tensors generate a state
$\ket{\phi_{\mathrm{pos}}}$~\cite[eq.~(18)]{Malz2024Preparation}; if it is close
to a product of nearest-neighbour pairs $\ket\Omega$, then preparing $\ket\Omega$
and applying the isometries of the polar decompositions prepares the state
with error $\varepsilon(\Omega,\phi_{\mathrm{pos}})$. This section proves the
exact identity behind the scheme and the depth and error of the prepared
state. Throughout, every bond of the ring has the same dimension $D$, every
blocked tensor is injective, and every block has at least $3D$ sites; the
source allows bond dimensions at most $D$, and assumes injectivity for its
polar decompositions in the footnote to the paragraph ``Approximation through
the fixed-point state''. The restrictions are recorded
in~\cite{gap:mswc24_inhomogeneous_scope}.

\begin{definition}[Blocked tensors of an inhomogeneous chain]
    \label{def:ldp_chain_block_tensor}
    \lean{MPSChainTensor.blockTensor,
        MPSChainTensor.blockTensor_decodeBlockEquiv_symm,
        MPSChainTensor.blockTensor_const, MPSChainTensor.eval_eq_prod_ofFn,
        MPSChainTensor.eval_const, MPSTensor.chainBlockTensor,
        MPSTensor.chainBlockTensor_const, MPSPreparation.chainState,
        MPSPreparation.chainState_apply}
    \leanok
    \uses{def:ldp_block_isometry_state}
    Let $A_1,\dots,A_N$ be tensors of physical dimension $d$ and bond dimension
    $D$, one on each site of a ring of $N$ sites, and let
    $\ket{\phi_N}=\sum_s\Tr(A_1^{s_1}\cdots A_N^{s_N})\ket s$ be their state.
    Cut the ring into $M\ge1$ consecutive blocks of lengths
    $\ell_1,\dots,\ell_M$ as in Definition~\ref{def:ldp_block_isometry_state}.
    The blocked tensor $B_k$ of block $k$ has physical dimension $d^{\ell_k}$ and
    the matrices
    $B_k^{s_{o_k+1}\cdots s_{o_k+\ell_k}}=A_{o_k+1}^{s_{o_k+1}}\cdots
        A_{o_k+\ell_k}^{s_{o_k+\ell_k}}$. When all tensors are equal to $A$, it is
    the $\ell_k$-site blocked tensor $\mathcal B_{\ell_k}$ of $A$.
\end{definition}

\begin{definition}[State of the positive parts and the block isometries]
    \label{def:ldp_chain_pos_state}
    \lean{MPSPreparation.chainPosState, MPSPreparation.chainPosState_apply,
        MPSTensor.blockIsoVector, MPSTensor.blockIsoVector_apply,
        MPSTensor.pairFamilyState, MPSTensor.pairFamilyState_const,
        MPSTensor.pairFamilyVector, MPSTensor.pairFamilyVector_apply,
        MPSTensor.chainBlockIsometryState,
        MPSTensor.chainBlockIsometryState_apply,
        MPSTensor.blockIsometryState_eq_chainBlockIsometryState}
    \leanok
    \uses{def:ldp_chain_block_tensor, thm:ldpolar_tensor_decomposition,
        def:prep_pair_product_state}
    Let $B_k=V_kP_k$ be the polar decompositions of the blocked tensors of
    Definition~\ref{def:ldp_chain_block_tensor}, with $P_k$ read as a tensor of
    physical dimension $D^2$ and bond dimension $D$ as in
    Theorem~\ref{thm:ldpolar_tensor_decomposition}. The state of the positive
    parts on the $M$ blocks is
    \begin{align}
        \ket{\phi_{\mathrm{pos}}}
         & =\sum_{\tau}\Tr\bigl(P_1^{\tau_1}\cdots P_M^{\tau_M}\bigr)
        \ket{\tau_1\cdots\tau_M},
        \label{eq:ldp_chain_pos_state}
    \end{align}
    the state of \cite[eq.~(18)]{Malz2024Preparation}. For vectors
    $\omega^1,\dots,\omega^M$ on $\C^D\otimes\C^D$ let
    $\ket\Omega=\bigotimes_{k=1}^M\ket{\omega^k}_{R_kL_{k+1}}$, the pair
    $\omega^k$ joining the right leg of block $k$ to the left leg of block
    $k+1$, cyclically; when all $\omega^k$ are equal to $\omega$ it is the pair
    product state of Definition~\ref{def:prep_pair_product_state}. The state
    $(\bigotimes_kV_k)\ket\Omega$ has the coefficient
    $\sum_c\prod_k\bra{s_k}V_k\ket{c_k}\,\Omega(c)$ at the configuration $s$
    read block by block; for equal tensors and equal pairs it is the state
    $\ket{\psi_{\ell,\omega}}$ of Definition~\ref{def:ldp_block_isometry_state}.
\end{definition}

\begin{theorem}[The state is the isometries applied to the positive parts]
    \label{thm:ldp_chain_state_eq_pos}
    \lean{MPSTensor.coeff_eq_mpvFamily_chainBlockTensor,
        MPSPreparation.chainState_eq_blockIsoVector}
    \leanok
    \uses{def:ldp_chain_pos_state}
    For every cutting of the ring into blocks,
    $\ket{\phi_N}=(\bigotimes_{k=1}^MV_k)\ket{\phi_{\mathrm{pos}}}$.
\end{theorem}
\begin{proof}
    \leanok
    The coefficient of $\ket{\phi_N}$ at $s$ is
    $\Tr(B_1^{s_1}\cdots B_M^{s_M})$ with $s_k$ the configuration of block $k$.
    Substituting $B_k^{s_k}=\sum_{\tau_k}\bra{s_k}V_k\ket{\tau_k}P_k^{\tau_k}$
    and expanding the product gives
    $\sum_\tau\prod_k\bra{s_k}V_k\ket{\tau_k}\Tr(P_1^{\tau_1}\cdots P_M^{\tau_M})$.
\end{proof}

\begin{theorem}[The isometries preserve overlaps]
    \label{thm:ldp_chain_block_iso_inner}
    \lean{MPSTensor.inner_blockIsoVector, MPSTensor.norm_blockIsoVector,
        MPSTensor.pairFamilyState_inner, MPSTensor.pairFamilyState_norm_sq,
        MPSTensor.norm_pairFamilyVector,
        MPSPreparation.blockIsoVector_smul, MPSPreparation.norm_chainState,
        MPSPreparation.inner_chainBlockIsometryState_normalize}
    \leanok
    \uses{def:ldp_chain_pos_state}
    Suppose every blocked tensor $B_k$ is injective. Then
    $\langle(\bigotimes_kV_k)x,(\bigotimes_kV_k)y\rangle=\langle x,y\rangle$ for
    all vectors $x,y$ on the $M$ blocks, and
    $\lVert\phi_N\rVert=\lVert\phi_{\mathrm{pos}}\rVert$. For products
    $\ket\Omega$ and $\ket{\Omega'}$ of pairs $\omega^k$ and $\omega'^k$,
    $\langle\Omega,\Omega'\rangle=\prod_k\langle\omega^k,\omega'^k\rangle$; in
    particular the norm of $\ket\Omega$ is $\prod_k\lVert\omega^k\rVert$.
    For all vectors $\omega^k$,
    \begin{align}
        \Bigl\langle\Bigl(\bigotimes_kV_k\Bigr)\Omega,
        \frac{\phi_N}{\lVert\phi_N\rVert}\Bigr\rangle
         & =\Bigl\langle\Omega,
        \frac{\phi_{\mathrm{pos}}}{\lVert\phi_{\mathrm{pos}}\rVert}\Bigr\rangle.
        \label{eq:ldp_chain_block_iso_inner}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_chain_state_eq_pos, thm:ldpolar_tensor_injective,
        def:prep_bond_regrouping}
    Injectivity makes each $V_k$ an isometry,
    Theorem~\ref{thm:ldpolar_tensor_injective}, so that
    $\sum_{s_k}\overline{(V_k)_{s_ka}}(V_k)_{s_kb}=\delta_{ab}$ for the
    configurations $s_k$ of block $k$ and all inputs $a,b$. Expanding the inner
    product over the configurations $s=(s_k)_k$ and applying this identity
    factor by factor gives
    \begin{align}
        \sum_s\overline{\Bigl(\Bigl(\bigotimes_kV_k\Bigr)x\Bigr)(s)}
        \Bigl(\Bigl(\bigotimes_kV_k\Bigr)y\Bigr)(s)
         & =\sum_{a,b}\overline{x(a)}\,y(b)
        \prod_k\sum_{s_k}\overline{(V_k)_{s_ka_k}}(V_k)_{s_kb_k}
        =\sum_a\overline{x(a)}\,y(a).
        \notag
    \end{align}
    The remaining identities follow from
    Theorem~\ref{thm:ldp_chain_state_eq_pos}. For the overlap of the pair
    products, expanding $\prod_k\langle\omega^k,\omega'^k\rangle$ as a sum
    over bond configurations $(p_k)_k$ of
    $\prod_k\overline{\omega^k_{p_k}}\,\omega'^k_{p_k}$ and reindexing along
    the bond regrouping of Definition~\ref{def:prep_bond_regrouping} gives
    $\sum_c\overline{\Omega(c)}\,\Omega'(c)$.
\end{proof}

\begin{definition}[Approximation of the positive parts by pairs]
    \label{def:ldp_pair_approximable}
    \lean{MPSPreparation.IsPairApproximable}
    \leanok
    \uses{def:ldp_chain_pos_state}
    For $\delta\in\R$, the chain is \emph{approximated by pairs with error
        $\delta$} on the given blocks if there are unit vectors
    $\omega^1,\dots,\omega^M$ on $\C^D\otimes\C^D$ with
    $1-\lvert\langle\Omega,\phi_{\mathrm{pos}}\rangle\rvert
        /\lVert\phi_{\mathrm{pos}}\rVert\le\delta$. The source defines a sequence of
    matrix product states to have finite correlation length if, after blocking
    $q=O(\log N)$ sites, this holds with an error tending to $0$ as
    $N\to\infty$~\cite[paragraph ``Inhomogeneous short-range correlated
        MPS'']{Malz2024Preparation}; the definition here is the condition for
    one ring.
\end{definition}

\begin{theorem}[Preparation for site-dependent tensors]
    \label{thm:ldp_chain_depth}
    \lean{MPSPreparation.exists_isPreparedInDepth_chainBlockIsometryState,
        MPSPreparation.chainBlockIsometryState_eq_mulVec,
        MPSPreparation.mul_self_le_pow_of_isInjective}
    \leanok
    \uses{def:ldp_chain_pos_state, def:ldp_local_circuit}
    There is $C$, depending only on $d$ and $D$, such that the following
    holds. Cut the ring of $N\ge1$ sites into $M\ge1$ blocks of lengths
    $3D\le\ell_k\le L$, let $A_1,\dots,A_N$ be tensors of bond dimension $D$
    whose blocked tensors are all injective, and let $\omega^k$ be unit
    vectors on $\C^D\otimes\C^D$. Then $(\bigotimes_kV_k)\ket\Omega$ is prepared
    in depth at most $CL$. An injective tensor of physical dimension $d^\ell$
    and bond dimension $D$ has $D^2\le d^\ell$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:ldp_unitary_implementing_isometry,
        thm:ldp_sequential_factorization,
        thm:ldp_bounded_unitary_two_site_gates}
    The circuit of the proof of
    Theorem~\ref{thm:ldp_approximating_state_depth} applies
    without change: the first layer prepares the pair $\omega^k$ on the window
    joining block $k$ to block $k+1$, one unitary per window, and on block $k$
    the unitary implementing $V_k$ is built from the sequential factorization
    of Theorem~\ref{thm:ldp_sequential_factorization}, which allows a
    different tensor on each site.
\end{proof}

\begin{theorem}[Preparation of inhomogeneous short-range correlated states]
    \label{thm:ldp_inhomogeneous}
    \lean{MPSPreparation.exists_isPreparedInDepth_inhomogeneous,
        MPSPreparation.exists_isPreparedInDepth_of_isPairApproximable}
    \leanok
    \uses{thm:ldp_chain_depth, thm:ldp_chain_block_iso_inner,
        def:ldp_pair_approximable}
    There is $C$, depending only on $d$ and $D$, such that the following
    holds. Cut the ring of $N\ge1$ sites into $M\ge1$ blocks of lengths
    $3D\le\ell_k\le L$, let $A_1,\dots,A_N$ be tensors of bond dimension $D$
    whose blocked tensors are all injective, and let $\omega^k$ be unit
    vectors on $\C^D\otimes\C^D$. Then
    $\ket\psi=(\bigotimes_kV_k)\ket\Omega$ is a unit vector prepared in depth
    at most $CL$, and
    \begin{align}
        1-\frac{\lvert\langle\psi,\phi_N\rangle\rvert}{\lVert\phi_N\rVert}
         & =1-\frac{\lvert\langle\Omega,\phi_{\mathrm{pos}}\rangle\rvert}
        {\lVert\phi_{\mathrm{pos}}\rVert},
        \label{eq:ldp_inhomogeneous_error}
    \end{align}
    that is, $\varepsilon(\psi,\phi_N)=\varepsilon(\Omega,\phi_{\mathrm{pos}})$
    for the normalized states. In particular, if the chain is approximated by
    pairs with error $\delta$ on the given blocks, some unit vector prepared in
    depth at most $CL$ has error at most $\delta$ against the normalized state
    $\ket{\phi_N}$. This is a scope-restricted variant, for one ring, of the
    claim of the paragraph ``Inhomogeneous short-range correlated MPS''
    of~\cite{Malz2024Preparation}, which states total depth
    $O(\log(N/\varepsilon))$ with error
    $\varepsilon(\Omega,\phi_{\mathrm{pos}})$. The common bond dimension, the
    injectivity of the blocked tensors and the lower bound $3D\le\ell_k$ are
    absent from that statement~\cite{gap:mswc24_inhomogeneous_scope}, and the
    unrestricted claim is not formalized. The bound $CL$ gives the stated
    depth when $L=O(\log(N/\varepsilon))$; the source's definition fixes
    blocks of length $q=O(\log N)$. The choice of $L$ in terms of $N$ and the
    error is not made here.
\end{theorem}
\begin{proof}
    \leanok
    The depth is Theorem~\ref{thm:ldp_chain_depth}. The norm of $\ket\psi$ is
    $\lVert\Omega\rVert=\prod_k\lVert\omega^k\rVert=1$ and the error identity is
    (\ref{eq:ldp_chain_block_iso_inner}), both by
    Theorem~\ref{thm:ldp_chain_block_iso_inner}.
\end{proof}
