\section{Union Closure for Normal PEPS}\label{sec:peps_normal_union}

A normal PEPS is injective only after blocking certain regions, so the injective
theorem applies once those regions are available. The closure property that
makes them available is that, for positive bond dimensions, the union of two
injective regions is again injective,
$\inj(A)\wedge\inj(B)\Longrightarrow\inj(A\cup B)$.
Writing $P_0=A\setminus B$, $P_1=A\cap B$, $P_2=B\setminus A$, and
$P_3=(A\cup B)^c$, a boundary operator $X$ attached to $P_3$ that is annihilated
by the contraction over $A\cup B$ descends through the contraction chain
\begin{align}
    \mathcal C_{\{0,1,2\}}(X)=0
    \Longrightarrow \mathcal C_{\{2\}}(X)=0
    \Longrightarrow \mathcal C_{\{1,2\}}(X)=0
    \Longrightarrow X=0,
    \notag
\end{align}
so the union contraction has trivial kernel. Iterating gives injectivity of
every finite union of injective regions.

\begin{definition}[Four-region decomposition for a union of regions]%
    \label{def:peps_regionUnionDecomposition}
    \lean{TNLean.PEPS.regionOnlyLeft, TNLean.PEPS.regionOverlap}
    \lean{TNLean.PEPS.regionOnlyRight, TNLean.PEPS.regionOutsideUnion}
    \lean{TNLean.PEPS.regionUnionPart}
    \leanok
    For $A,B\subseteq V$, define the indexed four-region family
    $(P_i)_{i=0}^3$ used in Lemma~\ref{lem:peps_injectiveRegion_union} by
    \begin{align}
        P_0 & = A\setminus B,        &
        P_1 & = A\cap B,             &
        P_2 & = B\setminus A,        &
        P_3 & = V\setminus(A\cup B).
        \label{eq:peps_four_region_parts}
    \end{align}
\end{definition}

\begin{definition}[Contraction maps for the union injectivity proof]%
    \label{def:peps_injectiveUnionContractionMaps}
    \lean{TNLean.PEPS.InjectiveUnionContractionMaps}
    \leanok
    For complex vector spaces $K,E_{012},E_2,E_{12}$, a contraction-map family
    for \cite[Section~4]{Molnar2018NormalPEPS} is a triple of linear maps
    $\mathcal C_{\{0,1,2\}}:K\to E_{012}$,
    $\mathcal C_{\{2\}}:K\to E_2$, and
    $\mathcal C_{\{1,2\}}:K\to E_{12}$.
\end{definition}

\begin{definition}[Contraction chain for the union injectivity proof]%
    \label{def:peps_injectiveUnionContractionChain}
    \lean{TNLean.PEPS.InjectiveUnionContractionChain}
    \leanok
    \uses{def:peps_injectiveUnionContractionMaps}
    A contraction chain is a contraction-map family
    $(\mathcal C_{\{0,1,2\}},\mathcal C_{\{2\}},\mathcal C_{\{1,2\}})$
    (Definition~\ref{def:peps_injectiveUnionContractionMaps}) satisfying
    \begin{align}
        \mathcal C_{\{0,1,2\}}(X)=0
        \Longrightarrow \mathcal C_{\{2\}}(X)=0
        \Longrightarrow \mathcal C_{\{1,2\}}(X)=0
        \Longrightarrow X=0
        \qquad (X\in K).
        \label{eq:peps_union_contraction_chain}
    \end{align}
\end{definition}

\begin{lemma}[Kernel criterion for the union contraction chain]%
    \label{lem:peps_injectiveRegion_union_contractionChain}
    \lean{TNLean.PEPS.InjectiveUnionContractionChain.union_contraction_ker_eq_bot}
    \lean{TNLean.PEPS.InjectiveUnionContractionChain.union_contraction_injective}
    \leanok
    \uses{def:peps_injectiveUnionContractionChain}
    For any contraction chain
    (Definition~\ref{def:peps_injectiveUnionContractionChain}) in
    \cite[Section~4]{Molnar2018NormalPEPS},
    $\ker \mathcal C_{\{0,1,2\}}=\{0\}$, so $\mathcal C_{\{0,1,2\}}$ is injective.
\end{lemma}
\begin{proof}\leanok
    Chaining the three implications in (\ref{eq:peps_union_contraction_chain})
    gives $\mathcal C_{\{0,1,2\}}(X)=0\Longrightarrow X=0$.
\end{proof}

    (Lemma~\ref{lem:peps_regionUnionDecomposition_identities}). Writing
    $\mathcal C_I$ for contraction over $\bigcup_{i\in I}P_i$, left-inverting
    the injective region $A=P_0\cup P_1$ gives
    $\mathcal C_{\{0,1,2\}}(X)=0\Longrightarrow\mathcal C_{\{2\}}(X)=0$;
    contracting with $P_1$ gives
    $\mathcal C_{\{2\}}(X)=0\Longrightarrow\mathcal C_{\{1,2\}}(X)=0$; and
    left-inverting $B=P_1\cup P_2$ gives
    $\mathcal C_{\{1,2\}}(X)=0\Longrightarrow X=0$. By
    Lemma~\ref{lem:peps_injectiveRegion_union_contractionChain},
    $\ker\mathcal C_{\{0,1,2\}}=\{0\}$, so $\inj(A\cup B)$.
\end{proof}

\begin{lemma}[Union of the edge-centred blocks]%
    \label{lem:peps_injectiveRegion_union_edgeBlocked}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_union}
    \leanok
    \uses{def:peps_regionInjectivityDataOf}
    For an edge $e$ and an edge-centred partition $V=R\sqcup B\sqcup C$ with
    $\inj(B)$, $\inj(C)$, and $D_f>0$ on every bond $f$, the union
    $B\cup C=V\setminus R$ satisfies $\inj(B\cup C)$.
\end{lemma}
\begin{proof}\leanok
    Suppose $\sum_\rho c_\rho T_{A,V\setminus R}^\rho=0$. Applying the left
    inverse of $\inj(B)$ strips the $B$-block, so the coupling of $c_\rho$ with
    $T_{A,C}$ vanishes for every $\partial B$ configuration. Applying $\inj(C)$
    then forces $c_\rho=0$ on every boundary configuration $\rho$ realized by a
    global virtual configuration; since $D_f>0$ makes the restriction to
    $\partial(V\setminus R)$ surjective, $c_\rho=0$ for all $\rho$.
\end{proof}

\begin{lemma}[Union of two disjoint injective regions]%
    \label{lem:peps_injectiveRegion_union_disjoint}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_union_disjoint}
    \leanok
    \uses{lem:peps_injectiveRegion_union_edgeBlocked}
    If $S,T\subseteq V$ satisfy $S\cap T=\varnothing$, $\inj(S)$, $\inj(T)$,
    and $D_f>0$ for all $f$, then $\inj(S\cup T)$.
\end{lemma}
\begin{proof}\leanok
    Since $S\cap T=\varnothing$, setting $R=(S\cup T)^c$, $B=S$, and $C=T$
    forms a three-part partition $V=R\sqcup B\sqcup C$, and the two-step
    inversion proof of Lemma~\ref{lem:peps_injectiveRegion_union_edgeBlocked}
    gives $\inj(S\cup T)$.
\end{proof}

\begin{lemma}[Union of two injective regions with positive bonds]%
    \label{lem:peps_injectiveRegion_union_pos}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_union_overlap}
    \leanok
    \uses{lem:peps_injectiveRegion_union_edgeBlocked}
    If $A,B\subseteq V$ satisfy $\inj(A)$, $\inj(B)$, and $D_f>0$ for all $f$,
    then $\inj(A\cup B)$.
\end{lemma}
\begin{proof}\leanok
    Set $(P_i)_{i=0}^3$ as in (\ref{eq:peps_four_region_parts}) and suppose
    $\sum_\rho c_\rho T_{A,A\cup B}^\rho=0$. Left-inverting $\inj(A)$ strips
    $P_0\cup P_1=A$, leaving the $P_2$-contraction zero for every $\partial A$
    configuration. Partitioning $\partial(A\cup B)$ into the edges incident to
    $B$ and the outer edges of $P_0$, fixing the outer-$P_0$ label to a
    reference value, reinserting $P_1$, and left-inverting $\inj(B)$ forces
    $c_\rho=0$ on each fiber; surjectivity of global configurations onto
    $\partial(A\cup B)$ (from $D_f>0$) then gives $c_\rho=0$ for all $\rho$.
\end{proof}

\begin{definition}[Injectivity predicate for blocked regions]%
    \label{def:peps_regionInjectivityData}
    \lean{TNLean.PEPS.RegionInjectivityData}
    \lean{TNLean.PEPS.RegionInjectivityUnionClosure}
    \leanok
    A region-injectivity predicate $\inj$ assigns an injectivity assertion
    $\inj(R)$ to each finite region $R\subseteq V$. It satisfies union closure
    when $\inj(R)\wedge\inj(S)\Longrightarrow\inj(R\cup S)$ for all $R,S\subseteq V$.
\end{definition}

\begin{definition}[Concrete region-injectivity predicate of a tensor]%

    asserts that the blocked-region tensor family $\{T_{A,R}^\mu\}_\mu$ is

\end{definition}

\begin{lemma}[Gauged blocked-region weight as a global sum]
    \label{thm:peps_regionBlockedWeight_applyGauge_eq_globalSum}
    \lean{TNLean.PEPS.regionBlockedWeight_applyGauge_eq_globalSum}
    \leanok
    \uses{def:applyGauge}
    Let $\widetilde B=\operatorname{Gauge}_X(B)$. For a region $R$, boundary
    configuration $\mu$, and physical configuration $\tau$ on $R$,
    \begin{align}
        T_{\widetilde B,R}^{\mu}(\tau)
         & =
        \sum_{\zeta}
        \left(\prod_{f\in\partial R}G_f^X(\mu_f,\zeta_f)\right)
        \prod_{v\in R}B_v(\zeta|_{\inc(v)},\tau_v),
        \label{eq:peps_gauged_global_sum}
    \end{align}
    where $\zeta$ ranges over global virtual configurations and $G_f^X$ is the
    boundary gauge on $f\in\partial R$.
\end{lemma}
\begin{proof}\leanok
    On every edge $e$ internal to $R$, the two endpoint gauges cancel via
    $\sum_a (X_e)_{\alpha,a}(X_e^{-1})_{a,\beta}=(X_eX_e^{-1})_{\alpha,\beta}=\delta_{\alpha,\beta}$,
    leaving only the boundary factors $G_f^X(\mu_f,\zeta_f)$ for $f\in\partial R$.
\end{proof}

\begin{lemma}[Boundary-coupling form of the gauged blocked weight]
    \label{thm:peps_regionBlockedWeight_applyGauge}
    \lean{TNLean.PEPS.regionBlockedWeight_applyGauge}
    \leanok
    \uses{thm:peps_regionBlockedWeight_applyGauge_eq_globalSum}
    With $K_R^X(\mu,\nu):=\prod_{f\in\partial R}G_f^X(\mu_f,\nu_f)$ as in
    \cite[Section~4, proof of Theorem~3]{Molnar2018NormalPEPS},
    \begin{align}
        T_{\widetilde B,R}^{\mu}(\tau)
         & =
        \sum_{\nu}K_R^X(\mu,\nu)\,T_{B,R}^{\nu}(\tau).
        \label{eq:peps_boundary_coupling}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Group the global sum (\ref{eq:peps_gauged_global_sum}) of
    Lemma~\ref{thm:peps_regionBlockedWeight_applyGauge_eq_globalSum} by the
    boundary restriction $\nu=\zeta|_{\partial R}$.
\end{proof}

\begin{lemma}[The boundary coupling has a right inverse]
    \label{thm:peps_sum_regionBoundaryCoupling_mul_inv}
    \lean{TNLean.PEPS.sum_regionBoundaryCoupling_mul_inv}
    \leanok
    For $K_R^X(\mu,\nu)=\prod_{f\in\partial R}G_f^X(\mu_f,\nu_f)$ and
    $(K_R^X)^{-1}(\nu,\rho)=\prod_{f\in\partial R}(G_f^X)^{-1}(\nu_f,\rho_f)$,
    \begin{align}
        \sum_{\nu}K_R^X(\mu,\nu)\,(K_R^X)^{-1}(\nu,\rho)
         & =
        \delta_{\mu,\rho}.
        \label{eq:peps_boundary_coupling_inverse}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Factoring the sum over $f\in\partial R$ gives
    \begin{align}
        \sum_{\nu}K_R^X(\mu,\nu)\,(K_R^X)^{-1}(\nu,\rho)
         & =
        \prod_{f\in\partial R}
        \sum_a G_f^X(\mu_f,a)\,(G_f^X)^{-1}(a,\rho_f)
        = \prod_{f\in\partial R}\delta_{\mu_f,\rho_f}
        = \delta_{\mu,\rho}.
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Gauge preservation of blocked-region injectivity]
    \label{thm:peps_regionBlockedTensorInjective_applyGauge}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_applyGauge}
    \leanok
    \uses{def:applyGauge, thm:peps_regionBlockedWeight_applyGauge, thm:peps_sum_regionBoundaryCoupling_mul_inv}
    For $\widetilde B=\operatorname{Gauge}_X(B)$,
    $\inj(T_{B,R})\Longrightarrow\inj(T_{\widetilde B,R})$.
\end{lemma}
\begin{proof}\leanok
    If $\sum_\mu c_\mu T_{\widetilde B,R}^{\mu}=0$, then (\ref{eq:peps_boundary_coupling})
    and linear independence of $\{T_{B,R}^{\nu}\}_{\nu}$ give
    $\sum_{\mu}c_\mu K_R^X(\mu,\nu)=0$ for all $\nu$; multiplying by
    $(K_R^X)^{-1}(\nu,\rho)$ and applying (\ref{eq:peps_boundary_coupling_inverse})
    yields $c_\rho=0$.
\end{proof}

\begin{lemma}[Kernel descent for blocked regions]%
    \label{thm:peps_regionBlockedTensorInjective_of_isVertexInjective}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_of_isVertexInjective}
    \leanok
    \uses{def:IsVertexInjective, def:peps_regionInjectivityDataOf}
    If $A$ is vertex-injective and $D_f>0$ for every virtual bond $f$, then
    $\inj_A(R)$ holds for every finite region $R\subseteq V$.
    % Paper-gap note: docs/paper-gaps/peps_injective_ft_section3_route.tex.
\end{lemma}
\begin{proof}\leanok
    If $\sum_{\rho} c_\rho\,T_{A,R}^{\rho}(\tau)=0$ for all $\tau$, inverting
    the vertex-injective local tensors $A_v$ ($v\in R$) one by one descends to
    the empty region and forces $c_\rho=0$ for all $\rho$.
\end{proof}

\begin{lemma}[Union closure for a vertex-injective tensor]%
    \label{thm:peps_regionInjectivityUnionClosure_of_isVertexInjective}
    \lean{TNLean.PEPS.regionInjectivityUnionClosure_of_isVertexInjective}
    \leanok
    \uses{def:IsVertexInjective, def:peps_regionInjectivityDataOf, thm:peps_regionBlockedTensorInjective_of_isVertexInjective}
    If $A$ is vertex-injective and $D_f>0$ for all $f$, then $\inj_A$ satisfies
    $\inj_A(R)\wedge\inj_A(S)\Longrightarrow\inj_A(R\cup S)$ for all $R,S\subseteq V$.
    % Paper-gap note: docs/paper-gaps/peps_injective_ft_section3_route.tex.
\end{lemma}
\begin{proof}\leanok
    Apply Lemma~\ref{thm:peps_regionBlockedTensorInjective_of_isVertexInjective}
    directly to $R\cup S$.
\end{proof}

\begin{lemma}[Union closure from the overlapping union lemma]%
