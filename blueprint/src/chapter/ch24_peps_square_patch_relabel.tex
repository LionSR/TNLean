\section{Relabelling the sites of a square patch}
\label{sec:peps_square_patch_relabel}

\begin{lemma}[Relabelled square patches]
    \label{lem:peps_square_patch_relabel}
    \lean{TNLean.PEPS.EncodedFrame.SquarePatch.exists_relabel,
        TNLean.PEPS.EncodedFrame.HasAdaptivePatch.relabel}
    \leanok
    Let $e\colon V\to W$ be a bijection of finite site sets, and give each site
    $w\in W$ the position of $e^{-1}(w)$. Square cylinder data on $V$ at $(z,u)$
    give square cylinder data on $W$ at $(z,u)$ with the same radii and the same
    inside dimensions, whose sum of cylinder projectors is the transport of the
    original sum along the configuration bijection induced by $e$. Consequently
    the conclusion of the adaptive patch constraint at $(z,u)$ for a vector
    $\Omega$ on $V$ holds, with the same dimension bound and error, for the
    transported vector on $W$. No nonemptiness condition is imposed on the
    sites or samples.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_frame_square_patch, def:peps_adaptive_patch_constraint}
    The bijection identifies the sample of each square on $V$ with the sample
    of the same square on $W$, and the induced configuration bijection is
    unitary. The cylinder description is independent of how the sites are
    labelled (\cite{OpenAI2026PolynomialPEPS}, \texttt{03-patches.tex}, lines
    24--49 and 563--603).
\end{proof}
