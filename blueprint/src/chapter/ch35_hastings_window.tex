\section{The window of large cut entropy}
\label{sec:hastings_window}

Hastings' proof of the one-dimensional area law \cite{Hastings2007AreaLaw}
argues by contradiction from a cut of large entropy. For any unit vector
$\Omega$ on the chain $\{0,\ldots,L-1\}$ of local dimension $q$, the cut
entropy $S(\lbrack 0,k))$ changes by at most $\log q$ per site, so
$S(\lbrack 0,i))>S_{\max}$ produces a window $[i,i+l_0)$ on which
$S(\lbrack 0,k))\ge 2S_{\max}/3$ and the maximal $l$-site interval entropy
$S_l$ satisfies $0\le S_l\le l\log q$ and $S_{2l}\le 2S_l$.

\begin{lemma}[The cut entropy moves slowly]
    \label{thm:hastings_cut_entropy_slow}
    \lean{
        TNLean.MPS.AreaLaw.disjoint_interval_interval,
        TNLean.MPS.AreaLaw.Hastings.interval_union_interval,
        TNLean.MPS.AreaLaw.Hastings.cutEntropy_sub_le
    }
    \leanok
    For any unit vector $\Omega$ on the chain of $L$ sites and $i\le k$,
    \begin{align}
        S(\lbrack 0,k))\ \ge\ S(\lbrack 0,i))-(k-i)\log q .
        \label{eq:hastings_cut_entropy_slow}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:chain_entropy_inequalities}
    Since $\lbrack 0,k)=\lbrack 0,i)\sqcup\lbrack i,k)$, the Araki--Lieb and
    dimension bounds of Lemma~\ref{lem:chain_entropy_inequalities} give
    $S(\lbrack 0,i))\le S(\lbrack 0,k))+S(\lbrack i,k))\le S(\lbrack 0,k))+(k-i)\log q$.
\end{proof}

\begin{lemma}[The window of large cut entropy]
    \label{thm:hastings_window_cut_entropy}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.windowLength,
        TNLean.MPS.AreaLaw.Hastings.windowLength_mul_log_le,
        TNLean.MPS.AreaLaw.Hastings.lt_windowLength_add_one,
        TNLean.MPS.AreaLaw.Hastings.two_mul_div_three_le_cutEntropy
    }
    \leanok
    Let $S_{\max}\ge 0$ and $l_0\in\N$ satisfy $l_0\log q\le S_{\max}/3$ (e.g.\
    $l_0=\lfloor S_{\max}/(3\log q)\rfloor$ for $q\ge 2$). If
    $S(\lbrack 0,i))>S_{\max}$, then for all $k\in[i,i+l_0]$,
    \begin{align}
        S(\lbrack 0,k))\ \ge\ \tfrac23\, S_{\max}.
        \label{eq:hastings_window_cut_entropy}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:hastings_cut_entropy_slow}
    By~(\ref{eq:hastings_cut_entropy_slow}),
    $S(\lbrack 0,k))\ge S(\lbrack 0,i))-(k-i)\log q>S_{\max}-l_0\log q\ge 2S_{\max}/3$
    (note $S(\lbrack 0,k))=0$ for $k\ge L$, so $S_{\max}>0$ forces $i+l_0<L$).
\end{proof}

\begin{definition}[The window entropy]
    \label{def:hastings_window_entropy}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.windowStarts,
        TNLean.MPS.AreaLaw.Hastings.mem_windowStarts,
        TNLean.MPS.AreaLaw.Hastings.windowStarts_nonempty,
        TNLean.MPS.AreaLaw.Hastings.windowEntropy,
        TNLean.MPS.AreaLaw.Hastings.regionEntropy_le_windowEntropy,
        TNLean.MPS.AreaLaw.Hastings.exists_windowEntropy_eq
    }
    \leanok
    For the window $\lbrack i,i+l_0)$ and $l\in\N$, define the \emph{window
        entropy}
    \begin{align}
        S_l=\begin{cases}
                \max\bigl\{S(\lbrack a,a+l)):i\le a,\ a+l\le i+l_0\bigr\}, & l\le l_0, \\
                0,                                                         & l>l_0.
            \end{cases}
        \label{eq:hastings_window_entropy}
    \end{align}
    The window $[i,i+l_0)$ consists of the $l_0$ sites between cuts $i$ and
    $i+l_0$, all of which satisfy~(\ref{eq:hastings_window_cut_entropy}) (omitting
    site $i-1$ of~\cite{Hastings2007AreaLaw}).
\end{definition}

\begin{lemma}[Bounds on the window entropy]
    \label{thm:hastings_window_entropy_bounds}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.regionEntropy_nonneg,
        TNLean.MPS.AreaLaw.Hastings.windowEntropy_nonneg,
        TNLean.MPS.AreaLaw.Hastings.windowEntropy_le,
        TNLean.MPS.AreaLaw.Hastings.windowEntropy_two_mul_le
    }
    \leanok
    \uses{def:hastings_window_entropy}
    For any unit vector and every $l\in\N$,
    \begin{align}
        0\ \le\ S_l\ \le\ l\log q,
        \qquad
        S_{2l}\ \le\ 2S_l .
        \notag
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:chain_entropy_inequalities}
    By Lemma~\ref{lem:chain_entropy_inequalities}, $0\le S(\lbrack a,a+l))\le l\log q$.
    For $2l\le l_0$, choosing $a$ attaining $S_{2l}=S(\lbrack a,a+2l))$ in
    (\ref{eq:hastings_window_entropy}) and using subadditivity on
    $\lbrack a,a+2l)=\lbrack a,a+l)\sqcup\lbrack a+l,a+2l)$ gives
    $S_{2l}\le S(\lbrack a,a+l))+S(\lbrack a+l,a+2l))\le 2S_l$; for $2l>l_0$,
    $S_{2l}=0\le 2S_l$.
\end{proof}

\begin{lemma}[The entropy gain on the window gives the strengthened doubling, stage form]
    \label{thm:hastings_window_claim_of_relent}
    \lean{TNLean.MPS.AreaLaw.Hastings.windowEntropy_two_mul_le_of_relent}
    \leanok
    \uses{def:hastings_window_entropy}
    Let $\xi',C_1,C_2\in\R$. If every $\lbrack a,a+2l)\subseteq\lbrack i,i+l_0)$
    satisfies
    \begin{align}
        S(\lbrack a,a+2l))\ \le\ S(\lbrack a,a+l))+S(\lbrack a+l,a+2l))
        -\bigl(1-2C_1e^{-l/\xi'}\bigr)\frac{l}{\xi'}+\ln C_1+C_2 ,
        \label{eq:hastings_window_relent_hyp}
    \end{align}
    then for every $l$ with $2l\le l_0$,
    \begin{align}
        S_{2l}\ \le\ 2S_l-\bigl(1-2C_1e^{-l/\xi'}\bigr)\frac{l}{\xi'}+\ln C_1+C_2 .
        \label{eq:hastings_window_claim}
    \end{align}
    This is the step from eq.~(relent) to eq.~(claim) in
    \cite[lines 201--204 and 234--242]{Hastings2007AreaLaw}.
\end{lemma}

\begin{proof}
    \leanok
    Choose $a$ attaining $S_{2l}=S(\lbrack a,a+2l))$ in
    (\ref{eq:hastings_window_entropy}); since
    $\lbrack a,a+l),\lbrack a+l,a+2l)\subseteq\lbrack i,i+l_0)$ have entropy at
    most $S_l$, (\ref{eq:hastings_window_relent_hyp}) yields
    (\ref{eq:hastings_window_claim}).
\end{proof}
