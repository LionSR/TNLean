\begin{definition}[Parent commuting Hamiltonian on the tripartite space]
    \label{def:tripartite_parent_commuting_hamiltonian}
    \lean{ManyBody.HasCommutingParentHamiltonian}
    \leanok
    \uses{def:tripartite_local_lifts}
    In \cite[Definition~D.2, lines 2205--2218]{Cirac2016MPDO_arXiv}, there are
    orthogonal projectors $P_{AX}$ and $P_{XB}$ on the two local factors whose
    lifts commute and whose ranges satisfy the source ground-space intersection
    condition $\ran P=\ran\widehat P_{AX}\cap\ran\widehat P_{XB}$, so that
    $Q_{AX}=\Id-P_{AX}$ and $Q_{XB}=\Id-P_{XB}$ are the commuting parent terms.
    Equivalently, on a system of sites with disjoint regions $A$ and $B$, there
    is a two-term interaction, one term supported on the complement of $B$ and
    one on the complement of $A$, whose terms are commuting orthogonal
    projectors and whose common kernel is $\ran P$.
\end{definition}

\begin{lemma}[Ground-space intersection and projector product]
    \label{thm:tripartite_ground_space_intersection_iff_product}
    \lean{TripartiteDecorrelation.range_mul_eq_inter,
        TripartiteDecorrelation.hermitian_idempotent_eq_of_range_eq,
        ManyBody.HasCommutingParentHamiltonian.eq_mul}
    \leanok
    \uses{def:tripartite_parent_commuting_hamiltonian}
    Let $P$, $P_{AX}$, and $P_{XB}$ be orthogonal projectors, and write their
    lifts to the tripartite space as
    $\widehat P_{AX}=P_{AX}\otimes\Id_B$ and
    $\widehat P_{XB}=\Id_A\otimes P_{XB}$.
    Suppose that the lifted projectors commute,
    $\widehat P_{AX}\widehat P_{XB}
        =\widehat P_{XB}\widehat P_{AX}$.
    In the passage from
    \cite[Definition~D.2, lines 2205--2218]{Cirac2016MPDO_arXiv} to the
    projector identity at lines 2279--2289,
    \begin{align}
        \ran P
        =\ran\widehat P_{AX}\cap\ran\widehat P_{XB}
        \quad\Longleftrightarrow\quad
        \widehat P_{AX}\widehat P_{XB}=P.
        \label{eq:parent_tripartite_projector_product}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    A vector lies in the range of
    $\widehat P_{AX}\widehat P_{XB}$ precisely when it is fixed by both
    commuting idempotents. Hence
    $\ran(\widehat P_{AX}\widehat P_{XB})
        =\ran\widehat P_{AX}\cap\ran\widehat P_{XB}$.
    The product is again an orthogonal projector. Orthogonal projectors with
    equal ranges are equal, giving the forward implication in
    (\ref{eq:parent_tripartite_projector_product}); the reverse implication
    follows by substituting the product identity.
\end{proof}

\begin{lemma}[Parent commuting Hamiltonian--complex-linear decorrelation equivalence]
    \label{thm:tripartite_parent_iff_complex_linear_decorrelated}
    \lean{ManyBody.HasCommutingParentHamiltonian.isDecorrelated,
        ManyBody.hasCommutingParentHamiltonian_of_isDecorrelated}
    \leanok
    \uses{def:tripartite_decorrelation, def:tripartite_parent_commuting_hamiltonian, thm:tripartite_support_products, thm:tripartite_ground_space_intersection_iff_product}
    Let $P=P^\dagger=P^2$. Then $P$ is the ground-space projector of a
    parent commuting Hamiltonian on $AX$ and $XB$ if and only if regions
    $A$ and $B$ satisfy complex-linear decorrelation.
\end{lemma}

\begin{proof}
    \leanok
    From decorrelation, the preceding product theorem applied to the support
    projectors of $\rho_{AX}$ and $\rho_{XB}$ proves their commutation and
    identifies their product with $P$.

    Conversely, from commuting parent projectors, the ground-space intersection
    theorem gives
    $P=\widehat P_{AX}\widehat P_{XB}
        =\widehat P_{XB}\widehat P_{AX}$. Locality gives
    $[O_A,\widehat P_{XB}]=[O_B,\widehat P_{AX}]=0$, while
    $\widehat P_{XB}(\Id-P)\widehat P_{AX}=0$.
    Therefore, using the two locality commutators,
    \begin{align}
        P O_A(\Id-P)O_B P
         & =\widehat P_{AX}O_A
        [\widehat P_{XB}(\Id-P)\widehat P_{AX}]
        O_B\widehat P_{XB}=0.
        \notag
    \end{align}
\end{proof}

\begin{theorem}[Parent commuting Hamiltonian--observable decorrelation equivalence]
    \label{thm:tripartite_parent_iff_decorrelated}
    \lean{ManyBody.hasCommutingParentHamiltonian_iff_isDecorrelated}
    \leanok
    \uses{def:tripartite_observable_decorrelation, thm:tripartite_observable_decorrelation_iff, thm:tripartite_parent_iff_complex_linear_decorrelated}
    For $P=P^\dagger=P^2$,
    \cite[Proposition~D.3, lines 2221--2289]{Cirac2016MPDO_arXiv} states that
    $P$ is the ground-space projector of a parent commuting Hamiltonian on $AX$
    and $XB$ if and only if regions $A$ and $B$ are decorrelated when tested on
    Hermitian observables. By
    Lemma~\ref{thm:tripartite_observable_decorrelation_iff}, the latter
    condition is complex-linear decorrelation.
\end{theorem}

\begin{proof}
    \leanok
    Combining the complex-linear parent equivalence with the decomposition
    $O=H+\mathrm{i}K$ from
    Lemma~\ref{thm:tripartite_observable_decorrelation_iff} gives the
    equivalence.
\end{proof}
