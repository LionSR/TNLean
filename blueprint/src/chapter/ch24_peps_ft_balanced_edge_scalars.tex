\section{Balanced Edge Scalars}\label{sec:peps_balanced_edge_scalars}

The edge gauges are determined only up to a scalar on each edge. The residual
freedom is the balanced edge scalars: a nonzero scalar $c_e$ on every edge,
whose endpoint readings $c_{v,e}$ multiply to one around each vertex,
$\prod_{e\ni v}c_{v,e}=1$.
% Formula: prod_{e incident to v} c_{v,e}=1 for an arbitrary graph vertex v.
% Ink-to-index: the three displayed rays are the first, a typical, and the last
% incident virtual half-edge; each ring is multiplication by its endpoint
% scalar; intervening incident edges are suppressed.
% Contraction and boundary signature: each scalar is contracted between the
% vertex and its virtual boundary; the physical leg stays open.  The exact
% signature is (V,P)=(deg(v),1), not the degree-four square specialization.
% Source verdict: the edgewise scalar freedom is the uniqueness statement in
% Papers/1804.04964/paper_normal.tex:1207--1212; the displayed product is the
% exact local balance condition of this definition, not a separate paper figure.
\begin{center}
    \tenkzkernel
    % Each incident half-edge carries its endpoint scalar as a ring whose
    % outward port stays open.
    \begin{tenkz}[rows={wire}, cols=1, bonds=none]
        \tn[name=balanceVertex, label pos=w,
            ports={225:virtual, 270:virtual, 315:virtual,
                    90:physical}]{v}
        \tn[at=1 sw of balanceVertex, name=balanceFirst, skin=ring,
            species=operator, label pos=w,
            ports={45:virtual, 225:virtual}]{c_{v,e_1}}
        \tn[at=1 s of balanceVertex, name=balanceTypical, skin=ring,
            species=operator, label pos=e,
            ports={90:virtual, 270:virtual}]{c_{v,e_k}}
        \tn[at=1 se of balanceVertex, name=balanceLast, skin=ring,
            species=operator, label pos=e,
            ports={135:virtual, 315:virtual}]{c_{v,e_{\deg(v)}}}
        \tnwire{balanceVertex.225}{balanceFirst.45}
        \tnwire{balanceVertex.270}{balanceTypical.90}
        \tnwire{balanceVertex.315}{balanceLast.135}
    \end{tenkz}
    \qquad $\prod_{e\ni v}c_{v,e}=1$.
\end{center}
Two gauge families related by such an assignment give the same gauge
equivalence, which is the precise sense in which the gauges of the Fundamental
Theorem are unique: modulo balanced edge scalars, not modulo a single global
constant.

\begin{definition}[Gauge equivalence modulo balanced edge scalars]%
    \label{def:GaugeEquivModEdgeScalars}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars}
    \leanok
    \uses{def:edgeGaugeAt, def:edgeScalarAt, def:IsVertexBalanced}
    Gauge families $X,Y$ are equivalent modulo balanced edge scalars, written
    $X\sim_{\mathrm{bal}}Y$, if there exist $c_e\in\C^\times$ ($e\in E$) such
    that $X_{v,e}=c_{v,e}Y_{v,e}$ for all $v\in e$ and
    $\prod_{e\ni v}c_{v,e}=1$ for all $v\in V$.
\end{definition}

\begin{theorem}[Multiplication of oriented edge scalars]%
    \label{thm:edgeScalarAt_mul}
    \lean{TNLean.PEPS.edgeScalarAt_mul}
    \leanok
    \uses{def:edgeScalarAt}
    For edge-scalar families $c,d$, $(cd)_{v,e}=c_{v,e}d_{v,e}$ on every
    oriented half-edge $(v,e)$.
    % The endpoint-action convention is the equation in Definition~\ref{def:edgeGaugeAt}.
    % No independent source picture occurs here; the old scalar diagram suppressed the
    % boundary (deg(u)+deg(w)-2,2) of the two-vertex tensor network.
\end{theorem}
\begin{proof}\leanok
    By Definition~\ref{def:edgeScalarAt}, $(cd)_{v,e}=c_ed_e=c_{v,e}d_{v,e}$ at
    the lower endpoint and $(cd)_{v,e}=(c_ed_e)^{-1}=c_e^{-1}d_e^{-1}=c_{v,e}d_{v,e}$
    at the upper endpoint.
\end{proof}

\begin{theorem}[Inversion of oriented edge scalars]%
    \label{thm:edgeScalarAt_inv}
    \lean{TNLean.PEPS.edgeScalarAt_inv}
    \leanok
    \uses{def:edgeScalarAt}
    For any edge-scalar family $c$, $(c^{-1})_{v,e}=c_{v,e}^{-1}$ on every
    oriented half-edge $(v,e)$.
\end{theorem}
\begin{proof}\leanok
    By Definition~\ref{def:edgeScalarAt}, $(c^{-1})_{v,e}=c_e^{-1}=c_{v,e}^{-1}$
    at the lower endpoint and $(c^{-1})_{v,e}=(c_e^{-1})^{-1}=c_{v,e}^{-1}$ at
    the upper endpoint.
\end{proof}

\begin{theorem}[Nonzero oriented edge scalars]%
    \label{thm:edgeScalarAt_ne_zero}
    \lean{TNLean.PEPS.edgeScalarAt_ne_zero}
    \leanok
    \uses{def:edgeScalarAt}
    For any edge-scalar family $c\in(\C^\times)^E$, $c_{v,e}\ne 0$ for all $v\in e$.
\end{theorem}
\begin{proof}\leanok
    By Definition~\ref{def:edgeScalarAt}, $c_{v,e}\in\{c_e,c_e^{-1}\}\subset\C^\times$.
\end{proof}

\begin{theorem}[Product of balanced edge scalars]%
    \label{thm:IsVertexBalanced_mul}
    \lean{TNLean.PEPS.IsVertexBalanced.mul}
    \leanok
    \uses{def:IsVertexBalanced, thm:edgeScalarAt_mul}
    If $c$ and $d$ are vertex-balanced, then $cd$ is vertex-balanced.
\end{theorem}
\begin{proof}\leanok
    By Theorem~\ref{thm:edgeScalarAt_mul},
    $\prod_{e\ni v}(cd)_{v,e}=(\prod_{e\ni v}c_{v,e})(\prod_{e\ni v}d_{v,e})=1\cdot 1=1$.
\end{proof}

\begin{theorem}[Inverse of balanced edge scalars]%
    \label{thm:IsVertexBalanced_inv}
    \lean{TNLean.PEPS.IsVertexBalanced.inv}
    \leanok
    \uses{def:IsVertexBalanced, thm:edgeScalarAt_inv}
    If $c$ is vertex-balanced, then $c^{-1}$ is vertex-balanced.
\end{theorem}
\begin{proof}\leanok
    By Theorem~\ref{thm:edgeScalarAt_inv},
    $\prod_{e\ni v}(c^{-1})_{v,e}=(\prod_{e\ni v}c_{v,e})^{-1}=1^{-1}=1$.
\end{proof}

\begin{theorem}[Reflexivity of balanced edge-scalar equivalence]%
    \label{thm:GaugeEquivModEdgeScalars_refl}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.refl}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars}
    For every edge-gauge family $X$, $X\sim_{\mathrm{bal}}X$ via $c_e=1$.
    % This is the same algebraic endpoint convention as
    % Definition~\ref{def:edgeGaugeAt}, not a separate tensor-network identity.
\end{theorem}
\begin{proof}\leanok
    With $c_e=1$, $c_{v,e}=1$, $X_{v,e}=c_{v,e}X_{v,e}$, and
    $\prod_{e\ni v}c_{v,e}=1$.
\end{proof}

\begin{theorem}[Symmetry of balanced edge-scalar equivalence]%
    \label{thm:GaugeEquivModEdgeScalars_symm}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.symm}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars, thm:IsVertexBalanced_inv, thm:edgeScalarAt_ne_zero}
    If $X\sim_{\mathrm{bal}}Y$ via $c$, then $Y\sim_{\mathrm{bal}}X$ via $c^{-1}$.
    % The inverse endpoint action is already stated exactly by
    % Definition~\ref{def:edgeGaugeAt}; there is no additional source figure.
\end{theorem}
\begin{proof}\leanok
    From $X_{v,e}=c_{v,e}Y_{v,e}$ and Theorem~\ref{thm:edgeScalarAt_ne_zero} we
    get $Y_{v,e}=c_{v,e}^{-1}X_{v,e}$, and $c^{-1}$ is balanced by
    Theorem~\ref{thm:IsVertexBalanced_inv}.
\end{proof}

\begin{theorem}[Transitivity of balanced edge-scalar equivalence]%
    \label{thm:GaugeEquivModEdgeScalars_trans}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.trans}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars, thm:IsVertexBalanced_mul, thm:edgeScalarAt_mul}
    If $X\sim_{\mathrm{bal}}Y$ via $c$ and $Y\sim_{\mathrm{bal}}Z$ via $d$,
    then $X\sim_{\mathrm{bal}}Z$ via $cd$.
\end{theorem}
\begin{proof}\leanok
    By Theorem~\ref{thm:edgeScalarAt_mul},
    $X_{v,e}=c_{v,e}Y_{v,e}=(c_{v,e}d_{v,e})Z_{v,e}=(cd)_{v,e}Z_{v,e}$, and $cd$
    is balanced by Theorem~\ref{thm:IsVertexBalanced_mul}.
\end{proof}

\begin{theorem}[The balanced edge-scalar quotient is an equivalence relation]%
    \label{thm:GaugeEquivModEdgeScalars_equivalence}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.equivalence}
    \leanok
    \uses{thm:GaugeEquivModEdgeScalars_refl, thm:GaugeEquivModEdgeScalars_symm, thm:GaugeEquivModEdgeScalars_trans}
    The relation $\sim_{\mathrm{bal}}$ on edge-gauge families is an equivalence
    relation.
\end{theorem}
\begin{proof}\leanok
    Combine Theorems~\ref{thm:GaugeEquivModEdgeScalars_refl},
    \ref{thm:GaugeEquivModEdgeScalars_symm},
    and~\ref{thm:GaugeEquivModEdgeScalars_trans}.
\end{proof}

\begin{definition}[The balanced edge-scalar quotient relation]%
    \label{def:GaugeEquivModEdgeScalars_setoid}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.setoid}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars, thm:GaugeEquivModEdgeScalars_equivalence}
    The setoid on edge-gauge families with equivalence relation
    $\sim_{\mathrm{bal}}$ from Theorem~\ref{thm:GaugeEquivModEdgeScalars_equivalence}.
\end{definition}

\begin{theorem}[Balanced edge scalars preserve the local gauge action]%
    \label{thm:GaugeEquivModEdgeScalars_gaugeVertex_eq}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.gaugeVertex_eq}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars, def:gaugeVertex}
    If $X\sim_{\mathrm{bal}}Y$, then $(A^X)_v(\eta,\sigma)=(A^Y)_v(\eta,\sigma)$
    for all $v,\eta,\sigma$.
\end{theorem}
\begin{proof}\leanok\uses{def:GaugeEquivModEdgeScalars, def:gaugeVertex}
    In Definition~\ref{def:gaugeVertex}, each summand for $(A^X)_v$ differs from
    that for $(A^Y)_v$ by the factor $\prod_{e\ni v}c_{v,e}=1$.
\end{proof}

\begin{theorem}[Balanced edge scalars preserve the gauged tensor]%
    \label{thm:GaugeEquivModEdgeScalars_applyGauge_eq}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.applyGauge_eq}
    \leanok
    \uses{thm:GaugeEquivModEdgeScalars_gaugeVertex_eq, def:applyGauge}
    If $X\sim_{\mathrm{bal}}Y$, then $A^X=A^Y$.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:GaugeEquivModEdgeScalars_gaugeVertex_eq} at each
    $v,\eta,\sigma$ in Definition~\ref{def:applyGauge}.
\end{proof}

\begin{theorem}[Balanced edge scalars preserve the gauged state]%
    \label{thm:GaugeEquivModEdgeScalars_applyGauge_sameState}
    \lean{TNLean.PEPS.GaugeEquivModEdgeScalars.applyGauge_sameState}
    \leanok
    \uses{thm:GaugeEquivModEdgeScalars_applyGauge_eq, def:peps_same_state}
    If $X\sim_{\mathrm{bal}}Y$, then $\Coeff_{A^X}(\sigma)=\Coeff_{A^Y}(\sigma)$
    for every physical configuration $\sigma$.
\end{theorem}
\begin{proof}\leanok
    Immediate from $A^X=A^Y$ (Theorem~\ref{thm:GaugeEquivModEdgeScalars_applyGauge_eq}).
\end{proof}

\begin{theorem}[Gauge uniqueness modulo balanced edge scalars]%
    \label{thm:gauge_unique_mod_edge_scalars}
    \lean{TNLean.PEPS.gauge_unique_mod_edge_scalars}
    \leanok
    \uses{def:GaugeEquivModEdgeScalars, def:peps_injective_tensor, thm:peps_piProductFormsScalar}
    If $A$ is an injective PEPS tensor and $A^X=A^Y$, then $X\sim_{\mathrm{bal}}Y$.
    % Correction documented in docs/paper-gaps/peps_gauge_edge_scalars.tex.
    % Nonzero bond spaces are part of the injectivity hypothesis, as in the
    % existence direction; see docs/paper-gaps/peps_injective_ft_section3_route.tex.
\end{theorem}
\begin{proof}\leanok
    For each $v\in V$, $(A^X)_v=(A^Y)_v$ and injectivity of $A_v$ equate the
    incident gauge products $\prod_{e\ni v}X_{v,e}=\prod_{e\ni v}Y_{v,e}$.
    Since $D_A(e)>0$ and each $X_{v,e},Y_{v,e}$ is invertible,
    Theorem~\ref{thm:peps_piProductFormsScalar} yields $c_{v,e}\in\C^\times$
    with $X_{v,e}=c_{v,e}Y_{v,e}$ and $\prod_{e\ni v}c_{v,e}=1$; taking $c_e$
    at the lower endpoint of $e$ forces $c_{v,e}=c_e^{-1}$ at the upper
    endpoint, so $X\sim_{\mathrm{bal}}Y$.
\end{proof}

\begin{remark}
    The local left inverse $\Psi_v$, the realization map $T \mapsto O_T$,
    the split at one incident edge, and the partition
    $V = \{u\} \sqcup (V \setminus \{u,v\}) \sqcup \{v\}$ provide the local
    structure for the edge-centered reduction of
    \cite[Section~3]{Molnar2018NormalPEPS}. The local gauge then follows from
    comparison with the corresponding three-site injective chain, with uniqueness
    determined modulo balanced edge scalars.
\end{remark}

\subsection{Inserted-site scalar extraction}
\label{subsec:peps_inserted_site_scalar_extraction}

The final comparison of \cite[Section~3, proof of Theorem~3]{Molnar2018NormalPEPS}
turns two region proportionalities into a single scalar at one site. Fix a finite
region $R$ and a site $v\notin R$, and write $S=R\cup\{v\}$ for the
region with $v$ adjoined. A local configuration $\eta$ at $v$ assigns a virtual
index to every edge incident to $v$. The label on its $v$-incident $R$-boundary
edges reads $\eta$ on the $v$-incident edges that bound $R$, while the base
boundary label $\mu$ fixes the remaining $v$-incident edges. Under inserted-site
consistency, the $R$-boundary label and the values fixed by $\mu$ on the
remaining $v$-incident edges determine $\eta$ completely.

\begin{lemma}[Inserted-site consistency pins a local configuration to its $R$-boundary label]%
    \label{lem:peps_localConfig_eq_of_insertConsistent}
    \lean{TNLean.PEPS.localConfig_eq_of_insertConsistent}
    \leanok
    For $v\notin R$ and $\mu\in\partial(R\cup\{v\})$, any two local
    configurations $\eta,\eta'\in\mc V_v$ consistent with $\mu$ that induce the
    same $R$-boundary label $b(\mu,\eta)=b(\mu,\eta')$ satisfy $\eta=\eta'$.
\end{lemma}
\begin{proof}\leanok
    On each $e\in\operatorname{Inc}(v)$, if $e\in\partial R$ then
    $\eta_e=b(\mu,\eta)_e=b(\mu,\eta')_e=\eta'_e$, and if $e\notin\partial R$
    then $e\in\partial(R\cup\{v\})$ so consistency with $\mu$ gives
    $\eta_e=\mu_e=\eta'_e$.
\end{proof}

\begin{theorem}[Inserted-site scalar extraction from two region proportionalities]%
    \label{thm:peps_component_eq_of_regionProportional}
    \lean{TNLean.PEPS.component_eq_of_regionProportional}
    \leanok
    \uses{lem:peps_localConfig_eq_of_insertConsistent}
    Suppose $D_A=D_C>0$, $\inj(T_{C,R})$ holds, and $A_R=c_RC_R$ and
    $A_S=c_SC_S$ for $S=R\cup\{v\}$ ($v\notin R$) with $c_R\ne 0$. Then for
    every $\eta\in\mc V_v$ and $\sigma$,
    \begin{align}
        A_v(\eta,\sigma)
         & =\frac{c_S}{c_R}\,C_v(\eta,\sigma).
        \label{eq:peps_inserted_ratio}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    With $b(\mu,\eta)\in\partial R$ the boundary label induced by $\mu\in\partial S$
    and $\eta\in\mc V_v$, and $m>0$ the interior bond multiplicity,
    \begin{align}
        m\,A_S(\mu,\sigma)
         & =
        \sum_{\eta\ \text{consistent with}\ \mu}
        A_v(\eta,\sigma(v))
        A_R(b(\mu,\eta),\sigma|_R),
        \label{eq:peps_insert_factorization}
    \end{align}
    and similarly for $C$. Substituting $A_S=c_SC_S$ and $A_R=c_RC_R$ into
    (\ref{eq:peps_insert_factorization}) and dividing by $m$ gives
    $\sum_{\eta}(c_RA_v(\eta,\sigma(v))-c_SC_v(\eta,\sigma(v)))C_R(b(\mu,\eta),\sigma|_R)=0$.
    Since $\eta\mapsto b(\mu,\eta)$ is injective by
    Lemma~\ref{lem:peps_localConfig_eq_of_insertConsistent} and $T_{C,R}$ is
    linearly independent, each coefficient vanishes, giving
    (\ref{eq:peps_inserted_ratio}).
\end{proof}

\begin{corollary}[Per-vertex gauge relation from two two-block proportionalities]%
    \label{cor:peps_component_eq_gaugeVertex_of_twoBlockProportional}
    \lean{TNLean.PEPS.component_eq_gaugeVertex_of_twoBlockProportional}
    \leanok
    \uses{thm:peps_component_eq_of_regionProportional, def:peps_regionTwoBlock, def:applyGauge, def:gaugeVertex}
    Suppose $D_A=D_B>0$, let $C=B^X$ reindexed along $D_A=D_B$, and assume
    $\inj(T_{C,R})$. If $B_{A,R}=c_RB_{C,R}$ and $B_{A,S}=c_SB_{C,S}$ for
    $S=R\cup\{v\}$ ($v\notin R$) with $c_R\ne 0$, then for every $\eta,\sigma$,
    \begin{align}
        A_v(\eta,\sigma)
         & =\frac{c_S}{c_R}\,C_v(\eta,\sigma).
        \label{eq:peps_vertex_gauge_ratio}
    \end{align}
\end{corollary}
\begin{proof}\leanok
    By Definition~\ref{def:peps_regionTwoBlock}, $B_{A,R}=c_RB_{C,R}$ and
    $B_{A,S}=c_SB_{C,S}$ mean $A_R=c_RC_R$ and $A_S=c_SC_S$, so
    Theorem~\ref{thm:peps_component_eq_of_regionProportional} yields
    (\ref{eq:peps_vertex_gauge_ratio}).
\end{proof}
