\section{One physical soft improving family before all energy estimates}
\label{sec:uniform_physical_soft_improving_family}

Theorem~\ref{thm:uniform_physical_soft_improving_family} combines the
uniform regional comparison with fixed middle tensors and a common
polynomial family. It constructs an improving operator for the very
same represented soft Hamiltonian, uniformly over the three estimated
regional energies. The numerical residual conditions remain explicit.
This is a corrected auxiliary to
\cite[Proposition~5.7 and Theorem~5.8]{Arad2016RigorousRG}; the source
qualifications remain those of
\cite{gap:alvv16_soft_regional_residuals,gap:alvv16_cluster_approximation}.

\begin{theorem}[A common physical family for all lower regional estimates]
    \label{thm:uniform_physical_soft_improving_family}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_uniform_physical_soft_improving_family}
    \leanok
    Fix $q\ge2$, $\ell,k\ge1$, $m\ge2$, $\gamma>0$, $0<\tau\le1$,
    $t\ge3m$, and real $T_0,T_1$ such that
    $T_0\ge\gamma/40+24$ and $t,T_1\ge156+2\gamma$.
    Suppose the corrected residuals satisfy
    \begin{align}
      3\left(32e^{-(T_0-\gamma/40-24)/8}
        +\frac{t}{m}(T_0/t)^m\right)&\le\tau\gamma/10^6,\notag\\
      3\left(32e^{-(T_1-156-2\gamma)/8}
        +\frac{t}{m}(T_1/t)^m\right)&\le\tau\gamma/10^6.
      \label{eq:uniform_soft_residual_budgets}
    \end{align}
    Set
    \begin{align}
      M&=3t(1+\log m)+4\ell+34, &
      \xi&=\min\left\{1,\frac{\tau\gamma^2}{20000(M+\gamma+8)}\right\},\notag\\
      U&=M+\xi+4+\gamma, &
      D&=(mk+1)(qk)^{36\ell+4\lfloor k/\ell\rfloor},\notag\\
      \Delta&=4\exp\left(-4k\sqrt{\frac{\gamma}{2U}}\right).
      \label{eq:uniform_soft_family_budgets}
    \end{align}
    Let $h_i$, $0\le i<N$, be positive-contraction nearest-neighbour
    interactions on $N+1$ sites of dimension $q$. Choose
    $a+2\ell\le b$ and $b+2\ell\le N$, and cut the chain at
    $0,a+1,a+2\ell,b+1,b+2\ell,N+1$. Denote the five internal
    Hamiltonians by $A_j$ and their crossing sum by $Y$.

    One can choose their minima $e_j$, a common unit minimizing vector
    $x$, and $E\in[0,4]$. They satisfy $A_jx=e_jx$ and $A_j-e_jI\ge0$.
    Thereafter one can choose native middle tensors $A_*$ and closing
    coefficients $C_0,\ldots,C_m$, and a family $W_0,\ldots,W_{n-1}$
    on $J=[a+\ell,b+\ell+1)$, before every estimated center below.
    The count satisfies $n\le D$.

    Write $b_*=2(e^{1/3}-1)$ and define
    \begin{align}
      s&=\max\left\{1,
        \left\lceil
          \frac{\log\max\{1,N^2\}-\log\log(1+\xi/(t2^{m+1}))}
               {-\log b_*}
        \right\rceil_+\right\},\notag\\
      B&=2s^2q^s(m+1), &
      Z(z)&=\mathcal T\left(A_*,\sum_{j=0}^{m}z^j C_j\right).
      \label{eq:uniform_soft_native_family}
    \end{align}
    The tensors $A_*$ have virtual dimension $B$. Each $W_i$ has an
    open-boundary matrix product representation in the paired physical
    alphabet, with all bonds at most $(\max\{1,B\})^{2k}$.

    For every $p_j\in[e_j-10,e_j]$ on $j\in\{0,2,4\}$, put
    \begin{align}
      F_j&=p_jI+h_{m,t}(A_j-p_jI),\notag\\
      G_p&=\sum_{j=0}^4
        \bigl(\mathbf1_{j\in\{0,2,4\}}F_j+
              \mathbf1_{j\notin\{0,2,4\}}A_j-e_jI\bigr)+Y,\notag\\
      z_p&=e^{(p_2-e_2)/t}, &
      \widehat G_p&=G_p-F_2+p_2I+\iota(Z(z_p)).
      \label{eq:uniform_soft_same_approximant}
    \end{align}
    Here $\iota$ is identity extension from the native middle interval
    $[a+2\ell,b+1)$. The matrix $\widehat G_p$ is Hermitian and
    \begin{align}
      \|h_{m,t}(A_2-p_2I)-\iota(Z(z_p))\|&\le\xi,\notag\\
      \|G_p-\widehat G_p\|&\le\xi, &
      \|\widehat G_p\|&\le M+\xi.
      \label{eq:uniform_soft_same_error}
    \end{align}

    Suppose $\mu_0$ is a spectral value of $H_0=\sum_i h_i$ and every
    spectral value is either $\mu_0$ or at least $\mu_0+\gamma$.
    Let $G_0$ be its ground space and let
    $T_p=\operatorname{ran}\mathbf1_{(-\infty,E+\gamma/4]}(\widehat G_p)$.
    Then $\mu_0=E+\sum_j e_j$, and
    \begin{align}
      T_p&=\operatorname{ran}
          \mathbf1_{(-\infty,E+3\gamma/4]}(\widehat G_p), &
      \dim T_p&=\dim G_0.
      \label{eq:uniform_soft_own_cluster}
    \end{align}
    The spaces $T_p,G_0$ are mutually $3\tau/200$-close.

    Let $\mathcal U$ be the canonical isometry placing $J$ first and
    put $T_p^c=\mathcal U T_p$. There are complementary operators
    $R_{p,i}$ and an actual operator
    $K_p=\sum_{i=0}^{n-1}W_i\otimes R_{p,i}$ that preserve
    $T_p^c$ and its orthogonal complement. They satisfy
    $\|K_pv\|\ge\|v\|$ on $T_p^c$ and
    $\|K_pv\|^2\le\Delta\|v\|^2$ on $(T_p^c)^\perp$.
    For every $r\ge0$ and every $\delta$-viable subspace $S$ for
    $T_p^c$, where $\delta<1$, the literal ordered-word image
    \begin{align}
      S_r&=\sum_{w\in\{0,\ldots,n-1\}^r}
           W_{w_0}\cdots W_{w_{r-1}}S
      \label{eq:uniform_soft_word_image}
    \end{align}
    has dimension at most $D^r\dim S$ and is
    $\Delta^r/(1-\delta)^2$-viable for $T_p^c$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:alvv_five_soft_uniform_clusters,lem:uniform_native_soft_middle,
      thm:common_closing_polynomial_family}
    Choose the regional minima once by the uniform physical comparison.
    At the true lower center $e_2$, construct the fixed heat atoms and
    native tensors of Lemma~\ref{lem:uniform_native_soft_middle}.
    Their closing matrix is polynomial in $z$.
    Lemma~\ref{thm:common_closing_polynomial_family} chooses one left
    family before $z$, the supported exterior pair, and the polynomial.

    Keep the two windows exact. With exterior terms
    $P_L=F_0+(p_2-\sum_j e_j)I$ and $P_R=F_4$, their alphabet sums
    exactly to $\widehat G_p$ in~(\ref{eq:uniform_soft_same_approximant}).
    Cancellation gives~(\ref{eq:uniform_soft_same_error}); the uniform
    comparison gives the cluster of this very same approximant in
    (\ref{eq:uniform_soft_own_cluster}).

    Apply the actual-cluster Chebyshev construction with lower cutoff
    $E+\gamma/4$, upper cutoff $E+3\gamma/4$, and spectrum ceiling $U$.
    Its polynomial decomposition uses the unchanged left family.
    Since $E\ge0$, the denominator $U-E-\gamma/4$ is at most $U$;
    monotonicity of the square root and exponential yields the
    contraction bound in~(\ref{eq:uniform_soft_family_budgets}).
    The minimum gain and complementary contraction give the ordered-word
    estimate in~(\ref{eq:uniform_soft_word_image}); its dimension is
    bounded by $n^r\dim S\le D^r\dim S$.
\end{proof}

The parameters in~(\ref{eq:uniform_soft_family_budgets}) precede the
chain length; the tensors and left family precede all three estimated
centers. The statement retains explicit residual conditions and does
not identify the source's fixed fourth-order choice. Adaptive selection
of centers, the complete generating algorithm and its running time
are separate conclusions.

\begin{lemma}[Retaining the represented matrix and its filter]
    \label{lem:uniform_physical_soft_filter_family}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_uniform_physical_soft_filter_family}
    \leanok
    Under the hypotheses and notation of
    Theorem~\ref{thm:uniform_physical_soft_improving_family}, one may
    retain just the five attained regional minima, their common unit
    minimizing vector, $E\in[0,4]$, and one core family $W$ with count
    at most $D$ and the same member bond bound.

    For every admissible triple of regional estimates, there is a
    Hermitian matrix $\widehat G_p$ with
    $\|G_p-\widehat G_p\|\le\xi$ and
    $\|\widehat G_p\|\le M+\xi$. For every original ground energy
    with spectral gap at least $\gamma$, this same matrix has the
    closed spectral target, rank equality and both $3\tau/200$
    comparisons of~(\ref{eq:uniform_soft_own_cluster}). Its original
    filter, tensor decomposition, four operator bounds and all
    ordered-word conclusions are retained unchanged.
    The family precedes all three estimates; the local tensor
    representation need not be specified again.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:uniform_physical_soft_improving_family}
    Use the preceding construction once, retain its represented
    Hermitian matrix as a witness for each triple of estimates, and
    keep its complete spectral and filter conclusions. No second
    approximant or filter is chosen.
\end{proof}

\begin{lemma}[One soft improving family on a prescribed long interval]
    \label{lem:uniform_long_interval_soft_improving_family}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_uniform_long_interval_soft_improving_family}
    \leanok
    Fix the numerical parameters and residual budgets
    of~(\ref{eq:uniform_soft_residual_budgets})--(\ref{eq:uniform_soft_family_budgets}).
    Let $h_i$, $0\le i<N$, be positive-contraction nearest-neighbour
    interactions on $N+1$ sites of dimension $q\ge2$. Prescribe
    $J=[p,r)$ with $0\le p\le r\le N+1$ and $4\ell<r-p$.

    There are integers $a,b$ with
    \begin{align}
      a+2\ell&\le b, & b+2\ell&\le N,\notag\\
      p&\le a+\ell, & b+\ell+1&\le r,\notag\\
      (a+2\ell)-p&\le2\ell, & r-(b+1)&\le2\ell.
      \label{eq:uniform_long_soft_middle_collars}
    \end{align}
    Moreover $J_0=[a+\ell,b+\ell+1)\subseteq J$ and
    $|J\setminus J_0|\le2\ell$.
    Form the five internal Hamiltonians $A_j$ and crossing sum $Y$
    at these cuts as in the preceding theorem. There are five attained
    minima $e_j$, one common unit minimizing vector $x$, $E\in[0,4]$,
    and a family $W_1,\ldots,W_n$ on $J$, chosen before every admissible
    triple of center estimates, such that
    \begin{align}
      A_jx&=e_jx, & A_j-e_jI&\ge0, & n&\le D^2.
    \end{align}
    In increasing site order, each $W_i$ has open matrix product bond
    bound $\max\{1,(\max\{1,B\})^{2k}\}$, with the same $B$ as
    in~(\ref{eq:uniform_soft_native_family}).

    For every admissible triple $p_j\in[e_j-10,e_j]$ on
    $j\in\{0,2,4\}$, define the actual soft matrix $G_p$ by
    (\ref{eq:uniform_soft_same_approximant}). There is a Hermitian
    $\widehat G_p$ with
    $\|G_p-\widehat G_p\|\le\xi$ and
    $\|\widehat G_p\|\le M+\xi$.
    For every ground energy $\mu_0$ of $H_0=\sum_i h_i$ separated
    from the rest of its spectrum by at least $\gamma$,
    $\mu_0=E+\sum_j e_j$. Let $G_0$ be the full ground space of $H_0$.
    The same matrix $\widehat G_p$ has
    \begin{align}
      T_p&=\operatorname{ran}\mathbf1_{(-\infty,E+\gamma/4]}(\widehat G_p)
          =\operatorname{ran}\mathbf1_{(-\infty,E+3\gamma/4]}(\widehat G_p),\notag\\
      \dim T_p&=\dim G_0,
    \end{align}
    and $T_p,G_0$ are mutually $3\tau/200$-close.

    Write $\Psi_J$ for the canonical unitary placing $J$ first, and
    set $T_{p,J}=\Psi_JT_p$. The original factory's filter, pulled
    back to the whole chain, is an operator $K_*$ with
    \begin{align}
      K_J=\Psi_JK_*\Psi_J^{-1}=\sum_{i=1}^n W_i\otimes R_i.
    \end{align}
    Its whole-chain matrix is unchanged. The conjugated operator $K_J$ preserves
    $T_{p,J}$ and its orthogonal complement, has minimum gain one
    on $T_{p,J}$, and has squared contraction at most $\Delta$ on
    $T_{p,J}^\perp$.
    For every $\nu\ge0$, $\delta<1$, and $\delta$-viable $S$ for
    $T_{p,J}$, the ordered-word image
    \begin{align}
      S_\nu=\sum_{w\in\{1,\ldots,n\}^\nu}W_{w_1}\cdots W_{w_\nu}S
    \end{align}
    has dimension at most $n^\nu\dim S$ and is
    $\Delta^\nu/(1-\delta)^2$-viable for $T_{p,J}$.
    No viability assertion is transferred directly between distinct
    tensor splits.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:uniform_physical_soft_filter_family,
      thm:common_collar_mpo_tensor_family,thm:precenter_collar_count,
      lem:common_collar_filter_words}
    Take the inward cuts of $J$. Their two added collars have total
    length at most $2\ell$, and they give
    (\ref{eq:uniform_long_soft_middle_collars}). Use the natural
    physical construction once on $J_0$, keeping its attained minima
    and its one common family before all three center estimates.
    Enlarge that family once by the collar matrix units.
    Its count is at most $Dq^{2|J\setminus J_0|}\le D^2$,
    and the member bond bound changes only by the displayed maximum.

    For each triple, keep the represented matrix and its own spectral
    target from the same construction. Pull its original filter back
    to the whole chain by the inverse canonical unitary. Its actual
    whole-matrix decomposition supplies the old-region input of
    Lemma~\ref{lem:common_collar_filter_words}. That lemma derives the
    decomposition, operator bounds and word estimate on $J$ using the
    already chosen enlarged family. The target rank and both original
    comparisons are unchanged throughout.
\end{proof}

\begin{lemma}[One-step improvement in the native regional coordinates]
    \label{lem:uniform_native_long_interval_soft_improving_family}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_uniform_native_long_interval_soft_improving_family}
    \leanok
    Under the hypotheses of
    Lemma~\ref{lem:uniform_long_interval_soft_improving_family}, retain
    the same attained regional minima, common minimizing vector,
    residual energy, and family $W_1,\ldots,W_n$ before all three
    admissible center estimates. The count and member bond bounds
    are unchanged. For each center, retain the same represented
    Hermitian matrix $\widehat G_p$, its own global spectral target
    $T_p$, its rank and both comparisons with the original ground space.

    Let $\mathcal R_J$ be the canonical retained-region/exterior unitary,
    which sends the coordinate vector at a configuration $\sigma$ to
    $e_{\sigma|_J}\otimes e_{\sigma|_{J^c}}$, and put
    $T_{p,J}=\mathcal R_JT_p$.
    The same whole-chain filter and exterior coefficients satisfy
    \begin{align}
      K_J=\mathcal R_JK_*\mathcal R_J^{-1}
          =\sum_{i=1}^n W_i\otimes R_i,
    \end{align}
    with target and complement invariant, minimum gain one on the
    target, and squared contraction at most $\Delta$ on its complement.
    For every $\delta$-viable regional subspace $S$ for $T_{p,J}$,
    where $\delta<1$, define the literal sum of images
    \begin{align}
      \mathcal V(S)=\sum_{i=1}^n W_iS.
    \end{align}
    Then $\dim\mathcal V(S)\le n\dim S$ and
    $\mathcal V(S)$ is $\Delta/(1-\delta)^2$-viable for $T_{p,J}$.
    All global matrix identities and the two individual middle-collar
    bounds of the preceding lemma are retained.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:uniform_long_interval_soft_improving_family}
    Apply the preceding physical family construction once.
    Its exterior-first coordinate expression followed by factor
    exchange equals $\mathcal R_J$: both send each coordinate vector
    to the same tensor of its two restrictions. Express the original
    filter and target in these identical coordinates, keeping the
    whole-chain operator and all coefficient matrices unchanged.

    The tensor decomposition places the filtered extension of $S$
    inside $\mathcal V(S)$ tensored with the full exterior.
    Minimum gain on the target and contraction on its complement
    therefore give the stated one-step error bound. The sum of $n$
    image spaces, each of dimension at most $\dim S$, gives the
    dimension estimate. No additional approximant, filter or family
    is selected.
\end{proof}

\begin{lemma}[Physical families and original-ground histories on one interval]
    \label{lem:physical_soft_interval_history}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_physical_soft_interval_history}\leanok
    Assume the hypotheses of
    Lemma~\ref{lem:uniform_native_long_interval_soft_improving_family}
    on a chain of $N+1$ sites and a long interval $J=[p,r)$.
    Write $G$ for the original ground space at a spectral minimum $\mu$,
    separated from the remaining spectrum by at least $g$.
    Suppose $D\ge2$ and $K\ge1$, and assume the numerical sampling bound
    \begin{align}
      K(\rho'+1)(s+2)^2\le Q\quad\Longrightarrow\quad
      4D^2\le Q,\qquad
      \operatorname{Fail}(D,Q,\rho')\le\frac{1}{q(s+2)^4}
      \label{eq:physical_interval_sampling_threshold}
    \end{align}
    for all natural $\rho',s,Q$. Assume also
    \begin{align}
      D^{12}\Delta\le10^{-5},\qquad
      \frac{3\tau}{200}\le
        \frac1{100}\left(\frac{3}{100D^2}\right)^2,
      \qquad \frac{3\tau}{200}\le\frac1{200}.
    \end{align}
    Fix an original-ground reference $R_0$ that is mutually
    $\varepsilon_0$-close to $G$, where
    $0\le\varepsilon_0\le10^{-5}$. Suppose its full-exterior Schmidt ranks
    across every internal cut of $J$ are at most $b_0$.
    Let $\zeta>0$ satisfy
    $\zeta\sqrt{(r-p-1)b_0}\le10^{-3}$, let $\dim G\le\rho$,
    and choose a number of rounds $R$ such that
    $q^{|J|}\le2^R$.

    The same five attained regional minima, common minimizing vector,
    residual energy and common family $W_1,\ldots,W_n$ may be chosen
    with the count and member bond bounds of the preceding lemma.
    These choices precede all admissible center triples, all regional
    candidates and all sample sequences. Put
    $\theta=K(\rho+1)(N+3)^2$. For each candidate $S$ that is
    $0.04$-viable for $G$, use the actual stopped half-sampling iteration
    with threshold $\theta$, trimming radius $\zeta$, and this family.
    Give its $R$ seeds the product Haar law $\mathbb P$.
    Let $\mathcal H_S$ be the event that every actual state through
    round $R$ remains $0.04$-viable for $G$ and every state immediately
    after an active step is $0.015$-viable for $G$. Let $V_S$ be its terminal
    parent, and let $Y_S$ be its last untrimmed family image.
    Then $\mathcal H_S$ and the true $0.015$-viability event for $V_S$
    are measurable, and
    \begin{align}
      \mathbb P(\mathcal H_S^c)&\le\frac{R}{q(N+3)^4},&
      \mathbb P\{V_S\text{ is not }0.015\text{-viable for }G\}
        &\le\frac{R}{q(N+3)^4}.
    \end{align}
    On $\mathcal H_S$, the same $Y_S$ is $27/2500$-viable for $G$
    and the same $V_S$ is $3/200$-viable for $G$.

    For each admissible center triple, its represented Hermitian matrix
    and own global target $T$ may be fixed before all candidates and
    sample sequences. They retain the matrix approximation bounds,
    target rank $\dim T=\dim G$, both $3\tau/200$ comparisons with $G$,
    the equality of the two closed spectral cutoffs and
    $E+\sum_j e_j=\mu$. For every $S$ and every sequence in the same
    event $\mathcal H_S$, its same last untrimmed image $Y_S$ is
    $\Delta/(91/100)^2$-viable for $T$ in the canonical regional
    coordinates. Both individual middle-collar bounds are retained.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:uniform_native_long_interval_soft_improving_family,
      lem:fixed_interval_raw_history_probability}
    Choose the native physical family once. For a fixed admissible
    center triple, retain its represented matrix and target, and use
    the preceding one-step image estimate as the improvement input to
    Lemma~\ref{lem:fixed_interval_raw_history_probability}.
    Its other inputs are exactly the reference and numerical bounds
    stated here. Apply that lemma with a single target.

    The history event depends only on $G$, the common family, the
    candidate and the fixed trimming radius. It therefore agrees for
    every center triple. The choice of the represented matrix remains
    before every candidate and sample sequence. In particular, the
    admissible triple $p_j=e_j$ gives the standalone original-ground
    history and parent probability bounds. Applying the same argument
    to any other triple gives its target estimate on that identical
    event; no intersection over targets is used for measurability.
\end{proof}
