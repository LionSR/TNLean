\section{Cumulative span}

The exact-word identities used in this section are proved in
Section~\ref{sec:app_wielandt_exact_word_support}. The stabilization and
spectral linear algebra behind the cumulative estimates are collected in
Section~\ref{sec:app_wielandt_cumulative_support}.


\begin{definition}[Word span]\label{def:word_span}
    \lean{Kraus.wordSpan}
    \leanok
    \uses{def:eval_word}
    Following~\cite[Section~II]{Sanz2010Quantum}, the \emph{word span} at
    length~$n$ is the subspace
    \begin{align}
        S_n(A)
         & =\spn_{\C}\{A^w:|w|=n\}\subseteq\MN{D}.
        \notag
    \end{align}
\end{definition}

\begin{definition}[Cumulative span]\label{def:cumulative_span}
    \lean{Kraus.cumulativeSpan}
    \leanok
    \uses{def:word_span}
    The \emph{cumulative span} at level~$n$ is
    $T_n(A)=S_0(A)+S_1(A)+\cdots+S_n(A)$.
\end{definition}


\begin{definition}[Fixed-length vector span of an MPS tensor]
    \label{def:mps_vector_spread_span}
    \lean{Kraus.vectorSpreadSpan}
    \leanok

    For an MPS tensor $A$, its fixed-length vector span is the fixed-length
    vector span of the underlying finite matrix family.
\end{definition}

\subsection{Paper primitivity and indices}

For the quantitative bounds, write $d'=\dim S_1(A)$ for the number of
linearly independent Kraus operators. The source definitions and
Proposition~3 of~\cite{Sanz2010Quantum} distinguish uniform vector spreading,
eventual exact-word spanning, and spectral strong irreducibility.

\begin{definition}[Kraus rank]\label{def:wielandt_kraus_rank}
    \lean{MPSTensor.krausRank}
    \leanok
    \uses{def:word_span}
    The \emph{Kraus rank} is
    $\krausrk(A)=\dim S_1(A)$.
\end{definition}

\begin{definition}[Eventually full Kraus rank]
    \label{def:eventually_full_kraus_rank}
    \lean{MPSTensor.HasEventuallyFullKrausRank}
    \leanok
    \uses{def:word_span}
    The Kraus family has \emph{eventually full Kraus rank} if there is a
    positive integer $i$ such that $S_i(A)=\MN{D}$.
\end{definition}

\begin{definition}[Paper primitivity]\label{def:paper_primitive}
    \lean{MPSTensor.IsPrimitivePaper}
    \leanok

    The transfer map is \emph{primitive in the sense of
        \cite{Sanz2010Quantum}} if there is a positive integer $q$ such that
    \begin{align}
        H_q(A,\varphi)
         & =\C^D
        \label{eq:wld_paper_primitive_spread}
    \end{align}
    for every nonzero $\varphi\in\C^D$. The same $q$ works for all
    nonzero vectors.
\end{definition}

\begin{definition}[Full-Kraus-rank index]\label{def:wielandt_i_index}
    \lean{MPSTensor.iIndex}
    \leanok
    \uses{def:eventually_full_kraus_rank}
    The \emph{full-Kraus-rank index} is
    \begin{align}
        \iota(A)
         & =\min\{n\ge1:S_n(A)=\MN{D}\}.
        \notag
    \end{align}
    If the set is empty, its value is defined to be zero.
\end{definition}

\begin{definition}[Primitivity index]\label{def:wielandt_q_index}
    \lean{MPSTensor.qIndex}
    \leanok
    \uses{def:paper_primitive}
    The \emph{primitivity index} is
    \begin{align}
        q(\E_A)
         & =\min\{q\ge1:H_q(A,\varphi)=\C^D
        \text{ for every }\varphi\neq0\}.
        \notag
    \end{align}
    If the set is empty, its value is defined to be zero.
\end{definition}

\begin{definition}[Strengthened strong irreducibility]
    \label{def:strongly_irreducible_paper}
    \lean{MPSTensor.IsStronglyIrreduciblePaper}
    \leanok
    \uses{def:transfer_map}
    An MPS tensor $A$ is \emph{strongly irreducible} if there exists a matrix
    $\rho>0$ such that
    \begin{align}
        \E_A(\rho)                 & =\rho, \notag \\
        \spec_{\mathrm{per}}(\E_A) & =\{1\},
        \notag
    \end{align}
    and $\E_A$ is irreducible. This formulation includes explicit irreducibility
    alongside the peripheral spectrum and fixed-point condition of
    Sanz et al.~\cite[Proposition~3(c)]{Sanz2010Quantum}; see
    \cite{gap:spwc10_wielandt_one_step_subspace}.
\end{definition}

The next seven results use the following notation. For nonnegative integers
$d$ and $D$, let $K=\{K^i\}_{i=0}^{d-1}$ be a finite family of $D\times D$
matrices, and let $\E_K$ be its Kraus map.


\begin{lemma}[Powers of a transfer-map eigenmatrix]
    \label{thm:transfer-map-eigenmatrix-power}
    \lean{Module.End.pow_apply_of_mem_eigenspace}
    \leanok
    \uses{def:transfer_map}
    If $\E_A(X)=\mu X$, then $\E_A^n(X)=\mu^nX$ for every $n\geq0$.
\end{lemma}

\begin{proof}\leanok
    Specializing the companion quantum-channel result~\cite[``Powers on an eigenspace'']{QICLean2026Blueprint}
    to the eigenspace of $\mu$ gives $\E_A^n(X)=\mu^nX$.
\end{proof}


\begin{lemma}[Hermitian power-fixed point from a peripheral eigenvector]
    \label{thm:hermitian-power-fixed-point}
    \lean{MPSTensor.exists_hermitian_ne_zero_trace_zero_pow_fixedPoint}
    \leanok
    \uses{def:transfer_map}
    Let $A$ be a normalized MPS tensor, and suppose
    $\E_A(X)=\mu X$ with $X\neq0$, $\mu\neq1$, and $\mu^p=1$. Then there is a
    nonzero Hermitian matrix $H$ such that
    \begin{align}
        \tr(H)
         & =0,
        \notag \\
        \E_A^p(H)
         & =H.
        \notag
    \end{align}
    In addition, $H$ is not positive semidefinite.
\end{lemma}

\begin{proof}\leanok
    Applying
    Theorem~\cite[``Hermitian power-fixed point from a finite-order eigenmatrix'']{QICLean2026Blueprint}
    to the transfer channel produces a nonzero Hermitian matrix of trace zero
    fixed by $\E_A^p$; since a nonzero positive-semidefinite matrix has
    positive trace, this matrix is not positive semidefinite.
\end{proof}

\begin{lemma}[Uniqueness of power-fixed positive matrices]
    \label{thm:mps-power-fixed-point-unique-from-spreading}
    \lean{Kraus.posSemidef_pow_fixedPoint_unique}
    \leanok
    \uses{def:transfer_map}
    Suppose there is a length $q\geq0$ such that
    $H_q(A,\varphi)=\C^D$ for every nonzero $\varphi\in\C^D$. Let $p>0$.
    If $\rho$ and $\sigma$ are nonzero positive-semidefinite fixed points of
    $\E_A^p$, then $\sigma=c\rho$ for some $c\in\C$.
\end{lemma}

\begin{proof}\leanok
    Applying
    Theorem~\cite[``Uniqueness of power-fixed positive matrices for a Kraus map'']{QICLean2026Blueprint}
    to the Kraus family formed by the matrices of $A$ yields $\sigma=c\rho$.
\end{proof}


\begin{lemma}[Kraus-span equivalence]%
    \label{thm:wolf_6_8_kraus_span}
    \lean{MPSTensor.primitivePaper_iff_hasEventuallyFullKrausRank}

\end{lemma}

\end{remark}

\subsection{Different indices for the three-Pauli channel}

\begin{definition}[Three-Pauli channel]
    \label{def:wielandt_three_pauli_channel}
    \lean{MPSTensor.threePauliTensor}
    \leanok
    The three-Pauli channel is the qubit channel with Kraus matrices
    $A_x=\sigma_x/\sqrt3$, $A_y=\sigma_y/\sqrt3$ and
    $A_z=\sigma_z/\sqrt3$, where
    \begin{align}
        \sigma_x & =\begin{pmatrix}0&1\\1&0\end{pmatrix},                  &
        \sigma_y & =\begin{pmatrix}0&-\mathrm i\\\mathrm i&0\end{pmatrix}, &
        \sigma_z & =\begin{pmatrix}1&0\\0&-1\end{pmatrix}.
    \end{align}
    This is the example following Proposition~2 of~\cite{Sanz2010Quantum}.
\end{definition}

\begin{lemma}[The three-Pauli channel is trace preserving]
    \label{lem:wielandt_three_pauli_trace_preserving}
    \lean{MPSTensor.threePauliTensor_isTP}
    \leanok
    \uses{def:wielandt_three_pauli_channel}
    The Kraus matrices of the three-Pauli channel satisfy
    $\sum_{k=x,y,z}A_k^\dagger A_k=\Id$.
\end{lemma}
\begin{proof}\leanok
    Each Pauli matrix is Hermitian and squares to the identity, so
    $A_k^\dagger A_k=\Id/3$ for $k=x,y,z$.
\end{proof}

\begin{lemma}[One-step spreading and two-step matrix spanning]
    \label{lem:wielandt_three_pauli_spans}
    \lean{MPSTensor.threePauliTensor_vectorSpreadSpan_one,
        MPSTensor.threePauliTensor_wordSpan_one_ne_top,
        MPSTensor.threePauliTensor_wordSpan_two}
    \leanok
    \uses{def:wielandt_three_pauli_channel, def:word_span, def:mps_vector_spread_span}
    For every $0\ne\varphi\in\mathbb C^2$, one has
    $H_1(A,\varphi)=\mathbb C^2$. Nevertheless $S_1(A)\ne M_2(\mathbb C)$,
    whereas $S_2(A)=M_2(\mathbb C)$.
\end{lemma}
\begin{proof}\leanok
    Write $\varphi=(a,b)$. The combinations
    $(\sigma_x+\mathrm i\sigma_y)\varphi=(2b,0)$ and
    $(\sigma_x-\mathrm i\sigma_y)\varphi=(0,2a)$, together with
    $\sigma_z\varphi=(a,-b)$, span both coordinate axes since $a$ and $b$
    do not both vanish. The common nonzero scalar $1/\sqrt3$ does not
    change this span. Every one-letter matrix has trace zero, so their span
    excludes the identity. The two-letter products satisfy
    \begin{align}
        \sigma_x^2=\sigma_y^2=\sigma_z^2=\Id,\quad
        \sigma_x\sigma_y=\mathrm i\sigma_z,\quad
        \sigma_y\sigma_z=\mathrm i\sigma_x,\quad
        \sigma_z\sigma_x=\mathrm i\sigma_y,
        \notag
    \end{align}
    so $\Id,\sigma_x,\sigma_y,\sigma_z\in S_2(A)$, and these four matrices
    span $M_2(\mathbb C)$.
\end{proof}

\begin{lemma}[Strict inequality between the two channel indices]
    \label{thm:wielandt_three_pauli_indices}
    \lean{MPSTensor.threePauliTensor_qIndex, MPSTensor.threePauliTensor_iIndex,
        MPSTensor.threePauliTensor_qIndex_lt_iIndex}
    \leanok
    \uses{lem:wielandt_three_pauli_spans, def:wielandt_q_index, def:wielandt_i_index}
    The three-Pauli channel satisfies $q(\mathcal E_A)=1<2=\iota(A)$.
    Thus the comparison of Proposition~1 in~\cite{Sanz2010Quantum} can be strict.
\end{lemma}
\begin{proof}\leanok
    The least positive spreading length is one by
    Lemma~\ref{lem:wielandt_three_pauli_spans}. The least positive
    full-matrix-span length is two, since the length-one span is proper
    and the length-two span is full.
\end{proof}

\begin{lemma}[Eventual full Kraus rank gives paper primitivity]
    \label{thm:paper_primitive_of_full_kraus_rank}
    \lean{MPSTensor.isPrimitivePaper_of_hasEventuallyFullKrausRank}
    \leanok
    \uses{def:eventually_full_kraus_rank, def:paper_primitive}
    If $S_i(A)=\MN{D}$ for some positive $i$, then $H_i(A,\varphi)=\C^D$ for
    every $\varphi\neq0$, so $A$ is primitive in the sense of
    Definition~\ref{def:paper_primitive}.  No normalization is imposed on
    $A$.
\end{lemma}

\begin{proof}
    \leanok
    Applying every matrix of $\MN{D}$ to a nonzero vector $\varphi$ gives all
    of $\C^D$, and $S_i(A)\varphi$ is the span of the length-$i$ words
    applied to $\varphi$.
\end{proof}

\begin{lemma}[Paper primitivity gives strong irreducibility]
    \label{thm:strongly_irreducible_of_paper_primitive}
    \lean{MPSTensor.isStronglyIrreduciblePaper_of_isPrimitivePaper}
    \leanok
    \uses{def:paper_primitive, def:strongly_irreducible_paper}
    Let $D>0$ and suppose $\sum_i(A^i)^\dagger A^i=\Id$.  If $A$ is primitive
    in the sense of Definition~\ref{def:paper_primitive}, then $\E_A$ is
    strongly irreducible.
\end{lemma}

\begin{proof}
    \leanok
    For the finite Kraus family of $A$,
    Theorems~\cite[``Positive-definite fixed point from vector spreading'']{QICLean2026Blueprint}
    and~\cite[``Primitivity from vector spreading'']{QICLean2026Blueprint}
    give a positive-definite fixed point and show that $1$ is the only
    peripheral eigenvalue, and
    Theorem~\cite[``Irreducibility from vector spreading'']{QICLean2026Blueprint}
    gives irreducibility of $\E_A$.  Together these are the three conditions
    of Definition~\ref{def:strongly_irreducible_paper}.
\end{proof}

\begin{lemma}[Strengthened form of the primitivity equivalence]
    \label{thm:wielandt_proposition_three}
    \lean{MPSTensor.primitivePaper_iff_stronglyIrreducible}
    \lean{MPSTensor.hasEventuallyFullKrausRank_iff_stronglyIrreducible}
    \leanok
    \uses{def:paper_primitive, def:eventually_full_kraus_rank, def:strongly_irreducible_paper}
    Let $D>0$ and suppose
    $\sum_i(A^i)^\dagger A^i=\Id$. The following are equivalent:
    \begin{enumerate}
        \item there is a positive $q$ for which
              $H_q(A,\varphi)=\C^D$ for every $\varphi\neq0$;
        \item there is a positive $i$ for which $S_i(A)=\MN{D}$;
        \item $\E_A$ is strongly irreducible.
    \end{enumerate}
    Proposition~3 of~\cite{Sanz2010Quantum} states the same equivalence with
    item~(3) replaced by the printed condition in the source: $1$ is the only
    peripheral eigenvalue, and its one-dimensional eigenspace is generated by
    a positive-definite fixed point. Here item~(3) includes the additional
    irreducibility condition.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:paper_primitive_of_full_kraus_rank,
        thm:strongly_irreducible_of_paper_primitive}
    Lemma~\ref{thm:paper_primitive_of_full_kraus_rank} is
    (2)$\Rightarrow$(1), and
    Lemma~\ref{thm:strongly_irreducible_of_paper_primitive} is
    (1)$\Rightarrow$(3).

    For (3)$\Rightarrow$(2), the power decomposition
    $\E_A^m=P_\rho+(\E_A-P_\rho)^m$ has complementary powers tending to zero.
    If $S_m(A)\neq\MN{D}$ for every $m$, a nonzero matrix $B$ orthogonal to
    $S_m(A)$ has vanishing trace pairing on the left, while positive
    definiteness gives a lower bound $c\lVert B\rVert^2$ for the $P_\rho$ term
    and the complementary term is $o(\lVert B\rVert^2)$, a contradiction for
    large $m$.
\end{proof}

\begin{lemma}[The simple peripheral eigenspace criterion]
    \label{thm:wielandt_literal_peripheral_criterion}
    \lean{MPSTensor.isStronglyIrreduciblePaper_of_simple_peripheral_fixedPoint,
        MPSTensor.primitivePaper_iff_exists_simple_peripheral_fixedPoint,
        MPSTensor.hasEventuallyFullKrausRank_iff_exists_simple_peripheral_fixedPoint}
    \leanok
    \uses{def:paper_primitive, def:eventually_full_kraus_rank,
        def:strongly_irreducible_paper, thm:wielandt_proposition_three,
        thm:wolf_6_8_kraus_span, thm:strongly_irreducible_of_paper_primitive,
        thm:primitive_from_strongly_irred}
    Let $D>0$ and suppose $\sum_i(A^i)^\dagger A^i=\Id$.
    The following conditions are equivalent:
    \begin{enumerate}
        \item $\E_A$ is primitive.
        \item The exact-length word spaces eventually equal $\MN{D}$.
        \item There is $\rho>0$ such that $\E_A(\rho)=\rho$, the fixed
              eigenspace is $\mathbb C\rho$, and $1$ is the only peripheral
              eigenvalue.
    \end{enumerate}
    In the third condition, irreducibility is a conclusion rather than
    an additional hypothesis. This is \cite[Proposition~3]{Sanz2010Quantum}.
\end{lemma}
\begin{proof}\leanok
    For a primitive channel, the positive-definite fixed point and the
    complementary spectral gap give the third condition. Conversely, assume
    the third condition and let $P\neq0,\Id$ be an orthogonal projection with
    $\E_A(P\MN{D}P)\subseteq P\MN{D}P$. The restriction of $\E_A$ to this
    corner is positive and trace preserving, so it fixes a density matrix
    $\sigma=P\sigma P$. Simplicity of the fixed eigenspace gives
    $\sigma=c\rho$ with $c\neq0$, hence
    \begin{align}
        (\Id-P)\rho(\Id-P) & =c^{-1}(\Id-P)\sigma(\Id-P)=0,
        \notag
    \end{align}
    contradicting $\rho>0$. Thus $\E_A$ is irreducible, and together with
    peripheral primitivity this gives eventual full Kraus rank and hence
    primitivity.
\end{proof}

\begin{lemma}[Normalized normal tensors are strongly irreducible]
    \label{thm:normalized_normal_strongly_irreducible}
    \lean{MPSTensor.isStronglyIrreduciblePaper_of_hasEventuallyFullKrausRank,
        MPSTensor.isNormal_implies_stronglyIrreducible}
    \leanok
    \uses{def:eventually_full_kraus_rank, def:normal,
        def:strongly_irreducible_paper}
    Let $D>0$ and suppose $\sum_i(A^i)^\dagger A^i=\Id$.  If $A$ has
    eventually full Kraus rank, or equivalently is normal, then $\E_A$ is
    strongly irreducible.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:paper_primitive_of_full_kraus_rank,
        thm:strongly_irreducible_of_paper_primitive, rem:wolf_6_8_pairwise}
    Normality is eventual full Kraus rank by
    Remark~\ref{rem:wolf_6_8_pairwise}.
    Lemma~\ref{thm:paper_primitive_of_full_kraus_rank} then gives paper
    primitivity, and Lemma~\ref{thm:strongly_irreducible_of_paper_primitive}
    gives strong irreducibility.
\end{proof}

\begin{lemma}[Comparison of the two channel indices]
    \label{thm:q_index_le_i_index_channel}
    \lean{MPSTensor.qIndex_le_iIndex_of_tracePreserving}
    \leanok
    \uses{def:wielandt_i_index, def:wielandt_q_index, thm:wolf_6_8_kraus_span}
    Let $A_1,\ldots,A_d\in\MN{D}$, where $D>0$, satisfy
    $\sum_i A_i^\dagger A_i=\Id$. With both indices set to zero when
    their defining sets are empty, one has $q(\E_A)\le\iota(A)$.
    This is \cite[Proposition~1]{Sanz2010Quantum}.
\end{lemma}

\begin{proof}\leanok
    If $\iota(A)$ is a positive integer with $S_{\iota(A)}(A)=\MN{D}$, then
    $H_{\iota(A)}(A,\varphi)=\C^D$ for every $\varphi\neq0$, so $\iota(A)$
    lies in the defining set for $q(\E_A)$ and $q(\E_A)\le\iota(A)$.
    Otherwise, Lemma~\ref{thm:wolf_6_8_kraus_span} excludes
    primitivity, so
    $\{q\ge1:H_q(A,\varphi)=\C^D\text{ for every }\varphi\neq0\}=\emptyset$
    and $q(\E_A)=0\le\iota(A)$.
\end{proof}

\begin{lemma}[Primitive-channel specialization of the index bound]
    \label{thm:q_index_le_i_index}
    \lean{MPSTensor.qIndex_le_iIndex_of_isPrimitivePaper}
    \leanok
    \uses{def:wielandt_i_index, def:wielandt_q_index, thm:wielandt_proposition_three}
    As the primitive-channel specialization of the universal inequality
    $q(\E)\le\iota$ of Sanz et al.~\cite[Proposition~1]{Sanz2010Quantum} (see
    \cite{gap:spwc10_wielandt_one_step_subspace}), under the hypotheses of
    Lemma~\ref{thm:wielandt_proposition_three}, paper primitivity implies
    \begin{align}
        q(\E_A) & \le\iota(A).
        \notag
    \end{align}
\end{lemma}

\begin{proof}\leanok
    By Lemma~\ref{thm:wielandt_proposition_three}, $\iota(A)$ is a
    positive length with $S_{\iota(A)}(A)=\MN{D}$. Therefore
    $H_{\iota(A)}(A,\varphi)=\C^D$ for every $\varphi\neq0$, so the defining
    minimum for $q(\E_A)$ is at most $\iota(A)$.
\end{proof}

\begin{lemma}[Unique fixed points of full rank]
    \label{thm:wolf_6_15}
    \lean{MPSTensor.wolf_theorem_6_15}
    \leanok
    \uses{thm:wielandt_proposition_three}
    Following~\cite[Theorem~6.15]{Wolf2012Quantum}, let $D>0$ and suppose
    $\sum_i(A^i)^\dagger A^i=\Id$. If there is a positive $n$ for which
    $S_n(A)=\MN{D}$, then there is a unique positive definite density matrix
    $\rho\in\MN{D}$ such that every fixed point of $\E_A$ is a scalar multiple
    of $\rho$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:wielandt_proposition_three, lem:fixedpoint_unique_spectral}
    By Lemma~\ref{thm:wielandt_proposition_three}, the span hypothesis
    makes $\E_A$ strongly irreducible, giving a positive-definite fixed
    point $\rho_0$ together with peripheral primitivity (unique peripheral
    eigenvalue $1$) and irreducibility of $\E_A$. Peripheral primitivity and
    irreducibility of $\E_A$ together give the complementary transfer-map
    gap at $\rho_0$ (Lemma~\ref{lem:fixedpoint_unique_spectral}): the
    restriction of $\E_A$ off the fixed-point
    projection at $\rho_0$ has spectral radius strictly less than one. This
    gap forces every fixed point of $\E_A$ to be a scalar multiple of
    $\rho_0$; rescaling $\rho_0$ by the inverse of its (positive) trace
    gives the unique density-matrix representative $\rho$.

    The source establishes this by showing that $\E_A^n(\ket\psi\!\bra\psi)$
    has full rank for each pure state and ruling out multidimensional fixed
    spaces via Corollary~6.5 of~\cite{Wolf2012Quantum}. The present argument
    proceeds via the Proposition~3 equivalence
    (Lemma~\ref{thm:wielandt_proposition_three}).
\end{proof}

\begin{lemma}[Monotonicity of cumulative span]\label{lem:cumulative_mono}
    \lean{Kraus.cumulativeSpan_mono}
    \leanok
    \uses{def:cumulative_span}
    For every $n$, one has $T_n(A)\le T_{n+1}(A)$.
\end{lemma}

\begin{proof}\leanok\uses{def:cumulative_span}
    The span $T_{n+1}$ includes all words of length at most $n+1$ and
    therefore contains all words of length at most~$n$.
\end{proof}

\begin{lemma}[Dimension bound]\label{lem:cumulative_finrank_le}
    \lean{Kraus.cumulativeSpan_finrank_le}
    \leanok
    \uses{def:cumulative_span}
    For every $n$, one has $\dim T_n(A)\le D^2$.
\end{lemma}

\begin{proof}\leanok
    Since $T_n(A)\subseteq\MN{D}$ and $\dim\MN{D}=D^2$, the result follows.
\end{proof}

\begin{lemma}[Cumulative span stabilisation]\label{thm:cumulative_stable}
    \lean{Kraus.cumulativeSpan_stable}
    \leanok
    \uses{def:cumulative_span}
    If $T_n(A)=T_{n+1}(A)$, then $T_m(A)=T_n(A)$ for every $m\ge n$.
\end{lemma}

\begin{proof}
    \leanok
    If $T_n=T_{n+1}$, then $S_{n+1}\subseteq T_n$.
    Left-multiplying any element of $S_{n+1}$ by $A^i$ gives
    an element of $S_{n+2}$, so $S_{n+2}\subseteq T_{n+1}=T_n$.
    By induction, $S_m\subseteq T_n$ for every $m>n$, hence $T_m=T_n$.
\end{proof}

\begin{lemma}[Dimension growth]\label{thm:cumulative_strict_mono}
    \lean{Kraus.cumulativeSpan_finrank_strict_mono}
    \leanok
    \uses{def:cumulative_span}
    If $T_n(A)\neq T_{n+1}(A)$, then

\end{proof}

\begin{lemma}[Sharp nonzero trace product for a primitive channel]%

    with
    \begin{align}
        1\le|w|  & \le D^2-\krausrk(A)+1,
        \notag                            \\

        \notag
    \end{align}
\end{lemma}

\begin{proof}\leanok

    rank, hence normality. For $b=D^2-\krausrk(A)+1$, let

    $\dim\MN{D}$ gives $V_b=\MN{D}$. Since $\Id\in V_b$ is a linear
    combination of the positive-length words defining $V_b$ and
    $\tr(\Id)=D\neq0$, at least one such word has nonzero trace.
\end{proof}

\begin{definition}[Cumulative vector span]\label{def:cumulative_vector_span}

    \begin{align}
        K_n(A,\varphi)
         & =\spn_{\C}\{A^w\varphi:|w|\le n\}\subseteq\C^D.
        \notag
    \end{align}
\end{definition}

\begin{lemma}[Eigenvector spreading]\label{thm:eigenvector_spreading}
    \lean{MPSTensor.eigenvector_spreading}
    \leanok
    \uses{def:cumulative_vector_span, def:normal}
    Providing the cumulative dimension-growth step in
    \cite[Lemma~2(a)]{Sanz2010Quantum}, let $A$ be a normal MPS tensor and let
    $A^{i_0}\varphi=\mu\varphi$, where $\mu\neq0$ and $\varphi\neq0$.
    Then $K_{D-1}(A,\varphi)=\C^D$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:cumulative_eq_top, lem:vector_span_from_matrix, thm:vector_span_stable}
    The vector span $K_n$ is monotone and $\dim K_n\le D$.
    If $K_n=K_{n+1}$ for some $n<D-1$, the span stabilises by
    Lemma~\ref{thm:vector_span_stable}.
    But $T_{D^2}(A)=\MN{D}$ by Lemma~\ref{thm:cumulative_eq_top}, so
    Lemma~\ref{lem:vector_span_from_matrix} gives
    $K_{D^2}(A,\varphi)=\C^D$, contradicting early stabilisation.
    Hence $\dim K_n$ grows by at least~$1$ at each step, reaching~$D$ by
    step $D-1$.
\end{proof}

\begin{lemma}[Rank-one extraction from a non-invertible eigenvector]%
    \label{lem:rank_one_noninvertible_eigenvector}
    \lean{MPSTensor.vecMulVec_eigenvector_exact_wordSpan,
        MPSTensor.rectSpan_nilpIndex_strict_growth_of_isNormal}
    \leanok
    \uses{def:word_span, def:normal}
    Let $A$ be a normal MPS tensor with bond dimension $D\ge1$.
    Suppose that a Kraus operator $A^{i_0}$ is non-invertible and that
    there exist a nonzero vector $\varphi\in\C^D$ and a nonzero scalar $\mu$
    such that $A^{i_0}\varphi=\mu\varphi$.
    Then, for every $\psi\in\C^D$,
    $\ketbra{\varphi}{\psi}\in S_{D^2-D+1}(A)$.
\end{lemma}

\begin{proof}
    \leanok
    Let $\mathcal L(M)=A^{i_0}M$, and choose $r$ so that
    $\mathcal L^r$ annihilates the generalized zero-eigenspace of $\mathcal L$;
    equivalently, $\range(\mathcal L^r)$ is its stabilized nonzero-spectral
    part. The dimension-growth argument gives a level $n_0$ for which
    \begin{align}
        (A^{i_0})^rS_{n_0}(A)
         & =\{(A^{i_0})^rM:M\in\MN{D}\}.
        \notag
    \end{align}
    Since $A^{i_0}\varphi=\mu\varphi$ and $\mu\neq0$, iterating gives
    $(A^{i_0})^r\varphi=\mu^r\varphi$ and hence
    $\varphi=\mu^{-r}(A^{i_0})^r\varphi\in\range((A^{i_0})^r)$.
    Therefore, for every $\psi\in\C^D$,
    $\ketbra{\varphi}{\psi}\in S_{r+n_0}(A)$.
    The corresponding Fitting-index estimate gives
    $r+n_0\le D^2-D+1$.
    Since
    \begin{align}
        A^{i_0}\ketbra{\varphi}{\psi}
         & =\ketbra{A^{i_0}\varphi}{\psi}
        =\mu\ketbra{\varphi}{\psi},
        \notag
    \end{align}
    one has
    $\ketbra{\varphi}{\psi}
        =\mu^{-1}A^{i_0}\ketbra{\varphi}{\psi}$.
    Thus
    $\ketbra{\varphi}{\psi}\in S_k(A)$ implies
    $\ketbra{\varphi}{\psi}\in S_{k+1}(A)$, because left multiplication
    by $A^{i_0}$ maps $S_k(A)$ into $S_{k+1}(A)$.
    Iterating from $k=r+n_0$ and using $r+n_0\le D^2-D+1$ gives
    $\ketbra{\varphi}{\psi}\in S_{D^2-D+1}(A)$.
\end{proof}

\begin{theorem}[Wielandt's inequality --- general bound]%
    \label{thm:wielandt_general}
    \lean{MPSTensor.iIndex_le_general_of_isPrimitivePaper}
    \leanok
    \uses{def:paper_primitive, def:wielandt_i_index, def:wielandt_kraus_rank}

    normalized MPS tensor with $D>0$, $\sum_i(A^i)^\dagger A^i=\Id$, that is
    primitive in the sense of Definition~\ref{def:paper_primitive}.
    Then
    \begin{align}
        \iota(A)
         & \le(D^2-\krausrk(A)+1)D^2,
        \notag
    \end{align}
    where $\iota(A)=\min\{n\ge1:S_n(A)=\MN{D}\}$ is the
    full-Kraus-rank index.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:wielandt_proposition_three, thm:nonzero_trace_word_primitive_sharp, thm:isNormal_blockTensor, thm:wielandt_case2_normal_generator, cor:wielandt_case3_generator}
    Lemma~\ref{thm:wielandt_proposition_three} gives normality.
    By Lemma~\ref{thm:nonzero_trace_word_primitive_sharp}, there exists a
    \emph{positive-length} word~$w$ with
    $|w|\le D^2-d'+1$ and $\tr(A^w)\neq0$, where
    $d'=\krausrk(A)$.
    The matrix $A^w$ has nonzero trace, hence a nonzero eigenvalue~$\mu$
    and eigenvector~$\varphi$ by
    Theorem~\cite[``Nonzero trace implies eigenvector'']{QICLean2026Blueprint}.
    For $n=|w|$, the blocked tensor $B=A^{[n]}$ is normal by
    Lemma~\ref{thm:isNormal_blockTensor}, and encoding $w$ as a single block
    index gives $B^{i_0}=A^w$.
    \begin{enumerate}
        \item If $A^w$ is invertible, then
              $S_{D^2}(B)=\MN{D}$, whence
              $S_{D^2n}(A)=\MN{D}$.
        \item If $A^w$ is non-invertible, then the eigenvector $\varphi$
              spans a proper subspace, and the rank-one extraction argument
              gives $S_{D^2}(B)=\MN{D}$, whence
              $S_{D^2n}(A)=\MN{D}$.
    \end{enumerate}
    In the invertible case, the invertible Kraus operator $B^{i_0}$ allows
    direct dimension growth, giving
    $S_{D^2-k_B+1}(B)=\MN{D}$, where $k_B=\krausrk(B)$.
    In the non-invertible case, the bound $S_{D^2}(B)=\MN{D}$ follows from
    the rank-one extraction argument of \cite[Lemma~2(b)]{Sanz2010Quantum}.
    In both cases, $\iota(A)\le D^2n\le D^2(D^2-d'+1)$.
\end{proof}

\begin{remark}[Quantitative and qualitative Wielandt bounds]%
    \leanok
    The quantitative bound $\iota(A)\le(D^2-d'+1)D^2$
    (\cite[Theorem~1, case~(1)]{Sanz2010Quantum}) gives an explicit upper bound
    on the injectivity length. For the Fundamental Theorem of MPS, the
    qualitative assertion that $S_N(A)=\MN{D}$ for some finite $N$ suffices.
\end{remark}

\subsection{The cyclic lower-bound example}

\begin{definition}[Cyclic shift and jump family]
    \label{def:cyclic_wielandt_family}
    \lean{MPSTensor.CyclicWielandt.shift, MPSTensor.CyclicWielandt.jump,
        MPSTensor.CyclicWielandt.tensor,
        MPSTensor.CyclicWielandt.tensor_zero, MPSTensor.CyclicWielandt.tensor_one}
    \leanok
    For $D\ge2$, let $S=\sum_{i=0}^{D-1}\ketbra{i+1}{i}$, with indices
    modulo $D$, and $R=\ketbra{1}{D-1}$. The family $A=\{S,R\}$ is the
    cyclic example following the main theorem in~\cite{Sanz2010Quantum}.
\end{definition}

\begin{lemma}[Cyclic matrix identities]
    \label{thm:cyclic_wielandt_identities}
    \lean{MPSTensor.CyclicWielandt.shift_mul_apply,
        MPSTensor.CyclicWielandt.shift_pow_apply,
        MPSTensor.CyclicWielandt.shift_pow_dim,
        MPSTensor.CyclicWielandt.shift_pow_mul_dim,
        MPSTensor.CyclicWielandt.shift_pow_mod,
        MPSTensor.CyclicWielandt.shift_isUnit,
        MPSTensor.CyclicWielandt.shift_pow_mul_single,
        MPSTensor.CyclicWielandt.single_mul_shift_pow,
        MPSTensor.CyclicWielandt.sandwich_eq_single,
        MPSTensor.CyclicWielandt.return_coordinate_iff,
        MPSTensor.CyclicWielandt.one_add_last,
        MPSTensor.CyclicWielandt.jump_shift_pow_jump,
        MPSTensor.CyclicWielandt.jump_return}
    \leanok
    \uses{def:cyclic_wielandt_family}
    The shift satisfies $S^D=I$ and is invertible. For $0\le k<D$,
    $RS^kR=R$ if $k=D-2$ and $RS^kR=0$ otherwise. For all $k,l\ge0$,
    $S^kRS^l=\ketbra{1+k}{D-1-l}$, with coordinates modulo $D$.
\end{lemma}

\begin{proof}\leanok
    The entries of $S^n$ are $\delta_{i,j+n}$. Multiplication by a
    matrix unit then gives the two sandwich identities.
\end{proof}

\begin{lemma}[Exact cyclic word normal form]
    \label{thm:cyclic_wielandt_word_normal_form}
    \lean{MPSTensor.CyclicWielandt.evalWord_normalForm}
    \leanok
    \uses{def:cyclic_wielandt_family}
    Every word of length $L$ evaluates to zero, to $S^L$, or to
    $S^kRS^l$ for some $0\le k,l<D$ and $n,m\ge0$ satisfying
    $L=1+k+l+n(D-1)+mD$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:cyclic_wielandt_identities}
    Induct by prepending a letter. Prepending $S$ increments $k$,
    carrying a full turn into $m$. Prepending $R$ to a shifted jump
    survives precisely when $k=D-2$, resets $k$ to zero, and increases
    $n$ by one. Prepending $R$ to a pure shift reduces its exponent
    modulo $D$.
\end{proof}

\begin{lemma}[Full span at the cyclic length]
    \label{thm:cyclic_wielandt_full_span}
    \lean{MPSTensor.CyclicWielandt.jump_mem_wordSpan,
        MPSTensor.CyclicWielandt.sandwich_mem_wordSpan,
        MPSTensor.CyclicWielandt.exists_full_length_padding,
        MPSTensor.CyclicWielandt.wordSpan_full_length}
    \leanok
    \uses{def:cyclic_wielandt_family, def:word_span}
    At $N=D(D-1)$, the cyclic family satisfies $S_N(A)=\MN{D}$.
    The omitted diagonal-unit step in the source proof is supplied as
    recorded in~\cite{gap:spwc10_cyclic_missing_diagonal}.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:cyclic_wielandt_identities}
    Return blocks give $R\in S_{1+n(D-1)}(A)$ and full turns give
    $I\in S_{mD}(A)$. For $s=k+l\le D-2$, choose $n=s+1$ and
    $m=D-s-2$. For $D-1\le s\le2D-3$, choose $n=s+1-D$ and
    $m=2D-s-3$. Thus every matrix unit except $\ketbra{0}{0}$ belongs
    to $S_N(A)$. Since $S^N=I$, subtraction of the other diagonal units
    gives $\ketbra{0}{0}\in S_N(A)$.
\end{proof}

\begin{lemma}[The predecessor entry obstruction]
    \label{thm:cyclic_wielandt_predecessor}
    \lean{MPSTensor.CyclicWielandt.evalWord_pred_entries_eq,
        MPSTensor.CyclicWielandt.wordSpan_pred_entries_eq,
        MPSTensor.CyclicWielandt.wordSpan_pred_ne_top}
    \leanok
    \uses{def:cyclic_wielandt_family, def:word_span}
    Every $X\in S_{D(D-1)-1}(A)$ satisfies $X_{D-1,0}=X_{0,1}$.
    Consequently $S_{D(D-1)-1}(A)\ne\MN{D}$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:cyclic_wielandt_word_normal_form,
        thm:cyclic_wielandt_identities}
    A shifted-jump word having a nonzero entry at either position
    requires $k+l=2D-3$. Its length constraint would give
    $D(D-1)=(n+m+2)(D-1)+m+1$, which forces $m+1<D-1$ and
    $n+m+2<D$ and is therefore impossible. Such words vanish at both
    positions; pure shifts have equal entries there. The equality
    persists under linear combinations, whereas $\ketbra{D-1}{0}$
    violates it.
\end{proof}

\begin{lemma}[Exact index of the cyclic example]
    \label{thm:cyclic_wielandt_index}
    \lean{MPSTensor.CyclicWielandt.isNormal,
        MPSTensor.CyclicWielandt.isPrimitivePaper,
        MPSTensor.CyclicWielandt.iIndex_eq,
        MPSTensor.CyclicWielandt.classical_bound_lt_iIndex}
    \leanok
    \uses{def:cyclic_wielandt_family, def:wielandt_i_index,
        def:paper_primitive, def:normal}
    The cyclic family is normal and primitive, and
    $\iota(A)=D(D-1)$. For $D>2$, this exceeds the classical
    Wielandt exponent $D^2-2D+2$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:cyclic_wielandt_full_span,
        thm:cyclic_wielandt_predecessor, thm:cyclic_wielandt_identities,
        thm:normal_irreducible_tensor}
    Full span holds at $D(D-1)$ and fails at its predecessor. Since
    $S$ is invertible, fullness at any earlier length would persist
    to the predecessor length, a contradiction. The index difference
    from the classical exponent is $D-2$.
\end{proof}
