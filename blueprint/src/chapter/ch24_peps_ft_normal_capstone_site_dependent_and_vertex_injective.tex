\subsection{Site-dependent chains}

The source states its Lemma~5 for site-dependent tensors: five sites
carrying tensors $A_1, \ldots, A_5$ whose two-site windows are injective.
While the previous results focus on a repeated tensor, the site-dependent setting
treats chains carrying a distinct tensor per site. Windows are arcs of consecutive sites,
and the closed-chain corollary delivers one gauge per bond. Most of the
matrix-chain statements below take one physical dimension $d$ and one bond
dimension $D$. For the Applications-section translational-invariance
corollary, the cycle-graph formulation instead retains one dimension $D_e$ per
bond, derives their equality from the injective Fundamental Theorem, and then
identifies the virtual spaces to construct the repeated tensor in the source's
standing regime of at least three sites, whereas Theorem~\ref{thm:peps_cycle_two_site_bond_uniformity_counterexample}
(\cite{gap:peps_cyclic_mps_two_site_bond_uniformity}) shows that the two-site case of the printed corollary is false.

\begin{definition}[Arc products and window injectivity of a site-dependent
        chain]%
    \label{def:chain_arc_window_injective}
    \lean{MPSChainTensor.arcEval}
    \lean{MPSChainTensor.IsWindowInjective}
    \leanok
    \uses{def:non_ti_chain}
    Let $A=(A_0, \ldots, A_{n-1})$ be a closed chain on $n \ge 1$ sites with
    $A_s\in\MN{D}^d$. The \emph{arc product} of $x=x_1\cdots x_\ell$ from site
    $s$ (indices modulo $n$) is
    \begin{align}
        A_{(s)}^{x}
         & =A_s^{x_1}A_{s+1}^{x_2}\cdots A_{s+\ell-1}^{x_\ell}.
        \label{eq:peps_chain_arc_product}
    \end{align}
    Extending \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}
    to window length $L$, the chain is \emph{window injective at length $L$}
    when $\spn\{A_{(s)}^x:|x|=L\}=\MN{D}$ for every site $s$.
\end{definition}

\begin{lemma}[Injectivity is window injectivity at length one]%
    \label{lem:isWindowInjective_one_of_isInjective}
    \lean{MPSChainTensor.isWindowInjective_one_of_isInjective}
    \leanok
    \uses{def:chain_arc_window_injective, def:injective}
    If every $A_s$ in a closed chain is injective
    ($\spn\{A_s^i\}_{i=0}^{d-1}=\MN{D}$), then the chain is window injective
    at length~$1$.
\end{lemma}
\begin{proof}\leanok
    For $|x|=1$, $A_{(s)}^{i}=A_s^i$ by (\ref{eq:peps_chain_arc_product}).
\end{proof}

\begin{lemma}[Arc spanning from window injectivity]%
    \label{lem:isWindowInjective_arc_span}
    \lean{MPSChainTensor.IsWindowInjective.arc_span}
    \leanok
    \uses{def:chain_arc_window_injective}
    If $A=(A_0,\ldots,A_{n-1})$ is window injective at length $L>0$
    (Definition~\ref{def:chain_arc_window_injective}), then
    $\spn\{A_{(s)}^x:|x|=m\}=\MN{D}$ for every $m \ge L$ and every site $s$
    (cf.\ Lemma~\ref{lem:isNBlkInjective_of_le};
    \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}).
\end{lemma}
\begin{proof}\leanok
    \uses{def:chain_arc_window_injective}
    By induction on $m\ge L$: writing $\Id=\sum_{|u|=L}\alpha_u A_{(s)}^u$
    expresses any $P\in\MN{D}$ as
    $\sum_{i,|w|=L-1}\alpha_{iw} A_s^i\bigl(A_{(s+1)}^w P\bigr)$, and expanding
    each $A_{(s+1)}^w P$ in $\{A_{(s+1)}^y:|y|=m\}$ yields a combination of
    $\{A_{(s)}^z:|z|=m+1\}$.
\end{proof}

\begin{lemma}[Bond-operator extraction from overlapping windows,
        site-dependent form]%
    \label{lem:chain_overlapWindow_exists_bondOperator}
    \lean{MPSChainTensor.overlapWindow_exists_bondOperator}
    \lean{MPSChainTensor.eq_of_span_mul_left}
    \leanok
    \uses{def:chain_arc_window_injective}
    Let $0<L$, $2L+1\le n$, let $A=(A_0,\ldots,A_{n-1})$ with $A_s\in\MN{D}^d$
    be window injective at length $L$, and fix a site $r$. Let
    $C_0, \ldots, C_L:\{0,\ldots,d-1\}^L\to\MN{D}$ satisfy, for all $0\le j<L$,
    $|p|=j$, $|u|=L+1$, and $|s|=n-j-L-1$,
    \begin{align}
         & \tr\bigl(A_{(r)}^p C_j(u_1\cdots u_L)
        A_{r+j+L}^{u_{L+1}}A_{(r+j+L+1)}^s\bigr)
        \notag                                   \\
         & \qquad
        =\tr\bigl(A_{(r)}^p A_{r+j}^{u_1}
        C_{j+1}(u_2\cdots u_{L+1})A_{(r+j+L+1)}^s\bigr).
        \notag
    \end{align}
    Then there exists a unique $X\in\MN{D}$ such that
    \begin{align}
        C_0(a) & = A_{(r)}^a X,  &
        C_L(b) & = X A_{(r+L)}^b
        \qquad (|a|=|b|=L),
        \label{eq:peps_chain_bond_operator}
    \end{align}
    establishing \cite[Lemma~5]{Molnar2018NormalPEPS}
    (with constant specialization Lemma~\ref{lem:overlapWindow_exists_bondOperator}).
\end{lemma}
\begin{proof}\leanok
    \uses{lem:isWindowInjective_arc_span, def:chain_arc_window_injective}
    Since $n-L-1\ge L$, Lemma~\ref{lem:isWindowInjective_arc_span} and trace
    nondegeneracy strip the complementary arc to give
    $C_j(u_1\cdots u_L)A_{r+j+L}^{u_{L+1}}=A_{r+j}^{u_1}C_{j+1}(u_2\cdots u_{L+1})$
    for $0\le j<L$. Chaining $j=0,\ldots,L-1$ yields
    $C_0(a)A_{(r+L)}^b=A_{(r)}^aC_L(b)$ for all $|a|=|b|=L$. Choosing $\alpha_b$
    with $\sum_b\alpha_bA_{(r+L)}^b=\Id$ and setting $X:=\sum_b\alpha_bC_L(b)$
    gives $A_{(r)}^a X=C_0(a)$ and $X A_{(r+L)}^b=C_L(b)$; spanning of
    $\{A_{(r)}^a\}$ makes $X$ unique.
\end{proof}

\begin{lemma}[Arc transfer between same-state chains]%
    \label{lem:exists_arcTransport}
    \lean{MPSChainTensor.exists_arcTransport}
    \leanok
    \uses{def:chain_arc_window_injective, def:non_ti_chain}
    Let site-dependent chains $A,B$ on $n$ sites generate the same closed-chain
    state, let $k+q=n$, and suppose $\spn\{A_{(s)}^x:|x|=k\}=\MN{D}$ and
    $\spn\{A_{(s+k)}^y:|y|=q\}=\MN{D}$. Then there exists a linear map
    $\Lambda:\MN{D}\to\MN{D}$ such that $\Lambda(A_{(s)}^x)=B_{(s)}^x$ for all
    $|x|=k$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:exists_mpvTransport, def:chain_arc_window_injective}
    As in Lemma~\ref{lem:exists_mpvTransport}, define
    $\Psi_A(M)_y=\tr(M A_{(s+k)}^y)$ and $\Psi_B(M)_y=\tr(M B_{(s+k)}^y)$
    ($|y|=q$). Injectivity of $\Psi_A$ and the equality
    $\Psi_A(A_{(s)}^x)_y=\tr(A_{(s)}^{xy})=\tr(B_{(s)}^{xy})=\Psi_B(B_{(s)}^x)_y$
    give $\Lambda=\Psi_B^{-1}\circ\Psi_A$.
\end{proof}

\begin{lemma}[The insertion correspondence is an algebra homomorphism,
        site-dependent form]%
    \label{lem:chain_exists_insertionHom}
    \lean{MPSChainTensor.exists_insertionHom}
    \lean{MPSChainTensor.insertionHom_unique}
    \leanok
    \uses{def:chain_arc_window_injective, def:non_ti_chain}
    Let $0<L$, $2L+1\le n$, let $A$ and $B$ be site-dependent chains on $n$
    sites, each window injective at length $L$, generating the same
    closed-chain state, and fix a site $r$. Then there exists a unique linear
    map $\Phi:\MN{D}\to\MN{D}$ such that
    \begin{align}
        \tr(X A_{(r+L)}^w) = \tr\bigl(\Phi(X) B_{(r+L)}^w\bigr)
        \qquad (X\in\MN{D},\ |w|=n),
        \label{eq:peps_chain_insertion_hom}
    \end{align}
    and $\Phi$ is a unital algebra homomorphism, establishing the closing clause
    of \cite[Lemma~5]{Molnar2018NormalPEPS}.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:chain_overlapWindow_exists_bondOperator, lem:exists_arcTransport, lem:isWindowInjective_arc_span}
    For $X\in\MN{D}$ and $0\le j\le L$, transfer the deformed window
    $u\mapsto A_{(r+j)}^{u_1\cdots u_{L-j}} X A_{(r+L)}^{u_{L-j+1}\cdots u_L}$
    to $B$ via $\Lambda_{r+j}$ (Lemma~\ref{lem:exists_arcTransport}) to obtain
    $C_j$, and extract $\Phi(X)\in\MN{D}$ via
    Lemma~\ref{lem:chain_overlapWindow_exists_bondOperator}. Uniqueness in
    (\ref{eq:peps_chain_bond_operator}) makes $\Phi$ linear, unital, and
    multiplicative; expanding $X$ in $\{A_{(r+L)}^b:|b|=L\}$ gives
    (\ref{eq:peps_chain_insertion_hom}), which uniquely determines $\Phi$ by
    Lemma~\ref{lem:isWindowInjective_arc_span} at length $n$ and trace
    nondegeneracy.
\end{proof}

\begin{lemma}[Conjugate arc products at every bond, site-dependent form]%
    \label{lem:chain_exists_conjugation_of_sameState}
    \lean{MPSChainTensor.exists_conjugation_of_sameState}
    \leanok
    \uses{def:chain_arc_window_injective, def:non_ti_chain}
    Under the hypotheses of Lemma~\ref{lem:chain_exists_insertionHom} with
    $D>0$, for every site $p$ there exists $Z_p\in\GL_D(\C)$ such that
    \begin{align}
        B_{(p)}^w = Z_p A_{(p)}^w Z_p^{-1}
        \qquad (|w|=n).
        \label{eq:peps_chain_conj_full_length}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{lem:chain_exists_insertionHom}
    By Lemma~\ref{lem:chain_exists_insertionHom}, $\Phi$ is a unital
    endomorphism of $\MN{D}$, hence an inner automorphism
    $\Phi(X)=Z_p^{-1} X Z_p$ by Skolem--Noether
    (Theorem~\cite[``Skolem--Noether'']{QICLean2026Blueprint}); substituting
    into (\ref{eq:peps_chain_insertion_hom}) at $r+L\equiv p$ and using trace
    nondegeneracy gives (\ref{eq:peps_chain_conj_full_length}).
\end{proof}

\begin{corollary}[Fundamental Theorem for site-dependent normal closed
        chains at $n \ge 2L + 1$]%
    \label{cor:fundamentalTheorem_normalMPSChain_of_overlap}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPSChain_of_overlap}
    \leanok
    \uses{def:chain_arc_window_injective, def:non_ti_chain}
    Let $0<L$, $2L+1\le n$, $D>0$, and let $A,B$ be site-dependent chains on
    $n$ sites, each window injective at length $L$, generating the same
    closed-chain state. Then there exist invertible matrices
    $(Z_v)_{v=0}^{n-1}$ ($Z_n\equiv Z_0$) such that
    \begin{align}
        B_v^i = Z_v A_v^i Z_{v+1}^{-1}
        \qquad (0\le v<n,\ 0\le i<d)
        \label{eq:peps_site_dependent_chain_gauge}
    \end{align}
    \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{lem:chain_exists_conjugation_of_sameState, lem:chain_overlapWindow_exists_bondOperator, lem:isWindowInjective_arc_span}
    For $L\le m\le n-L$, combining (\ref{eq:peps_chain_conj_full_length}) at
    $p$ and $p+m$ with Lemma~\ref{lem:chain_overlapWindow_exists_bondOperator}
    and Lemma~\ref{lem:isWindowInjective_arc_span} gives
    $B_{(p)}^u = c_{p,m} Z_p A_{(p)}^u Z_{p+m}^{-1}$ ($c_{p,m}\in\C^\times$,
    $|u|=m$). Comparing $m=L+1$ with $m=L$ strips a single site to yield
    $B_v^i = \mu_v Z_v A_v^i Z_{v+1}^{-1}$ ($\mu_v\in\C^\times$). Multiplying
    around the cycle against a nonzero length-$n$ arc product forces
    $\prod_{v=0}^{n-1}\mu_v=1$, and rescaling $Z_v$ by $\prod_{k<v}\mu_k^{-1}$
    absorbs $\mu_v$ into (\ref{eq:peps_site_dependent_chain_gauge}).
\end{proof}

\begin{theorem}[Cyclic gauges preserve closed-chain states]%
    \label{thm:mps_chain_gaugeEquiv_sameState}
    \lean{MPSChainTensor.arcEval_eq_gauge_conj}
    \lean{MPSChainTensor.GaugeEquiv.sameState}
    \leanok
    \uses{def:chain_arc_window_injective, def:non_ti_chain}
    If site-dependent chains $A,B$ on $n$ sites satisfy
    $B_v^i=Z_vA_v^iZ_{v+1}^{-1}$ for all $v,i$ ($Z_n\equiv Z_0$), then for every
    site $s$ and word $x$ of length $\ell$,
    $B_{(s)}^x = Z_s A_{(s)}^x Z_{s+\ell}^{-1}$, and in particular
    $\tr(B_{(0)}^w)=\tr(A_{(0)}^w)$ for all $|w|=n$.
\end{theorem}
\begin{proof}\leanok
    Telescoping $B_v^i=Z_v A_v^i Z_{v+1}^{-1}$ along (\ref{eq:peps_chain_arc_product})
    gives $B_{(s)}^x = Z_s A_{(s)}^x Z_{s+\ell}^{-1}$; at $\ell=n$,
    $Z_n\equiv Z_0$ and cyclicity of trace give $\tr(B_{(0)}^w)=\tr(A_{(0)}^w)$.
\end{proof}

\begin{corollary}[Fundamental Theorem for injective closed MPS chains]%
    \label{cor:fundamentalTheorem_injectiveMPSChain_of_sameState}
    \lean{TNLean.PEPS.fundamentalTheorem_injectiveMPSChain_of_sameState}
    \leanok
    \uses{cor:fundamentalTheorem_normalMPSChain_of_overlap, lem:isWindowInjective_one_of_isInjective}
    Let $A=(A_0,\ldots,A_{n-1})$ and $B=(B_0,\ldots,B_{n-1})$ be site-dependent
    closed chains on $n\ge 3$ sites with common physical dimension $d$ and
    bond dimension $D>0$. If every $A_v$ and $B_v$ is injective and $A,B$
    generate the same closed-chain state, then there exist invertible matrices
    $Z_0,\ldots,Z_{n-1}$ satisfying (\ref{eq:peps_site_dependent_chain_gauge})
    \cite[Theorem~\textup{thm:inj\_MPS}, lines~688--725]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{cor:fundamentalTheorem_normalMPSChain_of_overlap, lem:isWindowInjective_one_of_isInjective}
    Apply Corollary~\ref{cor:fundamentalTheorem_normalMPSChain_of_overlap} with
    $L=1$ ($2L+1=3\le n$), using
    Lemma~\ref{lem:isWindowInjective_one_of_isInjective}.
\end{proof}

\begin{definition}[Cyclic shift of a closed MPS chain]%
    \label{def:mps_chain_cyclic_shift}
    \lean{MPSChainTensor.cyclicShift}
    \lean{MPSChainTensor.IsCyclicShiftInvariantState}
    \leanok
    For a closed chain $A=(A_0,\ldots,A_{n-1})$, its cyclic shift is
    $\mathrm{shift}(A)_v=A_{v+1\bmod n}$. The state of $A$ is cyclic-shift
    invariant when $\tr(A_0^{\sigma_0}\cdots A_{n-1}^{\sigma_{n-1}})
        =\tr(A_1^{\sigma_0}\cdots A_0^{\sigma_{n-1}})$ for all $\sigma$.
\end{definition}

\begin{definition}[Cyclic shift of a cycle-graph tensor]%
    \label{def:peps_cycle_tensor_cyclic_shift}
    \lean{TNLean.PEPS.cycleRotate}
    \lean{TNLean.PEPS.Tensor.cycleShift}
    \lean{TNLean.PEPS.IsCycleShiftInvariantState}
    \leanok
    \uses{def:peps_tensor, def:peps_same_state}
    On the $n$-site cycle ($n\ge 3$) with rotation $r(v)=v+1\bmod n$, the cyclic
    shift of $A=(A_0,\ldots,A_{n-1})$ is the pullback $(r^*A)_v=A_{v+1}$
    \cite[lines~1807--1827]{Molnar2018NormalPEPS}. Its state is cyclic-shift
    invariant when $\Coeff_{r^*A}=\Coeff_A$.
\end{definition}

\begin{corollary}[Bond dimensions of a cyclic-shift-invariant injective MPS]%
    \label{cor:peps_cycle_shift_bond_uniform}
    \lean{TNLean.PEPS.gaugeEquiv_cycleShift_of_isCycleShiftInvariantState}
    \lean{TNLean.PEPS.bondDim_cycleSuccEdge_add_one_eq_of_isCycleShiftInvariantState}
    \lean{TNLean.PEPS.bondDim_eq_of_isCycleShiftInvariantState}
    \leanok
    \uses{def:peps_cycle_tensor_cyclic_shift, thm:fundamentalTheorem_PEPS, def:cycleTensorOfMPS,
        def:peps_injective_tensor}
    Let $A=(A_0,\ldots,A_{n-1})$ be a site-dependent injective closed MPS on
    $n\ge 3$ sites with bond dimensions $D_e>0$. If $\Coeff_{r^*A}=\Coeff_A$,
    then $D_e=D_f$ for all bonds $e,f$
    \cite[lines~1803--1842]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    Applying Theorem~\ref{thm:fundamentalTheorem_PEPS} to $A$ and $r^*A$ gives
    $D_e=D_{r(e)}$ for every bond $e=(v,v+1)$; iterating $r$ around the cycle
    gives $D_e=D_f$ for all $e,f$.
\end{proof}

\begin{theorem}[Two-site obstruction to cyclic MPS bond uniformity]%
    \label{thm:peps_cycle_two_site_bond_uniformity_counterexample}
    \lean{TNLean.PEPS.CycleMPSTwoSiteCounterexample.twoSiteBondUniformityStatement_false}
    \leanok
    \discussion{3371}
    For $n=2$, there exist injective tensors
    \begin{align}
        A_1^i & \in\operatorname{Mat}_{1\times2}(\mathbb C),
              & A_2^i                                        & \in\operatorname{Mat}_{2\times1}(\mathbb C)
        \qquad (i=0,1),
        \label{eq:peps_cycle_two_site_rectangular_tensors}
    \end{align}
    whose closed-chain state is cyclic-shift invariant while their bond
    dimensions are $1\neq 2$, refuting the $n=2$ case of
    \cite[lines~1803--1805]{Molnar2018NormalPEPS}
    (\cite{gap:peps_cyclic_mps_two_site_bond_uniformity}).
\end{theorem}
\begin{proof}\leanok
    Let $e_0,e_1$ be the standard basis of $\mathbb C^2$ and set
    $A_1^i=e_i^{\mathsf T}$, $A_2^i=e_i$. Both local maps are injective, and
    \begin{align}
        \tr(A_1^iA_2^j)=\delta_{ij}=\tr(A_2^iA_1^j),
        \label{eq:peps_cycle_two_site_shift_invariance}
    \end{align}
    whereas the two bonds have dimensions $1$ and $2$.
\end{proof}

\begin{corollary}[Translation-invariant description of an injective closed MPS chain,
        uniform bond dimension]%
    \label{cor:exists_constant_injectiveMPS_of_cyclicShiftInvariantState}
    \lean{TNLean.PEPS.exists_constant_injectiveMPS_of_cyclicShiftInvariantState}
    \leanok
    \uses{cor:fundamentalTheorem_injectiveMPSChain_cyclicShift, def:mps_chain_cyclic_shift}
    Let $A=(A_0,\ldots,A_{n-1})$ be an injective closed chain on $n\ge 3$ sites
    with physical dimension $d$ and bond dimension $D>0$. If its state is
    cyclic-shift invariant, then there exists an injective tensor
    $B\in\MN{D}^d$ such that, for every $\sigma$,
    \begin{align}
        \tr\!\left(A_0^{\sigma_0}\cdots A_{n-1}^{\sigma_{n-1}}\right)
         & =
        \tr\!\left(B^{\sigma_0}\cdots B^{\sigma_{n-1}}\right)
        \notag
    \end{align}
    \cite[lines~1803--1890]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    By Corollary~\ref{cor:fundamentalTheorem_injectiveMPSChain_cyclicShift},
    $A_{v+1}^i=Z_v A_v^i Z_{v+1}^{-1}$. With $P_m=Z_{m-1}\cdots Z_0$ and
    $\widetilde B^i=A_0^i Z_0$, telescoping gives
    $A_m^i=P_m \widetilde B^i P_{m+1}^{-1}$ and
    $\tr(A_0^{\sigma_0}\cdots A_{n-1}^{\sigma_{n-1}})=\tr(\widetilde B^{\sigma_0}\cdots\widetilde B^{\sigma_{n-1}}P_n^{-1})$.
    Across the seam $A_n\equiv A_0$, $P_n$ commutes with $\spn\{\widetilde B^i\}=\MN{D}$,
    so $P_n=\lambda\Id$ with $\lambda\in\C^\times$. Choosing $c\in\C^\times$
    with $c^n=\lambda^{-1}$, $B:=c\widetilde B$ is injective and satisfies
    $\tr(B^{\sigma_0}\cdots B^{\sigma_{n-1}})=\tr(A_0^{\sigma_0}\cdots A_{n-1}^{\sigma_{n-1}})$.
\end{proof}

    \label{cor:exists_constant_injectiveMPS_of_cyclicShiftInvariantState_per_bond_pos}
    \lean{TNLean.PEPS.Tensor.reindexMPSChain}

    \lean{TNLean.PEPS.exists_constant_injectiveMPS_of_cyclicShiftInvariantState_per_bond}
    \leanok

    \uses{cor:peps_cycle_shift_bond_uniform, cor:exists_constant_injectiveMPS_of_cyclicShiftInvariantState}
    Let $A=(A_0,\ldots,A_{n-1})$ be a site-dependent injective closed MPS on
    $n\ge 3$ sites with bond dimensions $D_e>0$. If $\Coeff_{r^*A}=\Coeff_A$
    \cite[lines~1803--1890]{Molnar2018NormalPEPS}, then there exist $D>0$ and an
    injective tensor $B\in\MN{D}^d$ such that $D_e=D$ for every bond $e$ and
    \begin{align}
        \Coeff_A(\sigma)
         & =
        \tr\!\left(B^{\sigma_0}\cdots B^{\sigma_{n-1}}\right)
        \qquad (\forall\,\sigma)
        \notag
    \end{align}
    (the $n=2$ case is false by
    Theorem~\ref{thm:peps_cycle_two_site_bond_uniformity_counterexample};
    \cite{gap:peps_cyclic_mps_two_site_bond_uniformity}).
\end{corollary}
\begin{proof}\leanok
    By Corollary~\ref{cor:peps_cycle_shift_bond_uniform}, $D_e=D$ for all $e$.
    Reindexing each bond $e_v=(v,v+1)$ to $\{0,\ldots,D-1\}$ via
    $\iota_v(\xi)=(\xi(e_{v-1}),\xi(e_v))$ and setting
    $C_v^i(a,b)=A_v^{\prime\,i}(\iota_v^{-1}(a,b))$ gives an injective uniform
    chain $C$ with
    \begin{align}
        \Coeff_A(\sigma)
         & =\sum_{g:\{0,\ldots,n-1\}\to\{0,\ldots,D-1\}}
        \prod_{v=0}^{n-1}
        C_v^{\sigma_v}\!\left(g(v-1),g(v)\right)
        =\Coeff_C(\sigma)
        \notag
    \end{align}
    and $\Coeff_{\mathrm{shift}(C)}(\sigma)=\Coeff_C(\sigma\circ r^{-1})=\Coeff_C(\sigma)$.
    Applying Corollary~\ref{cor:exists_constant_injectiveMPS_of_cyclicShiftInvariantState}
    to $C$ yields $B$.
\end{proof}

\begin{theorem}[Uniqueness of the injective closed-chain gauge]%
    \label{thm:fundamentalTheorem_injectiveMPSChain_gauge_unique}
    \lean{TNLean.PEPS.fundamentalTheorem_injectiveMPSChain_gauge_unique}
    \leanok
    \uses{cor:fundamentalTheorem_injectiveMPSChain_of_sameState, def:chain_arc_window_injective}
    Under the hypotheses of
    Corollary~\ref{cor:fundamentalTheorem_injectiveMPSChain_of_sameState}, if
    invertible families $(Z_v)_{v=0}^{n-1}$ and $(Z'_v)_{v=0}^{n-1}$ both satisfy
    $B_v^i=Z_vA_v^iZ_{v+1}^{-1}=Z'_vA_v^i(Z'_{v+1})^{-1}$ for all $v,i$, then
    there exists $c\in\C^\times$ such that $Z'_v=cZ_v$ for all $v$
    \cite[line~724]{Molnar2018NormalPEPS}.
\end{theorem}
\begin{proof}\leanok
    With $C_v=Z_v^{-1}Z'_v$, the two relations give $C_v A_v^i = A_v^i C_{v+1}$
    for all $i$. Since $\spn\{A_v^i\}=\MN{D}$, $C_v W = W C_{v+1}$ for all
    $W\in\MN{D}$; taking $W=\Id$ gives $C_v=C_{v+1}=:C$, so $C\in Z(\MN{D})=\C^\times\Id$.
\end{proof}

The injective MPS results of \cite[Section~3]{Molnar2018NormalPEPS} are the
$L=1$ specialization of the corollaries above for $n\ge 3$.
