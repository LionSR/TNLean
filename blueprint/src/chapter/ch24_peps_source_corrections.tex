\section{Source substitutions and corrected density terms}

Following \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
(\texttt{04-compression.tex}, lines 279--383),
let $W:\mathcal H_a\to\mathcal H_b$ be a chronological composition with finite
source-occurrence set $\mathcal E$. For $e=(g,j)\in\mathcal E$, let $\Xi_g$ be
the local branch labels of gate $g$, $c_{g,\alpha}$ its branch coefficients,
and $\nu_{e,\alpha}\in\C^{u_e}\otimes\C^{v_e}$ its original source vector. All
coordinate matrices use fixed finite orthonormal bases on $\mathcal H_a$ and
$\mathcal H_b$ without private-dimension bounds (manuscript lines 5--19), and
$\rho$ is an arbitrary input matrix.

\begin{definition}[Substitution in the original composition]
    \label{def:peps_source_substitution}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.branchWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.evalWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.selectedSourceVectors}
    \leanok
    \uses{def:peps_source_circuit,def:peps_prepared_source_gate,def:peps_partial_source_choices}
    For vectors $\eta_{e,\alpha}\in\C^{u_e}\otimes\C^{v_e}$, let $W(\eta)$ be
    the operator obtained by replacing the source preparations in branch
    $\alpha$ of each gate $g$ by $\eta_{e,\alpha}$, summing with $c_{g,\alpha}$,
    and composing in chronological order. For an affected party set
    $\mathcal A$, let $W_{\mathcal A,\xi}(\eta)$ be the partial branch formed by
    performing this substitution at touched gates while keeping untouched gates
    as their aggregate operators. For $S\subseteq\mathcal E$, let
    $\nu[S\leftarrow\eta]$ equal $\eta$ on $S$ and $\nu$ on
    $\mathcal E\setminus S$.
\end{definition}

\begin{lemma}[Summing untouched gates]
    \label{thm:peps_source_substitution_partial}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.evalWithSources_original}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialWithSources_original}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.evalWithSources_eq_sum_partialWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_selectedSourceVectors_eq_sum}
    \leanok
    \uses{def:peps_source_substitution}
    We have $W(\nu)=W$ and $W_{\mathcal A,\xi}(\nu)=W_{\mathcal A,\xi}$. If
    $\eta_{e,\alpha}=\nu_{e,\alpha}$ whenever the owning gate of $e$ does not
    touch $\mathcal A$ (in particular, for $\nu[S\leftarrow\eta]$ when every
    $e\in S$ has an endpoint in $\mathcal A$), then under the owner
    identifications $J_a,J_b$,
    \begin{align}
        J_b W(\eta)
         & =\sum_{\xi}c_{\mathcal A,\xi}
        W_{\mathcal A,\xi}(\eta)J_a.
        \label{eq:peps_substitution_partial}
    \end{align}
\end{lemma}

\begin{proof}
    \lean{TNLean.Lattice.PartyCircuit.Layout.affectedOwner_mapOwner_eq_none}
    \leanok
    \uses{thm:peps_partial_source_evaluation,thm:peps_source_circuit_incidence}
    By induction on $W$ using Lemma~\ref{thm:peps_partial_source_evaluation}:
    each touched gate expands into its branch sum, while each untouched gate has
    $\eta_{e,\alpha}=\nu_{e,\alpha}$ and exterior participants, so it evaluates
    to its original aggregate operator. For $W=W''W'$ with intermediate
    identification $J_c$,
    \begin{align}
        J_bW''(\eta)W'(\eta)
         & =\sum_{\zeta}c_{\mathcal A,\zeta}W''_{\mathcal A,\zeta}(\eta)
        J_cW'(\eta)
        =\sum_{\zeta,\zeta'}c_{\mathcal A,\zeta}c_{\mathcal A,\zeta'}
        W''_{\mathcal A,\zeta}(\eta)W'_{\mathcal A,\zeta'}(\eta)J_a,\notag
    \end{align}
    giving (\ref{eq:peps_substitution_partial}); Lemma~\ref{thm:peps_source_circuit_incidence}
    gives the specialization to $\nu[S\leftarrow\eta]$.
\end{proof}

\begin{definition}[Remaining operations after source extraction]
    \label{def:peps_canonical_source_residual}

    sources of a word to its initial registers
    (Lemma~\ref{thm:peps_source_preparation}) and identifying the source
    register list with a prescribed equal list. For $W_{\mathcal A,\xi}$ with
    fixed retained positions $\mathcal E_{\mathcal A}$
    (Lemma~\ref{thm:peps_partial_fixed_layout}), denote this residual by

\end{definition}

\begin{lemma}[Recovery from the remaining operations]
    \label{thm:peps_canonical_source_recovery}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_residualInSlots}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_residualInSlots}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_residualInSlots_prepare}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.slotLayout_partialSlots_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sources_partialResidual}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isAllowed_partialResidual}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_partialResidual_prepare}
    \leanok
    \uses{def:peps_canonical_source_residual}
    The composition $V$ is source-free, allowed when the original word is
    allowed, and satisfies $VP_\eta=W$ for its ordered source inventory $\eta$.
    In particular, for allowed partial branches,
    \begin{align}
        V_{\mathcal A,\xi}P_{\nu_\xi}
         & =W_{\mathcal A,\xi},
         & (\nu_\xi)_i          & =\nu_{e_i,\xi_{g_i}}.
        \label{eq:peps_canonical_recovery}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_source_preparation,thm:peps_partial_original_source_vector}
    Combine Lemma~\ref{thm:peps_source_preparation} with the inventory
    identification $(\nu_\xi)_i=\nu_{e_i,\xi_{g_i}}$ of
    Lemma~\ref{thm:peps_partial_original_source_vector}.
\end{proof}

\begin{lemma}[Uniform recovery after source substitution]
    \label{thm:peps_canonical_substitution_recovery}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.layout_sources_partialWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.residualInSlots_partialWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.ofSlots_partialWithSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_partialResidual_prepare_substituted}
    \leanok
    \uses{def:peps_source_substitution,def:peps_canonical_source_residual}
    For any $\eta$, $W_{\mathcal A,\xi}(\eta)$ has retained positions
    $\mathcal E_{\mathcal A}$ and inventory $(\eta_\xi)_i=\eta_{e_i,\xi_{g_i}}$.
    If $\eta_{e,\alpha}=\nu_{e,\alpha}$ whenever both endpoints of $e$ lie
    outside $\mathcal A$, its residual composition is $V_{\mathcal A,\xi}$ and
    \begin{align}
        V_{\mathcal A,\xi}P_{\eta_\xi}
         & =W_{\mathcal A,\xi}(\eta),
         & (\eta_\xi)_i               & =\eta_{e_i,\xi_{g_i}}.
        \label{eq:peps_substituted_recovery}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_partial_source_order,thm:peps_canonical_source_recovery}
    By Lemma~\ref{thm:peps_partial_source_order}, substitution preserves the
    retained positions $\mathcal E_{\mathcal A}$, and since exterior-only
    sources retain $\nu_{e,\alpha}$, the extracted residual equals
    $V_{\mathcal A,\xi}$; Lemma~\ref{thm:peps_canonical_source_recovery} then
    gives (\ref{eq:peps_substituted_recovery}).
\end{proof}

\begin{lemma}[Weighted matrix sums]
    \label{thm:peps_weighted_density_sum}
    \lean{Matrix.sum_smul_mul_mul_conjTranspose}
    \leanok
    For finite families $c_\alpha,d_\beta\in\C$ and compatible rectangular
    matrices $K_\alpha,L_\beta,R$,
    \begin{align}
        \Bigl(\sum_\alpha c_\alpha K_\alpha\Bigr)R
        \Bigl(\sum_\beta d_\beta L_\beta\Bigr)^\dagger
         & =\sum_{\alpha,\beta}c_\alpha\overline{d_\beta}
        K_\alpha R L_\beta^\dagger.
        \label{eq:peps_weighted_density_sum}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Expand by bilinearity of matrix multiplication and conjugate linearity of
    $(\cdot)^\dagger$.
\end{proof}

\begin{definition}[Corrected terms at original source positions]
    \label{def:peps_original_corrected_term}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.SourceBasisChoice}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceBasisChoiceFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.labelBasisVectors}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceBasisMatrix}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceTerm}
    \leanok
    \uses{def:peps_source_substitution}
    For $S\subseteq\mathcal E$, a choice $u=(\alpha,a)$ assigns to each
    $e=(g,j)\in S$ a label $\alpha_e\in\Xi_g$ and an endpoint-basis index $a_e$.
    Let $M_{S,u}$ be the matrix of $W(\eta)$ where $\eta_{e,\gamma}$ is the
    $a_e$-th basis tensor if $e\in S$ and $\gamma=\alpha_e$, $0$ if $e\in S$ and
    $\gamma\neq\alpha_e$, and $\nu_{e,\gamma}$ if $e\notin S$. For source
    matrices $E_{e,\alpha,\beta}$, define
    \begin{align}
        F_S(E;\rho)
         & :=\sum_{u=(\alpha,a),\,v=(\beta,b)}
        \left(\prod_{e\in S}E_{e,\alpha_e,\beta_e}(a_e,b_e)\right)
        M_{S,u}\rho M_{S,v}^{\dagger}.
        \label{eq:peps_original_corrected_term}
    \end{align}
\end{definition}

\begin{lemma}[The empty corrected set]
    \label{thm:peps_corrected_empty}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceTerm_empty}
    \leanok
    \uses{def:peps_original_corrected_term}
    For the original circuit matrix $M$, $F_\varnothing(E;\rho)=M\rho M^\dagger$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_source_substitution_partial}
    On $S=\varnothing$, there is a single empty choice $u$ with empty product
    $1$ and $M_{\varnothing,u}=M$ by Lemma~\ref{thm:peps_source_substitution_partial}.
\end{proof}

\begin{definition}[Partially expanded basis substitutions]
    \label{def:peps_partial_basis_substitution}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceBasisMatrix}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.SelectedSourceCoordinates}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.selectedSourceCoordinatesFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coordinateBasisVectors}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialCoordinateBasisMatrix}
    \leanok
    \uses{def:peps_original_corrected_term,def:peps_partial_source_choices}
    Let $M_{\mathcal A,\xi;S,u}$ be the matrix of the substituted partial branch
    $J_b^{-1}W_{\mathcal A,\xi}(\eta)J_a$ for $u=(\alpha,a)$ in
    Definition~\ref{def:peps_original_corrected_term}, and for an endpoint-basis
    assignment $a$ on $S$, let $K_{\mathcal A,\xi;S,a}$ be the partial-branch
    matrix with the $a_e$-th basis tensor at each $e\in S$ and $\nu$ outside $S$.
\end{definition}

\begin{lemma}[Partial expansion of the same corrected term]
    \label{thm:peps_corrected_partial}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceBasisMatrix_eq_sum_partial}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceTerm_eq_sum_partial}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceBasisMatrix_eq_ite}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceTerm_eq_sum_coordinates}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialCoordinateBasisMatrix_eq_preparedMatrix}
    \leanok
    \uses{def:peps_partial_basis_substitution}
    If every $e\in S$ has an endpoint in $\mathcal A$, then
    $M_{S,u}=\sum_\xi c_{\mathcal A,\xi}M_{\mathcal A,\xi;S,u}$ with
    $M_{\mathcal A,\xi;S,(\alpha,a)}=\mathbf 1_{\forall e\in S,\,\alpha_e=\xi_{g_e}}K_{\mathcal A,\xi;S,a}$,
    so
    \begin{align}
        F_S(E;\rho)
         & =\sum_{\xi,\zeta}c_{\mathcal A,\xi}\overline{c_{\mathcal A,\zeta}}
        \sum_{a,b}
        \left(\prod_{e\in S}
        E_{e,\xi_{g_e},\zeta_{g_e}}(a_e,b_e)\right)
        K_{\mathcal A,\xi;S,a}\rho K_{\mathcal A,\zeta;S,b}^{\dagger},
        \label{eq:peps_corrected_partial}
    \end{align}
    where $K_{\mathcal A,\xi;S,a}$ is the prepared matrix of
    $V_{\mathcal A,\xi}$ with basis vectors $a_e$ on $S$ and $\nu_{e_i,\xi_{g_i}}$
    on $\mathcal E_{\mathcal A}\setminus S$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_source_substitution_partial,thm:peps_weighted_density_sum,
        thm:peps_canonical_substitution_recovery,thm:peps_source_preparation_sums}
    Expand $M_{S,u}$ by (\ref{eq:peps_substitution_partial}) and $F_S(E;\rho)$
    by (\ref{eq:peps_weighted_density_sum}). By
    (\ref{eq:peps_substituted_recovery}) and multilinearity of source
    preparation (Lemma~\ref{thm:peps_source_preparation_sums}),
    $M_{\mathcal A,\xi;S,(\alpha,a)}$ vanishes unless $\alpha_e=\xi_{g_e}$ for
    all $e\in S$, in which case it equals $K_{\mathcal A,\xi;S,a}$, giving
    (\ref{eq:peps_corrected_partial}).
\end{proof}

\begin{lemma}[Prepared densities as source contractions]

\begin{lemma}[The prepared corrected-subset identity]
    \label{thm:peps_prepared_subset_identity}
    \lean{TNLean.Lattice.PartyCircuit.Word.sourceOperatorDensity_sub}
    \lean{TNLean.Lattice.PartyCircuit.Word.sourceOperatorDensity_exact}
    \lean{TNLean.Lattice.PartyCircuit.Word.sourceOperatorDensity_sub_prepared}
    \leanok
    \uses{def:peps_prepared_source_replacement}
    For any source matrix families $X,E$,
    \begin{align}
        \mathcal R(X+E)-\mathcal R(X)
         & =\sum_{\varnothing\ne S\subseteq\{0,\ldots,r-1\}}
        \mathcal R(E\text{ on }S,\ X\text{ off }S).
        \label{eq:peps_prepared_subsets}
    \end{align}
    When $X_{\xi,\zeta,i}=\ketbra{\eta_{\xi,i}}{\eta_{\zeta,i}}$,
    $\mathcal R(X)=M\rho M^\dagger$ for $M=\sum_\xi c_\xi K_{V_\xi}(\eta_\xi)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_prepared_source_contraction,thm:peps_weighted_density_sum}
    Expanding $\prod_i(X_i+E_i)-\prod_i X_i$ over nonempty
    $S\subseteq\{0,\ldots,r-1\}$ in (\ref{eq:peps_source_contraction}) gives
    (\ref{eq:peps_prepared_subsets}), and (\ref{eq:peps_prepared_contraction})
    with (\ref{eq:peps_weighted_density_sum}) gives $\mathcal R(X)=M\rho M^\dagger$.
\end{proof}

\begin{lemma}[The global corrected-subset identity]
    \label{thm:peps_global_corrected_subsets}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceReplacedDensity_exact}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceReplacedDensity_piecewise}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceReplacedDensity_sub_exact}
    \leanok
    \uses{def:peps_global_source_replacement}
    We have $\widehat\rho(X)=M\rho M^\dagger$,
    $\widehat\rho(E\text{ on }S,X\text{ off }S)=F_S(E;\rho)$, and
    \begin{align}
        \widehat\rho(X+E)-M\rho M^\dagger
         & =\sum_{\varnothing\ne S\subseteq\mathcal E}F_S(E;\rho).
        \label{eq:peps_global_corrected_subsets}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_corrected_source_contraction,thm:peps_corrected_empty,
        thm:peps_prepared_subset_identity,thm:peps_all_source_subsets}
    Apply (\ref{eq:peps_corrected_contraction}) with $\mathcal A=P$ to
    $S=\varnothing$ (using Lemma~\ref{thm:peps_corrected_empty}) and to
    general $S\subseteq\mathcal E$, then combine (\ref{eq:peps_prepared_subsets})
    with Lemma~\ref{thm:peps_all_source_subsets}.
\end{proof}

\begin{definition}[Physical output after discarding retained registers]
    \label{def:peps_physical_source_replacement}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalDensity}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalCorrectedSourceTerm}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalSourceReplacedDensity}
    \leanok
    \uses{def:peps_global_source_replacement,def:peps_original_corrected_term}
    In a product orthonormal basis of the physical and discarded output
    memories, define
    \begin{align}
        \rho_{\mathrm{phys}}            & :=\operatorname{Tr}_{\mathrm{disc}}(M\rho M^\dagger), &
        \widehat\rho_{\mathrm{phys}}(Y) & :=\operatorname{Tr}_{\mathrm{disc}}\widehat\rho(Y),   &
        F_S^{\mathrm{phys}}(E;\rho)     & :=\operatorname{Tr}_{\mathrm{disc}}F_S(E;\rho).
        \label{eq:peps_physical_source_replacement}
    \end{align}
\end{definition}
\begin{lemma}[The physical corrected-subset identity]
    \label{thm:peps_physical_corrected_subsets}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalSourceReplacedDensity_exact}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalSourceReplacedDensity_sub_exact}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.rectangularTraceNorm_physicalSourceReplacedDensity_sub_le}
    \leanok
    \uses{def:peps_physical_source_replacement}
    We have $\widehat\rho_{\mathrm{phys}}(X)=\rho_{\mathrm{phys}}$ and
    \begin{align}
        \widehat\rho_{\mathrm{phys}}(X+E)-\rho_{\mathrm{phys}}
         & =\sum_{\varnothing\ne S\subseteq\mathcal E}F_S^{\mathrm{phys}}(E;\rho),
        \label{eq:peps_physical_corrected_subsets}                                 \\
        \bigl\|\widehat\rho_{\mathrm{phys}}(X+E)-\rho_{\mathrm{phys}}\bigr\|_1
         & \leq\sum_{\varnothing\ne S\subseteq\mathcal E}
        \bigl\|F_S^{\mathrm{phys}}(E;\rho)\bigr\|_1.\notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_global_corrected_subsets}
    Apply $\operatorname{Tr}_{\mathrm{disc}}$ linearly to
    (\ref{eq:peps_global_corrected_subsets}) and take the trace-norm triangle
    inequality.
\end{proof}

\subsection{Preparing the original private input}

\begin{definition}[The source-to-output composition]
    \label{def:peps_actual_private_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourcePrivatePreparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sources_sourcePrivatePreparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceWord}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sources_partialSourceWord}
    \leanok
    \uses{def:peps_private_product_input,def:peps_canonical_source_residual,def:peps_all_party_source_registers,def:peps_affected_owner,def:peps_partial_source_choices,thm:peps_owner_composition,thm:peps_canonical_source_recovery}
    For a distributed composition $W$ with private product input
    $\psi\in\mathcal H_a$, fixed source memory $\mathcal H_S$, and all-party
    residual $V_\xi$, let $P_\psi:\mathcal H_S\otimes\C\to\mathcal H_S\otimes\mathcal H_a$
    extend the owner-relabeled preparation of $\psi$ by $\Id_{\mathcal H_S}$,
    and define the source-free source-to-output composition $T_\xi=V_\xi P_\psi$,
    as in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 137--139 and 246--267).
\end{definition}

\begin{lemma}[Action of the private preparation]
    \label{thm:peps_actual_private_preparation}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_sourcePrivatePreparation_tmul}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_partialSourceWord_tmul}
    \leanok
    \uses{def:peps_actual_private_preparation}
    Under the append and owner identifications, every $y\in\mathcal H_S$
    satisfies
    \begin{align}
        P_\psi(y\otimes1) & =y\otimes\psi,
        \label{eq:peps_actual_private_preparation}      \\
        T_\xi(y\otimes1)  & =V_\xi(y\otimes\psi).\notag
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_private_product_input,lem:lattice_register_reorderings,
        thm:peps_owner_composition}
    By Definition~\ref{def:peps_private_product_input},
    Lemma~\ref{lem:lattice_register_reorderings}, and
    (\ref{eq:peps_owner_composition}), $P_\psi(y\otimes 1)=y\otimes\psi$, and
    applying $V_\xi$ gives $T_\xi(y\otimes 1)=V_\xi(y\otimes\psi)$.
\end{proof}
