\section{The exterior source memory when all parties are selected}

Let $P$ be the set of original parties and let $W$ be a finite chronological
composition. Each source occurrence $e$ has distinct endpoints $p_e,q_e\in P$
and prescribed complex halfspaces $\C^{u_e}$ and $\C^{v_e}$, whose dimensions
may be zero. When every party is selected, write $Q=P\sqcup\{*\}$
for the owner set. An original endpoint $p$ retains its own owner, and
$*$ denotes the exterior owner.

\begin{definition}[Fixed all-party source registers]
    \label{def:peps_all_party_source_registers}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allResidualSourceLayout}
    \leanok
    \uses{def:peps_partial_fixed_positions,def:peps_affected_owner}
    Let $\mathcal E_W$ be the source occurrences in their original
    preparation order. Define $R_W$ to be the concatenation, in that order,
    of the two-register lists
    $[(p_e,\C^{u_e}),(q_e,\C^{v_e})]$ for $e\in\mathcal E_W$.
    Its owners and spaces depend only on the original composition, before
    a monomial or a prepared source vector is chosen.
\end{definition}

\begin{theorem}[No exterior source registers for the all-party selection]
    \label{thm:peps_all_party_source_exterior_empty}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_allResidualSourceLayout_isNone}
    \leanok
    \uses{def:peps_all_party_source_registers}
    The sublist of $R_W$ consisting of registers at owner
    $*$ is empty. This holds for every chronological composition,
    including when $P$ or the source inventory is empty.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_all_party_source_registers,thm:peps_partial_fixed_layout,
        def:peps_affected_owner}
    A register in $R_W$ belongs to one of the two halves of an actual
    source occurrence. The source-slot specification identifies its
    owner with the affected owner of the corresponding original
    endpoint. Under the all-party selection this is
    $p\in P$, which is distinct from $*$. No register is therefore retained by the
    exterior selection.
\end{proof}

This identity concerns the fixed source registers in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 253--267 and 342--434.
The exterior memory of this empty register list is the scalar space
$\C$. The identity does not remove the scalar contributed by source-free
operations on an empty memory.

\begin{definition}[Fixed coordinates of the actual source input]
    \label{def:peps_actual_source_input_basis}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceU}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceV}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceLabels}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceBasis}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualSourceInputBasis}
    \uses{def:peps_all_party_source_registers,def:peps_source_slot_basis}
    For each source occurrence $e$, take the standard bases of
    $\C^{u_e}$ and $\C^{v_e}$. Their ordered tensor product is an
    orthonormal basis $(b_x)_{x\in I_W}$ of $\mathcal H_{R_W}$, where
    $I_W=\prod_{e\in\mathcal E_W}
        (\{0,\ldots,u_e-1\}\times\{0,\ldots,v_e-1\})$.
    Transport this basis through the canonical identification
    $\mathcal H_{R_W}\simeq\mathcal H_{R_W\,\varnothing}$, obtaining
    $(b_x^+)_{x\in I_W}$ on the input of the actual private preparation.
    These bases depend on the original source positions and spaces, before
    a monomial choice or sampled source vector is specified. This is the
    fixed-coordinate convention of
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 253--267 and 279--355.
\end{definition}

\begin{definition}[Literal source-to-output columns]
    \label{def:peps_actual_source_column_matrix}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix}
    \uses{def:peps_actual_private_preparation,def:peps_actual_source_input_basis}
    Let $T_\xi$ be the actual source-to-output composition with recorded
    private input $\psi$, and choose any finite orthonormal output basis
    $(\beta_j)_{j\in J}$. Its literal column matrix is
    \begin{align}
        (M_\xi)_{j,x}
         & =\langle\beta_j,T_\xi J_{R_W,\varnothing}^{-1}
        (b_x\otimes1)\rangle,
        \label{eq:peps_actual_source_columns}
    \end{align}
    where $J_{R_W,\varnothing}$ is the canonical append isometry.
    The ket and bra compositions may have independent monomial choices.
\end{definition}

\begin{theorem}[Coordinates of the actual source-to-output word]
    \label{thm:peps_actual_source_column_coordinates}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualSourceInputBasis_apply}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_eq_toMatrix}
    \uses{def:peps_actual_source_column_matrix}
    Every $x\in I_W$ satisfies
    $b_x^+=J_{R_W,\varnothing}^{-1}(b_x\otimes1)$.
    Consequently $M_\xi$ is precisely the matrix of $T_\xi$ in the
    input basis $(b_x^+)$ and output basis $(\beta_j)$.
    Empty source inventories give the singleton scalar basis; a source
    halfspace of dimension zero gives an empty column-index set.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_actual_source_input_basis,def:peps_actual_source_column_matrix,thm:peps_basis_final_scalar}
    Appending an empty register list identifies its scalar unit with the
    original source vector. This gives the stated expression for $b_x^+$.
    Substituting it into the defining matrix entries yields
    (\ref{eq:peps_actual_source_columns}).
\end{proof}

\begin{definition}[Scalar contributions of zero-party gates]
    \label{def:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.emptyGateScalar_gate_of_isEmpty}
    \uses{def:peps_source_circuit_positions,def:peps_prepared_source_gate}
    At gate occurrence $g$, let $C_g$ be its participating party set,
    $\Xi_g$ its monomial labels, and $c_{g,\alpha}$ its recorded coefficients.
    Define
    \begin{align}
        \lambda_W & :=\prod_{g:C_g=\varnothing}
        \left(\sum_{\alpha\in\Xi_g}c_{g,\alpha}\right).
        \label{eq:peps_empty_party_gate_scalar}
    \end{align}
    The empty product is $1$, and an empty label set contributes $0$.
    Concatenating compositions multiplies their scalar contributions;
    extending by spectator registers preserves them. Zero-party gates
    are included in the gate convention of
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28: a private gate has exactly
    one participating party.
\end{definition}

\begin{theorem}[The operator and norm of a zero-party gate]
    \label{thm:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.PEPS.PairEffect.PreparedSourceGate.eval_heq_coeffSum_smul_id_of_isEmpty}
    \lean{TNLean.PEPS.PairEffect.PreparedSourceGate.norm_coeffSum_le_one_of_isEmpty}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.norm_emptyGateScalar_le_one}
    \uses{def:peps_empty_party_gate_scalar}
    A prepared gate with no participating parties acts on $\C$ as
    $z\mapsto(\sum_{\alpha\in\Xi_g}c_{g,\alpha})z$.
    Its contraction bound gives
    $|\sum_{\alpha\in\Xi_g}c_{g,\alpha}|\le1$.
    Consequently every distributed composition satisfies $|\lambda_W|\le1$,
    including compositions with vanishing scalar contributions.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_factorization,def:peps_prepared_source_gate,def:peps_empty_party_gate_scalar}
    A register list on an empty party set is empty, and a word with no
    participating parties acts as the identity on $\C$ by the no-party
    clause of Theorem~\ref{thm:peps_party_factorization}. Each monomial of
    the gate therefore has operator $1$, so its weighted sum has operator
    $\sum_{\alpha\in\Xi_g}c_{g,\alpha}$. Taking operator norms gives the
    coefficient-sum bound. Induction on the composition, using
    $|\lambda_1\lambda_2|\le|\lambda_1||\lambda_2|$, proves the final bound.
\end{proof}


\begin{theorem}[Exterior restriction of the private preparation]
    \label{thm:peps_actual_private_preparation_exterior}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_sourcePrivatePreparation_none_eval_heq_id}
    \leanok
    \uses{def:peps_actual_private_preparation,
        thm:peps_all_party_source_exterior_empty}
    Select all original parties. The restriction of the actual private
    preparation to the exterior owner is the identity on the empty memory
    $\mathbb C$. This identity holds before any choice of gate monomials.
    It is the private-preparation part of the local contraction in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 137--139, 246--267 and 409--434.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_all_party_source_exterior_empty,
        thm:peps_excluded_spectator_restriction,
        thm:peps_owner_disjoint_restriction,def:peps_actual_private_preparation}
    Write the private preparation as $F_\tau(V^f)$, where $V$ prepares
    the recorded private input, $V^f$ relabels each owner $p$ by the
    selected original owner $f(p)=p\in Q$, and $F_\tau$ adjoins the source
    register list $\tau$ unchanged. By
    Theorem~\ref{thm:peps_exterior_owner_selector}, restriction to $\{*\}$
    selects exactly the registers owned by $*$. Every source register
    belongs to an original party, so $\tau|_{\{*\}}$ is empty by
    Theorem~\ref{thm:peps_all_party_source_exterior_empty}, and removing
    these spectators gives the first equality below. Since $*\notin f(P)$,
    Theorem~\ref{thm:peps_owner_disjoint_restriction} gives the second:
    \begin{align}
        F_\tau(V^f)|_{\{*\}}
         & =V^f|_{\{*\}}=\operatorname{id}_{\mathbb C}.
    \end{align}
\end{proof}


\begin{theorem}[Exterior scalar of an actual empty-party gate]
    \label{thm:peps_actual_empty_party_gate_scalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_localPart_gate_heq_coeffSum_of_isEmpty}
    \leanok
    \uses{def:peps_partial_source_choices,def:peps_canonical_source_residual,
        thm:peps_empty_party_gate_scalar}
    Select all original parties, and consider an actual gate $G$ with
    no participating parties and branch coefficients $(c_i)_{i\in I}$.
    For every monomial choice, the exterior restriction of the remaining
    operations of its partial expansion is
    \begin{align}
        \left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C}.
        \label{eq:peps_actual_empty_party_gate_scalar}
    \end{align}
    An empty branch set gives the zero map. No coefficient sum is
    cancelled. This is the empty-party case of the chronological
    contraction in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28, 233--251 and 351--417.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_partial_source_choices,thm:peps_empty_party_gate_scalar,
        thm:peps_excluded_mapped_owners,thm:peps_party_restriction}
    The input and output register lists of $G$ are empty. No participating
    party is selected, so the remaining operations of its partial expansion
    are the local map $\Lambda_*(\widetilde G;\tau^f)$ at the exterior
    owner, beside the relabelled spectator list $\tau^f$, where
    $\widetilde G=UGU^{-1}$ for the canonical identifications $U$ of the
    empty memories. By
    Theorem~\ref{thm:peps_empty_party_gate_scalar},
    $G=(\sum_{i\in I}c_i)\operatorname{id}_{\mathbb C}$. Every original
    spectator belongs to a selected original party, so
    $\tau^f|_{\{*\}}$ is empty. Hence
    \begin{align}
        \Lambda_*(\widetilde G;\tau^f)|_{\{*\}}
         & =\Lambda_*(\widetilde G;\varnothing)
        =UGU^{-1}
        =\left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C},
    \end{align}
    which is (\ref{eq:peps_actual_empty_party_gate_scalar}).
\end{proof}


\begin{theorem}[Selecting the exterior owner]
    \label{thm:peps_exterior_owner_selector}
    \lean{TNLean.PEPS.PairEffect.partySelector_none}
    \leanok
    \uses{def:peps_single_party_selector}
    In the owner set $Q=P\sqcup\{*\}$, the singleton selector
    at $*$ is the indicator of the exterior owner.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_single_party_selector}
    The two selectors are $1$ at $*$ and $0$ at every original party.
\end{proof}

\begin{theorem}[Exterior restriction of a touched gate]
    \label{thm:peps_actual_touched_gate_exterior}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_all_gate_localPart_eval_heq_id}
    \leanok
    \uses{def:peps_partial_source_choices,thm:peps_source_word_parties}
    Select all original parties. If a gate has a participating party, then,
    for each choice of its monomial, the exterior restriction of its remaining
    operations is $\Id_{\C}$. The monomial vectors and spectator registers
    are arbitrary.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_exterior_owner_selector,def:peps_partial_source_choices,thm:peps_source_word_parties}
    The gate is replaced by the chosen monomial. Relabelling its operations
    and spectator registers assigns every owner to a selected original party.
    The exterior owner is outside that image. Moving its pair preparations to
    the beginning preserves the participating parties, so the exterior
    restriction has no participating party and acts as $\Id_{\C}$.
\end{proof}

\begin{theorem}[The actual chronological exterior scalar]
    \label{thm:peps_actual_chronological_exterior_scalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_all_localPart_eval_heq_emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialResidual_all_eval_heq_emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialSourceWord_all_eval_heq_emptyGateScalar}
    \leanok
    \uses{def:peps_empty_party_gate_scalar,def:peps_canonical_source_residual,
        thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_actual_private_preparation_exterior}
    Select all original parties of a composition $W$. For every monomial
    choice, the exterior restriction of its remaining operations acts as
    $\lambda_W\Id_{\C}$. The same identity holds after identifying the
    source registers with their prescribed slots and after preparing the
    recorded private product input. It includes an empty original party set
    and a vanishing coefficient sum.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_scalar_residual_composition,thm:peps_all_party_source_exterior_empty,
        thm:peps_actual_private_preparation_exterior}
    Induct on the composition, writing $\lambda_u$ for the exterior
    scalar of a subword $u$. A local operation or exchange at an original
    party, a gate with participating parties $C$, and a chronological
    composition contribute
    \begin{align}
        \lambda_{\mathrm{op}}=1,\qquad
        \lambda_{\mathrm{gate}}=
        \begin{cases}
            1,                & C\neq\varnothing, \\
            \sum_{i\in I}c_i, & C=\varnothing,
        \end{cases}
        \qquad
        \lambda_{u\circ v}=\lambda_u\lambda_v,
        \label{eq:peps_actual_exterior_scalar_cases}
    \end{align}
    by Theorems~\ref{thm:peps_actual_touched_gate_exterior},
    \ref{thm:peps_actual_empty_party_gate_scalar}
    and~\ref{thm:peps_scalar_residual_composition}, and adjoining an
    excluded register preserves $\lambda_u$. All exterior input,
    intermediate and source memories are empty. The prescribed slot
    identification adds no operation, and the private preparation
    contributes the identity, so $\lambda_W$ is the product of the
    factors in~(\ref{eq:peps_actual_exterior_scalar_cases}). No scalar
    is cancelled.
\end{proof}

\begin{definition}[Actual party matrices and their source coordinates]
    \label{def:peps_actual_party_source_matrices}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualPartySourceBasisChange}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourcePartyMatrix}
    \leanok
    \uses{def:peps_actual_source_input_basis,def:peps_actual_private_preparation,
        def:peps_party_grouped_basis}
    Let $\mathcal P$ be finite and set
    $\mathcal Q=\mathcal P\sqcup\{\star\}$, where $\star$ denotes the
    exterior owner. Choose orthonormal bases $B_q,C_q$ for the actual input
    and output register memories at each $q\in\mathcal Q$, and order all
    parties without repetition. Let $B,C$ be the corresponding grouped bases.
    Write $E$ for the literal source input basis. Define
    \[
        R_{y,x}=B^*_{y}E_x,\qquad
        A_{\xi,q}=[\widetilde W_\xi|_q]_{B_q,C_q},
    \]
    where $\widetilde W_\xi$ is the actual remaining word after preparing the
    recorded private input. Thus $R$ is the matrix of the identity from $E$
    to $B$ and is independent of the monomial choice.
\end{definition}

\begin{theorem}[Actual source columns in party coordinates]
    \label{thm:peps_actual_party_source_columns}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_eq_partyProduct_mul_basisChange}
    \leanok
    \uses{def:peps_actual_party_source_matrices,thm:peps_actual_source_column_coordinates,
        thm:peps_actual_party_word_entries}
    The literal prepared source-column matrix, with output basis $C$, satisfies
    \[
        D_\xi=\left(\bigotimes_{q\in\mathcal Q} A_{\xi,q}\right)R.
    \]
    Every owner, including $\star$, remains in this product. The same bases
    and the same $R$ may be used for independent ket and bra choices.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_actual_source_column_coordinates,thm:peps_actual_party_word_entries,
        def:peps_actual_party_source_matrices}
    First express the actual word in the grouped input and output bases.
    By Theorem~\ref{thm:peps_actual_party_word_entries} its matrix is the
    product of its singleton-restriction matrices,
    \[
        [\widetilde W_\xi]_{B,C}=\bigotimes_{q\in\mathcal Q}A_{\xi,q}.
    \]
    Composing with the identity from the literal source basis $E$ to the
    grouped input basis $B$, whose matrix is $R$, gives
    \[
        D_\xi=[\widetilde W_\xi]_{E,C}
            =[\widetilde W_\xi]_{B,C}\,[\operatorname{id}]_{E,B}
        =\left(\bigotimes_{q\in\mathcal Q}A_{\xi,q}\right)R.
    \]
\end{proof}
