\section{The exterior source memory when all parties are selected}

Let $P$ be the set of original parties and let $W$ be a finite chronological
composition. Each source occurrence $e$ has distinct endpoints $p_e,q_e\in P$
and complex halfspaces $\C^{u_e},\C^{v_e}$ ($u_e,v_e\ge 0$). When all parties
in $P$ are selected, the owner set is $Q=P\sqcup\{*\}$, where $*$ denotes the
exterior owner.

\begin{definition}[Fixed all-party source registers]
    \label{def:peps_all_party_source_registers}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allResidualSourceLayout}
    \leanok
    \uses{def:peps_partial_fixed_positions,def:peps_affected_owner}
    Let $\mathcal E_W$ be the source occurrences of $W$ in preparation order.
    Define $R_W$ as the concatenation in that order of the two-register lists
    $[(p_e,\C^{u_e}),(q_e,\C^{v_e})]$ for $e\in\mathcal E_W$.
\end{definition}

\begin{theorem}[No exterior source registers for the all-party selection]
    \label{thm:peps_all_party_source_exterior_empty}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_allResidualSourceLayout_isNone}
    \leanok
    \uses{def:peps_all_party_source_registers}
    The sublist $R_W|_{\{*\}}$ of registers in $R_W$ owned by $*$ is empty.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_all_party_source_registers,thm:peps_partial_fixed_layout,
        def:peps_affected_owner}
    By Definition~\ref{def:peps_all_party_source_registers} and
    Theorem~\ref{thm:peps_partial_fixed_layout}, every register in $R_W$ has
    owner $p_e\in P$ or $q_e\in P$, and $p\neq *$ for all $p\in P$ under
    Definition~\ref{def:peps_affected_owner}.
\end{proof}

For the fixed source registers of
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 253--267 and 342--434,
the empty sublist $R_W|_{\{*\}}=\varnothing$ has exterior memory $\C$.

\begin{definition}[Fixed coordinates of the source input]
    \label{def:peps_actual_source_input_basis}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allSourceU}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allSourceV}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allSourceLabels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.allSourceBasis}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.actualSourceInputBasis}
    \uses{def:peps_all_party_source_registers,def:peps_source_slot_basis}
    Taking the standard bases of $\C^{u_e}$ and $\C^{v_e}$ for $e\in\mathcal E_W$
    gives the product orthonormal basis $(b_x)_{x\in I_W}$ of $\mathcal H_{R_W}$,
    where
    $I_W=\prod_{e\in\mathcal E_W}(\{0,\ldots,u_e-1\}\times\{0,\ldots,v_e-1\})$.
    Its image under the canonical identification
    $\mathcal H_{R_W}\simeq\mathcal H_{R_W\,\varnothing}$ is the orthonormal
    basis $(b_x^+)_{x\in I_W}$ on the private-preparation input, as in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 253--267 and 279--355.
\end{definition}

\begin{definition}[Source-to-output column matrix]
    \label{def:peps_actual_source_column_matrix}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceGlobalMatrix}
    \uses{def:peps_actual_private_preparation,def:peps_actual_source_input_basis}
    For the source-to-output composition $T_\xi$ with private input $\psi$ and
    a finite orthonormal output basis $(\beta_j)_{j\in J}$, define the column
    matrix $M_\xi$ via the append isometry $J_{R_W,\varnothing}$ by
    \begin{align}
        (M_\xi)_{j,x}
         & =\langle\beta_j,T_\xi J_{R_W,\varnothing}^{-1}
        (b_x\otimes1)\rangle.
        \label{eq:peps_actual_source_columns}
    \end{align}
\end{definition}

\begin{theorem}[Coordinates of the source-to-output word]
    \label{thm:peps_actual_source_column_coordinates}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.actualSourceInputBasis_apply}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceGlobalMatrix_eq_toMatrix}
    \uses{def:peps_actual_source_column_matrix}
    For every $x\in I_W$, $b_x^+=J_{R_W,\varnothing}^{-1}(b_x\otimes1)$, so
    $M_\xi$ is the matrix of $T_\xi$ in the input basis $(b_x^+)_{x\in I_W}$
    and output basis $(\beta_j)_{j\in J}$.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_actual_source_input_basis,def:peps_actual_source_column_matrix,thm:peps_basis_final_scalar}
    By Definition~\ref{def:peps_actual_source_input_basis} and
    Theorem~\ref{thm:peps_basis_final_scalar},
    $b_x^+=J_{R_W,\varnothing}^{-1}(b_x\otimes1)$, which substituted into
    $\langle\beta_j,T_\xi b_x^+\rangle$ gives~(\ref{eq:peps_actual_source_columns}).
\end{proof}

\begin{definition}[Scalar contributions of zero-party gates]
    \label{def:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.emptyGateScalar}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.emptyGateScalar_gate_of_isEmpty}
    \uses{def:peps_source_circuit_positions,def:peps_prepared_source_gate}
    For each gate occurrence $g$ with participant set $C_g$, monomial labels
    $\Xi_g$, and branch coefficients $c_{g,\alpha}$, define
    \begin{align}
        \lambda_W & :=\prod_{g:C_g=\varnothing}
        \left(\sum_{\alpha\in\Xi_g}c_{g,\alpha}\right).
        \label{eq:peps_empty_party_gate_scalar}
    \end{align}
    Under the gate convention of
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28 (where private gates have
    $|C_g|=1$), $\lambda_{W_1\circ W_2}=\lambda_{W_1}\lambda_{W_2}$ and framing
    by spectator registers preserves $\lambda_W$.
\end{definition}

\begin{theorem}[The operator and norm of a zero-party gate]
    \label{thm:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.eval_heq_coeffSum_smul_id_of_isEmpty}
    \lean{TNLean.Lattice.PartyCircuit.PreparedSourceGate.norm_coeffSum_le_one_of_isEmpty}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.norm_emptyGateScalar_le_one}
    \uses{def:peps_empty_party_gate_scalar}
    A prepared gate $g$ with $C_g=\varnothing$ acts on $\C$ as
    $(\sum_{\alpha\in\Xi_g}c_{g,\alpha})\Id_{\C}$, with
    $|\sum_{\alpha\in\Xi_g}c_{g,\alpha}|\le1$, and thus $|\lambda_W|\le1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_factorization,def:peps_prepared_source_gate,def:peps_empty_party_gate_scalar}
    By Theorem~\ref{thm:peps_party_factorization}, each monomial on
    $C_g=\varnothing$ acts as $\Id_{\C}$, so the gate acts as
    $(\sum_{\alpha\in\Xi_g}c_{g,\alpha})\Id_{\C}$. Contractivity of the gate
    gives $|\sum_{\alpha\in\Xi_g}c_{g,\alpha}|\le 1$, and multiplicativity
    in~(\ref{eq:peps_empty_party_gate_scalar}) yields $|\lambda_W|\le 1$.
\end{proof}


\begin{theorem}[Exterior restriction of the private preparation]
    \label{thm:peps_actual_private_preparation_exterior}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_sourcePrivatePreparation_none_eval_heq_id}
    \leanok
    \uses{def:peps_actual_private_preparation,
        thm:peps_all_party_source_exterior_empty}
    When all original parties are selected, the exterior restriction of the
    private preparation in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 137--139, 246--267 and 409--434,
    equals $\operatorname{id}_{\mathbb C}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_all_party_source_exterior_empty,
        thm:peps_excluded_spectator_restriction,
        thm:peps_owner_disjoint_restriction,def:peps_actual_private_preparation}
    The private preparation is $F_\tau(V^f)$, where $V^f$ relabels $p\in P$ to
    $f(p)=p\in Q$ and $F_\tau$ adjoins the source layout $\tau=R_W$. Since
    $\tau|_{\{*\}}=\varnothing$ by
    Theorem~\ref{thm:peps_all_party_source_exterior_empty} and $*\notin f(P)$,
    Theorems~\ref{thm:peps_exterior_owner_selector},
    \ref{thm:peps_excluded_spectator_restriction},
    and~\ref{thm:peps_owner_disjoint_restriction} give
    \begin{align}
        F_\tau(V^f)|_{\{*\}}
         & =V^f|_{\{*\}}=\operatorname{id}_{\mathbb C}.\notag
    \end{align}
\end{proof}


\begin{theorem}[Exterior scalar of a zero-party gate]
    \label{thm:peps_actual_empty_party_gate_scalar}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_partialWord_localPart_gate_heq_coeffSum_of_isEmpty}
    \leanok
    \uses{def:peps_partial_source_choices,def:peps_canonical_source_residual,
        thm:peps_empty_party_gate_scalar}
    When all original parties are selected, for a gate $G$ with empty
    participant set and branch coefficients $(c_i)_{i\in I}$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28, 233--251 and 351--417,
    every monomial choice has exterior residual restriction
    \begin{align}
        \left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C}.
        \label{eq:peps_actual_empty_party_gate_scalar}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_partial_source_choices,thm:peps_empty_party_gate_scalar,
        thm:peps_excluded_mapped_owners,thm:peps_party_restriction}
    By Definition~\ref{def:peps_partial_source_choices}, the residual of $G$ is
    $\Lambda_*(\widetilde G;\tau^f)$ with $\widetilde G=UGU^{-1}$ on $\C$ and
    relabelled spectators $\tau^f$ owned by $P$. Since
    $\tau^f|_{\{*\}}=\varnothing$ by Theorem~\ref{thm:peps_excluded_mapped_owners}
    and $G=(\sum_{i\in I}c_i)\operatorname{id}_{\mathbb C}$ by
    Theorem~\ref{thm:peps_empty_party_gate_scalar},
    Theorem~\ref{thm:peps_party_restriction} gives
    \begin{align}
        \Lambda_*(\widetilde G;\tau^f)|_{\{*\}}
         & =\Lambda_*(\widetilde G;\varnothing)
        =UGU^{-1}
        =\left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C},\notag
    \end{align}
    proving~(\ref{eq:peps_actual_empty_party_gate_scalar}).
\end{proof}

\begin{theorem}[Chronological exterior scalar]
    \label{thm:peps_actual_chronological_exterior_scalar}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_partialWord_all_localPart_eval_heq_emptyGateScalar}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_partialResidual_all_eval_heq_emptyGateScalar}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.restrict_partialSourceWord_all_eval_heq_emptyGateScalar}
    \leanok
    \uses{def:peps_empty_party_gate_scalar,def:peps_canonical_source_residual,
        thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_actual_private_preparation_exterior}
    When all original parties of $W$ are selected, the exterior restriction at
    $*$ of its remaining operations, of its slot-identified residual
    $V_\xi$, and of its prepared source-to-output word $T_\xi$ equals
    $\lambda_W\Id_{\C}$ for every monomial choice $\xi$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_scalar_residual_composition,thm:peps_all_party_source_exterior_empty,
        thm:peps_actual_private_preparation_exterior}
    By induction on $W$ using Theorems~\ref{thm:peps_actual_touched_gate_exterior},
    \ref{thm:peps_actual_empty_party_gate_scalar},
    and~\ref{thm:peps_scalar_residual_composition}, the exterior scalar
    $\lambda_u$ of a subword $u$ satisfies
    \begin{align}
        \lambda_{\mathrm{op}}=1,\qquad
        \lambda_{\mathrm{gate}}=
        \begin{cases}
            1,                & C\neq\varnothing, \\
            \sum_{i\in I}c_i, & C=\varnothing,
        \end{cases}
        \qquad
        \lambda_{u\circ v}=\lambda_u\lambda_v.
        \label{eq:peps_actual_exterior_scalar_cases}
    \end{align}
    Since $R_W|_{\{*\}}=\varnothing$ (Theorem~\ref{thm:peps_all_party_source_exterior_empty})
    and the private preparation restricts to $\Id_{\C}$ at $*$
    (Theorem~\ref{thm:peps_actual_private_preparation_exterior}), the product of
    (\ref{eq:peps_actual_exterior_scalar_cases}) equals $\lambda_W\Id_{\C}$.
\end{proof}

\begin{theorem}[Source columns in party coordinates]
    \label{thm:peps_actual_party_source_columns}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceGlobalMatrix_eq_partyProduct_mul_basisChange}

        thm:peps_actual_party_word_entries}

    \begin{align}
        D_\xi=\left(\bigotimes_{q\in Q} A_{\xi,q}\right)R.
        \label{eq:peps_actual_party_source_columns}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_matrix_orthonormal_coordinates,thm:peps_actual_source_column_coordinates,thm:peps_actual_party_word_entries,
        def:peps_actual_party_source_matrices}
    By Theorem~\ref{thm:peps_actual_party_word_entries},
    $[\widetilde W_\xi]_{B,C}=\bigotimes_{q\in Q}A_{\xi,q}$. Composing with
    $R=[\operatorname{id}]_{E,B}$ via
    Theorem~\ref{thm:peps_matrix_orthonormal_coordinates} and
    Theorem~\ref{thm:peps_actual_source_column_coordinates} gives
    \begin{align}
        D_\xi=[\widetilde W_\xi]_{E,C}
            =[\widetilde W_\xi]_{B,C}\,[\operatorname{id}]_{E,B}
        =\left(\bigotimes_{q\in Q}A_{\xi,q}\right)R.\notag
    \end{align}
\end{proof}

\begin{theorem}[An input operator between prepared source columns]
    \label{thm:peps_actual_source_mixed_word}

    Theorem~\ref{thm:peps_actual_party_mixed_matrix}.

\begin{theorem}[Sampled contraction before the physical trace]
