\section{The exterior source memory when all parties are selected}

Let $P$ be the set of original parties and let $W$ be a finite chronological
composition. Each source occurrence $e$ has distinct endpoints $p_e,q_e\in P$
and complex halfspaces $\C^{u_e}$ and $\C^{v_e}$, whose dimensions
may be zero. When every party in $P$ is selected, the owner set is $Q=P\sqcup\{*\}$,
where each $p\in P$ retains its own owner and $*$ denotes the exterior owner.

\begin{definition}[Fixed all-party source registers]
    \label{def:peps_all_party_source_registers}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allResidualSourceLayout}
    \leanok
    \uses{def:peps_partial_fixed_positions,def:peps_affected_owner}
    Let $\mathcal E_W$ be the source occurrences in their original
    preparation order. Define $R_W$ to be the concatenation, in that order,
    of the two-register lists
    $[(p_e,\C^{u_e}),(q_e,\C^{v_e})]$ for $e\in\mathcal E_W$.
    Its owners and spaces depend only on the original composition, before
    a monomial or a prepared source vector is chosen.
\end{definition}

\begin{theorem}[No exterior source registers for the all-party selection]
    \label{thm:peps_all_party_source_exterior_empty}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_allResidualSourceLayout_isNone}
    \leanok
    \uses{def:peps_all_party_source_registers}
    The sublist of $R_W$ consisting of registers at owner
    $*$ is empty.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_all_party_source_registers,thm:peps_partial_fixed_layout,
        def:peps_affected_owner}
    Each register in $R_W$ belongs to one of the two halves of a
    source occurrence. Its owner is the affected owner of the corresponding
    endpoint in $P$. Under the all-party selection this owner is
    $p\in P$, which differs from $*$. Thus no register is retained at $*$.
\end{proof}

This describes the fixed source registers in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 253--267 and 342--434.
The exterior memory of this empty register list is the scalar space
$\C$.

\begin{definition}[Fixed coordinates of the source input]
    \label{def:peps_actual_source_input_basis}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceU}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceV}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceLabels}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.allSourceBasis}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualSourceInputBasis}
    \uses{def:peps_all_party_source_registers,def:peps_source_slot_basis}
    For each source occurrence $e$, take the standard bases of
    $\C^{u_e}$ and $\C^{v_e}$. Their ordered tensor product forms an
    orthonormal basis $(b_x)_{x\in I_W}$ of $\mathcal H_{R_W}$, where
    $I_W=\prod_{e\in\mathcal E_W}
        (\{0,\ldots,u_e-1\}\times\{0,\ldots,v_e-1\})$.
    Transporting this basis through the standard identification
    $\mathcal H_{R_W}\simeq\mathcal H_{R_W\,\varnothing}$ gives
    $(b_x^+)_{x\in I_W}$ on the input of the private preparation.
    These bases depend only on the source positions and spaces of $W$, as in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 253--267 and 279--355.
\end{definition}

\begin{definition}[Source-to-output column matrix]
    \label{def:peps_actual_source_column_matrix}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix}
    \uses{def:peps_actual_private_preparation,def:peps_actual_source_input_basis}
    Let $T_\xi$ be the source-to-output composition with
    private input $\psi$, and choose a finite orthonormal output basis
    $(\beta_j)_{j\in J}$. Define the column matrix $M_\xi$ by
    \begin{align}
        (M_\xi)_{j,x}
         & =\langle\beta_j,T_\xi J_{R_W,\varnothing}^{-1}
        (b_x\otimes1)\rangle,
        \label{eq:peps_actual_source_columns}
    \end{align}
    where $J_{R_W,\varnothing}$ is the append isometry.
\end{definition}

\begin{theorem}[Coordinates of the source-to-output word]
    \label{thm:peps_actual_source_column_coordinates}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualSourceInputBasis_apply}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_eq_toMatrix}
    \uses{def:peps_actual_source_column_matrix}
    Every $x\in I_W$ satisfies
    $b_x^+=J_{R_W,\varnothing}^{-1}(b_x\otimes1)$.
    Consequently $M_\xi$ is the matrix of $T_\xi$ in the
    input basis $(b_x^+)$ and output basis $(\beta_j)$.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_actual_source_input_basis,def:peps_actual_source_column_matrix,thm:peps_basis_final_scalar}
    Appending an empty register list identifies its scalar unit with the
    source vector, which gives the stated expression for $b_x^+$.
    Substituting it into the matrix entries of $T_\xi$ yields
    (\ref{eq:peps_actual_source_columns}).
\end{proof}

\begin{definition}[Scalar contributions of zero-party gates]
    \label{def:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.emptyGateScalar_gate_of_isEmpty}
    \uses{def:peps_source_circuit_positions,def:peps_prepared_source_gate}
    At gate occurrence $g$, let $C_g$ be its participating party set,
    $\Xi_g$ its monomial labels, and $c_{g,\alpha}$ its coefficients.
    Define
    \begin{align}
        \lambda_W & :=\prod_{g:C_g=\varnothing}
        \left(\sum_{\alpha\in\Xi_g}c_{g,\alpha}\right).
        \label{eq:peps_empty_party_gate_scalar}
    \end{align}
    The empty product is $1$, and an empty label set contributes $0$.
    Concatenating compositions multiplies their scalar contributions, and
    extending by spectator registers preserves them. Zero-party gates
    arise from the gate convention of
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28, under which a private gate has
    one participating party.
\end{definition}

\begin{theorem}[The operator and norm of a zero-party gate]
    \label{thm:peps_empty_party_gate_scalar}
    \leanok
    \lean{TNLean.PEPS.PairEffect.PreparedSourceGate.eval_heq_coeffSum_smul_id_of_isEmpty}
    \lean{TNLean.PEPS.PairEffect.PreparedSourceGate.norm_coeffSum_le_one_of_isEmpty}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.norm_emptyGateScalar_le_one}
    \uses{def:peps_empty_party_gate_scalar}
    A prepared gate with no participating parties acts on $\C$ as
    $z\mapsto(\sum_{\alpha\in\Xi_g}c_{g,\alpha})z$.
    Since the gate is a contraction,
    $|\sum_{\alpha\in\Xi_g}c_{g,\alpha}|\le1$, and hence $|\lambda_W|\le1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_factorization,def:peps_prepared_source_gate,def:peps_empty_party_gate_scalar}
    A register list on an empty party set is empty, and a word with no
    participating parties acts as the identity on $\C$ by
    Theorem~\ref{thm:peps_party_factorization}. Each monomial of
    the gate therefore acts as $1$, so the weighted sum acts as multiplication by
    $\sum_{\alpha\in\Xi_g}c_{g,\alpha}$. Taking operator norms gives the
    bound on the coefficient sum, and induction on the composition using
    $|\lambda_1\lambda_2|\le|\lambda_1||\lambda_2|$ gives $|\lambda_W|\le1$.
\end{proof}


\begin{theorem}[Exterior restriction of the private preparation]
    \label{thm:peps_actual_private_preparation_exterior}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_sourcePrivatePreparation_none_eval_heq_id}
    \leanok
    \uses{def:peps_actual_private_preparation,
        thm:peps_all_party_source_exterior_empty}
    Select all original parties. The restriction of the private
    preparation to the exterior owner $*$ is the identity on the empty memory
    $\mathbb C$.
    This is the private-preparation step of the local contraction in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 137--139, 246--267 and 409--434.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_all_party_source_exterior_empty,
        thm:peps_excluded_spectator_restriction,
        thm:peps_owner_disjoint_restriction,def:peps_actual_private_preparation}
    Write the private preparation as $F_\tau(V^f)$, where $V$ prepares
    the private input, $V^f$ relabels each owner $p$ to
    $f(p)=p\in Q$, and $F_\tau$ adjoins the source
    register list $\tau$ unchanged. By
    Theorem~\ref{thm:peps_exterior_owner_selector}, restriction to $\{*\}$
    selects the registers owned by $*$. Every source register
    belongs to an original party, so $\tau|_{\{*\}}$ is empty by
    Theorem~\ref{thm:peps_all_party_source_exterior_empty}, and removing
    these spectators gives the first equality below. Since $*\notin f(P)$,
    Theorem~\ref{thm:peps_owner_disjoint_restriction} gives the second:
    \begin{align}
        F_\tau(V^f)|_{\{*\}}
         & =V^f|_{\{*\}}=\operatorname{id}_{\mathbb C}.
    \end{align}
\end{proof}


\begin{theorem}[Exterior scalar of a zero-party gate]
    \label{thm:peps_actual_empty_party_gate_scalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_localPart_gate_heq_coeffSum_of_isEmpty}
    \leanok
    \uses{def:peps_partial_source_choices,def:peps_canonical_source_residual,
        thm:peps_empty_party_gate_scalar}
    Select all original parties, and consider a gate $G$ with
    no participating parties and branch coefficients $(c_i)_{i\in I}$.
    For every monomial choice, the exterior restriction of the remaining
    operations of its partial expansion is
    \begin{align}
        \left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C}.
        \label{eq:peps_actual_empty_party_gate_scalar}
    \end{align}
    This is the empty-party case of the chronological
    contraction in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{04-compression.tex}, lines 24--28, 233--251 and 351--417.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_partial_source_choices,thm:peps_empty_party_gate_scalar,
        thm:peps_excluded_mapped_owners,thm:peps_party_restriction}
    The input and output register lists of $G$ are empty. No participating
    party is selected, so the remaining operations of its partial expansion
    are the local map $\Lambda_*(\widetilde G;\tau^f)$ at the exterior
    owner beside the relabelled spectator list $\tau^f$, where
    $\widetilde G=UGU^{-1}$ for the standard identification $U$ of the
    empty memories. By
    Theorem~\ref{thm:peps_empty_party_gate_scalar},
    $G=(\sum_{i\in I}c_i)\operatorname{id}_{\mathbb C}$. Every
    spectator belongs to a selected original party, so
    $\tau^f|_{\{*\}}$ is empty. Hence
    \begin{align}
        \Lambda_*(\widetilde G;\tau^f)|_{\{*\}}
         & =\Lambda_*(\widetilde G;\varnothing)
        =UGU^{-1}
        =\left(\sum_{i\in I}c_i\right)\operatorname{id}_{\mathbb C},
    \end{align}
    which is (\ref{eq:peps_actual_empty_party_gate_scalar}).
\end{proof}


\begin{theorem}[Selecting the exterior owner]
    \label{thm:peps_exterior_owner_selector}
    \lean{TNLean.PEPS.PairEffect.partySelector_none}
    \leanok
    \uses{def:peps_single_party_selector}
    In the owner set $Q=P\sqcup\{*\}$, the singleton selector
    at $*$ is the indicator of the exterior owner.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_single_party_selector}
    Both selectors take the value $1$ at $*$ and $0$ at every $p\in P$.
\end{proof}

\begin{theorem}[Exterior restriction of a touched gate]
    \label{thm:peps_actual_touched_gate_exterior}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_all_gate_localPart_eval_heq_id}
    \leanok
    \uses{def:peps_partial_source_choices,thm:peps_source_word_parties}
    Select all original parties. If a gate has at least one participating party, then
    for each choice of its monomial, the exterior restriction of its remaining
    operations is $\Id_{\C}$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_exterior_owner_selector,def:peps_partial_source_choices,thm:peps_source_word_parties}
    Replacing the gate by the chosen monomial and relabelling its operations
    and spectator registers assigns every owner to a selected party in $P$.
    The exterior owner $*$ lies outside that image. Moving the pair preparations to
    the beginning preserves the participating parties, so the exterior
    restriction has no participating party and acts as $\Id_{\C}$.
\end{proof}

\begin{theorem}[Chronological exterior scalar]
    \label{thm:peps_actual_chronological_exterior_scalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialWord_all_localPart_eval_heq_emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialResidual_all_eval_heq_emptyGateScalar}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.restrict_partialSourceWord_all_eval_heq_emptyGateScalar}
    \leanok
    \uses{def:peps_empty_party_gate_scalar,def:peps_canonical_source_residual,
        thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_actual_private_preparation_exterior}
    Select all original parties of a composition $W$. For every monomial
    choice, the exterior restriction of its remaining operations acts as
    $\lambda_W\Id_{\C}$. The same identity holds after identifying the
    source registers with their slots and after preparing the
    private product input.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_exterior_owner_selector,thm:peps_actual_touched_gate_exterior,thm:peps_actual_empty_party_gate_scalar,
        thm:peps_scalar_residual_composition,thm:peps_all_party_source_exterior_empty,
        thm:peps_actual_private_preparation_exterior}
    Induct on the composition, writing $\lambda_u$ for the exterior
    scalar of a subword $u$. A local operation or exchange at an original
    party, a gate with participating parties $C$, and a chronological
    composition contribute
    \begin{align}
        \lambda_{\mathrm{op}}=1,\qquad
        \lambda_{\mathrm{gate}}=
        \begin{cases}
            1,                & C\neq\varnothing, \\
            \sum_{i\in I}c_i, & C=\varnothing,
        \end{cases}
        \qquad
        \lambda_{u\circ v}=\lambda_u\lambda_v,
        \label{eq:peps_actual_exterior_scalar_cases}
    \end{align}
    by Theorems~\ref{thm:peps_actual_touched_gate_exterior},
    \ref{thm:peps_actual_empty_party_gate_scalar}
    and~\ref{thm:peps_scalar_residual_composition}, and adjoining an
    excluded register preserves $\lambda_u$. All exterior input,
    intermediate, and source memories are empty. The slot
    identification adds no operation, and the private preparation
    contributes the identity, so $\lambda_W$ is the product of the
    factors in~(\ref{eq:peps_actual_exterior_scalar_cases}).
\end{proof}

\begin{definition}[Party matrices and their source coordinates]
    \label{def:peps_actual_party_source_matrices}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualPartySourceBasisChange}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourcePartyMatrix}
    \leanok
    \uses{def:peps_actual_source_input_basis,def:peps_actual_private_preparation,
        def:peps_party_grouped_basis}
    Let $P$ be finite and set
    $Q=P\sqcup\{*\}$, where $*$ denotes the
    exterior owner. Choose orthonormal bases $B_q,C_q$ for the input
    and output register memories at each $q\in Q$, and order all
    owners in $Q$ without repetition. Let $B,C$ be the corresponding grouped bases,
    and write $E$ for the source input basis $(b_x^+)_{x\in I_W}$. Define
    \[
        R_{y,x}=B^*_{y}E_x,\qquad
        A_{\xi,q}=[\widetilde W_\xi|_q]_{B_q,C_q},
    \]
    where $\widetilde W_\xi$ is the remaining word after preparing the
    private input. Thus $R$ is the change-of-basis matrix from $E$
    to $B$ and is independent of the monomial choice $\xi$.
\end{definition}

\begin{theorem}[Source columns in party coordinates]
    \label{thm:peps_actual_party_source_columns}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_eq_partyProduct_mul_basisChange}
    \leanok
    \uses{def:peps_actual_party_source_matrices,thm:peps_actual_source_column_coordinates,
        thm:peps_actual_party_word_entries}
    The prepared source-column matrix $D_\xi$, written in the output basis $C$, satisfies
    \[
        D_\xi=\left(\bigotimes_{q\in Q} A_{\xi,q}\right)R.
    \]
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_matrix_orthonormal_coordinates,thm:peps_actual_source_column_coordinates,thm:peps_actual_party_word_entries,
        def:peps_actual_party_source_matrices}
    In the grouped input and output bases $B$ and $C$,
    Theorem~\ref{thm:peps_actual_party_word_entries} expresses the matrix of
    $\widetilde W_\xi$ as the tensor product of its single-owner restrictions:
    \[
        [\widetilde W_\xi]_{B,C}=\bigotimes_{q\in Q}A_{\xi,q}.
    \]
    Composing with the change of basis $R=[\operatorname{id}]_{E,B}$ from the
    source input basis $E$ to $B$ gives
    \[
        D_\xi=[\widetilde W_\xi]_{E,C}
            =[\widetilde W_\xi]_{B,C}\,[\operatorname{id}]_{E,B}
        =\left(\bigotimes_{q\in Q}A_{\xi,q}\right)R.
    \]
\end{proof}

\begin{theorem}[An input operator between prepared source columns]
    \label{thm:peps_actual_source_mixed_word}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_mixed_word}
    \leanok
    \uses{def:peps_actual_party_source_matrices,
        thm:peps_actual_source_column_coordinates,thm:peps_actual_party_mixed_matrix}
    Let $U$ be a source-free word from the source input memory to
    itself, and let $[U]_E$ be its matrix in the source input basis $E$.
    Set $U_q=[U|_q]_{B_q,B_q}$. For independent ket and bra choices $\xi$ and $\zeta$,
    \begin{align}
        D_\xi[U]_E D_\zeta^\dagger
         & =\bigotimes_{q\in Q}
        A_{\xi,q}U_q A_{\zeta,q}^\dagger.
        \label{eq:peps_actual_source_mixed}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_matrix_orthonormal_coordinates,thm:peps_actual_source_column_coordinates,
        thm:peps_actual_party_mixed_matrix,def:peps_actual_party_source_matrices}
    Express the prepared words and $U$ in orthonormal coordinates.
    The matrix of their composition on the output memory is independent of the
    chosen orthonormal input basis. Evaluating it in the grouped input basis $B$
    via the single-owner restrictions gives~(\ref{eq:peps_actual_source_mixed}).
\end{proof}

\begin{definition}[Endpoint factors at the sampled sources]
    \label{def:peps_original_sampled_endpoint_factors}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledEndpointLeft}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledEndpointRight}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledSourceMatrix_eq_endpoint_average}
    \leanok
    \uses{def:peps_sampled_source_matrix,def:peps_source_gaussian_law}
    At a source occurrence $e$, fix ket and bra
    branch labels $\alpha,\beta$, a Gaussian array, and a sample index $s$.
    Denote the resulting left and right endpoint matrices by
    $L_{e,\alpha\beta,s}$ and $H_{e,\alpha\beta,s}$, where the right factor uses
    the conjugate of the same Gaussian array. The sampled source matrix is
    \begin{align}
        \widehat X_{e,\alpha\beta}
         & =k^{-1}\sum_{s=0}^{k-1}
        L_{e,\alpha\beta,s}\otimes H_{e,\alpha\beta,s},
        \label{eq:peps_sampled_source_endpoints}
    \end{align}
    with $0^{-1}=0$ when $k=0$.
\end{definition}

\begin{definition}[The sampled endpoint word on the source input]
    \label{def:peps_actual_sampled_endpoint_word}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledEndpointWord}

    assignment $j$ on the source occurrences. At each source $e$, form
    the endpoint word from the branch pair $(\xi_e,\zeta_e)$ and the two endpoint
    matrices for sample $j(e)$. Transport this word to the input of the
    private preparation, including the terminal scalar memory,
    and denote it by $U_{\xi,\zeta,j}$.
\end{definition}

\begin{theorem}[Sampled endpoint words contain no sources]
    \label{thm:peps_actual_sampled_endpoint_word_sources}

\begin{proof}\leanok

\begin{theorem}[The sampled source matrix as an average of endpoint words]
    \label{thm:peps_actual_sampled_endpoint_average}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledEndpointWord_average}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.actualSourceInputBasis_repr_sampledEndpointInputWord}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.sampledEndpointInputMatrix_average}
    \leanok
    \uses{def:peps_actual_sampled_endpoint_word,
        thm:peps_endpoint_matrix_average,def:peps_actual_source_input_basis}
    Let $n$ be the number of source slots and let
    $X_{\xi,\zeta}$ be the tensor product of the sampled source matrices
    in the source input basis $E$. Let $\sigma$ be the
    bijection from slot indices $\{0,\ldots,n-1\}$ to the source occurrences. Then
    \begin{align}
        X_{\xi,\zeta}
         & =k^{-n}\sum_{j:\{0,\ldots,n-1\}\to\{0,\ldots,k-1\}}
        [U_{\xi,\zeta,j\circ\sigma^{-1}}]_E.
        \label{eq:peps_actual_sampled_average}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_matrix_orthonormal_coordinates,thm:peps_endpoint_matrix_average,def:peps_actual_sampled_endpoint_word,
        def:peps_original_sampled_endpoint_factors,def:peps_actual_source_input_basis}
    Apply the finite endpoint-averaging identity to the ordered
    source list, reindexing sample assignments $j$ along $\sigma$. Conjugating
    the endpoint word by the input identification preserves its matrix entries in $E$,
    which gives~(\ref{eq:peps_actual_sampled_average}).
\end{proof}

\begin{theorem}[Sampled contraction before the physical trace]
    \label{thm:peps_actual_sampled_party_contraction}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceGlobalMatrix_sampled_eq_sum_partyProduct}
    \leanok
    \uses{thm:peps_actual_sampled_endpoint_average,thm:peps_actual_source_mixed_word}
    In the same notation, let
    $U_{\xi,\zeta,j,q}=[U_{\xi,\zeta,j\circ\sigma^{-1}}|_q]_{B_q,B_q}$.
    The sampled source contraction satisfies
    \begin{align}
        D_\xi X_{\xi,\zeta}D_\zeta^\dagger
         & =k^{-n}\sum_j\bigotimes_{q\in Q}
        A_{\xi,q}U_{\xi,\zeta,j,q}A_{\zeta,q}^\dagger,
        \label{eq:peps_actual_sampled_contraction}
    \end{align}
    where $j$ ranges over all sample assignments on $\{0,\ldots,n-1\}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_actual_sampled_endpoint_average,thm:peps_actual_source_mixed_word}
    Substitute the average~(\ref{eq:peps_actual_sampled_average}) for $X_{\xi,\zeta}$,
    distribute multiplication over the finite sum, and apply

\begin{theorem}[Physical traces of mixed source-column matrices]
    \label{thm:peps_actual_source_column_physical_trace}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.selectedPhysicalPrivateBasis_partialTrace_actualColumns}
    \leanok
    \uses{thm:peps_selected_physical_private_trace,thm:peps_actual_source_column_coordinates}
    Let $X$ be a matrix on the joint source-coordinate space,
    and let $\xi,\zeta$ be independent monomial choices. Let
    $D_\xi,D_\zeta$ be their prepared-column matrices in the
    party-grouped physical and private bases, and let $D_\xi^{\mathrm{ref}}$
    and $D_\zeta^{\mathrm{ref}}$ be their matrices in the physical basis
    tensored with any finite orthonormal reference basis of the private
    output memory $G$. Write $\chi$ for the coordinate bijection separating
    retained physical and discarded private indices and identifying the
    retained indices with the physical configurations. Then
    \begin{align*}
        \operatorname{tr}_{\prod_qE_q}
        \left((D_\xi X D_\zeta^\dagger)^\chi\right)
         & =\operatorname{tr}_{G}
        \left(D_\xi^{\mathrm{ref}}X
        (D_\zeta^{\mathrm{ref}})^\dagger\right).
    \end{align*}
    This identifies the party-coordinate partial trace with the physical partial trace in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 279--355, 409--434 and 565--588.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_selected_physical_private_trace,thm:peps_actual_source_column_coordinates}
    Expand $X$ in the standard basis $(e_s)$ of the joint source-coordinate space:
    \begin{align*}
        D_\xi XD_\zeta^\dagger
         & =\sum_{s,t}X_{s,t}\,(D_\xi e_s)(D_\zeta e_t)^\dagger,
    \end{align*}
    and similarly for $D_\xi^{\mathrm{ref}}X(D_\zeta^{\mathrm{ref}})^\dagger$.
    For each pair $(s,t)$, applying
    Theorem~\ref{thm:peps_selected_physical_private_trace} to the
    $s$-th ket column and $t$-th bra column gives
    \begin{align*}
        \operatorname{tr}_{\prod_qE_q}
        \left(\left((D_\xi e_s)(D_\zeta e_t)^\dagger\right)^\chi\right)
         & =\operatorname{tr}_{G}
        \left((D_\xi^{\mathrm{ref}}e_s)(D_\zeta^{\mathrm{ref}}e_t)^\dagger\right).
    \end{align*}
    Linearity of the reindexing $\chi$ and of both partial
    traces yields the identity for general $X$.
\end{proof}

\begin{theorem}[The exterior factor in the sampled physical trace]
    \label{thm:peps_actual_sampled_exterior_trace}
    \lean{TNLean.PEPS.PairEffect.SourceCircuit.partialSourceWord_sampled_none_partialTrace_eq_scalar}
    \leanok
    \uses{thm:peps_actual_chronological_exterior_scalar,
        thm:peps_actual_sampled_endpoint_exterior}

    Choose any finite orthonormal basis of the exterior input memory and
    any finite orthonormal basis of the exterior output memory indexed by
    $\{1\}\times J$. For independent ket and bra choices $\xi,\zeta$
    and any joint sample assignment $j$, the exterior partial trace
    satisfies
    \begin{align}
        \operatorname{tr}_{J}
        \left(A_{\xi,*}U_{\xi,\zeta,j,*}A_{\zeta,*}^\dagger\right)
         & =\lambda_W\overline{\lambda_W}\,I_{\{1\}}.
        \label{eq:peps_actual_exterior_trace}
    \end{align}
    In particular its single entry is $\lambda_W\overline{\lambda_W}$,
    independently of $j,\xi,\zeta$ and the chosen bases.

    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 137--139, 246--267 and 565--588.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_actual_chronological_exterior_scalar,
        thm:peps_actual_sampled_endpoint_exterior,thm:peps_all_party_source_exterior_empty}

    chronological restrictions act as $\lambda_W I$, and the sampled input
    restriction acts as $I$. If $V$ is the orthonormal change-of-basis matrix,

    $\lambda_W\overline{\lambda_W}\,I_{\{1\}\times J}$.

    tracing over $J$ gives (\ref{eq:peps_actual_exterior_trace}).
