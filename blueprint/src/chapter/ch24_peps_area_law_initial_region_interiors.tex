\section{Disjoint interiors of the initial regions}
\label{sec:area_law_initial_region_interiors}

The initial regions in \cite[Section~11]{OpenAI2026AreaLaw} may share
boundary segments or corners, but their open interiors are pairwise disjoint.

% These assertions concern the initial family before repairs.

\begin{lemma}[Distinct runs have disjoint interiors]
    \label{thm:area_law_fan_run_disjoint_interiors}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanRunRegions_disjoint_interiors}
    \leanok
    \uses{def:area_law_fan_runs}
    Fix a translated dyadic square, a midpoint subdivision, and a
    two-coloring of its fan triangles. If $R\ne T$ are two runs, then
    $\operatorname{int}(T_R)\cap\operatorname{int}(T_T)=\varnothing$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fan_triangle_contact,thm:area_law_cell_fan_intersections,
        def:area_law_fan_runs}
    Distinct runs consist of disjoint sets of fan triangles. For any
    triangle $P_e$ in $R$ and $P_f$ in $T$, the intersection $P_e\cap P_f$
    is either a radial segment or the center of the square, and is therefore
    closed with empty interior. Since $R$ and $T$ are finite,
    \begin{align}
        T_R\cap T_T
         & \subseteq\bigcup_{e\in R}\bigcup_{f\in T}(P_e\cap P_f)
    \end{align}
    is a finite union of nowhere dense sets, so its interior
    $\operatorname{int}(T_R)\cap\operatorname{int}(T_T)$ is empty.
\end{proof}

\begin{lemma}[Pairwise disjoint initial open regions]
    \label{thm:area_law_initial_open_disjoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialOpenRegion_pairwise_disjoint}
    \leanok
    \uses{def:area_law_initial_open_regions}
    Fix an origin, a finite endpoint set, an integer width $C\ge 2$,
    an initial layer $k_0\ge 50{,}000{,}000$, and valid pitch residues
    at every layer. For any distinct initial region identifiers $i\ne j$,
    $\operatorname{int}(X_i^0)\cap\operatorname{int}(X_j^0)=\varnothing$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fan_run_disjoint_interiors,
        thm:area_law_primary_fine_cell_cover,thm:area_law_fine_cells_disjoint,
        thm:area_law_dummy_layer_disjoint,thm:area_law_fan_run_cover,
        def:area_law_initial_birth_regions}
    Let $Q$ be a half-open dyadic square that is disjoint from each half-open
    cell $T_m$ of a finite family. Since $\operatorname{int}(Q)$ is open, it
    is disjoint from every $\overline{T_m}$ and hence from
    $\operatorname{int}\bigl(\bigcup_m\overline{T_m}\bigr)$. Taking the
    closure of $\operatorname{int}(Q)$ gives
    \begin{align}
        \overline Q\cap\operatorname{int}\Bigl(\bigcup_m\overline{T_m}\Bigr)
         & =\varnothing,
    \end{align}
    so two finite unions of closures of pairwise disjoint half-open dyadic
    squares have disjoint interiors.

    Each primary region is a finite union of closed non-belt fine cells, each
    fan run is contained in the closure of its belt cell, and the dummy tile
    is the closure of the half-open neighborhood $N_{k_0}$. Distinct fine
    cells and $N_{k_0}$ are pairwise disjoint before closure, so the
    identity above shows that initial regions supported on disjoint sets of
    cells have disjoint interiors. The remaining case of two distinct runs in
    the same belt cell follows from the preceding theorem.
\end{proof}
