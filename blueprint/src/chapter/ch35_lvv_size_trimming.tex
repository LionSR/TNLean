\section{Trimming a viable set to polynomial size}
\label{sec:lvv_size_trimming}

Claim~2.7 of~\cite{Landau2015Polynomial} replaces a viable set by one with
polynomially many vectors in the length of the chain. The new vectors are
chosen before the ground state is known: the boundary states at the cut are
discretized by a finite net, and at each point of the net one keeps the left
columns of a leading eigenvector of a density matrix of least energy among
those whose boundary state is close to that point. The resulting selection is
a function of the Hamiltonian, the gap parameter and the input set.

\begin{lemma}[Dyadic approximation in the open-chain unit ball]
    \label{thm:lvv_dyadic_open_mps_approximation}
    \lean{TNLean.MPS.AreaLaw.one_add_pow_sub_one_le_mul_two_pow,
        TNLean.MPS.AreaLaw.mul_inv_two_pow_clog_natCeil_le,
        TNLean.MPS.AreaLaw.LeftOpenMPSRepresentation.exists_unitBounded_of_norm_le_one,
        TNLean.MPS.AreaLaw.IsLeftOpenMPS.exists_dyadic_rational_approximation_of_norm_le_one,
        TNLean.MPS.AreaLaw.IsLeftOpenMPS.exists_dyadic_rational_approximation_at_error_of_norm_le_one,
        TNLean.MPS.AreaLaw.exists_rational_leftFamily_approximation_of_norm_le_one,
        TNLean.MPS.AreaLaw.IsLeftOpenMPS.exists_dyadic_rational_approximation_at_error}
    \leanok
    Let $1\le k\le L$, and let $v$ be a left-block matrix product
    state on $k$ sites with local dimension $q$ and bond bound $b$. Put
    $D=\max\{1,b\}$, and suppose $\|v\|\le1$. For every $\epsilon>0$, define
    \[
      K=q^k D^{k+1}(k+1)2^k,\qquad
      t=\left\lceil\log_2\max\{1,\lceil K/\epsilon\rceil\}\right\rceil.
    \]
    There is a vector $w$ with the same bond bound and Gaussian-rational
    tensor parameters of common coordinate height at most $2^{2t+5}$ such
    that $\|v-w\|\le\epsilon$.
    A finite family of such vectors admits these approximations
    simultaneously with the same height bound and without increasing its
    cardinality. This approximating family is chosen before any ground
    vector or viability witness is specified.
\end{lemma}
\begin{proof}\leanok
    Choose an equivalent open-chain representation with scalar cap and
    local entries of modulus at most one. Round its parameters at mesh
    $\delta=2^{-t}$. The contraction estimate gives error at most
    $q^kD^{k+1}((1+\delta)^{k+1}-1)$. Since $0\le\delta\le1$,
    \[
      (1+\delta)^{k+1}-1\le\delta(k+1)2^k.
    \]
    The choice of $t$ gives $K2^{-t}\le\epsilon$, as required.
\end{proof}

\begin{lemma}[Viability under rounding of subnormalized columns]
    \label{thm:lvv_subnormalized_column_rounding}
    \lean{TNLean.MPS.AreaLaw.isViable_of_close_subnormalized_leftColumns,
        TNLean.MPS.AreaLaw.isViable_of_approximating_leftFamily,
        TNLean.MPS.AreaLaw.isViable_half_of_approximating_trimmed_leftFamily}
    \leanok
    Let $u$ have $m$ left columns in a finite family $S$, let $W$ be a right
    isometry, and let $\Gamma$ be a unit target. Suppose $\|u\|\ge c>0$
    and the normalized extension of $u$ has overlap at least $1-\delta$
    with $\Gamma$. If a family $T$ approximates each member of $S$ within
    distance $\theta\ge0$, then $T$ is viable with error at most
    \[
      \delta+\frac{2\sqrt m\,\theta}{c},
      \qquad \text{provided }\frac{\sqrt m\,\theta}{c}<1.
    \]
    In particular, the trimmed witness bounds $c=37/48$ and
    $1-\delta=7/12$ give error at most $1/2$ whenever
    \[
      \theta\le \frac{37}{1152(\sqrt m+1)}.
    \]
\end{lemma}
\begin{proof}\leanok
    Replace the columns independently. Their joint Euclidean error is at
    most $\sqrt m\,\theta$. Extend by $W$ and divide by the original
    norm, losing at most a factor $c^{-1}$. Normalizing the perturbed
    witness costs at most twice this distance in overlap. The displayed
    mesh bounds the total overlap loss by $1/12$.
\end{proof}

\begin{lemma}[Bit bound for the tensor precision]
    \label{thm:lvv_tensor_dyadic_precision}
    \lean{TNLean.MPS.AreaLaw.clog_natCeil_div_le_of_dyadic_bounds,
        TNLean.MPS.AreaLaw.tensor_contraction_constant_le_two_pow,
        TNLean.MPS.AreaLaw.tensor_dyadic_precision_le_of_dimension_bounds,
        TNLean.MPS.AreaLaw.tensor_dyadic_precision_le_of_polynomial_bond}
    \leanok
    Suppose $q\le2^a$, $D\le2^c$, and $2^{-r}\le\epsilon$, where
    $a,c,k,r$ are natural numbers. For
    $K=q^kD^{k+1}(k+1)2^k$, the precision exponent satisfies
    \[
      \left\lceil\log_2\max\{1,\lceil K/\epsilon\rceil\}\right\rceil
      \le ak+c(k+1)+(k+1)+k+r.
    \]
    If $D\le CL^p$ and $k\le L$, this is bounded uniformly over cuts by
    \[
      aL+\bigl(\ell_2(C)+p\ell_2(L)\bigr)(L+1)+2L+1+r,
      \qquad \ell_2(x)=\lceil\log_2\max\{1,x\}\rceil.
    \]
\end{lemma}
\begin{proof}\leanok
    The dimension bounds and $k+1\le2^{k+1}$ give
    $K\le2^{ak+c(k+1)+(k+1)+k}$. Dividing by $\epsilon\ge2^{-r}$
    adds at most $r$ to this exponent. Taking the natural ceiling and
    then the ceiling binary logarithm preserves the bound.
    Finally $C\le2^{\ell_2(C)}$ and $L\le2^{\ell_2(L)}$ bound the
    polynomial bond dimension by the required power of two.
\end{proof}

\begin{lemma}[Uniform quadratic bound for the tensor precision]
    \label{lem:lvv_uniform_tensor_precision}
    \lean{TNLean.MPS.AreaLaw.inv_two_pow_le_polynomial_family_rounding_mesh,
        TNLean.MPS.AreaLaw.tensor_precision_expression_le_quadratic,
        TNLean.MPS.AreaLaw.dyadic_height_le_quadratic_of_precision_le,
        TNLean.MPS.AreaLaw.exists_tensor_dyadic_precision_quadratic_bound}
    \leanok
    Fix natural numbers $q,C,p,D,s,a$ and a real number $\alpha>0$.
    There is a natural number $E$, depending only on these parameters,
    such that the following holds simultaneously for every chain length
    $L\ge1$, cut $k\le L$, cardinality $v\le CL^p$, and bond budget
    $B\le DL^s$. Define
    \begin{align}
      \theta&=\frac{\alpha}{2L^a(v+1)},\notag\\
      K&=q^k\max\{1,B\}^{k+1}(k+1)2^k,\notag\\
      t&=\ell_2\bigl(\lceil K/\theta\rceil\bigr).\notag
    \end{align}
    Then both the precision exponent and the common rational
    tensor-parameter height satisfy
    \begin{align}
      t&\le E(L+1)^2,\notag\\
      2^{2t+5}&\le2^{(2E+5)(L+1)^2}.\notag
    \end{align}
    One possible constant is
    \begin{align}
      r_0&=\ell_2\bigl(\lceil2(C+1)/\alpha\rceil\bigr),\notag\\
      E&=\ell_2(q)+\ell_2(D+1)+s+2+r_0+a+p.\notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:lvv_tensor_dyadic_precision}
    Since $L\ge1$ and $v\le CL^p$, one has
    $v+1\le(C+1)L^p$. The choice of $r_0$ gives
    $2(C+1)\le\alpha2^{r_0}$, and $L\le2^L$ therefore yields
    $2^{-r_0-(a+p)L}\le\theta$.
    Also $\max\{1,B\}\le(D+1)L^s$.
    Lemma~\ref{thm:lvv_tensor_dyadic_precision} now bounds $t$ by
    \begin{align}
      \ell_2(q)L+
      \bigl(\ell_2(D+1)+s\ell_2(L)\bigr)(L+1)
      +2L+1+r_0+(a+p)L.\notag
    \end{align}
    Using $\ell_2(L)\le L$, each summand is bounded by its constant
    coefficient times $(L+1)^2$. Their sum is at most $E(L+1)^2$.
    Finally $2t+5\le(2E+5)(L+1)^2$, giving the stated height bound.
\end{proof}

\begin{lemma}[Enlarging the bounds on rational tensor data]
    \label{lem:lvv_rational_representation_monotone}
    \lean{TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentation.mono_bond,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.mono_height,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.mono_bond,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.mono}
    \leanok
    Consider an open-chain representation of a vector by a scalar and
    local tensors with Gaussian-rational entries. Its bond budget $b$
    bounds every internal virtual dimension by $\max\{1,b\}$.
    A common coordinate-height bound $H$ means that the reduced numerator
    and positive denominator of every real and imaginary coordinate have
    absolute value at most $H$.
    If $b\le b'$ and $H\le H'$, the same representation has bond budget
    $b'$ and coordinate-height bound $H'$. Enlarging either bound leaves
    all virtual spaces, tensor entries, and the represented vector fixed.
    Without a specified height bound, the bond budget may likewise be
    enlarged while retaining rationality.
\end{lemma}
\begin{proof}\leanok
    Each internal dimension is at most $\max\{1,b\}\le\max\{1,b'\}$,
    and every coordinate height at most $H$ is at most $H'$.
    These inequalities apply to the original scalar and tensors directly.
\end{proof}

\begin{lemma}[Sequences with prescribed successive relations]
    \label{lem:dependent_sequence_choice}
    \lean{Nat.exists_dependent_sequence}
    \leanok
    Let $(X_i)_{i\ge0}$ be a family of sets, and let
    $R_i\subseteq X_i\times X_{i+1}$ be a relation for each $i$.
    Suppose every $x\in X_i$ has a successor $y\in X_{i+1}$ with
    $(x,y)\in R_i$. For every prescribed $x_0\in X_0$, there is a sequence
    $(x_i)_{i\ge0}$ beginning at $x_0$ such that
    $(x_i,x_{i+1})\in R_i$ at every index.
\end{lemma}
\begin{proof}\leanok
    Choose for each $i$ a map $g_i:X_i\to X_{i+1}$ satisfying
    $(x,g_i(x))\in R_i$ for every $x\in X_i$.
    Recursively define $x_{i+1}=g_i(x_i)$ from the prescribed initial value.
\end{proof}

\begin{definition}[A common Gaussian-integer denominator]
    \label{def:lvv_gaussian_integer_denominator}
    \lean{TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator}
    \leanok
    A positive integer $M$ is a denominator for a complex number $z$ if
    $Mz\in\mathbb Z[i]$.
\end{definition}

\begin{lemma}[Clearing finitely many rational coordinates]
    \label{lem:lvv_common_rational_denominator}
    \lean{TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator.finset_sum,
        TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator.mul,
        TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator.finset_prod,
        TNLean.MPS.AreaLaw.exists_intCast_eq_natCast_mul_of_den_dvd,
        TNLean.MPS.AreaLaw.exists_common_gaussianInteger_denominator,
        TNLean.MPS.AreaLaw.rationalHeight_int_div_nat_le,
        TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator.height_of_norm_le,
        TNLean.MPS.AreaLaw.HasGaussianIntegerDenominator.height_of_norm_le_real}
    \leanok
    A family of $P$ Gaussian-rational numbers of coordinate height at most
    $H$ has a common positive denominator $M\le H^{2P}$.
    A finite sum preserves a common denominator. A product of $n$ factors
    with denominator $M$ has denominator $M^n$.
    If $M$ is a denominator for $z$ and $\|z\|\le R$, the reduced
    coordinate heights of $z$ are at most
    $M\max\{1,\lceil R\rceil\}$.
\end{lemma}
\begin{proof}\leanok
    Multiply the reduced positive denominators of all real and imaginary
    coordinates. Each divides the resulting product, which is at most
    $H^{2P}$. The Gaussian integers are closed under addition and
    multiplication, proving the sum and product assertions.
    For the height assertion, write $Mz=a+ib$ with integers $a,b$.
    Then $|a|,|b|\le M\lceil R\rceil$, and reducing $a/M$ and $b/M$
    can only decrease their numerator magnitudes and denominators.
\end{proof}

\begin{lemma}[Common denominators in open-chain contractions]
    \label{lem:lvv_open_chain_common_denominator}
    \lean{TNLean.MPS.AreaLaw.OBCChainTensor.coeff_gaussianInteger_denominator,
        TNLean.MPS.AreaLaw.OBCChainTensor.scalar_mul_coeff_gaussianInteger_denominator,
        TNLean.MPS.AreaLaw.OBCChainTensor.exists_common_parameter_denominator,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.exists_common_coordinate_denominator}
    \leanok
    Suppose an open chain has $k$ local tensors, and a positive integer
    $M$ is a denominator for every local entry. Every physical coefficient
    then has denominator $M^k$. A scalar cap with denominator $M$ changes
    this to $M^{k+1}$.
    In particular, if the scalar and the $P-1$ stored local entries have
    coordinate height at most $H$, one may choose a single
    $M\le H^{2P}$ such that all physical coefficients of the represented
    vector have denominator $M^{k+1}$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:lvv_common_rational_denominator}
    Each virtual path contributes a product of $k$ local entries, hence
    has denominator $M^k$. Summing these products preserves that
    denominator. Multiplication by the scalar cap adds one factor of
    $M$. Apply Lemma~\ref{lem:lvv_common_rational_denominator} to the
    scalar and all stored local entries simultaneously.
\end{proof}

\begin{definition}[Physical-coordinate height of a rational tensor]
    \label{def:lvv_contracted_height_bound}
    \lean{TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.coordinateHeightBound}

\begin{lemma}[Polynomial binary growth of contracted coordinates]
    \label{lem:lvv_contracted_coordinate_height}
    \lean{TNLean.MPS.AreaLaw.HasRationalComplexCoordinatesOfHeight.norm_le_two_mul,
        TNLean.MPS.AreaLaw.OBCChainTensor.norm_coeff_le_of_entry_norm_le,
        TNLean.MPS.AreaLaw.OBCChainTensor.norm_scalar_mul_coeff_le_of_height,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.coordinate_height_common_denominator,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.coordinate_height_le,
        TNLean.MPS.AreaLaw.contractionHeight_le_two_pow,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentationOfHeight.coordinate_height_le_two_pow}
    \leanok
    Every physical coordinate of a scalar-capped rational MPS of length
    $k$, parameter height $H$, and bond bound $b$ has coordinate height
    at most $\mathcal H(H,b,k)$. This estimate does not depend on the
    physical dimension. If $H\le2^h$ and $\max\{1,b\}\le B$, then
    \[
      \mathcal H(H,b,k)
        \le2^{\bigl(2h(1+kB^2)+1+h+B\bigr)(k+1)}.
    \]
\end{lemma}
\begin{proof}\leanok
    \uses{lem:lvv_common_rational_denominator,lem:lvv_open_chain_common_denominator}
    Fix a physical configuration. Only the selected matrix at each site
    contributes to its coefficient. These matrices and the scalar have
    at most $P=1+kD^2$ complex entries. Clear their denominators with
    $M\le H^{2P}$. The contracted coefficient has denominator $M^{k+1}$.
    Each local entry and the scalar have norm at most $2H$, while there
    are at most $D^{k+1}$ virtual paths. Hence the coefficient norm is
    at most $(2HD)^{k+1}$. The denominator and norm estimates give
    $\mathcal H(H,b,k)$.
    Finally $B\le2^B$ and $H\le2^h$ imply the displayed binary bound.
\end{proof}

\begin{definition}[Sequential rational row contraction]
    \label{def:lvv_rational_row_contraction}
    \lean{TNLean.MPS.AreaLaw.RationalRectMatrixCode.rowMul,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.rowProduct,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.rowProduct_nil,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.rowProduct_cons}
    \leanok
    Represent a Gaussian rational by its rational real and imaginary
    coordinates. For a row $u$ of length $d$ and a $d\times d'$ matrix
    $A$, define the rational row product by
    \begin{align*}
      (uA)_c&=\sum_{a=0}^{d-1}u_aA_{ac}.
    \end{align*}
    For a list of square $D\times D$ matrices $A_0,\ldots,A_{n-1}$,
    define the successive rows by $u^{(0)}=u$ and
    $u^{(p+1)}=u^{(p)}A_p$. Compute and retain every coordinate of
    $u^{(p+1)}$ before proceeding to the next matrix. The resulting
    row is $u^{(n)}$; an empty list leaves the initial row fixed.
\end{definition}

\begin{lemma}[Correctness of sequential rational contraction]
    \label{lem:lvv_rational_contraction_correct}
    \lean{TNLean.MPS.AreaLaw.RationalRectMatrixCode.decode_rowMul,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.decode_rowProduct,
        TNLean.MPS.AreaLaw.OBCChainTensor.zeroPadRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.decode_zeroPadRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.coefficientOfRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.decode_coefficientOfRationalParameters,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentation.decode_coefficientOfRationalParameters}
    \leanok
    \uses{def:lvv_rational_row_contraction}
    Interpreting the rational coordinates as complex numbers commutes
    with row multiplication. For the successive rows of
    Definition~\ref{def:lvv_rational_row_contraction}, this gives
    $u^{(n)}=uA_0\cdots A_{n-1}$ as a complex row.

    Consider an open-chain tensor representation with $n$ sites,
    scalar cap $c$, local matrices $A_p^{\sigma_p}$, and virtual
    dimensions $b_0,\ldots,b_n\le D$, where $b_0=b_n=1$.
    Suppose explicit rational coordinate pairs represent $c$ and every
    local tensor entry. Pad each selected matrix with zeros to a
    $D\times D$ matrix $\widetilde A_p^{\sigma_p}$. Starting from
    $u^{(0)}_a=\mathbf1_{a=0}$, perform the rational row contractions
    and return $cu^{(n)}_0$. Its complex value is the represented
    physical coefficient
    \begin{align*}
      cu^{(n)}_0
        &=c\bigl(\widetilde A_0^{\sigma_0}\cdots
          \widetilde A_{n-1}^{\sigma_{n-1}}\bigr)_{00}
          =c\,A(\sigma).
    \end{align*}
    Here $A(\sigma)$ is the open-chain contraction with its
    one-dimensional boundary indices.
    This identity includes $n=0$: the initial zero-coordinate basis
    row is unchanged, and the returned coefficient is $c$.
\end{lemma}
\begin{proof}\leanok
    Rational complex multiplication and finite addition agree with
    their complex counterparts. Induction along the list of matrices,
    using $(uA)B=u(AB)$, proves the row-product identity.
    Entrywise zero-padding also commutes with interpreting the rational
    coordinates as complex numbers. Apply the open-chain identity
    $A(\sigma)=(\widetilde A_0^{\sigma_0}\cdots
    \widetilde A_{n-1}^{\sigma_{n-1}})_{00}$ and multiply by the scalar
    cap. At length zero the ordered matrix product is the identity,
    whose $(0,0)$ entry is one.
\end{proof}

\begin{definition}[Arithmetic count of rational contraction]
    \label{def:lvv_rational_contraction_count}
    \lean{TNLean.MPS.AreaLaw.RationalContractionCost.Arithmetic,
        TNLean.MPS.AreaLaw.RationalContractionCost.rationalAdd,
        TNLean.MPS.AreaLaw.RationalContractionCost.rationalSub,
        TNLean.MPS.AreaLaw.RationalContractionCost.rationalMul,
        TNLean.MPS.AreaLaw.RationalContractionCost.complexAdd,
        TNLean.MPS.AreaLaw.RationalContractionCost.complexMul,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_bind,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_pure,
        TNLean.MPS.AreaLaw.RationalContractionCost.dot,
        TNLean.MPS.AreaLaw.RationalContractionCost.rowMul,
        TNLean.MPS.AreaLaw.RationalContractionCost.rowProduct,
        TNLean.MPS.AreaLaw.RationalContractionCost.contract,
        TNLean.MPS.AreaLaw.RationalContractionCost.coefficientOfRationalParameters}
    \leanok
    \uses{def:lvv_rational_row_contraction}
    A counted rational evaluation returns its value together with a
    natural number recording the arithmetic operations performed.
    Charge one unit for each rational addition, subtraction, or
    multiplication. Returning an already computed value has count zero,
    and successive evaluations add their counts.

    For Gaussian rationals $x=x_1+ix_2$ and $y=y_1+iy_2$, compute
    addition by the two rational additions in
    $(x_1+y_1)+i(x_2+y_2)$. Compute multiplication by the four rational
    products, one subtraction, and one addition in
    $(x_1y_1-x_2y_2)+i(x_1y_2+x_2y_1)$.
    Compute each scalar product by starting at zero and successively
    adding these computed complex products.
    In the row contraction of Definition~\ref{def:lvv_rational_row_contraction},
    evaluate and retain all coordinates of each new row before proceeding
    to the next matrix. After the last row, multiply the selected
    coordinate by the scalar cap.
    Only these rational arithmetic operations are charged; their
    bit-operation cost also depends on the lengths of the numerators
    and denominators.
\end{definition}

\begin{lemma}[Arithmetic count of one physical coefficient]
    \label{lem:lvv_rational_contraction_arithmetic_count}
    \lean{TNLean.MPS.AreaLaw.RationalContractionCost.run_complexAdd,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_complexMul,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_dot,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_vectorOfFnM,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_rowMul,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_rowProduct,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_contract,
        TNLean.MPS.AreaLaw.RationalContractionCost.run_coefficientOfRationalParameters}
    \leanok
    \uses{def:lvv_rational_contraction_count}
    The counted evaluations return the same rational values as the
    sequential contractions. Their accumulated arithmetic counts are
    exactly $2$ for complex addition, $6$ for complex multiplication,
    $8D$ for a scalar product of length $D$, and $8D^2$ for a row
    multiplied by a square matrix of order $D$.
    Thus propagation through $n$ such matrices has count $8nD^2$.
    For an open chain of $n$ sites with virtual dimensions at most $D$,
    including the scalar cap gives the physical coefficient together
    with the exact accumulated count
    \begin{align*}
      8nD^2+6.
    \end{align*}
    This formula includes $n=0$, when only the final scalar
    multiplication is performed.
\end{lemma}
\begin{proof}\leanok
    Each term of the scalar product performs six rational operations
    for multiplication and two for addition to the accumulated value.
    Induction on the number of terms gives $8D$ and the ordinary scalar
    product as its value. Evaluating $D$ output coordinates adds these
    counts to $8D^2$ and retains the resulting row.
    Induction along the list of matrices therefore gives the ordinary
    sequential row product and count $8nD^2$. The final multiplication
    by the scalar cap adds six operations. Selecting the zero-coordinate
    boundary entry gives the asserted open-chain coefficient evaluation.
\end{proof}

\begin{lemma}[Heights of computed rational coordinates]
    \label{lem:lvv_computed_rational_coordinate_height}
    \lean{TNLean.MPS.AreaLaw.RationalComplexCode.decode_injective,
        TNLean.MPS.AreaLaw.RationalComplexCode.decode_height_iff,
        TNLean.MPS.AreaLaw.RationalComplexCode.bitCount_le_of_decode_height,

    \leanok
    \uses{lem:lvv_contracted_coordinate_height,lem:lvv_rational_contraction_correct,
        def:lvv_rational_coordinate_storage}
    Rational real and imaginary coordinates are uniquely determined by
    their complex value. Consequently the actual fractions returned by
    sequential contraction of height-$H$, bond-$b$ tensor data on $k$
    sites have heights at most $\mathcal H(H,b,k)$. Their combined signed
    binary length is at most
    $6(\lfloor\log_2\mathcal H(H,b,k)\rfloor+1)$.
\end{lemma}
\begin{proof}\leanok
    Equality of complex values gives equality of real and imaginary
    coordinates. Injectivity of the rational-to-real inclusion then
    identifies the stored reduced fractions with the coordinates in the
    height certificate. Apply Lemma~\ref{lem:lvv_contracted_coordinate_height}
    and correctness of the sequential evaluator. Each signed fraction
    has length at most twice the height's binary length, plus one sign bit.
\end{proof}

\begin{definition}[Weighted mixed transfer matrices]
    \label{def:lvv_weighted_mixed_transfer}
    \lean{MPSChainTensor.weightedTransferMatrix,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.weightedTransfer,

\begin{lemma}[The weighted transfer product]
    \label{lem:lvv_weighted_transfer_product}

    \begin{align*}
      E_0\cdots E_{n-1}
        &=\sum_{\sigma,\tau\in\{0,\ldots,q-1\}^n}
          \left(\prod_{p=0}^{n-1}(W_p)_{\sigma_p\tau_p}\right)
          \bigl(\overline{A(\sigma)}\otimes B(\tau)\bigr).
    \end{align*}
    In particular, every boundary entry is obtained by taking the
    corresponding entries of $A(\sigma)$ and $B(\tau)$ in this sum.
    The identity includes empty chains, with ordered matrix products
    interpreted as identity matrices.
\end{lemma}

    The identity $(X\otimes Y)(X'\otimes Y')=XX'\otimes YY'$
    separates each term into its two ordered matrix products.
    Entrywise conjugation commutes with multiplication, while the
    scalar weights multiply to the displayed product.
    Indexing the resulting terms by the two physical configurations
    gives the formula, and taking a matrix entry gives its boundary form.
\end{proof}

\begin{lemma}[Rational weighted overlaps and Gram inner products]
    \label{lem:lvv_rational_weighted_overlap}
    \lean{TNLean.MPS.AreaLaw.OBCChainTensor.weightedTransferSiteList,
        TNLean.MPS.AreaLaw.OBCChainTensor.weightedOverlapOfRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.weightedOverlapOfRationalParameters_zero,
        TNLean.MPS.AreaLaw.OBCChainTensor.decode_weightedOverlapOfRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.decode_weightedOverlapOfRationalParameters_eq_productOperator,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.identity,
        TNLean.MPS.AreaLaw.RationalRectMatrixCode.decode_identity,
        TNLean.MPS.AreaLaw.OBCChainTensor.gramOfRationalParameters,
        TNLean.MPS.AreaLaw.OBCChainTensor.decode_gramOfRationalParameters,
        TNLean.MPS.AreaLaw.RationalLeftOpenMPSRepresentation.decode_gramOfRationalParameters}
    \leanok
    \uses{def:lvv_rational_row_contraction,def:lvv_weighted_mixed_transfer}
    Consider two open chains on $n$ sites with Gaussian-rational scalar
    caps $c_a,c_b$, rational local entries, and virtual dimensions bounded
    by $D_a,D_b$. Let $a(\sigma)$ and $b(\tau)$ denote their contractions
    before multiplication by the caps. Pad their virtual spaces with
    zeros and form the rational weighted transfers $E_p$.
    Start a row at the paired zero-coordinate basis vector, propagate it
    through these transfers, and return its paired zero coordinate
    multiplied by $\overline{c_a}c_b$.

    Let $\mathcal W=\bigotimes_{p=0}^{n-1}W_p$ be the finite Kronecker
    product. The resulting Gaussian rational has complex value
    \begin{align*}
      F&=\overline{c_a}c_b
        \sum_{\sigma,\tau}
          \overline{a(\sigma)}
          \left(\prod_{p=0}^{n-1}(W_p)_{\sigma_p\tau_p}\right)b(\tau)\\
       &=\overline{c_a}c_b\,
         \overline a\mathbin{\cdot}\bigl(\mathcal W b\bigr).
    \end{align*}
    The finite Kronecker product here has entries
    $\mathcal W_{\sigma\tau}=\prod_p(W_p)_{\sigma_p\tau_p}$.
    Taking every $W_p$ to be the identity gives the Gram inner product
    $F=\langle c_a a,c_b b\rangle$ of the actual physical coefficient
    vectors, with the inner product conjugate-linear in its first argument.
    Thus for rational representations of two vectors on the first
    $n\le L$ sites, the same evaluation gives their left-block inner product.
    At length zero, both contractions equal one and $F=\overline{c_a}c_b$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:lvv_rational_contraction_correct,lem:lvv_weighted_transfer_product}
    Rational arithmetic and zero-padding commute with their complex
    interpretations. Sequential row contraction therefore gives the
    paired zero-coordinate entry of $E_0\cdots E_{n-1}$.
    Lemma~\ref{lem:lvv_weighted_transfer_product} expands that entry
    into the configuration sum. The entry formula for the finite
    Kronecker product identifies this sum with the product-operator
    matrix element. For identity weights, $\mathcal W$ is the identity,
    giving the Gram inner product after including the scalar caps.
    The natural bijection between numbered configurations and left-block
    configurations is an isometry and preserves this inner product.
\end{proof}

\begin{lemma}[Expansion of a two-site operator into product operators]

    $\{p,r\}$. Choose a reference configuration $\sigma_0$, and write
    $\sigma_0[a,c]$ for its modification with labels $a$ at $p$ and
    $c$ at $r$. With $E_{ab}=|a\rangle\langle b|$, define
    \begin{align*}
      K_{(a,c),(b,d)}&=X_{\sigma_0[a,c],\sigma_0[b,d]},\\
      F_j^{abcd}&=\begin{cases}
        E_{ab},&j=p,\\
        E_{cd},&j=r,\\
        I,&j\notin\{p,r\}.
      \end{cases}
    \end{align*}

    coordinates of $K$.
\end{lemma}

    input and output labels at $p,r$ are precisely $a,b,c,d$, and all
    labels outside these sites agree. In that case its value is one.
    If the exterior labels agree, support of $X$ identifies its matrix
    element with the entry read at $\sigma_0[a,c],\sigma_0[b,d]$.
    If they disagree, support forces that matrix element to vanish.
    Taking matrix elements proves the expansion.
\end{proof}

      \|\phi-\psi\|\le r\theta\|\phi\|.
    \]
    Consequently, if the original family is $\delta$-viable for a unit
    target and $r\theta<1$, the rounded family is
    $(\delta+2r\theta)$-viable.
\end{lemma}
\begin{proof}\leanok
    Expand every right column in the orthonormal left family and replace
    each $e_j$ by $w_j$, keeping its coefficient. The triangle inequality
    and Cauchy--Schwarz bound the error of each column by $r\theta$
    times its norm. Summing squared column norms gives the asserted
    chain-vector bound, independently of the right-block dimension.
    Normalize the replacement to obtain the viability bound.
\end{proof}

\begin{lemma}[Rational approximation of a finite retained subspace]
    \label{thm:lvv_rational_orthonormal_family}
    \lean{TNLean.MPS.AreaLaw.exists_rational_leftFamily_approximation_of_orthonormalization}

    $|T|\le|V|$ and bond bound $b'$, chosen before every target and every
    viability error. Its common tensor-parameter height is bounded by
    the dyadic approximation at mesh $\theta$ with bond bound $b'$.
    For every unit target $\Gamma$ and every $\delta$,
    $\delta$-viability of $V$ implies $(\delta+\epsilon)$-viability of $T$.
\end{lemma}
\begin{proof}\leanok

    dimension by $b'$. Approximate its unit vectors at mesh $\theta$.
    Lemma~\ref{thm:lvv_orthonormal_family_rounding} gives the conclusion
    since $2r\theta\le\epsilon$. This construction uses the retained
    subspace alone and imposes no condition on the original family's

\end{proof}

\begin{definition}[The family of minimizing boundary states]
    \label{def:lvv_finite_boundary_selection}
    \lean{TNLean.MPS.AreaLaw.feasibleBoundaryDensities,
        TNLean.MPS.AreaLaw.feasibleBoundaryPoints,
        TNLean.MPS.AreaLaw.mem_feasibleBoundaryPoints,
        TNLean.MPS.AreaLaw.boundaryMinimizer,
        TNLean.MPS.AreaLaw.boundaryLeadingVector,
        TNLean.MPS.AreaLaw.boundaryLeadingValue,
        TNLean.MPS.AreaLaw.boundaryMinimizerFamily}
    \leanok
    \uses{def:lvv_finite_left_columns,def:chain_gluing_left_coordinates,def:lvv_viable_set}
    Fix a set $S\subseteq\mathcal L$, a linear operator $H_L$ on $\mathcal L$,
    a natural number $m$, a real number $\eta$, and a finite set $N$ of
    matrices on $\mathcal C\otimes\mathbb C^m$, where $\mathcal C$ is the
    Hilbert space of the region between consecutive clipped left blocks. Identify
    $\mathcal L=\mathcal A\otimes\mathcal C$ by joining the left coordinates.
    Let $P_S$ be the orthogonal projection onto $\operatorname{Span}(S)$, and put
    \begin{align}
        Q       & =P_S\otimes I_m, \notag                                          \\
        E(\rho) & =\operatorname{Re}\tr\bigl(\rho(H_L\otimes I_m)\bigr),\notag     \\
        \mathcal F_X
                & =\left\{\rho\ge0:\tr\rho=1,\ Q\rho=\rho,\
        \bigl\|\operatorname{tr}_{\mathcal A}\rho-X\bigr\|_1\le\eta\right\},\notag \\
        F       & =\{X\in N:\mathcal F_X\ne\varnothing\}.\notag
    \end{align}
    $\mathcal F_X$ is the set of feasible densities at $X$ and $F$ the set of
    feasible points of $N$. For every $X$ with $\mathcal F_X\ne\varnothing$,
    choose once a minimizer $\sigma_X\in\mathcal F_X$ of $E$ over
    $\mathcal F_X$, a unit eigenvector $u_X$ of $\sigma_X$ for its largest
    eigenvalue $\mu_X$, and that eigenvalue; at an infeasible point, all three are
    zero. After joining the left coordinates, write
    $u_X=\sum_{j<m}u_{X,j}\otimes|j\rangle$ and define the family of
    minimizing boundary states
    \begin{align}
        T=\{u_{X,j}:X\in F,\ j<m\}.\notag
    \end{align}
    These choices use only $S,H_L,m,\eta,N$. This is the fixed-bond
    construction for the size-trimming step
    of~\cite[Claim~2.7, source lines 472--478 and 520--527]{Landau2015Polynomial}.
\end{definition}

\begin{lemma}[Properties of the family of minimizing boundary states]
    \label{thm:lvv_finite_boundary_selection}
    \lean{TNLean.MPS.AreaLaw.boundaryMinimizer_spec,
        TNLean.MPS.AreaLaw.card_boundaryMinimizerFamily_le,
        TNLean.MPS.AreaLaw.norm_le_one_of_mem_boundaryMinimizerFamily,
        TNLean.MPS.AreaLaw.boundaryMinimizerFamily_subset_span}
    \leanok
    \uses{def:lvv_finite_boundary_selection}
    In the setting of Definition~\ref{def:lvv_finite_boundary_selection},
    for every $X\in F$ the chosen $\sigma_X$ lies in $\mathcal F_X$ and
    satisfies $E(\sigma_X)\le E(\rho)$ for every $\rho\in\mathcal F_X$;
    the chosen $u_X$ is a unit vector with $\sigma_Xu_X=\mu_Xu_X$,
    $\mu_X>0$ and $Qu_X=u_X$, and every real eigenvalue of $\sigma_X$ with a
    nonzero eigenvector is at most $\mu_X$. Moreover
    \begin{align}
        |T| & \le m|N|,\notag                        \\
        T   & \subseteq\operatorname{Span}(S).\notag
    \end{align}
    Each member of $T$ has norm at most one.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_gluing_left_projection,thm:lvv_finite_left_columns}
    At every $X\in F$, the choices are those of the minimizing-density selection
    theorem~\cite[``A minimizing density matrix and a leading eigenvector'']{QICLean2026Blueprint}.
    Its positive eigenvalue gives $Qu_X=u_X$. Joining the left coordinates
    transports this identity to $(P_S\otimes I_m)u_X=u_X$.
    Lemma~\ref{thm:lvv_finite_left_columns} gives
    $T\subseteq\operatorname{Span}(S)$ and $|T|\le m|F|\le m|N|$.
\end{proof}


\begin{lemma}[Size trimming for a one-state local space]
    \label{thm:lvv_size_trimmed_family_dimension_one}
    \lean{TNLean.MPS.AreaLaw.exists_size_trimmed_family_localDimension_one}
    \leanok
    \uses{def:lvv_viable_set,def:lvv_viable_mps}
    Suppose the local Hilbert space has dimension one.  If a finite family
    $S$ is $(k,\delta)$-viable for a normalized chain vector and every member
    of $S$ has matrix product state bond dimension at most $b$, then some
    singleton $T\subseteq S$ is $(k,0)$-viable and its member has bond
    dimension at most $b$.
\end{lemma}
\begin{proof}
    \leanok
    The chain Hilbert space and the left Hilbert space at every cut are both
    one-dimensional.  A unit viable witness is therefore a scalar multiple
    of the normalized chain vector.  Its support condition implies that
    $S$ contains a nonzero vector, and any such vector spans the left Hilbert
    space.  Taking this vector as the sole member of $T$ gives exact
    viability.
\end{proof}

\begin{lemma}[Size trimming for the empty chain]
    \label{thm:lvv_size_trimmed_family_empty_chain}
    \lean{TNLean.MPS.AreaLaw.exists_size_trimmed_family_chainLength_zero}
    \leanok

    Suppose the chain has no sites.  If a finite family $S$ is viable for a
    normalized chain vector and every member of $S$ has matrix product state
    bond dimension at most $b$, then some singleton $T\subseteq S$ is exactly
    viable and its member has bond dimension at most $b$.
\end{lemma}
\begin{proof}

    one-dimensional.  Viability supplies a nonzero member of
    $\operatorname{Span}(S)$, so $S$ contains a nonzero vector.  Its singleton
    spans the left Hilbert space and is exactly viable.  This formulation
    avoids the quotients by the chain length used for nonempty chains.
\end{proof}

\begin{lemma}[The size-trimming selection is viable before the ground state]
    \label{thm:lvv_size_trimmed_family}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.sizeTrimmingSelection_subset_span,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_le_one_of_mem_sizeTrimmingSelection,
        TNLean.MPS.AreaLaw.ChainHamiltonian.card_sizeTrimmingSelection_le,
        TNLean.MPS.AreaLaw.ChainHamiltonian.isViable_sizeTrimmingSelection}
    \leanok
    \uses{def:lvv_size_trimming_selection,def:lvv_viable_set,def:lvv_viable_mps}
    Let $L\ge1$ and $q\ge2$, and let $H$ have positive semidefinite terms of
    norm at most one. In the setting of
    Definition~\ref{def:lvv_size_trimming_selection}, put
    \begin{align}
        K & =\bigl(2\lceil(qD)^2/\eta\rceil+1\bigr)^{2(qD)^2}.\notag
    \end{align}
    The size-trimming selection $T$ of $S$ satisfies
    \begin{align}
        |T| & \le(D+1)DK,\notag                      \\
        T   & \subseteq\operatorname{Span}(S).\notag
    \end{align}
    Every member of $T$ has norm at most one.
    For every $E_0\in\mathbb R$ and every $\Gamma$, if $\Gamma$ is a normalized
    ground state of $H$ of energy $E_0$ with gap at least $\Delta$, then
    $(k,c/L)$-viability of $S$ for $\Gamma$ implies
    $(k,1/12)$-viability of $T$ for $\Gamma$.
    In this case there are $m\le D$, an isometry
    $W:\C^m\to\mathcal H_R$, and a unit vector
    $u\in\mathcal H_L\otimes\C^m$ such that every column
    $u_a=(I\otimes\langle a|)u$ belongs to $T$ and
    $|\langle\Gamma,(I\otimes W)u\rangle|\ge11/12$.

    This is an auxiliary version of
    \cite[Claim~2.7, source lines 472--478 and 520--527]{Landau2015Polynomial}.
    It uses the library bond constant, a union over actual bond dimensions,
    and the corrected matrix-space exponent. These qualifications are
    recorded in~\cite{gap:lvv13_constant_bond,gap:lvv13_trace_norm_net};
    the printed quantitative AKLV dependence remains separate.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_gluing_hamiltonian_decomposition,thm:chain_boundary_trace_norm_net,
        thm:lvv_finite_boundary_selection,thm:lvv_uniform_leading_bounds_library_bond,
        thm:chain_gluing_left_extension,thm:lvv_left_columns_right_extension}
    The decomposition $H=H_L+B$ of the left and boundary parts has the stated
    supports and $\|B\|\le2L$. Each net $N_m$ has at most $K$ points, so
    \begin{align}
        |T| & \le\sum_{m=0}^{D}|T_m|
        \le\sum_{m=0}^{D}m|N_m|\le(D+1)DK,\notag
    \end{align}
    and every $T_m$ lies in $\operatorname{Span}(S)$ by
    Lemma~\ref{thm:lvv_finite_boundary_selection}.

    Now fix a ground-state pair $E_0,\Gamma$ for which $S$ is
    $(k,c/L)$-viable. Lemma~\ref{thm:lvv_uniform_leading_bounds_library_bond}
    supplies an actual bond $m\le D$, an isometry $W$ and a unit left
    state $x$ fixed by the left support projection. Write $\mathcal A$
    for the Hilbert space on $\lbrack0,\max\{k-1,0\})$, intersected with the chain. For
    $\rho_x=|x\rangle\langle x|$ and its boundary state $Y$, one has
    \begin{align}
        \rho_x                 & \ge0,\notag                                      \\
        \tr\rho_x              & =\|x\|^2=1,\notag                                \\
        (P_S\otimes I_m)\rho_x & =\rho_x,\notag                                   \\
        Y                      & =\operatorname{tr}_{\mathcal A}\rho_x\ge0,\notag \\
        \|Y\|_1                & =\tr Y=1.\notag
    \end{align}
    Choose $X\in N_m$ with $\|Y-X\|_1\le\eta$. The density $\rho_x$ is
    feasible at $X$, so $X$ is a feasible point of $N_m$.
    The same uniform theorem applies to the chosen minimizer and its chosen
    leading eigenvector $u_{m,X}$. It gives
    a unit physical extension $v=(I\otimes W)u_{m,X}$ with
    $|\langle\Gamma,v\rangle|\ge11/12$.
    Joining the left coordinates and extending the right factor places
    $v$ in $\operatorname{Span}(T_m)\otimes\mathcal H_R$, hence in
    $\operatorname{Span}(T)\otimes\mathcal H_R$. Thus $v$ witnesses the
    asserted viability. The same chosen eigenvector $u_{m,X}$, after
    joining the left coordinates, has all its columns in $T_m\subseteq T$;
    its norm is one and the original right extension $W$ is an isometry.
\end{proof}

\begin{lemma}[Polynomial number of boundary minimizations]
    \label{thm:lvv_boundary_program_count}
    \lean{TNLean.MPS.AreaLaw.exists_boundaryNets_polynomial_total_cardinality,
        TNLean.MPS.AreaLaw.exists_boundaryNets_sweep_polynomial_total_cardinality}
    \leanok
    Fix a cut, $q\ge1$, $L\ge1$, $c>0$, and a bond bound $D$.  Put
    \begin{align*}
        p & =2(qD)^2, & A&=\lceil2(qD)^2/c\rceil.
    \end{align*}
    For every $0\le m\le D$, there is a $c/(2L)$-net $N_m$ of the
    trace-norm unit ball on the actual site-and-bond boundary space, and
    these nets can be chosen so that
    \begin{align*}
        |N_m|&\le(2A+1)^pL^p,
        &\sum_{m=0}^{D}|N_m|&\le(D+1)(2A+1)^pL^p.
    \end{align*}
    Hence the size-trimming construction poses at most the latter number
    of boundary-constrained minimization problems at one cut.  Across all
    positive cuts, the nets can be chosen simultaneously with
    \begin{align*}
      \sum_{i=1}^{L}\sum_{m=0}^{D}|N_{i,m}|
      \le(D+1)(2A+1)^pL^{p+1}.
    \end{align*}
\end{lemma}
\begin{proof}
    \leanok
    Choose the corrected trace-norm net separately for each actual bond
    dimension.  The same polynomial bound applies to all $D+1$ choices;
    summing these bounds proves the one-cut result.  Summing once more over
    the $L$ positive cuts introduces one further factor of $L$.
\end{proof}
