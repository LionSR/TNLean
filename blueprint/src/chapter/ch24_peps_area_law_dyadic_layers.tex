\section{Translated dyadic neighborhoods}
\label{sec:area_law_dyadic_neighborhoods}

\begin{definition}[Half-open dyadic cells]
    \label{def:area_law_dyadic_cells}
    \lean{TNLean.PEPS.AreaLaw.Geometry.dyadicCell,
        TNLean.PEPS.AreaLaw.Geometry.dyadicCellIndex,
        TNLean.PEPS.AreaLaw.Geometry.dyadicParent}
    \leanok
    Fix an origin $o\in\mathbb R^2$, covering the translated origin used in
    \cite[Section~11]{OpenAI2026AreaLaw}. For every integer $k\ge0$ and index
    $z\in\mathbb Z^2$, define the half-open cell, its index map, and the
    parent map by
    \begin{align}
        C_{k,z} & =o+2^k\bigl(z+[0,1)^2\bigr),                         \\
        c_k(x)  & =\left(\left\lfloor\frac{x_1-o_1}{2^k}\right\rfloor,
        \left\lfloor\frac{x_2-o_2}{2^k}\right\rfloor\right),           \\
        p(z)    & =\left(\left\lfloor\frac{z_1}{2}\right\rfloor,
        \left\lfloor\frac{z_2}{2}\right\rfloor\right).
    \end{align}
\end{definition}

\begin{lemma}[Cell indices and parents]
    \label{thm:area_law_dyadic_indices}
    \lean{TNLean.PEPS.AreaLaw.Geometry.mem_dyadicCell_iff,
        TNLean.PEPS.AreaLaw.Geometry.dyadicCellIndex_succ,
        TNLean.PEPS.AreaLaw.Geometry.dyadicParent_abs_sub_le,
        TNLean.PEPS.AreaLaw.Geometry.dyadicParent_mem_ambientDilation_image,
        TNLean.PEPS.AreaLaw.Geometry.mem_dyadicChildren_iff}
    \leanok
    \uses{def:area_law_dyadic_cells,def:area_law_cell_counts,def:area_law_template}
    Every point $x\in\mathbb R^2$ lies in the unique scale-$k$ cell indexed by
    $c_k(x)$, and passing to parents increases the scale by one:
    \begin{align}
        x\in C_{k,z} & \quad\Longleftrightarrow\quad c_k(x)=z, \\
        c_{k+1}(x)   & =p(c_k(x)).
    \end{align}
    For every integer $C_0\ge0$, two indices $z,w\in\mathbb Z^2$ with
    $\|z-w\|_\infty\le C_0$ satisfy $\|p(z)-p(w)\|_\infty\le C_0$, so $p$ maps
    the $C_0$-neighborhood of any finite index set $P\subset\mathbb Z^2$ into
    the $C_0$-neighborhood of $p(P)$. An index $z$ belongs to the child set
    $\operatorname{Ch}(P)$ if and only if $p(z)\in P$.
\end{lemma}
\begin{proof}\leanok
    The floor inequalities give the cell-membership equivalence, and
    $\lfloor t/2\rfloor=\lfloor\lfloor t\rfloor/2\rfloor$ gives
    $c_{k+1}(x)=p(c_k(x))$. Integer division by $2$ preserves coordinatewise
    bounds of the form $|z_i-w_i|\le C_0$. Finally, every integer $z_i$ has a
    unique decomposition $z_i=2p(z)_i+r_i$ with $r_i\in\{0,1\}$.
\end{proof}

\begin{definition}[Occupied indices, neighborhoods and layers]
    \label{def:area_law_dyadic_layers}
    \lean{TNLean.PEPS.AreaLaw.Geometry.occupiedCellIndices,
        TNLean.PEPS.AreaLaw.Geometry.dyadicNeighborhood,
        TNLean.PEPS.AreaLaw.Geometry.dyadicLayerIndices,
        TNLean.PEPS.AreaLaw.Geometry.dyadicLayer}
    \leanok
    \uses{def:area_law_dyadic_cells,def:area_law_cell_counts,def:area_law_template}
    Let $Z\subset\mathbb Z^2$ be finite and let $C_0\ge0$ be an integer.
    For each integer $k\ge0$, define
    \begin{align}
        \mathcal Z_k & =\{c_k(z):z\in Z\},                          \\
        P_k          & =(\mathcal Z_k+[-C_0,C_0]^2)\cap\mathbb Z^2, \\
        N_k          & =\bigcup_{z\in P_k}C_{k,z},                  \\
        D_k          & =N_{k+1}\setminus N_k,                       \\
        I_k          & =\operatorname{Ch}(P_{k+1})\setminus P_k.
    \end{align}
    The figure below illustrates one endpoint with $C_0=1$: the layer $D_k$ is
    the ring of scale-$k$ cells in $N_{k+1}\setminus N_k$.

\end{definition}

\begin{lemma}[Nested neighborhoods and layer cells]
    \label{thm:area_law_dyadic_nesting}
    \lean{TNLean.PEPS.AreaLaw.Geometry.occupiedCellIndices_succ,
        TNLean.PEPS.AreaLaw.Geometry.mem_dyadicNeighborhood_iff,
        TNLean.PEPS.AreaLaw.Geometry.dyadicNeighborhood_subset_succ,
        TNLean.PEPS.AreaLaw.Geometry.dyadicLayer_eq_iUnion}
    \leanok
    \uses{def:area_law_dyadic_layers,thm:area_law_dyadic_indices}
    For every origin $o$, finite $Z\subset\mathbb Z^2$, integer $C_0\ge0$,
    and integer $k\ge0$,
    \begin{align}
        \mathcal Z_{k+1} & =p(\mathcal Z_k),                            \\
        x\in N_k         & \quad\Longleftrightarrow\quad c_k(x)\in P_k, \\
        N_k              & \subseteq N_{k+1},                           \\
        D_k              & =\bigcup_{z\in I_k}C_{k,z}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Applying $c_{k+1}(z)=p(c_k(z))$ to $z\in Z$ gives
    $\mathcal Z_{k+1}=p(\mathcal Z_k)$. Because the scale-$k$ cells partition
    $\mathbb R^2$, $x\in N_k$ if and only if $c_k(x)\in P_k$. Since $p$ maps
    $P_k$ into $P_{k+1}$, every point of $N_k$ has $c_{k+1}(x)=p(c_k(x))\in P_{k+1}$,
    so $N_k\subseteq N_{k+1}$. Finally, $x\in N_{k+1}$ if and only if
    $p(c_k(x))\in P_{k+1}$, that is $c_k(x)\in\operatorname{Ch}(P_{k+1})$;
    removing $N_k$ removes the indices in $P_k$.
\end{proof}

\begin{lemma}[Uniform number of layer cells]
    \label{thm:area_law_dyadic_layer_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.card_dyadicLayerIndices_le,
        TNLean.PEPS.AreaLaw.Geometry.card_dyadicLayerIndices_boundary_le}
    \leanok
    \uses{thm:area_law_cell_counts,thm:area_law_dyadic_nesting}
    For every origin $o$, finite $Z\subset\mathbb Z^2$, integer $C_0\ge0$,
    and integer $k\ge0$, the layer-cell counts in
    \cite[Section~11]{OpenAI2026AreaLaw} satisfy
    \begin{align}
        |I_k| & \le4(2C_0+1)^2|Z|.
    \end{align}
    When $Z$ is the set of endpoints of the boundary edges $\partial_\Lambda A$
    of a cut $A$ in a finite induced domain $\Lambda$, with $b=|\partial_\Lambda A|$,
    \begin{align}
        |I_k| & \le8(2C_0+1)^2b.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $|\mathcal Z_{k+1}|\le|Z|$, its $C_0$-neighborhood $P_{k+1}$ has at
    most $(2C_0+1)^2|Z|$ elements, and $\operatorname{Ch}(P_{k+1})$ has at most
    $4|P_{k+1}|$ elements. Each edge in $\partial_\Lambda A$ contributes at
    most two endpoints to $Z$, so $|Z|\le 2b$.
\end{proof}


\begin{lemma}[Exhaustion of the plane]
    \label{thm:area_law_dyadic_exhaustion}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_mem_dyadicNeighborhood,
        TNLean.PEPS.AreaLaw.Geometry.iUnion_dyadicNeighborhood_eq_univ}
    \leanok
    \uses{def:area_law_dyadic_layers,thm:area_law_dyadic_indices}
    For every origin $o$, nonempty finite set $Z\subset\mathbb Z^2$, and
    integer $C_0\ge2$, the exhaustion property of
    \cite[Section~11]{OpenAI2026AreaLaw} gives
    \begin{align}
        \bigcup_{k\ge0}N_k & =\mathbb R^2.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    For any $z\in Z$ and $x\in\mathbb R^2$, choosing $k\ge0$ with
    $2^k>|x_1-o_1|+|x_2-o_2|+|z_1-o_1|+|z_2-o_2|$ places
    $(x_i-o_i)/2^k$ and $(z_i-o_i)/2^k$ in $(-1,1)$, so their floors
    belong to $\{-1,0\}$ and $\|c_k(x)-c_k(z)\|_\infty\le1\le C_0$. Since
    $c_k(z)\in\mathcal Z_k$, it follows that $c_k(x)\in P_k$ and $x\in N_k$.
\end{proof}


\begin{lemma}[Closures of cells and layers]
    \label{thm:area_law_dyadic_closures}
    \lean{TNLean.PEPS.AreaLaw.Geometry.closure_dyadicCell,
        TNLean.PEPS.AreaLaw.Geometry.convex_closure_dyadicCell,
        TNLean.PEPS.AreaLaw.Geometry.closure_dyadicLayer_eq_iUnion}
    \leanok
    \uses{thm:area_law_dyadic_nesting}
    For every origin $o$, integer $k\ge0$, cell index $z\in\mathbb Z^2$,
    finite set $Z\subset\mathbb Z^2$, and integer $C_0\ge0$, the closed cells
    and layers entering the separation estimates of
    \cite[Section~11]{OpenAI2026AreaLaw} satisfy
    \begin{align}
        \overline{C_{k,z}} & =o+2^k\bigl(z+[0,1]^2\bigr),           \\
        \overline{D_k}     & =\bigcup_{z\in I_k}\overline{C_{k,z}},
    \end{align}
    and in particular $\overline{C_{k,z}}$ is convex.
\end{lemma}
\begin{proof}\leanok
    Since $2^k>0$, the closure of each half-open coordinate interval is the
    corresponding closed interval, and closure commutes with finite products
    and with the finite union $D_k=\bigcup_{z\in I_k}C_{k,z}$. As a product of
    closed intervals, $\overline{C_{k,z}}$ is convex.
\end{proof}
