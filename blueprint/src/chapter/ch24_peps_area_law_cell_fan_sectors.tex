\section{Allowed lines and the eight open fan triangles}
\label{sec:area_law_cell_fan_sectors}

Fix an origin $o\in\mathbb R^2$, a dyadic exponent $\ell\in\mathbb N$, and a
cell index $z\in\mathbb Z^2$. Subdivide every side of the closed dyadic square
at its midpoint to obtain eight triangles $P_e$ with common center $c$.
These are the elementary sectors in the local fan construction of
\cite[Section~11]{OpenAI2026AreaLaw}. Put $\pi_1(x)=x_1$, $\pi_2(x)=x_2$, and
$\Phi=\{\pi_1,\pi_2,\pi_1+\pi_2,\pi_1-\pi_2\}$.
The level lines $\phi(x)=\phi(c)$ for $\phi\in\Phi$ are the four allowed lines
through $c$.

\begin{theorem}[Allowed lines miss open fan triangles]
    \label{thm:area_law_cell_fan_sector_directions}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_interior_not_isAllowedSlope_sub_center}
    \leanok
    \uses{def:area_law_cell_fan,def:area_law_template}
    Each of the eight triangles $P_e$ satisfies
    \begin{align}
        \operatorname{int}(P_e)\cap
        \bigcup_{\phi\in\Phi}\{x\in\mathbb R^2:\phi(x)=\phi(c)\}
         & =\varnothing.
    \end{align}
    Equivalently, for every $x\in\operatorname{int}(P_e)$, the displacement
    $x-c$ has none of the slopes $0,\infty,1,-1$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_unsplit_side_endpoints,
        thm:area_law_elementary_side_geometry}
    Let $\phi\in\Phi$. From the whole-side endpoint and midpoint-subdivision
    formulas, the two perimeter
    vertices $a_e,b_e$ of $P_e$ lie in a single closed half-plane through $c$:
    either $\phi(a_e),\phi(b_e)\ge\phi(c)$ or
    $\phi(a_e),\phi(b_e)\le\phi(c)$.

    In the first case, convexity places $P_e$ in
    $\phi^{-1}([\phi(c),\infty))$. Because $\phi$ is a surjective continuous
    linear functional, it is an open map, so the interior of this half-plane
    is $\phi^{-1}((\phi(c),\infty))$. Hence $\phi(x)>\phi(c)$ for every
    $x\in\operatorname{int}(P_e)$. Replacing $\phi$ by $-\phi$ handles the
    second case.
\end{proof}
