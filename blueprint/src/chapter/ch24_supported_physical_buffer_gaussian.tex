\section{A supported Gaussian operator for the physical buffer}

\begin{definition}[Localization constants of the doubled square Hamiltonian]
    \label{def:peps_square_localization_constants}
    \lean{TNLean.PEPS.Approximation.squareLocalizationPrefactor,
        TNLean.PEPS.Approximation.squareLocalizationVelocity,
        TNLean.PEPS.Approximation.squareLocalizationPrefactor_pos,
        TNLean.PEPS.Approximation.squareLocalizationVelocity_nonneg,
        TNLean.PEPS.Approximation.squareLocalizationPrefactor_eq,
        TNLean.PEPS.Approximation.squareLocalizationVelocity_eq}
    \leanok
    \uses{def:area_graph_ball_localization_constants}
    For $J\in\mathbb R$, with $C_0$ and $v$ as in
    Definition~\ref{def:area_graph_ball_localization_constants}, define the
    range-$1$, term-bound-$2J$ constants
    \begin{align}
        C_\square    & =2C_0(1)=\frac{64e^{3/2}}{1-e^{-1/2}}>0, &
        v_\square(J) & =v(1,2J)=320e^2J
        \label{eq:peps_square_localization_constants}
    \end{align}
    (so $v_\square(J)\geq0$ for $J\geq0$), matching the Lieb--Robinson and
    localization constants $C,v$ of \cite{OpenAI2026PolynomialPEPS},
    \texttt{02-information.tex}, lines 486--508, at decay rate $\mu=1/2$.
\end{definition}

\begin{theorem}[Supported Gaussian approximation of the physical buffer]
    \label{thm:peps_supported_physical_buffer_gaussian}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.exists_supported_buffer_gaussian_integral,
        TNLean.PEPS.Approximation.SquareHamiltonian.supported_buffer_gaussian_integral}
    \leanok
    \uses{def:polynomial_peps_model,def:qc_site_pair_coordinates,
        def:peps_frame_sheet_swap,def:qc_site_expectation,
        def:peps_square_localization_constants}
    Fix $q\ge1$ and $J\ge0$, and set $C=C_\square$, $v=v_\square(J)$, and
    $c=1/2$ from Definition~\ref{def:peps_square_localization_constants}
    (or positive constants $C,v+1,c$). Let $L\ge0$ and let
    $H=\sum_x h_x+\sum_e h_e$ be a Hermitian Hamiltonian on
    $\mathcal V=\{0,\ldots,L-1\}^2$ with each site or nearest-neighbour term
    satisfying $\|h_x\|,\|h_e\|\le J$. For disjoint $R,C_0\subseteq\mathcal V$,
    put $Q=(R\cup C_0)^c$, $\mathcal X_D=[q]^D$, and
    $P=[|\mathcal X_R|]\times[|\mathcal X_{C_0}|]$. Fixed enumerations of
    $\mathcal X_R,\mathcal X_{C_0}$ induce a bijection
    $s:\mathcal X_{\mathcal V}\to P\times\mathcal X_Q$ and doubled regrouping
    \begin{align}
        g(x,y) & =((s(x)_1,s(y)_1),(s(x)_2,s(y)_2)). \notag
    \end{align}
    With $p$ pairing the two copy values per site and $a:\mathcal V\to\Lambda$
    the bijection to the integer rectangle, put $u=a_*\circ p$ and write
    $\mathcal R_b(A)(x,y)=A(b^{-1}x,b^{-1}y)$. Let $F_R$ swap the two copies
    on $R$. For any matrix $W$ on $\mathcal X_Q\times\mathcal X_Q$, define
    \begin{align}
        K  & =\mathcal R_u(H\otimes I+I\otimes H),                &
        F  & =\mathcal R_u(F_R), \notag                             \\
        K' & =FKF^*,                                              &
        V  & =\mathcal R_{u\circ g^{-1}}(I_{P\times P}\otimes W).
        \label{eq:peps_supported_buffer_operators}
    \end{align}
    With $\partial R$ the nearest-neighbour edges meeting both $R$ and $R^c$,
    set $Z_R=\bigcup_{\{x,y\}\in\partial R}\{x,y\}$ and, for $\ell\ge0$,
    $B=N_\ell(a(Q)\cup a(Z_R))$ in the induced graph of $\Lambda$.

    There is a differentiable path $U$ of unitaries supported on $B$ solving
    \begin{align}
        U(0)  & =I,                                 &
        U'(t) & =iE_B(e^{itK'}(K'-K)e^{-itK'})U(t).
        \label{eq:peps_supported_buffer_unitary_ode}
    \end{align}
    For every $\eta,T\ge0$, with $\gamma_\eta$ the centered Gaussian measure of
    variance $\eta$, the operators
    \begin{align}
        M_{\eta,T}
         & =\int_{[-T,T]}E_B(e^{itK'}Ve^{-itK'})U(t)\,d\gamma_\eta(t), &
        \widetilde M_{\eta,T}
         & =\int_{[-T,T]}e^{itK'}Ve^{-itK}\,d\gamma_\eta(t)
        \label{eq:peps_supported_buffer_integrals}
    \end{align}
    satisfy $\operatorname{supp}(M_{\eta,T})\subseteq B$ and, including empty
    grids or regions and $\eta=0$ or $T=0$
    (\cite{OpenAI2026PolynomialPEPS}),
    \begin{align}
        \|M_{\eta,T}\| & \le\|W\|,                           &
        \|M_{\eta,T}-\widetilde M_{\eta,T}\|
                       & \le\|W\|CL^2(1+8JL^2T)e^{vT-c\ell}.
        \label{eq:peps_supported_buffer_bounds}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_physical_buffer_coordinates,
        thm:peps_swapped_doubled_boundary_bound,
        thm:swapped_square_hamiltonian_interaction,
        thm:area_supported_gaussian_integral,
        def:peps_square_localization_constants}
    Theorem~\ref{thm:swapped_square_hamiltonian_interaction} realizes $K'$ via
    a range-$1$ interaction family of local dimension $q^2$ and term bound
    $2J\ge0$ using the Hermitian parts of the terms of $H$. By
    Theorem~\ref{thm:peps_physical_buffer_coordinates}, $V$ is supported on
    $a(Q)$ with $\|V\|\le\|W\|$; by
    Theorem~\ref{thm:peps_swapped_doubled_boundary_bound}, $K'-K$ is supported
    on $a(Z_R)$ with $\|K'-K\|\le8JL^2$. Applying
    Theorem~\ref{thm:area_supported_gaussian_integral} on $B$ with $|\Lambda|=L^2$
    and the constants (\ref{eq:peps_square_localization_constants}) yields $U$
    solving (\ref{eq:peps_supported_buffer_unitary_ode}) independently of
    $\eta,T$, the bound $\|M_{\eta,T}\|\le\|V\|\le\|W\|$, and
    \begin{align}
        \|M_{\eta,T}-\widetilde M_{\eta,T}\|
         & \le\|V\|CL^2(1+\|K'-K\|T)e^{vT-c\ell}
        \le\|W\|CL^2(1+8JL^2T)e^{vT-c\ell}. \notag
    \end{align}
\end{proof}

\subsection{The partially swapped Hamiltonian at interaction range $R$}

The statements below extend the range-$1$ boundary and Gaussian buffer
estimates on the open square above to Hamiltonians of interaction range $R$
on a finite domain $\Lambda$.

\begin{definition}[Crossing supports and the swap coefficient]
    \label{def:peps_crossing_supports}
    \lean{TNLean.Lattice.crossingSites,
        TNLean.Lattice.swapBoundaryCoefficient}
    \leanok
    \uses{def:area_law_hamiltonian}
    For a finite domain $\Lambda$, an integer $R\ge0$, and $A\subseteq\Lambda$,
    a range-$R$ support $X$ \emph{crosses} $A$ if $X\cap A\ne\varnothing$ and
    $X\not\subseteq A$. Denote the union of all crossing supports by $Z_R(A)$,
    and set $v_R=1+2R(R+1)$, $\mu_R=2^{v_R-1}$, and $c_R=4\mu_R$.
\end{definition}

\begin{lemma}[Crossing supports lie near both sides and are few]
    \label{lem:peps_crossing_supports_near_and_few}
    \lean{TNLean.Lattice.exists_walks_of_mem_crossingSites,
        TNLean.Lattice.card_admissibleSupport_le}
    \leanok
    \uses{def:peps_crossing_supports}
    Every $z\in Z_R(A)$ is joined by walks of length at most $R$ to $A$ and to
    $A^c$, and there are at most $\mu_R|\Lambda|$ supports of range $R$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:area_law_crossing_supports}
    Any $z\in Z_R(A)$ belongs to a range-$R$ support $X$ containing some
    $x\in A$ and $y\notin A$, so walks of length at most $R$ join $z$ to $x$
    and $y$. Each support is nonempty and at most $\mu_R$ supports contain any
    given site by Theorem~\ref{thm:area_law_crossing_supports}, giving at most
    $\mu_R|\Lambda|$ supports in total.
\end{proof}

\begin{theorem}[The partially swapped Hamiltonian at range $R$]
    \label{thm:peps_swapped_local_hamiltonian}
    \lean{TNLean.Lattice.LocalHamiltonian.swappedDoubled,
        TNLean.Lattice.LocalHamiltonian.swappedDoubled_operator,
        TNLean.Lattice.LocalHamiltonian.sheetSwapOp_conj_doubled_operator_sub_eq_sum_crossing,
        TNLean.Lattice.LocalHamiltonian.swapped_doubled_boundary_bound}
    \leanok
    \uses{def:area_law_hamiltonian,def:qc_site_pair_coordinates,
        def:peps_frame_sheet_swap,def:peps_crossing_supports}
    Let $H=\sum_Xh_X$ be a range-$R$ Hamiltonian on a finite domain $\Lambda$
    with Hermitian terms satisfying $\|h_X\|\le J$, let $A\subseteq\Lambda$,
    and in paired-site coordinates set $K=H\otimes I+I\otimes H$ and
    $K'=F_AKF_A^*$. Then $K'$ is a range-$R$ Hamiltonian of local dimension
    $q^2$ on the same supports with term norms at most $2J$, and $K'-K$ sums
    only over supports crossing $A$, so $\operatorname{supp}(K'-K)\subseteq Z_R(A)$
    and (extending Theorem~\ref{thm:peps_swapped_doubled_boundary_bound})
    \begin{align}
        \|K'-K\| & \le c_RJ|\Lambda|.
        \label{eq:peps_swapped_local_hamiltonian_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_crossing_supports_near_and_few}
    Swap conjugation preserves support, scales doubled term norms by at most $2$,
    and commutes with terms supported inside $A$ or $A^c$. By
    Lemma~\ref{lem:peps_crossing_supports_near_and_few}, at most
    $\mu_R|\Lambda|$ crossing supports remain, each contributing at most $4J$
    to $\|K'-K\|$, yielding (\ref{eq:peps_swapped_local_hamiltonian_bound}).
\end{proof}

\begin{theorem}[Edge-boundary norm of the partially swapped Hamiltonian]
    \label{thm:peps_swapped_doubled_edge_boundary}
    \lean{TNLean.Lattice.LocalHamiltonian.swapped_doubled_boundary_bound_edgeBoundary}
    \leanok
    \uses{def:area_law_hamiltonian,def:qc_site_pair_coordinates,
        def:peps_frame_sheet_swap,def:peps_crossing_supports}
    Under the hypotheses of Theorem~\ref{thm:peps_swapped_local_hamiltonian},
    with $\partial_\Lambda A$ the edge boundary of $A$ in the induced
    nearest-neighbour graph of $\Lambda$,
    \begin{align}
        \|K'-K\| & \le c_Rv_RJ|\partial_\Lambda A|,
        \label{eq:peps_swapped_doubled_edge_boundary}
    \end{align}
    extending \cite{OpenAI2026PolynomialPEPS}, \texttt{02-information.tex},
    lines 416--424 and 513--524, to range $R$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_swapped_local_hamiltonian,
        thm:peps_sheet_swap_doubled_term_norm,thm:area_law_crossing_supports}
    By Theorems~\ref{thm:peps_swapped_local_hamiltonian} and
    \ref{thm:peps_sheet_swap_doubled_term_norm}, only crossing supports
    contribute to $K'-K$, each with norm at most $4J$.
    Theorem~\ref{thm:area_law_crossing_supports} bounds their count by
    $v_R\mu_R|\partial_\Lambda A|$, and $c_R=4\mu_R$ yields
    (\ref{eq:peps_swapped_doubled_edge_boundary}).
\end{proof}

\begin{theorem}[Supported Gaussian approximation of the physical buffer at range $R$]
    \label{thm:peps_supported_physical_buffer_gaussian_range}
    \lean{TNLean.Lattice.LocalHamiltonian.supported_buffer_gaussian_integral}
    \leanok
    \uses{thm:peps_swapped_local_hamiltonian,
        thm:peps_physical_buffer_coordinates,
        thm:area_supported_gaussian_integral,
        def:area_graph_ball_localization_constants}
    Fix $q\ge1$ and $J\ge0$, let $H$ be a range-$R$ Hamiltonian on a finite
    domain $\Lambda$ with term norms at most $J$, and let $A,C\subseteq\Lambda$
    be disjoint with $Q=(A\cup C)^c$. Define $K,K',V$ as in
    (\ref{eq:peps_supported_buffer_operators}) in $\Lambda$-coordinates, and
    set $B=N_\ell(Q\cup Z_R(A))$ for $\ell\ge0$. Then the conclusions of
    Theorem~\ref{thm:peps_supported_physical_buffer_gaussian} hold with
    \begin{align}
        \|M_{\eta,T}-\widetilde M_{\eta,T}\|
         & \le\|W\|\,2C_0(R)|\Lambda|(1+c_RJ|\Lambda|T)e^{v(R,2J)T-\ell/2}.
        \label{eq:peps_supported_buffer_range_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Combine Theorem~\ref{thm:peps_swapped_local_hamiltonian}
    ($\operatorname{supp}(K'-K)\subseteq Z_R(A)$ and
    $\|K'-K\|\le c_RJ|\Lambda|$) and
    Theorem~\ref{thm:peps_physical_buffer_coordinates}
    ($\operatorname{supp}(V)\subseteq Q$ and $\|V\|\le\|W\|$) with
    Theorem~\ref{thm:area_supported_gaussian_integral} at range $R$ and term
    bound $2J$.
\end{proof}
