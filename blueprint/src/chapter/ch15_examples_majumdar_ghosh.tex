\section{The Majumdar--Ghosh state}\label{sec:majumdar_ghosh}
%% ===================================================================

The Majumdar--Ghosh state is the exact dimer ground state of the spin-$\tfrac12$
antiferromagnetic Heisenberg chain with next-nearest-neighbour coupling at the
Majumdar--Ghosh point $J_2 = J_1 / 2$~\cite{MajumdarGhosh1969,MajumdarGhosh1969II}:
$H=\sum \boldsymbol{S}_i\cdot\boldsymbol{S}_{i+1}
    +\tfrac12\sum\boldsymbol{S}_i\cdot\boldsymbol{S}_{i+2}$.
On an even periodic chain it is the symmetric superposition of the two
nearest-neighbour singlet coverings $(1,2)(3,4)\cdots$ and
$(2,3)(4,5)\cdots(N,1)$, forming the one-dimensional prototype of a
resonating valence-bond state. The review
\cite[Appendix~A, ``The Majumdar-Ghosh model'']{Cirac2021Matrix} writes it as a
matrix product state with a tensor built from the singlet matrix
\begin{align}
    Y & =
    \begin{pmatrix}
        0 & -1 \\
        1 & 0
    \end{pmatrix},
    \label{eq:mg_singlet_matrix}
\end{align}
whose entry $Y_{ab}$ is the coefficient of $\ket{ab}$ in the singlet
$\ket{10}-\ket{01}$.

\begin{definition}[Majumdar--Ghosh tensor of the review]%
    \label{def:majumdar_ghosh_review_tensor}
    \lean{MPSTensor.majumdarGhoshSingletY, MPSTensor.majumdarGhoshReviewBond,
        MPSTensor.majumdarGhoshReviewTensor}
    \leanok
    \uses{def:mps_tensor}
    For $d=2$, let $E_{ab}$ denote the matrix units of $\MN{3}$, and let
    \begin{align}
        B^0 & = E_{02}+E_{20},
            & B^1              & = E_{12}+E_{21}.
        \label{eq:mg_review_bond}
    \end{align}
    In \cite[Appendix~A, ``The Majumdar-Ghosh model'']{Cirac2021Matrix}, the
    review prints the tensor
    $A=\bigl[\ket0{\bigl[(02|+(20|\bigr]}+\ket1{\bigl[(12|+(21|\bigr]}\bigr]\otimes Y$.
    Read directly, with $(ab|$ the bra $E_{ab}$ of the pair of virtual indices,
    $\otimes$ the Kronecker product, and $Y$ as in~(\ref{eq:mg_singlet_matrix}),
    this formula defines the tensor
    $\widetilde A^i=B^i\otimes Y\in\MN{6}$ on the bond
    $\C^3\otimes\C^2\cong\C^6$.
\end{definition}

The periodic coefficients are traces of products of these matrices.
As recorded in \cite{gap:rmp_majumdar_ghosh_tensor_gap} and
Lemma~\ref{thm:majumdar_ghosh_review_mpv}, the direct reading above and the
singlet-bond reading produce different periodic vectors.
Lemma~\ref{thm:majumdar_ghosh_dimer_superposition} identifies the
singlet-covering state, and Lemmas~\ref{thm:majumdar_ghosh_left_canonical},
\ref{thm:majumdar_ghosh_not_injective}, and \ref{thm:majumdar_ghosh_not_normal}
show that its bond-dimension-three tensor is left-canonical, non-injective, and
non-normal. Finally, Lemma~\ref{thm:majumdar_ghosh_ground_space} establishes
its two-dimensional parent ground space on even rings of at least four sites,
and Lemma~\ref{thm:majumdar_ghosh_review_hamiltonian} compares the parent
Hamiltonian with the physical Majumdar--Ghosh Hamiltonian.


\begin{lemma}[Coefficients of the review tensor factor]%
    \label{thm:majumdar_ghosh_review_coeff}
    \lean{MPSTensor.majumdarGhoshReview_coeff}
    \leanok
    \uses{def:majumdar_ghosh_review_tensor, def:eval_word}
    For every word $w$,
    $\tr\bigl(\widetilde A^{w}\bigr)=\tr\bigl(B^{w}\bigr)\,\tr\bigl(Y^{|w|}\bigr)$.
\end{lemma}

\begin{proof}
    \leanok
    By induction on $w$ and the mixed-product rule
    $(B^i\otimes Y)(M\otimes Y^n)=B^iM\otimes Y^{n+1}$, the word evaluation is
    $\widetilde A^{w}=B^{w}\otimes Y^{|w|}$ up to the identification of the
    bond with $\C^6$, which preserves traces. The trace of a Kronecker product
    is the product of the traces.
\end{proof}

\begin{definition}[Pair coverings]\label{def:majumdar_ghosh_pair_covering}
    \lean{MPSTensor.pairProduct, MPSTensor.pairCoveringEven,
        MPSTensor.pairCoveringOdd, MPSTensor.majumdarGhoshSinglet}
    \leanok
    Let $f(a,b)$ be a pair amplitude on two sites. For a chain of $N$ sites
    with configurations $\sigma=(\sigma_0,\ldots,\sigma_{N-1})$, the even and
    odd pair coverings are
    \begin{align}
        P^{\mathrm{even}}_f(\sigma) & = \prod_{k=0}^{\lfloor N/2\rfloor-1}
        f(\sigma_{2k},\sigma_{2k+1}),
                                    &
        P^{\mathrm{odd}}_f(\sigma)  & = \prod_{k=0}^{\lfloor N/2\rfloor-1}
        f(\sigma_{2k+1},\sigma_{(2k+2)\bmod N}).
        \label{eq:mg_pair_covering}
    \end{align}
    For even $N$ these are the coverings $(1,2)(3,4)\cdots(N-1,N)$ and
    $(2,3)(4,5)\cdots(N,1)$ of the periodic chain by the pair state
    $\sum_{ab}f(a,b)\ket{ab}$. The normalized singlet amplitude is
    $s(a,b)=Y_{ab}/\sqrt2$. For a word $w$, the pair product is the product of
    $f$ over the consecutive pairs $(w_0,w_1),(w_2,w_3),\ldots$, and zero if
    $|w|$ is odd.
\end{definition}

\begin{lemma}[Pair-covering form of periodic vectors]%
    \label{thm:mps_pair_covering_form}
    \lean{MPSTensor.evalWord_apply_hub_eq_pairProduct,
        MPSTensor.coeff_cons_eq_pairProduct_add,
        MPSTensor.mpv_eq_pairCovering_of_hub,
        MPSTensor.mpv_eq_zero_of_hub_of_odd, MPSTensor.pairProduct_eq_prod,
        MPSTensor.pairProduct_eq_zero_of_odd,
        MPSTensor.pairProduct_ofFn_eq_pairCoveringEven,
        MPSTensor.pairProduct_rotate_ofFn_eq_pairCoveringOdd}
    \leanok
    \uses{def:mpv, def:majumdar_ghosh_pair_covering}
    Let $K$ be a tensor with a bond level $z$ such that $K^i_{xy}=0$ whenever
    $x\neq z$ and $y\neq z$, and $K^i_{zz}=0$ for every $i$. With
    $f(a,b)=(K^aK^b)_{zz}$, the entry $(K^w)_{zz}$ is the pair product of $w$.
    For even $N>0$,
    \begin{align}
        \ket{V^{(N)}(K)} & = P^{\mathrm{even}}_f+P^{\mathrm{odd}}_f,
        \label{eq:mg_pair_vector}
    \end{align}
    and for odd $N$ the periodic vector vanishes.
\end{lemma}

\begin{proof}
    \leanok
    The row $z$ of $K^aK^b$ is $f(a,b)$ times the $z$-th unit row, since a path
    leaving $z$ must enter a level $x\neq z$ and then return to $z$. Induction
    on the word in steps of two letters gives the entry $(K^w)_{zz}$, and a word
    of length one has $(K^a)_{zz}=0$. For a word $a\,w$,
    \begin{align}
        \tr(K^aM) & = (K^aM)_{zz}+(MK^a)_{zz},
        \qquad M=K^{w},
        \label{eq:mg_pair_trace}
    \end{align}
    because the rows $x\neq z$ of $K^a$ are supported in column $z$ and
    $K^a_{zz}=0$. The two terms are the $(z,z)$ entries for $a\,w$ and for its
    rotation $w\,a$, which are the two pair coverings of the configuration.
    Pair products of words of odd length vanish.
\end{proof}

\begin{definition}[Review tensor with the singlet on the spin levels]%
    \label{def:majumdar_ghosh_bond_singlet_tensor}
    \lean{MPSTensor.majumdarGhoshBondSingletTensor}
    \leanok
    \uses{def:majumdar_ghosh_review_tensor}
    The tensor $C^i=B^i\,(Y\oplus1)\in\MN{3}$, with $B^i$ as
    in~(\ref{eq:mg_review_bond}), places the singlet~(\ref{eq:mg_singlet_matrix})
    on the two spin levels $0,1$ of the bond and carries the empty level $2$
    through. Explicitly $C^0=E_{02}-E_{21}$ and $C^1=E_{12}+E_{20}$.
\end{definition}

\begin{lemma}[The singlet-bond tensor gives the dimer superposition]%
    \label{thm:majumdar_ghosh_bond_singlet_mpv}
    \lean{MPSTensor.majumdarGhoshBondSinglet_mpv_eq_pairCovering}
    \leanok
    \uses{def:majumdar_ghosh_bond_singlet_tensor, thm:mps_pair_covering_form}
    With the tensor of \cite[Appendix~A, ``The Majumdar-Ghosh model'']{Cirac2021Matrix}
    read as in \cite{gap:rmp_majumdar_ghosh_tensor_gap}, every even $N>0$
    satisfies $\ket{V^{(N)}(C)}=P^{\mathrm{even}}_Y+P^{\mathrm{odd}}_Y$, the
    sum of the two nearest-neighbour coverings by the singlet
    $\ket{10}-\ket{01}$.
\end{lemma}

\begin{proof}
    \leanok
    Direct multiplication gives $(C^aC^b)_{22}=Y_{ab}$, and the level $z=2$
    satisfies the hypotheses of Lemma~\ref{thm:mps_pair_covering_form} for
    $C$.
\end{proof}

The representative used below has the same structure, with the distinguished
level $0$ and the normalized singlet as pair amplitude.

\begin{definition}[Majumdar--Ghosh tensor]\label{def:majumdar_ghosh_tensor}
    \lean{MPSTensor.majumdarGhoshTensor}
    \leanok
    \uses{def:mps_tensor}
    For $d = 2$ (spin-$\tfrac12$) and $D = 3$, the Majumdar--Ghosh tensor is
    the family $\{A^0, A^1\} \subset \MN{3}$ defined by
    \begin{align}
        A^0 & =
        \begin{pmatrix}
            0               & 1 & 0 \\
            0               & 0 & 0 \\
            \tfrac1{\sqrt2} & 0 & 0
        \end{pmatrix},
        \qquad
        A^1 =
        \begin{pmatrix}
            0                & 0 & 1 \\
            -\tfrac1{\sqrt2} & 0 & 0 \\
            0                & 0 & 0
        \end{pmatrix}.
        \notag
    \end{align}
    The level $0$ is the closed link between two dimers, the levels $1,2$ carry
    the spin of an open singlet partner, and the relative sign
    $-\tfrac1{\sqrt2}$ is the antisymmetry of the singlet.
\end{definition}

\begin{lemma}[The Majumdar--Ghosh tensor gives the dimer superposition]%
    \label{thm:majumdar_ghosh_dimer_superposition}
    \lean{MPSTensor.majumdarGhosh_mpv_eq_pairCovering,
        MPSTensor.majumdarGhosh_mpv_eq_zero_of_odd,
        MPSTensor.majumdarGhoshTensor_mul_apply_zero_zero}
    \leanok
    \uses{def:majumdar_ghosh_tensor, thm:mps_pair_covering_form}
    For every even $N>0$, $\ket{V^{(N)}(A)}=P^{\mathrm{even}}_s+P^{\mathrm{odd}}_s$,
    the sum of the two nearest-neighbour coverings by the normalized singlet
    $(\ket{10}-\ket{01})/\sqrt2$. For odd $N$ the periodic vector vanishes.
\end{lemma}

\begin{proof}
    \leanok
    The level $z=0$ satisfies the hypotheses of
    Lemma~\ref{thm:mps_pair_covering_form} for $A$: rows $1,2$ of $A^0$ and $A^1$
    are supported in column $0$, and $A^i_{00}=0$. Direct multiplication gives
    $(A^aA^b)_{00}=Y_{ab}/\sqrt2$.
\end{proof}

\begin{lemma}[Relation to the singlet-bond tensor]%
    \label{thm:majumdar_ghosh_bond_singlet_scaling}
    \lean{MPSTensor.majumdarGhoshBondSinglet_mpv_eq_smul}
    \leanok
    \uses{def:majumdar_ghosh_tensor, thm:majumdar_ghosh_bond_singlet_mpv,
        thm:majumdar_ghosh_dimer_superposition}
    For every even $N>0$,
    $\ket{V^{(N)}(C)}=\sqrt2^{\,N/2}\,\ket{V^{(N)}(A)}$.
\end{lemma}

\begin{proof}
    \leanok
    Both vectors are sums of the two pair coverings, with amplitudes $Y_{ab}$
    and $Y_{ab}/\sqrt2$; each covering is a product of $N/2$ amplitudes.
\end{proof}

\begin{lemma}[The printed review tensor does not give the dimer superposition]%
    \label{thm:majumdar_ghosh_review_mpv}
    \lean{MPSTensor.majumdarGhoshReview_mpv_eq,
        MPSTensor.majumdarGhoshReview_mpv_ne_smul,
        MPSTensor.majumdarGhoshReviewBond_mul_apply_two_two,
        MPSTensor.majumdarGhoshSingletY_sq,
        MPSTensor.majumdarGhoshSingletY_pow_two_mul}
    \leanok
    \uses{def:majumdar_ghosh_review_tensor, thm:majumdar_ghosh_review_coeff,
        thm:mps_pair_covering_form, thm:majumdar_ghosh_dimer_superposition}
    Let $N=2m>0$ and $\delta(a,b)=\delta_{ab}$. Then
    \begin{align}
        \ket{V^{(N)}(\widetilde A)} & =
        2(-1)^m\bigl(P^{\mathrm{even}}_\delta+P^{\mathrm{odd}}_\delta\bigr),
        \label{eq:mg_review_vector}
    \end{align}
    the sum of the two coverings by the pair state $\ket{00}+\ket{11}$, and
    $\ket{V^{(N)}(\widetilde A)}$ is not a scalar multiple of
    $\ket{V^{(N)}(A)}$.
\end{lemma}

\begin{proof}
    \leanok
    The level $z=2$ satisfies the hypotheses of
    Lemma~\ref{thm:mps_pair_covering_form} for $\{B^0,B^1\}$, with
    $(B^aB^b)_{22}=\delta_{ab}$,
    so Lemma~\ref{thm:mps_pair_covering_form} gives
    $\tr(B^{\sigma})=P^{\mathrm{even}}_\delta(\sigma)+P^{\mathrm{odd}}_\delta(\sigma)$.
    Since $Y^2=-I_2$, $\tr(Y^{2m})=2(-1)^m$, and
    Lemma~\ref{thm:majumdar_ghosh_review_coeff} gives~(\ref{eq:mg_review_vector}).
    On the configuration with all spins $0$ the right side
    of~(\ref{eq:mg_review_vector}) equals $4(-1)^m\neq0$, while
    Lemma~\ref{thm:majumdar_ghosh_dimer_superposition} gives
    $V^{(N)}(A)_{0\cdots0}=0$ because $s(0,0)=0$.
\end{proof}

\begin{lemma}[Majumdar--Ghosh tensor is in left-canonical form]%
    \label{thm:majumdar_ghosh_left_canonical}
    \lean{MPSTensor.majumdarGhosh_left_canonical}
    \leanok
    \uses{def:majumdar_ghosh_tensor}
    The Majumdar--Ghosh tensor is isometric:
    $(A^0)^\dagger A^0 + (A^1)^\dagger A^1 = \Id$.
\end{lemma}

\begin{proof}
    \leanok
    Direct multiplication of the two displayed matrices gives
    $(A^0)^\dagger A^0=\diag(\tfrac12,1,0)$ and
    $(A^1)^\dagger A^1=\diag(\tfrac12,0,1)$.
\end{proof}

\begin{lemma}[Transfer map of the Majumdar--Ghosh tensor]%
    \label{thm:majumdar_ghosh_transfer}
    \lean{MPSTensor.majumdarGhosh_transferMap_one}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:transfer_map}
    The transfer map of the Majumdar--Ghosh tensor sends the identity to the
    diagonal matrix $\E_A(\Id) = \diag(2, \tfrac12, \tfrac12)$.
\end{lemma}

\begin{proof}
    \leanok
    Evaluating $\E_A(\Id)=A^0(A^0)^\dagger+A^1(A^1)^\dagger$ on the two
    displayed matrices gives $\diag(2,\tfrac12,\tfrac12)$.
\end{proof}

\begin{lemma}[Majumdar--Ghosh tensor is not injective]%
    \label{thm:majumdar_ghosh_not_injective}
    \lean{MPSTensor.majumdarGhosh_not_isInjective}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:injective}
    The Majumdar--Ghosh tensor is not injective: every element of
    $\spn_{\C}\{A^0, A^1\}$ has vanishing $(2,2)$ entry, so the span is a proper
    subspace of~$\MN{3}$.
\end{lemma}

\begin{proof}
    \leanok
    The matrix unit $E_{22}$ has nonzero $(2,2)$ entry, while every linear
    combination of $A^0$ and $A^1$ has zero $(2,2)$ entry.
\end{proof}

\begin{lemma}[Majumdar--Ghosh tensor is not normal]%
    \label{thm:majumdar_ghosh_not_normal}
    \lean{MPSTensor.majumdarGhosh_not_isNormal}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:normal}
    The Majumdar--Ghosh tensor is not normal: no blocking length makes the
    products of that length span the full matrix algebra~$\MN{3}$. This
    non-normality is the algebraic signature expected from the two-periodic
    dimer-covering structure.
\end{lemma}

\begin{proof}
    \leanok
    The bond carries a two-periodic grading with complementary support sets
    \begin{align}
         & \{(0,1),(0,2),(1,0),(2,0)\}, \notag \\
         & \{(0,0),(1,1),(1,2),(2,1),(2,2)\}.
        \notag
    \end{align}
    Left multiplication with a generator exchanges these sets. Let $E_{ij}$
    denote the $(i,j)$ matrix
    unit. Then, for every blocking length $N$,
    \begin{align}
        N\text{ odd}
         & \Rightarrow
        (\forall w,\ |w|=N \Rightarrow (A^w)_{22}=0)
        \Rightarrow
        E_{22}\notin\spn_{\C}\{A^w: |w|=N\},
        \notag         \\
        N\text{ even}
         & \Rightarrow
        (\forall w,\ |w|=N \Rightarrow (A^w)_{10}=0)
        \Rightarrow
        E_{10}\notin\spn_{\C}\{A^w: |w|=N\}.
        \notag
    \end{align}
    Hence no fixed-length word span is all of $\MN{3}$.
\end{proof}

\begin{remark}\label{rem:majumdar_ghosh_bond_dimension}
    Read directly, the valence-bond formula of \cite{Cirac2021Matrix}
    specifies the six-dimensional tensor of
    Definition~\ref{def:majumdar_ghosh_review_tensor}, which by
    Lemma~\ref{thm:majumdar_ghosh_review_mpv} gives coverings by
    $\ket{00}+\ket{11}$ rather than by singlets. As detailed in
    \cite{gap:rmp_majumdar_ghosh_tensor_gap}, placing the singlet on the spin
    levels of a three-level bond
    (Definition~\ref{def:majumdar_ghosh_bond_singlet_tensor}) recovers the
    Majumdar--Ghosh superposition.
\end{remark}

\begin{definition}[Majumdar--Ghosh Hamiltonian]\label{def:majumdar_ghosh_hamiltonian}
    \lean{MPSTensor.spinHalfOperator, MPSTensor.spinExchange,
        MPSTensor.majumdarGhoshTerm, MPSTensor.majumdarGhoshHamiltonian}
    \leanok
    Let $S^\alpha=\sigma^\alpha/2$ for $\alpha=x,y,z$ be the spin-$\tfrac12$
    operators, and for two sites $j,k$ of a chain let
    $\mathbf S_j\cdot\mathbf S_k=\sum_\alpha S^\alpha_jS^\alpha_k$. Following
    \cite[Appendix~A, ``The Majumdar-Ghosh model'']{Cirac2021Matrix}, on a
    periodic chain of $N$ sites with indices modulo $N$, the Majumdar--Ghosh
    Hamiltonian is
    \begin{align}
        H_N & = \sum_{i}\mathbf S_i\cdot\mathbf S_{i+1}
        +\tfrac12\sum_{i}\mathbf S_i\cdot\mathbf S_{i+2}.
        \label{eq:mg_hamiltonian}
    \end{align}
    The three-site term is
    $h=\tfrac12(\mathbf S_0\cdot\mathbf S_1+\mathbf S_1\cdot\mathbf S_2
        +\mathbf S_0\cdot\mathbf S_2)$ on the sites $0,1,2$ of a window.
\end{definition}

\begin{lemma}[Exchange as a transposition]\label{thm:spin_exchange_swap}
    \lean{MPSTensor.spinExchange_spinHalfOperator_apply, MPSTensor.spinHalfOperator_sum}
    \leanok
    \uses{def:majumdar_ghosh_hamiltonian}
    For distinct sites $j\neq k$,
    $\mathbf S_j\cdot\mathbf S_k=\tfrac12P_{jk}-\tfrac14$, where $P_{jk}$
    exchanges the spins at $j$ and $k$.
\end{lemma}

\begin{proof}
    \leanok
    The completeness relation
    $\sum_\alpha S^\alpha_{pa}S^\alpha_{qb}
        =\tfrac12\delta_{pb}\delta_{qa}-\tfrac14\delta_{pa}\delta_{qb}$ is checked on
    the sixteen index values.
\end{proof}

\begin{lemma}[Three-site ground space of the Majumdar--Ghosh tensor]%
    \label{thm:majumdar_ghosh_local_ground_space}
    \lean{MPSTensor.majumdarGhosh_groundSpace_three_eq_eigenspace}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:ground_space_parent,
        def:majumdar_ghosh_hamiltonian, thm:spin_exchange_swap}
    The three-site local ground space $\mathcal G_3(A)$ equals the eigenspace
    of $h$ for the eigenvalue $-\tfrac38$.
\end{lemma}

\begin{proof}
    \leanok
    By Lemma~\ref{thm:spin_exchange_swap},
    $h=\tfrac14(P_{01}+P_{12}+P_{02})-\tfrac38$, so the eigenspace consists of
    the vectors $v$ with $v(\sigma P_{01})+v(\sigma P_{12})+v(\sigma P_{02})=0$
    for every $\sigma$, that is,
    \begin{align}
        v_{000} & = v_{111} = 0,
                & v_{001}+v_{010}+v_{100} & = 0,
                & v_{011}+v_{101}+v_{110} & = 0.
        \label{eq:mg_spin_half_conditions}
    \end{align}
    Every $\Gamma_3(X)$ satisfies~(\ref{eq:mg_spin_half_conditions}) by
    expansion. Conversely, a vector satisfying these conditions is
    $\Gamma_3(X)$ for the matrix $X$ with $X_{01}=-2v_{110}$,
    $X_{02}=-2v_{001}$, $X_{10}=\sqrt2\,v_{100}$, $X_{20}=-\sqrt2\,v_{011}$ and
    all other entries zero.
\end{proof}

\begin{lemma}[Twisted periodic vectors satisfy the window constraints]%
    \label{thm:twisted_mpv_chain_ground_space}
    \lean{MPSTensor.twistedMPV_window_mem_groundSpace,
        MPSTensor.twistedMPV_mem_chainGroundSpace}
    \leanok
    \uses{def:chain_ground_space}
    Let $G\in\MN{D}$ and $c\in\C$ satisfy $GA^i=c\,A^iG$ for every $i$. Then
    for $0<N$ and $L\le N$ the vector
    $\sigma\mapsto\tr(A^{\sigma_0}\cdots A^{\sigma_{N-1}}G)$ lies in
    $\mathcal G_{N,L}(A)$.
\end{lemma}

\begin{proof}
    \leanok
    For the window starting at site $i$, splitting the configuration as $u\,v$
    with $|u|=i$ ensures that $v\,u$ begins with the window. Then
    $\tr(A^uA^vG)=\tr(A^vGA^u)=c^{i}\tr(A^vA^uG)$, and $A^{vu}G$ is the window
    word followed by the boundary matrix $c^i A^{\mathrm{out}}G$, where
    $A^{\mathrm{out}}$ is the product over the sites outside the window.
\end{proof}

\begin{lemma}[Each dimer covering lies in the chain ground space]%
    \label{thm:majumdar_ghosh_coverings_chain_ground_space}
    \lean{MPSTensor.majumdarGhosh_pairCoveringEven_mem_chainGroundSpace,
        MPSTensor.majumdarGhosh_pairCoveringOdd_mem_chainGroundSpace}
    \leanok
    \uses{thm:majumdar_ghosh_dimer_superposition, thm:mps_pair_covering_form,
        thm:twisted_mpv_chain_ground_space, lem:mpv_mem_chain_ground_space}
    For even $N>0$ and $L\le N$, both coverings $P^{\mathrm{even}}_s$ and
    $P^{\mathrm{odd}}_s$ lie in $\mathcal G_{N,L}(A)$.
\end{lemma}

\begin{proof}
    \leanok
    The grading $G=\diag(1,-1,-1)$ satisfies $GA^i=-A^iG$. For even $N$,
    Lemma~\ref{thm:mps_pair_covering_form} and
    Lemma~\ref{thm:majumdar_ghosh_dimer_superposition} give
    $\tr(A^{\sigma}G)=2(A^{\sigma})_{00}-\tr(A^{\sigma})
        =P^{\mathrm{even}}_s(\sigma)-P^{\mathrm{odd}}_s(\sigma)$.
    By Lemma~\ref{thm:twisted_mpv_chain_ground_space}, both this twisted
    vector and $\ket{V^{(N)}(A)}$ lie in $\mathcal G_{N,L}(A)$, and the two
    coverings are half their sum and half their difference.
\end{proof}

\begin{lemma}[Energy of the dimer coverings]%
    \label{thm:majumdar_ghosh_hamiltonian_eigen}
    \lean{MPSTensor.majumdarGhoshHamiltonian_apply_of_mem_chainGroundSpace,
        MPSTensor.majumdarGhoshHamiltonian_pairCoveringEven,
        MPSTensor.majumdarGhoshHamiltonian_pairCoveringOdd,
        MPSTensor.majumdarGhoshHamiltonian_mpv}
    \leanok
    \uses{def:majumdar_ghosh_hamiltonian, thm:majumdar_ghosh_local_ground_space,
        thm:majumdar_ghosh_coverings_chain_ground_space,
        lem:mpv_mem_chain_ground_space}
    For $N\ge3$, every $\psi\in\mathcal G_{N,3}(A)$ satisfies
    $H_N\psi=-\tfrac{3N}{8}\psi$. In particular, for even $N\ge4$ each of the
    two singlet coverings, and for $N\ge3$ the periodic vector
    $\ket{V^{(N)}(A)}$, is an eigenvector of~(\ref{eq:mg_hamiltonian}) with
    eigenvalue $-\tfrac{3N}{8}$.
\end{lemma}

\begin{proof}
    \leanok
    Each window of three sites $i,i+1,i+2$ of $\psi$ lies in
    $\mathcal G_3(A)$, so by
    Lemma~\ref{thm:majumdar_ghosh_local_ground_space} the three transposed
    copies of $\psi$ in that window sum to zero. Summing over $i$ and shifting
    the index of the middle transposition gives
    $2\sum_iP_{i,i+1}\psi+\sum_iP_{i,i+2}\psi=0$. Lemma~\ref{thm:spin_exchange_swap}
    then gives
    $H_N\psi=\tfrac14\bigl(2\sum_iP_{i,i+1}\psi+\sum_iP_{i,i+2}\psi\bigr)
        -\tfrac{3N}{8}\psi=-\tfrac{3N}{8}\psi$.
\end{proof}

Because the Majumdar--Ghosh tensor is non-normal
(Lemma~\ref{thm:majumdar_ghosh_not_normal}), the intersection property of
Lemma~\ref{thm:ground_space_intersection} does not apply to it. The bond
grading $G=\diag(1,-1,-1)$, which satisfies $GA^i=-A^iG$, takes the place of
normality in the following two theorems.

\begin{lemma}[Graded injectivity of the boundary map]%
    \label{lem:majumdar_ghosh_graded_injectivity}
    \lean{MPSTensor.majumdarGhosh_exists_parity_rep,
        MPSTensor.majumdarGhosh_eq_zero_of_groundSpaceMap_eq_zero}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:ground_space_parent,
        thm:majumdar_ghosh_left_canonical}
    Every vector of $\mathcal G_{K}(A)$ equals $\Gamma_{K}(X)$ for a matrix $X$
    with $GX=(-1)^{K}XG$. For $K\ge3$, a matrix $X$ with $GX=(-1)^{K}XG$ and
    $\Gamma_{K}(X)=0$ is zero.
\end{lemma}

\begin{proof}
    \leanok
    Since $G^2=1$ and $GA^i=-A^iG$, the identity
    $\Gamma_{K}(GXG)=(-1)^{K}\Gamma_{K}(X)$ holds, so $X$ may be replaced by
    $\tfrac12\bigl(X+(-1)^{K}GXG\bigr)$. For $K=3$ the matrix $X$ is supported on
    the entries $(0,1)$, $(0,2)$, $(1,0)$, $(2,0)$, and the coordinates of
    $\Gamma_{3}(X)$ at $001$, $100$, $011$, $110$ are $-\tfrac12X_{02}$,
    $\tfrac1{\sqrt2}X_{10}$, $-\tfrac1{\sqrt2}X_{20}$ and $-\tfrac12X_{01}$. For the
    step from $K$ to $K+1$, fixing the last site to $a$ gives
    $\Gamma_{K}(A^aX)=0$, where $A^aX$ has parity $(-1)^{K}$; hence $A^aX=0$ for
    both $a$, and Lemma~\ref{thm:majumdar_ghosh_left_canonical} gives
    $X=\sum_a(A^a)^\dagger A^aX=0$.
\end{proof}

\begin{lemma}[Intersection property of the Majumdar--Ghosh tensor]%
    \label{lem:majumdar_ghosh_intersection}
    \lean{MPSTensor.majumdarGhosh_groundSpace_intersection,
        MPSTensor.majumdarGhosh_chainGroundSpace_le_groundSpace}
    \leanok
    \uses{lem:majumdar_ghosh_graded_injectivity,
        thm:majumdar_ghosh_local_ground_space, def:chain_ground_space}
    For $M\ge3$,
    $(\mathcal G_{M}(A)\otimes\C^2)\cap(\C^2\otimes\mathcal G_{M}(A))
        =\mathcal G_{M+1}(A)$. Consequently, for $N\ge3$ every vector of
    $\mathcal G_{N,3}(A)$ lies in $\mathcal G_{N}(A)$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:ground_space_restriction_overlap, thm:ground_space_grow_back}
    Let $\psi$ lie in the intersection. By
    Lemma~\ref{lem:majumdar_ghosh_graded_injectivity}, its restriction with
    first letter $i$ is $\Gamma_{M}(Y_i)$ and its restriction with last letter
    $j$ is $\Gamma_{M}(Z_j)$, with $Y_i$ and $Z_j$ of parity $(-1)^{M}$. Comparing
    on the middle $M-1$ sites gives
    $\Gamma_{M-1}(A^jY_i)=\Gamma_{M-1}(Z_jA^i)$
    (Lemma~\ref{thm:ground_space_restriction_overlap}), and both matrices have parity
    $(-1)^{M-1}$. For $M\ge4$ the second statement of
    Lemma~\ref{lem:majumdar_ghosh_graded_injectivity} gives $A^jY_i=Z_jA^i$; with
    $X=\sum_j(A^j)^\dagger Z_j$ the left-canonical equation
    $\sum_j(A^j)^\dagger A^j=1$ and
    Lemma~\ref{thm:ground_space_grow_back} give $Y_i=XA^i$, hence
    $\psi=\Gamma_{M+1}(X)$. For $M=3$, the
    conditions~(\ref{eq:mg_spin_half_conditions}) on the four three-site
    restrictions with a fixed first or last letter give $\psi=\Gamma_{4}(X)$ for
    the matrix $X$ with $X_{00}=-2\psi_{0110}$, $X_{11}=-2\psi_{1100}$,
    $X_{12}=2\psi_{0100}$, $X_{21}=2\psi_{1011}$, $X_{22}=-2\psi_{0011}$ and all
    other entries zero, as a check of the sixteen coordinates shows. The second
    statement follows by enlarging a window of three sites one site at a time.
\end{proof}

\begin{lemma}[Ground space of the Majumdar--Ghosh parent Hamiltonian]%
    \label{thm:majumdar_ghosh_ground_space}
    \lean{MPSTensor.majumdarGhosh_chainGroundSpace_eq_span,
        MPSTensor.majumdarGhosh_ker_parentHamiltonian_eq_span,
        MPSTensor.majumdarGhosh_pairCovering_linearIndependent,
        MPSTensor.majumdarGhosh_finrank_chainGroundSpace}
    \leanok
    \uses{def:majumdar_ghosh_tensor, def:ground_space_parent,
        thm:majumdar_ghosh_coverings_chain_ground_space, lem:majumdar_ghosh_intersection,
        lem:majumdar_ghosh_graded_injectivity,
        thm:parent_hamiltonian_kernel_eq_chain_ground_space}
    On an even periodic chain of length $N = 2m\ge4$, let
    $H_{N,\mathrm{MG}}^{(3)}$ be the range-three parent Hamiltonian of the
    Majumdar--Ghosh tensor, the periodic sum of translates of the local
    projector with kernel $\mathcal G_{3}(A_{\mathrm{MG}})$. Then
    $H_{N,\mathrm{MG}}^{(3)}$ has a two-dimensional ground space
    $\mathcal G_{N,3}(A_{\mathrm{MG}})$, spanned by the two nearest-neighbour
    singlet coverings $(1,2)(3,4)\cdots(N-1,N)$ and $(2,3)(4,5)\cdots(N,1)$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:parent_cyclic_translate_preserves_chain_space,
        thm:full_ring_boundary_change_of_cut}
    By Lemma~\ref{thm:majumdar_ghosh_coverings_chain_ground_space}, the two
    coverings lie in $\mathcal G_{N,3}(A)$. Conversely,
    let $\psi\in\mathcal G_{N,3}(A)$. By
    Lemma~\ref{lem:majumdar_ghosh_intersection}, $\psi=\Gamma_{N}(X)$ and its
    translate by one site, which lies in $\mathcal G_{N,3}(A)$ by
    Lemma~\ref{lem:parent_cyclic_translate_preserves_chain_space}, is
    $\Gamma_{N}(Y)$, with $X$ and $Y$ commuting with $G$. Comparing the two
    cuts on the remaining $N-1$ sites via
    Lemma~\ref{thm:full_ring_boundary_change_of_cut} gives
    $\Gamma_{N-1}(XA^a)=\Gamma_{N-1}(A^aY)$, so
    Lemma~\ref{lem:majumdar_ghosh_graded_injectivity} forces $XA^a=A^aY$. The
    entries of these two equations force
    $X=\diag(x,y,y)=\tfrac{x+y}2\,1+\tfrac{x-y}2\,G$, and
    $\Gamma_{N}(1)=P^{\mathrm{even}}_s+P^{\mathrm{odd}}_s$,
    $\Gamma_{N}(G)=P^{\mathrm{even}}_s-P^{\mathrm{odd}}_s$ as in the proof of
    Lemma~\ref{thm:majumdar_ghosh_coverings_chain_ground_space}. The two
    coverings are linearly independent because $\Gamma_{N}$ is injective on the
    matrices commuting with $G$. Finally,
    Lemma~\ref{thm:parent_hamiltonian_kernel_eq_chain_ground_space}
    identifies the kernel of $H_{N,\mathrm{MG}}^{(3)}$ with
    $\mathcal G_{N,3}(A)$.
\end{proof}

\begin{lemma}[No dimer ground state on odd rings]%
    \label{thm:majumdar_ghosh_ground_space_odd}
    \lean{MPSTensor.majumdarGhosh_chainGroundSpace_eq_bot_of_odd}
    \leanok
    \uses{lem:majumdar_ghosh_intersection, lem:majumdar_ghosh_graded_injectivity}
    On an odd periodic chain of length $N\ge5$, the space
    $\mathcal G_{N,3}(A_{\mathrm{MG}})$ is zero.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:parent_cyclic_translate_preserves_chain_space,
        thm:full_ring_boundary_change_of_cut}
    As in the proof of Lemma~\ref{thm:majumdar_ghosh_ground_space}, a vector
    $\Gamma_{N}(X)$ of $\mathcal G_{N,3}(A)$ and its translates by one and two
    sites give $XA^a=A^aY$ and $YA^a=A^aZ$, now with $X,Y,Z$ anticommuting with
    $G$, so that $X_{00}=X_{11}=X_{12}=X_{21}=X_{22}=0$. The entries of
    $XA^a=A^aY$ give
    \begin{align}
        X_{10} & =X_{20}=0,
               &
        Y_{10} & =\tfrac1{\sqrt2}X_{02},
               &
        Y_{20} & =-\tfrac1{\sqrt2}X_{01},
        \notag
    \end{align}
    and the same entries of $YA^a=A^aZ$ give $Y_{10}=Y_{20}=0$. Hence
    $X_{01}=X_{02}=0$, and $X=0$.
\end{proof}

\begin{lemma}[The three-site term as a projector]%
    \label{thm:majumdar_ghosh_term_projector}
    \lean{MPSTensor.majumdarGhoshTerm_apply,
        MPSTensor.majumdarGhoshTerm_shift_eq_parentInteraction}
    \leanok
    \uses{def:majumdar_ghosh_hamiltonian, def:parent_interaction,
        thm:spin_exchange_swap, thm:majumdar_ghosh_local_ground_space}
    The operator
    $\tfrac43\bigl(h+\tfrac38\bigr)=\tfrac13(P_{01}+P_{12}+P_{02})$ is the
    orthogonal projector onto $\mathcal G_{3}(A)^{\perp}$, the three-site parent
    interaction of the Majumdar--Ghosh tensor.
\end{lemma}

\begin{proof}
    \leanok
    By Lemma~\ref{thm:spin_exchange_swap}, the operator
    $Q=\tfrac13(P_{01}+P_{12}+P_{02})$ equals $\tfrac43(h+\tfrac38)$. Each
    transposition is a permutation of the configurations, hence self-adjoint,
    so $Q$ is self-adjoint. On three spin-$\tfrac12$ sites the antisymmetric part
    vanishes, so $1+C+C^{2}=P_{01}+P_{12}+P_{02}$ for the cyclic shift $C$, and
    $(P_{01}+P_{12}+P_{02})^{2}=3+3(C+C^{2})=3(P_{01}+P_{12}+P_{02})$; this is
    checked on the eight configurations. Hence $Q^{2}=Q$. By
    Lemma~\ref{thm:majumdar_ghosh_local_ground_space}, the kernel of $Q$,
    which is the eigenspace of $h$ for $-\tfrac38$, equals $\mathcal G_{3}(A)$.
    A self-adjoint idempotent is the orthogonal projector onto the orthogonal
    complement of its kernel.
\end{proof}

\begin{lemma}[The three-site term and total spin]%
    \label{thm:majumdar_ghosh_term_total_spin}
    \lean{MPSTensor.siteOperator, MPSTensor.totalSpin, MPSTensor.totalSpinSq,
        MPSTensor.spinExchange_self,
        MPSTensor.totalSpinSq_eq, MPSTensor.spinHalfOperator_sum_mul_self,
        MPSTensor.totalSpinSq_three_eq_majumdarGhoshTerm,
        MPSTensor.totalSpinSq_three_eq_parentInteraction,
        MPSTensor.majumdarGhosh_groundSpace_three_eq_totalSpinSq_eigenspace,
        MPSTensor.majumdarGhosh_parentInteraction_eq_spinThreeHalves_starProjection}
    \leanok
    \uses{thm:majumdar_ghosh_term_projector, thm:spin_exchange_swap,
        thm:majumdar_ghosh_local_ground_space}
    On three spin-$\tfrac12$ sites let $S^\alpha=S^\alpha_0+S^\alpha_1+S^\alpha_2$
    and $\mathbf S^{2}=\sum_\alpha(S^\alpha)^{2}$. Then
    $\mathbf S^{2}=\tfrac94+4h=\tfrac34+3Q$ with
    $Q=\tfrac43\bigl(h+\tfrac38\bigr)$ the projector of
    Lemma~\ref{thm:majumdar_ghosh_term_projector}. Hence $\mathcal G_{3}(A)$ is
    the eigenspace of $\mathbf S^{2}$ for $\tfrac34$, the subspace of total spin
    $\tfrac12$, and $Q$ is the orthogonal projector $P^{3/2}$ onto the eigenspace of
    $\mathbf S^{2}$ for $\tfrac{15}4$, the subspace of total spin $\tfrac32$.
\end{lemma}

\begin{proof}
    \leanok
    Expanding the squares gives $\mathbf S^{2}=\sum_{j,k}\mathbf S_j\cdot\mathbf S_k$,
    where $\sum_\alpha(S^\alpha)^{2}=\tfrac34$ on each site and
    Lemma~\ref{thm:spin_exchange_swap} gives
    $\mathbf S_j\cdot\mathbf S_k=\tfrac12P_{jk}-\tfrac14$ for $j\neq k$. On
    three sites this yields
    $\mathbf S^{2}=\tfrac94-\tfrac32+(P_{01}+P_{12}+P_{02})=\tfrac34+3Q$. Since
    $Q$ is an orthogonal projector, $\mathbf S^{2}v=\tfrac34v$ exactly when $Qv=0$,
    and $\mathbf S^{2}v=\tfrac{15}4v$ exactly when $Qv=v$; the kernel of $Q$ is
    $\mathcal G_{3}(A)$ and its range is $\mathcal G_{3}(A)^{\perp}$.
\end{proof}

\begin{lemma}[Ground space of the Majumdar--Ghosh Hamiltonian]%
    \label{thm:majumdar_ghosh_review_hamiltonian}
    \lean{MPSTensor.majumdarGhoshHamiltonian_apply,
        MPSTensor.majumdarGhosh_parentHamiltonian_three_apply,
        MPSTensor.majumdarGhoshHamiltonian_add_eq_parentHamiltonian,
        MPSTensor.majumdarGhoshHamiltonian_add_isPositive,
        MPSTensor.majumdarGhoshHamiltonian_eigenvalue_ge,
        MPSTensor.majumdarGhoshHamiltonian_eigenspace_eq_ker,
        MPSTensor.majumdarGhoshHamiltonian_eigenspace_eq_span,
        MPSTensor.majumdarGhoshHamiltonian_hasEigenvalue}
    \leanok
    \uses{def:majumdar_ghosh_hamiltonian, def:parent_hamiltonian,
        thm:majumdar_ghosh_term_projector, lem:transported_es_positive,
        thm:majumdar_ghosh_ground_space, thm:spin_exchange_swap}
    For $N\ge3$, the Hamiltonian~(\ref{eq:mg_hamiltonian}) satisfies
    \begin{align}
        H_{N}+\tfrac{3N}{8}=\tfrac34H_{N,\mathrm{MG}}^{(3)},
        \label{eq:mg_hamiltonian_projectors}
    \end{align}
    so $H_{N}\ge-\tfrac{3N}{8}$, and the eigenspace of $H_{N}$ for
    $-\tfrac{3N}{8}$ is the kernel of $H_{N,\mathrm{MG}}^{(3)}$. As stated in
    \cite[Appendix~A, ``The Majumdar-Ghosh model'']{Cirac2021Matrix}, for even
    $N\ge4$ the value $-\tfrac{3N}{8}$ is the smallest eigenvalue of $H_{N}$,
    its eigenspace is spanned by the two singlet coverings
    $(1,2)(3,4)\cdots(N-1,N)$ and $(2,3)(4,5)\cdots(N,1)$, and the
    Majumdar--Ghosh state is a ground-state superposition of those two
    coverings.
\end{lemma}

\begin{proof}
    \leanok
    Lemma~\ref{thm:spin_exchange_swap} gives
    $H_{N}=\tfrac12\sum_{i}P_{i,i+1}+\tfrac14\sum_{i}P_{i,i+2}-\tfrac{3N}{8}$.
    The cyclic shift of the summation index gives
    $\sum_{i}P_{i+1,i+2}=\sum_{i}P_{i,i+1}$, so
    $H_{N}=\tfrac14\sum_{i}\bigl(P_{i,i+1}+P_{i+1,i+2}+P_{i,i+2}\bigr)
        -\tfrac{3N}{8}$.
    By Lemma~\ref{thm:majumdar_ghosh_term_projector},
    $H_{N,\mathrm{MG}}^{(3)}
        =\tfrac13\sum_{i}\bigl(P_{i,i+1}+P_{i+1,i+2}+P_{i,i+2}\bigr)$.
    Comparing the two sums gives~(\ref{eq:mg_hamiltonian_projectors}). The parent
    Hamiltonian is positive by Lemma~\ref{lem:transported_es_positive}, so a
    nonzero $v$ with $H_{N}v=\mu v$ satisfies
    $0\le\langle v,(H_{N}+\tfrac{3N}{8})v\rangle
        =(\mu+\tfrac{3N}{8})\lVert v\rVert^{2}$,
    and hence $\mu+\tfrac{3N}{8}\ge0$. By
    Lemma~\ref{thm:majumdar_ghosh_ground_space}, the eigenspace for
    $-\tfrac{3N}{8}$, which is the kernel of the right side
    of~(\ref{eq:mg_hamiltonian_projectors}), is spanned by the two coverings.
    The coverings are nonzero, so $-\tfrac{3N}{8}$ is an eigenvalue.
\end{proof}

\begin{lemma}[The dimer energy on odd rings]%
    \label{thm:majumdar_ghosh_review_hamiltonian_odd}
    \lean{MPSTensor.majumdarGhoshHamiltonian_not_hasEigenvalue_of_odd}
    \leanok
    \uses{thm:majumdar_ghosh_review_hamiltonian,
        thm:majumdar_ghosh_ground_space_odd,
        thm:parent_hamiltonian_kernel_eq_chain_ground_space}
    For odd $N\ge5$, $-\tfrac{3N}{8}$ is not an eigenvalue of $H_{N}$, so every
    eigenvalue of $H_{N}$ is strictly larger than $-\tfrac{3N}{8}$.
\end{lemma}

\begin{proof}
    \leanok
    By Lemmas~\ref{thm:majumdar_ghosh_review_hamiltonian},
    \ref{thm:parent_hamiltonian_kernel_eq_chain_ground_space},
    and~\ref{thm:majumdar_ghosh_ground_space_odd}, the eigenspace of $H_N$ for
    $-\tfrac{3N}{8}$ equals
    $\ker H_{N,\mathrm{MG}}^{(3)}=\mathcal G_{N,3}(A)=0$.
\end{proof}

\begin{remark}\label{rem:majumdar_ghosh_review_hamiltonian}
    By Lemma~\ref{thm:majumdar_ghosh_review_hamiltonian}, on an even ring the
    ground space of $H_{N}$ is two-dimensional. The one-site translation
    exchanges the two singlet coverings, so this degeneracy reflects the
    spontaneous breaking of the one-site translation symmetry of the chain.
\end{remark}
