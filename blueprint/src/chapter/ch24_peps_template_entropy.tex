\section{Entropy cost of a partial depth row}

\begin{theorem}[Regional increment bound]
    \label{thm:area_partial_row_increment}
    \lean{TNLean.PEPS.AreaLaw.abs_regionalEntropy_union_sub_le}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $\Omega$ be a unit vector on a finite lattice domain with local dimension
    $q$. For disjoint regions $R,B$, one has
    \begin{align}
        |S_\Omega(R\cup B)-S_\Omega(R)| & \le S_\Omega(B).
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Subadditivity gives
    \begin{align}
        S_\Omega(R\cup B) & \le S_\Omega(R)+S_\Omega(B).
    \end{align}
    Complements are taken in the physical domain. Since $R\cap B=\varnothing$,
    \begin{align}
        (R\cup B)^c\cap B & =\varnothing, & (R\cup B)^c\cup B & =R^c.
    \end{align}
    Apply subadditivity to this disjoint union and use purity,
    $S_\Omega(X)=S_\Omega(X^c)$, to obtain
    \begin{align}
        S_\Omega(R) & =S_\Omega(R^c)
        \le S_\Omega((R\cup B)^c)+S_\Omega(B)
        =S_\Omega(R\cup B)+S_\Omega(B).
    \end{align}
\end{proof}

\begin{theorem}[Partial template row]
    \label{thm:area_template_partial_row}
    \lean{TNLean.PEPS.AreaLaw.Geometry.template_partial_row_entropy_le}
    \leanok
    \uses{def:area_law_ground_entropy,def:area_law_template}
    Let $T$ be a template with $C_{\rm tpl}\ge24$, let $1\le j\le s_0$,
    and let $Y\subseteq T_j\setminus T_{j-1}$. For a unit vector $\Omega$,
    local dimension $q\ge1$, and any physical region $A$, one has
    \begin{align}
        \left|S_\Omega\bigl(A\cap((T_{j-1}\setminus T)\cup Y)\bigr)
        -S_\Omega\bigl(A\cap(T_{j-1}\setminus T)\bigr)\right|
        \le n\log q.
    \end{align}
    This is the partial-row entropy step in the proof of
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_partial_row_increment,thm:area_law_entropy_dimension,thm:area_law_template_depth_layers}
    Set $R=A\cap(T_{j-1}\setminus T)$ and $B=A\cap Y$.
    These regions are disjoint and $|B|\le |Y|\le n$.
    The regional increment bound and $S_\Omega(B)\le|B|\log q$
    (Theorem~\ref{thm:area_law_entropy_dimension}) give the claim since $\log q\ge0$.
\end{proof}

\subsection{Finite weighted dyadic summation}

\begin{theorem}[From side-length bounds to a dyadic moment]
    \label{thm:area_law_weighted_dyadic_sum}
    \lean{TNLean.PEPS.AreaLaw.Geometry.sum_weighted_dyadic_rpow_le}
    \leanok
    Let $K\ge0$ be an integer, $e>0$, $B\ge0$, $A\in\mathbb R$, and
    $a_k\in\mathbb R$. Suppose $2^K a_K\le A$ and $2^k a_k\le B$ for
    $0\le k<K$. Then
    \begin{align}
        \sum_{k=0}^{K} a_k(2^k)^{1+e}
        \le \left(A+\frac{B}{2^e-1}\right)(2^K)^e.
    \end{align}
    This is the finite summation step in
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Factor each summand as $(2^k a_k)(2^k)^e$. The cap contributes at most
    $A(2^K)^e$. Since $2^e>1$, the lower scales contribute at most
    \begin{align}
        B\sum_{k=0}^{K-1}(2^e)^k
        =B\frac{(2^e)^K-1}{2^e-1}
        \le\frac{B}{2^e-1}(2^K)^e.
    \end{align}
\end{proof}

\section{Safe rectangles in template coverings}

\begin{theorem}[Safety from template separation]
    \label{thm:area_template_safe_rectangles}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.isSafe_of_subset_ambientDilation,
        TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.isSafe_cappedDyadicPartition_core,
        TNLean.PEPS.AreaLaw.Geometry.Template.IsSeparated.isSafe_cappedDyadicPartition_shell}
    \leanok
    \uses{def:area_law_template,def:area_law_safe_rectangle,thm:area_template_dyadic_rectangle,
        thm:area_law_capped_dyadic_partition}
    Let $T$ be separated from the cut, let $D_0\ge1$ be an integer, and let
    $0\le j\le s_0$. Every integer rectangle $Q\subseteq T_j$ of size at
    most $s_0$ is safe. In particular, if $2^K\le s_0$, every selected square
    of the capped partitions of $T$ and $T_j\setminus T$ is a safe rectangle.
    The standing choice $D_0>2R+10$ implies $D_0\ge1$.
    This is the clearance step of~\cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:area_law_template_dyadic_clearance,thm:area_template_dyadic_rectangle,
        thm:area_law_capped_dyadic_partition}
    For $x\in Q\subseteq T_j$, choose $p\in T$ with
    $\|p-x\|_\infty\le j$. For every cut endpoint $z$,
    \begin{align}
        4D_0s_0<\|p-z\|_\infty
        \le j+\|x-z\|_\infty.
    \end{align}
    Since $j\le s_0\le D_0s_0$, this gives
    $D_0\operatorname{size}(Q)\le D_0s_0<\|x-z\|_\infty$.
    Selected cells are contained in the partitioned set, and their sizes
    satisfy $2^k\le2^K\le s_0$.
\end{proof}


\begin{theorem}[Weighted covering of the template core]
    \label{thm:area_law_template_weighted_core_cover}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.sum_rpow_cappedDyadicPartition_core_le}
    \leanok
    \uses{def:area_law_template,def:area_law_capped_dyadic_partition}
    Let $T$ be a template with $C_{\rm tpl}\ge24$, let $e>0$, and choose
    $K\ge0$ with $2^K\le s_0<2^{K+1}$. For the capped dyadic
    partition $\mathcal P$ of $T$,
    \begin{align}
        \sum_{Q\in\mathcal P}\operatorname{side}(Q)^{1+e}
        \le \left(18+\frac{4}{2^e-1}\right)n s_0^e.
    \end{align}
    This is the core-covering moment estimate in
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_template_core_selected_count,thm:area_law_capped_dyadic_moment}
    The side-length bounds are $18n$ at scale $K$ and $4n$ below it. Apply
    Theorem~\ref{thm:area_law_capped_dyadic_moment} with $L=s_0$,
    $A=18n$ and $B=4n$, and factor out $n$.
\end{proof}

\section{Weighted coverings of template shells}

\begin{theorem}[Shell area and cap count]
    \label{thm:area_template_shell_cap}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.card_shell_le,TNLean.PEPS.AreaLaw.Geometry.Template.card_cappedDyadicPartition_shell_at_cap_le}
    \leanok
    \uses{def:area_law_template,def:area_law_capped_dyadic_partition}
    Let $T$ be a template with $C_{\rm tpl}\ge24$ and $0\le j\le L\le s_0$.
    Then $|T_j\setminus T|\le nj$. If $L<2^{K+1}$, the number $a_K$ of
    cap-scale squares in the capped dyadic partition of $T_j\setminus T$
    satisfies $2^K a_K\le2n$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_template_depth_layers,thm:area_law_capped_dyadic_partition}
    Sum the depth-layer estimates to obtain $|T_j\setminus T|\le nj$.
    Disjointness gives $4^K a_K\le nj\le nL<2n2^K$; divide by $2^K>0$.
\end{proof}

\begin{theorem}[Weighted shell covering]
    \label{thm:area_template_weighted_shell}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.sum_rpow_cappedDyadicPartition_shell_le}
    \leanok
    \uses{def:area_law_template,def:area_law_capped_dyadic_partition}
    Let $T$ be a template with $C_{\rm tpl}\ge24$,
    $0\le j\le L\le s_0$, $2^K\le L<2^{K+1}$, and $e>0$.
    For the maximal contained dyadic squares $Q$ of $T_j\setminus T$ with
    side capped at $2^K$, one has
    \begin{align}
        \sum_Q \operatorname{side}(Q)^{1+e}
        \le \left(2+\frac{14}{2^e-1}\right)nL^e.
    \end{align}
    This is the weighted covering step of
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_template_shell_cap,thm:area_law_template_shell_selected_count,thm:area_law_capped_dyadic_moment}
    Write $a_k$ for the number of selected squares of side $2^k$.
    The mixed-square bound gives $2^k a_k\le14n$ for $k<K$.
    The cap bound gives $2^K a_K\le2n$. Apply
    Theorem~\ref{thm:area_law_capped_dyadic_moment} with $A=2n$, $B=14n$
    and the cap bounds $2^K\le L<2^{K+1}$, and factor out $n$.
\end{proof}

\section{Entropy from a box estimate at one exponent}

The clearance parameter in the following estimate is real.
\begin{theorem}[Template entropy from a box estimate at one exponent]
    \label{thm:area_template_safe_box_entropy}
    \lean{TNLean.PEPS.AreaLaw.Geometry.template_entropy_bounds_of_real_safe_box}
    \leanok
    \uses{def:area_law_template,def:area_law_ground_entropy,def:area_law_cell_counts}
    Fix a finite physical domain $\Lambda$, a cut $A\subseteq\Lambda$,
    a unit vector $\Omega$, and a local dimension $q\ge1$. Let $Z$ be the
    endpoints of the edges crossing the cut. Fix real numbers $D_0\ge1$,
    $e>0$ and $C$. Suppose every integer rectangle $Q$ with
    $\operatorname{dist}_\infty(Q,Z)>D_0\operatorname{size}(Q)$ satisfies
    \begin{align}
        S_\Omega(A\cap Q)\le C\operatorname{size}(Q)^{1+e}.
    \end{align}
    Set $C_+=\max(C,0)$. For every template $T$ with $C_{\rm tpl}\ge24$
    and $\operatorname{dist}_\infty(T,Z)>4D_0s_0$, the core satisfies
    \begin{align}
        S_\Omega(A\cap T)
        \le C_+\left(18+\frac{4}{2^e-1}\right)ns_0^e.
    \end{align}
    For all integers $1\le L\le s_0$ and $0\le j\le L$,
    \begin{align}
        S_\Omega(A\cap(T_j\setminus T))
        \le C_+\left(2+\frac{14}{2^e-1}\right)nL^e.
    \end{align}
    If $1\le j\le L$ and $Y\subseteq T_j\setminus T_{j-1}$, then
    \begin{align}
        S_\Omega\bigl(A\cap((T_{j-1}\setminus T)\cup Y)\bigr)
        \le C_+\left(2+\frac{14}{2^e-1}\right)nL^e+n\log q.
    \end{align}
    The coefficients depend only on $C,e$, and on $q$ in the last-row term.
    Only the box bound at the chosen exponent is assumed. These are the
    three conditional entropy estimates of
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}, with the stronger range $e>0$.
\end{theorem}
\begin{proof}\leanok
    \lean{TNLean.PEPS.AreaLaw.cast_supDist}
    \uses{thm:area_template_physical_dyadic_entropy,
        lem:area_law_template_dyadic_clearance,
        thm:area_law_template_weighted_core_cover,thm:area_template_weighted_shell,
        thm:area_template_partial_row}
    The lattice sup-norm distance, viewed as a real number, is the maximum
    of the absolute coordinate differences. Increasing $C$ to $C_+$
    preserves every box bound. Partition the core
    into maximal contained dyadic squares, capped at the largest power of
    two not exceeding $s_0$. For a shell, cap at the largest power of two
    not exceeding $L$. Every selected square has side $u\le s_0$ and lies
    in $T_j$ with $j\le s_0$. Separation gives
    \begin{align}
        \operatorname{dist}_\infty(T_j,Z)
        >4D_0s_0-j>D_0s_0\ge D_0u.
    \end{align}
    Thus the box bound applies at the same real clearance and exponent.
    Subadditivity over the physical intersections and the weighted core
    and shell estimates give the first two inequalities. Their factors
    are nonnegative because $C_+\ge0$ and $2^e-1>0$.
    For a prefix, use the shell estimate at depth $j-1$ and then the
    partial-row bound $n\log q$ for the arbitrary subset $Y$.
\end{proof}
