\section{Charge selection with an arbitrary fixed weight}
\label{sec:arealaw-source-charge-selection}

The construction below is an auxiliary step in the proof of
\cite[Proposition~9.2]{OpenAI2026AreaLaw}, retaining the physical
state and label sequence while allowing any $W>0$ and
fixed interpolation interval $[p_0,p_1]\subset(0,1)$.

\begin{lemma}[Nonvanishing and the eigen-equation in augmented coordinates]
    \label{thm:arealaw-augmented-prevector-properties}
    \lean{TNLean.PEPS.AreaLaw.Scan.augmented_prevector_properties_of_global}
    \leanok
    \uses{def:al_scan_auxiliary_coordinates}
    For finite $V$, $q,d,k\in\mathbb N$, $\Omega\in\mathcal H_V$, and a symmetric-group label $\lambda$ on $k$ copies, let $v$ be the physical Schmidt--Bell prevector in global coordinates and $\mathrm{pre}=Uv$ its image under the augmented-site coordinate isometry $U$ (Definition~\ref{def:al_scan_auxiliary_coordinates}). For a physical operator $H$ with replica means $H_{\rm glob}$ and $H_{\rm aug}=UH_{\rm glob}U^{-1}$ (extended by $I$ on the two auxiliary registers), if $v\ne0$ and $H_{\rm glob}v=e v$ ($e\in\mathbb R$), then
    \begin{align}
        \mathrm{pre}            & \ne0,                    &
        \mathrm{pre}            & \in\mathcal H_{\rm sym}, &
        H_{\rm aug}\mathrm{pre} & =e\,\mathrm{pre},
        \label{eq:arealaw-augmented-prevector-properties}
    \end{align}
    where $\mathcal H_{\rm sym}$ is the replica-symmetric subspace (\cite[Section~8, lines~130--153 and~421--438]{OpenAI2026AreaLaw}).
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_prevector_symmetry,thm:al_scan_global_replica_energy,
        thm:al_augmented_prevector_norm}
    Follows from Lemmas~\ref{thm:al_augmented_prevector_norm}, \ref{thm:al_augmented_prevector_symmetry}, and~\ref{thm:al_scan_global_replica_energy} via $H_{\rm aug}\mathrm{pre}=UH_{\rm glob}v=e\,\mathrm{pre}$.
\end{proof}

\begin{lemma}[Selected density from the rough norm difference]
    \label{thm:arealaw-density-from-rough-endpoint}
    \lean{TNLean.PEPS.AreaLaw.Scan.selected_density_of_rough_endpoint}
    \leanok
    \uses{def:arealaw-rough-endpoint-remainder}
    Let $k,n,m,D,K\in\mathbb Z_{>0}$, $d\in\mathbb N$, $a,C_1,w>0$, $W=aK$, and $\rho=ka/(2(C_1+1)nD)$.
    If $A,B,T,b,c,C_i,C_b,\mathcal Q\in\mathbb R$ satisfy
    \begin{align}
        \frac A k  & \leq B+R_k^{\rm end}(d,c,K,C_i),        &
        b          & \leq C_b\log(k+2),                      &
        \mathcal Q & \leq\frac{A+2nm(kT+b)}{\rho nmw},\notag
    \end{align}
    then (\cite[Section~9, lines~457--490]{OpenAI2026AreaLaw})
    \begin{align}
        \frac{\mathcal Q}{KnD}
         & \leq\frac{2(C_1+1)}{Wnmw}
        \left[B+2nmT+R_k^{\rm end}(d,c,K,C_i+2nmC_b)\right].
        \label{eq:arealaw-density-from-rough-endpoint}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:arealaw-rough-selection-numerator,thm:arealaw-selected-density-normalization}
    By Lemma~\ref{thm:arealaw-rough-selection-numerator}, $(A+2nm(kT+b))/k\le B+2nmT+R_k^{\rm end}(d,c,K,C_i+2nmC_b)$; dividing by $KnD$ via Lemma~\ref{thm:arealaw-selected-density-normalization} and $aK=W$ gives (\ref{eq:arealaw-density-from-rough-endpoint}).
\end{proof}

\begin{lemma}[One physical state and label sequence for charge selection]
    \label{thm:arealaw-source-charge-selection}
    \lean{
        TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_sourceTarget_selected_actualCharge_density_energy}
    \leanok
    \uses{def:area_law_hamiltonian,def:al_scan_actual_histories,
        def:al_scan_actual_energy_terms,thm:al_scan_target_typical_window_uniform_dimension}
    Fix $R\in\mathbb N$, $J\geq0$, $\Delta>0$, and
    $C_T\geq0$. By Lemma~\ref{thm:al_scan_target_typical_window_uniform_dimension}, there exists $C_{\rm tr}>0$ (independent of $q,C_1,m_*,P$; cf.\ \cite[Proposition~4.5 and Section~9, lines~49--59]{OpenAI2026AreaLaw}) such that for every
    $q\geq1$, $C_1>0$, $m_*\in\mathbb N$ with $3m_*\leq C_1$, and $P\in\mathbb R$, there
    exist $C_{\rm target},c_t>0$ and $C_{\rm ent},e_{\rm ent}\in\mathbb R$, and for every scanner exponent tuple $(\ell,\mu,\kappa,\nu,e_{\rm in})$ with
    $0<\ell<1$, $0<\kappa<\mu<1-\ell$, $\nu>0$, and $\kappa\le e_{\rm in}<1$, there exist
    $C\geq1$, $C_l\geq0$, and $N\geq2$ (preceding all physical data and entropy inputs) with the following property.

    Let $\Lambda,T\subset\mathbb Z^2$ be finite with $T\ne\varnothing$,
    $H$ a range-$R$ Hamiltonian of bound $J$ with unit ground vector $\Omega$ and projector gap
    $\Delta$, and $d(x)=\operatorname{dist}_\infty(x,T)$.
    For $A\subseteq\Lambda$ and $n\geq N$, set $M=\lceil C_1nD\rceil$ and
    \begin{align}
        r_0 & =\lceil C_{\rm tr}(\log n)^2\rceil,     &
        L   & =\lfloor n^{1-\ell}\rfloor,             &
        m   & =\lfloor n^\mu\rfloor,                  &
        D   & =\lceil n^\kappa\rceil,                 &
        K   & =\left\lfloor\frac{L}{8m}\right\rfloor.
        \label{eq:arealaw-scanner-scales}
    \end{align}
    Assume $|T|\leq C_Tn^2$, each depth-$j$ dilation layer ($j\le L$) has $\le n$ sites,
    anchor multiplicity is $\le m_*$, $\operatorname{dist}_\infty(T,\partial_{\rm v}\Lambda\cup\partial_{\rm v}A)>2L+10r_0$,
    $1\le 2r_0\le D\le m/4$, and
    $S_\Omega(\{x\in A:0<d(x)\leq j\})\leq C_e(nL^\eta+n)$ for $0\leq j\leq L$.

    Put $X=A\cap T$, $S_0=A\cap T_L$, let $g>0$ be the normalized amplified gap in
    Lemma~\ref{thm:al_scan_target_typical_window}, and set
    $\varepsilon_t=\min\{n^{-1000},g/4\}$, $w=n^{3/5}$, and $\delta_t=n^{-100}$.
    Then $m,M>0$, $K\geq8$, $8Km\leq L$, $|S_0|\leq(C_T+1)n^2$,
    $X\subseteq S_0$, and the truncated Hamiltonian $\widetilde H$ has a unit ground vector
    $\widetilde\Omega$ with energy $e_t$ and projector gap $g/2$ satisfying
    \begin{align}
        0\leq e_t    & \leq\varepsilon_t,                       &
        \operatorname{dist}_{\rm tr}
        (|\widetilde\Omega\rangle\langle\widetilde\Omega|,
        |\Omega\rangle\langle\Omega|)
                     & \leq\sqrt{2\varepsilon_t/g},             &
        \mathcal B_X & \leq C_{\rm target}n(\log n)^{12}.\notag
    \end{align}
    With target-marginal eigenvalues $p_i$ and $S_X=S_{\widetilde\Omega}(X)$, the typical set
    $E=\{i:p_i>0,\ |{-\log p_i}-S_X|\leq w\}$ has cardinality $d=|E|>0$, mass $z=\sum_{i\in E}p_i\in[1-\delta_t,1]$, and $e^{-S_X-w}\leq p_i\leq e^{-S_X+w}$ for $i\in E$.

    Let $\psi$ be the normalized Schmidt vector restricted to $E$ with auxiliary entropy $S'$. There exist labels $\lambda_k$ and physical Schmidt--Bell prevectors $\mathrm{pre}_k$ (fixed before $W$) satisfying
    $0<\|\mathrm{pre}_k\|\leq1$ ($k>0$), $\log\dim\lambda_k-kS'=o(k)$, and eventually
    \begin{align}
        \|(I\otimes P_{\lambda_k})\psi^{\otimes k}\|^2
         & \geq\frac1{2(k+1)^{d^2}}.\notag
    \end{align}
    For any $W>0$, set $a=W/K$ and $\ell_{\dim}=1+(2r_0+1)^2\log q$, suppose $a\ell_{\dim}\leq c_t$, and define
    \begin{align}
        B_{\rm sh}  & =C_e(nL^\eta+n)+n^{-247},\notag                   \\
        B_{\rm rough}
                    & =2\log d-\log z+
        W\left(\frac{B_{\rm sh}}z+1+3nD\log q\right),\notag             \\
        T_{\rm ent} & =C_{\rm ent}aK\,a^{1/4}\ell_{\dim}^{e_{\rm ent}}.
        \label{eq:arealaw-charge-budgets}
    \end{align}
    Then there exist $C_b,c_U,c_R,C_i\ge0$, $0\leq b_k\leq C_b\log(k+2)$, and $r_k\geq0$ with $r_k\to0$ (chosen before $[p_0,p_1]$) such that
    \begin{align}
        R_k^{\rm Bell}(d,c,K)
         & =\frac{\log4+(d^2+cK)\log(k+2)}k,\notag \\
        R_k^{\rm end}
         & =R_k^{\rm Bell}(d,c_R,K)
        +(C_i+2nmC_b)\frac{\log(k+2)}k\to 0,\notag
    \end{align}
    and for every $0<p_0<p_1<1$ and all sufficiently large $k$, some charge
    round $s_k<nm$ and $p_k\in[p_0,p_1]$ with entropy defect $\mathcal Q_k$ and root metric $\mathcal M_k$, setting
    \begin{align}
        N_k^2 & =\|\mathcal M_k^{-1/4}\mathrm{pre}_k\|^2, &
        v_k   & =\mathcal M_k^{-1/4}\mathrm{pre}_k/N_k,   &
        \mu_k & =\operatorname{Re}\langle v_k,
        \overline{\widetilde H}_k v_k\rangle,\notag         \\
        \operatorname{cap}_k
              & =CW^2n^\ell D^2(\log n)^{C_l}
        \left[(1-p_k)\left(\frac{\mathcal Q_k}{KnD}\right)^{1/8}
            +(1-p_k)n^{-P}+p_k\right],\notag
    \end{align}
    satisfy
    \begin{align}
        \frac{\mathcal Q_k}{KnD}
              & \leq\frac{2(C_1+1)}{Wnm(p_1-p_0)}
        \left(B_{\rm rough}+2nmT_{\rm ent}+R_k^{\rm end}\right),\notag \\
        \frac{\mu_k-e_t}{g/2}
              & \leq\frac{e_t+\operatorname{cap}_k+r_k}{g/2},\notag    \\
        N_k^2 & >0,\notag                                              \\
        -\frac{\log N_k^2}{k}
              & \leq B_{\rm rough}+R_k^{\rm Bell}(d,c_U,K).
        \label{eq:arealaw-charge-selection}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_scan_target_typical_window,thm:al_global_schmidt_bell_prevector,
        thm:al_positive_prefix_entropy_transfer,thm:al_scan_actual_rough_upper,
        thm:arealaw-physical-endpoint-loss,thm:arealaw-density-from-rough-endpoint,
        thm:arealaw-augmented-prevector-properties,thm:arealaw-actual-selected-energy,
        thm:arealaw-generic-charge-energy,thm:arealaw-generic-selected-energy,
        lem:al_scanner_scale_facts,thm:area_law_band_count_growth,
        thm:arealaw-rough-endpoint-remainder-limit}
    Fix $C_{\rm tr}$ before $q$, then $(C_{\rm target},c_t,C_{\rm ent},e_{\rm ent})$, and then $(C,C_l,N)$. For $n\ge N$, Lemmas~\ref{thm:al_scan_target_typical_window}, \ref{thm:al_global_schmidt_bell_prevector}, and~\ref{thm:al_positive_prefix_entropy_transfer} yield $(\widetilde\Omega,E,z,\lambda_k,\mathrm{pre}_k)$ and transfer prefix entropies within $n^{-247}$. Lemmas~\ref{thm:al_scan_actual_rough_upper} and~\ref{thm:arealaw-physical-endpoint-loss} control the endpoint norms, and Lemma~\ref{thm:arealaw-augmented-prevector-properties} feeds $\mathrm{pre}_k$ into Lemmas~\ref{thm:arealaw-generic-charge-energy}, \ref{thm:arealaw-generic-selected-energy}, and~\ref{thm:arealaw-actual-selected-energy} to select $(s_k,p_k)$, which together with (\ref{eq:arealaw-density-from-rough-endpoint}) gives (\ref{eq:arealaw-charge-selection}).
\end{proof}

\begin{lemma}[Physical selected density for arbitrary scanner exponents]
    \label{thm:arealaw-source-selected-density}
    \lean{
        TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_sourceTarget_selected_actualCharge_source_density}
    \leanok
    \uses{thm:arealaw-source-charge-selection}
    Fix $R\in\mathbb N$, $J\geq0$, $\Delta>0$, and $C_T\geq0$, with $C_{\rm tr}>0$ as in Lemma~\ref{thm:arealaw-source-charge-selection}. For every $q\geq1$, $C_1>0$, $m_*\in\mathbb N$ ($3m_*\leq C_1$), and $P\in\mathbb R$, there exist $C_{\rm target}>0$ and $0<c\leq1/2$; for every admissible $(\ell,\mu,\kappa,\nu,e_{\rm in})$ there exist $C\geq1$ and $C_l\geq0$; and for every $C_e\geq0$ there exist $C_\delta>0$, $C_{l,\delta}\geq0$, and $N\geq2$ (preceding the physical instance and $W$).

    Under the geometric hypotheses and scales (\ref{eq:arealaw-scanner-scales}) of Lemma~\ref{thm:arealaw-source-charge-selection} for $n\ge N$ and $X=A\cap T$, assume that for $0\leq j\leq L$,
    \begin{align}
        S_\Omega(X) & \leq C_en^{1+e_{\rm in}},          &
        S_\Omega(\{x\in A:0<d(x)\leq j\})
                    & \leq C_e(nL^{e_{\rm in}}+n).\notag
    \end{align}
    Then $1\le 2r_0\le D\le m/4$, and $(\widetilde\Omega,E,z,\lambda_k,\mathrm{pre}_k)$ from Lemma~\ref{thm:arealaw-source-charge-selection} is fixed before $W$.
    For any $W\geq1$ with $a=W/K$, $\varepsilon=n^{-\nu}$, and $ar_0^2\leq c$, (\ref{eq:arealaw-charge-budgets}) at $\eta=e_{\rm in}$ satisfies
    \begin{align}
        B_{\rm rough}
         & \leq (6C_e+13+6\log q)Wn^{1+e_{\rm in}},\notag \\
        \delta_n
         & :=C_\delta\varepsilon^{-1}
        \left(n^{e_{\rm in}-\mu}+a^{1/4}(\log n)^{C_{l,\delta}}\right),
        \label{eq:arealaw-source-density}
    \end{align}
    and there exists $\rho_k\ge0$ with $\rho_k\to0$ such that for all sufficiently large $k$, the selected $(s_k,p_k)\in\{0,\ldots,nm-1\}\times[\varepsilon/2,\varepsilon]$ satisfies (\ref{eq:arealaw-charge-selection}) (with $p_0=\varepsilon/2,p_1=\varepsilon$) and
    $\mathcal Q_k/(KnD)\leq\delta_n+\rho_k$ (\cite[Proposition~9.2, lines~475--490]{OpenAI2026AreaLaw}).
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:arealaw-source-charge-selection,thm:scanner_rounded_admissibility,
        thm:al_scan_radius_dimension_smallness,thm:al_scan_transport_coefficients_log,
        thm:arealaw-target-entropy-transfer,thm:arealaw-source-rough-budget-power,
        thm:arealaw-selected-density-delta}
    By Lemmas~\ref{thm:scanner_rounded_admissibility}, \ref{thm:al_scan_radius_dimension_smallness}, \ref{thm:arealaw-target-entropy-transfer}, and~\ref{thm:arealaw-source-rough-budget-power}, $ar_0^2\le c$ implies $a\ell_{\dim}\le c_t$ and the bound on $B_{\rm rough}$ in (\ref{eq:arealaw-source-density}). By Lemma~\ref{thm:al_scan_transport_coefficients_log}, $G=C_{\rm ent}\ell_{\dim}^{e_{\rm ent}}\le C_t(\log n)^{C_{l,\delta}}$. Setting $C_\delta=4(C_1+1)(6C_e+13+6\log q+2C_t)C_s$, $\gamma=4(C_1+1)/(Wnm\varepsilon)$, and $\rho_k=|\gamma R_k^{\rm end}|\to0$, Lemma~\ref{thm:arealaw-selected-density-delta} and (\ref{eq:arealaw-charge-selection}) give $\mathcal Q_k/(KnD)\le\delta_n+\rho_k$.
\end{proof}

\end{proof}

\begin{lemma}[Uniform comparisons for a prescribed physical charge state]
    \label{thm:arealaw-supplied-target-norm-bounds}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_suppliedTarget_actualCharge_norm_bounds}
    \leanok
    \uses{thm:al_scan_common_status_budget,def:actual_sharp_lower_remainder}
    Fix $q\geq1$, $R\in\mathbb N$, and $C_r>0$.
    There exists $C_B>0$ (depending only on $q,R,C_r$) such that under the geometric hypotheses of
    Lemma~\ref{thm:al_scan_common_status_budget} with $K,m,M>0$ and
    $r_0=\lceil C_r(\log n)^2\rceil$, $\mathcal B=C_BnD(\log n)^{12}\geq1$.
    For a range-$R$ Hamiltonian $H$, unit vector $\Omega$, $\Delta>0$, and a ground vector $\widetilde\Omega$ of $\widetilde H$ with energy $e_t$ and projector gap $g/2$, set
    $C_0=512\mathrm e\,(2/g)$ and $c_0=(32\sqrt{1+2/g})^{-1}$.

    At $X=A\cap T$, prescribe a spectral set $E$ of size $d>0$ and mass $z>0$, normalized Schmidt truncation $\psi$ with auxiliary entropy $S'$, and labels $\lambda_k$ with $\mathrm{pre}_k=\pi_R^{\lambda_k}(\widetilde\Omega\otimes\mathrm{Bell}_{CR})^{\otimes k}$ satisfying
    $0<\|\mathrm{pre}_k\|\leq1$ ($k>0$), eventually $\|(I\otimes\pi_R^{\lambda_k})\psi^{\otimes k}\|^2\geq[2(k+1)^{d^2}]^{-1}$, and $\log\dim\lambda_k-kS'=o(k)$.

    Fix $W\geq1$, $a=W/K$, and assume $ar_0^2\leq1/2$.
    At each $(s,p)$ ($s<nm$, $p\in\mathbb R$ projected to $[0,1]$), let $\mathcal M_k$ be the root metric with band exponent $a/2$, and set
    \begin{align}
        N_k^2 & =\|\mathcal M_k^{-1/4}\mathrm{pre}_k\|^2, &
        v_k   & =\mathcal M_k^{-1/4}\mathrm{pre}_k/N_k,   &
        D_k   & =\frac{\operatorname{Re}\langle v_k,
            \overline{\widetilde H}_k v_k\rangle-e_t}{g/2}.\notag
    \end{align}
    Then there exist $c_R,C_{f,R}\geq0$ (fixed before $k,s,p$) such that $R_k^{(a)}\to0$ in
    (\ref{eq:actual_rough_weight_label_remainder}), and for every $k>0$ and $0<\tau\leq1/2$, $N_k^2>0$, $\|v_k\|=1$, $D_k\geq0$, and
    \begin{align}
        \frac W4\left[4S'
            -6\left(\tau\log d+\frac{h(\tau)}a\right)
            -3\frac{D_k}{\tau}\max\{0,2S'\}\right]-R_k^{(a)}
         & \leq-\frac{\log N_k^2}{k}.
        \label{eq:arealaw-supplied-rough-lower}
    \end{align}

    If moreover $a\sqrt{\mathcal B}\leq\min\{c_0,1\}/10$, then for each terminal $j$, band $b$, and regions $Q_{jb},Y_{jb},U_{jb}$, setting
    \begin{align}
        G^-_{jb}                       & =S_{\widetilde\Omega}(Q_{jb})
        +S_{\widetilde\Omega}((Q_{jb}\cup Y_{jb})^c)
        -S_{\widetilde\Omega}(Y_{jb}),\notag                                               \\
        G^+_{jb}                       & =S_{\widetilde\Omega}(U_{jb})
        +S_{\widetilde\Omega}(U_{jb}\cup Y_{jb})
        -S_{\widetilde\Omega}(Y_{jb}), &
        G_*                            & =\max\bigl(\{0\}\cup\{G^-_{jb}:j,b\}\bigr),\notag \\
        L_b                            & =2S'+(1-\tau)\sum_jw_jG^-_{jb}
        -2\tau\log d-(3+1/a)h(\tau)-25C_0\mathcal B a,\notag
    \end{align}
    with tree weights $w_j$ at $(s,p)$, there exist $c_U,c_L,C_{f,L}\geq0$ (fixed before $k,s,p,\tau$) such that $R_k^+=[\log4+(d^2+c_UK)\log(k+2)]/k\to0$, $R_k^-\to0$ (Definition~\ref{def:actual_sharp_lower_remainder}), and for all sufficiently large $k$, $s<nm$, $p\in\mathbb R$, and $0<\tau\leq1/2$,
    \begin{align}
        -\frac{\log N_k^2}{k}
         & \leq2\log d-\log z+\frac a2\sum_b\sum_jw_jG^+_{jb}
        +6KC_0\mathcal B a^2-\frac K2\log z+R_k^+,\notag      \\
        \frac a2\sum_b\left[L_b
            -\frac{D_k}{\tau}\max\{0,2S'+G_*\}-R_k^-\right]
         & \leq-\frac{\log N_k^2}{k},
        \label{eq:arealaw-supplied-sharp-bounds}
    \end{align}
    matching \cite[Proposition~9.2, lines~386--395]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_scan_supplied_status_moments,thm:actual_supplied_rough_lower_weight,
        thm:arealaw-sharp-smallness,thm:al_actual_sharp_upper,
        thm:actual_scan_supplied_prevector_sharp_lower}
    Choose $C_B$ from Lemma~\ref{thm:al_scan_supplied_status_moments}. Since $r_0\ge1$ and $ar_0^2\le1/2$ imply $0<a\le1/2$, Lemma~\ref{thm:actual_supplied_rough_lower_weight} yields (\ref{eq:arealaw-supplied-rough-lower}). When $a\sqrt{\mathcal B}\le\min\{c_0,1\}/10$, Lemma~\ref{thm:arealaw-sharp-smallness} gives $10a\le\min\{c_0/\sqrt{\mathcal B},1\}$, so Lemmas~\ref{thm:al_scan_supplied_status_moments}, \ref{thm:al_actual_sharp_upper}, and~\ref{thm:actual_scan_supplied_prevector_sharp_lower} yield (\ref{eq:arealaw-supplied-sharp-bounds}).
\end{proof}
