\section{Finite channel oscillation bounds}
\label{sec:al_channel_propagation}

Throughout this section, let $\mathcal H=(\C^q)^{\otimes I}$ ($q>0$) and
$\mathcal K=\C^J$ ($\dim\mathcal K\ge0$) be as in
Section~\ref{sec:al_channel_shell_oscillation}. Fix a positive contraction
$0\le k\le\mathbb 1$ on $\mathcal H$ and increasing regions
$K_l\subseteq K_m$ ($0\le l\le m$). Write
$\mathcal E_l=\mathcal E_{K_l}$ (Definition~\ref{def:al_local_root_channel}),
$\Delta_l$ for its shells (Definition~\ref{def:al_local_channel_shell}), and
$d_y$ for the site oscillation at $y\in I$
(Definition~\ref{def:qc_site_oscillation}). Operators $B$ on
$\mathcal H\otimes\mathcal K$ are arbitrary complex matrices with the
Euclidean operator norm.

\subsection{Initial oscillation from physical support}

For $S\subseteq I$, let $\mathcal A_S$ be the algebra of physical operators
supported on $S$, and $E_S^{\mathcal K}$ the expectation of
Definition~\ref{def:qc_spectator_expectation}. For $a,b\in J$, define the
physical block $B_{ab}$ by $(B_{ab})_{\sigma,\tau}=B_{(\sigma,a),(\tau,b)}$.

\begin{lemma}[Physical support and spectator expectation]
    \label{thm:al_spectator_support_fixed}
    \uses{def:qc_spectator_expectation,def:qc_local_circuit}
    \lean{QuantumCircuit.spectatorSiteExpectation_eq_self_iff}
    \leanok
    For every $S\subseteq I$ and arbitrary spectator index set $J$,
    \begin{align}
        E_S^{\mathcal K}(B)=B
         & \quad\Longleftrightarrow\quad
        \forall a,b\in J,\ B_{ab}\in\mathcal A_S.
        \label{eq:al_propagation_support_fixed}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_spectator_expectation,lem:qc_site_expectation_properties}
    By~(\ref{eq:qc_spectator_blocks}), $E_S^{\mathcal K}(B)=B$ is equivalent
    to $E_S(B_{ab})=B_{ab}$ for all $a,b\in J$, which holds if and only if
    $B_{ab}\in\mathcal A_S$ by Lemma~\ref{lem:qc_site_expectation_properties}.
\end{proof}

\begin{lemma}[Vanishing oscillation outside the physical support]
    \label{thm:al_spectator_support_oscillation_zero}
    \uses{def:qc_site_oscillation,def:qc_local_circuit}
    \lean{QuantumCircuit.siteOscillation_eq_zero_of_not_mem_support}
    \leanok
    If $B_{ab}\in\mathcal A_S$ for all $a,b\in J$ and $y\notin S$, then
    $d_y(B)=0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_spectator_support_fixed,thm:qc_spectator_commutant,
        thm:qc_site_oscillation_commutant}
    By Lemma~\ref{thm:al_spectator_support_fixed}, $E_S^{\mathcal K}(B)=B$.
    For $U\in\mathcal A_{\{y\}}\subseteq\mathcal A_{I\setminus S}$,
    Lemma~\ref{thm:qc_spectator_commutant} gives
    $[B,\widehat U]=[E_S^{\mathcal K}(B),\widehat U]=0$, so $d_y(B)=0$ by
    Lemma~\ref{thm:qc_site_oscillation_commutant}.
\end{proof}

\begin{lemma}[Initial oscillation bounded by the support indicator]
    \label{thm:al_spectator_support_indicator}
    \uses{def:qc_site_oscillation,def:qc_local_circuit}
    \lean{QuantumCircuit.siteOscillation_le_support_indicator}
    \leanok
    As the initial support estimate of
    \cite[Section~10]{OpenAI2026AreaLaw}, if $B_{ab}\in\mathcal A_S$ for all
    $a,b\in J$, then for every $y\in I$,
    \begin{align}
        d_y(B) & \le2\|B\|\mathbf1_S(y).
        \label{eq:al_propagation_support_indicator}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_spectator_support_oscillation_zero,
        thm:qc_site_oscillation_bounds}
    For $y\in S$, Lemma~\ref{thm:qc_site_oscillation_bounds} gives
    $d_y(B)\le2\|B\|$; for $y\notin S$, both sides of
    (\ref{eq:al_propagation_support_indicator}) vanish by
    Lemma~\ref{thm:al_spectator_support_oscillation_zero}.
\end{proof}

\subsection{Coefficients from the localization errors}

\begin{definition}[Shell coefficients from localization errors]
    \label{def:al_channel_shell_coefficient}
    \lean{TNLean.PEPS.AreaLaw.localRootChannelShellBound}
    \leanok
    For a real sequence $(\varepsilon_l)_{l\ge0}$ (with $\sqrt{r}=0$ for
    $r<0$), define
    \begin{align}
        \eta_0     & =2, \label{eq:al_propagation_coefficient_zero} \\
        \eta_{l+1} & =4\bigl(\sqrt{\varepsilon_{l+1}}
        +\sqrt{\varepsilon_l}\bigr).
        \label{eq:al_propagation_coefficient_successor}
    \end{align}
\end{definition}

\begin{lemma}[Nonnegativity of the shell coefficients]
    \label{thm:al_channel_shell_coefficient_nonneg}
    \uses{def:al_channel_shell_coefficient}
    \lean{TNLean.PEPS.AreaLaw.localRootChannelShellBound_nonneg}
    \leanok
    For every real sequence $(\varepsilon_l)$ and $l\ge0$, $\eta_l\ge0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:al_channel_shell_coefficient}
    Immediate from~(\ref{eq:al_propagation_coefficient_zero})
    and~(\ref{eq:al_propagation_coefficient_successor}) since
    $\sqrt{\varepsilon_l}\ge0$.
\end{proof}

\begin{lemma}[Finite-radius physical shell bound]
    \label{thm:al_channel_shell_finite_bound}
    \uses{def:al_local_channel_shell,def:qc_site_oscillation,
        def:al_channel_shell_coefficient}
    \lean{TNLean.PEPS.AreaLaw.norm_localRootChannelShell_le_sum_siteOscillation}
    \leanok
    Fix $l\in\N$ and $(\varepsilon_j)_{j\ge0}\subset\R$. If every
    $0\le j\le l$ satisfies
    \begin{align}
        \|k-E_{K_j}(k)\| & \le\varepsilon_j,
        \label{eq:al_propagation_finite_error}
    \end{align}
    then
    \begin{align}
        \|\Delta_l(B)\| & \le\eta_l\sum_{z\in K_l}d_z(B).
        \label{eq:al_propagation_finite_shell}
    \end{align}
    In particular, if $\|k-E_{K_j}(k)\|=0$ for all $0\le j\le L$, then
    $\Delta_j(B)=0$ for $1\le j\le L$, while $\eta_0=2$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_channel_shell_oscillation,def:al_channel_shell_coefficient}
    For $l=0$, (\ref{eq:al_propagation_finite_shell}) is
    (\ref{eq:al_shell_zero_oscillation}) with
    (\ref{eq:al_propagation_coefficient_zero}). For $l=m+1$, substituting
    (\ref{eq:al_propagation_finite_error}) at $j\in\{m,m+1\}$ into
    (\ref{eq:al_shell_successor_oscillation}) gives
    (\ref{eq:al_propagation_finite_shell}) with coefficient
    (\ref{eq:al_propagation_coefficient_successor}); when
    $\varepsilon_m=\varepsilon_{m+1}=0$, $\eta_{m+1}=0$.
\end{proof}

\begin{lemma}[Finite one-event oscillation increment]
    \label{thm:al_channel_finite_increment}
    \uses{def:al_local_root_channel,def:qc_site_oscillation,
        def:al_channel_shell_coefficient}
    \lean{TNLean.PEPS.AreaLaw.siteOscillation_localRootChannel_le}
    \leanok
    As the finite-radius one-event estimate of
    \cite[Section~10]{OpenAI2026AreaLaw}, fix $n\in\N$, $y\in I$, and
    $(\varepsilon_l)\subset\R$ satisfying
    $\|k-E_{K_l}(k)\|\le\varepsilon_l$ for all $0\le l\le n$. Then
    \begin{align}
        \label{eq:al_propagation_finite_increment}
        d_y(\mathcal E_n(B))
         & \le d_y(B)+2\sum_{0\le l\le n,\ y\in K_l}
        \eta_l\sum_{z\in K_l}d_z(B).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_channel_shell_finite_bound,
        thm:al_channel_shell_coefficient_nonneg,
        thm:al_root_channel_outside_oscillation,
        thm:qc_site_oscillation_seminorm,
        thm:qc_site_oscillation_bounds,def:al_local_channel_shell}
    Set $a_l=\eta_l\sum_{z\in K_l}d_z(B)\ge0$ and induct on $n$. If
    $y\notin K_n$, Lemma~\ref{thm:al_root_channel_outside_oscillation} gives
    $d_y(\mathcal E_n(B))\le d_y(B)$, which implies
    (\ref{eq:al_propagation_finite_increment}) since $a_l\ge0$. For $n=0$ and
    $y\in K_0$, $\mathcal E_0(B)=B+\Delta_0(B)$ and
    Lemma~\ref{thm:al_channel_shell_finite_bound} give
    \begin{align}
        d_y(\mathcal E_0(B))
         & \le d_y(B)+d_y(\Delta_0(B))\le d_y(B)+2a_0.
        \label{eq:al_propagation_increment_zero}
    \end{align}
    For $y\in K_{n+1}$, $\mathcal E_{n+1}(B)=\mathcal E_n(B)+\Delta_{n+1}(B)$
    and Lemma~\ref{thm:al_channel_shell_finite_bound} yield
    \begin{align}
        d_y(\mathcal E_{n+1}(B))
         & \le d_y(\mathcal E_n(B))+2\|\Delta_{n+1}(B)\|
        \le d_y(\mathcal E_n(B))+2a_{n+1},
        \label{eq:al_propagation_increment_step}
    \end{align}
    and applying the induction hypothesis to $d_y(\mathcal E_n(B))$ in
    (\ref{eq:al_propagation_increment_step}) closes the induction.
\end{proof}

\subsection{Stretched-exponential localization errors}

\begin{lemma}[Explicit stretched-exponential shell coefficients]
    \label{thm:al_channel_shell_coefficient_exp}
    \uses{def:al_channel_shell_coefficient}
    \lean{TNLean.PEPS.AreaLaw.localRootChannelShellBound_le_exp}
    \leanok
    Let $C\ge0$, $c>0$, $0<\alpha\le1$, and $\varepsilon_l=C\exp(-cl^\alpha)$.
    With
    \begin{align}
        A & =2+8\sqrt C\exp(c/2), & b & =c/2,
        \label{eq:al_propagation_exp_constants}
    \end{align}
    every $l\in\N$ satisfies
    \begin{align}
        \eta_l & \le A\exp(-bl^\alpha).
        \label{eq:al_propagation_exp_coefficient}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:al_channel_shell_coefficient}
    Here $\sqrt{\varepsilon_l}=\sqrt C\exp(-bl^\alpha)$. Since
    $0<\alpha\le1$ gives $(l+1)^\alpha\le l^\alpha+1$,
    \begin{align}
        \exp(-bl^\alpha)
         & \le\exp(b)\exp(-b(l+1)^\alpha).
        \label{eq:al_propagation_exp_previous}
    \end{align}
    Substituting $1\le\exp(b)$ and~(\ref{eq:al_propagation_exp_previous})
    into~(\ref{eq:al_propagation_coefficient_successor}) gives
    \begin{align}
        \eta_{l+1}
         & \le8\sqrt C\exp(b)\exp(-b(l+1)^\alpha)
        \le A\exp(-b(l+1)^\alpha),
        \label{eq:al_propagation_exp_successor}
    \end{align}
    while $\eta_0=2\le A\exp(-b\cdot0^\alpha)$ at $l=0$.
\end{proof}

\begin{lemma}[Stretched-exponential physical shell bound]
    \label{thm:al_channel_shell_exp_bound}
    \uses{def:al_local_channel_shell,def:qc_site_oscillation}

    \end{align}

         & \le A\exp(-bl^\alpha)\sum_{z\in K_l}d_z(B).

\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_channel_shell_finite_bound,
        thm:al_channel_shell_coefficient_exp,thm:qc_site_oscillation_bounds}
    Combine~(\ref{eq:al_propagation_finite_shell}) for
    $\varepsilon_j=C\exp(-cj^\alpha)$ with
    (\ref{eq:al_propagation_exp_coefficient}) and $\sum_{z\in K_l}d_z(B)\ge0$.
\end{proof}

\begin{lemma}[Stretched-exponential finite one-event increment]

\subsection{The full channel on an anchor component}

In the graph notation of Section~\ref{sec:qc_component_channel}, let $G$ be a
simple graph on $I$, $a\in I$ an anchor with connected component $C(a)$,
$N_l(a)=\{z\in I:d_G(a,z)\le l\}$, $R_a=\max_{z\in C(a)}d_G(a,z)$, and
$\mathcal E_k$ the full root channel~(\ref{eq:qc_component_root_channel}).

\begin{lemma}[Full-channel increment with an explicit finite cutoff]
    \label{thm:al_graph_channel_exp_increment}
    \uses{def:al_local_root_channel,def:qc_site_oscillation,
        def:qc_local_circuit}
    \lean{TNLean.PEPS.AreaLaw.siteOscillation_spectatorRootChannel_le_exp_of_component_support}
    \leanok
    Suppose $0\le k\le\mathbb 1$ and $k\in\mathcal A_{C(a)}$. Let $C\ge0$,
    $c>0$, $0<\alpha\le1$, $n\ge R_a$, and $A,b$ be as in
    (\ref{eq:al_propagation_exp_constants}). If
    $\|k-E_{N_l(a)}(k)\|\le C\exp(-cl^\alpha)$ for all $0\le l\le n$, then for
    every $y\in I$,
    \begin{align}
        \label{eq:al_propagation_graph_increment}
        d_y(\mathcal E_k(B))
         & \le d_y(B)+2\sum_{0\le l\le n,\ d_G(a,y)\le l}
        A\exp(-bl^\alpha)\sum_{z\in N_l(a)}d_z(B).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_channel_exp_increment,thm:qc_graph_ball_stabilization,
        thm:qc_component_channel_stabilization}
    Apply Lemma~\ref{thm:al_channel_exp_increment} with $K_l=N_l(a)$ and use
    $\mathcal E_{N_n(a)}(B)=\mathcal E_k(B)$ from
    Lemma~\ref{thm:qc_component_channel_stabilization} ($n\ge R_a$,
    $k\in\mathcal A_{C(a)}$).
\end{proof}

\subsection{Finite families of channel events}

Let $F$ be a finite label set with anchors $(a_i)_{i\in F}\in I^F$. For
$l\in\N$ and $y,z\in I$, set
\begin{align}
    \label{eq:al_propagation_ball_count}
    m_l(y,z) & =\bigl|\{i\in F:y,z\in N_l(a_i)\}\bigr|.
\end{align}

\begin{lemma}[Finite shell sums as counts of labels]
    \label{thm:al_graph_shell_sum_cardinality}
    \lean{TNLean.PEPS.AreaLaw.sum_graphBall_shell_eq_sum_card}
    \leanok
    For any $(w_l)_{l\ge0}\subset\R$, $(v_z)_{z\in I}\subset\R$, $n\in\N$, and
    $y\in I$,
    \begin{align}
        \label{eq:al_propagation_shell_count}
        \sum_{i\in F}\sum_{0\le l\le n,\ d_G(a_i,y)\le l}
        w_l\sum_{z\in N_l(a_i)}v_z
         & =\sum_{l=0}^{n}w_l\sum_{z\in I}m_l(y,z)v_z.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Interchanging the sums over $i\in F$ and $z\in I$ at each $0\le l\le n$
    with the indicator $\mathbf1_{N_l(a_i)}$ and using
    (\ref{eq:al_propagation_ball_count}) gives
    \begin{align}
        \label{eq:al_propagation_count_reorder}
        \sum_{i\in F}\mathbf1_{N_l(a_i)}(y)
        \sum_{z\in N_l(a_i)}v_z
         & =\sum_{z\in I}\left(\sum_{i\in F}
        \mathbf1_{N_l(a_i)}(y)\mathbf1_{N_l(a_i)}(z)\right)v_z
        =\sum_{z\in I}m_l(y,z)v_z,
    \end{align}
    and multiplying by $w_l$ and summing over $0\le l\le n$
    yields~(\ref{eq:al_propagation_shell_count}).
\end{proof}

\begin{lemma}[Sum of full-channel oscillation increments]
    \label{thm:al_graph_event_increment_cardinality}
    \uses{def:al_local_root_channel,def:qc_site_oscillation,
        def:qc_local_circuit}
    \lean{TNLean.PEPS.AreaLaw.sum_siteOscillation_spectatorRootChannel_sub_le_exp_card}
    \leanok
    Following \cite[Section~10]{OpenAI2026AreaLaw}, for each $i\in F$ let
    $0\le k_i\le\mathbb 1$ be supported on $C(a_i)$ with full root channel
    $\mathcal E_i=\mathcal E_{k_i}$. Let $C\ge0$, $c>0$, $0<\alpha\le1$, $A,b$
    from~(\ref{eq:al_propagation_exp_constants}), and $n\ge\max_{i\in F}R_{a_i}$.
    If every $i\in F$ and $0\le l\le n$ satisfy
    \begin{align}
        \label{eq:al_propagation_family_error}
        \|k_i-E_{N_l(a_i)}(k_i)\| & \le C\exp(-cl^\alpha),
    \end{align}
    then for every $y\in I$,
    \begin{align}
        \label{eq:al_propagation_family_increment}
        \sum_{i\in F}\bigl(d_y(\mathcal E_i(B))-d_y(B)\bigr)
         & \le2A\sum_{l=0}^{n}\exp(-bl^\alpha)
        \sum_{z\in I}m_l(y,z)d_z(B).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_graph_channel_exp_increment,
        thm:al_graph_shell_sum_cardinality}
    Summing~(\ref{eq:al_propagation_graph_increment}) over $i\in F$ and
    applying~(\ref{eq:al_propagation_shell_count}) with
    $w_l=A\exp(-bl^\alpha)$ and $v_z=d_z(B)$ gives
    \begin{align}
        \label{eq:al_propagation_family_shells}
        \sum_{i\in F}\bigl(d_y(\mathcal E_i(B))-d_y(B)\bigr)
         & \le2\sum_{i\in F}\sum_{0\le l\le n,\ d_G(a_i,y)\le l}
        A\exp(-bl^\alpha)\sum_{z\in N_l(a_i)}d_z(B),
    \end{align}
    which equals the right-hand side
    of~(\ref{eq:al_propagation_family_increment}).
\end{proof}

\subsection{Uniform weighted rows of the finite event kernel}

Write $d_G^0(y,z)=d_G(y,z)$ when $d_G(y,z)<\infty$ and $d_G^0(y,z)=0$ when
$d_G(y,z)=\infty$.

\begin{definition}[Finite graph-channel event kernel]
    \label{def:al_graph_event_kernel}
    \lean{TNLean.PEPS.AreaLaw.graphChannelEventKernel}
    \leanok
    For $D,b,\alpha\in\R$ (with $0^0=1$ and $0^\alpha=0$ for $\alpha\ne0$) and
    $N\in\N$, define
    \begin{align}
        \label{eq:al_propagation_event_kernel}
        \mathsf A^{(N)}_{yz}
         & =D\sum_{l=0}^{N}\exp(-bl^\alpha)m_l(y,z).
    \end{align}
\end{definition}

\begin{lemma}[Vanishing kernel between distinct components]
    \label{thm:al_graph_kernel_disconnected}
    \uses{def:al_graph_event_kernel}
    \lean{TNLean.PEPS.AreaLaw.graphChannelEventKernel_eq_zero_of_not_reachable}
    \leanok
    If $d_G(y,z)=\infty$, then $\mathsf A^{(N)}_{yz}=0$ for all $N\in\N$ and
    $D,b,\alpha\in\R$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:al_graph_event_kernel}
    If $y,z\in N_l(a_i)$, then $d_G(y,z)\le d_G(y,a_i)+d_G(a_i,z)\le2l<\infty$.
    Hence $m_l(y,z)=0$ for all $l$, so
    (\ref{eq:al_propagation_event_kernel}) vanishes.
\end{proof}

\begin{lemma}[Finite weighted row bound from ball growth]
    \label{thm:al_graph_kernel_finite_row}
    \uses{def:al_graph_event_kernel}
    \lean{TNLean.PEPS.AreaLaw.sum_graphChannelEventKernel_mul_exp_le}
    \leanok
    Let $D,K,\mu\ge0$ and $b,\alpha>0$, and set
    \begin{align}
        \label{eq:al_propagation_row_weight}
        \theta & =\frac{b}{2\,2^\alpha}.
    \end{align}
    If every $x\in I$ and $l\in\N$ satisfy
    \begin{align}
        \label{eq:al_propagation_ball_growth}
        |N_l(x)|           & \le K(l+1)^2, \\
        \label{eq:al_propagation_anchor_fiber}
        |\{i\in F:a_i=x\}| & \le\mu,
    \end{align}
    then for every $N\in\N$ and $y\in I$,
    \begin{align}
        \label{eq:al_propagation_finite_row}
        \sum_{z\in I}\mathsf A^{(N)}_{yz}
        \exp\bigl(\theta(d_G^0(y,z))^\alpha\bigr)
         & \le D\mu K^2\sum_{l=0}^{N}(l+1)^4\exp\bigl(-(b/2)l^\alpha\bigr).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:al_graph_event_kernel,thm:al_graph_shell_sum_cardinality}

         & =\sum_{x\in N_l(y)}|\{i\in F:a_i=x\}|
        \le\mu K(l+1)^2.
    \end{align}

    \end{align}
    and substituting~(\ref{eq:al_propagation_anchor_count}) into
    (\ref{eq:al_propagation_row_count}) gives
    (\ref{eq:al_propagation_finite_row}).
\end{proof}

\begin{lemma}[A uniform weighted row constant]
    \label{thm:al_graph_kernel_uniform_row}

        \label{eq:al_propagation_uniform_row}

         & \le V.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_graph_kernel_finite_row,lem:al_truncation_error}

    $N\in\N$,

        \label{eq:al_propagation_polynomial_bound}
        (u+1)^2\exp\bigl(-(b/4)u^\alpha\bigr)           & \le M, \\
        \label{eq:al_propagation_series_bound}
        \sum_{l=0}^{N}(l+1)^2
        \exp\bigl(-(b/4)\lfloor l/2\rfloor^\alpha\bigr) & \le S.
    \end{align}

    (\ref{eq:al_propagation_polynomial_bound}) gives

    (\ref{eq:al_propagation_series_bound}), and
    (\ref{eq:al_propagation_fourth_power}) give
    (\ref{eq:al_propagation_uniform_row}) with $V=D\mu K^2MS$.
\end{proof}

\begin{definition}[Canonical extended graph pseudometric]
    \label{def:al_graph_extended_metric}
    \lean{TNLean.PEPS.AreaLaw.graphPseudoEMetricSpace,
        TNLean.PEPS.AreaLaw.graphPseudoEMetricSpace_edist}
    \leanok
    For a simple graph $G$ on $V$, let $d_G(x,y)\in\N\cup\{\infty\}$ be the
    shortest-path distance and $\iota:\N\cup\{\infty\}\to[0,\infty]$ the
    canonical inclusion. Define the extended pseudometric
    \begin{align}
        \label{eq:al_propagation_extended_distance}
        \rho_G(x,y) & =\iota(d_G(x,y)).
    \end{align}
\end{definition}
\begin{proof}
    \leanok
    Since $\iota$ preserves $0$, $+$, and $\le$, symmetry and
    $d_G(x,x)=0$ carry over, and the triangle inequality holds by
    \begin{align}
        \label{eq:al_propagation_extended_triangle}
        \rho_G(x,z) & \le\iota(d_G(x,y)+d_G(y,z))
        =\rho_G(x,y)+\rho_G(y,z).
    \end{align}
\end{proof}

\begin{lemma}[Finite distance and connected components]
    \label{thm:al_graph_extended_components}
    \uses{def:al_graph_extended_metric}
    \lean{TNLean.PEPS.AreaLaw.graphPseudoEMetricSpace_edist_ne_top,
        TNLean.PEPS.AreaLaw.graphPseudoEMetricSpace_edist_eq_top}
    \leanok
    For all $x,y\in V$, $\rho_G(x,y)\ne\infty$ if and only if $x$ and $y$ are
    connected in $G$ (and $\rho_G(x,y)=\infty$ otherwise).
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:al_graph_extended_metric}
    Since $\iota^{-1}(\{\infty\})=\{\infty\}$, $\rho_G(x,y)<\infty$ holds if
    and only if $d_G(x,y)<\infty$.
\end{proof}

For $r\in[0,\infty]$, write $r_{\R}=r$ for $r<\infty$ and $\infty_{\R}=0$.
