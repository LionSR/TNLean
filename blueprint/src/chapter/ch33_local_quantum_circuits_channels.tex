\section{Local channel circuits and conversions}
\label{sec:qc_channels}

Lemmas~\ref{lem:qc_light_cone} and~\ref{lem:qc_light_cone_correlations}
extend from unitary gates to channels, the operations
of a local circuit that may add product ancillas and discard them
\cite{Piroli2021LOCC}.

\begin{definition}[Local channel circuit of depth $T$]
    \label{def:qc_local_channel_circuit}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.Layer, QuantumCircuit.Layer.dual,
        QuantumCircuit.Circuit.seq, QuantumCircuit.Circuit.map_seq_cons,
        QuantumCircuit.Circuit.depth_seq_le_length, QuantumCircuit.Circuit.dual}
    \leanok
    Fix a bond geometry $(B_b)_{b\in\beta}$. A \emph{channel on bond $B$} is a map $\Phi_B(\rho)=\sum_j K_j\rho K_j^\dagger$ with $K_j\in\mathcal A_B$ and $\sum_j K_j^\dagger K_j=\mathbb 1$, with Heisenberg dual $\Phi_B^\dagger(A)=\sum_j K_j^\dagger AK_j$. A \emph{channel layer} is a composition of channels on pairwise disjoint bonds, a \emph{local channel circuit of depth $T$} is a composition $\Phi=\Phi_T\circ\cdots\circ\Phi_1$ of $T$ channel layers with dual $\Phi^\dagger=\Phi_1^\dagger\circ\cdots\circ\Phi_T^\dagger$, and $\rho$ is \emph{prepared in depth $T$ by local channels} if $\rho=\Phi(\bigotimes_i\sigma_i)$ for single-site density matrices $\sigma_i$.
\end{definition}

\begin{lemma}[Duality and complete positivity of local channel circuits]
    \label{lem:qc_local_channel_circuit_properties}
    \uses{def:qc_local_channel_circuit,def:qc_local_circuit}
    \lean{QuantumCircuit.Circuit.isKrausCPTP_map,
        QuantumCircuit.Circuit.trace_map_mul,
        QuantumCircuit.Circuit.map_eq_of_subsingleton}
    \leanok
    Every local channel circuit $\Phi$ is CPTP and satisfies
    \begin{align}
        \Tr(\Phi(\rho)A) & =\Tr(\rho\,\Phi^\dagger(A)).
        \label{eq:qc_channel_circuit_duality}
    \end{align}
    A depth-$T$ unitary circuit $U$ is the depth-$T$ channel circuit $\rho\mapsto U\rho U^\dagger$.
\end{lemma}
\begin{proof}
    \leanok
    For a single bond channel, cyclicity of trace gives $\Tr(\sum_j K_j\rho K_j^\dagger A)=\Tr(\rho\sum_j K_j^\dagger AK_j)$, and $\sum_j K_j^\dagger K_j=\mathbb 1$ implies CPTP; both properties are preserved under composition.
\end{proof}

\begin{theorem}[Light cone of a local channel circuit]
    \label{lem:qc_channel_light_cone}
    \uses{def:qc_local_channel_circuit,def:qc_light_cone_geometry,lem:qc_light_cone_geometries}
    \lean{QuantumCircuit.Circuit.dual_mem_supportedOperators_lightCone,
        QuantumCircuit.Layer.dual_mem_supportedOperators_lightCone}
    \leanok
    If $\Phi$ is a local channel circuit of depth $T$ and $Z\in\mathcal A_X$, then $\Phi^\dagger(Z)\in\mathcal A_{\mathcal L_T(X)}$ (on the ring, within ring distance $T$ of $X$).
\end{theorem}
\begin{proof}
    \leanok
    By induction on $T$ using Lemma~\ref{lem:qc_light_cone_geometries}, it suffices to treat one layer. For a bond $B\cap X=\varnothing$, $K_j\in\mathcal A_B$ commutes with $Z\in\mathcal A_X$, so $\sum_j K_j^\dagger ZK_j=(\sum_j K_j^\dagger K_j)Z=Z$; for bonds $B\cap X\ne\varnothing$, $B\subseteq\mathcal N_1(X)$, so $\sum_j K_j^\dagger ZK_j\in\mathcal A_{\mathcal N_1(X)}$.
\end{proof}

\begin{lemma}[No connected correlations beyond the light cone of channels]
    \label{lem:qc_channel_light_cone_correlations}
    \uses{lem:qc_channel_light_cone,lem:qc_local_channel_circuit_properties}
    \lean{QuantumCircuit.Circuit.dual_mul_of_disjoint_lightCone,
        QuantumCircuit.Layer.dual_mul_of_disjoint_lightCone,
        QuantumCircuit.IsConversion.trace_mul_mul_eq}
    \leanok
    If $\rho$ is prepared in depth $T$ by local channels and $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $\mathcal L_T(X)\cap\mathcal L_T(X')=\varnothing$ (e.g., $d_{\rm ring}(X,X')>2T$), then $\Tr(\rho ZZ')=\Tr(\rho Z)\Tr(\rho Z')$.
\end{lemma}
\begin{proof}
    \leanok
    For a single layer $L$ and $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$ with $\mathcal N_1(X)\cap\mathcal N_1(Y)=\varnothing$, no bond of $L$ meets both $X$ and $Y$. Since $[K_j,B]=0\implies\Phi_B^\dagger(AB)=\Phi_B^\dagger(A)B$, splitting $L^\dagger$ into bonds meeting $X$ and the remaining bonds (and applying Theorem~\ref{lem:qc_channel_light_cone}) gives $L^\dagger(ZZ')=L^\dagger(Z)L^\dagger(Z')$. Induction on $T$ yields $\Phi^\dagger(ZZ')=\Phi^\dagger(Z)\Phi^\dagger(Z')$ with disjoint supports $\mathcal L_T(X),\mathcal L_T(X')$, and (\ref{eq:qc_channel_circuit_duality}) on $\rho=\Phi(\bigotimes_i\sigma_i)$ ($\Tr\sigma_i=1$) factors the trace.
\end{proof}

Ancillas are attached and discarded site by site, so the operations that
change the local dimension act on single sites.

\begin{definition}[Onsite channel]
    \label{def:qc_onsite_channel}
    \lean{ManyBody.prodFamily,
        QuantumCircuit.IsProtocol.ofOnsite,
        KrausFamily.ancillaState, KrausFamily.attach,
        KrausFamily.attach_op_apply, KrausFamily.map_attach,
        KrausFamily.traceRight}
    \leanok
    An \emph{onsite channel} $\Phi=\bigotimes_i\Phi_i$ from local dimension $d$ to $e$ is defined by sitewise Kraus operators $K_{ij}\colon\C^d\to\C^e$ with $\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1_d$:
    \begin{align}
        \Phi(\rho)      & =\sum_J K_J\rho K_J^\dagger,    &
        \Phi^\dagger(A) & =\sum_J K_J^\dagger AK_J,       &
        K_J             & =\bigotimes_i K_{i,J(i)}.\notag
    \end{align}
    Attaching an $a$-level ancilla in state $\tau=\sum_k w_k w_k^\dagger$ ($\sum_k\|w_k\|^2=1$) is the onsite channel $X_i\mapsto X_i\otimes\tau$ ($d\to da$), and discarding it is the partial trace $\operatorname{Tr}_{\C^a}$ ($da\to d$)~\cite{Piroli2021LOCC}.
\end{definition}

\begin{lemma}[Onsite channels are product channels]
    \label{lem:qc_onsite_channel_properties}
    \uses{def:qc_onsite_channel}
    \lean{QuantumCircuit.Circuit.map_ofLayer_ofOnsite_piKronecker,
        ManyBody.map_prodFamily_piKronecker,
        KrausFamily.isTP_attach,
        KrausFamily.map_traceRight_map_attach}
    \leanok
    Every onsite channel $\Phi$ is CPTP, satisfies $\Phi(\bigotimes_i\sigma_i)=\bigotimes_i\Phi_i(\sigma_i)$ and $\Tr(\Phi(\rho)A)=\Tr(\rho\,\Phi^\dagger(A))$, and satisfies $\operatorname{discard}_\tau\circ\operatorname{attach}_\tau=\operatorname{id}$.
\end{lemma}
\begin{proof}
    \leanok
    We have $\sum_J K_J^\dagger K_J=\bigotimes_i\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1$, $\sum_J K_J(\bigotimes_i\sigma_i)K_J^\dagger=\bigotimes_i\sum_j K_{ij}\sigma_i K_{ij}^\dagger$, $\Tr(\Phi(\rho)A)=\Tr(\rho\,\Phi^\dagger(A))$ by trace cyclicity, and $\operatorname{Tr}_{\C^a}(X_i\otimes\tau)=(\Tr\tau)X_i=X_i$.
\end{proof}

\begin{lemma}[Onsite channels do not spread supports]
    \label{lem:qc_onsite_channel_locality}
    \uses{def:qc_onsite_channel}
    \lean{QuantumCircuit.Layer.lightCone_ofOnsite}
    \leanok
    For any onsite channel $\Phi$, the backward light cone of a region $X$ is $X$ itself. Consequently, $Z\in\mathcal A_X\implies\Phi^\dagger(Z)\in\mathcal A_X$, and if $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$ with $X\cap Y=\varnothing$, then $\Phi^\dagger(ZZ')=\Phi^\dagger(Z)\Phi^\dagger(Z')$.
\end{lemma}
\begin{proof}
    \leanok
    An onsite channel is a layer without bonds, so no bond enlarges a region. Since $\Phi_i^\dagger(\mathbb 1_e)=\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1_d$, $\Phi^\dagger$ preserves identity factors outside $X$; for $X\cap Y=\varnothing$, at least one of the two operators is the identity at each site $i$, so the dual factors over the sites.
\end{proof}

\begin{definition}[Local channel conversion]
    \label{def:qc_local_channel_conversion}
    \uses{def:qc_local_channel_circuit,def:qc_onsite_channel}
    \lean{QuantumCircuit.IsProtocol,
        QuantumCircuit.IsConversion}
    \leanok
    Modeling the ancilla-assisted circuits $V'=U_\ell V_\ell\cdots U_1V_1U_0$ of~\cite{Piroli2021LOCC}, a \emph{local channel protocol of depth $T$} is a composition
    \begin{align}
        \Psi & =\Phi_T\circ L_T\circ\Phi_{T-1}\circ\cdots\circ L_1\circ\Phi_0,\notag
    \end{align}
    where each $L_t$ is a channel layer and each $\Phi_t$ is a composition of onsite channels (changing local dimensions as needed). A matrix $\rho$ is \emph{converted in depth $T$} into $\sigma$ if $\sigma=\Psi(\rho)$ for a protocol $\Psi$ of depth at most $T$.
\end{definition}

\begin{lemma}[Basic properties of local channel conversions]
    \label{lem:qc_local_channel_conversion_properties}
    \uses{def:qc_local_channel_conversion,def:qc_local_channel_circuit,lem:qc_onsite_channel_properties}
    \lean{QuantumCircuit.IsProtocol.isKrausCPTP,
        QuantumCircuit.IsProtocol.ofLayer,
        QuantumCircuit.IsProtocol.seq,
        QuantumCircuit.IsConversion.ofOnsite_seq,
        QuantumCircuit.IsConversion.refl,
        QuantumCircuit.IsConversion.mono,
        QuantumCircuit.IsConversion.density}
    \leanok
    Every local channel protocol is CPTP (hence maps density matrices to density matrices). Attaching ancillas, running a depth-$T$ local channel circuit, and discarding ancillas is a depth-$T$ protocol. Conversion is reflexive at depth $0$, monotone in $T$, and every state prepared in depth $T$ by local channels is converted in depth $T$ from a product state.
\end{lemma}
\begin{proof}
    \leanok
    By induction on Definition~\ref{def:qc_local_channel_conversion}, onsite channels (Lemma~\ref{lem:qc_onsite_channel_properties}) and channel layers (Lemma~\ref{lem:qc_local_channel_circuit_properties}) are CPTP, onsite steps add $0$ to depth, and channel layers add $1$.
\end{proof}

\begin{lemma}[Composition of conversions]
    \label{lem:qc_local_channel_conversion_trans}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsConversion.trans,
        QuantumCircuit.IsProtocol.comp}
    \leanok
    If $\rho$ is converted in depth $T_1$ into $\sigma$ and $\sigma$ is converted in depth $T_2$ into $\tau$, then $\rho$ is converted in depth $T_1+T_2$ into $\tau$.
\end{lemma}
\begin{proof}
    \leanok
    Composing protocols $\Psi_2\circ\Psi_1$ merges the initial onsite block of $\Psi_2$ into the final onsite block $\Phi_{T_1}$ of $\Psi_1$, summing the layer counts to $T_1+T_2$.
\end{proof}

\begin{lemma}[Light cone of a local channel protocol]
    \label{lem:qc_protocol_light_cone}
    \uses{def:qc_local_channel_conversion,lem:qc_onsite_channel_locality,lem:qc_channel_light_cone,lem:qc_channel_light_cone_correlations}
    \lean{QuantumCircuit.IsProtocol.traceAdjointMap_mem_supportedOperators_graphNeighborhood,
        QuantumCircuit.IsProtocol.traceAdjointMap_mul_of_disjoint_graphNeighborhood,
        Matrix.trace_traceAdjointMap_mul}
    \leanok
    Extending~\cite[Theorem~1]{Malz2024Preparation}, every depth-$T$ local channel protocol $\Psi$ has a dual $\Psi^\dagger$ satisfying $\Tr(\Psi(\rho)A)=\Tr(\rho\,\Psi^\dagger(A))$, mapping $\mathcal A_X$ into $\mathcal A_{\mathcal L_T(X)}$, and satisfying
    \begin{align}
        \Psi^\dagger(ZZ') & =\Psi^\dagger(Z)\,\Psi^\dagger(Z')
        \label{eq:qc_conversion_dual_mul}
    \end{align}
    whenever $\mathcal L_T(X)\cap\mathcal L_T(Y)=\varnothing$ for $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$.
\end{lemma}
\begin{proof}
    \leanok
    By induction on Definition~\ref{def:qc_local_channel_conversion}: onsite channels preserve supports and disjoint-support products by Lemma~\ref{lem:qc_onsite_channel_locality}, and each channel layer expands the light-cone radius by $1$ and preserves disjoint-neighborhood products by Theorem~\ref{lem:qc_channel_light_cone} and Lemma~\ref{lem:qc_channel_light_cone_correlations}.
\end{proof}

\begin{lemma}[No connected correlations after a conversion]
    \label{lem:qc_local_channel_conversion_correlations}
    \uses{def:qc_local_channel_conversion,lem:qc_protocol_light_cone}
    \lean{QuantumCircuit.IsConversion.trace_mul_mul_eq_of_isGraphSeparated}
    \leanok
    If $\sigma$ is converted in depth $T$ from $\rho=\bigotimes_i\sigma_i$ ($\Tr\sigma_i=1$) and $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $d_{\rm ring}(X,X')>2T$, then $\Tr(\sigma ZZ')=\Tr(\sigma Z)\Tr(\sigma Z')$~\cite{Piroli2021LOCC}.
\end{lemma}
\begin{proof}
    \leanok
    Write $\sigma=\Psi(\rho)$ with $\operatorname{depth}(\Psi)=T'\le T$. Since $d_{\rm ring}(X,X')>2T\ge2T'$, $\mathcal L_{T'}(X)\cap\mathcal L_{T'}(X')=\varnothing$, so Lemma~\ref{lem:qc_protocol_light_cone} and (\ref{eq:qc_conversion_dual_mul}) give
    \begin{align}
        \Tr(\sigma ZZ') & =\Tr\bigl(\rho\,\Psi^\dagger(ZZ')\bigr)
        =\Tr\bigl(\rho\,\Psi^\dagger(Z)\Psi^\dagger(Z')\bigr)
        =\Tr\bigl(\rho\,\Psi^\dagger(Z)\bigr)\Tr\bigl(\rho\,\Psi^\dagger(Z')\bigr)
        =\Tr(\sigma Z)\Tr(\sigma Z').\notag
    \end{align}
\end{proof}

\begin{definition}[Blocked local channel conversion]
    \label{def:qc_blocked_local_channel_conversion}
    \uses{def:qc_local_channel_conversion,def:peps_owner_layout}
    \lean{ManyBody.blockConfigEquiv,
        ManyBody.reindex_blockConfigEquiv_symm_piKronecker}
    \leanok
    Following~\cite{Piroli2021LOCC}, grouping $k\ge1$ consecutive sites $jk,\dots,jk+k-1$ of an $Mk$-site ring of local dimension $d$ into block $j\in\Z_M$ of dimension $d^k$ defines the $*$-isomorphism
    \begin{align}
        \beta_k\colon M_d^{\otimes Mk}\xrightarrow{\sim} M_{d^k}^{\otimes M},\qquad
        \beta_k\Bigl(\bigotimes_{i=0}^{Mk-1}m_i\Bigr)=\bigotimes_{j=0}^{M-1}(m_{jk}\otimes\cdots\otimes m_{jk+k-1}).\notag
    \end{align}
    A matrix $\rho$ on $Mk$ sites is \emph{converted in depth $T$ after blocking $k$ sites} into $\sigma$ on $M$ sites if $\beta_k(\rho)$ is converted in depth $T$ into $\sigma$ on the blocked ring (compiling blocked gates into original nearest-neighbour gates is separate; see~\cite{gap:psc21_local_channel_phase_scope}).
\end{definition}

\begin{lemma}[Blocking preserves supports]
    \label{lem:qc_blocking_supports}
    \uses{def:qc_blocked_local_channel_conversion}
    \lean{ManyBody.reindex_blockConfigEquiv_place,
        ManyBody.exists_eq_place_of_mem_supportedOperators,
        ManyBody.preimage_graphNeighborhood_cycleGraph_subset}
    \leanok
    Let $\pi_k\colon\Z_{Mk}\to\Z_M$ map $jk+t\mapsto j$ ($0\le t<k$). An operator $K$ placed on a
    set $Y$ of blocks is, under $\beta_k^{-1}$, the same operator placed on the sites
    $\pi_k^{-1}(Y)$. Hence $Z\in\mathcal A_{\pi_k^{-1}(Y)}\iff\beta_k(Z)\in\mathcal A_Y$, and
    $\pi_k^{-1}(\mathcal L_r(X))\subseteq\mathcal L_{kr}(\pi_k^{-1}(X))$.
\end{lemma}
\begin{proof}
    \leanok
    A configuration of the blocks of $Y$ is a configuration of the sites of $\pi_k^{-1}(Y)$, so
    placing $K$ on $Y$ and relabelling the basis gives $K$ placed on $\pi_k^{-1}(Y)$. Every
    operator supported on a region is a placed operator on that region, and every placed operator
    is supported there. Finally, $(j+m)k+t=(jk+t)+km\pmod{Mk}$ with $|km|\le kr$ for $|m|\le r$.
\end{proof}

\begin{lemma}[Light cone and correlations after blocking]
    \label{lem:qc_blocked_conversion_light_cone}
    \uses{def:qc_blocked_local_channel_conversion,lem:qc_blocking_supports,lem:qc_protocol_light_cone,lem:qc_local_channel_conversion_correlations,lem:qc_local_channel_conversion_properties,lem:qc_local_channel_conversion_trans,lem:qc_light_cone_geometries}
    \lean{QuantumCircuit.IsProtocol.traceAdjointMap_comp_block_mem_supportedOperators_graphNeighborhood,
        QuantumCircuit.IsProtocol.traceAdjointMap_comp_block_mul,
        Matrix.trace_piKronecker}
    \leanok
    Let $\Psi$ be a local channel protocol of depth $T$ on the blocked ring $\Z_M$. The dual of
    $\Psi\circ\beta_k$ maps $\mathcal A_X$ ($X\subseteq\Z_M$) into
    $\mathcal A_{\mathcal L_{kT}(\pi_k^{-1}(X))}$ and is multiplicative on block sets $X,Y$ with
    $\mathcal L_T(X)\cap\mathcal L_T(Y)=\varnothing$; on the ring, the $T$-fold neighbourhood of a
    set is its ball of radius $T$. For $\rho=\bigotimes_{i=0}^{Mk-1}\sigma_i$ with
    $\Tr\sigma_i=1$, the blocked matrix $\beta_k(\rho)$ is a product over the blocks of factors of
    unit trace, so Lemma~\ref{lem:qc_local_channel_conversion_correlations} applies to it on the
    blocked ring: if $\sigma$ is converted in depth $T$ after blocking from $\rho$ and
    $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $d_{\rm ring}(X,X')>2T$, then
    $\Tr(\sigma ZZ')=\Tr(\sigma Z)\Tr(\sigma Z')$.
\end{lemma}
\begin{proof}
    \leanok
    Combine Lemma~\ref{lem:qc_protocol_light_cone} on the blocked ring with the support
    preservation of $\beta_k$ from Lemma~\ref{lem:qc_blocking_supports}. The trace of a tensor
    product is the product of the traces, and $\beta_k$ groups the factors of $\rho$ blockwise by
    Definition~\ref{def:qc_blocked_local_channel_conversion}.
\end{proof}

\begin{lemma}[Matching schedules after grouping sites]
    \label{lem:qc_regrouped_matching_schedule}
    \uses{def:qc_blocked_local_channel_conversion}
    \lean{QuantumCircuit.Geometry.ringBlocking}
    \leanok
    Let $M,k\ge1$. Every original bond $\{i,i+1\}$ lies in the union of blocks
    $\pi_k(i)$ and $\pi_k(i)+1$. Color the block bond $j\to j+1$ by $j\bmod2$,
    except that the closing bond receives color $2$ when $M$ is odd.
    Each color consists of disjoint bonds; when $M$ is even only colors $0,1$ occur.
\end{lemma}
\begin{proof}
    \leanok
    The quotient $\lfloor(i+1)/k\rfloor$ equals either $\lfloor i/k\rfloor$ or
    $\lfloor i/k\rfloor+1$, with cyclic reduction at the seam. Distinct intersecting block bonds
    have distinct colors, including the closing bond.
\end{proof}

\begin{definition}[Regrouped channels]
    \label{def:qc_regrouped_channel}
    \uses{def:qc_blocked_local_channel_conversion,def:qc_local_channel_circuit}
    \lean{QuantumCircuit.Layer.exists_coarsen}
    \leanok
    Write $\beta_{k,d}$ for blocking $k$ sites of local dimension $d$.
    For a linear map $\Psi$ from $Mk$ sites of dimension $d$ to $Mk$ sites of
    dimension $e$, set $\widetilde\Psi=\beta_{k,e}\circ\Psi\circ\beta_{k,d}^{-1}$.
    Then $\widetilde\Psi\circ\beta_{k,d}=\beta_{k,e}\circ\Psi$ on every operator.
    Let $L$ be a channel layer each of whose bonds lies, after grouping, in a
    bond of blocks, distinct covering bonds being disjoint. Then $\widetilde L$ is
    a channel layer on the blocks with the covering bonds, trace preserving when
    $L$ is. Its map on a covering bond is the product of the maps of $L$ on the
    bonds it covers and of the onsite maps at the remaining sites of its blocks;
    at a block met by no covering bond it is the tensor product of the $k$
    original onsite maps, with the corresponding basis relabelling.
\end{definition}

\begin{lemma}[Depth and dimension bounds for regrouped protocols]
    \label{lem:qc_regrouped_protocol}
    \uses{def:qc_regrouped_channel,lem:qc_regrouped_matching_schedule,def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsProtocol.block}
    \leanok
    Let $M,k\ge1$ and $d\ge1$. A depth-$T$ local channel protocol on $Mk$ sites
    has a regrouped protocol of depth at most $C(M)T$, where $C(M)=2$ for even
    $M$ and $C(M)=3$ for odd $M$. Onsite operations cost zero depth. For $M=2$
    the bound is $2T$, including both orientations of the periodic bond.
    These are enlarged-site channel costs; the fixed physical-port comparison in
    the source is separate~\cite[Appendix E]{Piroli2021LOCC}.
\end{lemma}
\begin{proof}
    \leanok
    Conjugate each original Kraus operator by blocking. For every block $j$,
    compose all gates whose left site belongs to $j$. Their Kraus operators are
    products of the original Kraus operators and are supported on $\{j,j+1\}$.
    Completeness follows from $\sum_{a,b}(A_aB_b)^\dagger(A_aB_b)=I$.
    Partition these composite channels by the explicit colors of
    Lemma~\ref{lem:qc_regrouped_matching_schedule}. Original matching channels
    commute, so regrouping their products preserves the entire channel.
    Induct on protocol composition, regrouping each onsite map blockwise.
    The depths add.
\end{proof}

\begin{corollary}[An unblocked conversion followed by a blocked conversion]
    \label{cor:qc_unblocked_then_blocked_conversion}
    \uses{lem:qc_regrouped_protocol,lem:qc_local_channel_conversion_trans}
    \lean{QuantumCircuit.IsConversion.block_trans}
    \leanok
    Let $M,k,d\ge1$. If $\rho$ converts to $\sigma$ on the $Mk$-site ring in
    depth at most $T$, and $\sigma$ converts to $\tau$ after grouping $k$ sites
    in depth at most $U$, then $\rho$ converts to $\tau$ after the same grouping
    in depth at most $C(M)T+U$. The input matrices are arbitrary operators.
\end{corollary}
\begin{proof}
    \leanok
    Regroup the first channel and compose it with the second.
    The equality $\widetilde\Psi(\beta_{k,d}(\rho))=\beta_{k,e}(\Psi(\rho))$
    identifies the intermediate operator, and protocol composition adds the depths.
\end{proof}
