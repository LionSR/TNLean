\section{Local channel circuits and conversions}
\label{sec:qc_channels}

Lemmas~\ref{lem:qc_light_cone} and~\ref{lem:qc_light_cone_correlations}
extend from unitary gates to channels, the operations
of a local circuit that may add product ancillas and discard them
\cite{Piroli2021LOCC}.

\begin{definition}[Local channel circuit of depth $T$]
    \label{def:qc_local_channel_circuit}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.ChannelLayer, QuantumCircuit.BondChannelLayer,
        QuantumCircuit.BondChannelLayer.map,
        QuantumCircuit.BondChannelLayer.dual, QuantumCircuit.channelCircuitMap,
        QuantumCircuit.channelCircuitDual,
        QuantumCircuit.IsChannelPreparedInDepth,
        QuantumCircuit.IsBondChannelPreparedInDepth}
    \leanok
    Fix a bond geometry $(B_b)_{b\in\beta}$. A \emph{channel on bond $B$} is a map $\Phi_B(\rho)=\sum_j K_j\rho K_j^\dagger$ with $K_j\in\mathcal A_B$ and $\sum_j K_j^\dagger K_j=\mathbb 1$, with Heisenberg dual $\Phi_B^\dagger(A)=\sum_j K_j^\dagger AK_j$. A \emph{channel layer} is a composition of channels on pairwise disjoint bonds, a \emph{local channel circuit of depth $T$} is a composition $\Phi=\Phi_T\circ\cdots\circ\Phi_1$ of $T$ channel layers with dual $\Phi^\dagger=\Phi_1^\dagger\circ\cdots\circ\Phi_T^\dagger$, and $\rho$ is \emph{prepared in depth $T$ by local channels} if $\rho=\Phi(\bigotimes_i\sigma_i)$ for single-site density matrices $\sigma_i$.
\end{definition}

\begin{lemma}[Duality and complete positivity of local channel circuits]
    \label{lem:qc_local_channel_circuit_properties}
    \uses{def:qc_local_channel_circuit, def:qc_local_circuit}
    \lean{QuantumCircuit.channelCircuitMap_isKrausCPTP,
        QuantumCircuit.trace_channelCircuitMap_mul,
        QuantumCircuit.channelCircuitMap_map_toChannelLayer}
    \leanok
    Every local channel circuit $\Phi$ is CPTP and satisfies
    \begin{align}
        \Tr(\Phi(\rho)A) & =\Tr(\rho\,\Phi^\dagger(A)).
        \label{eq:qc_channel_circuit_duality}
    \end{align}
    A depth-$T$ unitary circuit $U$ is the depth-$T$ channel circuit $\rho\mapsto U\rho U^\dagger$.
\end{lemma}
\begin{proof}
    \leanok
    For a single bond channel, cyclicity of trace gives $\Tr(\sum_j K_j\rho K_j^\dagger A)=\Tr(\rho\sum_j K_j^\dagger AK_j)$, and $\sum_j K_j^\dagger K_j=\mathbb 1$ implies CPTP; both properties are preserved under composition.
\end{proof}

\begin{lemma}[Light cone of a local channel circuit]
    \label{lem:qc_channel_light_cone}
    \uses{def:qc_local_channel_circuit, def:qc_light_cone_geometry,
        lem:qc_light_cone_geometries}
    \lean{QuantumCircuit.channelCircuitDual_mem_supportedOperators,
        QuantumCircuit.channelCircuitDual_mem_supportedOperators_lightCone,
        QuantumCircuit.BondChannelLayer.dual_mem_supportedOperators,
        QuantumCircuit.ChannelLayer.dual_mem_supportedOperators}
    \leanok
    If $\Phi$ is a local channel circuit of depth $T$ and $Z\in\mathcal A_X$, then $\Phi^\dagger(Z)\in\mathcal A_{\mathcal L_T(X)}$ (on the ring, within ring distance $T$ of $X$).
\end{lemma}
\begin{proof}
    \leanok
    By induction on $T$ using Lemma~\ref{lem:qc_light_cone_geometries}, it suffices to treat one layer. For a bond $B\cap X=\varnothing$, $K_j\in\mathcal A_B$ commutes with $Z\in\mathcal A_X$, so $\sum_j K_j^\dagger ZK_j=(\sum_j K_j^\dagger K_j)Z=Z$; for bonds $B\cap X\ne\varnothing$, $B\subseteq\mathcal N_1(X)$, so $\sum_j K_j^\dagger ZK_j\in\mathcal A_{\mathcal N_1(X)}$.
\end{proof}

\begin{lemma}[No connected correlations beyond the light cone of channels]
    \label{lem:qc_channel_light_cone_correlations}
    \uses{lem:qc_channel_light_cone, lem:qc_local_channel_circuit_properties}
    \lean{QuantumCircuit.trace_mul_mul_eq_of_isChannelPreparedInDepth,
        QuantumCircuit.IsBondChannelPreparedInDepth.trace_mul_mul_eq,
        QuantumCircuit.channelCircuitDual_mul,
        QuantumCircuit.channelCircuitDual_mul_of_disjoint_lightCone,
        QuantumCircuit.BondChannelLayer.dual_mul,
        QuantumCircuit.ChannelLayer.dual_mul,
        QuantumCircuit.trace_rectKronecker_mul_mul}
    \leanok
    If $\rho$ is prepared in depth $T$ by local channels and $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $\mathcal L_T(X)\cap\mathcal L_T(X')=\varnothing$ (e.g., $d_{\rm ring}(X,X')>2T$), then $\Tr(\rho ZZ')=\Tr(\rho Z)\Tr(\rho Z')$.
\end{lemma}
\begin{proof}
    \leanok
    For a single layer $L$ and $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$ with $\mathcal N_1(X)\cap\mathcal N_1(Y)=\varnothing$, no bond of $L$ meets both $X$ and $Y$. Since $[K_j,B]=0\implies\Phi_B^\dagger(AB)=\Phi_B^\dagger(A)B$, splitting $L^\dagger$ into bonds meeting $X$ and the remaining bonds (and applying Lemma~\ref{lem:qc_channel_light_cone}) gives $L^\dagger(ZZ')=L^\dagger(Z)L^\dagger(Z')$. Induction on $T$ yields $\Phi^\dagger(ZZ')=\Phi^\dagger(Z)\Phi^\dagger(Z')$ with disjoint supports $\mathcal L_T(X),\mathcal L_T(X')$, and (\ref{eq:qc_channel_circuit_duality}) on $\rho=\Phi(\bigotimes_i\sigma_i)$ ($\Tr\sigma_i=1$) factors the trace.
\end{proof}

Ancillas are attached and discarded site by site, so the operations that
change the local dimension act on single sites.

\begin{definition}[Onsite channel]
    \label{def:qc_onsite_channel}
    \lean{Matrix.rectKronecker, Matrix.rectKronecker_one,
        QuantumCircuit.OnsiteChannel,
        QuantumCircuit.OnsiteChannel.map, QuantumCircuit.OnsiteChannel.dual,
        QuantumCircuit.OnsiteChannel.id, QuantumCircuit.OnsiteChannel.attach,
        QuantumCircuit.OnsiteChannel.attachZero,
        QuantumCircuit.OnsiteChannel.attach_siteMap_apply,
        QuantumCircuit.OnsiteChannel.discard,
        QuantumCircuit.OnsiteChannel.discard_siteMap_apply}
    \leanok
    An \emph{onsite channel} $\Phi=\bigotimes_i\Phi_i$ from local dimension $d$ to $e$ is defined by sitewise Kraus operators $K_{ij}\colon\C^d\to\C^e$ with $\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1_d$:
    \begin{align}
        \Phi(\rho)      & =\sum_J K_J\rho K_J^\dagger,    &
        \Phi^\dagger(A) & =\sum_J K_J^\dagger AK_J,       &
        K_J             & =\bigotimes_i K_{i,J(i)}.\notag
    \end{align}
    Attaching an $a$-level ancilla in state $\tau=\sum_k w_k w_k^\dagger$ ($\sum_k\|w_k\|^2=1$) is the onsite channel $X_i\mapsto X_i\otimes\tau$ ($d\to da$), and discarding it is the partial trace $\operatorname{Tr}_{\C^a}$ ($da\to d$)~\cite{Piroli2021LOCC}.
\end{definition}

\begin{lemma}[Onsite channels are product channels]
    \label{lem:qc_onsite_channel_properties}
    \uses{def:qc_onsite_channel}
    \lean{QuantumCircuit.OnsiteChannel.map_isKrausCPTP,
        QuantumCircuit.OnsiteChannel.map_rectKronecker,
        QuantumCircuit.OnsiteChannel.map_finKronecker,
        QuantumCircuit.OnsiteChannel.trace_map_mul,
        QuantumCircuit.OnsiteChannel.discard_siteMap_attach_siteMap}
    \leanok
    Every onsite channel $\Phi$ is CPTP, satisfies $\Phi(\bigotimes_i\sigma_i)=\bigotimes_i\Phi_i(\sigma_i)$ and $\Tr(\Phi(\rho)A)=\Tr(\rho\,\Phi^\dagger(A))$, and satisfies $\operatorname{discard}_\tau\circ\operatorname{attach}_\tau=\operatorname{id}$.
\end{lemma}
\begin{proof}
    \leanok
    We have $\sum_J K_J^\dagger K_J=\bigotimes_i\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1$, $\sum_J K_J(\bigotimes_i\sigma_i)K_J^\dagger=\bigotimes_i\sum_j K_{ij}\sigma_i K_{ij}^\dagger$, $\Tr(\Phi(\rho)A)=\Tr(\rho\,\Phi^\dagger(A))$ by trace cyclicity, and $\operatorname{Tr}_{\C^a}(X_i\otimes\tau)=(\Tr\tau)X_i=X_i$.
\end{proof}

\begin{lemma}[Onsite channels do not spread supports]
    \label{lem:qc_onsite_channel_locality}
    \uses{def:qc_onsite_channel}
    \lean{QuantumCircuit.OnsiteChannel.dual_mem_supportedOperators,
        QuantumCircuit.OnsiteChannel.dual_mul,
        QuantumCircuit.OnsiteChannel.dual_rectKronecker,
        QuantumCircuit.OnsiteChannel.dual_finKronecker,
        QuantumCircuit.rectangularKrausMap_rectKronecker_rectKronecker}
    \leanok
    For any onsite channel $\Phi$, $\Phi^\dagger(\bigotimes_i m_i)=\bigotimes_i\Phi_i^\dagger(m_i)$. Consequently, $Z\in\mathcal A_X\implies\Phi^\dagger(Z)\in\mathcal A_X$, and if $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$ with $X\cap Y=\varnothing$, then $\Phi^\dagger(ZZ')=\Phi^\dagger(Z)\Phi^\dagger(Z')$.
\end{lemma}
\begin{proof}
    \leanok
    Since $\Phi_i^\dagger(\mathbb 1_e)=\sum_j K_{ij}^\dagger K_{ij}=\mathbb 1_d$, $\Phi^\dagger(\bigotimes_i m_i)=\bigotimes_i\Phi_i^\dagger(m_i)$ preserves identity factors outside $X$; for $X\cap Y=\varnothing$, at least one of $m_i,m'_i$ is $\mathbb 1_e$ at each site $i$, so $\Phi_i^\dagger(m_im'_i)=\Phi_i^\dagger(m_i)\Phi_i^\dagger(m'_i)$.
\end{proof}

\begin{definition}[Local channel conversion]
    \label{def:qc_local_channel_conversion}
    \uses{def:qc_local_channel_circuit, def:qc_onsite_channel}
    \lean{QuantumCircuit.IsLocalChannelProtocol,
        QuantumCircuit.IsLocalChannelConversion}
    \leanok
    Modeling the ancilla-assisted circuits $V'=U_\ell V_\ell\cdots U_1V_1U_0$ of~\cite{Piroli2021LOCC}, a \emph{local channel protocol of depth $T$} is a composition
    \begin{align}
        \Psi & =\Phi_T\circ L_T\circ\Phi_{T-1}\circ\cdots\circ L_1\circ\Phi_0,\notag
    \end{align}
    where each $L_t$ is a channel layer and each $\Phi_t$ is a composition of onsite channels (changing local dimensions as needed). A matrix $\rho$ is \emph{converted in depth $T$} into $\sigma$ if $\sigma=\Psi(\rho)$ for a protocol $\Psi$ of depth at most $T$.
\end{definition}

\begin{lemma}[Basic properties of local channel conversions]
    \label{lem:qc_local_channel_conversion_properties}
    \uses{def:qc_local_channel_conversion, def:qc_local_channel_circuit,
        lem:qc_onsite_channel_properties}
    \lean{QuantumCircuit.IsLocalChannelProtocol.isKrausCPTP,
        QuantumCircuit.IsLocalChannelProtocol.channelCircuitMap_comp,
        QuantumCircuit.isLocalChannelConversion_circuit,
        QuantumCircuit.IsChannelPreparedInDepth.exists_isLocalChannelConversion,
        QuantumCircuit.IsLocalChannelConversion.refl,
        QuantumCircuit.IsLocalChannelConversion.mono,
        QuantumCircuit.IsLocalChannelConversion.exists_isKrausCPTP,
        QuantumCircuit.IsLocalChannelConversion.density}
    \leanok
    Every local channel protocol is CPTP (hence maps density matrices to density matrices). Attaching ancillas, running a depth-$T$ local channel circuit, and discarding ancillas is a depth-$T$ protocol. Conversion is reflexive at depth $0$, monotone in $T$, and every state prepared in depth $T$ by local channels is converted in depth $T$ from a product state.
\end{lemma}
\begin{proof}
    \leanok
    By induction on Definition~\ref{def:qc_local_channel_conversion}, onsite channels (Lemma~\ref{lem:qc_onsite_channel_properties}) and channel layers (Lemma~\ref{lem:qc_local_channel_circuit_properties}) are CPTP, onsite steps add $0$ to depth, and channel layers add $1$.
\end{proof}

\begin{lemma}[Composition of conversions]
    \label{lem:qc_local_channel_conversion_trans}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsLocalChannelConversion.trans,
        QuantumCircuit.IsLocalChannelProtocol.comp}
    \leanok
    If $\rho$ is converted in depth $T_1$ into $\sigma$ and $\sigma$ is converted in depth $T_2$ into $\tau$, then $\rho$ is converted in depth $T_1+T_2$ into $\tau$.
\end{lemma}
\begin{proof}
    \leanok
    Composing protocols $\Psi_2\circ\Psi_1$ merges the initial onsite block of $\Psi_2$ into the final onsite block $\Phi_{T_1}$ of $\Psi_1$, summing the layer counts to $T_1+T_2$.
\end{proof}

\begin{lemma}[Light cone of a local channel protocol]
    \label{lem:qc_protocol_light_cone}
    \uses{def:qc_local_channel_conversion, lem:qc_onsite_channel_locality,
        lem:qc_channel_light_cone, lem:qc_channel_light_cone_correlations}
    \lean{QuantumCircuit.IsLocalChannelProtocol.exists_dual}
    \leanok
    Extending~\cite[Theorem~1]{Malz2024Preparation}, every depth-$T$ local channel protocol $\Psi$ has a dual $\Psi^\dagger$ satisfying $\Tr(\Psi(\rho)A)=\Tr(\rho\,\Psi^\dagger(A))$, mapping $\mathcal A_X$ into $\mathcal A_{\mathcal L_T(X)}$, and satisfying
    \begin{align}
        \Psi^\dagger(ZZ') & =\Psi^\dagger(Z)\,\Psi^\dagger(Z')
        \label{eq:qc_conversion_dual_mul}
    \end{align}
    whenever $\mathcal L_T(X)\cap\mathcal L_T(Y)=\varnothing$ for $Z\in\mathcal A_X$, $Z'\in\mathcal A_Y$.
\end{lemma}
\begin{proof}
    \leanok
    By induction on Definition~\ref{def:qc_local_channel_conversion}: onsite channels preserve supports and disjoint-support products by Lemma~\ref{lem:qc_onsite_channel_locality}, and each channel layer expands the light-cone radius by $1$ and preserves disjoint-neighborhood products by Lemmas~\ref{lem:qc_channel_light_cone} and~\ref{lem:qc_channel_light_cone_correlations}.
\end{proof}

\begin{lemma}[No connected correlations after a conversion]
    \label{lem:qc_local_channel_conversion_correlations}
    \uses{def:qc_local_channel_conversion, lem:qc_protocol_light_cone}
    \lean{QuantumCircuit.trace_mul_mul_eq_of_isLocalChannelConversion}
    \leanok
    If $\sigma$ is converted in depth $T$ from $\rho=\bigotimes_i\sigma_i$ ($\Tr\sigma_i=1$) and $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $d_{\rm ring}(X,X')>2T$, then $\Tr(\sigma ZZ')=\Tr(\sigma Z)\Tr(\sigma Z')$~\cite{Piroli2021LOCC}.
\end{lemma}
\begin{proof}
    \leanok
    Write $\sigma=\Psi(\rho)$ with $\operatorname{depth}(\Psi)=T'\le T$. Since $d_{\rm ring}(X,X')>2T\ge2T'$, $\mathcal L_{T'}(X)\cap\mathcal L_{T'}(X')=\varnothing$, so Lemma~\ref{lem:qc_protocol_light_cone} and (\ref{eq:qc_conversion_dual_mul}) give
    \begin{align}
        \Tr(\sigma ZZ') & =\Tr\bigl(\rho\,\Psi^\dagger(ZZ')\bigr)
        =\Tr\bigl(\rho\,\Psi^\dagger(Z)\Psi^\dagger(Z')\bigr)
        =\Tr\bigl(\rho\,\Psi^\dagger(Z)\bigr)\Tr\bigl(\rho\,\Psi^\dagger(Z')\bigr)
        =\Tr(\sigma Z)\Tr(\sigma Z').\notag
    \end{align}
\end{proof}

\begin{definition}[Blocked local channel conversion]
    \label{def:qc_blocked_local_channel_conversion}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.siteBlock, QuantumCircuit.blockSites,
        QuantumCircuit.blockKronecker,
        QuantumCircuit.IsBlockedLocalChannelConversion}
    \leanok
    Following~\cite{Piroli2021LOCC}, grouping $k\ge1$ consecutive sites $jk,\dots,jk+k-1$ of an $Mk$-site ring of local dimension $d$ into block $j\in\Z_M$ of dimension $d^k$ defines the $*$-isomorphism
    \begin{align}
        \beta_k\colon M_d^{\otimes Mk}\xrightarrow{\sim} M_{d^k}^{\otimes M},\qquad
        \beta_k\Bigl(\bigotimes_{i=0}^{Mk-1}m_i\Bigr)=\bigotimes_{j=0}^{M-1}(m_{jk}\otimes\cdots\otimes m_{jk+k-1}).\notag
    \end{align}
    A matrix $\rho$ on $Mk$ sites is \emph{converted in depth $T$ after blocking $k$ sites} into $\sigma$ on $M$ sites if $\beta_k(\rho)$ is converted in depth $T$ into $\sigma$ on the blocked ring (compiling blocked gates into original nearest-neighbour gates is separate; see~\cite{gap:psc21_local_channel_phase_scope}).
\end{definition}

\begin{lemma}[Blocking is a channel that preserves supports]
    \label{lem:qc_blocking_supports}
    \uses{def:qc_blocked_local_channel_conversion}
    \lean{QuantumCircuit.blockSites_isKrausCPTP,
        QuantumCircuit.blockSites_symm_isKrausCPTP,
        QuantumCircuit.trace_blockSites,
        QuantumCircuit.blockSites_rectKronecker,
        QuantumCircuit.blockSites_mem_supportedOperators,
        QuantumCircuit.blockSites_symm_mem_supportedOperators,
        QuantumCircuit.finProdFinEquiv_add_intCast,
        QuantumCircuit.preimage_siteBlock_neighbourhood_subset}
    \leanok
    Let $\pi_k\colon\Z_{Mk}\to\Z_M$ map $jk+t\mapsto j$ ($0\le t<k$). Then $\beta_k$ and $\beta_k^{-1}$ are CPTP, $Z\in\mathcal A_{\pi_k^{-1}(Y)}\implies\beta_k(Z)\in\mathcal A_Y$, $W\in\mathcal A_X\implies\beta_k^{-1}(W)\in\mathcal A_{\pi_k^{-1}(X)}$ (for $d\ge1$), and $\pi_k^{-1}(\mathcal L_r(X))\subseteq\mathcal L_{kr}(\pi_k^{-1}(X))$.
\end{lemma}
\begin{proof}
    \leanok
    Basis relabelling is unitary conjugation, hence CPTP. On product generators, $\beta_k$ groups factors blockwise, so $Z\in\mathcal A_{\pi_k^{-1}(Y)}\implies\beta_k(Z)\in\mathcal A_Y$, and commutant duality gives $\beta_k^{-1}(\mathcal A_X)\subseteq\mathcal A_{\pi_k^{-1}(X)}$. Finally, $(j+m)k+t=(jk+t)+km\pmod{Mk}$ with $|km|\le kr$ for $|m|\le r$.
\end{proof}

\begin{lemma}[Light cone and correlations after blocking]
    \label{lem:qc_blocked_conversion_light_cone}
    \uses{def:qc_blocked_local_channel_conversion, lem:qc_blocking_supports,
        lem:qc_protocol_light_cone, lem:qc_local_channel_conversion_correlations,
        lem:qc_local_channel_conversion_properties,
        lem:qc_local_channel_conversion_trans}
    \lean{QuantumCircuit.IsLocalChannelProtocol.exists_dual_comp_blockSites,
        QuantumCircuit.trace_blockKronecker,
        QuantumCircuit.IsBlockedLocalChannelConversion.blockSites,
        QuantumCircuit.IsBlockedLocalChannelConversion.mono,
        QuantumCircuit.IsBlockedLocalChannelConversion.trans,
        QuantumCircuit.IsBlockedLocalChannelConversion.exists_isKrausCPTP,
        QuantumCircuit.IsBlockedLocalChannelConversion.density,
        QuantumCircuit.IsBlockedLocalChannelConversion.exists_dual,
        QuantumCircuit.trace_mul_mul_eq_of_isBlockedLocalChannelConversion}
    \leanok
    Every $\rho$ is converted in depth $0$ after blocking into $\beta_k(\rho)$, and blocked conversion is monotone in $T$, composes with blocked-ring conversions ($T+T'$), and is realized by a CPTP map $\Psi$ (preserving density matrices). For $d\ge1$, $\Psi^\dagger$ maps $\mathcal A_X$ ($X\subseteq\Z_M$) into $\mathcal A_{\mathcal L_{kT}(\pi_k^{-1}(X))}$, is multiplicative on block sets $X,Y$ with $\mathcal L_T(X)\cap\mathcal L_T(Y)=\varnothing$, and for $\rho=\bigotimes_{i=0}^{Mk-1}\sigma_i$ ($\Tr\sigma_i=1$) and $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $d_{\rm ring}(X,X')>2T$, satisfies $\Tr(\sigma ZZ')=\Tr(\sigma Z)\Tr(\sigma Z')$.
\end{lemma}
\begin{proof}
    \leanok
    Combine Lemmas~\ref{lem:qc_local_channel_conversion_properties}, \ref{lem:qc_local_channel_conversion_trans}, \ref{lem:qc_protocol_light_cone}, and~\ref{lem:qc_local_channel_conversion_correlations} on the blocked ring with the support and product-state preservation of $\beta_k$ from Lemma~\ref{lem:qc_blocking_supports}.
\end{proof}
