\section{Localization from an arbitrary region}

\begin{theorem}[Elementary commutator bound]
    \label{thm:qc_commutator_norm_bound}
    \lean{norm_commutator_le}
    \leanok
    For any two elements $X,A$ of a normed ring,
    $\|XA-AX\|\leq2\|A\|\|X\|$.
\end{theorem}
\begin{proof}\leanok
    The triangle inequality and submultiplicativity give
    $\|XA-AX\|\leq\|X\|\|A\|+\|A\|\|X\|$.
\end{proof}

\begin{theorem}[Transferring time evolution across a commutator]
    \label{thm:qc_commutator_reverse_evolution}
    \lean{norm_exp_commutator_eq}
    \leanok
    Let $\mathcal A$ be a complete real normed star algebra satisfying the
    $C^*$ identity, and let $K^*=-K$. For $A,B\in\mathcal A$ and real $t$,
    \begin{align*}
        \|[e^{tK}Ae^{-tK},B]\|=\|[A,e^{-tK}Be^{tK}]\|.
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    Since $K^*=-K$, the element $U=e^{tK}$ is unitary with $U^*=e^{-tK}$.
    Using $U^*U=UU^*=1$,
    \begin{align}
        U\bigl(Ae^{-tK}Be^{tK}-e^{-tK}Be^{tK}A\bigr)U^*
        =UAU^*B-BUAU^*=[e^{tK}Ae^{-tK},B].
    \end{align}
    By the $C^*$ identity, $\|UXU^*\|=\|X\|$ for every $X\in\mathcal A$.
    Taking $X=[A,e^{-tK}Be^{tK}]$ gives the claim.
\end{proof}

\begin{definition}[Constants of graph-ball localization]
    \label{def:area_graph_ball_localization_constants}
    \lean{TNLean.PEPS.AreaLaw.graphBallLocalizationPrefactor,
        TNLean.PEPS.AreaLaw.graphBallLocalizationVelocity,
        TNLean.PEPS.AreaLaw.graphBallLocalizationPrefactor_pos,
        TNLean.PEPS.AreaLaw.graphBallLocalizationVelocity_nonneg}
    \leanok
    For an integer $R\geq0$ and a real number $J$, put $v_R=1+2R(R+1)$,
    $\mu_R=2^{v_R-1}$, and
    \begin{align*}
        C_0(R) & =2e^R\cdot2\cdot\frac{2^2\,2!\,e^{1/2}}{1-e^{-1/2}}, &
        v(R,J) & =2v_R\mu_RJe^{2R}.
    \end{align*}
    Then $C_0(R)>0$, and $v(R,J)\geq0$ when $J\geq0$. These are the prefactor
    and velocity of the graph-ball localization estimate of
    \cite[Lemma~4.1]{OpenAI2026AreaLaw}, whose decay rate is $1/2$.
\end{definition}

\begin{theorem}[Localization from a region]
    \label{thm:area_region_localization}
    \lean{TNLean.PEPS.AreaLaw.exists_quasilocal_region_localization,
        TNLean.PEPS.AreaLaw.quasilocal_region_localization}
    \leanok
    \uses{def:area_law_hamiltonian,def:qc_site_expectation,
        def:area_graph_ball_localization_constants}
    To establish the operator localization estimate used in
    \cite{OpenAI2026PolynomialPEPS}, fix integers $q\geq1$ and $R\geq0$,
    and a real number $J$.
    There are positive constants $C,v,c$ with the following property.
    Let $\Lambda\subseteq\Z^2$ be any finite domain, and let $H$ be a
    finite-range Hamiltonian of range $R$ and term-norm bound $J$ on
    $\Lambda$. For a region $S\subseteq\Lambda$, write
    $N_l(S)=\{y\in\Lambda:d_\Lambda(y,S)\leq l\}$, with
    $N_l(\varnothing)=\varnothing$. If $A$ is supported on $S$, then for
    every integer $l\geq0$ and real $t$,
    \begin{align*}
        \|\tau_t(A)-E_{N_l(S)}(\tau_t(A))\|
        \leq\|A\|\min\{2,C|\Lambda|e^{v|t|-cl}\}.
    \end{align*}
    The same bound holds with $C=2C_0(R)$, $v=v(R,J)$ and $c=1/2$, the
    constants of Definition~\ref{def:area_graph_ball_localization_constants}.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:quasilocal-lr,thm:qc_commutator_norm_bound,
        thm:qc_commutator_reverse_evolution,
        lem:qc_site_expectation_properties,lem:qc_site_expectation_commutator}
    Let $C_0=C_0(R)$, $v=v(R,J)$ and $c=1/2$. By
    Lemma~\ref{lem:quasilocal-lr}, the singleton localization estimate holds
    with these constants.
    If $y\notin N_l(S)$, then $N_l(y)$ is disjoint from $S$.
    For an on-site unitary $U$ at $y$, the operator
    $E_{N_l(y)}(\tau_{-t}(U))$ therefore commutes with $A$. For the
    difference $D=\tau_{-t}(U)-E_{N_l(y)}(\tau_{-t}(U))$, the singleton
    localization estimate gives $\|D\|\leq C_0e^{v|t|-cl}$, so the two
    commutator identities yield
    \begin{align*}
        \|[\tau_t(A),U]\|=\|[A,D]\|
        \leq2C_0\|A\|e^{v|t|-cl}.
    \end{align*}
    Lemma~\ref{lem:qc_site_expectation_commutator} sums these bounds over
    at most $|\Lambda|$ sites:
    \begin{align*}
        \|\tau_t(A)-E_{N_l(S)}(\tau_t(A))\|
         & \leq\sum_{y\in\Lambda\setminus N_l(S)}2C_0\|A\|e^{v|t|-cl}
        \leq2C_0|\Lambda|\,\|A\|e^{v|t|-cl}.
    \end{align*}
    Independently, since $\|\tau_t(A)\|=\|A\|$ and
    $\|E_{N_l(S)}(X)\|\leq\|X\|$,
    \begin{align*}
        \|\tau_t(A)-E_{N_l(S)}(\tau_t(A))\|
         & \leq\|\tau_t(A)\|+\|E_{N_l(S)}(\tau_t(A))\|
        \leq2\|A\|.
    \end{align*}
    Taking the smaller of the two bounds proves the assertion with
    $C=2C_0$, $v=v(R,J)$ and $c=1/2$. Replacing $v$ by $\max\{v,1\}$ only
    enlarges the bound and makes the velocity positive.
\end{proof}
