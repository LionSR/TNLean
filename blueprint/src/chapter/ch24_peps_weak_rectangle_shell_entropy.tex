\section{Entropy of safe rectangle shells}

\begin{definition}[Integer rectangles of dyadic cells]
    \label{def:area_template_dyadic_rectangle}
    \lean{TNLean.PEPS.AreaLaw.Geometry.latticeDyadicRect}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_lattice_dyadic_square}
    For $k\in\N$ and $z\in\Z^2$, let
    \begin{align}
        R_{k,z}
         & =\prod_{i=1}^{2}\bigl([2^kz_i,2^kz_i+2^k-1]\cap\Z\bigr).
    \end{align}
    This is the integer rectangle underlying the dyadic square in the
    covering argument of \cite[Lemma~9.4, lines~641--649]{OpenAI2026AreaLaw}.
\end{definition}

\begin{theorem}[Exact rectangle interpretation]
    \label{thm:area_template_dyadic_rectangle}
    \lean{TNLean.PEPS.AreaLaw.Geometry.toFinset_latticeDyadicRect,
        TNLean.PEPS.AreaLaw.Geometry.size_latticeDyadicRect,
        TNLean.PEPS.AreaLaw.Geometry.rectRegion_latticeDyadicRect}
    \leanok
    \uses{def:area_template_dyadic_rectangle,def:area_law_safe_rectangle}
    For every $k\in\N$ and $z\in\Z^2$, the lattice points of $R_{k,z}$
    are exactly $C_{k,z}$, and $\operatorname{size}(R_{k,z})=2^k$.
    For a finite physical domain $\Lambda$ and a region $A\subseteq\Lambda$,
    its physical region is $A\cap C_{k,z}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:area_template_dyadic_rectangle,def:area_law_lattice_dyadic_square}
    The half-open coordinate interval from $2^kz_i$ to $2^k(z_i+1)$
    contains exactly the integers from $2^kz_i$ to $2^kz_i+2^k-1$.
    Each coordinate interval contains $2^k$ sites. Intersecting with
    $A$ gives the equality of physical regions.
\end{proof}

\begin{theorem}[Physical dyadic partition and subadditivity]
    \label{thm:area_template_physical_dyadic_entropy}
    \lean{TNLean.PEPS.AreaLaw.Geometry.biUnion_rectRegion_cappedDyadicPartition,
        TNLean.PEPS.AreaLaw.Geometry.pairwiseDisjoint_rectRegion_cappedDyadicPartition,
        TNLean.PEPS.AreaLaw.Geometry.regionalEntropy_filter_le_sum_rectRegion}
    \leanok
    \uses{def:area_law_capped_dyadic_partition,
        def:area_template_dyadic_rectangle,def:area_law_ground_entropy}
    Fix a finite lattice domain $\Lambda$, a region $A\subseteq\Lambda$,
    a finite lattice set $S$, and $K\in\N$. The physical regions
    $A\cap R_{k,z}$ for $(k,z)\in\mathcal P_K(S)$ are pairwise disjoint,
    and their union is $A\cap S$. For a unit physical vector $\Omega$
    on $\Lambda$ with local dimension $q\in\N$, one has
    \begin{align}
        S_\Omega(A\cap S)
         & \le\sum_{(k,z)\in\mathcal P_K(S)}S_\Omega(A\cap R_{k,z}).
    \end{align}
    This is the physical summation step of
    \cite[Lemma~9.4, lines~651--664]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_capped_dyadic_partition,
        thm:area_template_dyadic_rectangle}
    The selected ambient squares partition $S$. Intersecting them with
    $A$ preserves pairwise disjointness and gives union $A\cap S$.
    Apply finite disjoint-union subadditivity, starting with the empty
    region, whose entropy is zero. This also includes $S=\varnothing$.
\end{proof}

\begin{theorem}[Safe selected squares in a rectangle shell]
    \label{thm:area_law_rectangle_shell_safe_cells}
    \lean{TNLean.PEPS.AreaLaw.IsSafe.isSafe_cappedDyadicPartition_shell}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_template_dyadic_rectangle,
        def:area_law_capped_dyadic_partition}
    Fix a finite lattice domain $\Lambda$, a physical region $A$, and
    $D_0\in\N$. Let $Q$ be a safe integer rectangle and put
    $s=\operatorname{size}(Q)$. Suppose $j,L,K\in\N$ satisfy
    $j\le L$, $2^K\le L$, and
    \begin{align}
        D_0L+L & \le D_0s.
    \end{align}
    Then each selected square of $\mathcal P_K(Q^{+j}\setminus Q)$,
    interpreted as its integer rectangle, is safe.
    This is a sufficient clearance condition for the covering argument in
    \cite[proof of Proposition~9.5, lines~723--729]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_capped_dyadic_partition,
        thm:area_template_dyadic_rectangle,thm:area_law_safe_rectangle_dilation}
    A selected square $R_{k,z}$ is contained in $Q^{+j}$ and has
    $k\le K$ and $\operatorname{size}(R_{k,z})=2^k\le L$. Hence
    \begin{align}
        D_0\operatorname{size}(R_{k,z})+j
         & \le D_0L+L\le D_0s.
    \end{align}
    Safety inside a dilated parent rectangle gives the result.
\end{proof}

\begin{theorem}[Entropy from safe selected squares]
    \label{thm:area_law_capped_dyadic_entropy_cover}
    \lean{TNLean.PEPS.AreaLaw.Geometry.regionalEntropy_filter_le_weighted_cover}
    \leanok
    \uses{def:area_law_ground_entropy,def:area_law_safe_rectangle,
        def:area_template_dyadic_rectangle,def:area_law_capped_dyadic_partition}
    Fix a finite lattice domain $\Lambda$ with local dimension $q$,
    a unit physical vector $\Omega$, a region $A$, and $D_0\in\N$.
    Let $S$ be a finite set of lattice sites, let $K\in\N$, and let
    $e,C\in\R$. Suppose that each square selected by $\mathcal P_K(S)$
    is safe, and that every safe integer rectangle $R$ satisfies
    \begin{align}
        S_\Omega(A\cap R) & \le C\operatorname{size}(R)^{1+e}.
    \end{align}
    Then
    \begin{align}
        S_\Omega(A\cap S)
         & \le C\sum_{(k,z)\in\mathcal P_K(S)}(2^k)^{1+e}.
    \end{align}
    This is the conditional summation step in
    \cite[proof of Lemma~9.4, lines~651--664]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_template_physical_dyadic_entropy,
        thm:area_template_dyadic_rectangle}
    The physical regions $A\cap R_{k,z}$ are pairwise disjoint and
    have union $A\cap S$. Each $R_{k,z}$ is an integer rectangle
    of size $2^k$. Subadditivity and the assumed estimate give
    \begin{align}
        S_\Omega(A\cap S)
         & \le\sum_{(k,z)\in\mathcal P_K(S)}S_\Omega(A\cap R_{k,z}) \\
         & \le\sum_{(k,z)\in\mathcal P_K(S)}C(2^k)^{1+e}
        =C\sum_{(k,z)\in\mathcal P_K(S)}(2^k)^{1+e}.
    \end{align}
\end{proof}

\begin{theorem}[Rectangle-shell entropy from a safe-box estimate]
    \label{thm:area_law_rectangle_shell_entropy}
    \lean{TNLean.PEPS.AreaLaw.IntRect.regionalEntropy_shell_le_of_safe_box}
    \leanok
    \uses{def:area_law_ground_entropy,def:area_law_safe_rectangle}
    Fix a finite lattice domain $\Lambda$ with local dimension $q$,
    a unit physical vector $\Omega$, a region $A$, and $D_0\in\N$.
    Let $e>0$ and $C\ge0$, and assume that every safe integer
    rectangle $R$ satisfies
    \begin{align}
        S_\Omega(A\cap R) & \le C\operatorname{size}(R)^{1+e}.
    \end{align}
    Let $Q$ be safe and put $s=\operatorname{size}(Q)$. For $j,L,K\in\N$
    with $j\le L\le s$, $2^K\le L<2^{K+1}$, and $D_0L+L\le D_0s$,
    one has
    \begin{align}
        S_\Omega\bigl(A\cap(Q^{+j}\setminus Q)\bigr)
         & \le C\left(24+\frac{64}{2^e-1}\right)sL^e.
    \end{align}
    This is the conditional shell entropy estimate used in
    \cite[proof of Proposition~9.5, lines~717--729]{OpenAI2026AreaLaw}.
    % This auxiliary estimate does not assert the one-step entropy improvement.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_capped_dyadic_entropy_cover,
        thm:area_law_rectangle_shell_safe_cells,
        thm:area_law_rectangle_shell_weighted_cover}
    The actual selected squares give disjoint physical rectangle regions
    whose union is $A\cap(Q^{+j}\setminus Q)$. Each is safe.
    Subadditivity and the assumed safe-box estimate give
    \begin{align}
        S_\Omega\bigl(A\cap(Q^{+j}\setminus Q)\bigr)
         & \le C\sum_{(k,z)\in\mathcal P_K(Q^{+j}\setminus Q)}(2^k)^{1+e} \\
         & \le C\left(24+\frac{64}{2^e-1}\right)sL^e.
    \end{align}
    The last step multiplies the weighted covering estimate by $C\ge0$.
\end{proof}
