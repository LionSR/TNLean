\section{Isolated stars of the initial mesh}
\label{sec:area_law_initial_stars}

We use the normalized fan of Definition~\ref{def:area_law_repair_fan},
with its eight triangles $T_s$, $s\in J$, the final endpoint $b_s$ of
segment $s$, and the counterclockwise successor $\tau$ of
Definition~\ref{def:area_law_cell_fan_successor}. For $v\in\mathbb R^2$
and $\rho>0$ put $W_s(v,\rho)=\{v+\rho x:x\in T_s\}$ and
$Q_\rho(v)=v+[-\rho,\rho]^2$. Distances are maximum-norm distances.
The endpoints $b_s$ are the eight nonzero vectors with coordinates in
$\{-1,0,1\}$, that is, the axis and diagonal directions.

\begin{definition}[Isolated star]
    \label{def:area_law_isolated_star}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairFanDirection,
        TNLean.PEPS.AreaLaw.Geometry.repairFanDirection_mem,
        TNLean.PEPS.AreaLaw.Geometry.IsIsolatedStar,
        TNLean.PEPS.AreaLaw.Geometry.RepairStage.HasIsolatedStars}
    \leanok
    \uses{def:area_law_repair_fan,def:area_law_repair_current_assignment,
        def:area_law_cell_fan_successor,def:area_law_repair_stage}
    Let $(R_i)_{i\in I}$ be open regions with colors
    $\xi:I\to\{0,1\}$. The square $Q_\rho(v)$ is an isolated star of
    this family when there is $\sigma:J\to I$ with the following
    properties.
    \begin{enumerate}
        \item For every $s$, $\operatorname{int}W_s(v,\rho)\subseteq
                  R_{\sigma(s)}$.
        \item For every $i$,
              $\overline{R_i}\cap Q_\rho(v)
                  =\bigcup_{\sigma(s)=i}W_s(v,\rho)$.
        \item For every $x\in Q_\rho(v)$, the point $x$ lies on some
              frontier $\partial R_i$ if and only if $x-v$ is a
              nonnegative multiple of $b_s$ for some $s$ with
              $\sigma(s)\ne\sigma(\tau(s))$.
        \item For every $s$, $\xi(\sigma(s))=\xi(\sigma(\tau(s)))$ if
              and only if $\sigma(s)=\sigma(\tau(s))$.
        \item If $\sigma(s)=\sigma(\tau(s))$ for every $s$, then
              $Q_\rho(v)\subseteq R_i$ for some $i$.
    \end{enumerate}
    The rays in the third property are the active rays. A run of
    triangles between consecutive active rays has one identifier, so it
    lies in one region together with its seams. A stage
    $\mathcal D$ has isolated stars with constant $a_0$ when
    $\operatorname{dist}(v_p,v_q)\ge a_0\max(S_p,S_q)$ for distinct
    occurrences $p\ne q$, and $Q_{a_0S_p}(v_p)$ is an isolated star of
    the current regions for every occurrence $p$.
\end{definition}

\begin{lemma}[Shrinking an isolated star]
    \label{lem:area_law_isolated_star_mono}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairCurrentFanTriangle_inter_closedBall,
        TNLean.PEPS.AreaLaw.Geometry.IsIsolatedStar.mono,
        TNLean.PEPS.AreaLaw.Geometry.RepairStage.HasIsolatedStars.mono}
    \leanok
    \uses{def:area_law_isolated_star}
    For $0<\rho'\le\rho$ one has
    $W_s(v,\rho)\cap Q_{\rho'}(v)=W_s(v,\rho')$. Hence an isolated star
    $Q_\rho(v)$ gives an isolated star $Q_{\rho'}(v)$ with the same
    assignment. If every scale of a stage is positive, isolated stars
    with constant $a_0$ give isolated stars with every constant
    $a_0'\in(0,a_0]$.
\end{lemma}
\begin{proof}\leanok
    A point of $T_s$ is $ux$ with $u\in[0,1]$ and $x$ on the outer side,
    where $|x|=1$. Thus $v+\rho ux$ lies in $Q_{\rho'}(v)$ exactly when
    $\rho u\le\rho'$, that is, when it equals $v+\rho'u'x$ with
    $u'=\rho u/\rho'\in[0,1]$. The five properties restrict to the
    smaller square by this identity.
\end{proof}

\begin{theorem}[Isolated star at an actual fine-cell mark]
    \label{thm:area_law_initial_mark_isolated_star}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialRegion_isIsolatedStar}
    \leanok
    \uses{def:area_law_isolated_star,thm:area_law_actual_sector_assignment,
        thm:area_law_initial_active_ray_closed,
        thm:area_law_initial_sector_colors,
        thm:area_law_initial_sector_no_active,
        thm:area_law_concentric_fan_restriction,
        thm:area_law_initial_birth_regularity}
    Fix an origin, a finite endpoint set $Z$, a width $C_0\ge2$, a lower
    layer $k_0\ge50{,}000{,}000$ and arbitrary valid residues at every
    layer. Let $k\ge k_0$, let $z$ index an actual fine cell of layer $k$,
    and let $v$ be one of its nine marks. Then $Q_{t_k/128}(v)$ is an
    isolated star of the actual initial open regions with their colors.
\end{theorem}
\begin{proof}\leanok
    Put $r=t_k/64$, so that $t_k/128=r/2$. Take $\sigma$ to be the
    actual sector assignment. Each scaled triangle $W_s(v,r/2)$ lies in
    the working triangle of half-side $2r$, which lies in its assigned
    open region. Each birth region is the closure of its open region and
    meets $Q_r(v)$ in its assigned triangles $W_s(v,r)$, hence meets
    $Q_{r/2}(v)$ in the triangles $W_s(v,r/2)$. Birth and open regions
    have the same frontier. In the closed square $Q_{r/2}(v)$ these
    frontiers are exactly the directed rays at changes of identifier.
    Neighboring sectors have equal colors exactly when their identifiers
    agree. Without a change, the whole square $Q_r(v)$ lies in one open
    region.
\end{proof}

\begin{lemma}[Isolated stars of the initial mesh]
    \label{lem:area_law_initial_stars}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialMarks_isolatedStars,
        TNLean.PEPS.AreaLaw.Geometry.initialRepairStage_hasIsolatedStars,
        TNLean.PEPS.AreaLaw.Geometry.InitialIsolatedStarsStatement,
        TNLean.PEPS.AreaLaw.Geometry.initialIsolatedStarsStatement_holds}
    \leanok
    \uses{thm:area_law_initial_mark_isolated_star,
        thm:area_law_fine_layer_mark_separation,
        thm:area_law_least_incident_mark_scales,def:area_law_repair_stage}
    Put $a_0=1/128$. Fix an origin, a finite endpoint set, a width
    $C_0\ge2$, a lower layer $k_0\ge50{,}000{,}000$ and arbitrary valid
    residues at every layer. Let $M$ be the set of deduplicated marks of
    the selected belt cells in the layers $k\ge k_0$. Give each $v\in M$
    a positive scale $S_v$ equal to the side of some selected belt cell
    that marks $v$, for instance the smallest such side. Then:
    \begin{enumerate}
        \item distinct marks $v,w$ satisfy
              $\operatorname{dist}(v,w)\ge a_0\max(S_v,S_w)$;
        \item every square $Q_{a_0S_v}(v)$ is an isolated star of the
              actual initial regions.
    \end{enumerate}
    In particular, for every finite domain and cut, the initial stage
    of the recursive family has isolated stars with constant $a_0$.
\end{lemma}
\begin{proof}\leanok
    Let $S_v=t_k$ be realized by a selected belt cell of layer $k$. The
    star is the preceding theorem applied to that cell, since
    $a_0S_v=t_k/128$. For the separation, distinct marks of actual fine
    cells satisfy $\operatorname{dist}(v,w)\ge\max(S_v,S_w)/4$, which
    exceeds $a_0\max(S_v,S_w)$.
\end{proof}
