\section{Sampling an original construction}
\label{sec:peps_original_compression_sampling}

We now combine the local approximation, pair-effect replacement, and source-sampling
estimates for the original gate maps in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
% Source: 04-compression.tex, lines 137--175, 199--229 and 590--600.
The sampled physical operator need not be positive; its identification with a
contraction of local tensors is proved separately.

\begin{definition}[Distributed construction]
    \label{def:peps_distributed_construction}
    \lean{TNLean.Lattice.PartyCircuit.DistributedConstruction}
    \leanok
    \uses{def:peps_original_circuit}
    A distributed construction consists of a finite nonempty party set $P$
    ordered without repetition, finite-dimensional initial private memories
    $H_p$ with unit vectors $x_p\in H_p$ and physical dimensions $d_p$ for
    $p\in P$, a finite ordered list $R$ of finite-dimensional private output
    registers, and a contraction circuit
    \begin{align}
        W:\bigotimes_{p\in P}H_p
        \longrightarrow
        \left(\bigotimes_{p\in P}\mathbb C^{d_p}\right)\otimes\mathcal H_R
        \label{eq:peps_distributed_construction_map}
    \end{align}
    retaining the original gate expansions and participant labels, with no
    numerical bound on $\dim H_p$ or $\dim\mathcal H_R$.

    Assuming $P\neq\varnothing$ permits scalar factors to be absorbed into a
    local tensor as in the proof of Theorem~5.2 without excluding empty-participant
    gates; the $P=\varnothing$ scalar boundary case is recorded in
    \cite{gap:polypeps_nonempty_parties}.
\end{definition}

\begin{definition}[Original input and physical density]
    \label{def:peps_distributed_construction_density}
    \lean{TNLean.Lattice.PartyCircuit.DistributedConstruction.input,
        TNLean.Lattice.PartyCircuit.DistributedConstruction.physicalDensity}
    \leanok
    \uses{def:peps_distributed_construction}
    The input of a distributed construction is the unit product vector
    $\psi=\bigotimes_{p\in P}x_p$, and its unnormalized physical density in the
    party-indexed physical coordinates is
    \begin{align}
        \rho=\operatorname{Tr}_{\mathcal H_R}
        \bigl(|W\psi\rangle\langle W\psi|\bigr),
        \label{eq:peps_distributed_construction_density}
    \end{align}
    where $\operatorname{Tr}_{\mathcal H_R}$ is taken in a finite orthonormal
    basis of $\mathcal H_R$.
\end{definition}

\begin{lemma}[Subnormalized physical density]
    \label{thm:peps_distributed_construction_density}
    \lean{TNLean.Lattice.PartyCircuit.DistributedConstruction.physicalDensity_posSemidef,
        TNLean.Lattice.PartyCircuit.DistributedConstruction.trace_physicalDensity_le_one}
    \leanok
    \uses{def:peps_distributed_construction_density}
    The physical density $\rho$ in Definition~\ref{def:peps_distributed_construction_density}
    satisfies $\rho\ge 0$ and $\operatorname{Tr}\rho\leq1$.
\end{lemma}
\begin{proof}
    \leanok
    Since $|W\psi\rangle\langle W\psi|\ge 0$ and $\operatorname{Tr}_{\mathcal H_R}$
    is positive and trace-preserving, $\rho\ge 0$ and
    $\operatorname{Tr}\rho=\|W\psi\|^2\le\|\psi\|^2=1$ because $\|W\|\le 1$ and
    $\|x_p\|=1$.
\end{proof}

\begin{lemma}[A fixed inverse power meets the per-gate error tolerance]
    \label{thm:peps_fixed_power_gate_budget}
    \lean{TNLean.Lattice.PartyCircuit.inversePower_le_sourceGateBudget_half}
    \leanok
    For $L\ge2$, $C\ge1$, $M,n\in\mathbb N$ with $M\le CL^n$, and $p\in\mathbb R$,
    setting $q=n+\lceil p\rceil_++\lceil16C\rceil_+$ gives
    \begin{align}
        L^{-q}\le \frac{L^{-p}}{16\max\{1,M\}}.
        \label{eq:peps_fixed_power_gate_budget}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_approximate_gate_real_accuracy}
    Since $C,L\ge1$, we have $1\le\max\{1,M\}\le CL^n$; applying
    Lemma~\ref{thm:peps_approximate_gate_real_accuracy} to $\max\{1,M\}$ and
    dividing by $16\max\{1,M\}>0$ yields~(\ref{eq:peps_fixed_power_gate_budget}).
\end{proof}

\begin{lemma}[Monotonicity in the local approximation tolerance]
    \label{thm:peps_original_approximation_budget}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.IsGateApproximation.mono}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.isGateApproximation_budget_of_power_bound}
    \leanok
    If local gate expansions approximate the original gates within $\delta_1\le\delta_2$,
    they also do so within $\delta_2$ with the same operators, monomial lists,
    contraction bounds, and allowedness. Under the hypotheses of
    Lemma~\ref{thm:peps_fixed_power_gate_budget}, expansions at accuracy
    $\delta_1=L^{-q}$ satisfy the per-gate tolerance
    $\delta_2=L^{-p}/(16\max\{1,M\})$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_fixed_power_gate_budget}
    Each local operator-norm bound $\|G_g-\widetilde G_g\|\le\delta_1$ implies
    $\|G_g-\widetilde G_g\|\le\delta_2$ by transitivity and~(\ref{eq:peps_fixed_power_gate_budget}).
\end{proof}

\begin{lemma}[A positive polynomial coefficient bound]
    \label{thm:peps_original_coefficient_majorant}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_one_le_isExpansionBounded_of_termwise_power_bounds}
    \leanok
    Let $L\geq1$, $C_K,C_C\geq0$, and $u,v\in\mathbb N$. If every nonprivate gate
    of an original circuit has at most $K\leq C_KL^u$ monomials, each with at
    most $r$ pair effects and coefficient modulus $|c_\xi|\le C_CL^v$, then the
    absolute coefficient sum at every gate is bounded by $B\ge 1$ with
    \begin{align}
        B=(1+C_KC_C)L^{u+v}.
        \label{eq:peps_original_coefficient_majorant}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_original_coefficient_sum}
    Since $L\ge 1$, $B\ge 1$, and by Lemma~\ref{thm:peps_original_coefficient_sum}
    each gate's coefficient sum satisfies
    $\sum_\xi|c_\xi|\le K C_CL^v\le C_KC_CL^{u+v}\le B$.
\end{proof}

\begin{lemma}[Finite expansion bounds and replacement coefficients]
    \label{thm:peps_original_finite_expansion_bound}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isMonomialBounded_mono}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_isMonomialBounded}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.sum_norm_branchCoefficient_physicalFirstReplacement_le}
    \leanok
    Every finite original circuit admits $K\in\mathbb N$ (and hence any
    $K'\ge K$) bounding all its gate-expansion lengths. If each monomial has at
    most $r$ pair effects and each gate has coefficient sum $\sum_\xi|c_\xi|\le B$,
    then for any replacement accuracy $\delta>0$, every gate of the physical-first
    source replacement $R$ has branch-coefficient sum at most $B$.
\end{lemma}
\begin{proof}
    \uses{thm:peps_physical_first_occurrences,thm:peps_source_only_resources}
    \leanok
    Finiteness of the gate list yields a common length bound $K$.
    The coefficient sum bound $\le B$ follows from
    Lemma~\ref{thm:peps_physical_first_occurrences} and
    Lemma~\ref{thm:peps_source_only_resources}, as the final exchange of
    physical and auxiliary registers adds no gate and alters no coefficient.
\end{proof}

\begin{lemma}[Physical approximation after replacement and sampling]
    \label{thm:peps_sampled_original_density}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_physicalFirst_sourceSample_original_error_le}
    \leanok
    In the density-approximation steps of
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $W$ be a circuit on a finite party set $P$ with initial unit product
    vector $\psi=\bigotimes_{p\in P}\psi_p$, physical dimensions $d_p\le D$
    ($D\ge 1$), and finite-dimensional discarded private registers. Suppose $W$
    has at most $M$ nonprivate gate occurrences, each party participates in at
    most $\ell$ of them, and each expansion has at most $r$ pair effects per
    monomial and coefficient sum at most $B\ge1$.

    Fix finite orthonormal initial-memory coordinates and $0<\varepsilon\le1$.
    Let $R$ be the physical-first replacement at per-gate tolerance
    $\delta=\varepsilon/(8\max\{1,M\})$, with source occurrence set
    $\mathcal E_R$ of cardinality $N_R=|\mathcal E_R|$. Set
    \begin{align}
        Q & =B^{4\ell}D^4, &
        k & =\left\lceil
        \left(\frac{8\max\{1,N_R\}Q}{\varepsilon}\right)^2
        \right\rceil+1.
        \label{eq:peps_original_sample_count}
    \end{align}
    Then there exist Schmidt data for every source and branch and a Gaussian
    sample array $\omega$ such that the sampled physical operator
    $\widehat\rho_R(\omega)$ and $\rho_W=\operatorname{Tr}_{\mathrm{priv}}(|W\psi\rangle\langle W\psi|)$
    satisfy
    \begin{align}
        \|\widehat\rho_R(\omega)-\rho_W\|_1
        \le\frac{3\varepsilon}{4}.
        \label{eq:peps_original_sample_error}
    \end{align}
    % Source: 04-compression.tex, lines 137--151, 199--229 and 342--381.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_physical_first_exact_sources,thm:peps_actual_source_sampling,
        thm:peps_auxiliary_register_finiteness,thm:peps_original_finite_expansion_bound}
    Preparing each $\psi_p:\mathbb C\to H_p$ locally prepares $\psi$ with no
    pair source. By Lemmas~\ref{thm:peps_auxiliary_register_finiteness}
    and~\ref{thm:peps_original_finite_expansion_bound}, $R$ is an allowed
    contraction circuit with finite-dimensional discarded registers, participation
    bound $\ell$, and coefficient-sum bound $B$.
    Lemma~\ref{thm:peps_actual_source_sampling} provides Schmidt data and a
    sample $\omega$ with $\|\widehat\rho_R(\omega)-\rho_R\|_1\le\varepsilon/4$,
    while Lemma~\ref{thm:peps_physical_first_exact_sources} gives
    $\|\rho_R-\rho_W\|_1\le\varepsilon/2$. The triangle inequality yields~(\ref{eq:peps_original_sample_error}).
\end{proof}


\begin{lemma}[Sampling from approximate original gates]
    \label{thm:peps_approximate_source_sample}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.exists_sourceSample_approximate_error_le}
    \leanok
    Let a finite distributed construction have normalized initial party vectors,
    contraction gates, at most $M$ nonprivate gates, party participation bound
    $\ell$, finite-dimensional private spaces, and physical dimensions $d_p\le D$
    ($D\ge1$).
    Fix $0<\varepsilon\le1$. Suppose each nonprivate gate $G_g$ is approximated
    within $\delta_a=\varepsilon/(16\max(1,M))$ in operator norm by a sum of
    allowed monomials with at most $r$ pair effects and absolute coefficient sum
    at most $B\ge1$. Rescale each expansion by $(1+\delta_a)^{-1}$ to form a
    contraction circuit $W$, and apply the pair-effect replacement of stack
    length $m$ at tolerance $\delta_e$ with
    \begin{align}
        \delta_e & =\frac{\varepsilon}{8\max(1,M)},                              &
        m        & =\left\lceil\left(\frac{rB}{\delta_e}\right)^2\right\rceil+1.
        \label{eq:peps_approximate_effect_params}
    \end{align}
    If $N$ is the source count of the resulting circuit and
    \begin{align}
        Q & =B^{4\ell}D^4,                                                            &
        k & =\left\lceil\left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2\right\rceil+1,
        \label{eq:peps_approximate_sample_params}
    \end{align}
    there exist normalized Schmidt data and a size-$k$ Gaussian sample whose
    sampled physical operator $\widehat\rho$ satisfies
    $\|\widehat\rho-\rho_G\|_1\le\varepsilon$, where $\rho_G$ is the physical
    density of the original gate circuit.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_sampled_original_density,thm:peps_approximate_physical_error,thm:peps_finite_whole_party_input}
    By Lemma~\ref{thm:peps_finite_whole_party_input}, the initial party memories
    admit finite orthonormal bases. Rescaling by $(1+\delta_a)^{-1}\le 1$
    preserves participants, pair-effect counts $\le r$, and coefficient sums
    $\le B$. Lemma~\ref{thm:peps_sampled_original_density} gives
    $\|\widehat\rho-\rho_W\|_1\le3\varepsilon/4$, and
    Lemma~\ref{thm:peps_approximate_physical_error} gives
    $\|\rho_W-\rho_G\|_1\le4M\delta_a\le\varepsilon/4$, so
    \begin{align}
        \|\widehat\rho-\rho_G\|_1
         & \le\|\widehat\rho-\rho_W\|_1+\|\rho_W-\rho_G\|_1
        \le\frac{3\varepsilon}{4}+\frac{\varepsilon}{4}=\varepsilon.\notag
    \end{align}
\end{proof}

\begin{lemma}[Explicit polynomial dimensions of the replacement links]
    \label{thm:peps_physical_first_link_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.card_physicalFirstNetworkAlphabet_le_power}
    \leanok
    Let an original circuit have at most $N$ nonprivate gate occurrences, each
    with at most $b$ participants, at most $K$ monomials, at most $r$ pair
    effects per monomial, and coefficient sum at most $B$. For $L\geq1$,
    $C_K,C_B\geq0$, $C_N\geq1$, and $u,n,s,d,p,\tau\in\mathbb N$, assume
    \begin{align}
        K\leq C_KL^u,\qquad N\leq C_NL^n,\qquad
        0\leq B\leq C_BL^s,\qquad 0\leq D\leq C_DL^d.\notag
    \end{align}
    Let $R$ be the physical-first source replacement at per-gate tolerance
    $\delta=L^{-p}/(8\max(1,N))$ with sample count
    \begin{align}
        k=\left\lceil
        \bigl(8\max(1,\lvert\operatorname{Sources}(R)\rvert)
        B^{4\tau}D^4L^p\bigr)^2
        \right\rceil+1,\notag
    \end{align}
    and define
    \begin{align}
        C_{\mathrm{src}} & =\max\left(1,C_N\binom b2\right),                          \notag                              \\
        C_{\mathrm{br}}  & =C_K\bigl(64C_N^2(rC_B)^2+2\bigr)^r,
                         & a_{\mathrm{br}}                                                   & =u+2r(n+s+p),       \notag \\
        C_{\mathrm{sam}} & =64C_{\mathrm{src}}^2
        (C_B^{4\tau}C_D^4)^2+2,
                         & a_{\mathrm{sam}}                                                  & =2(n+4\tau s+4d+p).
        \label{eq:peps_physical_first_link_constants}
    \end{align}
    Then for every retained occurrence set, root assignment, and star or sample
    link $e$, its alphabet $\mathcal A_e$ satisfies
    \begin{align}
        \lvert\mathcal A_e\rvert
        \leq (C_{\mathrm{br}}^2+C_{\mathrm{sam}})
        L^{\max(2a_{\mathrm{br}},a_{\mathrm{sam}})}.
        \label{eq:peps_physical_first_link_bound}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_physical_first_branch_polynomial,thm:peps_physical_first_sample_polynomial,thm:peps_source_alphabet_polynomial}
    By Lemma~\ref{thm:peps_physical_first_branch_polynomial}, each gate has at
    most $C_{\mathrm{br}}L^{a_{\mathrm{br}}}$ branches. Since
    $|\operatorname{Sources}(R)|\le N\binom{b}{2}$,
    Lemma~\ref{thm:peps_physical_first_sample_polynomial} gives
    $k<C_{\mathrm{sam}}L^{a_{\mathrm{sam}}}$. Because $|\mathcal A_e|$ is the
    squared branch count on star links and $k$ on sample links,
    Lemma~\ref{thm:peps_source_alphabet_polynomial} yields~(\ref{eq:peps_physical_first_link_bound}).
\end{proof}

\begin{lemma}[Trace bounded by the trace norm]
    \label{thm:peps_trace_le_trace_norm}
    \lean{TNLean.Lattice.Compression.norm_trace_le_rectangularTraceNorm}
    \leanok
    Every complex square matrix $X$ satisfies
    $\lvert\operatorname{Tr}X\rvert\le\|X\|_1$.
\end{lemma}
\begin{proof}
    \leanok
    Since $\|I\|\le 1$, trace-norm duality gives
    $\lvert\operatorname{Tr}X\rvert=\lvert\overline{\operatorname{Tr}(X^\dagger I)}\rvert\le\|X\|_1$.
\end{proof}

\begin{lemma}[Adding an absolute error to a pure-state approximation]
    \label{thm:peps_absolute_error_pure_state}
    \lean{TNLean.Lattice.Compression.rectangularTraceNorm_sub_pure_le_of_retained_density}
    \leanok
    Let $\sigma,\rho$ be complex square matrices and $\Omega$ a unit vector.
    If $\|\sigma-\rho\|_1\le\kappa$ and
    $\|\rho-|\Omega\rangle\langle\Omega|\|_1\le\eta$, then
    \begin{align}
        \|\sigma-|\Omega\rangle\langle\Omega|\|_1 & \le\kappa+\eta, &
        \lvert\operatorname{Tr}\rho-1\rvert       & \le\eta.
        \label{eq:peps_absolute_error_pure_state}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_trace_le_trace_norm}
    The first bound in~(\ref{eq:peps_absolute_error_pure_state}) is the triangle
    inequality; the second follows from Lemma~\ref{thm:peps_trace_le_trace_norm}
    applied to $\rho-|\Omega\rangle\langle\Omega|$ since
    $\operatorname{Tr}|\Omega\rangle\langle\Omega|=\|\Omega\|^2=1$.
\end{proof}

\begin{lemma}[Absolute error and retained norm]

\begin{lemma}[Separate arity and lifetime bounds for link incidence]
    \label{thm:peps_separate_arity_incidence}
    \lean{TNLean.Lattice.Compression.card_incidentDistributedLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.SourceCircuit.card_incidentSourceDensityLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.card_incidentPhysicalFirstDensityLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.card_participants_produce_le_of_arity}
    \leanok
    \uses{def:compression_operator_links,def:compression_operator_incidences}
    Let $G$ be a finite set of gate occurrences with finite participant sets
    $P_g$ and roots $r_g\in P_g$. For $a,b\in\mathbb N$ and $p\in P$, if
    $|P_g|\le a$ for all $g\in G$ and $|\{g\in G:p\in P_g\}|\le b$, then the
    set $I_p$ of tagged star and sample links incident to $p$ satisfies
    \begin{align}
        |I_p| & \le2b(a-1),
        \label{eq:peps_separate_arity_incidence}
    \end{align}
    with natural subtraction $a-1$. The same bound holds for the nonempty-participant
    occurrences of a source circuit, and for the physical-first replacement $R$
    at any tolerance $\delta>0$ of an original circuit $W$ with expansion bound
    $(r,S)$, arity bound $a$, and lifetime bound $b$, as in
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:compression_incidence_containment,thm:compression_gate_incidence_bound,
        thm:compression_tagging_cardinality,thm:peps_physical_first_original_locality,
        thm:peps_source_only_resources,thm:peps_physical_first_occurrences,
        thm:peps_original_produced_resource_bounds}
    By Lemmas~\ref{thm:compression_incidence_containment},
    \ref{thm:compression_gate_incidence_bound}, and~\ref{thm:compression_tagging_cardinality},
    each $g\in G_p=\{g\in G:p\in P_g\}$ contributes at most $|P_g|-1$ star links
    and $|P_g|-1$ sample links incident to $p$, so
    \begin{align}
        |I_p| & \le\sum_{g\in G_p}2(|P_g|-1)
        \le |G_p|\,2(a-1)\le2b(a-1).\notag
    \end{align}
    Restricting to nonempty occurrences gives the source-circuit bound. By
    Lemmas~\ref{thm:peps_physical_first_original_locality},
    \ref{thm:peps_source_only_resources}, \ref{thm:peps_physical_first_occurrences},
    and~\ref{thm:peps_original_produced_resource_bounds}, the physical-first
    replacement $R$ and recorded-expansion reading preserve participant sets and
    do not increase party lifetimes, so (\ref{eq:peps_separate_arity_incidence})
    applies to $R$.
\end{proof}

\begin{theorem}[Uniform compression with separate arity and lifetime]
    \label{thm:peps_separate_arity_compression}
    \lean{TNLean.Lattice.PartyCircuit.DistributedConstruction.exists_uniform_sampled_network_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.DistributedConstruction.exists_uniform_sampled_network_sub_pure_le_of_arity_lifetime}
    \leanok
    \uses{def:peps_distributed_construction,def:peps_distributed_construction_density,
        def:compression_operator_operator,def:compression_operator_alphabets}
    In the message-free case of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $(\mathcal C_i)_{i\in I}$ be a family of distributed constructions with
    sizes $L_i\ge2$. Fix $a,b,n,d,r,u,v\in\mathbb N$ with $a\le b$ and
    $C_N,C_D,C_K,C_S\in\mathbb R$. Suppose each $\mathcal C_i$ has at most
    $C_NL_i^n$ nonprivate gates, physical dimensions $d_p\le C_DL_i^d$, gate
    arity at most $a$, party participation at most $b$, and recorded gate
    expansions of length at most $K_i\le C_KL_i^u$ whose monomials have at most
    $r$ pair effects and coefficient sum $\sum_\xi|c_\xi|\le C_SL_i^v$.

    For every $p>0$, there exist $C_p\ge0$ and $\alpha\in\mathbb N_{>0}$
    independent of $i$ such that each $\mathcal C_i$ admits a sampled operator
    tensor network with contraction $\sigma_i$, link alphabets $D_e$, and
    party-incident link sets $I_j$ satisfying
    \begin{align}
        \|\sigma_i-\rho_i\|_1 & \le L_i^{-p},      &
        |D_e|                 & \le C_pL_i^\alpha, & |I_j| & \le2b(a-1),
        \label{eq:peps_separate_arity_network}
    \end{align}
    where $\rho_i$ is the physical density of $\mathcal C_i$ and both endpoints
    of every link $e$ belong to a common nonprivate gate of $\mathcal C_i$.
    If unit vectors $\omega_i$ satisfy
    $\|\rho_i-|\omega_i\rangle\langle\omega_i|\|_1\le\eta_i$, as in
    \cite[Remark~5.3]{OpenAI2026PolynomialPEPS}, then also
    \begin{align}
        \|\sigma_i-|\omega_i\rangle\langle\omega_i|\|_1
                 & \le L_i^{-p}+\eta_i,                             &
        1-\eta_i & \le\operatorname{Re}\operatorname{Tr}\rho_i\le1.
        \label{eq:peps_separate_arity_pure}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_uniform_approximate_operator_network_of_le,
        thm:peps_distributed_construction_gate_maps_density,
        thm:peps_separate_arity_incidence,thm:peps_compression_retained_norm}
    By Definition~\ref{def:peps_distributed_construction_gate_maps} and
    Lemma~\ref{thm:peps_distributed_construction_gate_maps_density}, the
    recorded expansions reproduce each gate operator with zero approximation
    error and preserve $\rho_i$. Since $a\le b$, applying
    Lemma~\ref{thm:peps_uniform_approximate_operator_network_of_le} with
    common bound $b$ yields $\sigma_i$ and $D_e$ satisfying the first two bounds
    in~(\ref{eq:peps_separate_arity_network}) and the common-gate locality of
    links. Because reexpansion and rescaling preserve participant sets and
    lifetimes, Lemma~\ref{thm:peps_separate_arity_incidence} gives
    $|I_j|\le 2b(a-1)$ on the same replacement network. Finally,
    Lemma~\ref{thm:peps_compression_retained_norm} yields~(\ref{eq:peps_separate_arity_pure}).
\end{proof}
