\section{Sampling an original construction}
\label{sec:peps_original_compression_sampling}

We now combine the local approximation, pair-effect replacement and source
sampling estimates for the original gate maps. These are the approximation
steps in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
% Source: 04-compression.tex, lines 137--175, 199--229 and 590--600.
The sampled physical operator need not be positive. Its identification with a
contraction of local tensors is a separate assertion.

\begin{definition}[Distributed construction]
    \label{def:peps_distributed_construction}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction}
    \leanok
    \uses{def:peps_original_circuit}
    A distributed construction has a finite nonempty set $P$ of parties.
    For each $p\in P$, it specifies a finite-dimensional initial private
    memory $H_p$ and a unit vector $x_p\in H_p$. It also specifies a physical
    dimension $d_p$, an ordering of the parties without repetition, a finite
    ordered collection $R$ of finite-dimensional private output registers,
    and a contraction circuit
    \begin{align}
        W:\bigotimes_{p\in P}H_p
        \longrightarrow
        \left(\bigotimes_{p\in P}\mathbb C^{d_p}\right)\otimes\mathcal H_R.
    \end{align}
    The circuit retains the original gate expansions and participant labels.
    There is no numerical bound on private dimensions in this definition.
    Resource estimates and approximation conclusions are separate statements.

    The nonempty-party convention permits scalar factors to be absorbed in
    an existing local tensor, as in the proof of Theorem~5.2. It does not
    exclude gates with no participating parties. The zero-party scalar
    boundary case is recorded in \cite{gap:polypeps_nonempty_parties}.
\end{definition}

\begin{definition}[Original input and physical density]
    \label{def:peps_distributed_construction_density}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction.input,
        TNLean.PEPS.PairEffect.DistributedConstruction.physicalDensity}
    \leanok
    \uses{def:peps_distributed_construction}
    The input of a distributed construction is the normalized product
    $\psi=\bigotimes_{p\in P}x_p$. Its physical density is
    \begin{align}
        \rho=\operatorname{Tr}_{\mathcal H_R}
        \bigl(|W\psi\rangle\langle W\psi|\bigr),
    \end{align}
    expressed in the physical coordinates indexed by the original parties.
    The private trace uses a finite orthonormal basis of the retained
    registers. No normalization by $\|W\psi\|$ is performed.
\end{definition}

\begin{theorem}[Subnormalized physical density]
    \label{thm:peps_distributed_construction_density}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction.physicalDensity_posSemidef,
        TNLean.PEPS.PairEffect.DistributedConstruction.trace_physicalDensity_le_one}
    \leanok
    \uses{def:peps_distributed_construction_density}
    The physical density of a distributed construction is positive
    semidefinite and satisfies $\operatorname{Tr}\rho\leq1$.
\end{theorem}
\begin{proof}
    \leanok
    The outer product of the original output vector is positive semidefinite,
    and partial trace preserves positivity. Partial trace also preserves the
    full trace, so $\operatorname{Tr}\rho=\|W\psi\|^2$. Each gate is a
    contraction and $\|\psi\|=1$, giving the asserted bound.
\end{proof}

\begin{theorem}[A fixed inverse power meets the per-gate error tolerance]
    \label{thm:peps_fixed_power_gate_budget}
    \lean{TNLean.PEPS.PairEffect.inversePower_le_sourceGateBudget_half}
    \leanok
    Let $L\ge2$, $C\ge1$, $M\le CL^n$, where $M,n$ are nonnegative
    integers, and let $p$ be real. Set
    $q=n+\lceil p\rceil_++\lceil16C\rceil_+$. Then
    \begin{align}
        L^{-q}\le \frac{L^{-p}}{16\max\{1,M\}}.
    \end{align}
    The exponent is fixed by the original count bound and the target accuracy,
    and works simultaneously at every size $L\ge2$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_approximate_gate_real_accuracy}
    The bound $\max\{1,M\}\le CL^n$ follows from $C,L\ge1$, and
    applying Theorem~\ref{thm:peps_approximate_gate_real_accuracy}
    with this enlarged count
    and dividing by its positive denominator gives the bound.
\end{proof}

\begin{theorem}[Monotonicity in the local approximation tolerance]
    \label{thm:peps_original_approximation_budget}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.IsGateApproximation.mono}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.isGateApproximation_budget_of_power_bound}
    \leanok
    Increasing the permitted local approximation error preserves the original
    operators, approximate monomial lists and all contraction and allowedness
    conditions. Consequently, under the preceding count assumptions, gate
    expansions available at accuracy $L^{-q}$ satisfy the local error tolerance
    $L^{-p}/(16\max\{1,M\})$ required to reserve one quarter of the final
    physical error for this approximation step.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_fixed_power_gate_budget}
    Only the inequality bounding each local operator difference changes;
    applying Theorem~\ref{thm:peps_fixed_power_gate_budget} and transitivity
    at every original occurrence gives the tolerance, while all the other local hypotheses are unchanged.
\end{proof}

\begin{theorem}[A positive polynomial coefficient bound]
    \label{thm:peps_original_coefficient_majorant}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_one_le_isExpansionBounded_of_termwise_power_bounds}
    \leanok
    Let $L\geq1$. Suppose every nonprivate gate of the original circuit
    has at most $K\leq C_KL^u$ monomials, each with at most $r$ pair
    effects and coefficient of absolute value at most $C_CL^v$, where
    $C_K,C_C\geq0$ and $u,v$ are nonnegative integers. Then there is
    a bound $B\geq1$ for the absolute coefficient sum of every gate,
    with the same effect bound $r$, such that
    \begin{align}
        B\leq (1+C_KC_C)L^{u+v}.
    \end{align}
    Thus the lower bound $B\geq1$ required in the source-sampling
    estimate follows from the original polynomial assumptions.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_original_coefficient_sum}
    Choosing $B=(1+C_KC_C)L^{u+v}$ gives $B\geq1$ since $L\geq1$, and
    each coefficient sum is at most
    $K C_CL^v\leq C_KC_CL^{u+v}\leq B$.
\end{proof}

\begin{theorem}[Finite expansion bounds and replacement coefficients]
    \label{thm:peps_original_finite_expansion_bound}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isMonomialBounded_mono}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_isMonomialBounded}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.sum_norm_branchCoefficient_physicalFirstReplacement_le}
    \leanok
    Every finite original circuit has an integer $K$ bounding the lengths
    of all its gate expansions. Any larger integer is also such a bound.
    Suppose its monomials contain at most $r$ pair effects and the absolute
    coefficient sum at every gate is at most $B$. For any positive
    replacement accuracy, every gate of the source replacement,
    with the physical output registers placed first, has absolute branch
    coefficient sum at most $B$.
\end{theorem}
\begin{proof}
    \uses{thm:peps_physical_first_occurrences,thm:peps_source_only_resources}
    \leanok
    Taking a common bound on the finitely many original gate lists, the
    coefficient estimate of Theorem~\ref{thm:peps_physical_first_occurrences}
    gives the bound, and the final exchange of physical and auxiliary registers
    introduces no gate and changes no coefficient.
\end{proof}

\begin{theorem}[Physical approximation after replacement and sampling]
    \label{thm:peps_sampled_original_density}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.exists_physicalFirst_sourceSample_original_error_le}
    \leanok
    In the density-approximation steps of
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
    let $W$ be a circuit on a finite party set $P$, with initial vector
    $\psi=\bigotimes_{p\in P}\psi_p$, where each $\psi_p$ is normalized
    in the whole initial private memory of party $p$. The final physical
    register at $p$ has dimension $d_p$, and every discarded private
    register is finite-dimensional. Suppose that the original nonprivate
    gate count is at most $M$, each party participates in at most $\ell$
    nonprivate occurrences, and the original expansion bounds hold with
    pair-effect bound $r$ and coefficient-sum bound $B\ge1$. Let $D\ge1$
    satisfy $d_p\le D$ for every $p$.

    Fix finite orthonormal coordinates on the whole initial memory.
    Fix $0<\varepsilon\le1$. Set
    $\delta=\varepsilon/(8\max\{1,M\})$, and form the physical-first
    replacement $R$ using the corresponding prepared stack length.
    Let $\mathcal E_R$ be its source occurrences and
    $N_R=|\mathcal E_R|$. Choose
    \begin{align}
        Q & =B^{4\ell}D^4, &
        k & =\left\lceil
        \left(\frac{8\max\{1,N_R\}Q}{\varepsilon}\right)^2
        \right\rceil+1.
        \label{eq:peps_original_sample_count}
    \end{align}
    There exist Schmidt data for every source and branch, and
    one Gaussian sample array $\omega$, such that the sampled physical
    operator $\widehat\rho_R(\omega)$ satisfies
    \begin{align}
        \|\widehat\rho_R(\omega)-\rho_W\|_1
        \le\frac{3\varepsilon}{4}.
        \label{eq:peps_original_sample_error}
    \end{align}
    Here $\rho_W$ is the physical partial trace of $|W\psi\rangle
        \langle W\psi|$, in the original party-labelled coordinates, and
    no numerical bound on private-memory dimensions is assumed.
    % Source: 04-compression.tex, lines 137--151, 199--229 and 342--381.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_physical_first_exact_sources,thm:peps_actual_source_sampling,
        thm:peps_auxiliary_register_finiteness,thm:peps_original_finite_expansion_bound}
    Preparing each $\psi_p$ by a local contraction from the scalar space,
    their chronological composition prepares $\psi$ and introduces no
    pair source. The replacement is a contraction, preserves the original
    participating occurrences, and retains coefficient-sum bound $B$.
    A finite bound on its monomial counts follows from the finite original
    gate lists; it is not an additional hypothesis. Every auxiliary
    register is finite-dimensional by its Euclidean construction, so the
    individual private-register assumption suffices for all discarded
    registers of $R$.

    Applying Theorem~\ref{thm:peps_actual_source_sampling} to this
    circuit, preparation, and discarded memory gives original-source
    Schmidt data and one sample $\omega$ with
    $\|\widehat\rho_R(\omega)-\rho_R\|_1\le\varepsilon/4$.
    Theorem~\ref{thm:peps_physical_first_exact_sources} gives
    $\|\rho_R-\rho_W\|_1\le\varepsilon/2$, and the trace-norm triangle
    inequality yields~(\ref{eq:peps_original_sample_error}).
\end{proof}


\begin{theorem}[Sampling from approximate original gates]
    \label{thm:peps_approximate_source_sample}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.exists_sourceSample_approximate_error_le}
    \leanok
    Let a finite distributed construction begin in a product of normalized
    party vectors. Suppose its original gates are contractions, its number of
    nonprivate gates is at most $M$, and each party participates in at most $\ell$
    such gates. All private spaces are finite-dimensional, without numerical
    dimension bounds. Let the physical dimension at each party be at most
    $D\ge1$.

    Fix $0<\varepsilon\le1$. Suppose the monomial list for each original
    nonprivate gate approximates that gate in operator norm within
    $\delta_a=\varepsilon/(16\max(1,M))$. Its monomials are allowed, each has at
    most $r$ pair effects, and its absolute coefficient sum is at most $B\ge1$.
    Rescale each such list by $(1+\delta_a)^{-1}$ to obtain a contraction
    circuit $W$. Set
    \begin{align}
        \delta_e & =\frac{\varepsilon}{8\max(1,M)},                              &
        m        & =\left\lceil\left(\frac{rB}{\delta_e}\right)^2\right\rceil+1.
    \end{align}
    Apply the constructed pair-effect replacement of length $m$ to $W$, and
    let $N$ be the number of sources of the resulting circuit. With
    \begin{align}
        Q & =B^{4\ell}D^4,                                                            &
        k & =\left\lceil\left(\frac{8\max(1,N)Q}{\varepsilon}\right)^2\right\rceil+1,
    \end{align}
    there are normalized Schmidt data for these sources and a Gaussian
    sample of size $k$ whose sampled physical operator $\widehat\rho$ satisfies
    $\|\widehat\rho-\rho_G\|_1\le\varepsilon$. Here $\rho_G$ is the physical
    density computed from the original gate operators, with all private
    registers traced out.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sampled_original_density,thm:peps_approximate_physical_error,thm:peps_finite_whole_party_input}
    The input coordinates are obtained from finite-dimensionality of the
    original party memories. Rescaling preserves the original gate
    occurrences and participating sets, and does not increase the coefficient
    sums or pair-effect counts. Applying the exact sampled-density bound to $W$
    therefore gives $\|\widehat\rho-\rho_W\|_1\le3\varepsilon/4$ with the
    stated replacement and sample counts. The local approximation estimate
    gives $\|\rho_W-\rho_G\|_1\le4M\delta_a\le\varepsilon/4$. Thus
    \begin{align}
        \|\widehat\rho-\rho_G\|_1
         & \le\|\widehat\rho-\rho_W\|_1+\|\rho_W-\rho_G\|_1
        \le\varepsilon.
    \end{align}
\end{proof}

\begin{theorem}[Explicit polynomial dimensions of the replacement links]
    \label{thm:peps_physical_first_link_polynomial}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.card_physicalFirstNetworkAlphabet_le_power}
    \leanok
    Let an original circuit have at most $N$ nonprivate gate occurrences,
    each with at most $b$ participants. Suppose each gate has at most $K$
    monomials, at most $r$ pair effects in each monomial, and absolute
    coefficient sum at most $B$. Let $L\geq1$, and assume
    \begin{align}
        K\leq C_KL^u,\qquad N\leq C_NL^n,\qquad
        0\leq B\leq C_BL^s,\qquad 0\leq D\leq C_DL^d,
    \end{align}
    where $C_K,C_B\geq0$, $C_N\geq1$, and $u,n,s,d,p$ are nonnegative
    integers. Use the per-gate error tolerance
    $\delta=L^{-p}/(8\max(1,N))$, and let $R$ be the resulting source
    replacement, with the physical output registers ordered first.
    For any nonnegative integer $\tau$, choose the sample count
    \begin{align}
        k=\left\lceil
        \bigl(8\max(1,\lvert\operatorname{Sources}(R)\rvert)
        B^{4\tau}D^4L^p\bigr)^2
        \right\rceil+1.
    \end{align}
    Define constants and exponents by
    \begin{align}
        C_{\mathrm{src}} & =\max\left(1,C_N\binom b2\right),                          \\
        C_{\mathrm{br}}  & =C_K\bigl(64C_N^2(rC_B)^2+2\bigr)^r,
                         & a_{\mathrm{br}}                      & =u+2r(n+s+p),       \\
        C_{\mathrm{sam}} & =64C_{\mathrm{src}}^2
        (C_B^{4\tau}C_D^4)^2+2,
                         & a_{\mathrm{sam}}                     & =2(n+4\tau s+4d+p).
    \end{align}
    For every retained set of gate occurrences, every root assignment,
    and every corresponding star or sample link $e$, its link alphabet
    $\mathcal A_e$ satisfies
    \begin{align}
        \lvert\mathcal A_e\rvert
        \leq (C_{\mathrm{br}}^2+C_{\mathrm{sam}})
        L^{\max(2a_{\mathrm{br}},a_{\mathrm{sam}})}.
    \end{align}
    These constants depend only on the displayed uniform resource bounds.
    No private-memory dimension or source Schmidt rank occurs. This is a
    dimension estimate for the constructed alphabets; the error estimate
    and equality with a contraction of local tensors are separate assertions.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_physical_first_branch_polynomial,thm:peps_physical_first_sample_polynomial,thm:peps_source_alphabet_polynomial}
    Theorem~\ref{thm:peps_physical_first_branch_polynomial} gives at most
    $C_{\mathrm{br}}L^{a_{\mathrm{br}}}$ branches per gate.
    There are at most $N\binom b2$ source occurrences, so the
    estimate of Theorem~\ref{thm:peps_physical_first_sample_polynomial}
    gives $k<C_{\mathrm{sam}}L^{a_{\mathrm{sam}}}$.
    A star alphabet has cardinality equal to the square of its gate's
    branch count, and a sample alphabet has cardinality $k$, so applying the
    bound of Theorem~\ref{thm:peps_source_alphabet_polynomial}
    to these two estimates gives the result; the choice
    $C_{\mathrm{src}}\geq1$ includes circuits without sources.
\end{proof}

\begin{theorem}[Trace bounded by the trace norm]
    \label{thm:peps_trace_le_trace_norm}
    \lean{TNLean.PEPS.Approximation.norm_trace_le_rectangularTraceNorm}
    \leanok
    Every complex square matrix $X$ satisfies
    $\lvert\operatorname{Tr}X\rvert\le\|X\|_1$.
\end{theorem}
\begin{proof}
    \leanok
    The identity matrix has operator norm at most one, so trace-norm duality
    gives $\lvert\operatorname{Tr}(X^\dagger\cdot 1)\rvert\le\|X\|_1$, and
    $\operatorname{Tr}X^\dagger$ is the complex conjugate of
    $\operatorname{Tr}X$.
\end{proof}

\begin{theorem}[Adding an absolute error to a pure-state approximation]
    \label{thm:peps_absolute_error_pure_state}
    \lean{TNLean.PEPS.Approximation.rectangularTraceNorm_sub_pure_le_of_retained_density}
    \leanok
    Let $\sigma,\rho$ be complex square matrices and $\Omega$ a unit vector.
    If $\|\sigma-\rho\|_1\le\kappa$ and
    $\|\rho-|\Omega\rangle\langle\Omega|\|_1\le\eta$, then
    \begin{align}
        \|\sigma-|\Omega\rangle\langle\Omega|\|_1 & \le\kappa+\eta, \\
        \lvert\operatorname{Tr}\rho-1\rvert       & \le\eta.
    \end{align}
    Neither matrix is assumed positive, Hermitian or normalized.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_trace_le_trace_norm}
    The first bound is the triangle inequality for the trace norm, and the
    second follows by applying Theorem~\ref{thm:peps_trace_le_trace_norm} to
    $\rho-|\Omega\rangle\langle\Omega|$ and using
    $\operatorname{Tr}|\Omega\rangle\langle\Omega|=1$.
\end{proof}

\begin{theorem}[Absolute error and retained norm]
    \label{thm:peps_compression_retained_norm}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction.rectangularTraceNorm_compression_sub_pure_le}
    \leanok
    \uses{def:peps_distributed_construction_density}
    % Source: 04-compression.tex, lines 602--611.
    As noted in \cite[Remark~5.3]{OpenAI2026PolynomialPEPS},
    the compression of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}
    approximates the physical density $\rho$ of a distributed
    construction, which need not have trace one, and does not normalize it.
    Let $\sigma$ be any operator with $\|\sigma-\rho\|_1\le\kappa$; it need be
    neither positive nor Hermitian. If $\Omega$ is a unit vector with
    $\|\rho-|\Omega\rangle\langle\Omega|\|_1\le\eta$, then
    \begin{align}
        \|\sigma-|\Omega\rangle\langle\Omega|\|_1       & \le\kappa+\eta, \\
        1-\eta\le\operatorname{Re}\operatorname{Tr}\rho & \le1.
    \end{align}
    No bound on private dimensions or on Schmidt ranks of the pair
    vectors enters, and the compression theorem is used only through its error
    estimate $\|\sigma-\rho\|_1\le\kappa$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_absolute_error_pure_state,thm:peps_distributed_construction_density}
    Theorem~\ref{thm:peps_absolute_error_pure_state} gives the trace-norm bound
    and $\lvert\operatorname{Tr}\rho-1\rvert\le\eta$, hence the lower bound on
    the real part. The upper bound $\operatorname{Tr}\rho\le1$ is
    Theorem~\ref{thm:peps_distributed_construction_density}.
\end{proof}

\begin{theorem}[Separate arity and lifetime bounds for link incidence]
    \label{thm:peps_separate_arity_incidence}
    \lean{TNLean.PEPS.Approximation.card_incidentDistributedLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.SourceCircuit.card_incidentSourceDensityLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.card_incidentPhysicalFirstDensityLinks_le_of_arity_lifetime,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.card_participants_produce_le_of_arity}
    \leanok
    \uses{def:compression_operator_links,def:compression_operator_incidences}
    Let $G$ be a finite set of gate occurrences, each with a finite
    participant set $P_g$ and a root $r_g\in P_g$.
    Let $a,b\in\mathbb N$ and fix a party $p$.
    If $|P_g|\le a$ for every $g\in G$ and $p$ belongs to at most
    $b$ of these sets, then its actual tagged star and sample links satisfy
    \begin{align}
        |I_p| & \le2b(a-1).
        \label{eq:peps_separate_arity_incidence}
    \end{align}
    Here $a-1$ denotes natural subtraction. The bound includes
    $a=0,1$ and $b=0$, and counts parallel link tags separately.

    For a source circuit, the same bound holds for the link family
    of its nonempty participant occurrences, if each such occurrence
    has at most $a$ participants and every party participates at most $b$ times.
    Reading an original circuit as its recorded expansions preserves
    the participant set of each occurrence, and hence its arity bound.
    More precisely, let $W$ have expansion bound $(r,S)$, arity at
    most $a$ and lifetime at most $b$ at every party. For each
    $\delta>0$, let $R$ be its constructed physical-output-first
    replacement at that precision. Then the link family of this same
    $R$ has $|I_p(R)|\le2b(a-1)$ at every party.
    These are incidence estimates for the specified tagged links in
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:compression_incidence_containment,thm:compression_gate_incidence_bound,
        thm:compression_tagging_cardinality,thm:peps_physical_first_original_locality,
        thm:peps_source_only_resources,thm:peps_physical_first_occurrences,
        thm:peps_original_produced_resource_bounds}
    At a gate containing $p$, at most $|P_g|-1$ star links and
    $|P_g|-1$ sample links meet $p$. The union over its actual
    incident gate occurrences therefore gives
    \begin{align}
        |I_p| & \le\sum_{g\in G_p}2(|P_g|-1)
        \le |G_p|\,2(a-1)\le2b(a-1).\notag
    \end{align}
    Each link retains its occurrence and kind, so the union does not
    identify parallel links. Empty participant occurrences contribute
    to no party's count. The source-circuit count follows by restricting
    to the nonempty occurrences. For the physical-output-first
    replacement, the occurrence correspondence preserves the exact
    participant set, and its lifetime does not exceed the original
    lifetime. Apply (\ref{eq:peps_separate_arity_incidence}) to that
    replacement. The recorded expansion correspondence preserves
    participants without any approximation or precision assumption.
\end{proof}

\begin{theorem}[Uniform compression with separate arity and lifetime]
    \label{thm:peps_separate_arity_compression}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction.exists_uniform_sampled_network_of_arity_lifetime,
        TNLean.PEPS.PairEffect.DistributedConstruction.exists_uniform_sampled_network_sub_pure_le_of_arity_lifetime}
    \leanok
    \uses{def:peps_distributed_construction,def:peps_distributed_construction_density,
        def:compression_operator_operator,def:compression_operator_alphabets}
    Consider the case of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}
    without messages. Let $(\mathcal C_i)_{i\in I}$ be an arbitrary
    family of distributed constructions with real sizes $L_i\ge2$.
    Fix natural $a,b,n,d,r,u,v$ with $a\le b$ and real
    $C_N,C_D,C_K,C_S$, before all members. Suppose every member has
    at most $C_NL_i^n$ original nonprivate gates and physical
    dimensions at most $C_DL_i^d$. Each such gate has at most $a$
    participants, each party participates in at most $b$ gates, and
    there is a natural monomial bound $K_i\le C_KL_i^u$.
    Every recorded monomial has at most $r$ pair effects, and at every
    original nonprivate gate the moduli of the recorded coefficients sum
    to at most $C_SL_i^v$.

    For every $p>0$, there are $C_p\ge0$ and a natural $\alpha>0$
    before all members, such that each member admits actual finite
    gate occurrences, roots, branch alphabets, sample count,
    coefficients and party matrices $T$, whose contraction $\sigma_i$
    satisfies
    \begin{align}
        \|\sigma_i-\rho_i\|_1 & \le L_i^{-p},      &
        |D_e|                 & \le C_pL_i^\alpha, & |I_j| & \le2b(a-1).
        \label{eq:peps_separate_arity_network}
    \end{align}
    Here $\rho_i$ is the original unnormalized physical density.
    Both endpoints of every actual tagged link participate in a
    common original nonprivate gate of $\mathcal C_i$.
    All these conclusions concern the same contraction and alphabets.

    If additionally unit vectors $\omega_i$ and real errors $\eta_i$
    satisfy $\|\rho_i-|\omega_i\rangle\langle\omega_i|\|_1\le\eta_i$,
    the same conclusion can be chosen with
    \begin{align}
        \|\sigma_i-|\omega_i\rangle\langle\omega_i|\|_1
                 & \le L_i^{-p}+\eta_i,                             &
        1-\eta_i & \le\operatorname{Re}\operatorname{Tr}\rho_i\le1.
        \label{eq:peps_separate_arity_pure}
    \end{align}
    This is the near-pure-state consequence of
    \cite[Remark~5.3]{OpenAI2026PolynomialPEPS}. No positivity or
    normalization of $\sigma_i$ is asserted.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_uniform_approximate_operator_network_of_le,
        thm:peps_distributed_construction_gate_maps_density,
        thm:peps_separate_arity_incidence,thm:peps_compression_retained_norm}
    Use each gate's own recorded expansion. It gives its original
    operator exactly, with the stated monomial and coefficient bounds.
    Since $a\le b$, apply the original-gate sampling theorem with
    common sampling bound $b$ at its lower construction level.
    Keep its selected expansion lists, rescaled circuit, actual
    replacement $R$, sample count, coefficients and local matrices
    $T$. Its error and alphabet estimates already concern these data.
    Reexpansion and rescaling preserve each original participant set
    and lifetime. Theorem~\ref{thm:peps_separate_arity_incidence}
    therefore gives $2b(a-1)$ for this very replacement's links,
    without selecting another network. Transporting its common-gate
    witness through the recorded occurrence correspondence gives an
    original gate of $\mathcal C_i$ with the same participants.
    The absorbed exterior scalar changes one local matrix but no link
    or alphabet. This proves (\ref{eq:peps_separate_arity_network}).
    Apply the retained-density trace-norm estimate to this same
    $\sigma_i$ and $\rho_i$ to obtain
    (\ref{eq:peps_separate_arity_pure}).
\end{proof}

\begin{theorem}[Compression of a family of distributed constructions]
    \label{thm:peps_distributed_construction_compression}
    \lean{TNLean.PEPS.PairEffect.DistributedConstruction.exists_uniform_sampled_network}
    \leanok
    In the case of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} without messages
    (the recorded expansions of a distributed construction consist of allowed
    monomials and carry no messages; the case with messages is not covered by this statement),
    let $(\mathcal C_i)_{i\in I}$ be a family of distributed constructions,
    each index carrying a size $L=L_i\ge2$. Suppose that there are constants
    $b,n,d,r,u,v$ and $C_N,C_D,C_K,C_C$, independent of the index, such that
    every member has at most $C_NL^n$ nonprivate gates and physical
    dimensions at most $C_DL^d$; every nonprivate gate involves at most $b$
    parties and every party participates in at most $b$ nonprivate gates over
    the whole construction; and the recorded expansion of every nonprivate
    gate has at most $C_KL^u$ monomials, each with at most $r$ pair effects and
    a coefficient of modulus at most $C_CL^v$. Then for every $p>0$ there are
    $C_p\ge0$ and an integer $\alpha_p>0$, chosen before the family member,
    such that every member admits an operator tensor network $\sigma_L$ with
    one node per party, the physical ket and bra legs of that party, and
    \begin{align}
        \|\sigma_L-\rho_L\|_1\le L^{-p},\qquad \dim\mathcal V_e\le C_pL^{\alpha_p}
    \end{align}
    for the physical density $\rho_L$. Every node meets at most
    $2b(b-1)$ virtual links, and the two endpoints of every link participate
    in a common nonprivate gate of the original construction.
    % Source: 04-compression.tex, lines 137--175.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_distributed_construction_compression_of_le,def:peps_compression_constants}
    Take $C_p=C(r,b,C_N,C_D,C_K,C_C)$ and
    $\alpha_p=\alpha(r,b,n,d,u,v,p)$, the positive constants of
    Definition~\ref{def:peps_compression_constants}, in
    Theorem~\ref{thm:peps_distributed_construction_compression_of_le}.
    Reading every member with the operators of its own expansions, the recorded
    expansions approximate these operators with error zero, hence at every
    inverse-power accuracy, and keep the original monomial counts,
    coefficients, gate occurrences and participating sets. Applying
    Theorem~\ref{thm:peps_uniform_approximate_operator_network} and using
    Theorem~\ref{thm:peps_distributed_construction_gate_maps_density}
    identifies the approximated density with $\rho_L$, while each gate occurrence of
    the reading corresponds to a nonprivate gate of the original
    construction with the same participants.
\end{proof}
