\section{Removal squares and repair children}
\label{sec:area_law_simultaneous_repair}

In the recursive repair of \cite[Section~11]{OpenAI2026AreaLaw}, each
removal step deletes a union of closed squares from all existing regions
before any replacement regions are adjoined.

\begin{definition}[Dyadic removal square]
    \label{def:area_law_repair_square}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.repairSquare}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    Retain the fixed exponent $\rho_0=(1+\beta)/2$. For a parent scale
    $S>0$ and a center $v\in\mathbb R^2$, define the dyadic half-side and
    removal square by
    \begin{align}
        d(S)   & =2^{\lfloor\rho_0\log_2 S\rfloor}, &
        Q(v,S) & =v+[-d(S),d(S)]^2.
        \label{eq:area_law_repair_square}
    \end{align}
\end{definition}

% Source: 10-geometry.tex, geometry:trim-margins, lines 386--391.
\begin{theorem}[Removal margins]
    \label{thm:area_law_repair_margins}
    \lean{TNLean.PEPS.AreaLaw.Geometry.rho_pos_lt_one,
        TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_le_rpow,
        TNLean.PEPS.AreaLaw.Geometry.half_rpow_le_repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_lt_self,
        TNLean.PEPS.AreaLaw.Geometry.eventually_repairHalfSide_le,
        TNLean.PEPS.AreaLaw.Geometry.eventually_repairHalfSide_margins}
    \leanok
    \uses{def:area_law_repair_square}
    The exponent satisfies $0<\rho_0<1$. For every $S>0$,
    \begin{equation}
        \tfrac12S^{\rho_0}\le d(S)\le S^{\rho_0},
        \label{eq:area_law_repair_margins}
    \end{equation}
    and $d(S)<S$ whenever $S>1$. For every $a_0>0$ there exists $S_0$ such
    that $d(S)\le a_0S/1000$ for all $S\ge S_0$. Hence a stopping
    scale $S_*$ satisfying the margin conditions of
    \cite[Section~11]{OpenAI2026AreaLaw} exists and may be chosen
    arbitrarily large.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    Put $x=\rho_0\log_2 S$, so that $S^{\rho_0}=2^x$ and
    $d(S)=2^{\lfloor x\rfloor}$. The inequalities
    $\lfloor x\rfloor\le x<\lfloor x\rfloor+1$ give
    \eqref{eq:area_law_repair_margins}. Since $\rho_0<1$, $S^{\rho_0}<S$
    for $S>1$, and $S^{\rho_0}/S=S^{\rho_0-1}\to0$ as $S\to\infty$.
\end{proof}

\begin{definition}[Simultaneous old-region remnants]
    \label{def:area_law_simultaneous_remnants}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairRemovalSet,
        TNLean.PEPS.AreaLaw.Geometry.simultaneousRepairRemnants}
    \leanok
    \uses{def:area_law_repair_square}
    Let $(v_j,S_j)_{j\in J}$ be a finite family of marks with centers
    $v_j\in\mathbb R^2$ and scales $S_j>0$, and let $(R_i)_{i\in I}$ be
    a family of regions in $\mathbb R^2$. Define the removal set and the trimmed
    remnants by
    \begin{align}
        Q_J  & =\bigcup_{j\in J}Q(v_j,S_j), &
        R'_i & =R_i\setminus Q_J.
        \label{eq:area_law_simultaneous_remnants}
    \end{align}
\end{definition}

\begin{definition}[Children of nonterminal parents]
    \label{def:area_law_repair_children}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairChildOffset,
        TNLean.PEPS.AreaLaw.Geometry.RepairChildIndex,
        TNLean.PEPS.AreaLaw.Geometry.repairChildLocation,
        TNLean.PEPS.AreaLaw.Geometry.repairChildScale}
    \leanok
    \uses{def:area_law_repair_square}
    Let $(v_i,S_i)_{i\in I}$ be a family of parent marks with
    $v_i\in\mathbb R^2$ and $S_i>0$, and fix a stopping threshold $S_*$.
    Set $I_+=\{i\in I:d(S_i)\ge S_*\}$ and $P=\{-1,0,1\}^2$. The child
    marks are indexed by $I_+\times P$, with centers and scales
    \begin{align}
        v'_{i,p} & =v_i+d(S_i)p, & S'_{i,p} & =d(S_i).
        \label{eq:area_law_repair_children}
    \end{align}
    As $p$ ranges over $P$, the points $v'_{i,p}$ are the center, corners,
    and side midpoints of $Q(v_i,S_i)$. A parent with $d(S_i)<S_*$ is
    terminal and produces no children. This is the child-marking rule of
    \cite[Section~11]{OpenAI2026AreaLaw}.
\end{definition}

\subsection{Depth and descendant count of a repair branch}
\label{sec:area_law_repair_depth}

Each child of a parent of scale $S$ has scale $d(S)$. Thus the generation-$j$
descendants of an initial mark of scale $S$ all have scale $d^{j}(S)$, and
generation $j$ contains at most $9^j$ marks.

% Source: OpenAI2026AreaLaw, Section 11, 10-geometry.tex, lines 668--686;
% OpenAI2026PolynomialPEPS, Lemma 3.2, 02-information.tex, lines 302--311.
\begin{theorem}[Depth of a power-contracting recursion]
    \label{thm:area_law_power_recursion_depth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.log_iterate_le_of_le_rpow,
        TNLean.PEPS.AreaLaw.Geometry.one_le_log_add_exp_one,
        TNLean.PEPS.AreaLaw.Geometry.exists_iterate_depth_bound,
        TNLean.PEPS.AreaLaw.Geometry.exists_iterate_generation_count_bound,
        TNLean.PEPS.AreaLaw.Geometry.mul_rpow_le_max_mul_rpow,
        TNLean.PEPS.AreaLaw.Geometry.forall_le_iterate_of_le_iterate}
    \leanok
    Let $0<\rho<1$ and $S_*>1$, and let $f$ satisfy $0<f(u)\le u^{\rho}$
    for every $u\ge S_*$. If a scale $S>0$ satisfies $f^{i}(S)\ge S_*$ for
    $0\le i<m$, then $\log f^{i}(S)\le\rho^{i}\log S$ for $0\le i\le m$,
    and there exists a constant $C>0$, independent of $S$ and $m$, such that
    \begin{align}
        m & \le C+C\log\log(S+e).
        \notag
    \end{align}
    For every $b\ge0$ there also exists $C>0$ such that
    $\sum_{j=0}^{m}b^{j}\le C\bigl(\log(S+e)\bigr)^{C}$ under the same
    hypothesis. If $0<f(u)\le u^{\rho}$ for all $u>0$ and $S_*\ge1$, then
    $f^{m}(S)\ge S_*$ implies $f^{i}(S)\ge S_*$ for all $0\le i\le m$.
\end{theorem}

\begin{proof}
    \leanok
    Induction on $i$ gives $\log f^{i}(S)\le\rho^{i}\log S$. If $m\ge1$,
    then $0<\log S_*\le\rho^{m-1}\log S$, so $\log S>0$ and
    $(m-1)\log(1/\rho)\le\log\log S-\log\log S_*\le\log\log(S+e)-\log\log S_*$.
    For the geometric sum, set $B=b+2$; then
    $\sum_{j=0}^{m}b^{j}\le B^{m+1}\le B^{C+1}\bigl(\log(S+e)\bigr)^{C\log B}$.
    For the final claim, if $x<S_*$ then $f(x)\le x^{\rho}<S_*^{\rho}\le S_*$,
    so once $f^{i}(S)<S_*$ all subsequent iterates remain below $S_*$.
\end{proof}

\begin{theorem}[Depth of a repair branch]
    \label{thm:area_law_repair_depth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_pos,
        TNLean.PEPS.AreaLaw.Geometry.log_repairHalfSide_le,
        TNLean.PEPS.AreaLaw.Geometry.log_repairHalfSide_iterate_le,
        TNLean.PEPS.AreaLaw.Geometry.exists_repair_depth_bound}
    \leanok
    \uses{def:area_law_repair_square}
    For every $S>0$ and $j\ge0$,
    \begin{align}
        \log d^{j}(S) & \le\rho_0^{j}\log S.
        \label{eq:area_law_repair_depth_log}
    \end{align}
    For every $S_*>1$ there exists a constant $C>0$ such that every $S>0$
    and $j\ge0$ with $d^{j}(S)\ge S_*$ satisfy
    \begin{align}
        j & \le C+C\log\log\bigl(e^{e}+S\bigr).
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_repair_margins, thm:area_law_power_recursion_depth}
    Since $0<d(u)\le u^{\rho_0}$ for all $u>0$, we have
    $\log d(S)\le\rho_0\log S$, and induction on $j$ gives
    (\ref{eq:area_law_repair_depth_log}). If $d^{j}(S)\ge S_*$,
    Theorem~\ref{thm:area_law_power_recursion_depth} shows that
    $d^{i}(S)\ge S_*$ for all $0\le i\le j$ and bounds $j+1$ by
    $C+C\log\log(S+e)\le C+C\log\log(e^{e}+S)$.
\end{proof}

\begin{theorem}[Descendant count of an initial mark]
    \label{thm:area_law_repair_descendant_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_repair_descendant_count_bound,
        TNLean.PEPS.AreaLaw.Geometry.exists_repair_descendant_count_layer_bound}
    \leanok
    \uses{def:area_law_repair_square}
    For every $S_*>1$ there exists a constant $C>0$ such that every $S>0$
    and $m\ge0$ with $d^{m}(S)\ge S_*$ satisfy
    \begin{align}
        \sum_{j=0}^{m}9^{j} & \le C\bigl(\log(S+2)\bigr)^{C}.
    \end{align}
    For every $M>0$ there also exists a constant $C>0$, depending only on
    $S_*$ and $M$, such that $\sum_{j=0}^{m}9^{j}\le C(k+1)^{C}$ whenever
    $S\le M2^{k}$ for an integer $k\ge0$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_power_recursion_depth}
    If $d^{m}(S)\ge S_*$, Theorem~\ref{thm:area_law_power_recursion_depth}
    with $b=9$ gives $S\ge S_*>1$ and bounds $\sum_{j=0}^{m}9^{j}$ by
    $C\bigl(\log(S+e)\bigr)^{C}$. Since $S+e\le e(S+2)$ and
    $\log(S+2)\ge1$, we have $\log(S+e)\le2\log(S+2)$, which gives the first
    bound. When $S\le M2^{k}$, $S+2\le(M+2)2^{k}$ and hence
    $\log(S+2)\le(\log(M+2)+1)(k+1)$.
\end{proof}

\begin{theorem}[Fine clearance threshold]
    \label{thm:area_law_fine_clearance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.eventually_fine_clearance}
    \leanok
    Let $c_1>0$ and $C_B\ge0$. Every sufficiently large $u\in\mathbb R$
    satisfies
    \begin{align}
        c_1u^{\rho_0} & >2\lceil C_Bu\rceil^{\beta}.
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    For $u\ge1$, $\lceil C_Bu\rceil\le(C_B+1)u$, so
    $2\lceil C_Bu\rceil^{\beta}\le2(C_B+1)^{\beta}u^{\beta}$. Since
    $\beta<\rho_0$, we have $u^{\beta-\rho_0}\to0$ as $u\to\infty$.
\end{proof}
