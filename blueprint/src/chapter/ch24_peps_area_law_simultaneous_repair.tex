\section{Removal squares and repair children}
\label{sec:area_law_simultaneous_repair}

In the recursive repair of \cite[Section~11]{OpenAI2026AreaLaw}, each
removal step deletes a union of closed squares from all existing regions
before any replacement regions are adjoined.

\begin{definition}[Dyadic removal square]
    \label{def:area_law_repair_square}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.repairSquare}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    Retain the fixed exponent $\rho_0=(1+\beta)/2$. For a parent scale
    $S>0$ and a center $v\in\mathbb R^2$, define the dyadic half-side and
    removal square by
    \begin{align}
        d(S)   & =2^{\lfloor\rho_0\log_2 S\rfloor}, &
        Q(v,S) & =v+[-d(S),d(S)]^2.
        \label{eq:area_law_repair_square}
    \end{align}
\end{definition}

\begin{lemma}[Removal margins]
    \label{thm:area_law_repair_margins}
    \lean{TNLean.PEPS.AreaLaw.Geometry.rho_pos_lt_one,
        TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_le_rpow,
        TNLean.PEPS.AreaLaw.Geometry.half_rpow_le_repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_lt_self,
        TNLean.PEPS.AreaLaw.Geometry.eventually_repairHalfSide_le,
        TNLean.PEPS.AreaLaw.Geometry.eventually_repairHalfSide_margins}
    \leanok
    \uses{def:area_law_repair_square}
    The exponent satisfies $0<\rho_0<1$. For every $S>0$,
    \begin{align}
        \tfrac12S^{\rho_0}\le d(S)\le S^{\rho_0},
        \label{eq:area_law_repair_margins}
    \end{align}
    and $d(S)<S$ whenever $S>1$. For every $a_0>0$ there exists $S_0$ such
    that $d(S)\le a_0S/1000$ for all $S\ge S_0$; hence an arbitrarily large
    stopping scale $S_*$ satisfies the margin conditions of
    \cite[Section~11]{OpenAI2026AreaLaw}.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    For $x=\rho_0\log_2 S$, $\lfloor x\rfloor\le x<\lfloor x\rfloor+1$ gives
    (\ref{eq:area_law_repair_margins}) since $S^{\rho_0}=2^x$ and
    $d(S)=2^{\lfloor x\rfloor}$. Because $\rho_0<1$, $S^{\rho_0}<S$ for
    $S>1$ and $S^{\rho_0}/S=S^{\rho_0-1}\to0$ as $S\to\infty$.
\end{proof}

\begin{lemma}[Integral removal half-sides]
    \label{thm:area_law_repair_halfside_integral}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_natPow_repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.exists_natCast_repairHalfSide}
    \leanok
    \uses{def:area_law_repair_square}
    For every real parent scale $S\ge1$, there exists $e\in\mathbb N$
    such that $d(S)=2^e\in\mathbb N$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_square,thm:area_law_repair_margins}
    Since $\rho_0>0$ and $\log_2 S\ge0$, $e=\lfloor\rho_0\log_2 S\rfloor\in\mathbb N$,
    so (\ref{eq:area_law_repair_square}) gives $d(S)=2^e\in\mathbb N$.
\end{proof}

\begin{definition}[Simultaneous old-region remnants]
    \label{def:area_law_simultaneous_remnants}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairRemovalSet,
        TNLean.PEPS.AreaLaw.Geometry.simultaneousRepairRemnants}
    \leanok
    \uses{def:area_law_repair_square}
    For a finite family of marks $(v_j,S_j)_{j\in J}$ with
    $v_j\in\mathbb R^2$ and $S_j>0$, and a family of regions $(R_i)_{i\in I}$
    in $\mathbb R^2$, define the removal set and trimmed remnants by
    \begin{align}
        Q_J  & =\bigcup_{j\in J}Q(v_j,S_j), &
        R'_i & =R_i\setminus Q_J.
        \label{eq:area_law_simultaneous_remnants}
    \end{align}
\end{definition}

\begin{definition}[Children of nonterminal parents]
    \label{def:area_law_repair_children}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairChildOffset,
        TNLean.PEPS.AreaLaw.Geometry.RepairChildIndex,
        TNLean.PEPS.AreaLaw.Geometry.repairChildLocation,
        TNLean.PEPS.AreaLaw.Geometry.repairChildScale}
    \leanok
    \uses{def:area_law_repair_square}
    For parent marks $(v_i,S_i)_{i\in I}$ with $v_i\in\mathbb R^2$ and
    $S_i>0$, and a stopping threshold $S_*$, set
    $I_+=\{i\in I:d(S_i)\ge S_*\}$ and $P=\{-1,0,1\}^2$. The child marks
    of \cite[Section~11]{OpenAI2026AreaLaw} are indexed by $I_+\times P$,
    with centers (the center, corners, and side midpoints of $Q(v_i,S_i)$)
    and scales
    \begin{align}
        v'_{i,p} & =v_i+d(S_i)p, & S'_{i,p} & =d(S_i),
        \label{eq:area_law_repair_children}
    \end{align}
    while terminal parents ($d(S_i)<S_*$) produce no children.
\end{definition}

\subsection{Depth and descendant count of a repair branch}
\label{sec:area_law_repair_depth}

By (\ref{eq:area_law_repair_children}), generation-$j$ descendants of a mark
of scale $S$ have scale $d^{j}(S)$, and generation $j$ has at most $9^j$ marks.

% Source: OpenAI2026AreaLaw, Section 11, 10-geometry.tex, lines 668--686;
% OpenAI2026PolynomialPEPS, Lemma 3.2, 02-information.tex, lines 302--311.
\begin{lemma}[Depth of a power-contracting recursion]
    \label{thm:area_law_power_recursion_depth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.log_iterate_le_of_le_rpow,
        TNLean.PEPS.AreaLaw.Geometry.one_le_log_add_exp_one,
        TNLean.PEPS.AreaLaw.Geometry.exists_iterate_depth_bound,
        TNLean.PEPS.AreaLaw.Geometry.exists_iterate_generation_count_bound,
        TNLean.PEPS.AreaLaw.Geometry.mul_rpow_le_max_mul_rpow,
        TNLean.PEPS.AreaLaw.Geometry.forall_le_iterate_of_le_iterate}
    \leanok
    Let $0<\rho<1$, $S_*>1$, and $0<f(u)\le u^{\rho}$ for all $u\ge S_*$. If
    $S>0$ satisfies $f^{i}(S)\ge S_*$ for $0\le i<m$, then
    $\log f^{i}(S)\le\rho^{i}\log S$ for $0\le i\le m$, and for every $b\ge0$
    there exists $C>0$ (independent of $S$ and $m$) such that
    \begin{align}
        m                   & \le C+C\log\log(S+e),
        \label{eq:area_law_power_recursion_depth}             \\
        \sum_{j=0}^{m}b^{j} & \le C\bigl(\log(S+e)\bigr)^{C}.
        \label{eq:area_law_power_recursion_sum}
    \end{align}
    If $0<f(u)\le u^{\rho}$ for all $u>0$ and $S_*\ge1$, then
    $f^{m}(S)\ge S_*$ implies $f^{i}(S)\ge S_*$ for all $0\le i\le m$.
\end{lemma}

\begin{proof}
    \leanok
    Induction on $i$ gives $\log f^{i}(S)\le\rho^{i}\log S$. For $m\ge1$,
    $0<\log S_*\le\rho^{m-1}\log S$ yields
    $(m-1)\log(1/\rho)\le\log\log(S+e)-\log\log S_*$, proving
    (\ref{eq:area_law_power_recursion_depth}). With $B=b+2$,
    $\sum_{j=0}^{m}b^{j}\le B^{m+1}\le B^{C+1}\bigl(\log(S+e)\bigr)^{C\log B}$
    gives (\ref{eq:area_law_power_recursion_sum}). Finally, $x<S_*$ implies
    $f(x)\le x^{\rho}<S_*^{\rho}\le S_*$, so iterates stay below $S_*$.
\end{proof}

\begin{lemma}[Depth of a repair branch]
    \label{thm:area_law_repair_depth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairHalfSide_pos,
        TNLean.PEPS.AreaLaw.Geometry.log_repairHalfSide_le,
        TNLean.PEPS.AreaLaw.Geometry.log_repairHalfSide_iterate_le,
        TNLean.PEPS.AreaLaw.Geometry.exists_repair_depth_bound}
    \leanok
    \uses{def:area_law_repair_square}
    For every $S>0$ and $j\ge0$,
    \begin{align}
        \log d^{j}(S) & \le\rho_0^{j}\log S.
        \label{eq:area_law_repair_depth_log}
    \end{align}
    For every $S_*>1$ there exists $C>0$ such that $d^{j}(S)\ge S_*$ implies
    \begin{align}
        j & \le C+C\log\log\bigl(e^{e}+S\bigr).
        \label{eq:area_law_repair_depth_bound}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_repair_margins, thm:area_law_power_recursion_depth}
    By (\ref{eq:area_law_repair_margins}), $0<d(u)\le u^{\rho_0}$, so
    induction on $j$ gives (\ref{eq:area_law_repair_depth_log}). If
    $d^{j}(S)\ge S_*$, Lemma~\ref{thm:area_law_power_recursion_depth} gives
    $d^{i}(S)\ge S_*$ for $0\le i\le j$ and bounds $j+1$ via
    (\ref{eq:area_law_power_recursion_depth}) by
    $C+C\log\log(S+e)\le C+C\log\log(e^{e}+S)$.
\end{proof}

\begin{lemma}[Descendant count of an initial mark]
    \label{thm:area_law_repair_descendant_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_repair_descendant_count_bound,
        TNLean.PEPS.AreaLaw.Geometry.exists_repair_descendant_count_layer_bound}
    \leanok
    \uses{def:area_law_repair_square}
    For every $S_*>1$ there exists $C>0$ such that every $S>0$ and $m\ge0$
    with $d^{m}(S)\ge S_*$ satisfy
    \begin{align}
        \sum_{j=0}^{m}9^{j} & \le C\bigl(\log(S+2)\bigr)^{C}.
        \label{eq:area_law_simultaneous_descendant_count}
    \end{align}
    For every $M>0$ there exists $C>0$, depending only on $S_*$ and $M$, such
    that $\sum_{j=0}^{m}9^{j}\le C(k+1)^{C}$ whenever $S\le M2^{k}$ for an
    integer $k\ge0$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_power_recursion_depth}
    If $d^{m}(S)\ge S_*$, (\ref{eq:area_law_power_recursion_sum}) with $b=9$
    gives $S\ge S_*>1$ and bounds $\sum_{j=0}^{m}9^{j}$ by
    $C\bigl(\log(S+e)\bigr)^{C}$; since $S+e\le e(S+2)$ and $\log(S+2)\ge1$,
    $\log(S+e)\le2\log(S+2)$, giving (\ref{eq:area_law_simultaneous_descendant_count}).
    When $S\le M2^{k}$, $\log(S+2)\le(\log(M+2)+1)(k+1)$.
\end{proof}

\begin{lemma}[Fine clearance threshold]
    \label{thm:area_law_fine_clearance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.eventually_fine_clearance}
    \leanok
    For $c_1>0$ and $C_B\ge0$, every sufficiently large $u\in\mathbb R$
    satisfies
    \begin{align}
        c_1u^{\rho_0} & >2\lceil C_Bu\rceil^{\beta}.
        \notag
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_exponent_gaps}
    For $u\ge1$, $2\lceil C_Bu\rceil^{\beta}\le2(C_B+1)^{\beta}u^{\beta}$,
    and $u^{\beta-\rho_0}\to0$ as $u\to\infty$ since $\beta<\rho_0$.
\end{proof}
