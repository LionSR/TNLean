\section{The sampled construction at successive cuts}
\label{sec:lvv_sampled_construction}

This section follows the sampled projector of \cite{Landau2015Polynomial}.
Its uniform polynomial bounds assume a fixed upper bound $B$ on
$\varepsilon_0/\Delta$, which the printed statement does not assume. The
reason, and the published form of the estimate, are recorded
in~\cite{gap:lvv13_runtime_ground_energy}. The construction of
Section~\ref{sec:lvv_successive_construction} does not need this bound.

\begin{lemma}[Simultaneous bounds for sampled factors]
    \label{thm:lvv_sampled_cut_decomposition}
    \lean{TNLean.MPS.AreaLaw.operatorSchmidtRank_cutOperator_bondComplementWord_le_all,
        TNLean.MPS.AreaLaw.exists_sampledProjector_cutDecomposition}
    \leanok
    Suppose that each letter occurs at most $\kappa$ times in each of $\ell$
    sampled words. Put $D=q^{2\kappa}$. At any cut, the sampled operator has
    a product decomposition with at most $\ell D$ terms. Each left factor
    has operator Schmidt rank at most $D$ at every earlier internal cut.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:lvv_cut_decomposition}
    Every internal cut of a word has rank at most $q^{2\kappa}$; the
    endpoint ranks are at most one. Apply Lemma~\ref{thm:lvv_cut_decomposition}
    to each word separately. Concatenate their decompositions and absorb
    the common sampling scalar into the left factors. Scalar multiplication
    does not increase operator Schmidt rank.
\end{proof}

\begin{lemma}[Sampled error reduction]
    \label{thm:lvv_sampled_error_reduction}
    \lean{TNLean.MPS.AreaLaw.exists_sampled_errorReduction}
    \leanok
    Let $H=\sum_jH_j$ satisfy the positive nearest-neighbour contraction
    assumptions, with a unit ground vector $\Gamma$, ground energy
    $\varepsilon_0$ and positive gap $\Delta$. For $0<\delta\le1$, choose
    the sampling precision $\eta=\delta/8$, the power
    $m=\lceil((L-\varepsilon_0)/\Delta)\log(1/\eta)\rceil$, and the sample
    count $\ell$ and multiplicity bound $\kappa$ of the sampled projector.
    There is a sample with error less than $\delta/4$ from
    $|\Gamma\rangle\langle\Gamma|$. At every cut its left factors can be
    chosen before the input family, so that every $(i,s,b,1/2)$-viable
    family is sent to an $(i,\ell Ds,D\max(1,b),\delta)$-viable family,
    where $D=q^{2\kappa}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:lvv_sampled_cut_decomposition, thm:lvv_mps_error_reduction}
    The power error is at most $\eta$ and the sampling error is less than
    $\eta$. Their sum is less than $\delta/4$. The simultaneous bounds of
    Lemma~\ref{thm:lvv_sampled_cut_decomposition} supply the hypotheses
    of Lemma~\ref{thm:lvv_mps_error_reduction}.
\end{proof}

\begin{lemma}[Rational output of sampled error reduction]
    \label{thm:lvv_rational_sampled_error_reduction}
    \lean{TNLean.MPS.AreaLaw.exists_rational_sampled_errorReduction}
    \leanok
    \uses{thm:lvv_sampled_error_reduction,thm:lvv_rational_orthonormal_family}
    Under the assumptions of Lemma~\ref{thm:lvv_sampled_error_reduction},
    use sampling precision $\eta=\delta/16$ and the corresponding
    $m,\ell,D$. There is a sample within $\delta/8$ of the ground
    projection, with the same simultaneous cut decompositions and internal
    rank bounds as before. At every positive cut $i\le L$, each
    $(i,s,b,1/2)$-viable input family has a rational replacement $T$ of
    the exact error-reduced family $V$, with
    \[
      |T|\le |V|\le\ell Ds,\qquad
      D(T)\le(\ell Ds)\max\{1,D\max\{1,b\}\}.
    \]
    The family $T$ is $\delta$-viable. More precisely, put
    \[
      b'=|V|\max\{1,D\max\{1,b\}\},\quad
      \theta=\frac{\delta}{4(|V|+1)},\quad
      K=q^i\max\{1,b'\}^{i+1}(i+1)2^i,\quad
      t=\left\lceil\log_2\max\{1,\lceil K/\theta\rceil\}\right\rceil.
    \]
    Every member of $T$ has Gaussian-rational tensor parameters of
    common height at most $2^{2t+5}$.
\end{lemma}
\begin{proof}\leanok
    Apply sampled error reduction at target $\delta/2$, retaining the
    sample and every cut decomposition. The exact output $V$ is
    $\delta/2$-viable and has bond bound $D\max\{1,b\}$.
    Apply Lemma~\ref{thm:lvv_rational_orthonormal_family} with
    permitted loss $\delta/2$. Its cardinality and finite-span bond bounds
    give the displayed estimates, and the two viability losses add to
    $\delta$.
\end{proof}

\begin{lemma}[Uniform polynomial sampling bounds]
    \label{thm:lvv_sampled_polynomial_bounds}
    \lean{TNLean.MPS.AreaLaw.sampledProjectorWordDegree,
        TNLean.MPS.AreaLaw.exists_sampledProjectorWordCount_polynomial_explicit_degree,
        TNLean.MPS.AreaLaw.exists_sampledProjectorWordCount_polynomial,
        TNLean.MPS.AreaLaw.sampledProjectorLetterSlope,
        TNLean.MPS.AreaLaw.exists_sampledProjectorLetterBound_logarithmic_explicit_slope,
        TNLean.MPS.AreaLaw.exists_sampledProjectorLetterBound_logarithmic,
        TNLean.MPS.AreaLaw.sampledProjectorRankDegree,
        TNLean.MPS.AreaLaw.sampledProjectorTotalDegree,
        TNLean.MPS.AreaLaw.sampledProjectorEnergyDegreeSlope,
        TNLean.MPS.AreaLaw.sampledProjectorEnergyDegreeIntercept,
        TNLean.MPS.AreaLaw.sampledProjectorTotalDegree_le_affine,
        TNLean.MPS.AreaLaw.sampledGroundStateDegree_le_affine,
        TNLean.MPS.AreaLaw.exists_sampledProjector_uniform_polynomial_explicit_degree,
        TNLean.MPS.AreaLaw.exists_sampledProjector_uniform_polynomial}
    \leanok
    Fix $q\ge1$, $\Delta>0$, $0<c\le1$, a natural number $s$ and a real bound $B$.
    There are constants $C\ge1$ and $p\in\mathbb N$, chosen before $L$
    and $\varepsilon_0$, such that whenever $L\ge1$,
    $0\le\varepsilon_0\le L-1$ and $\varepsilon_0/\Delta\le B$, sampling
    at precision $c/L^s$ gives $\ell D\le CL^p$ and $D\le CL^p$.
    More precisely, with $r=\lceil B\rceil$, the count $\ell$ has degree
    \begin{align*}
      p_{\rm word}&=2(sr+1)+(2s+1)=2sr+2s+3,\\
      a_{\rm letter}&=\frac{2s}{\Delta}+2(p_{\rm word}+4),\\
      p_{\rm rank}&=\left\lceil2a_{\rm letter}\log q\right\rceil,
      \qquad p_{\rm total}=p_{\rm word}+p_{\rm rank}.
    \end{align*}
    The count satisfies $\ell\le C_1L^{p_{\rm word}}$, every letter occurs
    at most $a_{\rm letter}\log L+b$ times, and both $\ell D$ and $D$ are
    bounded by $C_2L^{p_{\rm total}}$.
    Quantitatively,
    \begin{align*}
      p_{\rm slope}&=2s+\left\lceil8s\log q\right\rceil,\\
      p_{\rm int}&=2s+3+
        \left\lceil\left(\frac{4s}{\Delta}+8s+28\right)\log q\right\rceil,
    \end{align*}
    and $p_{\rm total}\le p_{\rm int}+p_{\rm slope}\lceil B\rceil$.
    Consequently a final-MPS degree $2p_{\rm total}+p_0$ is at most
    $(2p_{\rm int}+p_0)+2p_{\rm slope}\lceil B\rceil$.
\end{lemma}
\begin{proof}
    \leanok
    Set $r=\lceil B\rceil$. The normalization obeys
    $C_0^m\le L(L^s/c)^r$. Substitution in the sample-count formula gives
    $\ell\le C_1L^{2sr+2s+3}$. Also
    $m/L\le\Delta^{-1}(\log(1/c)+s\log L)+1$.
    Substitution in $\kappa=\lceil2m/L+2\log(4\ell L^4)\rceil$ gives
    $\kappa\le a\log L+b$. Thus
    $q^{2\kappa}\le\exp(2b\log q)L^{\lceil2a\log q\rceil}$.
    Multiplying these bounds and rounding the coefficient proves the claim.
\end{proof}

\begin{definition}[Successive viability precision]
    \label{def:lvv_successive_precision}
    \lean{TNLean.MPS.AreaLaw.lvvViabilityPrecision}
    \leanok
    For a positive gap $\Delta$ and target accuracy $\eta>0$, set
    $\alpha=\min\{1,\eta,(\Delta/169)^2\}$.
\end{definition}

\begin{lemma}[A complete step with fixed polynomial bounds]
    \label{thm:lvv_sampled_successor}
    \lean{TNLean.MPS.AreaLaw.exists_sampled_viable_successor_polynomial_explicit_energy_degree,
        TNLean.MPS.AreaLaw.exists_sampled_viable_successor_polynomial}
    \leanok
    \uses{def:lvv_successive_precision, def:lvv_successive_step,
        def:lvv_size_trimming_selection}
    Fix $q\ge2$, $\Delta>0$, $\eta>0$, an integer $a\ge1$, and an upper bound $B$ for
    $\varepsilon_0/\Delta$. There are natural constants $C,D\ge1$ and
    $p,r$ such that, for every positive chain length and every cut
    $1\le i\le L$, any finite family $S$ of left vectors
    that is $(i-1,\alpha/L^a)$-viable admits a sampled projector $K$
    within $\alpha/(4L^a)$ of $|\Gamma\rangle\langle\Gamma|$ in operator
    norm, with left factors $A_1,\ldots,A_N$ at the cut after $i$ sites,
    such that the step of Definition~\ref{def:lvv_successive_step}
    applied, with these factors and the corrected radius, to the
    size-trimming selection of $S\otimes\{|1\rangle,\ldots,|q\rangle\}$ is
    $(i,CL^p,DL^r,\alpha/L^a)$-viable.
    The bounds are independent of the size and bond bound of the input family.
    More precisely, there are offsets $p_0,r_0$, independent of $B$, for
    which one may take $p=p_{\rm total}+p_0$ and
    $r=p_{\rm total}+r_0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:lvv_sampled_error_reduction, thm:lvv_sampled_polynomial_bounds,
        thm:lvv_size_trimmed_family_bond_trim_set, lem:lvv_bond_trimming_radius_polynomial}
    Extension preserves viability. Since $\alpha\le(\Delta/169)^2$ and $L^a\ge L$,
    the trimmed size-trimming selection of the extended family is
    half-viable, of polynomial cardinality and individual bond
    bound, with the bond trimming radius corrected as
    in~\cite{gap:lvv13_trimming_radius}. Sampled error reduction by the
    left factors of the sampled projector restores error $\alpha/L^a$ on
    the step. Multiply its polynomial factor counts and ranks by the
    corresponding trimming bounds. All coefficients and degrees are chosen
    before $L$.
\end{proof}

\begin{lemma}[Successive viable families]
    \label{thm:lvv_sampled_successive_families}
    \lean{TNLean.MPS.AreaLaw.exists_sampled_successive_viable_sets_polynomial_explicit_energy_degree,
        TNLean.MPS.AreaLaw.exists_sampled_successive_viable_sets_polynomial_total_cardinality,
        TNLean.MPS.AreaLaw.exists_sampled_successive_viable_sets_polynomial}
    \leanok
    \uses{def:lvv_successive_step, def:lvv_size_trimming_selection}
    Under the hypotheses of Lemma~\ref{thm:lvv_sampled_successor}, there
    are families $S_i$ at all cuts $0\le i\le L$, beginning with the scalar
    singleton, such that each is $(i,CL^p,DL^r,\alpha/L^a)$-viable, with
    $p=p_{\rm total}+p_0$ and $r=p_{\rm total}+r_0$ for offsets independent
    of $B$.  The families may be chosen so that the total number of retained
    candidate vectors throughout the sweep obeys
    \begin{align*}
      \sum_{i=0}^{L}|S_i|\le CL^{p+1}.
    \end{align*}
    This counts all retained candidates; it does not include the arithmetic
    cost of the convex minimizations or spectral computations.
    Moreover, each successor is the step of
    Definition~\ref{def:lvv_successive_step}, applied to the size-trimming
    selection of the extended preceding family, with a sampled projector
    within $\alpha/(4L^a)$ of $|\Gamma\rangle\langle\Gamma|$ and the corrected radius.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:lvv_sampled_successor, lem:lvv_initial_scalar_family}
    The scalar singleton is exactly viable. Choose, for every family at a
    cut, the sampled projector and the step of
    Lemma~\ref{thm:lvv_sampled_successor}, and apply this choice
    inductively at the successive cuts. The same size and bond bounds
    apply at each step.
    Summing the $L+1$ cardinality bounds and using $L+1\le2L$ gives the total count.
\end{proof}
