\section{Restricting the source-register operators}
\label{sec:peps_source_register_operators}

Let $\mathcal P$ be the set of parties, and let
$R=((p_i,q_i))_{i=0}^{n-1}$ be an ordered list of pairs of distinct parties.
For each $i$, let $U_i,V_i$ be complex inner product spaces. The ordered
register list is
\begin{align}
    \mathcal L_R
    = ((p_0,U_0),(q_0,V_0),\ldots,(p_{n-1},U_{n-1}),(q_{n-1},V_{n-1})).
    \label{eq:peps_source_register_order}
\end{align}
An empty register list has memory space $\C$. Register memories are tensored
in the order~(\ref{eq:peps_source_register_order}), with a final scalar factor.

\begin{definition}[The word of local source-register operators]
    \label{def:peps_source_register_operator_word}
    \lean{TNLean.PEPS.PairEffect.Word.localEndpoint}
    \lean{TNLean.PEPS.PairEffect.Word.sources_localEndpoint}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.endpointWord}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.sources_endpointWord}
    \leanok
    \uses{def:peps_party_layout,def:peps_source_slots}
    Let $A_i:U_i\to U_i$ and $B_i:V_i\to V_i$ be continuous linear maps.
    The word $Z_R$ applies $A_i$ to the register $(p_i,U_i)$ and $B_i$ to
    $(q_i,V_i)$, leaving every other register unchanged, in increasing order
    of $i$. It contains no source preparations. In particular,
    $Z_{\varnothing}=\Id_{\C}$.
    % Source: 04-compression.tex, lines 279--309 and 565--588.
    These are the local matrices used in the random source replacement
    and the contraction within each party in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
\end{definition}

\begin{theorem}[Restriction away from every source owner]
    \label{thm:peps_source_register_operator_exclusion}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.restrict_endpointWord_eq_nil_and_eval_eq_id}
    \leanok
    \uses{def:peps_source_register_operator_word,thm:peps_party_restriction}
    Let $f:\mathcal P\to\{0,1\}$ select parties, and suppose that
    $f(p_i)=f(q_i)=0$ for every $i$. Then the selected register list is
    empty, and the restricted word $Z_R|_f$ acts as $\Id_{\C}$.
    The register spaces and their local maps are arbitrary, and $n$ may be
    zero.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_source_register_operator_word,thm:peps_party_restriction,
        thm:peps_party_factorization}
    Every party occurring in $Z_R$ owns a register in $\mathcal L_R$.
    Indeed, adjoining the first pair introduces only its two owners through
    its local maps and unchanged registers; the claim for the remaining
    pairs follows by induction. Thus the assumptions exclude every
    participating party. Restriction retains exactly the selected
    participating parties, so none remain. A source-free word with no
    participating party has empty input and output lists and acts as
    $\Id_{\C}$.
\end{proof}


\begin{theorem}[Action on the original endpoint factors]
    \label{thm:peps_source_register_operator_evaluation}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.eval_endpointWord_slotVector_tmul}
    \leanok
    \uses{def:peps_source_register_operator_word,def:peps_joint_source_vector}
    Let $u_i\in U_i$ and $v_i\in V_i$. In the original register order, the
    actual endpoint word satisfies
    \[
        Z_R\left(\bigotimes_i(u_i\otimes v_i)\right)
        =\bigotimes_i(A_i u_i\otimes B_i v_i).
    \]
    The vectors and maps are arbitrary. An empty list of slots gives the
    scalar identity, with both empty tensors equal to $1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_source_register_operator_word,def:peps_joint_source_vector}
    Induct on the ordered source list. At the first pair, the two recorded
    endpoint operations send $u_0\otimes v_0$ to
    $A_0u_0\otimes B_0v_0$, leaving the remaining registers unchanged.
    Apply the induction hypothesis to those registers. The canonical
    association and register identifications preserve the original order.
\end{proof}

\begin{theorem}[Coordinates of the original joint source vector]
    \label{thm:peps_joint_source_vector_coordinates}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotVector_eq_sum_slotBasis}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotBasis_repr_slotVector}
    \leanok
    \uses{def:peps_joint_source_vector,def:peps_source_slot_basis}
    Choose finite orthonormal bases $b_i,c_i$ of $U_i,V_i$ and let $e_x$ be their
    ordered joint source basis. For arbitrary $\eta_i\in U_i\otimes V_i$,
    \begin{align}
        \bigotimes_i\eta_i
         & =\sum_x\left(\prod_i
        \langle b_{i,x_i^{(1)}}\otimes c_{i,x_i^{(2)}},\eta_i\rangle\right)e_x.
        \label{eq:peps_joint_source_coordinates}
    \end{align}
    Thus each coordinate is the product of the corresponding individual
    source coordinates. The assertion includes empty source lists and
    zero-dimensional register spaces.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_source_slot_basis,def:peps_joint_source_vector}
    Expand each factor in its orthonormal basis and distribute the finite
    tensor product over the finite sums. Taking the coordinate of $e_x$
    leaves the coefficient in~(\ref{eq:peps_joint_source_coordinates}).
\end{proof}

\begin{definition}[The operator with specified orthonormal coordinates]
    \label{def:peps_operator_from_basis_matrix}
    \lean{TNLean.PEPS.PairEffect.basisMatrixOperator}
    \lean{TNLean.PEPS.PairEffect.basisMatrixOperator_repr}
    \leanok
    Let $b=(b_j)_{j\in I}$ be an orthonormal basis of a finite-dimensional
    complex inner product space. For a matrix $M$, define $T_b(M)$ by
    $T_b(M)b_j=\sum_i M_{ij}b_i$. Its matrix in $b$ is exactly $M$.
    The matrix may be arbitrary, and the coordinate set may be empty.
\end{definition}

\begin{theorem}[Coordinates of the source-endpoint operator]
    \label{thm:peps_endpoint_operator_coordinates}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotBasis_repr_endpointWord}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.endpointMatrixWord}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.slotBasis_repr_endpointMatrixWord}
    \leanok
    \uses{thm:peps_source_register_operator_evaluation,
        thm:peps_joint_source_vector_coordinates,def:peps_operator_from_basis_matrix}
    In the joint source basis, the endpoint operator satisfies
    \begin{align}
        [Z_R]_{x,y} & =\prod_i
        \langle b_{i,x_i^{(1)}},A_i b_{i,y_i^{(1)}}\rangle
        \langle c_{i,x_i^{(2)}},B_i c_{i,y_i^{(2)}}\rangle.
        \label{eq:peps_endpoint_coordinates}
    \end{align}
    In particular, constructing the endpoint maps from matrices $L_i,H_i$
    gives $[Z_R]_{x,y}=\prod_i (L_i)_{x_i^{(1)},y_i^{(1)}}
        (H_i)_{x_i^{(2)},y_i^{(2)}}$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_source_register_operator_evaluation,
        thm:peps_joint_source_vector_coordinates,def:peps_operator_from_basis_matrix}
    Apply the endpoint action to the basis vector $e_y$, then take its
    coordinate at $e_x$. Inner products of elementary tensors multiply.
\end{proof}

\begin{theorem}[Finite averaging of endpoint operators]
    \label{thm:peps_endpoint_matrix_average}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.endpointMatrixAverage_eq_product}
    \leanok
    \uses{thm:peps_endpoint_operator_coordinates}
    Let $L_{i,s},H_{i,s}$ be endpoint matrices for $s\in\{0,\ldots,k-1\}$.
    Then, entrywise in the joint source basis,
    \begin{align}
        \bigotimes_{i=0}^{n-1}
        \left(k^{-1}\sum_{s=0}^{k-1} L_{i,s}\otimes H_{i,s}\right)
         & =k^{-n}\sum_{j:\{0,\ldots,n-1\}\to\{0,\ldots,k-1\}} Z_j,
        \label{eq:peps_endpoint_average}
    \end{align}
    where $Z_j$ is the actual endpoint operator with maps constructed from
    $L_{i,j(i)},H_{i,j(i)}$. The same assignment $j(i)$ is used at both
    endpoints. The algebraic identity includes $k=0$, with $0^{-1}=0$,
    and $n=0$, with the empty assignment contributing the scalar identity.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_endpoint_operator_coordinates}
    Substitute the endpoint-coordinate formula and distribute the product
    of finite sums. One choice of a summand at every original slot gives
    exactly one global assignment $j$.
\end{proof}

\begin{theorem}[Matrices in orthonormal coordinates]
    \label{thm:peps_matrix_orthonormal_coordinates}
    \lean{TNLean.PEPS.PairEffect.matrix_repr_eval_eq_toMatrix}
    \leanok
    Let $b,c$ be finite orthonormal bases and let $A$ be a continuous linear

\begin{theorem}[Dependence on the selected endpoint maps]
    \label{thm:peps_source_register_operator_local_congruence}
    \lean{TNLean.PEPS.PairEffect.SourceInventory.restrict_endpointWord_eval_eq_of_local_eq}
    \leanok
    \uses{def:peps_source_register_operator_word,thm:peps_party_restriction}
    Fix the ordered source-register list and a party selector $f$.
    Consider endpoint maps $A_i,B_i$ and $A'_i,B'_i$ on the same spaces.
    Suppose that $A_i=A'_i$ whenever $f(p_i)=1$, and $B_i=B'_i$ whenever
    $f(q_i)=1$. Then the two restricted source-register operators agree:
    \begin{align*}
        Z_R(A,B)|_f & =Z_R(A',B')|_f.
    \end{align*}
    No equality is required at an unselected endpoint. The register order
    and spectator memories are unchanged; an empty inventory and
    zero-dimensional spaces are included. This is the pointwise endpoint
    locality used in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 279--309 and 565--588.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_source_register_operator_word,thm:peps_party_restriction}
    Induct on the ordered source inventory. An unselected endpoint map
    is omitted in either restriction. At a selected endpoint the two maps
    agree by hypothesis. Tensoring with the unchanged spectator identities
    and composing with the inductive equality preserves equality. The
    canonical identifications of the fixed register lists are the same on
    both sides.
\end{proof}

\begin{theorem}[Restriction of an exchange of register blocks]
    \label{thm:peps_party_restriction_exchange_blocks}
    \lean{TNLean.PEPS.PairEffect.Word.restrict_exchangeBlocks_eval_heq}
    \uses{thm:peps_party_restriction,thm:peps_basis_memory_equalities}
    Let $a,b,t$ be arbitrary ordered register lists, and let $f$ select
    owners. Write $\ell|_f$ for the retained list, in its original order.
    Let $S_{a,b;t}$ exchange the successive blocks $a,b$ in front of the
    tail $t$, preserving the order within each block and leaving the tail unchanged.
    After the canonical identifications of the selected input and output
    register lists, its restricted operator satisfies
    \begin{align}
        (S_{a,b;t})|_f
         & =S_{a|_f,b|_f;t|_f}.
        \label{eq:peps_restricted_block_exchange}
    \end{align}
    No retained block or spectator memory is required to be empty.
    Empty lists and zero-dimensional register spaces are included.
    This is the register-order identity used when grouping sources in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 233--267, 409--434 and 565--588.
\end{theorem}

\begin{proof}
    \uses{thm:peps_party_restriction,thm:peps_basis_memory_equalities}
    First move one register through a prefix by successive exchanges.
    If that register is retained, exchanges with retained prefix registers
    give the corresponding move through the retained prefix; exchanges
    with erased registers act as identities. If the moving register is
    erased, every exchange restricts to an identity on the retained
    memory. Framing preserves these identities with the retained
    spectators. Now induct on the second block. Moving its first retained
    register through the first block, followed by the exchange of the
    remaining blocks, gives (\ref{eq:peps_restricted_block_exchange}).
    An erased first register instead contributes the identity before
    the inductive exchange. The canonical list identifications preserve
    each composition.
\end{proof}

\begin{theorem}[Operator identities under composition and framing]
    \label{thm:peps_operator_composition_framing}

    $I_{\mathcal H_r}\otimes U=I_{\mathcal H_r}\otimes U'$.
    These assertions do not require the compositions to be
    source-free or contractive. They are elementary compatibility
    identities for the register identifications used in

\end{theorem}

\begin{proof}
    \uses{thm:peps_basis_memory_equalities}
    Identify the equal input, intermediate and output register lists.
    The first assertion follows by composing equal operators in the
    indicated chronological order. The second follows by tensoring
    their equality with $I_{\mathcal H_r}$.
\end{proof}

\begin{theorem}[Framing a local part with erased sources]

    Let $W:a\to b$ be an arbitrary chronological composition, with
    ordered source-register list $s$. Let $f$ select owners such
    that $s|_f$ is empty, and let $t$ be any spectator list.
    Write $F_t(W)$ for $W$ framed in front by the registers of $t$,
    and $L(W):sa\to b$ for its local part. Under the canonical
    identifications of the retained register lists,
    \begin{align}
        (L(F_t(W)))|_f
         & =F_{t|_f}((L(W))|_f).
        \label{eq:peps_local_part_framing}
    \end{align}
    The original source inventory need not be empty. The input,
    output and retained spectator memories may be arbitrary,
    including empty lists and zero-dimensional spaces.
    This is an auxiliary identity used in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 233--267, 351--381, 409--434 and 565--588.
\end{theorem}

\begin{proof}
    \uses{thm:peps_source_preparation,thm:peps_party_restriction,
        thm:peps_erased_prefix_register_move,
        thm:peps_operator_composition_framing,thm:peps_basis_memory_equalities}
    Induct on $t$. The empty list gives the identity directly.
    For $t=rt'$, forming the local part first moves $r$ through
    the source prefix, then frames the local part for $t'$ by $r$.
    The source registers are erased, so the restricted move is the
    identity by Theorem~\ref{thm:peps_erased_prefix_register_move}.
    If $r$ is retained, frame the inductive identity by $r$; otherwise
    erase it. Compatibility with composition and the canonical list
    identifications gives (\ref{eq:peps_local_part_framing}).
\end{proof}

\begin{theorem}[Identities and input identifications under restriction]
    \label{thm:peps_identity_input_identification}
    \lean{TNLean.PEPS.PairEffect.Word.eval_id_heq,
        TNLean.PEPS.PairEffect.Word.restrict_castInput_eval_heq}
    \uses{thm:peps_basis_memory_equalities,thm:peps_party_restriction}
    Let $a=a'$ be an equality of ordered register lists. The identity
    operators on $\mathcal H_a$ and $\mathcal H_{a'}$ agree under
    the canonical memory identification. Moreover, let $W:a\to b$
    be source-free and let $f$ select owners. Replacing its input
    list by the equal list $a'$ before restriction gives the same
    selected operator as restricting $W$ directly, under the induced
    identification $a|_f=a'|_f$. The register spaces and selected
    memories are arbitrary. These are elementary compatibility
    identities for the register identifications in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 233--251 and 409--434.
\end{theorem}

\begin{proof}
    \uses{thm:peps_basis_memory_equalities,thm:peps_party_restriction}
    Identify the equal register lists. Both assertions reduce to
    equality of an operator with itself.
\end{proof}

\begin{theorem}[Spectators outside the owners of a sourced composition]
    \label{thm:peps_sourced_spectator_owner_exclusion}
    \lean{TNLean.PEPS.PairEffect.Word.restrict_localPart_mapOwner_appendTail_eval_heq_id_of_disjoint}
    \uses{thm:peps_source_word_parties,def:peps_append_spectators,
        thm:peps_party_restriction,thm:peps_basis_memory_equalities}
    Let $U:a\to b$ be an arbitrary chronological composition whose
    owners lie in $C$, and let $\gamma:C\to\mathcal P$ relabel them.
    Fix $p\in\mathcal P$ with $\gamma(c)\ne p$ for every $c\in C$.
    Extend $U^\gamma$ by an arbitrary spectator list $t$, obtaining
    $(U^\gamma)_t$, and move its source preparations to the beginning.
    The local part then satisfies, under the canonical retained-list
    identifications,
    \begin{align}
        \bigl(L((U^\gamma)_t)\bigr)|_p
         & =\Id_{\mathcal H_{t|_p}}.
        \label{eq:peps_sourced_spectator_exclusion}
    \end{align}
    The source inventory of $U$ may be nonempty, its source vectors
    are arbitrary, and the spectator memory at $p$ may be arbitrary.
    Empty lists and zero-dimensional spaces are included.
    This is an auxiliary owner-exclusion identity for
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 233--267, 351--381, 409--434 and 565--588.
\end{theorem}

\begin{proof}
    \uses{thm:peps_source_word_parties,def:peps_append_spectators,
        thm:peps_source_free_local_part_restriction,
        thm:peps_party_restriction_exchange_blocks,
        thm:peps_party_restriction_canonical_layouts,
        thm:peps_local_part_framing_restriction,
        thm:peps_excluded_spectator_restriction,
        thm:peps_operator_composition_framing,
        thm:peps_identity_input_identification}
    No core input, output or source register is owned by $p$.
    By Theorem~\ref{thm:peps_source_word_parties}, the restricted
    core local part is the identity on the empty memory.
    In the spectator extension, each of the initial and final
    exchanges therefore exchanges the retained spectators with an
    empty block, giving their identity. For the middle framed
    composition, Theorem~\ref{thm:peps_local_part_framing_restriction}
    tensors the scalar identity with the retained spectator identity.
    When taking the local part of either chronological composition,
    the additional source prefix is erased. Its framing contributes
    no selected register. Compose the three identity operators under
    the canonical list identifications to obtain
    (\ref{eq:peps_sourced_spectator_exclusion}).
\end{proof}
