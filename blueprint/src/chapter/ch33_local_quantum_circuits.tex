\chapter{Local Quantum Circuits}
\label{ch:local_quantum_circuits}

A chain of $N$ sites of local dimension $d$ carries $(\C^d)^{\otimes N}$, with
the computational basis labelled by the configurations of the chain. A local
circuit is a product of layers of unitaries on disjoint pairs of neighbouring
sites of the chain closed into a ring \cite{Malz2024Preparation}, or more
generally on disjoint bonds of a geometry such as the edges of a lattice
\cite{Piroli2021LOCC}. A circuit of depth $T$ enlarges the set of sites on
which an operator acts by at most $T$ steps along the bonds, on the ring by at
most $T$ sites on each side, so a state that it prepares from a product state
has no connected correlations beyond distance $2T$, and every unitary on a bounded
number of sites is a product of a bounded number of gates on neighbouring
sites. Measurements in the computational basis, followed by corrections on
single sites chosen from all the outcomes \cite{Piroli2021LOCC}, prepare
GHZ-type states in constant depth, teleport sites along chains of neighbouring
pairs, and apply gates on pairs of distant sites or of distant registers in
constant depth. Circuits of channels on neighbouring pairs, with ancillas
attached and discarded site by site, obey the same light cone.
This chapter treats unitary circuits;
Chapters~\ref{ch:circuits_measurements}--\ref{ch:circuits_lieb_robinson} treat
measurements and long-range gates, circuits of channels, and Lieb--Robinson
bounds. Chapter~\ref{ch:log_depth_preparation} applies these results to the
preparation of matrix product states.

\section{Operators on sets of sites and local circuits}
\label{sec:qc_local_circuits}

\begin{definition}[Local circuit of depth $T$]
    \label{def:qc_local_circuit}
    \lean{QuantumCircuit.IsLocalCircuitOfDepth, QuantumCircuit.IsPreparedInDepth,
        QuantumCircuit.Layer, QuantumCircuit.circuitOp,
        QuantumCircuit.BondLayer, QuantumCircuit.IsBondCircuitOfDepth,
        QuantumCircuit.IsBondPreparedInDepth, QuantumCircuit.ringBond,
        QuantumCircuit.supportedOperators,
        QuantumCircuit.rectKronecker_mem_supportedOperators}
    \leanok
    A \emph{bond geometry} on a finite site set $I$ is a family $(B_b)_{b\in\beta}$ of subsets $B_b\subseteq I$. An operator on $(\C^d)^{\otimes I}$ \emph{acts on} $X\subseteq I$ if it lies in $\mathcal A_X=\MN{d}^{\otimes X}\otimes\mathbb 1_{I\setminus X}=\operatorname{span}\{\bigotimes_{i\in I}m_i:m_i=\mathbb 1\text{ for }i\notin X\}$. A \emph{layer} is a product of unitaries on pairwise disjoint bonds, a \emph{local circuit of depth $T$} is a product $U=W_T\cdots W_1$ of $T$ layers, and a state $\ket\psi$ is \emph{prepared in depth $T$} if $\ket\psi=U\bigotimes_{i\in I}\ket{v_i}$.

    On the $N$-site chain $I=\{1,\dots,N\}$, the ring bonds are $\{k,k+1\}\pmod N$ ($\{1\}$ for $N=1$), the open-chain bonds are $\{k,k+1\}$ ($1\le k<N$), and graph bonds are the edges~\cite{Piroli2021LOCC}.
\end{definition}

\begin{lemma}[Matrix entries of a supported operator]
    \label{thm:area_law_supported_identity}
    \lean{QuantumCircuit.isSupportedOn_of_mem_supportedOperators,
        QuantumCircuit.mem_supportedOperators_of_isSupportedOn}
    \leanok
    \uses{def:qc_local_circuit}
    Let $I$ be a finite set, $q\in\mathbb N$, $S\subseteq I$, and $\mathcal C=\{0,\ldots,q-1\}^I$. A matrix $A\in\C^{\mathcal C\times\mathcal C}$ belongs to $\mathcal A_S=\operatorname{span}\{\bigotimes_{i\in I}m_i:m_i=\mathbb 1\text{ for }i\notin S\}$ if and only if:
    \begin{enumerate}
        \item $\sigma|_{S^c}\ne\tau|_{S^c}\implies A_{\sigma\tau}=0$;
        \item $\sigma|_S=\sigma'|_S,\,\tau|_S=\tau'|_S,\,\sigma|_{S^c}=\tau|_{S^c},\,\sigma'|_{S^c}=\tau'|_{S^c}\implies A_{\sigma\tau}=A_{\sigma'\tau'}$.
    \end{enumerate}
    This characterizes the support condition of~\cite[Lemma~3.1]{OpenAI2026AreaLaw}, \texttt{02-initial.tex}, line~39.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_local_circuit}
    Every generator $A=\bigotimes_i m_i$ ($m_i=\mathbb 1$ for $i\notin S$) has entries
    \begin{align}
        A_{\sigma\tau}
         & =\prod_{i\in S}(m_i)_{\sigma_i\tau_i}
        \prod_{i\notin S}\delta_{\sigma_i\tau_i},\notag
    \end{align}
    satisfying (1)--(2), which pass to linear combinations. Conversely, if $\mathcal C=\varnothing$ then $A=0$; otherwise fix $\rho\in\mathcal C$ and set $K_{ab}=A_{a\rho,\,b\rho}$ for $a,b\in\{0,\dots,q-1\}^S$, where $a\rho|_S=a$ and $a\rho|_{S^c}=\rho|_{S^c}$. By (1)--(2),
    \begin{align}
        A_{\sigma\tau}
         & =K_{\sigma|_S\,\tau|_S}\,\delta_{\sigma|_{S^c}\,\tau|_{S^c}},\notag \\
        A
         & =\sum_{a,b}K_{ab}\,(e_{ab}\otimes\mathbb 1)
        =\sum_{a,b}K_{ab}\Bigl(\bigotimes_{i\in S}e_{a_ib_i}\otimes\bigotimes_{i\notin S}\mathbb 1\Bigr)\in\mathcal A_S.\notag
    \end{align}
\end{proof}

\begin{definition}[Pairing local coordinates]
    \label{def:qc_site_pair_coordinates}
    \lean{QuantumCircuit.sitePairConfigurationEquiv}
    \leanok
    For a finite site set $I$ and integers $d,e\ge0$, the sitewise bijection $(a,b)\mapsto ea+b$ from $\{0,\dots,d-1\}\times\{0,\dots,e-1\}$ to $\{0,\dots,de-1\}$ induces a configuration bijection $p\colon\{0,\dots,d-1\}^I\times\{0,\dots,e-1\}^I\xrightarrow{\sim}\{0,\dots,de-1\}^I$.
\end{definition}

\begin{lemma}[Support of a tensor product on paired sites]
    \label{thm:qc_site_pair_tensor_support}
    \lean{QuantumCircuit.reindex_kronecker_mem_supportedOperators}
    \leanok
    \uses{def:qc_local_circuit,def:qc_site_pair_coordinates}
    As in~\cite{OpenAI2026PolynomialPEPS}, \texttt{02-information.tex}, lines 416--424 and 513--524, if $A$ on $(\C^d)^{\otimes I}$ is supported on $S\subseteq I$ and $B$ on $(\C^e)^{\otimes I}$ is supported on $T\subseteq I$, then $A\otimes B$ reindexed by $p$ is supported on $S\cup T$ in $(\C^{de})^{\otimes I}$.
\end{lemma}
\begin{proof}\leanok
    By bilinearity, for $A=\bigotimes_i A_i$ ($A_i=\mathbb 1$ outside $S$) and $B=\bigotimes_i B_i$ ($B_i=\mathbb 1$ outside $T$), reindexing by $p$ identifies $A\otimes B$ with $\bigotimes_i(A_i\otimes B_i)$, where $A_i\otimes B_i=\mathbb 1$ for $i\notin S\cup T$.
\end{proof}

\begin{definition}[Light cone of a bond geometry]
    \label{def:qc_light_cone_geometry}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.bondNeighbourhood, QuantumCircuit.lightCone,
        QuantumCircuit.neighbourhood, QuantumCircuit.IsSeparatedBy,
        QuantumCircuit.openBond, QuantumCircuit.edgeBond}
    \leanok
    For a bond geometry $(B_b)_{b\in\beta}$ on $I$ and $X\subseteq I$, the \emph{one-step neighbourhood} is $\mathcal N_1(X)=X\cup\bigcup_{B_b\cap X\ne\varnothing}B_b$, and the \emph{light cone of radius $r$} is the $r$-fold iterate $\mathcal L_r(X)=\mathcal N_1^r(X)$. On the ring, the $r$-neighbourhood of $X$ is $\{i+m\pmod N:i\in X,\,|m|\le r\}$, and $X,Y$ are separated by $s$ if $d_{\rm ring}(X,Y)>s$. On a graph, bonds are edges and distance is the path metric $d_G$~\cite{Piroli2021LOCC}.
\end{definition}

\begin{lemma}[Light cones of the ring, the open chain and a graph]
    \label{lem:qc_light_cone_geometries}
    \uses{def:qc_light_cone_geometry}
    \lean{QuantumCircuit.lightCone_ringBond,
        QuantumCircuit.bondNeighbourhood_ringBond, QuantumCircuit.neighbourhood_succ,
        QuantumCircuit.disjoint_lightCone_ringBond_of_isSeparatedBy,
        QuantumCircuit.lightCone_openBond_subset, QuantumCircuit.disjoint_lightCone_openBond,
        QuantumCircuit.lightCone_edgeBond_subset,
        QuantumCircuit.lightCone_edgeBond_subset_dist,
        QuantumCircuit.disjoint_lightCone_edgeBond,
        QuantumCircuit.disjoint_lightCone_edgeBond_of_dist,
        QuantumCircuit.lightCone_lightCone}
    \leanok
    For every bond geometry, $\mathcal L_s(\mathcal L_r(X))=\mathcal L_{r+s}(X)$. Furthermore:
    \begin{enumerate}
        \item On the ring ($N\ge1$), $\mathcal L_r(X)=\{i+m\pmod N:i\in X,\,|m|\le r\}$.
        \item On the open chain, $\mathcal L_r(X)\subseteq\{j:|j-i|\le r\text{ for some }i\in X\}$.
        \item On a graph $G$, $\mathcal L_r(X)\subseteq\{v:d_G(v,X)\le r\}$.
    \end{enumerate}
    In each case, $d(X,Y)>2r\implies\mathcal L_r(X)\cap\mathcal L_r(Y)=\varnothing$.
\end{lemma}
\begin{proof}
    \leanok
    Since $\mathcal L_r=\mathcal N_1^r$, $\mathcal L_s(\mathcal L_r(X))=\mathcal L_{r+s}(X)$. On the ring, the bonds meeting $i\in X$ are $\{i-1,i\}$ and $\{i,i+1\}$, so $\mathcal N_1(X)=\{i+m:|m|\le1\}$, and induction on $r$ gives (1). Items (2)--(3) and disjointness for $d(X,Y)>2r$ follow by induction on $r$ and the triangle inequality.
\end{proof}

\begin{lemma}[Light cone of a local circuit]
    \label{lem:qc_light_cone}
    \uses{def:qc_local_circuit, def:qc_light_cone_geometry,
        lem:qc_light_cone_geometries}
    \lean{QuantumCircuit.conj_circuitOp_mem_supportedOperators,
        QuantumCircuit.IsBondCircuitOfDepth.conj_mem_supportedOperators,
        QuantumCircuit.conj_circuitOp_mem_supportedOperators_lightCone,
        QuantumCircuit.BondLayer.conj_op_mem_supportedOperators}
    \leanok
    As in~\cite[Theorem~1]{Malz2024Preparation}, if $U$ is a local circuit of depth $T$ on a bond geometry and $Z\in\mathcal A_X$, then $U^\dagger ZU\in\mathcal A_{\mathcal L_T(X)}$ (on the ring, within ring distance $T$ of $X$).
\end{lemma}
\begin{proof}
    \leanok
    By induction on $T$ using Lemma~\ref{lem:qc_light_cone_geometries}, it suffices to treat a single layer $W=HG$, where $G$ is the product of gates on bonds meeting $X$ and $H$ is the product of the remaining gates. Since $H\in\mathcal A_{X^c}$ is unitary, $[H,Z]=0$, so $W^\dagger ZW=G^\dagger ZG\in\mathcal A_{\mathcal N_1(X)}$.
\end{proof}

\begin{lemma}[No connected correlations beyond the light cone]
    \label{lem:qc_light_cone_correlations}
    \uses{lem:qc_light_cone}
    \lean{QuantumCircuit.expect_mul_eq_of_isPreparedInDepth,
        QuantumCircuit.expect_mul_mul_expect_one_of_isPreparedInDepth,
        QuantumCircuit.IsBondPreparedInDepth.expect_mul_eq,
        QuantumCircuit.IsBondPreparedInDepth.expect_mul_mul_expect_one,
        QuantumCircuit.expect_productVector_rectKronecker,
        QuantumCircuit.expect_productVector_mul}
    \leanok
    Following~\cite[Theorem~1]{Malz2024Preparation}, let $\ket\psi$ be prepared in depth $T$, and let $Z\in\mathcal A_X$, $Z'\in\mathcal A_{X'}$ with $\mathcal L_T(X)\cap\mathcal L_T(X')=\varnothing$ (e.g., $d_{\rm ring}(X,X')>2T$). Then
    \begin{align}
        \bra{\psi}ZZ'\ket{\psi}\braket{\psi}{\psi}
         & =\bra{\psi}Z\ket{\psi}\bra{\psi}Z'\ket{\psi},\notag
    \end{align}
    and $\bra{\psi}ZZ'\ket{\psi}=\bra{\psi}Z\ket{\psi}\bra{\psi}Z'\ket{\psi}$ when $\braket{\psi}{\psi}=1$. On any product vector $\ket\pi=\bigotimes_i\ket{v_i}$, $\bra\pi\bigotimes_i m_i\ket\pi=\prod_i\bra{v_i}m_i\ket{v_i}$ and $\bra\pi AB\ket\pi\braket\pi\pi=\bra\pi A\ket\pi\bra\pi B\ket\pi$ for $A,B$ on disjoint supports.
\end{lemma}
\begin{proof}
    \leanok
    Write $\ket\psi=U\ket\pi$ with $\ket\pi=\bigotimes_i\ket{v_i}$. By Lemma~\ref{lem:qc_light_cone}, $A=U^\dagger ZU\in\mathcal A_{\mathcal L_T(X)}$ and $B=U^\dagger Z'U\in\mathcal A_{\mathcal L_T(X')}$ have disjoint supports. For product generators $A=\bigotimes_i m_i$ and $B=\bigotimes_i m'_i$ with disjoint supports, at least one of $m_i,m'_i$ is $\mathbb 1$ at each site $i$, so $\bra\pi AB\ket\pi\braket\pi\pi=\prod_i\bra{v_i}m_im'_i\ket{v_i}\braket{v_i}{v_i}=\bra\pi A\ket\pi\bra\pi B\ket\pi$, which extends by bilinearity.
\end{proof}

\begin{lemma}[Inverses and series composition of local circuits]
    \label{lem:qc_circuit_inverse_composition}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.BondLayer.adjoint, QuantumCircuit.BondLayer.adjoint_partialOp,
        QuantumCircuit.BondLayer.adjoint_op, QuantumCircuit.circuitOp_append,
        QuantumCircuit.circuitOp_map_adjoint_reverse,
        QuantumCircuit.IsBondCircuitOfDepth.mem_unitary,
        QuantumCircuit.IsBondCircuitOfDepth.star,
        QuantumCircuit.IsBondCircuitOfDepth.mul}
    \leanok
    For a layer $W=\prod_k g_k$, replacing each gate $g_k$ by $g_k^\dagger$ on the same bond implements $W^\dagger$ (and likewise on any bond subset). If $U=W_T\cdots W_1$ has depth $T$ and $U'=W'_{T'}\cdots W'_1$ has depth $T'$, then $U$ is unitary, $U^\dagger=W_1^\dagger\cdots W_T^\dagger$ has depth $T$, and $U'U=W'_{T'}\cdots W'_1 W_T\cdots W_1$ has depth $T+T'$.
\end{lemma}
\begin{proof}
    \leanok
    Gates on disjoint bonds commute, so $(\prod_k g_k)^\dagger=\prod_k g_k^\dagger$, $(W_T\cdots W_1)^\dagger=W_1^\dagger\cdots W_T^\dagger$, and concatenating layer lists multiplies the circuit operators and adds their depths.
\end{proof}

\begin{lemma}[Preparation from the all-zero state]
    \label{lem:qc_preparation_from_zero}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.rectKronecker_mulVec_productVector,
        QuantumCircuit.finKronecker_mulVec_productVector,
        Matrix.exists_mem_unitary_eq_smul_mulVec_single_zero,
        QuantumCircuit.productVector_pi_smul,
        QuantumCircuit.exists_productVector_eq_smul_finKronecker_mulVec,
        QuantumCircuit.isLocalCircuitOfDepth_finKronecker,
        QuantumCircuit.IsPreparedInDepth.exists_eq_smul_mulVec_productVector_single_zero}
    \leanok
    Let $d\ge1$. On any finite site set, $(\bigotimes_i m_i)(\bigotimes_i\ket{v_i})=\bigotimes_i m_i\ket{v_i}$ and $\bigotimes_i c_i\ket{v_i}=(\prod_i c_i)\bigotimes_i\ket{v_i}$. Every nonzero $\ket v\in\C^d$ equals $c\,u\ket 0$ ($c=\|v\|\ne0$, $u\in\mathrm U(d)$), every nonzero product vector equals $c(\bigotimes_i u_i)\ket{0\cdots0}$, $\bigotimes_i u_i$ has depth $2$, and every nonzero vector prepared in depth $T$ equals $c\,U\ket{0\cdots0}$ for a local circuit $U$ of depth $T+2$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:qc_circuit_inverse_composition}
    Factoring $\ket{v_i}=\|v_i\|u_i\ket0$ gives $\bigotimes_i\ket{v_i}=(\prod_i\|v_i\|)(\bigotimes_i u_i)\ket{0\cdots0}$. The product $\bigotimes_i u_i$ is implemented in $2$ layers (applying $u_k\otimes u_{k+1}$ on odd bonds $\{k,k+1\}$ with $k<N$, then $u_N$ on $\{N\}$ when $N$ is odd), and Lemma~\ref{lem:qc_circuit_inverse_composition} gives depth $T+2$.
\end{proof}

\begin{lemma}[Operators placed on sets of sites]
    \label{lem:qc_placed_operators}
    \lean{QuantumCircuit.embedOp, QuantumCircuit.embedOp_embedOp,
        QuantumCircuit.embedOp_rectKronecker,
        QuantumCircuit.commute_embedOp_of_disjoint,
        QuantumCircuit.noncommProd_embedOp_apply}
    \leanok
    \uses{def:qc_local_circuit}
    For a map $e\colon J\to I$ between finite site sets and $X$ on $(\C^d)^{\otimes J}$, the \emph{placed operator} $\operatorname{embed}_e(X)$ has entries
    \begin{align}
        \operatorname{embed}_e(X)_{x,y}
         & =\begin{cases}X_{x\circ e,\,y\circ e},&\text{if }x|_{I\setminus e(J)}=y|_{I\setminus e(J)},\\0,&\text{otherwise.}\end{cases}\notag
    \end{align}
    For injective $e$ and $f\colon I\hookrightarrow K$, $\operatorname{embed}_e(\bigotimes_{j\in J}m_j)$ has factor $m_j$ at $e(j)$ and $\mathbb 1$ outside $e(J)$, and $\operatorname{embed}_f(\operatorname{embed}_e(X))=\operatorname{embed}_{f\circ e}(X)$. If $\{e_k\colon J_k\hookrightarrow I\}_{k\in K}$ have pairwise disjoint ranges, the operators $\operatorname{embed}_{e_k}(X_k)$ commute and
    \begin{align}
        \Bigl(\prod_{k\in K}\operatorname{embed}_{e_k}(X_k)\Bigr)_{x,y}
         & =\begin{cases}\prod_{k\in K}(X_k)_{x\circ e_k,\,y\circ e_k},&\text{if }x|_{I\setminus\bigcup_k e_k(J_k)}=y|_{I\setminus\bigcup_k e_k(J_k)},\\0,&\text{otherwise.}\end{cases}\notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    The composition rule follows from $x|_{(f\circ e)(J)^c}=y|_{(f\circ e)(J)^c}\iff x|_{f(I)^c}=y|_{f(I)^c}$ and $(x\circ f)|_{e(J)^c}=(y\circ f)|_{e(J)^c}$. Disjoint ranges give commuting supports, and induction on $K$ collapses the intermediate matrix-product sum to the unique configuration matching the output on the new range and the input elsewhere.
\end{proof}

\begin{lemma}[Operators supported on a set of sites]
    \label{thm:qc_supported_operator_placement}
    \lean{QuantumCircuit.exists_embedOp_eq_of_mem_supportedOperators}
    \leanok
    \uses{def:qc_local_circuit,lem:qc_placed_operators}
    For any injective $e\colon\{0,\ldots,m-1\}\to\{0,\ldots,n-1\}$ ($m,n,d\ge 0$),
    every $Z\in\operatorname{End}((\C^d)^{\otimes n})$ with
    $\operatorname{supp}(Z)\subseteq\operatorname{im}(e)$ satisfies
    $Z=\iota_e(X)$ for some $X\in\operatorname{End}((\C^d)^{\otimes m})$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:qc_local_circuit,lem:qc_placed_operators}
    Writing $Z$ as a linear combination of $\bigotimes_{i<n}A_i$ with $A_i=I$
    for $i\notin\operatorname{im}(e)$, Lemma~\ref{lem:qc_placed_operators} gives
    $\bigotimes_{i<n}A_i=\iota_e\bigl(\bigotimes_{j<m}A_{e(j)}\bigr)$, and
    linearity of $\iota_e$ concludes.
\end{proof}

\begin{lemma}[Products of matrices placed on disjoint windows]
    \label{lem:qc_placed_configurations}
    \lean{QuantumCircuit.placeCfg, QuantumCircuit.placeCfg_apply,
        QuantumCircuit.placeCfg_comp, QuantumCircuit.placeCfg_apply_of_notMem,
        QuantumCircuit.eq_placeCfg, QuantumCircuit.placeCfg_injective,
        QuantumCircuit.injective_window, QuantumCircuit.disjoint_range_window,
        QuantumCircuit.list_prod_embedOp_placeCfg, Matrix.mul_apply_extend}
    \leanok
    \uses{lem:qc_placed_operators}
    Let $d\ge1$, and let $W_1,\dots,W_n\colon\{1,\dots,m\}\to I$ be windows such that $(p,t)\mapsto W_p(t)$ is injective. For $c=(c_1,\dots,c_n)$ with $c_p\in\{0,\dots,d-1\}^m$, define $\operatorname{place}(c)\in\{0,\dots,d-1\}^I$ by $\operatorname{place}(c)(W_p(t))=c_p(t)$ and $0$ outside $\bigcup_p\operatorname{im}W_p$; then $\operatorname{place}$ is injective and fixes every configuration vanishing outside $\bigcup_p\operatorname{im}W_p$ upon restricting to $(W_p)_p$.
    \begin{enumerate}
        \item If $Y_p\ket{\iota_p(c_p)}=\sum_{e_p}V_p(e_p,c_p)\ket{o_p(e_p)}$ for injective maps $o_p$, then
              \begin{align}
                  \Bigl(\prod_{p=1}^n\operatorname{embed}_{W_p}(Y_p)\Bigr)\ket{\operatorname{place}(\iota_p(c_p))}
                   & =\sum_{e}\Bigl(\prod_{p=1}^n V_p(e_p,c_p)\Bigr)\ket{\operatorname{place}(o_p(e_p))}.\notag
              \end{align}
        \item For injective $R,R'$ into a basis, if $M\ket{x_0}=\sum_a f(a)\ket{R(a)}$ and $G\ket{R(a)}=\sum_b g(a,b)\ket{R'(b)}$, then $GM\ket{x_0}=\sum_b\bigl(\sum_a g(a,b)f(a)\bigr)\ket{R'(b)}$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    Item (1) follows from the disjoint-range product formula of Lemma~\ref{lem:qc_placed_operators}, since every output with nonzero amplitude vanishes outside $\bigcup_p\operatorname{im}W_p$. Item (2) is linearity of matrix multiplication.
\end{proof}

\begin{lemma}[Circuits inside sets of sites, in series and in parallel]
    \label{lem:qc_parallel_circuits}
    \lean{QuantumCircuit.BondLayer.IsIn, QuantumCircuit.IsBondCircuitOn,
        QuantumCircuit.IsCircuitOn, QuantumCircuit.IsBondCircuitOn.mul,
        QuantumCircuit.IsBondCircuitOn.finset_noncommProd,
        QuantumCircuit.IsCircuitOn.finset_noncommProd,
        QuantumCircuit.IsPairProduct.isCircuitOn,
        QuantumCircuit.isCircuitOn_list_prod_embedOp}
    \leanok
    \uses{def:qc_local_circuit, lem:qc_circuit_inverse_composition,
        lem:qc_placed_operators}
    A layer acts \emph{inside} $R\subseteq I$ if all its bonds lie in $R$, and $U$ is a \emph{local circuit of depth $T$ inside $R$} if $U=W_T\cdots W_1$ with each $W_t$ acting inside $R$.
    \begin{enumerate}
        \item Series composition inside $R$ adds depths: $T$ and $T'$ give $T+T'$.
        \item If all bonds are nonempty and $\{U_i\}_{i\in K}$ are commuting depth-$T$ circuits inside pairwise disjoint sets $R_i$, then $\prod_{i\in K}U_i$ has depth $T$ inside $\bigcup_{i\in K}R_i$.
        \item If $e_i\colon\{1,\dots,m_i\}\hookrightarrow\Z_N$ satisfy $e_i(j+1)=e_i(j)+1$ with pairwise disjoint ranges and each $X_i$ is a product of at most $K$ nearest-neighbour unitaries on $\{1,\dots,m_i\}$, then $\prod_i\operatorname{embed}_{e_i}(X_i)$ has depth $K$ inside $\bigcup_i\operatorname{im}e_i$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    Item (1) is layer concatenation (Lemma~\ref{lem:qc_circuit_inverse_composition}). For (2), merging the $t$-th layers across $i\in K$ forms a single layer inside $\bigcup_i R_i$ because nonempty bonds in disjoint $R_i$ are disjoint. For (3), placing each of the $K$ gates of $X_i$ by $e_i$ (padded with empty layers) gives depth $K$ inside $\operatorname{im}e_i$, and (2) merges them in parallel.
\end{proof}

\section{Two-site gates suffice}
\label{sec:qc_two_site_gates}

\begin{theorem}[Unitaries on a bounded number of sites]
    \label{thm:qc_bounded_unitary_two_site_gates}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.exists_isPairProduct,
        QuantumCircuit.isTwoLevelWord_of_det_eq_one,
        QuantumCircuit.exists_isPairProduct_ctrlOp,
        QuantumCircuit.isPairProduct_embedOp_pairSites}
    \leanok
    Realizing~\cite[caption of Fig.~1]{Malz2024Preparation}, for every $d\ge1$ and $n\ge2$ there exists $K$ such that every $U\in\mathrm U((\C^d)^{\otimes n})$ is a product of at most $K$ nearest-neighbour two-site unitaries on the open chain $\{1,\dots,n\}$.
\end{theorem}
\begin{proof}
    \leanok
    Factoring out a $d^n$-th root of $\det U$ (a $1$-site gate) reduces to $\det U=1$. Givens elimination decomposes $U$ into two-level operators $G_{ab}(g)$ ($g\in\mathrm{SU}(2)$ a rotation or diagonal phase on $\operatorname{span}\{\ket a,\ket b\}$). With $Q=\bigl(\begin{smallmatrix}0&1\\-1&0\end{smallmatrix}\bigr)$ and distinct $a,b,c$,
    \begin{align}
        G_{ab}(g) & =G_{cb}(Q)^\dagger\,G_{ac}(g)\,G_{cb}(Q);
        \label{eq:qc_two_level_quarter_turn}
    \end{align}
    choosing $c$ to reduce the Hamming distance $d_H(a,b)$ by $1$ reduces to $d_H(a,b)=1$, where $G_{ab}(g)=C_S(u)$ is a $1$-site gate $u=vwv^\dagger w^\dagger\in\mathrm{SU}(d)$ at $t$ controlled by $S\not\ni t$. For $s\neq t$,
    \begin{align}
        C_{S\cup\{s\}}(vwv^\dagger w^\dagger)
         & =C_S(v)\,C_{\{s\}}(w)\,C_S(v^\dagger)\,C_{\{s\}}(w^\dagger),
        \label{eq:qc_controlled_commutator}
    \end{align}
    which inductively reduces $|S|$ to $1$, and nearest-neighbour SWAP gates route any two-site gate $\{s,t\}$ to adjacent sites.
\end{proof}

\begin{definition}[Gates on windows of consecutive sites]
    \label{def:qc_window_gate}
    \lean{QuantumCircuit.windowSites, QuantumCircuit.IsWindowGate,
        QuantumCircuit.IsWindowProduct}
    \leanok
    \uses{def:qc_local_circuit}
    Modeling the sequential-RG gates of~\cite[paragraph ``The sequential-RG circuit'']{Malz2024Preparation}, a \emph{gate on a window of $m$ sites} on $\{1,\dots,n\}$ is a unitary supported on $\{p:a\le p<\min(a+m,n+1)\}$ for some $a$. An operator is a \emph{product of at most $K$ window-$m$ gates} if it factors into at most $K$ such gates.
\end{definition}

\begin{lemma}[Products of gates on windows]
    \label{lem:qc_window_products}
    \lean{QuantumCircuit.isWindowGate_embedOp, QuantumCircuit.isWindowGate_of_le,
        QuantumCircuit.IsWindowProduct.mul, QuantumCircuit.IsWindowProduct.embedOp,
        QuantumCircuit.IsWindowGate.isNeighbourGate,
        QuantumCircuit.IsWindowProduct.isPairProduct_two,
        QuantumCircuit.IsWindowProduct.exists_isPairProduct}
    \leanok
    \uses{def:qc_window_gate, lem:qc_placed_operators,
        thm:qc_bounded_unitary_two_site_gates}
    On the open chain of $n$ sites:
    \begin{enumerate}
        \item Any unitary placed on $l\le m$ consecutive sites (and any chain unitary when $n\le m$) is a window-$m$ gate.
        \item Products of $K$ and $K'$ window-$m$ gates multiply to at most $K+K'$ window-$m$ gates and are preserved under contiguous subchain placement.
        \item For $n\ge2$, a window-$2$ gate is a nearest-neighbour two-site gate.
        \item For $m\ge2$, there exists $C(d,m)$ such that for all $n\ge2$, any product of at most $K$ window-$m$ gates is a product of at most $KC(d,m)$ nearest-neighbour two-site gates.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    Items (1)--(2) follow from Lemma~\ref{lem:qc_placed_operators}. For (3), a window of length $2$ lies in $\{a',a'+1\}$ with $a'=\min(a,n-1)$. For (4), each gate lies in a contiguous window of length $\min(m,n)\ge2$, hence factors into at most $K_{\min(m,n)}$ nearest-neighbour gates by Theorem~\ref{thm:qc_bounded_unitary_two_site_gates}, so $C(d,m)=\sum_{l=2}^m K_l$ suffices.
\end{proof}
