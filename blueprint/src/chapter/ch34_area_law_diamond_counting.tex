\section{Exact lattice constants}

\begin{definition}[Lattice diamond]
    \label{def:area_lattice_diamond}
    \uses{def:area_lattice_displacement}
    \lean{TNLean.PEPS.AreaLaw.latticeDiamond,
        TNLean.PEPS.AreaLaw.mem_latticeDiamond}
    \leanok
    For $a\in\Z^2$ and $R\in\N$, define the lattice diamond
    $D_R(a)=\{x\in\Z^2:\ell(a,x)\le R\}$.
\end{definition}

\begin{theorem}[Exact lattice diamond count]
    \label{thm:area_diamond_count}
    \uses{def:area_lattice_diamond}
    \lean{TNLean.PEPS.AreaLaw.card_latticeDiamond,
        TNLean.PEPS.AreaLaw.card_le_diamond_of_latticeL1Distance_le,
        TNLean.PEPS.AreaLaw.card_le_diamond_of_walks,
        TNLean.PEPS.AreaLaw.card_le_diamond_of_edist_le}
    \leanok
    As in \cite[Section~2]{OpenAI2026AreaLaw},
    $|D_R(a)|=v_R=1+2R(R+1)$. If $\iota:V\to\Z^2$ is injective and
    $S\subseteq V$ is finite with $\ell(\iota(a),\iota(x))\le R$ for all
    $x\in S$, then $|S|\le v_R$. If every edge of a graph $G$ on $V$ has
    $\ell$-displacement at most $1$, the bound $|S|\le v_R$ also holds
    whenever each $x\in S$ is joined to $a$ by a $G$-walk of length at most
    $R$, or when $d_G(a,x)\le R$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:area_lattice_diamond,thm:area_walk_displacement}
    Translation reduces $a$ to $0$. Partitioning $D_R(0)$ into vertical
    fibers of size $2(R-|x|)+1$ ($x\in[-R,R]\cap\Z$) and inducting on $R$
    (increment $(2R+1)\cdot2+2=4R+4$) gives
    \begin{align}
        |D_R(0)| & =\sum_{x=-R}^R\bigl(2(R-|x|)+1\bigr)
        =1+2R(R+1).
        \label{eq:area_diamond_rows}
    \end{align}
    For injective $\iota$, $\iota(S)\subseteq D_R(\iota(a))$ gives
    $|S|\le v_R$, and Theorem~\ref{thm:area_walk_displacement} reduces the
    walk and $d_G(a,x)\le R$ hypotheses to $\ell(\iota(a),\iota(x))\le R$.
\end{proof}

\begin{theorem}[Exact lattice interaction weight bounds]
    \label{thm:area_diamond_budget}
    \uses{def:area_chain_weight}
    \lean{TNLean.PEPS.AreaLaw.card_support_le_diamond,
        TNLean.PEPS.AreaLaw.card_supports_containing_le_diamond,
        TNLean.PEPS.AreaLaw.sum_supportWeights_containing_le_diamond,
        TNLean.PEPS.AreaLaw.sum_supportNorms_containing_le_diamond,
        TNLean.PEPS.AreaLaw.interactionChainWeightSum_le_diamond_pow,
        TNLean.PEPS.AreaLaw.interactionChainWeightSum_norm_le_diamond_pow}
    \leanok
    Following \cite[Sections~2 and~4]{OpenAI2026AreaLaw}, let $V$ be a finite
    graph embedded injectively into $\Z^2$ with edge displacement at most $1$,
    and let $\mathcal F$ be a finite family of distinct supports $X\subseteq V$
    of $V$-walk diameter at most $R\in\N$. With $v_R=1+2R(R+1)$ and
    $\mu_R=2^{v_R-1}$, every $X\in\mathcal F$ and $a\in V$ satisfy
    $|X|\le v_R$ and $|\{X\in\mathcal F:a\in X\}|\le\mu_R$. For $J\ge0$ and
    weights $w_X\le J$ (in particular $w_X=\lVert h_X\rVert\le J$),
    \begin{align}
        \sum_{X\in\mathcal F:\,a\in X}w_X & \le\mu_RJ,
        \label{eq:area_diamond_site}
    \end{align}
    and if $0\le w_X\le J$ for all $X\in\mathcal F$, then for every
    $X\in\mathcal F$ and $n\in\N$,
    \begin{align}
        W_n(X) & \le(v_R\mu_RJ)^n.
        \label{eq:area_diamond_chain}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_diamond_count,thm:area_support_count,
        thm:area_chain_growth}
    Every nonempty $X\in\mathcal F$ lies in a radius-$R$ ball around any
    $a\in X$, so $|X|\le v_R$ by Theorem~\ref{thm:area_diamond_count}, and
    $X\mapsto X\setminus\{a\}$ injects $\{X\in\mathcal F:a\in X\}$ into the
    power set of at most $v_R-1$ sites, giving
    $|\{X\in\mathcal F:a\in X\}|\le\mu_R$ by
    Theorem~\ref{thm:area_support_count}. Bounding $w_X\le J$ yields
    (\ref{eq:area_diamond_site}), and Theorem~\ref{thm:area_chain_growth}
    with $v=v_R$ and $b=\mu_RJ$ yields~(\ref{eq:area_diamond_chain}).
\end{proof}

\section{Chains reaching a target}

Let $\mathcal F$ be a finite family of distinct finite supports in a graph $G$
on $V$, and let $w_X\in\R$.

\begin{definition}[Chain weight at a target]
    \label{def:area_target_chain}
    \lean{TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum}
    \leanok
    For $y\in V$ and finite $X\subseteq V$, define recursively
    \begin{align}
        T_0^y(X)     & =\mathbf1_{\{y\in X\}},
        \label{eq:area_target_initial}         \\
        T_{n+1}^y(X) & =
        \sum_{Y\in\mathcal F:\,Y\cap X\ne\varnothing}w_YT_n^y(Y).
        \label{eq:area_target_step}
    \end{align}
\end{definition}

\begin{theorem}[Nonnegative target weights]
    \label{thm:area_target_domination}
    \uses{def:area_target_chain,def:area_chain_weight}
    \lean{TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_nonneg,
        TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_le}
    \leanok
    If $w_Y\ge0$ for all $Y\in\mathcal F$, then $0\le T_n^y(X)\le W_n(X)$ for
    every finite $X\subseteq V$ and $n\ge0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:area_target_chain,def:area_chain_weight}
    By induction on $n$ from $T_0^y(X)=\mathbf1_{\{y\in X\}}\in\{0,1\}$,
    (\ref{eq:area_target_step}), and~(\ref{eq:area_chain_step}) since
    $w_Y\ge0$.
\end{proof}

\begin{theorem}[Exact exclusion of distant targets]
    \label{thm:area_target_zero}
    \uses{def:area_target_chain}
    \lean{TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_eq_zero_of_no_walk,
        TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_eq_zero_of_lt_edist,
        TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_eq_zero_of_edist_eq_top}
    \leanok
    Suppose every pair of sites in each $X\in\mathcal F$ is joined by a
    $G$-walk of length at most $R\ge0$. For $X\in\mathcal F$, $a\in X$, and
    any $w_Y\in\R$, $T_n^y(X)=0$ whenever no walk of length at most $(n+1)R$
    joins $a$ to $y$, and in particular whenever $(n+1)R<d_G(a,y)$ (including
    $d_G(a,y)=\infty$).
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:area_target_chain}
    Induct on $n$: at $n=0$, $y\in X$ implies $d_G(a,y)\le R$; for the step,
    if $z\in Y\cap X$ and a walk of length $\le(n+1)R$ joins $z$ to $y$,
    concatenating with a walk of length $\le R$ from $a$ to $z$ gives a walk of
    length $\le(n+2)R$, so otherwise $T_n^y(Y)=0$ for all $Y\cap X\ne\varnothing$
    in~(\ref{eq:area_target_step}).
\end{proof}

\begin{theorem}[Graph-distance indicator bound]
    \label{thm:area_target_indicator}
    \uses{def:area_target_chain}
    \lean{TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_le_indicator_pow}
    \leanok
    Under the range hypothesis of Theorem~\ref{thm:area_target_zero}, suppose
    $w_Y\ge0$, $|Y|\le v\in\N$ for all $Y\in\mathcal F$, and
    $\sum_{Y\in\mathcal F:\,x\in Y}w_Y\le b$ for all $x\in V$ ($b\ge0$). For
    $X\in\mathcal F$, $a\in X$, and $n\ge0$,
    \begin{align}
        T_n^y(X) & \le(vb)^n\mathbf1_{\{d_G(a,y)\le(n+1)R\}},
        \label{eq:area_target_indicator}
    \end{align}
    which holds with $v=v_R$ and $b=\mu_RJ$ in the setting of
    Theorem~\ref{thm:area_diamond_budget}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_target_domination,thm:area_target_zero,
        thm:area_chain_growth}
    Combine $T_n^y(X)\le W_n(X)\le(vb)^n$ (Theorems~\ref{thm:area_target_domination}
    and~\ref{thm:area_chain_growth}) when $d_G(a,y)\le(n+1)R$ with
    $T_n^y(X)=0$ (Theorem~\ref{thm:area_target_zero}) when
    $d_G(a,y)>(n+1)R$.
\end{proof}

\section{Exponential generating series}

\begin{theorem}[Exponential coefficient bound]
    \label{thm:area_chain_exponential}
    \uses{def:area_target_chain}
    \lean{TNLean.PEPS.AreaLaw.interactionChainTargetWeightSum_le_exponential}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:area_target_indicator}, for
    $d_G(a,y)=D\in\N$, $\mu\ge0$, and $\kappa=vb$,
    \begin{align}
        T_n^y(X) & \le e^{\mu R-\mu D}(\kappa e^{\mu R})^n.
        \label{eq:area_chain_exponential}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_target_domination,thm:area_chain_growth,thm:area_target_zero}
    Since $e^{\mu R-\mu D}(\kappa e^{\mu R})^n=\kappa^n e^{\mu((n+1)R-D)}$,
    the bound follows from $T_n^y(X)\le W_n(X)\le\kappa^n\le\kappa^n e^{\mu((n+1)R-D)}$
    for $D\le(n+1)R$ and $T_n^y(X)=0\le\kappa^n e^{\mu((n+1)R-D)}$ for
    $D>(n+1)R$.
\end{proof}

\begin{definition}[Interaction-chain series]
    \label{def:area_chain_series}
    \uses{def:area_target_chain}
    \lean{TNLean.PEPS.AreaLaw.interactionChainPropagationSeries}
    \leanok
    For $t\in\R$, define the exponential generating series
    \begin{align}
        S_t^y(X) & =\sum_{n=0}^{\infty}\frac{(2|t|)^n}{n!}T_n^y(X).
        \label{eq:area_chain_series}
    \end{align}
\end{definition}

\begin{theorem}[Summability and spatial decay]
    \label{thm:area_chain_series_bound}
    \uses{def:area_chain_series}
    \lean{TNLean.PEPS.AreaLaw.interactionChainPropagationSeries_summable_and_le}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:area_chain_exponential}, the
    series~(\ref{eq:area_chain_series}) from the proof of
    \cite[Lemma~4.1]{OpenAI2026AreaLaw} converges and satisfies
    \begin{align}
        S_t^y(X) & \le e^{\mu R-\mu D}
        \exp\!\bigl(2|t|\,vb\,e^{\mu R}\bigr).
        \label{eq:area_chain_series_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_chain_exponential,thm:area_target_domination,
        def:area_chain_series}
    With $C=e^{\mu R-\mu D}$ and $A=vb\,e^{\mu R}$,
    (\ref{eq:area_chain_exponential}) gives
    \begin{align}
        0\le\frac{(2|t|)^n}{n!}T_n^y(X)
         & \le C\frac{(2|t|A)^n}{n!},
        \label{eq:area_series_comparison}
    \end{align}
    and summing~(\ref{eq:area_series_comparison}) over $n\ge0$ yields
    summability and~(\ref{eq:area_chain_series_bound}).
\end{proof}

\begin{theorem}[Series vanishing between components]
    \label{thm:area_chain_series_zero}
    \uses{def:area_chain_series}
    \lean{TNLean.PEPS.AreaLaw.interactionChainPropagationSeries_eq_zero_of_edist_eq_top}
    \leanok
    Under the range hypothesis of Theorem~\ref{thm:area_target_zero}, if
    $d_G(a,y)=\infty$, then $S_t^y(X)=0$ for all $t\in\R$ and $w_Y\in\R$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_target_zero,def:area_chain_series}
    By Theorem~\ref{thm:area_target_zero}, $T_n^y(X)=0$ for all $n\ge0$ in
    (\ref{eq:area_chain_series}).
\end{proof}
