\section{Primary birth regions and adjacent fine meshes}
\label{sec:area_law_primary_regions}

Retain the half-open cells $C_{k,z}$, the layer $D_k$, and its finite
index set $I_k$ from Definition~\ref{def:area_law_dyadic_layers}, with
arbitrary origin $o\in\R^2$, finite endpoint set $Z\subset\Z^2$, and
neighborhood radius $C_0\in\N$. All distances and diameters use the sup metric
on $\R^2$. The primary regions in \cite[Section~11]{OpenAI2026AreaLaw} are
formed by intersecting an open pitch interior with $D_k$ and taking its
closure, assigning all fragments in one pitch interior the same primary
identifier even when disconnected.

\begin{definition}[Pitch interiors and primary birth regions]
    \label{def:area_law_primary_regions}
    \lean{TNLean.PEPS.AreaLaw.Geometry.pitchInterior,
        TNLean.PEPS.AreaLaw.Geometry.primaryFragment,
        TNLean.PEPS.AreaLaw.Geometry.primaryFragmentIndices,
        TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion}
    \leanok
    \uses{def:area_law_dyadic_cells,def:area_law_dyadic_layers}
    Fix $k,\ell,p\in\N$, put $r=2^k$, $t=2^\ell$, $s=2^p$, and choose
    $a,b\in\Z$ (in the geometric construction, $\ell=\ell_k$, $p=p_k$, and
    $a,b$ represent the selected residue classes modulo $s/t$). For every
    $j\in\Z^2$, define the open pitch interior $U_j$, closed cell fragments
    $F_{j,z}$, active index set $J_j$, and primary birth region $R_j$ (allowing
    empty interiors and fragments) by
    \begin{align}
        v_j     & =o+t(a,b)+sj,\notag                                \\
        U_j     & =(v_{j,1}+t,v_{j,1}+s)
        \times(v_{j,2}+t,v_{j,2}+s),\notag                           \\
        F_{j,z} & =\overline{C_{k,z}\cap U_j},\notag                 \\
        J_j     & =\{z\in I_k:C_{k,z}\cap U_j\ne\varnothing\},\notag \\
        R_j     & =\overline{D_k\cap U_j}.
        \label{eq:area_law_primary_regions}
    \end{align}
    In the picture below, with $s=4t$ and $r=2t$, the open pitch interiors
    $U_j$ and $U_{j'}$, for $j'=j+(1,0)$, share one vertical interval and
    are separated by the vertical strip of width $t$ between them, and each
    scale-$r$ cell of the layer
    contributes one closed fragment to the primary region of each pitch
    interior it meets.

\end{definition}

\begin{lemma}[Finite rectangular decomposition]
    \label{thm:area_law_primary_fragment_union}
    \lean{TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_eq_iUnion_primaryFragment}
    \leanok
    \uses{def:area_law_primary_regions}
    For the data of Definition~\ref{def:area_law_primary_regions},
    $R_j=\bigcup_{z\in J_j}F_{j,z}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dyadic_nesting}
    Intersecting $D_k=\bigcup_{z\in I_k}C_{k,z}$
    (Lemma~\ref{thm:area_law_dyadic_nesting}) with $U_j$, commuting closure
    with the finite union, and dropping the empty terms $z\notin J_j$ gives
    $R_j=\bigcup_{z\in J_j}F_{j,z}$.
\end{proof}

\begin{lemma}[Closed cell fragments]
    \label{thm:area_law_primary_fragment_rectangles}
    \lean{TNLean.PEPS.AreaLaw.Geometry.primaryFragment_eq_closedRectangle,
        TNLean.PEPS.AreaLaw.Geometry.primaryFragment_subset_closure_dyadicCell,
        TNLean.PEPS.AreaLaw.Geometry.primaryFragment_dist_le,
        TNLean.PEPS.AreaLaw.Geometry.primaryFragment_diam_le}
    \leanok
    \uses{def:area_law_primary_regions}
    Write $\alpha_{i,z}=o_i+rz_i$ and $\beta_{i,z}=o_i+r(z_i+1)$ for
    $i\in\{1,2\}$. If $C_{k,z}\cap U_j\ne\varnothing$, then both coordinate
    intervals below have strictly positive length and
    \begin{align}
        F_{j,z}
         & =\prod_{i=1}^2
        [\max\{\alpha_{i,z},v_{j,i}+t\},
        \min\{\beta_{i,z},v_{j,i}+s\}].
        \label{eq:area_law_primary_fragment_rect}
    \end{align}
    For every fragment $F_{j,z}$ (including empty ones) and all $x,y\in F_{j,z}$,
    $F_{j,z}\subseteq\overline{C_{k,z}}$, $\|x-y\|_\infty\le r$, and
    $\operatorname{diam}_\infty F_{j,z}\le r$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dyadic_cell_distance}
    Any point of $C_{k,z}\cap U_j$ satisfies
    $\max\{\alpha_{i,z},v_{j,i}+t\}<x_i<\min\{\beta_{i,z},v_{j,i}+s\}$ for
    $i=1,2$, so taking closures of the coordinate intersections gives
    (\ref{eq:area_law_primary_fragment_rect}). Monotonicity of closure gives
    $F_{j,z}\subseteq\overline{C_{k,z}}$, and
    Lemma~\ref{thm:area_law_dyadic_cell_distance} at index radius $0$ gives
    $\|x-y\|_\infty\le r$ and $\operatorname{diam}_\infty F_{j,z}\le r$.
\end{proof}

\begin{lemma}[Primary diameter and separation]
    \label{thm:area_law_primary_diameter_separation}
    \lean{TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_subset_closure_pitchInterior,
        TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_dist_le,
        TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_diam_le,
        TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_dist_separation}
    \leanok
    \uses{def:area_law_primary_regions}
    For every $j\in\Z^2$ and all $x,y\in R_j$, one has
    $R_j\subseteq\overline{U_j}$, $\|x-y\|_\infty\le s$, and
    $\operatorname{diam}_\infty R_j\le s$. For distinct $i\ne j$ in $\Z^2$,
    every $x\in R_i$ and $y\in R_j$ satisfy $\|x-y\|_\infty\ge t$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:area_law_primary_regions}
    By (\ref{eq:area_law_primary_regions}),
    $R_j\subseteq\overline{U_j}\subseteq\prod_{q=1}^2[v_{j,q}+t,v_{j,q}+s]$,
    giving $\|x-y\|_\infty\le s$ and $\operatorname{diam}_\infty R_j\le s$. If
    $i_q<j_q$, then $v_{i,q}+s\le v_{j,q}=(v_{j,q}+t)-t$, separating the $q$th
    coordinates of $R_i$ and $R_j$ by at least $t$.
\end{proof}

\begin{lemma}[Separation of primary births in layers of the same parity]
    \label{thm:area_law_primary_same_parity_separation}
    \lean{TNLean.PEPS.AreaLaw.Geometry.primaryBirthRegion_dist_separation_of_same_parity}
    \leanok
    \uses{def:area_law_primary_regions,def:area_law_fine_belt_scales}
    With the scales $r_k,t_k,s_k$ of
    Definition~\ref{def:area_law_fine_belt_scales} and $C_0\ge2$, choose
    shifts $a_k,b_k\in\Z$ for each layer $k$ and write $R_{k,i}$ for the
    resulting primary birth region of
    Definition~\ref{def:area_law_primary_regions}. If $k\equiv h\pmod2$ and
    $(k,i)\ne(h,j)$, then every $x\in R_{k,i}$ and $y\in R_{h,j}$ satisfy
    $\|x-y\|_\infty\ge\min\{t_k,r_k/2\}$ (an auxiliary separation estimate for
    \cite[Section~11]{OpenAI2026AreaLaw}).
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:area_law_primary_diameter_separation,
        thm:area_law_dyadic_nonadjacent,def:area_law_primary_regions}
    If $k=h$, then $i\ne j$ and
    Lemma~\ref{thm:area_law_primary_diameter_separation} gives
    $\|x-y\|_\infty\ge t_k$. If $k\ne h$, then $|k-h|\ge2$, and since
    $R_{k,i}\subseteq\overline{D_k}$ and $R_{h,j}\subseteq\overline{D_h}$,
    Lemma~\ref{thm:area_law_dyadic_nonadjacent} with $C_0\ge2$ gives
    $\|x-y\|_\infty\ge(C_0-1)\max\{r_k,r_h\}/2\ge r_k/2$.
\end{proof}

\begin{lemma}[Number of fragments of one primary]
    \label{thm:area_law_primary_fragment_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.card_primaryFragmentIndices_le}
    \leanok
    \uses{def:area_law_primary_regions}
    If $k\le p$, then for every $j\in\Z^2$ (independently of $Z,C_0,o,a,b$),
    \begin{align}
        |J_j| & \le\bigl(2^{p-k}+2\bigr)^2
        =\bigl(s/r+2\bigr)^2.
        \label{eq:area_law_primary_fragment_count}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:area_law_primary_regions}
    With $N=2^{p-k}=s/r$ and $q_i=\lfloor(v_{j,i}-o_i)/r\rfloor$, any
    $z\in J_j$ satisfies $q_i\le z_i\le q_i+N$ for $i=1,2$, so
    $|J_j|\le(N+1)^2\le(N+2)^2$.
\end{proof}

\begin{lemma}[Adjacent fine meshes]
    \label{thm:area_law_adjacent_fine_meshes}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineScaleIndex_mono,
        TNLean.PEPS.AreaLaw.Geometry.fineScaleIndex_succ,
        TNLean.PEPS.AreaLaw.Geometry.fineScale_side_succ}
    \leanok
    \uses{def:area_law_fine_belt_scales}
    The fine exponents $\ell_k=\lfloor\zeta k\rfloor$ of
    Definition~\ref{def:area_law_fine_belt_scales} are nondecreasing, and for
    every $k\in\N$, $\ell_{k+1}\in\{\ell_k,\ell_k+1\}$ and
    $t_{k+1}\in\{t_k,2t_k\}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_exponent_gaps}
    By Lemma~\ref{thm:area_law_exponent_gaps}, $0\le\zeta\le1$, so
    $\zeta k\le\zeta(k+1)\le\zeta k+1$; taking floors gives
    $\ell_k\le\ell_{k+1}\le\ell_k+1$, and exponentiating gives
    $t_{k+1}=2^{\ell_{k+1}}\in\{t_k,2t_k\}$.
\end{proof}
