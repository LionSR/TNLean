\section{Fundamental Theorem for normal matrix product states}%
\label{sec:peps_normal_mps}

The closed chain is the one-dimensional PEPS: the cycle graph on $n$ vertices
carrying one tensor of $D \times D$ matrices per site, so that a physical
configuration $\sigma = (i_1, \ldots, i_n)$ receives the amplitude
$\tr(A^{i_1} \cdots A^{i_n})$. The Fundamental Theorem for normal PEPS
therefore specializes to matrix product states once its blocking hypotheses are
read off the chain, yielding the matrix-product-state corollaries of
\cite[Section~4]{Molnar2018NormalPEPS} and the sharper system-size bound in its
closing part \emph{Normal MPS: alternative proof}.

The development proceeds in four steps. The closed chain is first identified
with a cycle of tensors, and the normal blocking hypotheses are obtained from
the injectivity of every arc of $L$ consecutive sites, giving the Fundamental
Theorem for normal matrix product states on a ring of $n \ge 3L$ sites with one
invertible gauge per bond. The translation-invariant case then collapses the
per-bond gauges to one gauge $Z$ and one scalar $\lambda$, with the length-$n$
trace identity forcing $\lambda^n = 1$.

For the sharper bound, the matrix-product-state form of
\cite[Lemma~5]{Molnar2018NormalPEPS} compares the $L + 1$ overlapping
windows around a fixed bond. If $C_j$ is the deformation of the $j$-th
window, then for length-$L$ words $a$ and $b$ the family of
consecutive-window trace identities strips to
$C_0(a)\,B^b = B^a\,C_L(b)$, and hence to
$C_0(a) = B^a X$ and $C_L(b) = X B^b$.
This is the step that lowers the system size from $n \ge 3L$ to
$n \ge 2L + 1$. Finally the same argument is run for site-dependent
chains, one tensor per site, which is the setting of
\cite[Lemma~5]{Molnar2018NormalPEPS}.

\subsection{Matrix product states on the closed chain}

\begin{lemma}[Cycle blocking hypotheses from arc injectivity]%
    \label{thm:peps_cycleNormalPEPSBlockingHypotheses}
    \lean{TNLean.PEPS.cycleArcFrom}
    \lean{TNLean.PEPS.NormalCycleArcInjectivityHypotheses}
    \lean{TNLean.PEPS.cycleEdgeBlockingData}
    \lean{TNLean.PEPS.cycleOneSiteSeparation}
    \lean{TNLean.PEPS.cycleNormalPEPSBlockingHypotheses}
    \lean{TNLean.PEPS.isCrossingEdge_cycleEdge_iff}
    \leanok
    \uses{def:peps_normalPEPSBlockingHypotheses, def:peps_regionInjectivityData}
    On the cycle graph of $n$ vertices, let $A(v,\ell)=\{v,v+1,\ldots,v+\ell-1\}\pmod n$
    be the clockwise arc of length $\ell$ from $v$. Suppose $0<L$, $3L\le n$,
    $\inj\bigl(A(v,L)\bigr)$ holds for all $v$, and $\inj$ satisfies union
    closure (Definition~\ref{def:peps_regionInjectivityData}). Then the cycle
    satisfies Definition~\ref{def:peps_normalPEPSBlockingHypotheses}:
    \begin{enumerate}
        \item Around each edge $e=(v,v+1)$, the blocks
              $R=A(v-L+1,L)$, $B=A(v+1,L)$, and $C=A(v+L+1,n-2L)$ form a
              partition $V=R\sqcup B\sqcup C$ with $\inj(R),\inj(B),\inj(C)$
              and unique crossing edge $\partial(R,B)=\{e\}$.
        \item At each site $v$, the regions $S_v=A(v,L+1)$ and
              $T_v=A(v+1,L)$ satisfy $T_v\subseteq S_v$, $S_v\setminus T_v=\{v\}$,
              and $\inj(S_v),\inj(T_v),\inj(S_v^c),\inj(T_v^c)$.
    \end{enumerate}
\end{lemma}
\begin{proof}\leanok
    Disjointness, covering, and $\partial(R,B)=\{e\}$ follow from cyclic-distance
    arithmetic since $n-2L\ge L>0$. Every arc $A(v,\ell)$ with $L\le\ell<n$ is a
    union of length-$L$ arcs, hence satisfies $\inj\bigl(A(v,\ell)\bigr)$ by
    union closure; this covers $C$, $S_v$, $S_v^c=A(v+L+1,n-L-1)$, and
    $T_v^c=A(v+L+1,n-L)$.
\end{proof}

\begin{definition}[Normal MPS on a closed chain]%
    \label{def:peps_normalCycleArcMPS}
    \lean{TNLean.PEPS.IsNormalCycleArcMPS}
    \leanok
    \uses{thm:peps_cycleNormalPEPSBlockingHypotheses, def:peps_regionInjectivityDataOf}
    A tensor $A$ on the $n$-site cycle graph is a \emph{normal MPS with
        blocking length $L$} when $\inj_A\bigl(A(v,L)\bigr)$ holds for every site
    $v$ and $D_e>0$ for every bond $e$
    \cite[Section~4]{Molnar2018NormalPEPS}
    (\cite{gap:peps_injective_ft_section3_route}).
\end{definition}

\begin{corollary}[Fundamental Theorem for normal MPS on a closed chain]%
    \label{cor:fundamentalTheorem_normalMPS_cycle}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_cycle}
    \leanok
    \uses{def:peps_normalCycleArcMPS, def:GaugeEquivPEPS}
    Let $A$ and $B$ be normal MPS with blocking length $L>0$
    (Definition~\ref{def:peps_normalCycleArcMPS}) on the $n$-site cycle with
    $n\ge 3L$ and common physical dimension $d>0$. If $\Coeff_A=\Coeff_B$, then
    $A$ and $B$ are related by a scalar-free local gauge
    (Definition~\ref{def:GaugeEquivPEPS}): there exist invertible bond matrices
    $(Z_i)_{i=1}^n$ ($Z_{n+1}\equiv Z_1$) such that
    \begin{align}
        B_i = Z_i^{-1} A_i Z_{i+1}
        \qquad (1\le i\le n),
        \label{eq:peps_normal_mps_cycle_gauge}
    \end{align}
    giving the first corollary of \cite[Section~4]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{thm:peps_cycleNormalPEPSBlockingHypotheses, thm:fundamentalTheorem_normalPEPS_connected, lem:peps_injectiveRegion_union_pos}
    Since $D_e>0$, Lemma~\ref{lem:peps_injectiveRegion_union_pos} gives union
    closure for $\inj_A\wedge\inj_B$.
    Lemma~\ref{thm:peps_cycleNormalPEPSBlockingHypotheses} then supplies the
    normal blocking hypotheses with unique crossings on the connected cycle, and
    Corollary~\ref{thm:fundamentalTheorem_normalPEPS_connected} yields
    (\ref{eq:peps_normal_mps_cycle_gauge}).
\end{proof}

\begin{corollary}[Uniqueness of the closed-chain gauges up to a constant]%
    \label{cor:fundamentalTheorem_normalMPS_cycle_gauge_unique}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_cycle_gauge_unique}
    \leanok
    \uses{def:peps_normalCycleArcMPS, def:gaugeVertex}
    In the setting of Corollary~\ref{cor:fundamentalTheorem_normalMPS_cycle}, if
    $A$ and $B$ are normal MPS with blocking length $L>0$ ($n\ge 3L$) and equal
    bond dimensions, any two invertible bond-gauge families $Z,Z'$ satisfying
    (\ref{eq:peps_normal_mps_cycle_gauge}) satisfy $Z'_i=c_i Z_i$ for nonzero
    scalars $c_i\in\C^\times$ \cite[Section~4]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{thm:peps_cycleNormalPEPSBlockingHypotheses, thm:fundamentalTheorem_normalPEPS_gauge_unique, lem:peps_injectiveRegion_union_pos}
    Combine Lemma~\ref{lem:peps_injectiveRegion_union_pos} and
    Lemma~\ref{thm:peps_cycleNormalPEPSBlockingHypotheses} with the per-edge
    uniqueness Lemma~\ref{thm:fundamentalTheorem_normalPEPS_gauge_unique}.
\end{proof}

\begin{definition}[Cycle tensor of a matrix tensor]%
    \label{def:cycleTensorOfMPS}
    \lean{TNLean.PEPS.cycleSuccEdge}
    \lean{TNLean.PEPS.cycleLeftIncident}
    \lean{TNLean.PEPS.cycleRightIncident}
    \lean{TNLean.PEPS.cycleTensorOfMPS}
    \leanok
    \uses{def:peps_tensor, def:mps_tensor}
    On the $n$-site cycle ($n\ge 3$), each site $v$ has incident bonds
    $e_{v-1}=(v-1,v)$ and $e_v=(v,v+1)$. The \emph{cycle tensor} of an MPS
    tensor $A=(A^i)_{i=0}^{d-1}\in\MN{D}^d$ places $A$ at every site $v$ with
    bond dimension $D$ on every edge, reading the row index of $A^i$ on
    $e_{v-1}$ and the column index on $e_v$.
\end{definition}

\begin{lemma}[State coefficients of the cycle tensor]%
    \label{lem:stateCoeff_cycleTensorOfMPS}
    \lean{MPSTensor.evalWord_ofFn_apply}
    \lean{TNLean.PEPS.stateCoeff_cycleTensorOfMPS}
    \leanok

    For $n \ge 3$ and $\sigma = (i_1, \ldots, i_n)$, the state coefficient of
    the cycle tensor of $A$ (Definition~\ref{def:cycleTensorOfMPS}) is
    \begin{align}
        \Coeff_A(\sigma) = V^{(n)}(A)_\sigma = \tr\!(A^{i_1} \cdots A^{i_n}).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:trace_eval_word_eq_sum_cyclic}
    By Lemma~\ref{thm:trace_eval_word_eq_sum_cyclic},
    $\tr(A^{i_1}\cdots A^{i_n})=\sum_{\alpha}\prod_{v=1}^n (A^{i_v})_{\alpha_{v-1},\alpha_v}$,
    which matches the cycle contraction after identifying $\alpha_v$ with the
    bond label on $e_v$.
\end{proof}

\begin{lemma}[Arc injectivity of the cycle tensor from block injectivity]%
    \label{lem:regionBlockedTensorInjective_cycleTensorOfMPS}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_cycleArcFrom_iff}
    \lean{TNLean.PEPS.regionBlockedWeight_cycleTensorOfMPS}
    \lean{TNLean.PEPS.matrix_eq_zero_of_isNBlkInjective}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_cycleTensorOfMPS}
    \leanok
    \uses{def:cycleTensorOfMPS, def:l_blk_injective, def:peps_regionInjectivityDataOf}
    If $0 < L < n$, $n \ge 3$, $D>0$, and $A$ is $L$-block injective
    (Definition~\ref{def:l_blk_injective}), then the cycle tensor of $A$
    satisfies $\inj_A\bigl(A(v,L)\bigr)$ for every site $v$.
\end{lemma}
\begin{proof}\leanok
    The arc $A(v,L)$ has boundary $\partial A(v,L)=\{e_{v-1},e_{v+L-1}\}$ and
    $n-(L+1)$ exterior bonds, so its blocked weight at boundary indices $(a,b)$
    and word $x\in\{0,\ldots,d-1\}^L$ is
    $T_{A,A(v,L)}^{(a,b)}(x)=D^{\,n-(L+1)}(A^{x_1}\cdots A^{x_L})_{ab}$.
    If $\sum_{a,b}C_{ab}T_{A,A(v,L)}^{(a,b)}(x)=0$ for all $x$, then
    $\tr(C^{\mathsf T}A^{x_1}\cdots A^{x_L})=0$; since $\{A^{x_1}\cdots A^{x_L}\}$
    spans $\MN{D}$, nondegeneracy of the trace gives $C=0$.
\end{proof}

\subsection{Translation-invariant matrix product states}

When the same tensor sits at every site, if $B^i = \lambda\,Z^{-1}A^iZ$ for all
$i$, then for $\sigma=(i_1,\ldots,i_n)$,
\begin{align}
    \tr(B^{i_1}\cdots B^{i_n})
     & = \lambda^n\,\tr(Z^{-1}A^{i_1}\cdots A^{i_n}Z)
    = \lambda^n\,\tr(A^{i_1}\cdots A^{i_n}),
    \notag
\end{align}
so $V^{(n)}(A)=V^{(n)}(B)$ forces $\lambda^n = 1$, capturing the freedom in the
translation-invariant corollary of \cite[Section~4]{Molnar2018NormalPEPS}.

\begin{corollary}[Fundamental Theorem for translation-invariant normal MPS
        on a closed chain, matrix form]%
    \label{cor:fundamentalTheorem_normalMPS}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS}
    \leanok
    \uses{def:l_blk_injective, def:mpv}
    Let $0 < L$, $2L + 1 \le n$, $D>0$, and let $A,B\in\MN{D}^d$ be $L$-block
    injective with $V^{(n)}(A)=V^{(n)}(B)$. Then there exist invertible matrices
    $(Z_v)_{v=1}^n$ ($Z_{n+1}\equiv Z_1$) such that
    \begin{align}
        B^i = Z_v^{-1}\, A^i\, Z_{v+1}
        \qquad (1\le v\le n,\ 0\le i<d),
        \label{eq:peps_normal_mps_per_bond}
    \end{align}
    specializing the first corollary of \cite[Section~4]{Molnar2018NormalPEPS}
    at the optimal bound $n\ge 2L+1$.
\end{corollary}
\begin{proof}\leanok
    \uses{cor:fundamentalTheorem_normalMPS_of_overlap}
    Immediate from Corollary~\ref{cor:fundamentalTheorem_normalMPS_of_overlap}.
\end{proof}

\begin{lemma}[Two per-bond gauge families are proportional at every bond]%
    \label{lem:gaugeFamily_bond_proportional}
    \lean{TNLean.PEPS.gaugeFamily_bond_proportional}
    \leanok
    \uses{def:l_blk_injective, def:eval_word}
    Let $n\ge 1$, let $A,B\in\MN{D}^d$ with $A$ being $L$-block injective, and
    let $(Z_v)_{v=1}^n,(Z'_v)_{v=1}^n$ ($Z_{n+1}\equiv Z_1$, $Z'_{n+1}\equiv Z'_1$)
    be invertible matrices satisfying $B^i = Z_v^{-1} A^i Z_{v+1} = Z'^{-1}_v A^i Z'_{v+1}$
    for all $v,i$. Then for each $v$ there exists $c_v\in\C^\times$ such that
    $Z'_v = c_v Z_v$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:eval_word}
    For any word $x$ of length $L$,
    $B^x = Z_v^{-1} A^x Z_{v+L} = Z'^{-1}_v A^x Z'_{v+L}$. Since $\{A^x:|x|=L\}$
    spans $\MN{D}$, $Z_v^{-1} W Z_{v+L} = Z'^{-1}_v W Z'_{v+L}$ for all
    $W\in\MN{D}$. Setting $W=\Id$ gives $Z_v^{-1} Z_{v+L} = Z'^{-1}_v Z'_{v+L}$,
    and right-cancelling yields $Z_v^{-1} W Z_v = Z'^{-1}_v W Z'_v$ for all $W$,
    so $Z'_v Z_v^{-1}\in Z(\MN{D})=\C\Id$.
\end{proof}

\begin{corollary}[Uniqueness of the per-bond gauges up to one constant,
        matrix form]%
    \label{cor:fundamentalTheorem_normalMPS_gauge_unique}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_gauge_unique}
    \leanok
    \uses{def:l_blk_injective}
    Let $0 < L$, $2L + 1 \le n$, $D>0$, and let $A,B\in\MN{D}^d$ be $L$-block
    injective. If invertible families $Z,Z'$ both satisfy
    (\ref{eq:peps_normal_mps_per_bond}), then there is a single scalar
    $c\in\C^\times$ such that $Z'_v = c\,Z_v$ for all $v$
    \cite[Section~4]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{lem:gaugeFamily_bond_proportional}
    By Lemma~\ref{lem:gaugeFamily_bond_proportional}, $Z'_v = \mu_v Z_v$ with
    $\mu_v\in\C^\times$. Substituting into (\ref{eq:peps_normal_mps_per_bond})
    gives $B^i = \mu_v^{-1}\mu_{v+1} B^i$; since $B\neq 0$ by $L$-block
    injectivity and $D>0$, $\mu_{v+1}=\mu_v$ for all $v$.
\end{proof}

\begin{lemma}[Consecutive per-bond gauges are proportional]%
    \label{lem:gaugeFamily_succ_proportional}
    \lean{TNLean.PEPS.gaugeFamily_succ_proportional}
    \leanok
    \uses{def:l_blk_injective, def:eval_word}
    Let $n\ge 1$, let $A,B\in\MN{D}^d$ with $A$ being $L$-block injective, and
    let $(Z_v)_{v=1}^n$ ($Z_{n+1}\equiv Z_1$) be invertible matrices satisfying
    (\ref{eq:peps_normal_mps_per_bond}). Then for each $v$ there exists
    $c_v\in\C^\times$ such that $Z_{v+1} = c_v Z_v$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:eval_word}
    For any word $x$ of length $L$,
    $B^x = Z_v^{-1} A^x Z_{v+L} = Z_{v+1}^{-1} A^x Z_{v+1+L}$. By $L$-block
    injectivity of $A$, $Z_v^{-1} W Z_{v+L} = Z_{v+1}^{-1} W Z_{v+1+L}$ for all
    $W\in\MN{D}$. Setting $W=\Id$ gives $Z_v^{-1} Z_{v+L} = Z_{v+1}^{-1} Z_{v+1+L}$,
    and cancelling yields $Z_v^{-1} W Z_v = Z_{v+1}^{-1} W Z_{v+1}$ for all $W$,
    so $Z_{v+1} Z_v^{-1}\in\C^\times\Id$.
\end{proof}

\begin{corollary}[Fundamental Theorem for translation-invariant normal MPS
        on a closed chain, single-gauge form]%
    \label{cor:fundamentalTheorem_normalMPS_translationInvariant}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_translationInvariant}
    \leanok
    \uses{def:l_blk_injective, def:mpv}

    injective with $V^{(n)}(A) = V^{(n)}(B)$. Then there exist an invertible
    matrix $Z\in\GL_D(\C)$ and $\lambda\in\C$ with $\lambda^n = 1$ such that
    \begin{align}
        B^i = \lambda\, Z^{-1} A^i Z
        \qquad (0\le i<d),
        \label{eq:peps_normal_mps_ti_gauge}
    \end{align}
    giving the translation-invariant corollary of
    \cite[Section~4]{Molnar2018NormalPEPS} at $n\ge 2L+1$.
\end{corollary}
\begin{proof}\leanok
    \uses{cor:fundamentalTheorem_normalMPS_translationInvariant_of_overlap}
    Immediate from
    Corollary~\ref{cor:fundamentalTheorem_normalMPS_translationInvariant_of_overlap}.
\end{proof}

\begin{corollary}[Uniqueness of the translation-invariant gauge up to a
        constant]%
    \label{cor:fundamentalTheorem_normalMPS_translationInvariant_gauge_unique}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_translationInvariant_gauge_unique}
    \leanok
    \uses{def:l_blk_injective}
    Let $0 < L$, $0<D$, and let $A,B\in\MN{D}^d$ be $L$-block injective. If
    $Z,Z'\in\GL_D(\C)$ and $\lambda,\lambda'\in\C$ satisfy
    $B^i = \lambda\, Z^{-1} A^i Z = \lambda'\, Z'^{-1} A^i Z'$ for all $i$, then
    $Z' = c\,Z$ for some $c\in\C^\times$
    \cite[Section~4]{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{def:eval_word}
    Since $B$ is $L$-block injective and $D>0$, some $B^{i_0}\neq 0$, so
    $\lambda\neq 0$. For $|x|=L$,
    $B^x = \lambda^L Z^{-1} A^x Z = \lambda'^L Z'^{-1} A^x Z'$, which extends by
    $L$-block injectivity of $A$ to $\lambda^L Z^{-1} W Z = \lambda'^L Z'^{-1} W Z'$
    for all $W\in\MN{D}$. Setting $W=\Id$ gives $\lambda^L=\lambda'^L\neq 0$,
    so $Z^{-1} W Z = Z'^{-1} W Z'$ for all $W$ and $Z' Z^{-1}\in\C^\times\Id$.
\end{proof}

\subsection{\texorpdfstring{The optimal system size $n \ge 2L + 1$}{The optimal system size n >= 2L+1}}

A bond of the closed chain is straddled by $L + 1$ overlapping windows of $L$
sites. For consecutive windows, the deformed closed-chain coefficients satisfy
\begin{align}
    \tr(B^{p}\, C_j(u_1 \cdots u_L)\, B^{u_{L+1}}\, B^{s})
     & =
    \tr(B^{p}\, B^{u_1}\, C_{j+1}(u_2 \cdots u_{L+1})\, B^{s}).
    \notag
\end{align}
Injectivity on the complementary arc strips these identities to
\begin{align}
    C_j(u_1 \cdots u_L)\,B^{u_{L+1}}
     & =
    B^{u_1}\,C_{j+1}(u_2 \cdots u_{L+1}),
    \notag
\end{align}
and chaining across the $L + 1$ windows gives $C_0(a)\,B^b = B^a\,C_L(b)$, the
overlap mechanism of \cite[Lemma~5]{Molnar2018NormalPEPS} that lowers the system
size to $n \ge 2L + 1$.

\begin{lemma}[Block injectivity extends to longer blocks]%
    \label{lem:isNBlkInjective_of_le}
    \leanok
    \uses{def:l_blk_injective}
    If $A\in\MN{D}^d$ is $L$-block injective with $0 < L$, then $A$ is $m$-block
    injective for every $m \ge L$
    \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}.
\end{lemma}
\begin{proof}\leanok
    \uses{def:eval_word, thm:wordspan_blk_inj_succ}
    By induction on $m\ge L$ via Lemma~\ref{thm:wordspan_blk_inj_succ}: writing
    $\Id=\sum_{|u|=L}\alpha_u A^u$ expresses any $P\in\MN{D}$ as
    $\sum_{i,|w|=L-1}\alpha_{iw} A^i(A^w P)$, and expanding each $A^w P$ in
    $\{A^y:|y|=m\}$ yields a combination of length-$(m+1)$ word products.
\end{proof}

\begin{lemma}[Word transfer between same-state networks]%
    \label{lem:exists_mpvTransport}
    \lean{MPSTensor.exists_mpvTransport}
    \lean{MPSTensor.exists_evalWordTransport}
    \leanok
    \uses{def:l_blk_injective, def:mpv, def:eval_word}
    Let $A,B\in\MN{D}^d$, let $n = k + q$, and suppose $\{A^x:|x|=k\}$ and
    $\{A^y:|y|=q\}$ each span $\MN{D}$. If $V^{(n)}(A)=V^{(n)}(B)$ (or if
    $A^w=B^w$ for all $|w|=n$), then there exists a linear map
    $\Lambda:\MN{D}\to\MN{D}$ such that $\Lambda(A^x)=B^x$ for all $|x|=k$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:mpv, def:eval_word}
    Define $\Psi_A,\Psi_B:\MN{D}\to\C^{d^q}$ by
    $\Psi_A(M)_y=\tr(M A^y)$ and $\Psi_B(M)_y=\tr(M B^y)$ ($|y|=q$). Since
    $\{A^y:|y|=q\}$ spans $\MN{D}$ and the trace pairing is nondegenerate,
    $\Psi_A$ is injective. Since $\Psi_A(A^x)_y=V^{(n)}(A)_{xy}=V^{(n)}(B)_{xy}=\Psi_B(B^x)_y$
    and $\{A^x:|x|=k\}$ spans $\MN{D}$, $\operatorname{im}\Psi_A\subseteq\operatorname{im}\Psi_B$,
    so $\Psi_B$ is injective by dimension count and $\Lambda=\Psi_B^{-1}\circ\Psi_A$
    satisfies $\Lambda(A^x)=B^x$. Replacing $\tr(M A^y)$ by $M A^y$ proves the
    word-product variant.
\end{proof}

\begin{lemma}[Bond-operator extraction from overlapping windows]%
    \label{lem:overlapWindow_exists_bondOperator}
    \lean{MPSTensor.overlapWindow_exists_bondOperator}
    \lean{MPSTensor.bondOperator_unique}
    \leanok
    \uses{def:l_blk_injective, def:eval_word}
    Let $0 < L$, $2L + 1 \le n$, let $B\in\MN{D}^d$ be $L$-block injective, and
    let $C_0, \ldots, C_L:\{0,\ldots,d-1\}^L\to\MN{D}$ satisfy, for all
    $0\le j < L$, $|p|=j$, $|u|=L+1$, and $|s|=n-j-L-1$,
    \begin{align}
        \tr(B^{p}\, C_j(u_1 \cdots u_L)\, B^{u_{L+1}}\, B^{s})
         & =
        \tr\bigl(B^{p}\, B^{u_1}\, C_{j+1}(u_2 \cdots u_{L+1})
        B^{s}\bigr).
        \notag
    \end{align}
    Then there exists a unique $X\in\MN{D}$ such that, for all $|a|=|b|=L$,
    \begin{align}
        C_0(a) & = B^a X, &
        C_L(b) & = X B^b,
        \notag
    \end{align}
    giving the constant specialization of \cite[Lemma~5]{Molnar2018NormalPEPS}
    (Lemma~\ref{lem:chain_overlapWindow_exists_bondOperator}).
\end{lemma}
\begin{proof}\leanok
    \uses{lem:chain_overlapWindow_exists_bondOperator, def:eval_word}
    Apply Lemma~\ref{lem:chain_overlapWindow_exists_bondOperator} to the
    constant chain with tensor $B$ at every site.
\end{proof}

\begin{lemma}[Conjugate word products at one length from Lemma~5]%
    \label{lem:exists_conjugation_of_mpv_eq}
    \lean{MPSTensor.exists_conjugation_of_mpv_eq}
    \leanok
    \uses{def:l_blk_injective, def:mpv, def:eval_word}
    Let $0 < L$, $2L + 1 \le n$, $0<D$, and let $A,B\in\MN{D}^d$ be $L$-block
    injective with $V^{(n)}(A)=V^{(n)}(B)$. Then there exists $Z\in\GL_D(\C)$
    such that
    \begin{align}
        B^w & = Z\, A^w\, Z^{-1}
        \qquad (|w|=n).
        \label{eq:peps_mps_length_n_conj}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{cor:fundamentalTheorem_normalMPSChain_of_overlap, thm:mps_chain_gaugeEquiv_sameState}
    Regarding $A$ and $B$ as constant chains on $n$ sites,
    Corollary~\ref{cor:fundamentalTheorem_normalMPSChain_of_overlap} gives bond
    gauges $(Z_v)_{v=0}^{n-1}$ with $B^i=Z_v A^i Z_{v+1}^{-1}$, and
    Lemma~\ref{thm:mps_chain_gaugeEquiv_sameState} at length $n$ yields
    $B^w = Z_0 A^w Z_0^{-1}$.
\end{proof}

\begin{lemma}[Rigidity of word products at one length]%
    \label{lem:proportional_of_evalWord_eq}
    \lean{MPSTensor.proportional_of_evalWord_eq}
    \leanok
    \uses{def:l_blk_injective, def:eval_word}
    Let $0 < L$, $2L + 1 \le n$, $0 < D$, and let $A,C\in\MN{D}^d$ be $L$-block
    injective with $C^w = A^w$ for all $|w|=n$. Then there exists $\lambda\in\C$
    with $\lambda^n = 1$ such that $C^i = \lambda A^i$ for all $i$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:exists_mpvTransport, lem:isNBlkInjective_of_le, def:eval_word}
    Since $2L+1\le n$, Lemma~\ref{lem:isNBlkInjective_of_le} and the word-product
    form of Lemma~\ref{lem:exists_mpvTransport} give linear bijections
    $\Lambda_k:\MN{D}\to\MN{D}$ with $\Lambda_k(A^x)=C^x$ for $|x|=k\in\{L,L+1,n-1-L,n-L\}$.
    For $k\in\{L,n-1-L\}$, $C^i\Lambda_k(\Id)=\Lambda_{k+1}(A^i)=\Lambda_k(\Id)C^i$,
    so $\Lambda_k(\Id)=c_k\Id$ with $c_k\in\C^\times$ by $L$-block injectivity
    of $C$. Splitting $A^{xiy}=C^{xiy}$ ($|x|=L$, $|y|=n-1-L$) and evaluating on
    combinations summing to $\Id$ gives $C^i=(c_L c_{n-1-L})^{-1}A^i=:\lambda A^i$,
    and $C^w=A^w\neq 0$ forces $\lambda^n=1$.
\end{proof}

\begin{corollary}[Fundamental Theorem for translation-invariant normal MPS
        at $n \ge 2L + 1$, single-gauge form]%
    \label{cor:fundamentalTheorem_normalMPS_translationInvariant_of_overlap}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_translationInvariant_of_overlap}
    \leanok
    \uses{def:l_blk_injective, def:mpv}
    Let $0 < L$, $2L + 1 \le n$, $0<D$, and let $A,B\in\MN{D}^d$ be $L$-block
    injective with $V^{(n)}(A) = V^{(n)}(B)$. Then there exist $Z\in\GL_D(\C)$
    and $\lambda\in\C$ with $\lambda^n = 1$ satisfying
    (\ref{eq:peps_normal_mps_ti_gauge})
    \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{lem:exists_conjugation_of_mpv_eq, lem:proportional_of_evalWord_eq}
    By Lemma~\ref{lem:exists_conjugation_of_mpv_eq}, there exists $Z\in\GL_D(\C)$
    such that $\widetilde B^i := Z B^i Z^{-1}$ satisfies $\widetilde B^w = A^w$
    for all $|w|=n$. Applying Lemma~\ref{lem:proportional_of_evalWord_eq} to $A$
    and $\widetilde B$ gives $\widetilde B^i = \lambda A^i$ with $\lambda^n=1$,
    so $B^i = \lambda Z^{-1} A^i Z$.
\end{proof}

\begin{corollary}[Fundamental Theorem for translation-invariant normal MPS
        at $n \ge 2L + 1$, per-bond form]%
    \label{cor:fundamentalTheorem_normalMPS_of_overlap}
    \lean{TNLean.PEPS.fundamentalTheorem_normalMPS_of_overlap}
    \leanok
    \uses{def:l_blk_injective, def:mpv}
    Let $0 < L$, $2L + 1 \le n$, $0<D$, and let $A,B\in\MN{D}^d$ be $L$-block
    injective with $V^{(n)}(A)=V^{(n)}(B)$. Then there exist invertible matrices
    $(Z_v)_{v=1}^n$ ($Z_{n+1}\equiv Z_1$) satisfying
    (\ref{eq:peps_normal_mps_per_bond})
    \cite[Section~``Normal MPS: alternative proof'']{Molnar2018NormalPEPS}.
\end{corollary}
\begin{proof}\leanok
    \uses{cor:fundamentalTheorem_normalMPS_translationInvariant_of_overlap}
    Take $Z,\lambda$ from
    Corollary~\ref{cor:fundamentalTheorem_normalMPS_translationInvariant_of_overlap}
    with $\lambda^n=1$ and $B^i=\lambda Z^{-1}A^i Z$, and set
    $Z_v := \lambda^v Z$; since $\lambda^{1-n}=\lambda$,
    $Z_v^{-1} A^i Z_{v+1} = \lambda Z^{-1} A^i Z = B^i$ for all $1\le v\le n$.
\end{proof}
