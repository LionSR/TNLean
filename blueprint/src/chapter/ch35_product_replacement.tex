\section{Product replacement at a cut}

The product replacement argument of~\cite{Schuch2008EntropyScaling}
(arXiv:0705.0292v2, lines 436--470) can be expressed directly in terms of
coefficient spaces and cut ranks.

\begin{theorem}[Product replacement does not increase cut ranks]
    \label{thm:chain_product_replacement_ranks}
    \lean{TNLean.MPS.AreaLaw.rank_cutMatrix_le_of_product_mem_ranges}
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi,\chi$ be vectors on $L$ sites of local dimension $q$, and
    write $C=M_k(\psi)$ for the coefficient matrix at a cut $k$.
    Suppose $u\in\operatorname{range}C$, $v\in\operatorname{range}C^{\mathsf T}$,
    and $M_k(\chi)=uv^{\mathsf T}$. Then
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ at every cut $j$.
    No nonvanishing or normalization is required.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_supported_cut_rank}
    Choose $w,z$ with $Cw=u$ and $C^{\mathsf T}z=v$. The identities
    \begin{align}
        \label{eq:chain_product_replacement_local_maps}
        (uz^{\mathsf T})C=uv^{\mathsf T}
        =C{(vw^{\mathsf T})}^{\mathsf T}
    \end{align}
    express $\chi$ both as a left-local and as a right-local linear image of
    $\psi$. For $j\le k$, use the right-local image; for $j>k$, use the
    left-local image. The corresponding operator acts on only one side of
    cut $j$, so cannot increase its rank. If there are no configurations,
    both vectors vanish and the assertion is immediate.
\end{proof}

\begin{theorem}[Preservation of the bond bound]
    \label{thm:chain_product_replacement_mps}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.of_product_mem_cut_ranges}
    \leanok
    \uses{thm:chain_product_replacement_ranks,def:chain_open_mps}
    Under the hypotheses of Theorem~\ref{thm:chain_product_replacement_ranks},
    if $\psi$ is an open-boundary MPS of bond dimension at most $D$, then
    so is $\chi$. This includes $D=0$ and empty or one-site chains.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_product_replacement_ranks,thm:chain_mps_of_cut_rank}
    Each internal cut rank of $\chi$ is at most the corresponding rank of
    $\psi$, hence at most $D$. Apply the cut-rank characterization.
\end{proof}

\begin{theorem}[Normalized product replacement]
    \label{thm:chain_normalized_product_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_product_overlap_ge}
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi$ be a unit open-boundary MPS of bond dimension at most $D$.
    At any cut $k$, let $a,b$ be arbitrary target vectors on the two sides.
    There are unit vectors $u,v$ and a unit open-boundary MPS $\chi$ of
    bond dimension at most $D$ such that $M_k(\chi)=uv^{\mathsf T}$,
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ for every $j$, and
    $|\langle a\otimes b,\psi\rangle|
        \le |\langle a,u\rangle|\,|\langle b,v\rangle|$.
    The inner product on the left is taken in cut coordinates.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_product_replacement_mps}
    The product-overlap theorem in~\cite{QICLean2026Blueprint} gives unit
    $u\in\operatorname{range}M_k(\psi)$ and
    $v\in\operatorname{range}{M_k(\psi)}^{\mathsf T}$ with the stated overlap
    bound. Transport $u\otimes v$ back to the chain coordinates to define
    $\chi$. This transport preserves norms, so $\|\chi\|=\|u\|\,\|v\|=1$.
    Theorem~\ref{thm:chain_product_replacement_mps} preserves the bond bound.
\end{proof}

\begin{theorem}[Replacement at finitely many product cuts]
    \label{thm:chain_product_cut_iteration}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_rank_one_at_product_cuts}
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi$ be a unit open-boundary MPS of bond dimension at most $D$,
    let $\xi$ be a target vector, and let $S$ be a finite set of cuts.
    Suppose that $\xi$ factors across each cut in $S$.
    Then there is a unit open-boundary MPS $\chi$ of bond dimension at most
    $D$ such that $|\langle\xi,\psi\rangle|\le|\langle\xi,\chi\rangle|$,
    no cut rank of $\chi$ exceeds the corresponding cut rank of $\psi$,
    and every cut in $S$ has rank at most one for $\chi$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_normalized_product_replacement,thm:chain_product_replacement_ranks}
    Induct on $|S|$. At a newly selected cut, apply normalized product
    replacement to the current approximant and the two factors of the target.
    The overlap does not decrease. The new cut has rank at most one, and
    all earlier rank bounds persist because no cut rank increases.
    Normalization and the bond bound persist at each step.
\end{proof}

\begin{definition}[Consecutive block products]
    \label{def:chain_copy_product}
    \lean{TNLean.MPS.AreaLaw.copyConfigurationEquiv,TNLean.MPS.AreaLaw.copyProduct}
    \leanok
    Group the sites of a chain of length $KL$ into $K$ consecutive blocks
    of length $L$, identifying block $i$, site $j$ with site $Li+j$.
    For block vectors $\eta_0,\ldots,\eta_{K-1}$, define their product by
    \begin{align}
        \label{eq:chain_copy_product_coefficients}
        \left(\bigotimes_{i=0}^{K-1}\eta_i\right)(\sigma)
        =\prod_{i=0}^{K-1}\eta_i\bigl(j\mapsto\sigma(Li+j)\bigr).
    \end{align}
\end{definition}

\begin{theorem}[Norms and inner products of block products]
    \label{thm:chain_copy_product_inner}
    \lean{TNLean.MPS.AreaLaw.norm_copyProduct,TNLean.MPS.AreaLaw.inner_copyProduct,TNLean.MPS.AreaLaw.sq_norm_inner_copyProduct}
    \leanok
    \uses{def:chain_copy_product}
    Norms and inner products of block products are the products of the
    corresponding block norms and inner products. In particular, their
    squared fidelities multiply.
\end{theorem}
\begin{proof}\leanok
    The map $\sigma\mapsto(\tau_0,\ldots,\tau_{K-1})$, with
    $\tau_i(j)=\sigma(Li+j)$, is a bijection from configurations on $KL$ sites
    to families of block configurations. Reindexing the configuration sum
    along it and using~\eqref{eq:chain_copy_product_coefficients} gives
    \begin{align}
        \label{eq:chain_copy_product_inner}
        \Bigl\langle\bigotimes_{i=0}^{K-1}\eta_i,\bigotimes_{i=0}^{K-1}\theta_i\Bigr\rangle
         & =\sum_{\tau_0,\ldots,\tau_{K-1}}\,\prod_{i=0}^{K-1}
        \overline{\eta_i(\tau_i)}\,\theta_i(\tau_i)
        =\prod_{i=0}^{K-1}\langle\eta_i,\theta_i\rangle.
    \end{align}
    Taking $\theta_i=\eta_i$ and square roots gives
    $\bigl\|\bigotimes_{i=0}^{K-1}\eta_i\bigr\|=\prod_{i=0}^{K-1}\|\eta_i\|$,
    and taking squared absolute values gives
    $\bigl|\bigl\langle\bigotimes_{i=0}^{K-1}\eta_i,\bigotimes_{i=0}^{K-1}\theta_i\bigr\rangle\bigr|^2
        =\prod_{i=0}^{K-1}|\langle\eta_i,\theta_i\rangle|^2$.
\end{proof}

\begin{theorem}[Factorization at copy boundaries]
    \label{thm:chain_copy_boundary_factorization}
    \lean{TNLean.MPS.AreaLaw.copyProduct_factors_at_boundary}
    \leanok
    \uses{def:chain_copy_product}
    A block product factors across every boundary $tL$, for $t\ge0$.
\end{theorem}
\begin{proof}\leanok
    A site $Li+j$ lies before $tL$ exactly when $i<t$.
    Split the coefficient product into those blocks and the remaining blocks.
\end{proof}

\begin{theorem}[Replacement at all copy boundaries]
    \label{thm:chain_copy_boundary_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_rank_one_at_copy_boundaries}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    Given a unit MPS $\psi$ of bond dimension at most $D$ and a block-product
    target, there is a unit MPS $\chi$ of the same bond bound with no smaller
    overlap, no larger cut ranks, and rank at most one at every copy boundary.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_boundary_factorization,thm:chain_product_cut_iteration}
    Apply finite-cut replacement to the boundaries $tL$, $0\le t\le K$.
\end{proof}

\begin{definition}[An interval slice]
    \label{def:chain_interval_slice}
    \lean{TNLean.MPS.AreaLaw.insertInterval,TNLean.MPS.AreaLaw.intervalSlice}
    \leanok
    Fix a configuration outside an interval of $L$ sites starting at $a$.
    The corresponding interval slice of a chain vector is obtained by varying
    the configuration on that interval and keeping the exterior fixed.
\end{definition}

\begin{theorem}[Bond bounds for interval slices]
    \label{thm:chain_interval_slice_mps}
    \lean{TNLean.MPS.AreaLaw.rank_cutMatrix_intervalSlice_le,TNLean.MPS.AreaLaw.IsOpenMPS.intervalSlice,TNLean.MPS.AreaLaw.IsOpenMPS.of_intervalSlice_eq_mul}
    \leanok
    \uses{def:chain_interval_slice,def:chain_open_mps}
    At a local cut $0\le j\le L$, the coefficient matrix of an interval slice
    has rank at most that of the original vector at cut $a+j$.
    If the interval lies within the chain, every slice of an MPS of bond
    dimension at most $D$ is an MPS of that same bond bound.
    A vector differing from such a slice by a nonzero scalar also has this bound.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_mps_of_cut_rank}
    The sliced coefficient matrix is a submatrix of the full coefficient matrix:
    extend each row and column configuration by the fixed exterior values.
    Taking a submatrix cannot increase rank. Apply the cut-rank characterization,
    and absorb any nonzero scalar into the overall MPS coefficient.
\end{proof}

\begin{theorem}[Bond bounds for normalized block factors]
    \label{thm:chain_copy_product_factor_mps}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.copyProduct_factor}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    If a product of unit block vectors is an MPS of bond dimension at most $D$,
    then each block factor is an MPS of bond dimension at most $D$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_interval_slice_mps}
    Fix the factor $\eta_i$. Each unit factor $\eta_j$ has a block
    configuration $\sigma_j$ with $\eta_j(\sigma_j)\neq0$. Fix the exterior of
    the interval of block $i$, which starts at site $Li$, to $\sigma_j$ on every
    block $j\neq i$. By~\eqref{eq:chain_copy_product_coefficients}, the
    resulting interval slice $\phi$ of $\bigotimes_{j=0}^{K-1}\eta_j$ is
    \begin{align}
        \label{eq:chain_copy_product_slice}
        \phi(x)=\Bigl(\prod_{j\neq i}\eta_j(\sigma_j)\Bigr)\,\eta_i(x),
    \end{align}
    a nonzero multiple of $\eta_i$. Theorem~\ref{thm:chain_interval_slice_mps}
    gives $\eta_i$ the bond bound $D$.
\end{proof}

\begin{theorem}[Scalar error amplification for a block product]
    \label{thm:chain_copy_error_amplification}
    \lean{TNLean.MPS.AreaLaw.exists_copy_optimizedPureTraceError_le}
    \leanok
    \uses{def:chain_copy_product}
    Let $K\ge1$, and let $\eta,\theta_0,\ldots,\theta_{K-1}$ be unit block vectors.
    Write $f_i=|\langle\eta,\theta_i\rangle|^2$, $F=\prod_i f_i$, and
    $T(f)=2\sqrt{f(1-f)}$ for $f\ge1/2$, with $T(f)=1$ otherwise.
    If $T(F)<1$, then some $i$ satisfies
    \begin{align}
        \label{eq:chain_copy_error_amplification}
        T(f_i^K)\le T(F),\qquad
        \sqrt{K/4}\,T(f_i)\le T(F).
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_product_inner}
    Choose a factor with maximal squared fidelity, so $f_i^K\ge F$.
    The scalar function $T$ is decreasing. Thus $T(f_i^K)\le T(F)<1$,
    and the scalar amplification theorem in~\cite{QICLean2026Blueprint}
    gives $\sqrt{K/4}\,T(f_i)\le T(f_i^K)$.
\end{proof}

\begin{theorem}[Normalized factors from rank-one copy boundaries]
    \label{thm:chain_unit_copy_factors}
    \lean{TNLean.MPS.AreaLaw.exists_unit_block_factors}
    \leanok
    \uses{def:chain_copy_product}
    Let $K\ge1$ and let $\psi$ be a unit vector on $KL$ sites.
    If its coefficient matrix has rank at most one at every internal copy
    boundary $tL$, $0<t<K$, then $\psi$ is a product of $K$ unit block vectors.
    This includes blocks of length zero.
\end{theorem}
\begin{proof}\leanok
    Write $\Psi(x_0,\ldots,x_{K-1})$ for the coefficient of $\psi$ at the
    configuration whose block $i$ is $x_i$, and $(x_{<n},y_{\ge n})$ for the
    configuration agreeing with $x$ on blocks $i<n$ and with $y$ on blocks
    $i\ge n$. The coefficient matrix at $nL$ has rank at most one for every
    $n$, since the cuts $0$ and $KL$ have rank at most one automatically.
    Its two-by-two minors therefore vanish:
    \begin{align}
        \label{eq:chain_copy_cross_ratio}
        \Psi(x)\,\Psi(y)=\Psi(x_{<n},y_{\ge n})\,\Psi(y_{<n},x_{\ge n}).
    \end{align}
    Choose $x^0$ with $\Psi(x^0)\neq0$, and write $x^0[i\mapsto a]$ for $x^0$
    with block $i$ replaced by $a$. Applying~\eqref{eq:chain_copy_cross_ratio}
    at the boundary $nL$ to $(x_{<n+1},x^0_{\ge n+1})$ and $x^0$ gives
    \begin{align}
        \Psi(x_{<n+1},x^0_{\ge n+1})\,\Psi(x^0)
        =\Psi(x_{<n},x^0_{\ge n})\,\Psi\bigl(x^0[n\mapsto x_n]\bigr),
        \notag
    \end{align}
    and induction on $n$ yields
    \begin{align}
        \label{eq:chain_copy_coefficient_factorization}
        \Psi(x)=\Psi(x^0)\prod_{i=0}^{K-1}
        \frac{\Psi\bigl(x^0[i\mapsto x_i]\bigr)}{\Psi(x^0)}.
    \end{align}
    Put $w_i(a)=\Psi(x^0[i\mapsto a])/\Psi(x^0)$, absorb the scalar into the
    first factor by setting $v_0=\Psi(x^0)\,w_0$ and $v_i=w_i$ for $i>0$, and
    conclude $\psi=\bigotimes_{i=0}^{K-1}v_i$. Then
    $\prod_{i=0}^{K-1}\|v_i\|=\|\psi\|=1$, so every $\|v_i\|$ is nonzero and
    the unit vectors $v_i/\|v_i\|$ have the same product $\psi$.
\end{proof}

\begin{theorem}[Product replacement by normalized MPS factors]
    \label{thm:chain_unit_mps_copy_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_unit_copy_product_overlap_ge}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    Let $K\ge1$, let $\psi$ be a unit MPS on $KL$ sites of bond dimension
    at most $D$, and let $\xi$ be a block-product target.
    There are unit block vectors $\theta_0,\ldots,\theta_{K-1}$, each an MPS
    of bond dimension at most $D$, whose product has overlap with $\xi$
    at least as large as that of $\psi$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_boundary_replacement,thm:chain_unit_copy_factors,thm:chain_copy_product_factor_mps}
    First replace $\psi$ so that every copy boundary has rank at most one.
    Extract unit block factors, and then apply the bond bound for factors
    of a product MPS.
\end{proof}

\begin{theorem}[Scalar amplification from an arbitrary MPS approximant]
    \label{thm:chain_mps_copy_error_amplification}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_copy_error_amplification}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps,thm:chain_copy_error_amplification}
    Let $K\ge1$, let $\eta$ be a unit block vector, and let $\psi$ be any unit
    MPS on $KL$ sites of bond dimension at most $D$. Put
    $F=\bigl|\bigl\langle\bigotimes_{i=0}^{K-1}\eta,\psi\bigr\rangle\bigr|^2$ and suppose $T(F)<1$,
    with $T$ as in Theorem~\ref{thm:chain_copy_error_amplification}.
    There is a unit block MPS $\theta$ of bond dimension at most $D$ such that,
    writing $f=|\langle\eta,\theta\rangle|^2$,
    \begin{align}
        \label{eq:chain_mps_copy_error_amplification}
        T(f^K)\le T(F),\qquad \sqrt{K/4}\,T(f)\le T(F).
    \end{align}
    These are estimates for the scalar function of squared fidelity.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_unit_mps_copy_replacement,thm:chain_copy_product_inner,thm:chain_copy_error_amplification}
    Replace the approximant by a product of unit MPS block factors with
    no smaller overlap. Its squared fidelity is no smaller, and hence its
    scalar error is no larger than $T(F)$. Apply the block-product
    amplification theorem and select the factor it supplies.
\end{proof}
