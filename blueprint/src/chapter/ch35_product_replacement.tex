\section{Product replacement at a cut}

The product replacement argument of~\cite{Schuch2008EntropyScaling}
(arXiv:0705.0292v2, lines 436--470) is formulated below in terms of cut
coefficient matrices $M_k(\psi)$ and their ranks.

\begin{theorem}[Product replacement does not increase cut ranks]
    \label{thm:chain_product_replacement_ranks}
    \lean{TNLean.MPS.AreaLaw.rank_cutMatrix_le_of_product_mem_ranges}
    \leanok
    \uses{def:chain_open_mps}
    Let $\psi,\chi$ be vectors on $L$ sites of local dimension $q$, and set
    $C=M_k(\psi)$. If $u\in\operatorname{range}C$,
    $v\in\operatorname{range}C^{\mathsf T}$, and $M_k(\chi)=uv^{\mathsf T}$,
    then $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ for every
    cut $j$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_supported_cut_rank}
    Choose $w,z$ with $Cw=u$ and $C^{\mathsf T}z=v$. Then
    \begin{align}
        \label{eq:chain_product_replacement_local_maps}
        (uz^{\mathsf T})C=uv^{\mathsf T}
        =C{(vw^{\mathsf T})}^{\mathsf T}
    \end{align}
    realizes $\chi$ as both a left-local ($j>k$) and right-local ($j\le k$)
    image of $\psi$, which cannot increase $\operatorname{rank}M_j$ by
    Theorem~\ref{thm:hastings_supported_cut_rank}.
\end{proof}

\begin{theorem}[Preservation of the bond bound]
    \label{thm:chain_product_replacement_mps}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.of_product_mem_cut_ranges}
    \leanok
    \uses{thm:chain_product_replacement_ranks,def:chain_open_mps}
    Under the hypotheses of Theorem~\ref{thm:chain_product_replacement_ranks},
    if $\psi$ is an open-boundary MPS of bond dimension at most $D\ge0$, then
    so is $\chi$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_product_replacement_ranks,thm:chain_mps_of_cut_rank}
    By Theorem~\ref{thm:chain_product_replacement_ranks},
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)\le D$ for
    $0<j<L$; apply Theorem~\ref{thm:chain_mps_of_cut_rank}.
\end{proof}

\begin{theorem}[Normalized product replacement]
    \label{thm:chain_normalized_product_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_product_overlap_ge}
    \leanok
    \uses{def:chain_open_mps}
    For a unit open-boundary MPS $\psi$ of bond dimension at most $D$, a cut
    $k$, and target vectors $a,b$ on the two sides, there exist unit vectors
    $u,v$ and a unit open-boundary MPS $\chi$ of bond dimension at most $D$
    such that $M_k(\chi)=uv^{\mathsf T}$,
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ for all $j$,
    and $|\langle a\otimes b,\psi\rangle|\le|\langle a,u\rangle|\,|\langle b,v\rangle|$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_product_replacement_mps}
    The product-overlap theorem of~\cite{QICLean2026Blueprint} gives unit
    $u\in\operatorname{range}M_k(\psi)$ and
    $v\in\operatorname{range}{M_k(\psi)}^{\mathsf T}$ with
    $|\langle a\otimes b,\psi\rangle|\le|\langle a,u\rangle|\,|\langle b,v\rangle|$;
    defining $\chi$ via $M_k(\chi)=uv^{\mathsf T}$ gives $\|\chi\|=1$ and the
    bond bound by Theorem~\ref{thm:chain_product_replacement_mps}.
\end{proof}

\begin{theorem}[Replacement at finitely many product cuts]
    \label{thm:chain_product_cut_iteration}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_rank_one_at_product_cuts}
    \leanok
    \uses{def:chain_open_mps}
    If $\psi$ is a unit open-boundary MPS of bond dimension at most $D$ and
    $\xi$ factors across each cut in a finite set $S$, then there exists a unit
    open-boundary MPS $\chi$ of bond dimension at most $D$ with
    $|\langle\xi,\psi\rangle|\le|\langle\xi,\chi\rangle|$,
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ for all $j$,
    and $\operatorname{rank}M_k(\chi)\le1$ for all $k\in S$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_normalized_product_replacement,thm:chain_product_replacement_ranks}
    Induct on $|S|$, applying
    Theorem~\ref{thm:chain_normalized_product_replacement} at each cut and
    preserving prior rank bounds via
    Theorem~\ref{thm:chain_product_replacement_ranks}.
\end{proof}

\begin{definition}[Consecutive block products]
    \label{def:chain_copy_product}
    \lean{TNLean.MPS.AreaLaw.copyConfigurationEquiv,TNLean.MPS.AreaLaw.copyProduct}
    \leanok
    Partition $KL$ sites into $K$ blocks of length $L$ via $(i,j)\mapsto Li+j$,
    and define the block product of $\eta_0,\dots,\eta_{K-1}$ by
    \begin{align}
        \label{eq:chain_copy_product_coefficients}
        \left(\bigotimes_{i=0}^{K-1}\eta_i\right)(\sigma)
        =\prod_{i=0}^{K-1}\eta_i\bigl(j\mapsto\sigma(Li+j)\bigr).
    \end{align}
\end{definition}

\begin{theorem}[Norms and inner products of block products]
    \label{thm:chain_copy_product_inner}
    \lean{TNLean.MPS.AreaLaw.norm_copyProduct,TNLean.MPS.AreaLaw.inner_copyProduct,TNLean.MPS.AreaLaw.sq_norm_inner_copyProduct}
    \leanok
    \uses{def:chain_copy_product}
    For block vectors $\eta_i,\theta_i$,
    \begin{align}
        \label{eq:chain_copy_product_inner}
        \Bigl\langle\bigotimes_{i=0}^{K-1}\eta_i,\bigotimes_{i=0}^{K-1}\theta_i\Bigr\rangle
         & =\prod_{i=0}^{K-1}\langle\eta_i,\theta_i\rangle, &
        \Bigl\|\bigotimes_{i=0}^{K-1}\eta_i\Bigr\|
         & =\prod_{i=0}^{K-1}\|\eta_i\|.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Reindexing $\sigma\mapsto(\tau_0,\dots,\tau_{K-1})$ with
    $\tau_i(j)=\sigma(Li+j)$ in (\ref{eq:chain_copy_product_coefficients})
    factors $\sum_{\tau_0,\dots,\tau_{K-1}}\prod_{i=0}^{K-1}\overline{\eta_i(\tau_i)}\theta_i(\tau_i)$
    into (\ref{eq:chain_copy_product_inner}).
\end{proof}

\begin{theorem}[Factorization at copy boundaries]
    \label{thm:chain_copy_boundary_factorization}
    \lean{TNLean.MPS.AreaLaw.copyProduct_factors_at_boundary}
    \leanok
    \uses{def:chain_copy_product}
    Every block product $\bigotimes_{i=0}^{K-1}\eta_i$ factors across each cut
    $tL$ ($t\ge0$).
\end{theorem}
\begin{proof}\leanok
    Since $Li+j<tL\iff i<t$, (\ref{eq:chain_copy_product_coefficients}) splits
    into $\prod_{i<t}$ and $\prod_{i\ge t}$.
\end{proof}

\begin{theorem}[Replacement at all copy boundaries]
    \label{thm:chain_copy_boundary_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_rank_one_at_copy_boundaries}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    For a unit MPS $\psi$ of bond dimension at most $D$ and a block-product
    target $\xi=\bigotimes_{i=0}^{K-1}\eta_i$, there exists a unit MPS $\chi$
    of bond dimension at most $D$ with
    $|\langle\xi,\psi\rangle|\le|\langle\xi,\chi\rangle|$,
    $\operatorname{rank}M_j(\chi)\le\operatorname{rank}M_j(\psi)$ for all $j$,
    and $\operatorname{rank}M_{tL}(\chi)\le1$ for $0\le t\le K$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_boundary_factorization,thm:chain_product_cut_iteration}
    Apply Theorems~\ref{thm:chain_copy_boundary_factorization}
    and~\ref{thm:chain_product_cut_iteration} with $S=\{tL:0\le t\le K\}$.
\end{proof}

\begin{definition}[An interval slice]
    \label{def:chain_interval_slice}
    \lean{TNLean.MPS.AreaLaw.insertInterval,TNLean.MPS.AreaLaw.intervalSlice}
    \leanok
    For an interval $\{a,\dots,a+L-1\}$ and a fixed exterior configuration, the
    \emph{interval slice} of a chain vector restricts its argument to
    configurations on $\{a,\dots,a+L-1\}$ extended by that exterior
    configuration.
\end{definition}

\begin{theorem}[Bond bounds for interval slices]
    \label{thm:chain_interval_slice_mps}
    \lean{TNLean.MPS.AreaLaw.rank_cutMatrix_intervalSlice_le,TNLean.MPS.AreaLaw.IsOpenMPS.intervalSlice,TNLean.MPS.AreaLaw.IsOpenMPS.of_intervalSlice_eq_mul}
    \leanok
    \uses{def:chain_interval_slice,def:chain_open_mps}
    For any interval slice $\phi$ of $\psi$ on $\{a,\dots,a+L-1\}$ and
    $0\le j\le L$, $\operatorname{rank}M_j(\phi)\le\operatorname{rank}M_{a+j}(\psi)$.
    If $\{a,\dots,a+L-1\}$ lies in the chain and $\psi$ is an MPS of bond
    dimension at most $D$, then $\phi$ and any nonzero scalar multiple of $\phi$
    are MPS of bond dimension at most $D$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_mps_of_cut_rank}
    Since $M_j(\phi)$ is a submatrix of $M_{a+j}(\psi)$,
    $\operatorname{rank}M_j(\phi)\le\operatorname{rank}M_{a+j}(\psi)$; apply
    Theorem~\ref{thm:chain_mps_of_cut_rank}.
\end{proof}

\begin{theorem}[Bond bounds for normalized block factors]
    \label{thm:chain_copy_product_factor_mps}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.copyProduct_factor}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    If $\bigotimes_{j=0}^{K-1}\eta_j$ with $\|\eta_j\|=1$ is an MPS of bond
    dimension at most $D$, then each $\eta_i$ is an MPS of bond dimension at
    most $D$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_interval_slice_mps}
    Pick $\sigma_j$ with $\eta_j(\sigma_j)\ne0$ ($j\ne i$). By
    (\ref{eq:chain_copy_product_coefficients}), the interval slice $\phi$ on
    block $i$ with exterior $(\sigma_j)_{j\ne i}$ satisfies
    \begin{align}
        \label{eq:chain_copy_product_slice}
        \phi(x)=\Bigl(\prod_{j\neq i}\eta_j(\sigma_j)\Bigr)\,\eta_i(x),
    \end{align}
    so Theorem~\ref{thm:chain_interval_slice_mps} applies.
\end{proof}

\begin{theorem}[Scalar error amplification for a block product]
    \label{thm:chain_copy_error_amplification}
    \lean{TNLean.MPS.AreaLaw.exists_copy_optimizedPureTraceError_le}
    \leanok
    \uses{def:chain_copy_product}
    For $K\ge1$, unit block vectors $\eta,\theta_0,\dots,\theta_{K-1}$,
    $f_i=|\langle\eta,\theta_i\rangle|^2$, $F=\prod_if_i$, and
    $T(f)=2\sqrt{f(1-f)}$ for $f\ge1/2$ ($T(f)=1$ for $f<1/2$), if $T(F)<1$
    then some $i$ satisfies
    \begin{align}
        \label{eq:chain_copy_error_amplification}
        T(f_i^K)\le T(F),\qquad
        \sqrt{K/4}\,T(f_i)\le T(F).
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_product_inner}
    Choosing $i$ maximizing $f_i$ gives $f_i^K\ge F$, so $T(f_i^K)\le T(F)<1$;
    the scalar amplification lemma of~\cite{QICLean2026Blueprint} gives
    $\sqrt{K/4}\,T(f_i)\le T(f_i^K)$.
\end{proof}

\begin{theorem}[Normalized factors from rank-one copy boundaries]
    \label{thm:chain_unit_copy_factors}
    \lean{TNLean.MPS.AreaLaw.exists_unit_block_factors}
    \leanok
    \uses{def:chain_copy_product}
    If $K\ge1$, $\|\psi\|=1$ on $KL$ sites, and
    $\operatorname{rank}M_{tL}(\psi)\le1$ for all $0<t<K$, then
    $\psi=\bigotimes_{i=0}^{K-1}\theta_i$ for unit block vectors $\theta_i$.
\end{theorem}
\begin{proof}\leanok
    Write $\Psi(x_0,\dots,x_{K-1})$ for $\psi$ and $(x_{<n},y_{\ge n})$ for the
    spliced configuration at block $n$. Since $\operatorname{rank}M_{nL}(\psi)\le1$
    for all $0\le n\le K$, all $2\times2$ minors vanish:
    \begin{align}
        \label{eq:chain_copy_cross_ratio}
        \Psi(x)\,\Psi(y)=\Psi(x_{<n},y_{\ge n})\,\Psi(y_{<n},x_{\ge n}).
    \end{align}
    Fix $x^0$ with $\Psi(x^0)\neq0$. Applying (\ref{eq:chain_copy_cross_ratio})
    to $(x_{<n+1},x^0_{\ge n+1})$ and $x^0$ inductively gives
    \begin{align}
        \label{eq:chain_copy_coefficient_factorization}
        \Psi(x)=\Psi(x^0)\prod_{i=0}^{K-1}
        \frac{\Psi\bigl(x^0[i\mapsto x_i]\bigr)}{\Psi(x^0)}.
    \end{align}
    Normalizing the resulting factors $v_i$ using
    $\prod_{i=0}^{K-1}\|v_i\|=\|\psi\|=1$ yields unit $\theta_i=v_i/\|v_i\|$.
\end{proof}

\begin{theorem}[Product replacement by normalized MPS factors]
    \label{thm:chain_unit_mps_copy_replacement}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_unit_copy_product_overlap_ge}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps}
    For $K\ge1$, a unit MPS $\psi$ on $KL$ sites of bond dimension at most $D$,
    and a block-product target $\xi$, there exist unit MPS block vectors
    $\theta_0,\dots,\theta_{K-1}$ of bond dimension at most $D$ with
    $|\langle\xi,\psi\rangle|\le\bigl|\bigl\langle\xi,\bigotimes_{i=0}^{K-1}\theta_i\bigr\rangle\bigr|$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_copy_boundary_replacement,thm:chain_unit_copy_factors,thm:chain_copy_product_factor_mps}
    Combine Theorems~\ref{thm:chain_copy_boundary_replacement},
    \ref{thm:chain_unit_copy_factors}, and~\ref{thm:chain_copy_product_factor_mps}.
\end{proof}

\begin{theorem}[Scalar amplification from an arbitrary MPS approximant]
    \label{thm:chain_mps_copy_error_amplification}
    \lean{TNLean.MPS.AreaLaw.IsOpenMPS.exists_copy_error_amplification}
    \leanok
    \uses{def:chain_copy_product,def:chain_open_mps,thm:chain_copy_error_amplification}
    For $K\ge1$, a unit block vector $\eta$, and a unit MPS $\psi$ on $KL$ sites
    of bond dimension at most $D$, if
    $F=\bigl|\bigl\langle\bigotimes_{i=0}^{K-1}\eta,\psi\bigr\rangle\bigr|^2$
    satisfies $T(F)<1$, then there exists a unit block MPS $\theta$ of bond
    dimension at most $D$ such that $f=|\langle\eta,\theta\rangle|^2$ satisfies
    \begin{align}
        \label{eq:chain_mps_copy_error_amplification}
        T(f^K)\le T(F),\qquad \sqrt{K/4}\,T(f)\le T(F).
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chain_unit_mps_copy_replacement,thm:chain_copy_product_inner,thm:chain_copy_error_amplification}
    Combine Theorems~\ref{thm:chain_unit_mps_copy_replacement},
    \ref{thm:chain_copy_product_inner}, and~\ref{thm:chain_copy_error_amplification}.
\end{proof}
