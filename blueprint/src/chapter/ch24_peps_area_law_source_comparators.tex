\section{Upper and lower norm comparisons from an auxiliary pair}
\label{sec:al_comparators_source}

The norm comparisons of \cite[Proposition~8.1]{OpenAI2026AreaLaw} apply to a finite product of finite-dimensional local spaces, using a single label sequence across all status families, weights, and mean trees, with remainders uniform in the tree, terminal probabilities, and symmetric vector.

\begin{definition}[Statement of the norm comparisons]
    \label{def:al_comparators_source_statement}
    \lean{TNLean.PEPS.AreaLaw.Scan.Comparators.IsTypicalSchmidtSet,
        TNLean.PEPS.AreaLaw.Scan.Comparators.IsStatusFamily,
        TNLean.PEPS.AreaLaw.Scan.Comparators.IsPrefixData,
        TNLean.PEPS.AreaLaw.Scan.Comparators.IsMarginalMomentData,
        TNLean.PEPS.AreaLaw.Scan.Comparators.prevector,
        TNLean.PEPS.AreaLaw.Scan.Comparators.bandMetric,
        TNLean.PEPS.AreaLaw.Scan.Comparators.meanRoot,
        TNLean.PEPS.AreaLaw.Scan.Comparators.ComparatorsStatement}
    \leanok
    On a finite-site tensor-product space $\mathcal H_{\rm phys}$, let $\widetilde H\ge0$ satisfy $\widetilde H\widetilde\Omega=\widetilde E_0\widetilde\Omega$ and $\widetilde H-\widetilde E_0\ge\widetilde g(1-|\widetilde\Omega\rangle\langle\widetilde\Omega|)$ for a unit vector $\widetilde\Omega$ and gap $\widetilde g>0$.
    For a site subset $X$, a typical Schmidt set is a nonempty index set $E$ of positive eigenvalues $\{p_i\}_{i\in E}$ of $\rho_X$ satisfying $e^{-S_X-w}\le p_i\le e^{-S_X+w}$ ($w\ge0$) and $z=\sum_{i\in E}p_i>0$; set $d_*=|E|$, $p_i'=p_i/z$, and $S'=-\sum_{i\in E}p_i'\log p_i'$.
    Appending $d_*$-dimensional systems $C,R$ with $\mathrm{Bell}_{CR}=d_*^{-1/2}\sum_i|i\rangle_C|i\rangle_R$, $\mathcal V=\mathcal H_{\rm phys}\otimes\mathcal H_C\otimes\mathcal H_R$, $\mathcal S_k=\operatorname{Sym}^k\mathcal V$, and symmetric-group projections $\pi^\lambda_R,\pi^\lambda_C$, set
    \begin{align}
        \mathrm{pre}_k=\pi_R^{\lambda_k}
        (\widetilde\Omega\otimes\mathrm{Bell}_{CR})^{\otimes k},
        \qquad
        \overline{\widetilde H}=\frac1k\sum_{i=1}^k\widetilde H^{(i)}.\notag
    \end{align}

    A status family is a finite set of statuses $j$, each comprising $K$ site partitions $(X,U_{j,g},Y_{j,g},V_{j,g})$ ($1\le g\le K$) with $XU_{j,g}Y_{j,g}\subseteq XU_{j',g'}$ for all $j,j'$ and $g<g'$.
    For $a>0$, set $t=a/2$, $s=1/4$, $W=aK$, and with the parameter-$t$ Schur metrics $W_Q$, define
    \begin{align}
        A_{j,g}=(W_{CXU_{j,g}}^{-1}W_{RV_{j,g}}^{-1}W_{Y_{j,g}})^2,
        \qquad A_j=\prod_{g=1}^KA_{j,g}.\notag
    \end{align}
    Let $\mathcal M$ be the root on $\mathcal S_k$ of a finite binary mean tree with leaf operators $A_j$ and terminal probabilities $w_j$, and write $\langle f\rangle_{\rm av}=K^{-1}\sum_{g,j}w_jf_{j,g}$.
    For $\xi\in\mathcal S_k$, set $N=\|\mathcal M^{-s}\xi\|$, $v=\mathcal M^{-s}\xi/N$, $E_{\rm def}=(\langle v,\overline{\widetilde H}v\rangle-\widetilde E_0)/\widetilde g$,
    \begin{align}
        g^{\rm pre}_{j,g}  =S(XU_{j,g})+S(V_{j,g})-S(Y_{j,g}),\qquad
        g^{\rm post}_{j,g} =S(U_{j,g})+S(U_{j,g}Y_{j,g})-S(Y_{j,g}), \notag
    \end{align}
    and $g_{\max}=\max(\{0\}\cup\{g^{\rm pre}_{j,g}\}_{j,g})$ over the tree leaves.

    Prefix data consist of a finite nested family $\mathcal R$ disjoint from $X$ and bounds $B_{\rm sh},B_{\rm exc}$ with $S(B)\le B_{\rm sh}$ ($\forall B\in\mathcal R$) such that every $Q\in\{U_{j,g},U_{j,g}Y_{j,g}\}$ admits $B\in\mathcal R$ with $Q\subseteq B$ or $B\subseteq Q$ and $1+\log\dim\mathcal H_{B\triangle Q}\le B_{\rm exc}$.
    Moment data with constants $C_0,c_0$ consist of $\mathcal B\ge1$ such that every $Q\in\{XU_{j,g},V_{j,g},Y_{j,g},U_{j,g},U_{j,g}Y_{j,g}\}$ obeys
    \begin{align}
        \log\Tr\rho_Q^{1-u}\le uS(Q)+C_0\mathcal Bu^2,
        \qquad |u|\le c_0/\sqrt{\mathcal B},\notag
    \end{align}
    tracing over positive eigenvalues of $\rho_Q$.

    The norm-comparison statement asserts the existence of universal constants $a_0,C>0$ and, for each $C_0,c_0>0$, constants $c,C'>0$ such that for every $(\widetilde H,X,E)$ there is a label sequence $\lambda_k$ satisfying $0<\|\mathrm{pre}_k\|\le1$, $\mathrm{pre}_k\in\mathcal S_k$, $\overline{\widetilde H}\,\mathrm{pre}_k=\widetilde E_0\,\mathrm{pre}_k$, and $\pi_C^{\lambda_k}\mathrm{pre}_k=\pi_R^{\lambda_k}\mathrm{pre}_k=\mathrm{pre}_k$ for all $k>0$; for every site partition $(X,U,Y,V)$,
    \begin{align}
        2S_X+g^{\rm pre}-g^{\rm post}=I(X:V)+I(X:YV);
        \label{eq:al_comparators_source_identity}
    \end{align}
    and for every status family and $0<a\le a_0$, $g^{\rm pre}_{j,g}\ge0$, $N>0$, and (with $o_k(1)\to0$ independent of the tree, $w_j$, isometric coordinates of $\mathcal S_k$, and $\xi$):
    \begin{enumerate}
        \item For prefix data, $\xi=\mathrm{pre}_k$, and $k\gg1$,
              \begin{align}
                  -\frac{\log N^2}{k}\le2\log d_*-\log z
                  +2sW\Bigl(\frac{2B_{\rm sh}}z+2B_{\rm exc}\Bigr)+o_k(1).
                  \label{eq:al_comparators_source_rough_upper}
              \end{align}
        \item For all $k>0$, $0<\tau<1/2$, and $\xi\in\mathcal S_k$ with $0<\|\xi\|\le1$ fixed by $\pi_C^{\lambda_k},\pi_R^{\lambda_k}$ (and binary entropy $h$),
              \begin{align}
                  -\frac{\log N^2}{k}\ge2sW\Bigl[2S'
                      -C\Bigl(\tau\log d_*+\frac{h(\tau)}a\Bigr)
                      -C\frac{E_{\rm def}}{\tau}S'\Bigr]-o_k(1).
                  \label{eq:al_comparators_source_rough_lower}
              \end{align}
        \item For moment data with $a\le c/\sqrt{\mathcal B}$: for $\xi=\mathrm{pre}_k$ and $k\gg1$,
              \begin{align}
                  -\frac{\log N^2}{k}\le2\log d_*-\log z
                  +2sW\Bigl[\langle g^{\rm post}\rangle_{\rm av}
                      +C'\Bigl(a\mathcal B+\frac{|\log z|}a\Bigr)\Bigr]+o_k(1),
                  \label{eq:al_comparators_source_sharp_upper}
              \end{align}
              and for all $k>0$, $0<\tau<1/2$, and $\xi$ as in (\ref{eq:al_comparators_source_rough_lower}),
              \begin{align}
                  -\frac{\log N^2}{k}\ge2sW\Bigl[2S'+(1-\tau)
                      \langle g^{\rm pre}\rangle_{\rm av}
                      -C'\Bigl(\tau\log d_*+\frac{h(\tau)}a+a\mathcal B\Bigr)
                      -C'\frac{E_{\rm def}}\tau(S'+g_{\max})\Bigr]-o_k(1).
                  \label{eq:al_comparators_source_sharp_lower}
              \end{align}
    \end{enumerate}
\end{definition}

\begin{theorem}[Upper and lower norm comparisons]
    \label{thm:al_comparators_source}
    \lean{TNLean.PEPS.AreaLaw.Scan.Comparators.comparatorsStatement}
    \leanok
    \uses{def:al_comparators_source_statement}
    Definition~\ref{def:al_comparators_source_statement} (\cite[Proposition~8.1]{OpenAI2026AreaLaw}) holds with $a_0=1/10$, $C=4$, $C'=4+25C_0$, and $c=c_0/10$, with $\lambda_k$ chosen before the status family and $a$, and (\ref{eq:al_comparators_source_rough_lower}), (\ref{eq:al_comparators_source_sharp_lower}) valid for all $\xi\in\mathcal S_k$ of norm $0<\|\xi\|\le1$ fixed by $\pi_C^{\lambda_k},\pi_R^{\lambda_k}$ (in particular $\xi=\mathrm{pre}_k$).
\end{theorem}

\begin{proof}
    \leanok
    Choosing $\lambda_k$ from the normalized typical Schmidt vector (whose auxiliary marginal has entropy $S'$) gives $\log\dim\lambda_k=kS'+o(k)$ and inverse-polynomial label mass in $k$.
    Each $A_{j,g}$ preserves $\mathcal S_k$ and commutes across $g$ after compression, so $\mathcal M=\prod_{g=1}^K\mathcal M_g$ where $\mathcal M_g$ has leaf operators $A_{j,g}$.

    At $t=a/2$ and $s=1/4$, (\ref{eq:al_comparators_source_rough_upper})--(\ref{eq:al_comparators_source_sharp_lower}) follow from the regional bounds of \cite[Section~8]{OpenAI2026AreaLaw}: (\ref{eq:al_comparators_source_rough_upper}) uses prefix data; (\ref{eq:al_comparators_source_sharp_upper}) uses moment data on $\{U_{j,g},U_{j,g}Y_{j,g},Y_{j,g}\}$ with $6a\le c_0/\sqrt{\mathcal B}$; and (\ref{eq:al_comparators_source_rough_lower}), (\ref{eq:al_comparators_source_sharp_lower}) use the gap $\widetilde g$, $2kS'-\epsilon_k\le2\log\dim\lambda_k$ ($\epsilon_k=2|\log\dim\lambda_k-kS'|=o(k)$), and (for (\ref{eq:al_comparators_source_sharp_lower})) moment data on $\{XU_{j,g},Y_{j,g},V_{j,g}\}$ with $10a\le c_0/\sqrt{\mathcal B}$.
    The finite-$k$ error $O((\log k+a\epsilon_k)/k)=o_k(1)$, and for $a\le1$ and $S',g_{\max},E_{\rm def}\ge0$,
    \begin{align}
        2\tau\log d_*+(3+a^{-1})h(\tau)  & \le 4(\tau\log d_*+h(\tau)/a),         &
        25C_0\mathcal Ba                 & \le C'a\mathcal B, \notag                \\
        (E_{\rm def}/\tau)(2S'+g_{\max}) & \le C'(E_{\rm def}/\tau)(S'+g_{\max}), &
        6KC_0\mathcal Ba^2               & \le 2sWC'a\mathcal B, \notag
    \end{align}
    together with $-(K/2)\log z\le2sWC'|\log z|/a$.
    Purity of $\widetilde\Omega$ gives (\ref{eq:al_comparators_source_identity}), and subadditivity gives $g^{\rm pre}_{j,g}=I(XU_{j,g}:V_{j,g})\ge0$.
\end{proof}
