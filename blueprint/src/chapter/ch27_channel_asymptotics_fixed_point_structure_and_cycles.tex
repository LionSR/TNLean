\section{Further irreducibility and primitivity equivalences}

This section collects criteria and equivalences from
\cite[Theorems~6.7 and~6.8]{Wolf2012Quantum} that are used later. The
completely positive exponential characterization corresponding to Wolf's
Theorem~6.2 appears in the corresponding theorem in the companion quantum-channel volume~\cite[``Exponential characterization of irreducibility'']{QICLean2026Blueprint}.


\begin{theorem}[Kraus-span equivalence]%
    \label{thm:wolf_6_8_kraus_span}
    \lean{MPSTensor.wolf_theorem_6_8_kraus_span}
    \leanok
    Under normalization, primitivity in Wolf's sense is equivalent to eventual
    full Kraus rank.
\end{theorem}

\begin{theorem}[Combined primitivity equivalence]%
    \label{thm:wolf_6_8_conjunction}
    \lean{MPSTensor.wolf_theorem_6_8_conjunction}
    \leanok
    \uses{thm:wolf_6_8_kraus_span}
    Under normalization, primitivity in Wolf's sense is equivalent to the conjunction
    of eventual full Kraus rank, normality, and strong irreducibility.
\end{theorem}

\begin{remark}[Eventually full Kraus rank and normality]
    \label{rem:wolf_6_8_pairwise}
    \lean{MPSTensor.hasEventuallyFullKrausRank_iff_isNormal}
    \leanok
    Eventual full Kraus rank is equivalent to normality. The remaining
    pairwise primitivity equivalences are stated in
    Theorem~\ref{thm:wielandt_proposition_three}.
\end{remark}
