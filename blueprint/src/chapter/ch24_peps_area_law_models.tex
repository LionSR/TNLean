\section{Finite-domain area laws and polynomial PEPS approximation}
\label{sec:finite_domain_area_law}

An entropy bound for every cut and an approximation of the entire vector are
different assertions. The first concerns arbitrary finite induced lattice
domains. The second asks for one compatible tensor representation on the
original open square. We state the two models separately, following
\cite{OpenAI2026AreaLaw,OpenAI2026PolynomialPEPS}.

\begin{definition}[Induced lattice domain]
    \label{def:area_law_domain}
    \lean{
        TNLean.Lattice.Site,
        TNLean.Lattice.Configuration,
        TNLean.Lattice.StateSpace,
        TNLean.Lattice.domainGraph,
        TNLean.Lattice.graphNeighborhood,
        TNLean.Lattice.configurationSplit,
        TNLean.Lattice.card_configuration,
        TNLean.Lattice.card_regionConfig
    }
    \leanok
    Let $\Lambda\subset\mathbb Z^2$ be finite. Join sites at coordinate distance
    one, and let $d_\Lambda$ be distance in this induced graph, with value
    $+\infty$ between distinct components. For $X\subseteq\Lambda$ and an integer
    $r\ge0$, the closed graph neighborhood
    \begin{align}
        N_r^\Lambda(X) & =\{y\in\Lambda:\ d_\Lambda(x,y)\le r
        \text{ for some }x\in X\}
        \label{eq:area_law_graph_neighborhood}
    \end{align}
    consists of the sites joined to some site of $X$ by an induced walk of
    length at most $r$. At every site put $\mathbb C^q$,
    $q\ge1$. In configuration coordinates the physical Hilbert space is
    \begin{align}
        \label{eq:area_law_state_space}
        \mathcal H_\Lambda & =\mathbb C^{\{\sigma:\Lambda\to\{0,\ldots,q-1\}\}},
    \end{align}
    \begin{align}
        \label{eq:area_law_dimension}
        \dim\mathcal H_\Lambda & =q^{|\Lambda|}.
    \end{align}
    Restriction to $A$ and $\Lambda\setminus A$ identifies the configuration set
    with the product of the two regional configuration sets. The configuration
    set of a region $A$ has $q^{|A|}$ elements. Empty domains,
    holes, and disconnected components are permitted.
\end{definition}

\begin{definition}[Finite-range interactions and boundary]
    \label{def:area_law_hamiltonian}
    \lean{
        TNLean.Lattice.IsAdmissibleSupport,
        TNLean.Lattice.AdmissibleSupport,
        TNLean.Lattice.LocalHamiltonian,
        TNLean.Lattice.LocalHamiltonian.operator,
        TNLean.Lattice.LocalHamiltonian.operator_isHermitian,
        TNLean.Lattice.LocalHamiltonian.zero,
        TNLean.Lattice.localLift,
        TNLean.Lattice.localLift_mem_supportedOperators,
        TNLean.Lattice.localLift_eq_entropyLocalLift,
        TNLean.Lattice.edgeBoundary,
        TNLean.Lattice.edgeBoundary_empty,
        TNLean.Lattice.edgeBoundary_univ
    }
    \leanok
    \uses{def:area_law_domain}
    Fix an integer $R\ge0$ and $J>0$. For each nonempty
    $X\subseteq\Lambda$ with $\operatorname{diam}_{d_\Lambda}X\le R$, assign
    one Hermitian operator $h_X$ supported on $X$, with $\|h_X\|\le J$.
    The assigned operator may be zero, and acts as the identity on the
    complementary tensor factors. Define
    \begin{align}
        \label{eq:area_law_hamiltonian}
        H & =\sum_{\varnothing\ne X\subseteq\Lambda,\ \operatorname{diam}_{d_\Lambda}X\le R}h_X,
    \end{align}
    \begin{align}
        \label{eq:area_law_boundary}
        \partial_\Lambda A & =
        \{\{x,y\}:x\in A,\ y\in\Lambda\setminus A,\ d_\Lambda(x,y)=1\}.
    \end{align}
    The norm is the Euclidean operator norm. Boundary edges are unordered and
    counted once; empty and full cuts have empty boundary.
\end{definition}

\begin{theorem}[Interaction range and rectangular coordinates]
    \label{thm:area_law_domain_identifications}
    \lean{
        TNLean.Lattice.exists_walk_length_le_iff_edist_le,
        TNLean.Lattice.isAdmissibleSupport_zero_iff,
        TNLean.Lattice.isAdmissibleSupport_singleton,
        TNLean.Lattice.isAdmissibleSupport_pair_of_adj,
        TNLean.PEPS.AreaLaw.rectangleEmbedding,
        TNLean.PEPS.AreaLaw.rectangleEmbedding_injective,
        TNLean.PEPS.AreaLaw.rectangularDomain,
        TNLean.PEPS.AreaLaw.rectangleSiteEquiv,
        TNLean.PEPS.AreaLaw.rectangleSiteEquiv_val,
        TNLean.PEPS.AreaLaw.rectangleGraphIso,
        TNLean.PEPS.AreaLaw.rectangleConfigurationEquiv,
        TNLean.PEPS.AreaLaw.isAdmissibleSupport_rectangle_pair_of_adj
    }
    \leanok
    \uses{def:area_law_domain,def:area_law_hamiltonian}
    A pair has induced distance at most $R$ if and only if an induced walk of
    length at most $R$ joins it. Every singleton is an admissible support at
    every range, and at range zero the admissible supports are exactly the
    singletons. Adjacent pairs are admissible at range one.
    The nearest-neighbor graph on
    $\{0,\ldots,w-1\}\times\{0,\ldots,h-1\}$ is the induced graph of that
    integer rectangle, with the corresponding identification of configurations
    and range-one pair supports.
\end{theorem}
\begin{proof}\leanok
    The extended graph distance is the least walk length, or infinity when no
    walk exists. A zero-length walk has equal endpoints. A nonempty support of
    diameter zero therefore has one element. The coordinate embedding preserves
    horizontal and vertical nearest neighbors and is bijective onto the integer
    rectangle. Restricting configurations along this bijection gives the last
    identification.
\end{proof}

\begin{definition}[Ground vector and regional entropy]
    \label{def:area_law_ground_entropy}
    \lean{
        Matrix.IsGappedGroundState,
        Matrix.isGappedGroundState_iff,
        TNLean.Lattice.IsGappedGroundState,
        TNLean.Lattice.reducedState,
        TNLean.Lattice.reducedState_isHermitian,
        TNLean.Lattice.regionalEntropy,
        TNLean.Lattice.reducedState_eq_regionState,
        TNLean.Lattice.regionalEntropy_eq_regionEntropy,
        TNLean.PEPS.AreaLaw.UniformAreaLaw
    }
    \leanok
    \uses{def:area_law_hamiltonian}
    For a unit vector $\Omega$, a real $E_0$, and $\Delta>0$, the eigenvector and
    full-system projector-gap conditions are
    \begin{align}
        \label{eq:area_law_ground_vector}
        H\Omega & =E_0\Omega,
    \end{align}
    \begin{align}
        \label{eq:area_law_projector_gap}
        H-E_0I & \ge\Delta(I-|\Omega\rangle\langle\Omega|).
    \end{align}
    For $A\subseteq\Lambda$, put
    \begin{align}
        \label{eq:area_law_reduced_state}
        \rho_A & =\operatorname{Tr}_{\Lambda\setminus A}|\Omega\rangle\langle\Omega|,
    \end{align}
    \begin{align}
        \label{eq:area_law_regional_entropy}
        S_\Omega(A) & =-\operatorname{Tr}\rho_A\log\rho_A.
    \end{align}
    All logarithms are natural, with $0\log0=0$. The reduced matrix is Hermitian,
    including when its spectrum contains zero. No condition is imposed on the
    gap of a Hamiltonian restricted to a smaller region.
    The uniform boundary assertion requires a single finite constant for each
    $q,R,J,\Delta$, chosen before every domain, Hamiltonian, ground vector, and cut.
\end{definition}

\begin{theorem}[Uniqueness from a positive global gap]
    \label{thm:area_law_ground_eigenspace}
    \lean{Matrix.IsGappedGroundState.eigenspace_eq_span}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $H$ be a complex matrix indexed by a finite set, let $E_0$ be real,
    and let $\Omega$ be a unit vector satisfying
    (\ref{eq:area_law_ground_vector}) and
    (\ref{eq:area_law_projector_gap}) with $\Delta>0$.
    As an elementary consequence of the global gap giving the uniqueness
    assertion in \cite[Proposition~4.5]{OpenAI2026AreaLaw}, the eigenspace of
    $H$ at $E_0$ is $\operatorname{span}_{\mathbb C}\{\Omega\}$.
\end{theorem}
\begin{proof}\leanok
    For an eigenvector $\psi$ at $E_0$, the gap inequality gives
    $0\ge\Delta\|\psi-\langle\Omega,\psi\rangle\Omega\|^2$.
    Since $\Delta>0$, $\psi=\langle\Omega,\psi\rangle\Omega$.
    Conversely, $H\Omega=E_0\Omega$ puts every scalar multiple of $\Omega$
    in this eigenspace.
\end{proof}

\begin{theorem}[Regional reductions of pure vectors]
    \label{thm:area_law_regional_state_identities}
    \lean{
        TNLean.Lattice.reducedState_posSemidef,
        TNLean.PEPS.AreaLaw.trace_reducedState_eq_norm_sq,
        TNLean.PEPS.AreaLaw.trace_reducedState,
        TNLean.PEPS.AreaLaw.regionalEntropy_nonneg,
        TNLean.PEPS.AreaLaw.regionalEntropy_empty,
        TNLean.PEPS.AreaLaw.regionalEntropy_univ
    }
    \leanok
    \uses{def:area_law_domain,def:area_law_ground_entropy}
    Let $\Lambda\subset\mathbb Z^2$ be finite, let $q\ge0$ be an integer,
    and let $\Omega\in\mathcal H_\Lambda$. Under the density and entropy
    conventions of \cite[Section~2]{OpenAI2026AreaLaw}, for every
    $A\subseteq\Lambda$, the regional reduction defined in
    Definition~\ref{def:area_law_ground_entropy} satisfies
    \begin{align}
        \rho_A                  & \ge0,
        \label{eq:area_law_reduced_positive}     \\
        \operatorname{Tr}\rho_A & =\|\Omega\|^2.
        \label{eq:area_law_reduced_trace}
    \end{align}
    If $\|\Omega\|=1$, then for every $A\subseteq\Lambda$,
    \begin{align}
        \operatorname{Tr}\rho_A & =1,
        \label{eq:area_law_reduced_unit_trace} \\
        S_\Omega(A)             & \ge0.
        \label{eq:area_law_entropy_nonnegative}
    \end{align}
    If $\|\Omega\|=1$, the empty region and the full region also satisfy
    \begin{align}
        S_\Omega(\varnothing) & =0,
        \label{eq:area_law_empty_entropy} \\
        S_\Omega(\Lambda)     & =0.
        \label{eq:area_law_full_entropy}
    \end{align}
    No Hamiltonian or spectral-gap hypothesis is required. The identities include
    empty domains and regions and use the one-dimensional empty tensor product.
    The site dimension $q$ may here be zero, in which case $\mathcal H_\Lambda$
    is the zero space for nonempty $\Lambda$; the standing assumption $q\ge1$
    of Definition~\ref{def:area_law_domain} is not needed.
    % Source revision: openai/math@adc7f1241b42e322a6451854ab7e4b4c146bf78a,
    % build/sections/01-preliminaries.tex, sec:prelim, lines 10--25.
    % This entry records regional-state identities only; it does not mark a
    % source-labelled continuity, gap-equivalence, or area-law theorem complete.
\end{theorem}
\begin{proof}\leanok
    The pure-state matrix $|\Omega\rangle\langle\Omega|$ is positive semidefinite.
    Restriction of configurations to $A$ and its complement changes only the
    coordinates, and partial trace preserves positivity and trace. Thus
    $\operatorname{Tr}\rho_A=\operatorname{Tr}|\Omega\rangle\langle\Omega|
        =\|\Omega\|^2$. For a unit vector, the reduced matrix is positive with trace
    one. Its eigenvalues lie in $[0,1]$, so each term $-\lambda\log\lambda$ in
    its entropy is nonnegative, with $0\log0=0$.

    For the empty region, the reduced matrix acts on a one-dimensional space.
    For the full region, write $\rho_\Lambda=MM^*$ using the coefficient matrix
    of the vector under the configuration splitting. The complement has one
    configuration, so $M$ has one column and $\rho_\Lambda$ has rank at most one.
    A positive trace-one matrix of rank at most one has one eigenvalue equal to
    one and all remaining eigenvalues zero. Its entropy therefore vanishes in
    both cases.
\end{proof}

\begin{theorem}[Marginal trace norm from the full-system gap]
    \label{thm:peps_gap_marginal_trace_norm}
    \lean{FiniteProduct.re_trace_abs_reducedPure_sub_le_sqrt_energy}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $\mathcal H=\bigotimes_{v\in V}\mathcal H_v$ be a finite tensor product
    of finite-dimensional complex Hilbert spaces. Let $H$ be an operator on
    $\mathcal H$ and let $E_0\in\R$. Suppose that $\Omega$ is a unit vector
    and $\Delta>0$, with
    \begin{align}
        H\Omega & = E_0\Omega, \notag                         \\
        H-E_0I  & \geq\Delta(I-|\Omega\rangle\langle\Omega|).
        \label{eq:peps_gap_projector_bound}
    \end{align}
    For a unit vector $\phi$ and a region $X\subseteq V$, put
    $\rho_X^\phi=\operatorname{Tr}_{V\setminus X}|\phi\rangle\langle\phi|$
    and define $\rho_X^\Omega$ in the same way. As in the gap-to-marginal
    estimate in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L339-L365}{proof of Proposition 4.1},
    \begin{align}
        \|\rho_X^\phi-\rho_X^\Omega\|_1
         & \leq 2\sqrt{\frac{\operatorname{Re}\langle\phi,H\phi\rangle-E_0}{\Delta}}.
        \label{eq:peps_gap_marginal_distance}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:al_amp_pure_trace_distance}
    With $e=\operatorname{Re}\langle\phi,H\phi\rangle-E_0$, taking the
    expectation of~(\ref{eq:peps_gap_projector_bound}) in $\phi$ gives
    $1-|\langle\Omega,\phi\rangle|^2\leq e/\Delta$.
    Contractivity under partial trace and the trace norm formula for two pure
    states give
    \begin{align}
        \|\rho_X^\phi-\rho_X^\Omega\|_1
         & \leq\bigl\||\phi\rangle\langle\phi|-|\Omega\rangle\langle\Omega|\bigr\|_1
        =2\sqrt{1-|\langle\Omega,\phi\rangle|^2}
        \leq2\sqrt{e/\Delta}.
    \end{align}
\end{proof}

\begin{theorem}[A sufficient energy bound for marginal approximation]
    \label{thm:peps_gap_marginal_one_sixteenth}
    \lean{FiniteProduct.re_trace_abs_reducedPure_sub_le_one_sixteenth_of_energy}
    \leanok
    \uses{thm:peps_gap_marginal_trace_norm}
    Under the hypotheses of Theorem~\ref{thm:peps_gap_marginal_trace_norm}, suppose
    in addition that
    \begin{align}
        \operatorname{Re}\langle\phi,H\phi\rangle-E_0\leq\frac{\Delta}{1024}.
    \end{align}
    Then, for every region $X\subseteq V$,
    \begin{align}
        \|\rho_X^\phi-\rho_X^\Omega\|_1\leq\frac1{16}.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_gap_marginal_trace_norm}
    Since $\Delta>0$, the quotient of the excess energy by $\Delta$ is at
    most $1/1024$. Theorem~\ref{thm:peps_gap_marginal_trace_norm} bounds the
    full trace norm by $2\sqrt{1/1024}=1/16$.
\end{proof}

\begin{theorem}[Uniform finite-domain area law]
    \label{thm:openai_uniform_area_law}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw}
    \leanok
    \uses{def:area_law_domain,def:area_law_hamiltonian,def:area_law_ground_entropy}
    In Theorem~1.1 of \cite{OpenAI2026AreaLaw}, for fixed integers $q\ge1$,
    $R\ge0$ and reals $J,\Delta>0$, there is a finite constant
    $C=C(q,R,J,\Delta)$ such that every finite induced domain with the
    above interaction bounds and a unique unit ground vector, up to phase,
    satisfying the full-system projector gap obeys
    $S_\Omega(A)\le C|\partial_\Lambda A|$ for every $A\subseteq\Lambda$.
    The same constant applies to all domains, interaction families, and cuts.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_closed_constant}
    Theorem~\ref{thm:area_law_closed_constant} has the same hypotheses and gives
    the bound with $C=C_{\rm AL}(q,R,J,\Delta)$. That constant is assembled
    along the following argument.

    For disjoint $B,C\subseteq\Lambda$ write
    $I_\Omega(B:C)=S_\Omega(B)+S_\Omega(C)-S_\Omega(B\cup C)$, and let $Z$ be the
    set of endpoints of the edges in $\partial_\Lambda A$. Empty and full cuts
    have zero entropy, and so does every cut with $\partial_\Lambda A=\varnothing$.
    Put $b=|\partial_\Lambda A|>0$.

    The global gap first gives an exponent below volume: for a box $Q$ of side
    at most $r$, sufficiently far from $Z$,
    \begin{align}
        S_\Omega(A\cap Q) & \le Cr^{1+e_0},\qquad e_0<1.
        \label{eq:area_law_initial_box}
    \end{align}
    Spectral filtering replaces $H$ by positive quasi-local constraints that
    annihilate $\Omega$ and keep a global gap. A replica argument then compares
    two bounds on a filtered norm while a finite randomized scan moves the
    partition through a collar. Repeating the comparison improves the exponent
    in (\ref{eq:area_law_initial_box}) and gives, for a template $T$ at scale
    $n$ and $L_n=\lfloor n^{1-\varepsilon}\rfloor$,
    \begin{align}
        I_\Omega\bigl(A\cap T:\Lambda\setminus(A\cap T_{L_n})\bigr)
         & \le Cn^{1-\varepsilon}.
        \label{eq:area_law_collar_information}
    \end{align}
    Here $\varepsilon$, $\alpha$ and $\beta$ are the exponents of
    Theorem~\ref{thm:area_law_exponent_gaps}. Products of the positive
    constraints amplify this bound. For every $M\ge1$
    there is a radius $r\le C_Mn^{(1-\varepsilon)/\alpha}$ with
    $L_n+2r=o(n^\beta)$ such that
    \begin{align}
        I_\Omega(A\cap T:J)
         & \le C_Mn^{-M}
        \qquad\text{for every }
        J\subseteq\Lambda\setminus N_{2r}^\Lambda(A\cap T_{L_n}).
        \label{eq:area_law_amplified_information}
    \end{align}

    The planar construction partitions $A$ into a residual set $D$ and an
    ordered family of pieces $X_i\subseteq A\cap T_i$ with labels
    $f_i\in\{0,1\}$, templates $T_i$ at scales $n_i$, and
    \begin{align}
        |D| & \le Cb,
        \qquad
        \sum_in_i^{-100}\le Cb,
        \label{eq:area_law_partition_counts} \\
        \operatorname{dist}_\infty(T_i,Z)
            & >2n_i^\beta,
        \qquad
        \operatorname{dist}_\infty\Bigl(T_i,
        \bigcup_{j<i,\,f_j=f_i}X_j\Bigr)>2n_i^\beta.
        \label{eq:area_law_partition_clearance}
    \end{align}
    Put $S_i=A\cap(T_i)_{L_{n_i}}$ and
    $J_i=(\Lambda\setminus A)\cup\bigcup_{j<i,\,f_j=f_i}X_j$, and let $r_i$ be
    the radius in (\ref{eq:area_law_amplified_information}) at scale $n_i$, with
    the scale threshold large enough that $L_{n_i}+2r_i<n_i^\beta$. A walk of
    length at most $2r_i$ starting in $S_i$ stays within ambient distance
    $n_i^\beta$ of $T_i$. By (\ref{eq:area_law_partition_clearance}) it meets no
    earlier piece of the same family, and it cannot leave $A$, since its first
    step out of $A$ would end at a point of $Z$. Hence
    $J_i\cap N_{2r_i}^\Lambda(S_i)=\varnothing$. With $M=100$,
    (\ref{eq:area_law_amplified_information}) and monotonicity of mutual
    information give
    \begin{align}
        I_\Omega(X_i:J_i)\le I_\Omega(A\cap T_i:J_i)
         & \le Cn_i^{-100}=:\eta_i.
        \label{eq:area_law_piece_information}
    \end{align}
    For one family $f$, with $A_f=\bigcup_{f_i=f}X_i$ and
    $E_f=\sum_{f_i=f}\eta_i$, the chain rule, purity and subadditivity
    give
    \begin{align}
        S_\Omega(A)+\sum_{f_i=f}S_\Omega(X_i)-E_f
         & \le S_\Omega\bigl((\Lambda\setminus A)\cup A_f\bigr)
        =S_\Omega(D\cup A_{1-f})
        \nonumber                                               \\
         & \le S_\Omega(D)+\sum_{f_i=1-f}S_\Omega(X_i).
        \label{eq:area_law_family_chain}
    \end{align}
    Adding (\ref{eq:area_law_family_chain}) for $f=0$ and $f=1$ cancels every
    piece entropy. With (\ref{eq:area_law_partition_counts}),
    \begin{align}
        S_\Omega(A)\le S_\Omega(D)+\frac12\sum_i\eta_i
         & \le|D|\log q+\frac C2\sum_in_i^{-100}\le C'b.
        \label{eq:area_law_final_bound}
    \end{align}
    Every threshold and constant above depends only on $q,R,J,\Delta$, and is
    chosen before the domain and the cut.
\end{proof}

\begin{definition}[Open-square Hamiltonian and PEPS approximation]
    \label{def:polynomial_peps_model}
    \lean{
        TNLean.PEPS.Approximation.Configuration,
        TNLean.PEPS.Approximation.StateSpace,
        TNLean.PEPS.Approximation.SquareHamiltonian,
        TNLean.PEPS.Approximation.SquareHamiltonian.operator,
        TNLean.PEPS.Approximation.SquareHamiltonian.IsGappedGroundState,
        TNLean.PEPS.Approximation.pepsVector,
        TNLean.PEPS.Approximation.HasPEPSApproximation,
        TNLean.PEPS.Approximation.PolynomialPEPSApproximation
    }
    \leanok
    On the original open $L\times L$ graph, let
    \begin{align}
        H       & =\sum_v h_v+\sum_e h_e,
        \label{eq:polynomial_peps_hamiltonian} \\
        \|h_v\| & \le J,
        \label{eq:polynomial_peps_site_bound}  \\
        \|h_e\| & \le J,
        \label{eq:polynomial_peps_edge_bound}
    \end{align}
    where terms are supported on their designated sites or edges, and only the
    total $H$ is assumed Hermitian. For a unit vector $\Omega$, a real $E_0$,
    and $\Delta>0$, the gapped ground-vector condition for this $H$ is
    \begin{align}
        H\Omega=E_0\Omega,
        \qquad
        H-E_0I & \ge\Delta(I-|\Omega\rangle\langle\Omega|).
        \label{eq:polynomial_peps_projector_gap}
    \end{align}
    A PEPS approximation with parameters $C,c$
    is a tensor family on those same edges with positive dimensions $D_e$
    satisfying $D_e\le CL^c$ for every edge $e$, and a nonzero contracted vector
    $\Phi$ such that, for some $\theta\in\mathbb R$,
    \begin{align}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
        \label{eq:polynomial_peps_vector_error}
    \end{align}
    The norm in (\ref{eq:polynomial_peps_vector_error}) is the Euclidean norm
    of the entire vector.
    The local tensor maps need not be isometries and the bond dimensions need not
    be equal. The uniform approximation assertion requires $C,c>0$ to be chosen
    for fixed $q\ge2$, $J,\Delta>0$ before all $L\ge2$, Hamiltonians, and ground vectors.
\end{definition}

\begin{lemma}[Minimizing phase]
    \label{lem:polynomial_peps_minimizing_phase}
    \lean{
        TNLean.PEPS.Approximation.exists_phase_isMinOn,
        TNLean.PEPS.Approximation.exists_phase_le_iff_iInf_le,
        TNLean.PEPS.Approximation.hasPEPSApproximation_iff_iInf
    }
    \leanok
    For vectors $u,v$ in a complex normed space, the function
    $\theta\mapsto\|u-e^{i\theta}v\|$ attains its minimum over $\mathbb R$.
    Consequently a bound $\varepsilon$ holds at some phase if and only if
    \begin{align}
        \min_{\theta\in\mathbb R}\|u-e^{i\theta}v\| & \le\varepsilon,
    \end{align}
    and (\ref{eq:polynomial_peps_vector_error}) holds for some $\theta$ if and only
    if it holds with the minimum over $\theta$, as in the printed target error.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:polynomial_peps_model}
    The function is continuous and $2\pi$-periodic, so its range is the image
    of the compact interval $[0,2\pi]$ and contains its least element. A bound
    at one phase bounds the infimum, and a bound on the infimum holds at a
    minimizing phase.
\end{proof}

\begin{theorem}[Support and norm of the Hermitian terms]
    \label{thm:square_hermitian_term_properties}
    \lean{
        TNLean.PEPS.Approximation.SquareHamiltonian.hermitianTerm_supported,
        TNLean.PEPS.Approximation.SquareHamiltonian.hermitianTerm_isHermitian,

    satisfies $\|A_i^{\mathrm h}\|\le J$. These conclusions hold for
    every $L,q\ge0$ and every real interaction bound allowed by the
    original model, including empty grids and zero local dimension.
\end{theorem}
\begin{proof}
    \leanok

    hence under real linear combinations. Thus the Hermitian part has
    the same support. Its adjoint equals itself, and
    $\|A_i^{\mathrm h}\|\le\|A_i\|\le J$.
\end{proof}

\begin{definition}[The original square Hamiltonian as a finite-domain interaction]
    \label{def:square_area_law_hamiltonian}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.toAreaLawHamiltonian}
    \leanok

    \end{align*}

\begin{theorem}[Exact Hamiltonian in integer coordinates]
    \label{thm:square_area_law_hamiltonian_operator}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.toAreaLawHamiltonian_operator}
    \leanok
    \uses{def:square_area_law_hamiltonian,thm:square_hermitian_term_sum}

    \begin{align*}
        H_{\mathrm{int}} & =UHU^*.
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    The support map is injective and every unused support contributes
    zero. Thus the sum is $\sum_i U A_i^{\mathrm h}U^*$.
    Conjugation is linear and $\sum_i A_i^{\mathrm h}=H$.
\end{proof}

\begin{theorem}[The original nearest-neighbor interaction family]
    \label{thm:square_nearest_neighbor_terms}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.exists_nearestNeighbor_terms}
    \leanok
    \uses{def:polynomial_peps_model,def:square_area_law_hamiltonian,
        def:area_law_nearest_neighbor_support}
    Let $H$ be an original square Hamiltonian with interaction bound $J\ge0$.
    Index the sites and unordered nearest-neighbor edges of $\Lambda$ by
    $\mathcal I$, and let $D_j$ be the singleton or endpoint pair of $j$.
    There exist operators $(T_j)_{j\in\mathcal I}$ such that each $T_j$ is
    supported on $D_j$ and
    \begin{align*}
        T_j^*                    & =T_j,              & \|T_j\| & \le J, &
        \sum_{j\in\mathcal I}T_j & =H_{\mathrm{int}}.
    \end{align*}
    The assertion holds for every side length and local dimension, and for
    every choice of order on the rectangle sites.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:square_interaction_support_relabel,
        thm:square_area_law_hamiltonian_operator,thm:square_hermitian_term_properties,
        thm:square_hermitian_term_supported_algebra,thm:square_hermitian_term_sum,
        thm:area_law_supported_identity,def:square_area_law_support,
        thm:area_law_domain_identifications,def:peps_square_edge_correspondence,
        def:square_hermitian_term}
    Let $e$ be the site and edge correspondence, and put
    $T_{e(i)}=U A_i^{\mathrm h}U^*$. The support identity gives $D_{e(i)}=X_i$.
    The Hermitian part preserves the original support and norm bound, and
    unitary conjugation preserves Hermiticity and the norm. Finally, the
    bijection $e$ preserves the finite sum, so
    $\sum_jT_j=U(\sum_i A_i^{\mathrm h})U^*=UHU^*=H_{\mathrm{int}}$.
\end{proof}

\begin{theorem}[The same ground vector and full-system gap]
    \label{thm:square_area_law_ground_gap}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.isGappedGroundState_toAreaLawHamiltonian}
    \leanok
    \uses{thm:square_area_law_hamiltonian_operator,thm:norm_square_area_law_state}
    Let $E_0,\Delta\in\mathbb R$. If an original unit vector $\Omega$
    satisfies
    \begin{align*}
        H\Omega & =E_0\Omega,                                \\
        H-E_0I  & \ge\Delta(I-|\Omega\rangle\langle\Omega|),
    \end{align*}
    then $U\Omega$ is a unit vector satisfying
    \begin{align*}
        H_{\mathrm{int}}U\Omega & =E_0U\Omega,                                  \\
        H_{\mathrm{int}}-E_0I
                                & \ge\Delta(I-|U\Omega\rangle\langle U\Omega|).
    \end{align*}
    The conclusion holds for any real lower-gap parameter $\Delta$.
    In particular, a positive original full-system gap is preserved.
\end{theorem}
\begin{proof}\leanok
    Conjugating the eigenvector equation and the positive semidefinite
    gap defect by the unitary $U$, together with norm preservation, gives both
    relations.
\end{proof}

\begin{theorem}[Regional entropy under relabeling of sites]
    \label{thm:finite_product_entropy_relabel}
    \lean{TNLean.PEPS.Approximation.finiteProductEntropy_relabel}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $\sigma:V\to W$ be a bijection of finite site sets, let $q\ge0$, and
    let $U_\sigma$ be the unitary that relabels the tensor factors of
    $(\mathbb C^q)^{\otimes V}$ along $\sigma$. For every vector $\Omega$ and
    every $A\subseteq V$,
    \begin{align*}
        S_{U_\sigma\Omega}(\sigma(A)) & =S_\Omega(A).
    \end{align*}
    No normalization of $\Omega$ is required.
\end{theorem}
\begin{proof}\leanok
    Relabeling the sites of $\sigma(A)$ and of its complement identifies the
    reduced matrix of $U_\sigma\Omega$ on $\sigma(A)$ with the reduced matrix
    of $\Omega$ on $A$, conjugated by a permutation of configurations. A
    permutation leaves the spectrum, and hence the entropy, unchanged.
\end{proof}

\begin{definition}[The square-grid entropy area-law statement]
    \label{def:square_entropy_area_law}
    \lean{TNLean.PEPS.Approximation.SquareEntropyAreaLaw}
    \leanok
    \uses{def:polynomial_peps_model,def:area_law_ground_entropy}
    The square-grid entropy area law is the assertion that for every integer
    $q\ge2$ and all reals $J,\Delta>0$ there is a constant
    $C_A=C_A(q,J,\Delta)\ge0$ with the following property. For every $L$, every
    Hamiltonian of the original model with interaction bound $J$, and every
    unit vector $\Omega$ and real $E_0$ satisfying
    $H\Omega=E_0\Omega$ and $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$,
    every set $A$ of sites of the grid satisfies
    \begin{align*}
        S_\Omega(A) & \le C_A|\partial_\Lambda A|,
    \end{align*}
    where $\partial_\Lambda A$ is the set of grid edges with exactly one
    endpoint in $A$. The constant is chosen before $L$, $H$, $\Omega$ and $A$.
\end{definition}

    constant $C_A$ if $C_A\ge0$ and, for every $L$, every state $\Omega$ of
    the family and every set $A$ of sites, $S_\Omega(A)\le C_A|\partial_\Lambda A|$.
    The ground states of the model at $J$ and $\Delta$ form the family of unit
    vectors $\Omega$ for which some Hamiltonian of the original model with
    interaction bound $J$ and some real $E_0$ satisfy $H\Omega=E_0\Omega$ and
    $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$. The square-grid
    entropy area law of Definition~\ref{def:square_entropy_area_law} says that
    for $q\ge2$ and $J,\Delta>0$ some constant $C_A$ has this property for

\end{definition}

\begin{theorem}[Polynomial PEPS approximation]
    \label{thm:openai_polynomial_peps}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation}
    \leanok
    \uses{def:polynomial_peps_model,lem:polynomial_peps_minimizing_phase}
    In Theorem~1.1 of \cite{OpenAI2026PolynomialPEPS}, fix an integer $q\ge2$
    and reals $J,\Delta>0$. There are $C,c>0$, depending only on these
    parameters, such that for every $L\ge2$, every Hamiltonian in the
    preceding model with a unique unit ground vector, up to phase, satisfying
    (\ref{eq:polynomial_peps_projector_gap}) has a nonzero PEPS on the original open
    grid with $D_e\le CL^c$ for every edge $e$ and
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
        \label{eq:polynomial_peps_phase_error}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:polynomial_peps_explicit_constants}
    Theorem~\ref{thm:polynomial_peps_explicit_constants} has the same
    hypotheses and gives the approximation with $C=C_{\rm PEPS}$ and
    $c=c_{\rm PEPS}$. Those constants are assembled along the following
    argument.

    Write $\Lambda_L$ for the sites of the grid and $\mathcal H_X$ for the
    tensor product of the site spaces in $X\subseteq\Lambda_L$. The physical
    registers are distributed among parties indexed by dyadic squares, starting
    from one party holding $\Omega$ and ending with one party per site. The
    area-law and collar estimates make the mutual information across tapered
    buffers smaller than any fixed inverse power of $L$, which
    lets an owner split its purification approximately. Near a square of
    logarithmic size, the ground vector is projected with high accuracy onto a
    sum of mutually orthogonal subspaces
    \begin{align}
        \operatorname{span}\{v_{jk}\}_{k=1}^{d_j}
        \otimes\mathcal H_{\Lambda_L\setminus X_j},
        \qquad
        \sum_jd_j & \le C_0L^{c_0},
        \label{eq:polynomial_peps_rank_bound}
    \end{align}
    with nested square samples $X_j$ and orthonormal vectors
    $v_{jk}\in\mathcal H_{X_j}$. Each party takes part in boundedly many
    operations with other parties during its whole lifetime, and each such
    operation is a sum of polynomially many products of private contractions
    and bipartite vectors or covectors.

    Telescoping the accumulated errors of these contractions, the final vector
    $\Psi$ and the final reference vector $\Omega\otimes\Xi$ satisfy
    \begin{align}
        \|\Psi-\Omega\otimes\Xi\| & \le CL^{-17},
        \qquad\|\Psi\|\le1,
        \qquad 1-CL^{-57}\le\|\Xi\|\le1,
        \label{eq:polynomial_peps_protocol_error} \\
        \bigl\|\operatorname{Tr}_{\rm private}|\Psi\rangle\langle\Psi|
        -|\Omega\rangle\langle\Omega|\bigr\|_1
                                  & \le CL^{-10}.
        \label{eq:polynomial_peps_protocol_density}
    \end{align}
    Density compression replaces the bipartite resources by polynomial sums of
    product operators and gives an operator network $\sigma$ on the party graph
    with polynomial link dimensions and
    \begin{align}
        \bigl\|\sigma-|\Omega\rangle\langle\Omega|\bigr\|_1
         & \le\eta:=CL^{-5}.
        \label{eq:polynomial_peps_compressed_density}
    \end{align}
    If $\eta<1$, write $E=\sigma-|\Omega\rangle\langle\Omega|$ and
    $a_z=\langle\Omega|z\rangle$ for product basis vectors $|z\rangle$. Since
    $\sum_z\|E|z\rangle\|^2\le\eta^2$ and $\sum_z|a_z|^2=1$, some $z$ has
    $a_z\ne0$ and $\|E|z\rangle\|\le\eta|a_z|$. The column $v=\sigma|z\rangle$
    is then nonzero, has the same virtual dimensions, and satisfies
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{v}{\|v\|}-e^{i\theta}\Omega\right\|
         & \le2\eta.
        \label{eq:polynomial_peps_column_error}
    \end{align}
    Routing each link horizontally and then vertically through the anchors of
    the dyadic blocks has edge congestion at most a constant $\chi$, also after
    folding the padded grid into the $L\times L$ grid. If every link has
    dimension at most $C_1L^k$, grouping the links crossing each grid edge
    gives
    \begin{align}
        D_e & \le(C_1L^k)^\chi=C_1^\chi L^{k\chi}.
        \label{eq:polynomial_peps_routed_bond}
    \end{align}
    For $L\ge L_0$ the error $2CL^{-5}$ is at most $L^{-1}$, so
    (\ref{eq:polynomial_peps_column_error}) gives
    (\ref{eq:polynomial_peps_phase_error}). Each of the finitely many sizes
    $2\le L<L_0$ has an exact representation with bond dimensions bounded
    independently of the Hamiltonian, which is absorbed by enlarging the
    prefactor in (\ref{eq:polynomial_peps_routed_bond}).
\end{proof}

    \label{def:area_law_template}
    \lean{

        \label{eq:area_law_template_diameter}
    \end{align}
    Each $P_a$ is a closed convex rectangle or triangle of sup-norm diameter at
    most $s_0$, with side slopes in $\{0,\infty,1,-1\}$. Triangles have
    noncollinear vertices and rectangles have nonzero sides; this ordinary
    reading of the source's polygons is recorded in
    \cite{gap:arealaw2d_nondegenerate_templates}. The union may be
    disconnected. Geometric estimates require a uniform lower bound on
    $C_{\rm tpl}$, chosen independently of the template.
    Separation from a fixed cut set $Z$ means
    $\operatorname{dist}_\infty(T,Z)>4D_0s_0$. For the fixed cut
    $A\subseteq\Lambda$ of (\ref{eq:area_law_boundary}), put
    \begin{align}
        T_j & =T+([-j,j]^2\cap\mathbb Z^2),
        \label{eq:area_law_template_dilation} \\
        X   & =A\cap T,
        \label{eq:area_law_template_core}     \\
        Q_j & =A\cap(T_j\setminus T).
        \label{eq:area_law_template_collar}
    \end{align}
    In particular, $T_0=T$. These are ambient dilations, not the induced-graph
    neighborhoods (\ref{eq:area_law_graph_neighborhood}).
\end{definition}

\begin{definition}[Rectangle templates, unions and omitted pieces]
    \label{def:area_law_template_instances}

\begin{definition}[The sublinear collar bound]
    \label{def:area_law_collar_bound}
    \lean{TNLean.PEPS.AreaLaw.SublinearCollarBound,
        TNLean.PEPS.AreaLaw.SublinearCollarBoundAt,
        TNLean.PEPS.AreaLaw.sublinearCollarBound_iff_exists_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_ground_entropy,def:area_law_template,thm:area_law_exponent_gaps}
    Fix $q,R,J,\Delta$ and the template constants $D_0,C_{\rm tpl}$. For
    disjoint $B,C\subseteq\Lambda$ write
    $I_\Omega(B:C)=S_\Omega(B)+S_\Omega(C)-S_\Omega(B\cup C)$. The sublinear
    collar bound of Proposition~9.6 of \cite{OpenAI2026AreaLaw} for these
    constants is the assertion that there are $C$ and
    $N$ with the following property. Let $n\ge N$, let $\Omega$ be the gapped
    ground vector of an admissible Hamiltonian on a finite induced domain
    $\Lambda$, let $A\subseteq\Lambda$, and let $T$ be a template with parameters
    $(n,s_0)$ separated from the endpoints of the cut edges. Put
    $L=\lfloor n^{1-\varepsilon}\rfloor$, $X=A\cap T$ and $S_0=A\cap T_L$, and
    suppose $L\le s_0$. Then
    \begin{align*}
        I_\Omega(X:\Lambda\setminus S_0) & \le Cn^{1-\varepsilon},
        \qquad |S_0|\le Cn^2,
    \end{align*}
    and each of the cuts $X$, $S_0$ and $S_0\setminus X$ has at most $Cn$
    crossing edges. When this property holds for a given pair $(C,N)$, we say
    that the sublinear collar bound holds with constant $C$ and threshold $N$;
    the sublinear collar bound is the existence of such a pair.
\end{definition}

\begin{definition}[The amplification bound]
    \label{def:area_law_amplification_bound}
    \lean{TNLean.PEPS.AreaLaw.AmplificationBound,
        TNLean.PEPS.AreaLaw.AmplificationBoundAt,
        TNLean.PEPS.AreaLaw.amplificationBound_iff_forall_exists_amplificationBoundAt}
    \leanok
    \uses{def:area_law_ground_entropy,def:area_law_radius_exponent}
    Fix $q,R,J,\Delta$. The amplification bound of Proposition~10.2 of
    \cite{OpenAI2026AreaLaw} is the assertion that for every
    $C_0>0$ and every integer $M\ge1$ there are $C_M$ and $N$ with the
    following property. Let $n\ge\max(2,N)$, let $\Omega$ be the gapped ground
    vector of an admissible Hamiltonian on a finite induced domain $\Lambda$,
    let $A\subseteq\Lambda$ and let $T\subset\mathbb Z^2$ be finite. Put
    $L=\lfloor n^{1-\varepsilon}\rfloor$, $X=A\cap T$, $S_0=A\cap T_L$ and
    $Y=S_0\setminus X$. Suppose that $|S_0|\le C_0n^2$, that each of the cuts
    $X$, $Y$, $S_0$ has at most $C_0n$ crossing edges, and that
    $I_\Omega(X:\Lambda\setminus S_0)\le C_0n^{1-\varepsilon}$. Then there is
    an integer $r\le C_Mn^{(1-\varepsilon)/\alpha}$ such that
    \begin{align*}
        I_\Omega(X':J) & \le C_Mn^{-M}
        \qquad\text{for every }X'\subseteq X\text{ and every }
        J\subseteq\Lambda\setminus N_{2r}^\Lambda(S_0).
    \end{align*}

    set of endpoints of the cut edges, then $N_{2r}^\Lambda(S_0)\subseteq A$;
    its further assertion $L+2r=o(n^\beta)$ follows from the radius bound by
    Theorem~\ref{thm:area_law_uniform_collar_threshold}.
    When this property holds for given $C_0$, $M$ and a given pair $(C_M,N)$,
    we say that the amplification bound holds at $(C_0,M)$ with constant $C_M$
    and threshold $N$.
\end{definition}

\begin{lemma}[Retained primaries]

\begin{theorem}[Leaf entropy identity and its lower bound]
    \label{thm:area_law_collar_leaf_identity}

    entropy identity and the lower bound after the leaf average in the proof
    of Proposition~9.6 of \cite{OpenAI2026AreaLaw}, for every vector $\psi$,
    \begin{align}
        2S(X)+g_{\rm pre}-g_{\rm post} & =I(X:W)+I(X:Y\cup W).
        \label{eq:area_law_collar_leaf_identity}
    \end{align}
    If a set $F$ lies in $W$, then $2I(X:F)\le2S(X)+g_{\rm pre}-g_{\rm post}$.
    Consequently, for finitely many leaves with weights $p_i\ge0$ summing to one,
    each with its own partition $X,U_i,Y_i,W_i$ of $V$ and with $F\subseteq W_i$
    for every $i$,
    \begin{align*}
        2I(X:F) & \le2S(X)+\sum_ip_i\bigl(g_{{\rm pre},i}-g_{{\rm post},i}\bigr).
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    Purity gives $S(X\cup W)=S(U\cup Y)$, $S(X\cup Y\cup W)=S(U)$ and
    $S(Y\cup W)=S(X\cup U)$, and the identity follows by expanding both mutual
    informations. Strong subadditivity makes mutual information monotone in each
    argument, and both $W$ and $Y\cup W$ contain $F$. Multiplying the bound at
    each leaf by $p_i$ and summing gives the average inequality.
\end{proof}

\begin{theorem}[Transfer of mutual information to the original state]
    \label{thm:area_law_collar_transfer}
    \lean{FiniteProduct.abs_mutualInformation_sub_le_of_traceDistance,
        FiniteProduct.mutualInformation_le_add_of_traceDistance_le_rpow}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $\Omega,\widetilde\Omega$ be unit vectors at trace distance $\delta$, and
    let $X,J$ be disjoint. Then
    \begin{align*}
        |I_\Omega(X:J)-I_{\widetilde\Omega}(X:J)|
         & \le8\sqrt\delta\,\log(e\dim\mathcal H_X).
    \end{align*}
    In particular, if $\delta\le C_1n^{-500}$ and
    $\log\dim\mathcal H_X\le C_2n^2$ for an integer $n\ge1$, then
    $I_\Omega(X:J)\le I_{\widetilde\Omega}(X:J)+8\sqrt{C_1}(1+C_2)n^{-248}$.
    The dimension of $J$ does not enter.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:al_amp_pure_trace_distance}
    Applying \cite[Lemma~2.1]{OpenAI2026AreaLaw} to the joint reductions on
    $X\cup J$, whose trace distance is at most $\delta$ by
    Lemma~\ref{lem:al_amp_pure_trace_distance}, gives the first inequality;
    then $\sqrt{C_1n^{-500}}=\sqrt{C_1}\,n^{-250}$ and
    $\log(e\dim\mathcal H_X)\le(1+C_2)n^2$.
\end{proof}

\begin{theorem}[Cut and size assertions of the collar bound]

\end{theorem}
\begin{proof}

\section{Companion area-law inputs}
\label{sec:peps_companion_inputs}

The proof of Theorem~\ref{thm:openai_polynomial_peps} uses several results of
the area-law companion \cite{OpenAI2026AreaLaw}, which are listed in
\cite[Appendix~A]{OpenAI2026PolynomialPEPS}. All of them concern the original
ground vector $\Omega$ of the full system; no subregion is assigned a gap of its
own. The remarks below compare each listed result with the statement of this
chapter that expresses it. No statement below adds a hypothesis to the
printed one. The appendix also records that the companion takes its replica
limit at each fixed finite instance; this concerns the companion's proof of
the collar bound, and the statements below concern only the original one-copy
state.

\begin{remark}[Model, metric and entropy]
    \label{rem:peps_companion_model}
    \uses{def:area_law_domain,def:area_law_hamiltonian,
        def:area_law_ground_entropy,thm:area_law_ground_eigenspace}
    The companion allows every finite induced domain $\Gamma\subset\mathbb Z^2$
    with local dimension $q$, and one Hermitian term $h_X$ with
    $\|h_X\|\le J$ for each nonempty support $X$ of induced-graph diameter at
    most $R$. It assumes a unit ground vector $\Omega$ with
    $H-E_0I\ge\Delta(I-|\Omega\rangle\langle\Omega|)$, and its entropies use
    natural logarithms \cite[Appendix~A]{OpenAI2026PolynomialPEPS}.
    Definitions~\ref{def:area_law_domain}, \ref{def:area_law_hamiltonian}
    and~\ref{def:area_law_ground_entropy} state exactly this model: zero terms
    are allowed, the domain need not be connected, norms are operator norms,
    and entropy is the von Neumann entropy with natural logarithms. Uniqueness
    of the ground vector up to phase is not a separate hypothesis. It follows
    from the gap inequality by Theorem~\ref{thm:area_law_ground_eigenspace}.
    % Status: proved.
\end{remark}

\begin{remark}[The area law and its square-grid specialization]
    \label{rem:peps_companion_area_law}
    \uses{thm:openai_uniform_area_law,cor:rectangles,
        thm:area_law_conditional_collar_two_families,
        thm:area_law_sublinear_collar,thm:area_law_amplification,
        thm:square_hermitian_term_sum,thm:square_area_law_from_uniform,
        thm:rectangular_equal_side_entropy,thm:openai_square_area_law}
    Theorem~1.1 and Corollary~1.2 of \cite{OpenAI2026AreaLaw} give
    Proposition~2.1 of \cite{OpenAI2026PolynomialPEPS}, with $\Gamma$ the
    $L\times L$ grid and $R=1$. Theorem~\ref{thm:openai_uniform_area_law}
    states the uniform finite-domain area law with the printed quantifiers: one
    constant $C(q,R,J,\Delta)$ serves every domain, interaction family and cut.
    Theorem~\ref{thm:area_law_conditional_collar_two_families} derives it from
    Propositions~9.6 and~11.2 of \cite{OpenAI2026AreaLaw}.
    The collar proposition is Theorem~\ref{thm:area_law_sublinear_collar};
    amplification is Theorem~\ref{thm:area_law_amplification}.
    % Status: the collar, amplification and two-family propositions are
    % formalized, and so are the uniform area law and its square-grid passage.
    The site and edge terms of the
    original model need not be Hermitian individually. Replacing each by its
    Hermitian part keeps its support and norm bound and leaves the total
    Hamiltonian unchanged (Theorem~\ref{thm:square_hermitian_term_sum}).
    Theorem~\ref{thm:square_area_law_from_uniform} then specializes the uniform
    area law to the square at range one with $C_A(q,J,\Delta)=C(q,1,J,\Delta)$,
    and Theorem~\ref{thm:rectangular_equal_side_entropy} gives the same
    conclusion through the rectangle corollary, giving Proposition~2.1
    (Theorem~\ref{thm:openai_square_area_law}) from the uniform area law.
    % Status: Proposition 2.1 is formalized through the uniform area law.
\end{remark}

\begin{remark}[Positive replacement]
    \label{rem:peps_companion_positive_replacement}
    \uses{thm:al_lattice_positive_replacement,thm:al_lattice_counts}
    For every integer $k\ge1$ and $\alpha=k/(k+1)$, the appendix uses positive
    contractions $k_i$ at the original anchors, of uniformly bounded anchor
    multiplicity, with $k_i\Omega=0$,
    $\sum_ik_i\ge c_*^{-1}(H-E_0I)\ge(\Delta/c_*)(I-|\Omega\rangle\langle\Omega|)$
    and $\|k_i-\mathbb E_{N_r(a_i)}k_i\|\le Ce^{-cr^\alpha}$ for every integer
    $r\ge0$ \cite[Proposition~4.3]{OpenAI2026AreaLaw}.
    Theorem~\ref{thm:al_lattice_positive_replacement} states this for every
    choice of anchors $a_i\in X_i$, with constants depending only on
    $k,R,J,\Delta$ and under the weaker condition $J\ge0$.
    Its tail estimate is stated for conditional expectation onto every
    induced-graph ball. At most $2^{2R(R+1)}$ labels share an anchor, by
    Theorem~\ref{thm:al_lattice_counts}.
    % Status: proved.
\end{remark}

\begin{remark}[Templates and the sublinear collar bound]
    \label{rem:peps_companion_collar}
    \uses{def:area_law_template,def:area_law_collar_bound,
        thm:area_law_collar_cut_bounds,thm:area_law_collar_bound_of_information,
        thm:area_law_sublinear_collar}
    Definition~\ref{def:area_law_template} states the templates of
    \cite[Definition~9.3]{OpenAI2026AreaLaw} with the printed conditions
    $n\ge C_{\rm tpl}N_0(s_0+1)$, $\operatorname{diam}_\infty T\le n$ and
    $\operatorname{dist}_\infty(T,Z_A)>4D_0s_0$; the polygons are read as
    nondegenerate \cite{gap:arealaw2d_nondegenerate_templates}.
    Definition~\ref{def:area_law_collar_bound} states the sublinear collar
    bound, Proposition~9.6 of \cite{OpenAI2026AreaLaw}, with
    $\varepsilon=10^{-5}$, $\ell=\lfloor n^{1-\varepsilon}\rfloor\le s_0$,
    $X=A\cap T$ and $S_0=A\cap T_\ell$: at every scale beyond a uniform
    threshold, $I_\Omega(X:\Gamma\setminus S_0)\le Cn^{1-\varepsilon}$,
    $|S_0|\le Cn^2$, and each of $X$, $S_0\setminus X$ and $S_0$ has at most $Cn$
    crossing edges.
    % Status: Proposition 9.6 is proved by thm:area_law_sublinear_collar.
    For $C_{\rm tpl}\ge24$ and $D_0\ge1$, which
    the standing choices of a large $C_{\rm tpl}$ and of $D_0>2R+10$ ensure, the
    size and crossing-edge assertions hold for every separated template
    (Theorem~\ref{thm:area_law_collar_cut_bounds}), so
    Theorem~\ref{thm:area_law_collar_bound_of_information} reduces the
    proposition to the information inequality alone. This inequality,
    together with the size and crossing-edge bounds, is
    Theorem~\ref{thm:area_law_sublinear_collar}.
    % Status: all assertions of Proposition 9.6 are proved.
\end{remark}

\begin{remark}[Amplification]
    \label{rem:peps_companion_amplification}
    \uses{def:area_law_amplification_bound,thm:area_law_amplification,

    which derives it from the sublinear collar bound.

    conclusion using the physical collar theorem.
\end{remark}

\begin{remark}[Two-family entropy cancellation]
    \label{rem:peps_companion_two_family_entropy}
    \uses{thm:area_law_two_family_entropy}
    For a pure state and an ordered partition
    $A=D\mathbin{\dot\cup}\bigcup_iX_i$ into two labelled families, the bounds
    $I(X_i:A^c\cup\bigcup_{j<i,\,f_j=f_i}X_j)\le\delta_i$ give
    $S(A)\le S(D)+\frac12\sum_i\delta_i$
    \cite[Lemma~11.1]{OpenAI2026AreaLaw}.
    Theorem~\ref{thm:area_law_two_family_entropy} states this for a unit vector
    in any finite tensor product of finite-dimensional spaces.
    % Status: proved.
\end{remark}

\begin{theorem}[Sublinear information across a template collar]
    \label{thm:area_law_sublinear_collar}
    \lean{TNLean.PEPS.AreaLaw.sublinearCollarBound_real}
    \leanok
    \uses{def:area_law_collar_bound}
    In~\cite[Proposition~9.6]{OpenAI2026AreaLaw}, fix $q,R\in\mathbb N$ with
    $q\geq1$, real parameters $J\geq0$ and $\Delta>0$, and real template
    constants $D_0>2R+10$ and $C_{\mathrm{tpl}}\geq24$. There are
    $C\in\mathbb R$ and
    $N\in\mathbb N$, chosen before all the following finite data.
    Let $\Lambda$ be a finite induced square-lattice domain, let $H$ be
    an admissible Hamiltonian with parameters $(q,R,J)$, and let
    $\Omega$ be its normalized ground vector at energy $E_0$ with
    full-system projector gap $\Delta$. Let $A\subseteq\Lambda$ and
    let $Z$ be the endpoints of its cut edges. For every template $T$
    with parameters $(C_{\mathrm{tpl}},n,s_0)$ satisfying $n\geq N$,
    $\operatorname{dist}_\infty(T,Z)>4D_0s_0$, and
    $L=\lfloor n^{1-1/100000}\rfloor\leq s_0$, put $X=A\cap T$ and
    $S_0=A\cap T_L$. Then
    \begin{align}
        I_\Omega(X:\Lambda\setminus S_0)
              & \leq Cn^{1-1/100000}, \\
        |S_0| & \leq Cn^2,
    \end{align}
    and each of $X$, $S_0$, and $S_0\setminus X$ has at most $Cn$
    crossing edges in $\Lambda$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:arealaw-physical-box-fixed-exponent,
        thm:arealaw-box-clearance-comparison,
        thm:template-final-information,
        thm:area_law_collar_bound_of_information}
    Put $d=2R+10$. The physical safe-box bound at the integer clearance
    $d+1$ gives $C_b>0$ at exponent $1+e_*$, where
    $e_*=2\cdot10^{-6}$. The clearance comparison gives
    \begin{align}
        F_{\mathrm{box},d}(r)
         & \leq (4C_b+4\log q+1)r^{1+e_*}
    \end{align}
    at every positive integer scale. The conditional template
    information theorem at clearance $d$, which requires only $d\geq1$,
    gives $C_m>0$ and a threshold, before all finite instances, for
    which $2I_\Omega(X:\Lambda\setminus S_0)\leq C_m n^{1-1/100000}$.
    The hypothesis $D_0>d$ ensures that every template separated at
    the prescribed real clearance $D_0$ is also separated at $d$.
    The estimate therefore applies to the same original vector and
    the unchanged regions $X$ and $S_0$.

    Dividing the information coefficient by two and applying
    Theorem~\ref{thm:area_law_collar_bound_of_information} shows that the
    single constant $\max(C_m/2,10)$ also bounds the three cuts and the
    collar cardinality, with the same threshold.
\end{proof}
