\section{Finite-domain area laws and polynomial PEPS approximation}
\label{sec:finite_domain_area_law}

An entropy bound for every cut on arbitrary finite induced lattice domains and a
compatible tensor approximation on the open square are distinct assertions; we
state the two models separately following
\cite{OpenAI2026AreaLaw,OpenAI2026PolynomialPEPS}.

\begin{definition}[Induced lattice domain]
    \label{def:area_law_domain}
    \lean{
        TNLean.Lattice.Site,
        TNLean.Lattice.Configuration,
        TNLean.Lattice.StateSpace,
        TNLean.Lattice.domainGraph,
        TNLean.Lattice.graphNeighborhood,
        TNLean.Lattice.configurationSplit,
        TNLean.Lattice.card_configuration,
        TNLean.Lattice.card_regionConfig
    }
    \leanok
    Let $\Lambda\subset\mathbb Z^2$ be finite (possibly empty, perforated, or
    disconnected). Join sites at $\ell_1$-distance $1$, and let $d_\Lambda$ be
    the induced-graph distance ($d_\Lambda=+\infty$ between distinct components).
    For $X\subseteq\Lambda$ and an integer $r\ge0$, the closed graph neighborhood is
    \begin{align}
        N_r^\Lambda(X) & =\{y\in\Lambda:\ d_\Lambda(x,y)\le r
        \text{ for some }x\in X\}.
        \label{eq:area_law_graph_neighborhood}
    \end{align}
    With site space $\mathbb C^q$ ($q\ge1$), the physical Hilbert space in
    configuration coordinates is
    \begin{align}
        \label{eq:area_law_state_space}
        \mathcal H_\Lambda     & =\mathbb C^{\{\sigma:\Lambda\to\{0,\ldots,q-1\}\}}, \\
        \label{eq:area_law_dimension}
        \dim\mathcal H_\Lambda & =q^{|\Lambda|}.
    \end{align}
    For $A\subseteq\Lambda$, restriction to $A$ and $\Lambda\setminus A$
    identifies $\{0,\ldots,q-1\}^\Lambda\simeq
        \{0,\ldots,q-1\}^A\times\{0,\ldots,q-1\}^{\Lambda\setminus A}$, with
    $|\{0,\ldots,q-1\}^A|=q^{|A|}$.
\end{definition}

\begin{definition}[Finite-range interactions and boundary]
    \label{def:area_law_hamiltonian}
    \lean{
        TNLean.Lattice.IsAdmissibleSupport,
        TNLean.Lattice.AdmissibleSupport,
        TNLean.Lattice.LocalHamiltonian,
        TNLean.Lattice.LocalHamiltonian.operator,
        TNLean.Lattice.LocalHamiltonian.operator_isHermitian,
        TNLean.Lattice.LocalHamiltonian.zero,
        TNLean.Lattice.localLift,
        TNLean.Lattice.localLift_mem_supportedOperators,
        TNLean.Lattice.localLift_eq_entropyLocalLift,
        TNLean.Lattice.edgeBoundary,
        TNLean.Lattice.edgeBoundary_empty,
        TNLean.Lattice.edgeBoundary_univ
    }
    \leanok
    \uses{def:area_law_domain}
    Fix an integer $R\ge0$ and $J>0$. For each nonempty $X\subseteq\Lambda$ with
    $\operatorname{diam}_{d_\Lambda}X\le R$, assign one Hermitian operator
    $h_X=h_X^*$ supported on $X$ (extended by $I_{\Lambda\setminus X}$, possibly
    $h_X=0$) with Euclidean operator norm $\|h_X\|\le J$. For
    $A\subseteq\Lambda$, define
    \begin{align}
        \label{eq:area_law_hamiltonian}
        H                  & =\sum_{\varnothing\ne X\subseteq\Lambda,\ \operatorname{diam}_{d_\Lambda}X\le R}h_X, \\
        \label{eq:area_law_boundary}
        \partial_\Lambda A & =
        \{\{x,y\}:x\in A,\ y\in\Lambda\setminus A,\ d_\Lambda(x,y)=1\},
    \end{align}
    so $\partial_\Lambda\varnothing=\partial_\Lambda\Lambda=\varnothing$.
\end{definition}

\begin{lemma}[Interaction range and rectangular coordinates]
    \label{thm:area_law_domain_identifications}
    \lean{
        TNLean.Lattice.exists_walk_length_le_iff_edist_le,
        TNLean.Lattice.isAdmissibleSupport_zero_iff,
        TNLean.Lattice.isAdmissibleSupport_singleton,
        TNLean.Lattice.isAdmissibleSupport_pair_of_adj,
        TNLean.PEPS.AreaLaw.rectangleEmbedding,
        TNLean.PEPS.AreaLaw.rectangleEmbedding_injective,
        TNLean.PEPS.AreaLaw.rectangularDomain,
        TNLean.PEPS.AreaLaw.rectangleSiteEquiv,
        TNLean.PEPS.AreaLaw.rectangleSiteEquiv_val,
        TNLean.PEPS.AreaLaw.rectangleGraphIso,
        TNLean.PEPS.AreaLaw.rectangleConfigurationEquiv,
        TNLean.PEPS.AreaLaw.isAdmissibleSupport_rectangle_pair_of_adj
    }
    \leanok
    \uses{def:area_law_domain,def:area_law_hamiltonian}
    For $x,y\in\Lambda$, $d_\Lambda(x,y)\le R$ if and only if an induced walk of
    length at most $R$ joins $x$ and $y$. Every singleton $\{x\}$ is admissible
    for every $R\ge0$; at $R=0$ the admissible supports are precisely the
    singletons; adjacent pairs $\{x,y\}$ ($d_\Lambda(x,y)=1$) are admissible for
    $R\ge1$. The nearest-neighbor graph on $\{0,\ldots,w-1\}\times\{0,\ldots,h-1\}$
    is isomorphic to the induced graph on that integer rectangle, identifying
    their configurations and range-$1$ pair supports.
\end{lemma}
\begin{proof}\leanok
    Since $d_\Lambda(x,y)$ is the infimum of walk lengths (or $+\infty$), a walk
    of length $0$ has $x=y$, so $\operatorname{diam}_{d_\Lambda}X\le0$ with
    $X\ne\varnothing$ forces $|X|=1$. The coordinate embedding is a graph
    isomorphism onto $\{0,\ldots,w-1\}\times\{0,\ldots,h-1\}$, inducing the
    configuration and support identifications.
\end{proof}

\begin{definition}[Ground vector and regional entropy]
    \label{def:area_law_ground_entropy}
    \lean{
        Matrix.IsGappedGroundState,
        Matrix.isGappedGroundState_iff,
        TNLean.Lattice.IsGappedGroundState,
        TNLean.Lattice.reducedState,
        TNLean.Lattice.reducedState_isHermitian,
        TNLean.Lattice.regionalEntropy,
        TNLean.Lattice.reducedState_eq_regionState,
        TNLean.Lattice.regionalEntropy_eq_regionEntropy,
        TNLean.PEPS.AreaLaw.UniformAreaLaw
    }
    \leanok
    \uses{def:area_law_hamiltonian}
    For a unit vector $\Omega$, $E_0\in\mathbb R$, and $\Delta>0$, the gapped
    ground-state conditions (imposed only on the full Hamiltonian $H$) are
    \begin{align}
        \label{eq:area_law_ground_vector}
        H\Omega & =E_0\Omega,                                \\
        \label{eq:area_law_projector_gap}
        H-E_0I  & \ge\Delta(I-|\Omega\rangle\langle\Omega|).
    \end{align}
    For $A\subseteq\Lambda$, the Hermitian reduced state and natural-logarithm
    von Neumann entropy ($0\log0=0$) are
    \begin{align}
        \label{eq:area_law_reduced_state}
        \rho_A      & =\operatorname{Tr}_{\Lambda\setminus A}|\Omega\rangle\langle\Omega|, \\
        \label{eq:area_law_regional_entropy}
        S_\Omega(A) & =-\operatorname{Tr}(\rho_A\log\rho_A).
    \end{align}
    The uniform area law asserts that for each $(q,R,J,\Delta)$ there exists
    $C<\infty$ such that $S_\Omega(A)\le C|\partial_\Lambda A|$ for all finite
    $\Lambda$, $H$, $\Omega$, and $A\subseteq\Lambda$.
\end{definition}

\begin{lemma}[Uniqueness from a positive global gap]
    \label{thm:area_law_ground_eigenspace}
    \lean{Matrix.IsGappedGroundState.eigenspace_eq_span}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $H$ be a finite-dimensional complex matrix, $E_0\in\mathbb R$, and
    $\Omega$ a unit vector satisfying (\ref{eq:area_law_ground_vector}) and
    (\ref{eq:area_law_projector_gap}) with $\Delta>0$. Then
    $\ker(H-E_0I)=\operatorname{span}_{\mathbb C}\{\Omega\}$, giving the
    uniqueness assertion in \cite[Proposition~4.5]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    If $H\psi=E_0\psi$, then (\ref{eq:area_law_projector_gap}) gives
    $0\ge\Delta\|\psi-\langle\Omega,\psi\rangle\Omega\|^2$, so $\Delta>0$
    implies $\psi=\langle\Omega,\psi\rangle\Omega$. Conversely,
    (\ref{eq:area_law_ground_vector}) gives
    $\operatorname{span}_{\mathbb C}\{\Omega\}\subseteq\ker(H-E_0I)$.
\end{proof}

\begin{lemma}[Regional reductions of pure vectors]
    \label{thm:area_law_regional_state_identities}
    \lean{
        TNLean.Lattice.reducedState_posSemidef,
        TNLean.PEPS.AreaLaw.trace_reducedState_eq_norm_sq,
        TNLean.PEPS.AreaLaw.trace_reducedState,
        TNLean.PEPS.AreaLaw.regionalEntropy_nonneg,
        TNLean.PEPS.AreaLaw.regionalEntropy_empty,
        TNLean.PEPS.AreaLaw.regionalEntropy_univ
    }
    \leanok
    \uses{def:area_law_domain,def:area_law_ground_entropy}
    Let $\Lambda\subset\mathbb Z^2$ be finite, $q\ge0$ an integer (allowing
    $q=0$ or $\Lambda=\varnothing$), and $\Omega\in\mathcal H_\Lambda$. Under
    the conventions of \cite[Section~2]{OpenAI2026AreaLaw} and
    Definition~\ref{def:area_law_ground_entropy}, every $A\subseteq\Lambda$
    satisfies
    \begin{align}
        \rho_A                  & \ge0,
        \label{eq:area_law_reduced_positive}     \\
        \operatorname{Tr}\rho_A & =\|\Omega\|^2.
        \label{eq:area_law_reduced_trace}
    \end{align}
    If $\|\Omega\|=1$, then for every $A\subseteq\Lambda$,
    \begin{align}
        \operatorname{Tr}\rho_A & =1,
        \label{eq:area_law_reduced_unit_trace}  \\
        S_\Omega(A)             & \ge0,
        \label{eq:area_law_entropy_nonnegative} \\
        S_\Omega(\varnothing)   & =0,
        \label{eq:area_law_empty_entropy}       \\
        S_\Omega(\Lambda)       & =0.
        \label{eq:area_law_full_entropy}
    \end{align}
    % Source revision: openai/math@adc7f1241b42e322a6451854ab7e4b4c146bf78a,
    % build/sections/01-preliminaries.tex, sec:prelim, lines 10--25.
    % This entry records regional-state identities only; it does not mark a
    % source-labelled continuity, gap-equivalence, or area-law theorem complete.
\end{lemma}
\begin{proof}\leanok
    Since $|\Omega\rangle\langle\Omega|\ge0$ and $\operatorname{Tr}_{\Lambda\setminus A}$
    preserves positivity and trace, (\ref{eq:area_law_reduced_positive}) and
    (\ref{eq:area_law_reduced_trace}) hold. When $\|\Omega\|=1$, $\rho_A\ge0$
    and $\operatorname{Tr}\rho_A=1$ force $\operatorname{spec}(\rho_A)\subseteq[0,1]$,
    so $-\lambda\log\lambda\ge0$ gives (\ref{eq:area_law_entropy_nonnegative}).
    For $A=\varnothing$ or $A=\Lambda$, the complement has $q^0=1$ configuration,
    so the coefficient matrix $M$ with $\rho_A=MM^*$ has rank at most $1$ and
    $\operatorname{spec}(\rho_A)\subseteq\{0,1\}$, giving
    (\ref{eq:area_law_empty_entropy})--(\ref{eq:area_law_full_entropy}).
\end{proof}

\begin{lemma}[Marginal trace norm from the full-system gap]
    \label{thm:peps_gap_marginal_trace_norm}
    \lean{FiniteProduct.re_trace_abs_reducedPure_sub_le_sqrt_energy}
    \leanok
    \uses{def:area_law_ground_entropy}
    On $\mathcal H=\bigotimes_{v\in V}\mathcal H_v$, let $H$ be an operator,
    $E_0\in\R$, $\Delta>0$, and $\Omega\in\mathcal H$ a unit vector satisfying
    \begin{align}
        H\Omega & = E_0\Omega, \notag                         \\
        H-E_0I  & \geq\Delta(I-|\Omega\rangle\langle\Omega|).
        \label{eq:peps_gap_projector_bound}
    \end{align}
    For any unit vector $\phi\in\mathcal H$ and $X\subseteq V$, with
    $\rho_X^\psi=\operatorname{Tr}_{V\setminus X}|\psi\rangle\langle\psi|$, as in
    the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L339-L365}{proof of Proposition 4.1},
    \begin{align}
        \|\rho_X^\phi-\rho_X^\Omega\|_1
         & \leq 2\sqrt{\frac{\operatorname{Re}\langle\phi,H\phi\rangle-E_0}{\Delta}}.
        \label{eq:peps_gap_marginal_distance}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:al_amp_pure_trace_distance}
    With $e=\operatorname{Re}\langle\phi,H\phi\rangle-E_0$, pairing
    (\ref{eq:peps_gap_projector_bound}) with $\phi$ gives
    $1-|\langle\Omega,\phi\rangle|^2\leq e/\Delta$. Partial-trace contractivity
    and Lemma~\ref{lem:al_amp_pure_trace_distance} yield
    \begin{align}
        \|\rho_X^\phi-\rho_X^\Omega\|_1
         & \leq\bigl\||\phi\rangle\langle\phi|-|\Omega\rangle\langle\Omega|\bigr\|_1
        =2\sqrt{1-|\langle\Omega,\phi\rangle|^2}
        \leq2\sqrt{e/\Delta}.\notag
    \end{align}
\end{proof}

\begin{lemma}[A sufficient energy bound for marginal approximation]
    \label{thm:peps_gap_marginal_one_sixteenth}
    \lean{FiniteProduct.re_trace_abs_reducedPure_sub_le_one_sixteenth_of_energy}
    \leanok
    \uses{thm:peps_gap_marginal_trace_norm}
    Under the hypotheses of Lemma~\ref{thm:peps_gap_marginal_trace_norm}, if
    $\operatorname{Re}\langle\phi,H\phi\rangle-E_0\leq\Delta/1024$, then
    $\|\rho_X^\phi-\rho_X^\Omega\|_1\leq1/16$ for every $X\subseteq V$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_gap_marginal_trace_norm}
    Substitute $(\operatorname{Re}\langle\phi,H\phi\rangle-E_0)/\Delta\le1/1024$
    and $2\sqrt{1/1024}=1/16$ into (\ref{eq:peps_gap_marginal_distance}).
\end{proof}

\begin{theorem}[Uniform finite-domain area law]
    \label{thm:openai_uniform_area_law}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw}
    \leanok
    \uses{def:area_law_domain,def:area_law_hamiltonian,def:area_law_ground_entropy}
    In \cite[Theorem~1.1]{OpenAI2026AreaLaw}, for fixed integers $q\ge1$,
    $R\ge0$ and reals $J,\Delta>0$, there exists $C=C(q,R,J,\Delta)<\infty$ such
    that for every finite induced domain $\Lambda$, every Hamiltonian $H$ of
    Definition~\ref{def:area_law_hamiltonian}, every unit ground vector $\Omega$
    satisfying (\ref{eq:area_law_ground_vector})--(\ref{eq:area_law_projector_gap}),
    and every $A\subseteq\Lambda$, one has
    $S_\Omega(A)\le C|\partial_\Lambda A|$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_closed_constant}
    Corollary~\ref{thm:area_law_closed_constant} proves the bound with
    $C=C_{\rm AL}(q,R,J,\Delta)$ assembled as follows. Write
    $I_\Omega(B:C)=S_\Omega(B)+S_\Omega(C)-S_\Omega(B\cup C)$ for disjoint
    $B,C\subseteq\Lambda$, and let $Z$ be the endpoints of $\partial_\Lambda A$.
    If $b=|\partial_\Lambda A|=0$, then $S_\Omega(A)=0$; assume $b>0$.

    The global gap yields a sub-volume box bound: for a box $Q$ of side at most
    $r$ far from $Z$,
    \begin{align}
        S_\Omega(A\cap Q) & \le Cr^{1+e_0},\qquad e_0<1.
        \label{eq:area_law_initial_box}
    \end{align}
    Spectral filtering replaces $H$ by positive quasi-local constraints
    annihilating $\Omega$, and a replica-scanner bootstrap improves
    (\ref{eq:area_law_initial_box}) to the collar bound at scale $n$ with
    $L_n=\lfloor n^{1-\varepsilon}\rfloor$:
    \begin{align}
        I_\Omega\bigl(A\cap T:\Lambda\setminus(A\cap T_{L_n})\bigr)
         & \le Cn^{1-\varepsilon},
        \label{eq:area_law_collar_information}
    \end{align}
    for the exponents $\varepsilon,\alpha,\beta$ of
    Lemma~\ref{thm:area_law_exponent_gaps}. Constraint-product amplification
    then gives, for each $M\ge1$, a radius $r\le C_Mn^{(1-\varepsilon)/\alpha}$
    with $L_n+2r=o(n^\beta)$ and
    \begin{align}
        I_\Omega(A\cap T:J)
         & \le C_Mn^{-M}
        \qquad\bigl(J\subseteq\Lambda\setminus N_{2r}^\Lambda(A\cap T_{L_n})\bigr).
        \label{eq:area_law_amplified_information}
    \end{align}

    The two-family decomposition partitions $A$ into $D$ and ordered pieces
    $X_i\subseteq A\cap T_i$ with labels $f_i\in\{0,1\}$ and templates $T_i$ at
    scales $n_i$ satisfying
    \begin{align}
        |D| & \le Cb,
        \qquad
        \sum_in_i^{-100}\le Cb,
        \label{eq:area_law_partition_counts} \\
        \operatorname{dist}_\infty(T_i,Z)
            & >2n_i^\beta,
        \qquad
        \operatorname{dist}_\infty\Bigl(T_i,
        \bigcup_{j<i,\,f_j=f_i}X_j\Bigr)>2n_i^\beta.
        \label{eq:area_law_partition_clearance}
    \end{align}
    Put $S_i=A\cap(T_i)_{L_{n_i}}$ and
    $J_i=(\Lambda\setminus A)\cup\bigcup_{j<i,\,f_j=f_i}X_j$, with $L_{n_i}+2r_i<n_i^\beta$.
    By (\ref{eq:area_law_partition_clearance}), any walk of length at most
    $2r_i$ from $S_i$ stays in $A$ and avoids $\bigcup_{j<i,\,f_j=f_i}X_j$, so
    $J_i\cap N_{2r_i}^\Lambda(S_i)=\varnothing$. At $M=100$,
    (\ref{eq:area_law_amplified_information}) and monotonicity give
    \begin{align}
        I_\Omega(X_i:J_i)\le I_\Omega(A\cap T_i:J_i)
         & \le Cn_i^{-100}=:\eta_i.
        \label{eq:area_law_piece_information}
    \end{align}
    For $f\in\{0,1\}$, $A_f=\bigcup_{f_i=f}X_i$, and $E_f=\sum_{f_i=f}\eta_i$,
    the chain rule, purity, and subadditivity give
    \begin{align}
        S_\Omega(A)+\sum_{f_i=f}S_\Omega(X_i)-E_f
         & \le S_\Omega\bigl((\Lambda\setminus A)\cup A_f\bigr)
        =S_\Omega(D\cup A_{1-f})
        \le S_\Omega(D)+\sum_{f_i=1-f}S_\Omega(X_i).
        \label{eq:area_law_family_chain}
    \end{align}
    Summing (\ref{eq:area_law_family_chain}) over $f\in\{0,1\}$ and applying
    (\ref{eq:area_law_partition_counts}) yields
    \begin{align}
        S_\Omega(A)\le S_\Omega(D)+\frac12\sum_i\eta_i
         & \le|D|\log q+\frac C2\sum_in_i^{-100}\le C'b.
        \label{eq:area_law_final_bound}
    \end{align}
\end{proof}

\begin{definition}[Open-square Hamiltonian and PEPS approximation]
    \label{def:polynomial_peps_model}
    \lean{
        TNLean.PEPS.Approximation.Configuration,
        TNLean.PEPS.Approximation.StateSpace,
        TNLean.PEPS.Approximation.SquareHamiltonian,
        TNLean.PEPS.Approximation.SquareHamiltonian.operator,
        TNLean.PEPS.Approximation.SquareHamiltonian.IsGappedGroundState,
        TNLean.PEPS.Approximation.pepsVector,
        TNLean.PEPS.Approximation.HasPEPSApproximation,
        TNLean.PEPS.Approximation.PolynomialPEPSApproximation
    }
    \leanok
    On the open $L\times L$ grid, let $H=H^*$ be a sum of site and edge terms
    \begin{align}
        H       & =\sum_v h_v+\sum_e h_e,
        \label{eq:polynomial_peps_hamiltonian} \\
        \|h_v\| & \le J,
        \label{eq:polynomial_peps_site_bound}  \\
        \|h_e\| & \le J.
        \label{eq:polynomial_peps_edge_bound}
    \end{align}
    For a unit vector $\Omega$, $E_0\in\mathbb R$, and $\Delta>0$, the gapped
    ground-state condition is
    \begin{align}
        H\Omega=E_0\Omega,
        \qquad
        H-E_0I & \ge\Delta(I-|\Omega\rangle\langle\Omega|).
        \label{eq:polynomial_peps_projector_gap}
    \end{align}
    A PEPS approximation with parameters $C,c>0$ is a tensor family on the grid
    edges with bond dimensions $1\le D_e\le CL^c$ and nonzero contracted vector
    $\Phi$ satisfying, for some $\theta\in\mathbb R$,
    \begin{align}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
        \label{eq:polynomial_peps_vector_error}
    \end{align}
    Uniform polynomial PEPS approximation requires $C,c>0$ depending only on
    $(q,J,\Delta)$ ($q\ge2$, $J,\Delta>0$) prior to all $L\ge2$, $H$, and
    $\Omega$.
\end{definition}

\begin{lemma}[Minimizing phase]
    \label{lem:polynomial_peps_minimizing_phase}
    \lean{
        TNLean.PEPS.Approximation.exists_phase_isMinOn,
        TNLean.PEPS.Approximation.exists_phase_le_iff_iInf_le,
        TNLean.PEPS.Approximation.hasPEPSApproximation_iff_iInf
    }
    \leanok
    For vectors $u,v$ in a complex normed space, $\theta\mapsto\|u-e^{i\theta}v\|$
    attains its minimum on $\mathbb R$. Thus $\exists\theta\in\mathbb R$,
    $\|u-e^{i\theta}v\|\le\varepsilon$ if and only if
    $\min_{\theta\in\mathbb R}\|u-e^{i\theta}v\|\le\varepsilon$, matching
    (\ref{eq:polynomial_peps_vector_error}).
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:polynomial_peps_model}
    Since $\theta\mapsto\|u-e^{i\theta}v\|$ is continuous and $2\pi$-periodic,
    it attains its minimum on the compact interval $[0,2\pi]$.
\end{proof}

\begin{lemma}[Support and norm of the Hermitian terms]
    \label{thm:square_hermitian_term_properties}
    \lean{

        TNLean.PEPS.Approximation.SquareHamiltonian.hermitianTerm_isHermitian,
        TNLean.PEPS.Approximation.SquareHamiltonian.hermitianTerm_norm_le
    }
    \leanok

    For every $L,q\ge0$, $J\in\mathbb R$, and $i\in\mathcal I_L$, the term
    $A_i^{\mathrm h}$ is supported on $S_i$ and satisfies
    $(A_i^{\mathrm h})^*=A_i^{\mathrm h}$ and $\|A_i^{\mathrm h}\|\le J$.
\end{lemma}
\begin{proof}
    \leanok

    $A_i^{\mathrm h}=(A_i+A_i^*)/2$, with $(A_i^{\mathrm h})^*=A_i^{\mathrm h}$

\begin{lemma}[The original nearest-neighbor interaction family]

    Lemma~\ref{thm:square_interaction_support_relabel}, $D_{e(i)}=X_i$; by
    Lemma~\ref{thm:square_hermitian_term_properties}, $T_{e(i)}$ is Hermitian

    Lemma~\ref{thm:square_hermitian_term_sum} and
    Lemma~\ref{thm:square_area_law_hamiltonian_operator},
    $\sum_{j\in\mathcal I}T_j=U\bigl(\sum_{i\in\mathcal I_L}A_i^{\mathrm h}\bigr)U^*

\end{proof}

\begin{lemma}[The same ground vector and full-system gap]

\begin{theorem}[Polynomial PEPS approximation]
    \label{thm:openai_polynomial_peps}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation}
    \leanok
    \uses{def:polynomial_peps_model,lem:polynomial_peps_minimizing_phase}
    In \cite[Theorem~1.1]{OpenAI2026PolynomialPEPS}, for fixed $q\ge2$ and
    $J,\Delta>0$, there exist $C,c>0$ such that for every $L\ge2$, every
    Hamiltonian $H$ of Definition~\ref{def:polynomial_peps_model}, and every unit
    ground vector $\Omega$ satisfying (\ref{eq:polynomial_peps_projector_gap}),
    there is a nonzero PEPS $\Phi$ on the open $L\times L$ grid with
    $D_e\le CL^c$ for every edge $e$ and
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
        \label{eq:polynomial_peps_phase_error}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:polynomial_peps_explicit_constants}
    Corollary~\ref{thm:polynomial_peps_explicit_constants} gives the result with
    $C=C_{\rm PEPS}$ and $c=c_{\rm PEPS}$, assembled as follows. Write
    $\Lambda_L$ for the grid sites and $\mathcal H_X$ for the tensor product
    over $X\subseteq\Lambda_L$. A dyadic party tree distributes $\Omega$ from a
    single root party to one party per site. Across tapered buffers, the
    area-law and collar bounds reduce mutual information below any fixed
    $L^{-p}$, allowing approximate purification splits. Near logarithmic-scale
    squares, $\Omega$ projects onto a sum of orthogonal subspaces
    \begin{align}
        \operatorname{span}\{v_{jk}\}_{k=1}^{d_j}
        \otimes\mathcal H_{\Lambda_L\setminus X_j},
        \qquad
        \sum_jd_j & \le C_0L^{c_0},
        \label{eq:polynomial_peps_rank_bound}
    \end{align}
    for nested squares $X_j$ and orthonormal $v_{jk}\in\mathcal H_{X_j}$.
    Telescoping the polynomial-rank party operations yields a final vector
    $\Psi$ and reference $\Omega\otimes\Xi$ satisfying
    \begin{align}
        \|\Psi-\Omega\otimes\Xi\| & \le CL^{-17},
        \qquad\|\Psi\|\le1,
        \qquad 1-CL^{-57}\le\|\Xi\|\le1,
        \label{eq:polynomial_peps_protocol_error} \\
        \bigl\|\operatorname{Tr}_{\rm private}|\Psi\rangle\langle\Psi|
        -|\Omega\rangle\langle\Omega|\bigr\|_1
                                  & \le CL^{-10}.
        \label{eq:polynomial_peps_protocol_density}
    \end{align}
    Density compression replaces the bipartite resources by polynomial sums of
    product operators, producing an operator network $\sigma$ on the party graph
    with polynomial link dimensions and
    \begin{align}
        \bigl\|\sigma-|\Omega\rangle\langle\Omega|\bigr\|_1
         & \le\eta:=CL^{-5}.
        \label{eq:polynomial_peps_compressed_density}
    \end{align}
    For $\eta<1$, let $E=\sigma-|\Omega\rangle\langle\Omega|$ and
    $a_z=\langle\Omega|z\rangle$. Since $\sum_z\|E|z\rangle\|^2\le\eta^2$ and
    $\sum_z|a_z|^2=1$, some product basis state $|z\rangle$ has $a_z\ne0$ and
    $\|E|z\rangle\|\le\eta|a_z|$, so $v=\sigma|z\rangle\ne0$ satisfies
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{v}{\|v\|}-e^{i\theta}\Omega\right\|
         & \le2\eta.
        \label{eq:polynomial_peps_column_error}
    \end{align}
    Routing links with congestion at most $\chi$ and link dimension at most
    $C_1L^k$ gives bond dimensions
    \begin{align}
        D_e & \le(C_1L^k)^\chi=C_1^\chi L^{k\chi}.
        \label{eq:polynomial_peps_routed_bond}
    \end{align}
    For $L\ge L_0$, $2CL^{-5}\le L^{-1}$, so (\ref{eq:polynomial_peps_column_error})
    yields (\ref{eq:polynomial_peps_phase_error}); sizes $2\le L<L_0$ are
    absorbed by enlarging the prefactor in (\ref{eq:polynomial_peps_routed_bond}).
\end{proof}

        \label{eq:area_law_template_pieces}                                        \\

    \begin{align}
        T_j & =T+([-j,j]^2\cap\mathbb Z^2),
        \label{eq:area_law_template_dilation} \\
        X   & =A\cap T,
        \label{eq:area_law_template_core}     \\
        Q_j & =A\cap(T_j\setminus T),
        \label{eq:area_law_template_collar}

\begin{lemma}[Sublinear information across a template collar]
    \label{thm:area_law_sublinear_collar}
    \lean{TNLean.PEPS.AreaLaw.sublinearCollarBound_real}
    \leanok
    \uses{def:area_law_collar_bound}
    In~\cite[Proposition~9.6]{OpenAI2026AreaLaw}, fix integers $q\geq1$,
    $R\ge0$, reals $J\geq0$, $\Delta>0$, $D_0>2R+10$, and
    $C_{\mathrm{tpl}}\geq24$. There exist $C\in\mathbb R$ and $N\in\mathbb N$
    such that for every finite induced domain $\Lambda$, admissible Hamiltonian
    $H$ with parameters $(q,R,J)$, unit ground vector $\Omega$ with gap
    $\Delta$, cut $A\subseteq\Lambda$ with cut endpoints $Z$, and template $T$
    with parameters $(C_{\mathrm{tpl}},n,s_0)$ satisfying $n\geq N$,
    $\operatorname{dist}_\infty(T,Z)>4D_0s_0$, and
    $L=\lfloor n^{1-1/100000}\rfloor\leq s_0$, setting $X=A\cap T$ and
    $S_0=A\cap T_L$ gives
    \begin{align}
        I_\Omega(X:\Lambda\setminus S_0)                                                   & \leq Cn^{1-1/100000}, &
        |S_0|                                                                              & \leq Cn^2,            &
        |\partial_\Lambda X|,\,|\partial_\Lambda S_0|,\,|\partial_\Lambda(S_0\setminus X)| & \le Cn.\notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:arealaw-physical-box-fixed-exponent,
        thm:arealaw-box-clearance-comparison,
        thm:template-final-information,
        thm:area_law_collar_bound_of_information}
    With $d=2R+10$ and $e_*=2\cdot10^{-6}$,
    Theorem~\ref{thm:arealaw-physical-box-fixed-exponent} at clearance $d+1$
    gives $C_b>0$, and Lemma~\ref{thm:arealaw-box-clearance-comparison} gives
    \begin{align}
        F_{\mathrm{box},d}(r)
         & \leq (4C_b+4\log q+1)r^{1+e_*}\qquad(r\ge1).\notag
    \end{align}
    Since $D_0>d$, Theorem~\ref{thm:template-final-information} at clearance
    $d$ supplies $C_m>0$ and $N$ with
    $2I_\Omega(X:\Lambda\setminus S_0)\leq C_m n^{1-1/100000}$ for $n\ge N$, and
    Lemma~\ref{thm:area_law_collar_bound_of_information} gives the remaining
    bounds with $C=\max(C_m/2,10)$.
\end{proof}
