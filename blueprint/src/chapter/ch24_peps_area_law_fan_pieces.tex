\section{Connected pieces of the midpoint fan}
\label{sec:area_law_fan_pieces}

A vector $(u_1,u_2)\in\R^2$ has an allowed direction if
$u_1=0$, $u_2=0$, $u_1=u_2$ or $u_1=-u_2$.
The following observation will be used for the radial segments of
the elementary fans in \cite[Section~11]{OpenAI2026AreaLaw}.

\begin{theorem}[Allowed directions in an affine span]
    \label{thm:area_law_allowed_direction_affine_span}
    \lean{TNLean.PEPS.AreaLaw.Geometry.isAllowedSlope_sub_of_mem_affineSpan_pair}
    Let $p,q\in\R^2$ and suppose that $q-p$ has an allowed direction.
    If $v,x\in\operatorname{aff}\{p,q\}$, then $x-v$ has an allowed
    direction. The points $p,q$ may coincide.
\end{theorem}
\begin{proof}
    Write $v=(1-r)p+rq$ and $x=(1-s)p+sq$ with $r,s\in\R$.
    Then $x-v=(s-r)(q-p)$. Each of the four equations defining an
    allowed direction is preserved by real scalar multiplication.
\end{proof}

\begin{theorem}[Allowed base and radial directions]
    \label{thm:area_law_fan_base_radial_directions}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_base_and_radial_isAllowedSlope}
    \uses{def:area_law_cell_fan}
    In an elementary fan, the vectors $b_i-a_i$ and $c-b_i$
    have allowed directions.
    Each side may be left whole or subdivided at its midpoint.
\end{theorem}
\begin{proof}
    \uses{def:area_law_template,def:area_law_cell_fan}
    The fan triangle with vertices $c,a_i,b_i$ is a polygonal template.
    Its edge directions include $b_i-a_i$ and $c-b_i$, so both have
    one of the four prescribed slopes.
\end{proof}

For the next result, subdivide every side of the dyadic square at its
midpoint. There are then eight triangles $P_i$ and eight radial segments
$R_i=[c,b_i]$, with the notation of the preceding section.
The plane carries the maximum norm.

\begin{theorem}[Nonempty connected pieces after radial deletion]
    \label{thm:area_law_fan_cut_piece_connected}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_sdiff_radials_inter_ball_nonempty}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_sdiff_radials_inter_ball_isConnected}
    \uses{def:area_law_cell_fan}
    Let $A$ be any subset of the eight radial indices and let $r>0$.
    For each triangle index $i$, the set
    \begin{align}
        \left(P_i\setminus\bigcup_{t\in A}R_t\right)\cap B_\infty(c,r)
    \end{align}
    is nonempty and connected. The set $A$ may be empty.
\end{theorem}
\begin{proof}
    \uses{thm:area_law_cell_fan_regularity,
        thm:area_law_cell_fan_sector_directions,
        thm:area_law_allowed_direction_affine_span,
        thm:area_law_fan_base_radial_directions,
        thm:area_law_fan_radial_union_deletion_convex}
    The triangle $P_i$ is the closure of its interior and contains $c$.
    Hence $B_\infty(c,r)$ contains a point $x$ of the interior of $P_i$.
    The vector $x-c$ has none of the four allowed directions.
    Each radial has an allowed direction, and so does every difference
    of points in its affine span. Thus $x$ lies on no radial, proving
    nonemptiness even when all eight radials are deleted.

    The triangle with the selected radials deleted is convex, as is
    the ball. Their intersection is therefore a nonempty convex subset
    of a real normed space. It is path connected and hence connected.
    If $A$ is empty, no radial is deleted and the center is retained.
\end{proof}
