\subsection{Full-matrix presentation of a factor with scalar centre}

\begin{lemma}[Scalar centre forces a full matrix algebra of a definite size]
    \label{thm:qca_central_star_subalgebra_matrix}
    \lean{StarSubalgebra.exists_starAlgEquiv_matrix_and_finrank_of_isCentral}
    \leanok
    Let $I$ be a finite nonempty set and let $S\subseteq M_I(\C)$ be a unital
    star-subalgebra whose centre consists of the multiples of its unit,
    \begin{align}
        \{z\in S\mid zs=sz\text{ for every }s\in S\}
         & =\C\mathbf 1.
        \label{eq:qca_central_star_subalgebra_scalar_centre}
    \end{align}
    As in the single-block case of the structure theorem for
    finite-dimensional $C^*$-algebras
    \cite[Proposition~\texttt{Csform}, lines~2082--2098]{SchumacherWerner2004QCA}
    for an algebra presented inside a matrix algebra
    \cite{gap:sw04_csform_matrix_scope}, there is an $r\geq1$ with
    \begin{align}
        S            & \cong\MN{r}, \notag \\
        \dim_{\C}(S) & =r^2,
    \end{align}
    where the first isomorphism is an isomorphism of star-algebras.
\end{lemma}

\begin{proof}\leanok
    A star-subalgebra of a complex matrix algebra is semisimple, so it is
    isomorphic to a finite product $\prod_{k=1}^{m}\MN{d_k}$ with every
    $d_k\geq1$.  The element that is the unit in the $k$-th coordinate and
    zero in the others is central, hence by
    (\ref{eq:qca_central_star_subalgebra_scalar_centre}) a multiple
    $\lambda\mathbf 1$; comparing its $k$-th coordinate with another one gives
    $\lambda=1$ and $\lambda=0$, so $m\leq1$.  The unit of $M_I(\C)$ is
    nonzero because $I$ is nonempty, so $m=1$ and $S$ is a simple ring.  By the
    Wedderburn theorem over the algebraically closed field $\C$
    \cite[``A $*$-subalgebra with scalar centre is a full matrix
        algebra'']{QICLean2026Blueprint}, there are an $r\geq1$ and an
    isomorphism of $\C$-algebras $S\cong\MN{r}$, which preserves adjoints after
    conjugation by a positive definite matrix.  Since a star-isomorphism is in
    particular a $\C$-linear isomorphism
    \cite[``Dimension of a subalgebra presented as a full matrix
        algebra'']{QICLean2026Blueprint}, $\dim_{\C}(S)=\dim_{\C}\MN{r}=r^2$.
\end{proof}

Gross, Nesme, Vogts and Werner apply this conclusion to the site-indexed
support algebras \cite[lines~1276--1282]{Gross2012IndexTheory}: an element of
$\mathcal R_x$ commuting with all of $\mathcal R_x$ commutes with every
$\mathcal R_y$, hence with the algebra they generate, which is the whole
quasi-local algebra; its centre is scalar, so
(\ref{eq:qca_central_star_subalgebra_scalar_centre}) holds for $\mathcal R_x$
and $\mathcal R_x\cong M_{r(x)}(\C)$.


\begin{lemma}[A full-matrix star-isomorphism has a unitary implementer]
    \label{lem:qca_full_matrix_spatial_implementer}
    \lean{Matrix.exists_unitary_conj_of_starAlgEquiv_matrix_between}
    \notready
    Let $I,J$ be nonempty finite coordinate sets and let
    $f:M_I(\mathbb C)\to M_J(\mathbb C)$ be a complex star-algebra
    isomorphism. Then $|I|=|J|$ and there is a unitary map
    $U:\mathbb C^I\to\mathbb C^J$ with
    \begin{align}
        U^\dagger U=I,\qquad UU^\dagger=I,\qquad
        f(A)=UAU^\dagger\quad(A\in M_I(\mathbb C)).
    \end{align}
    This applies between distinct coordinate types, as required for the
    unitaries in \cite[equations~\textup{bruleU} and~\textup{bruleV},
        lines~980--1001]{SchumacherWerner2004QCA}.
\end{lemma}
\begin{proof}
    \notready
    The algebra isomorphism is complex linear, so $|I|^2=|J|^2$ and
    $|I|=|J|$. Choose an equivalence of the two coordinate sets and
    transport $f$ to an automorphism of one full matrix algebra.
    The unitary Skolem--Noether theorem implements that automorphism.
    Transport its columns back to the original input coordinates;
    multiplication and adjoints respect this reindexing, yielding the
    displayed formula and both unitary identities.
\end{proof}
