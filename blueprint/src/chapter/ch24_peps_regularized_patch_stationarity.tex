\begin{definition}[One-coordinate variation and insertion]
    \label{def:peps_patch_coordinate_insertion}
    \lean{
        TNLean.PEPS.regularizedPatchCoordinateUpdate,
        TNLean.PEPS.regularizedPatchInsertion,
        TNLean.PEPS.regularizedPatchInsertion_apply,
        TNLean.PEPS.regularizedPatchInsertion_self,
        TNLean.PEPS.regularizedPatchUnitaryCurve
    }
    \leanok
    \uses{def:peps_regularized_patch_objective}
    Fix $0\le j<m$ and a unitary $U$ on $\mathcal H_{X_j}$.
    Define $x^{j,U}$ by $(x^{j,U})_j=Ux_jU^*$ and $(x^{j,U})_k=x_k$ for $k\ne j$, and set
    \begin{align}
        P_j^+           & =L_{m-1}\cdots L_{j+1},            &
        P_j^-           & =L_{j-1}\cdots L_0,\notag            \\
        \mathcal I_j(K) & =P_j^+\widehat K_{X_j}P_j^-\Omega,
        \label{eq:peps_coordinate_insertion}
    \end{align}
    with empty products equal to $I$. Then $\mathcal I_j$ is $\C$-linear,
    $\mathcal I_j(K_j)=\Psi(x)$, and $U(t)=e^{tB}$ is unitary for $B^*=-B$ and $t\in\R$.
\end{definition}

\begin{lemma}[Feasibility and covariance]
    \label{thm:peps_patch_coordinate_covariance}
    \lean{
        TNLean.PEPS.regularizedPatchCoordinateUpdate_mem,
        TNLean.PEPS.regularizedPatchCoordinateUpdate_rpow,
        TNLean.PEPS.regularizedPatchOutput_coordinateUpdate
    }
    \leanok
    \uses{def:peps_patch_coordinate_insertion}
    For $x\in\mathcal D$, $b>0$, $a_j\in\R$, and a unitary $U$
    on $\mathcal H_{X_j}$, one has $x^{j,U}\in\mathcal D$ and
    \begin{align}
        (Ux_jU^*+bI)^{-a_j/2} & =UK_jU^*,
        \label{eq:peps_coordinate_covariance_rpow}      \\
        \Psi(x^{j,U})         & =\mathcal I_j(UK_jU^*).
        \label{eq:peps_coordinate_covariance_apply}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Unitary conjugation preserves $x_j\ge0$, $\tr x_j=1$, and $I$,
    so $Ux_jU^*+bI=U(x_j+bI)U^*>0$. Unitary covariance of $t\mapsto t^{-a_j/2}$ gives
    (\ref{eq:peps_coordinate_covariance_rpow}), and substituting into (\ref{eq:peps_coordinate_insertion}) gives~(\ref{eq:peps_coordinate_covariance_apply}).
\end{proof}

\begin{lemma}[Ordered first variation]
    \label{thm:peps_patch_first_variation}
    \lean{
        TNLean.PEPS.hasDerivAt_regularizedPatchOutput_coordinateUpdate,
        TNLean.PEPS.regularizedPatchFirstVariation_eq_zero
    }
    \leanok
    \uses{def:peps_patch_coordinate_insertion}
    Let $x\in\mathcal D$, $b>0$, $a_j\in\R$, and $B^*=-B$ on $\mathcal H_{X_j}$. Then
    \begin{align}
        \left.\frac{d}{dt}\Psi(x^{j,e^{tB}})\right|_{t=0}
         & =\mathcal I_j(BK_j-K_jB).
        \label{eq:peps_coordinate_derivative}
    \end{align}
    If $F(x)\le F(y)$ for all $y\in\mathcal D$, then
    \begin{align}
        \operatorname{Re}\langle\Psi(x),\mathcal I_j(BK_j-K_jB)\rangle=0.
        \label{eq:peps_coordinate_first_variation}
    \end{align}

    With $D_j=\widehat{BK_j-K_jB}_{X_j}$ and
    $\Delta_j=P_j^+D_jP_j^-\Omega$, every line below carries the full space $\mathcal H_V$:
    \input{figures/peps_patch_first_variation}
    Starting at $\Omega$ on the right, $P_j^-$ acts first, $D_j$ second,
    and $P_j^+$ last ($P_j^\pm$ need not be supported on $X_j$).
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_patch_coordinate_covariance}
    Differentiating $e^{tB}K_je^{-tB}$ at $t=0$ gives $BK_j-K_jB$, so linearity of $\mathcal I_j$ and (\ref{eq:peps_coordinate_covariance_apply}) yield
    (\ref{eq:peps_coordinate_derivative}). Since $x^{j,e^{tB}}\in\mathcal D$ for all $t\in\R$,
    $t\mapsto\|\Psi(x^{j,e^{tB}})\|^2$ is minimized at $t=0$, where its derivative
    $2\operatorname{Re}\langle\Psi(x),\mathcal I_j(BK_j-K_jB)\rangle$ vanishes.
\end{proof}

\begin{lemma}[Radius derivative of the ordered patch output]
    \label{thm:peps_patch_radius_derivative}
    \lean{TNLean.PEPS.hasDerivAt_regularizedPatchOutput_exp_neg}
    \leanok
    \uses{def:peps_patch_coordinate_insertion}
    Let $V$ be a finite ordered set with finite local alphabets $A_v$
    and $\mathcal H_X=\ell^2(\prod_{v\in X}A_v;\C)$.
    Fix $X_0,\ldots,X_{m-1}\subseteq V$, $a_j\in\R$,
    $x_j\ge0$ on $\mathcal H_{X_j}$, and $\Omega\in\mathcal H_V$.
    For $r\in\R$, set $K_j(r)=(x_j+e^{-r}I_{X_j})^{-a_j/2}$ and
    \begin{align}
        \Psi(r)             & =\widehat K_{m-1}(r)\cdots\widehat K_0(r)\Omega,\notag \\
        \mathcal I_{j,r}(H) & =
        \widehat K_{m-1}(r)\cdots\widehat K_{j+1}(r)
        \widehat H_{X_j}
        \widehat K_{j-1}(r)\cdots\widehat K_0(r)\Omega.
        \label{eq:peps_patch_radius_output}
    \end{align}
    Then $\Psi$ is differentiable at every $R\in\R$, with
    \begin{align}
        \Psi'(R)
         & =\sum_{j=0}^{m-1}\mathcal I_{j,R}\!\left(
        \frac{a_j}{2}e^{-R}
        (x_j+e^{-R}I_{X_j})^{-a_j/2-1}\right).
        \label{eq:peps_patch_radius_derivative}
    \end{align}
    This is the ordered product rule underlying the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L425-L437}{radius derivative estimate for regularized patches}.
\end{lemma}
\begin{proof}
    \leanok
    Since $x_j+e^{-R}I>0$, $K_j'(R)=\frac{a_j}{2}e^{-R}(x_j+e^{-R}I)^{-a_j/2-1}$.
    Linearity of $H\mapsto\widehat H_{X_j}$ and the Leibniz rule on (\ref{eq:peps_patch_radius_output}) yield (\ref{eq:peps_patch_radius_derivative}).
\end{proof}

\begin{lemma}[Regional expectation and density properties]
    \label{thm:peps_patch_marginal_expectation}
    \lean{
        TNLean.PEPS.inner_dependentRegionOperatorLift_eq_trace_reducedPure,
        TNLean.PEPS.normalizedRegularizedPatchMarginal_posSemidef,
        TNLean.PEPS.trace_normalizedRegularizedPatchMarginal,
        TNLean.PEPS.inner_normalizedRegularizedPatchOutput_lift_eq_trace
    }
    \leanok
    \uses{def:peps_patch_canonical_marginal}
    For every $\xi\in\mathcal H_V$, $X\subseteq V$, and $K\in\operatorname{End}(\mathcal H_X)$,
    \begin{align}
        \langle\xi,\widehat K_X\xi\rangle
         & =\tr\left(\operatorname{Tr}_{V\setminus X}
        \ketbra{J\xi}{J\xi}\,K\right).
        \label{eq:peps_complex_expectation}
    \end{align}
    In particular, $\langle\phi,\widehat K_X\phi\rangle=\tr(\rho_XK)$ and $\rho_X\ge0$; if $a_j\ge0$, $b>0$,
    $\|\Omega\|=1$, and $x\in\mathcal D$, then $\tr\rho_X=1$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regularized_patch_minimum}
    Under $J$, $\widehat K_X$ corresponds to $K\otimes I_{V\setminus X}$, so $\tr(\operatorname{Tr}_{V\setminus X}(P)K)=\tr(P(K\otimes I))$ for $P=\ketbra{J\xi}{J\xi}$ gives (\ref{eq:peps_complex_expectation}). Partial trace preserves positivity and trace, and $\|\phi\|=1$ by Lemma~\ref{thm:peps_regularized_patch_minimum}.
\end{proof}

\begin{lemma}[Trace pairing for an insertion in a nested product]
    \label{thm:peps_patch_insertion_trace}
    \lean{TNLean.PEPS.inner_regularizedPatchInsertion_eq_trace_of_outer_commute}
    \leanok
    \uses{def:peps_patch_coordinate_insertion,
        def:peps_patch_canonical_marginal}
    Let $X_0\subseteq\cdots\subseteq X_{m-1}\subseteq V$, $x_k\ge0$ on
    $\mathcal H_{X_k}$, $a_k\in\R$, $b>0$, and $\Omega\in\mathcal H_V$. Set
    $K_k=(x_k+bI_{X_k})^{-a_k/2}$, $L_k=\widehat{K_k}_{X_k}$,
    $\Psi(x)=L_{m-1}\cdots L_0\Omega$, $F=\|\Psi(x)\|$,
    and $\phi=F^{-1}\Psi(x)$ ($0$ if $F=0$), with canonical marginal $\rho_k$ on $X_k$.
    Fix $0\le j<m$ and suppose $[K_k,\rho_k]=0$ for all $k>j$.
    Then for every $H\in\operatorname{End}(\mathcal H_{X_j})$, the insertion $\mathcal I_j(H)$ of (\ref{eq:peps_coordinate_insertion}) satisfies
    \begin{align}
        \langle\Psi(x),\mathcal I_j(H)\rangle
         & =F^2\operatorname{Tr}(\rho_j H K_j^{-1}).
        \label{eq:peps_patch_insertion_trace}
    \end{align}
    This is the trace calculation in the proof of
    \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_patch_marginal_expectation,
        thm:peps_patch_basis_relabelling}
    If $F=0$, both sides of (\ref{eq:peps_patch_insertion_trace}) vanish. Assume $F>0$.
    Since $L_k$ is invertible, setting $A=L_{m-1}\cdots L_{j+1}$ and
    $Y=\widehat{H K_j^{-1}}_{X_j}$ gives $\mathcal I_j(H)=A Y A^{-1}\Psi(x)$ and
    $\langle\Psi(x),\mathcal I_j(H)\rangle=F^2\langle\phi,A Y A^{-1}\phi\rangle$.
    For any $T\in\operatorname{End}(\mathcal H_{X_k})$ with $k>j$, (\ref{eq:peps_complex_expectation}), trace cyclicity, and $[K_k,\rho_k]=0$ give
    \begin{align}
        \langle\phi,L_k\widehat T_{X_k}L_k^{-1}\phi\rangle
         & =\operatorname{Tr}(K_k^{-1}\rho_k K_k T)
        =\operatorname{Tr}(\rho_k T)
        =\langle\phi,\widehat T_{X_k}\phi\rangle.\notag
    \end{align}
    Since $X_j\subseteq\cdots\subseteq X_k$, descending induction on $k=m-1,\ldots,j+1$ gives
    $\langle\phi,A Y A^{-1}\phi\rangle=\langle\phi,Y\phi\rangle=\operatorname{Tr}(\rho_j H K_j^{-1})$.
\end{proof}


\begin{lemma}[Logarithmic radius derivative under outer commutations]
    \label{thm:peps_patch_log_radius_derivative_of_outer_commute}
    \lean{TNLean.PEPS.hasDerivAt_log_regularizedPatchObjective_exp_neg_of_outer_commute}
    \leanok
    \uses{thm:peps_patch_radius_derivative,thm:peps_patch_insertion_trace,
        def:peps_patch_canonical_marginal}
    Let $X_0\subseteq\cdots\subseteq X_{m-1}\subseteq V$, $a_j\ge0$,
    $x_j\in\mathcal D_{X_j}$, and $\|\Omega\|=1$.
    For $r\in\R$, set $K_j(r)=(x_j+e^{-r}I_{X_j})^{-a_j/2}$,
    $\Psi(r)=\widehat K_{m-1}(r)\cdots\widehat K_0(r)\Omega$,
    and $F(r)=\|\Psi(r)\|$, with reduced density matrix $\rho_j(r)$ of $\Psi(r)/F(r)$ on $X_j$.
    If $[K_k(R),\rho_k(R)]=0$ for all $0\le j<k<m$ at a fixed $R\in\R$, then
    \begin{align}
        \left.\frac{d}{dr}\log F(r)\right|_{r=R}
        =\sum_{j=0}^{m-1}\frac{a_j}{2}
        \operatorname{Re}\Tr\!\left(
        \rho_j(R)e^{-R}(x_j+e^{-R}I_{X_j})^{-1}\right).
        \label{eq:peps_patch_log_radius_derivative}
    \end{align}
    This is the trace computation in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L453-L466}{radius argument for regularized patches}.
\end{lemma}
\begin{proof}
    \leanok
    Since $K_j(r)$ is invertible, $F(r)>0$.
    Differentiating $\frac12\log\|\Psi(r)\|^2$ via (\ref{eq:peps_patch_radius_derivative}) and applying (\ref{eq:peps_patch_insertion_trace}) with $b=e^{-R}$ and $H K_j(R)^{-1}=\frac{a_j}{2}b(x_j+bI)^{-1}$ cancels $F(R)^2$ and yields (\ref{eq:peps_patch_log_radius_derivative}).
\end{proof}

\begin{lemma}[Uniform bound for the logarithmic radius derivative]
    \label{thm:peps_patch_uniform_log_radius_derivative}
    \lean{TNLean.PEPS.exists_hasDerivAt_log_regularizedPatchObjective_exp_neg_uniform}

    and a unit vector $\Omega\in\mathcal H_V$.
    For $x\in\mathcal D$, set
    $\Psi_x(r)=\widehat{(x_{m-1}+e^{-r}I)^{-a_{m-1}/2}}\cdots\widehat{(x_0+e^{-r}I)^{-a_0/2}}\Omega$
    and $F_x(r)=\|\Psi_x(r)\|$.
    Then for every $T\ge0$, $x\in\mathcal D$, and $R\in[0,T]$,
    \begin{align}
        \left|\frac{d}{dr}\log F_x(r)\bigg|_{r=R}\right|
         & \leq \frac{\kappa}{2}\,2^{\kappa/2}e^{\kappa T/2}.
        \label{eq:peps_patch_uniform_log_radius_derivative}
    \end{align}
    Source: the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L433-L451}{uniform local Lipschitz estimate for regularized patches}.
\end{lemma}
\begin{proof}
    \uses{thm:peps_patch_radius_derivative,thm:peps_regularized_patch_bounds}
    Set $b=e^{-R}$ and $c=\kappa/2$.
    For $K_j=(x_j+bI)^{-a_j/2}$ and $D_j=\frac{a_j}{2}b(x_j+bI)^{-a_j/2-1}$, Lemma~\ref{thm:peps_regularized_patch_bounds} gives
    $\|K_j\|\le b^{-a_j/2}$, $\|D_j\|\le\frac{a_j}{2}b^{-a_j/2}$, and $F_x(R)\ge(1+b)^{-c}$.
    Summing over $j$ in (\ref{eq:peps_patch_radius_derivative}) yields $\|\Psi_x'(R)\|\le c b^{-c}$.
    By Cauchy--Schwarz and $0\le R\le T$,
    \begin{align}
        |(\log F_x)'(R)|
         & \leq\frac{\|\Psi_x'(R)\|}{F_x(R)}
        \leq c\frac{b^{-c}}{(1+b)^{-c}}
        =c(1+e^R)^c
        \le c(2e^T)^c.\notag
    \end{align}
\end{proof}

\begin{definition}[Diagonal coordinate objective]
    \label{def:peps_patch_diagonal_objective}
    \lean{
        TNLean.PEPS.regularizedPatchDiagonalUpdate,
        TNLean.PEPS.regularizedPatchDiagonalObjective
    }
    \leanok
    \uses{def:peps_regularized_patch_objective}
    Fix $0\le j<m$ and a unitary $U$ on $\mathcal H_{X_j}$.
    For a real vector $y=(y_i)$ indexed by an orthonormal basis of $\mathcal H_{X_j}$,
    define $x[y]$ by $x[y]_k=x_k$ for $k\ne j$ and
    \begin{align}
        x[y]_j & =U\operatorname{diag}(y)U^*, &
        f(y)   & =F(x[y])^2.
        \label{eq:peps_patch_diagonal_objective}
    \end{align}
    This is the diagonal variation in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L170-L191}{proof of Proposition~4.1}.
\end{definition}

\begin{lemma}[Restriction of a patch minimum to the diagonal simplex]
    \label{thm:peps_patch_diagonal_minimum}
    \lean{TNLean.PEPS.isMinOn_regularizedPatchDiagonalObjective}
    \leanok
    \uses{def:peps_patch_diagonal_objective}
    If $b>0$, $x\in\mathcal D$ minimizes $F$ on $\mathcal D$, and
    $x_j=U\operatorname{diag}(s)U^*$, then for every $y$ with
    $y_i\ge0$ and $\sum_i y_i=1$,
    \begin{align}
        f(s) & \le f(y).
        \label{eq:peps_patch_diagonal_minimum}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:peps_patch_diagonal_objective}
    Since $y_i\ge0$ and $\sum_i y_i=1$, $x[y]_j\ge0$ and $\tr x[y]_j=1$, so $x[y]\in\mathcal D$.
    From $x[s]=x$ and minimality of $x$, $0\le F(x[s])=F(x)\le F(x[y])$, which squares to (\ref{eq:peps_patch_diagonal_minimum}).
\end{proof}

\begin{lemma}[Clipped spectra at a regularized patch minimum]
    \label{thm:peps_patch_minimum_clipped}
    \leanok
    \lean{TNLean.PEPS.exists_unitary_diagonal_clipped_of_isMinOn_regularizedPatchObjective}
    \uses{def:peps_regularized_patch_objective,def:peps_patch_canonical_marginal}
    Let $X_0\subseteq\cdots\subseteq X_{m-1}$, $a_j>0$, $b>0$,
    and $\|\Omega\|=1$. If $x\in\mathcal D$
    minimizes $F$ on $\mathcal D$ and $\rho_j$ is the marginal on $X_j$
    of $\Psi(x)/\|\Psi(x)\|$, then for each $j$ there exist a unitary $U_j$, real tuples $s_j,p_j$,
    and $0<\lambda_j\le1$ such that
    \begin{align}
        x_j       & = U_j\operatorname{diag}(s_j)U_j^*, \notag \\
        \rho_j    & = U_j\operatorname{diag}(p_j)U_j^*, \notag \\
        s_{j,i}+b & = \max\{p_{j,i}/\lambda_j,b\}.
        \label{eq:peps_minimum_clipped}
    \end{align}
    This is the clipped spectral formula in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L170-L200}{proof of Proposition 4.1}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_ordered_patch_minimizer,thm:peps_ordered_patch_marginal_coordinates,thm:peps_patch_marginal_expectation}
    By Lemmas~\ref{thm:peps_ordered_patch_minimizer} and~\ref{thm:peps_ordered_patch_marginal_coordinates}, $x$ minimizes the ordered-list objective and shares the normalized marginals $\rho_j$. Stationarity and the first-order inequalities in~\cite[``Stationarity of the ordered patch objective'' and ``First-order inequalities for normalized patch outputs'']{QICLean2026Blueprint} yield $[x_j,\rho_j]=0$ and, for every density matrix $y$ commuting with $x_j$,
    \begin{align}
        \operatorname{Re}\tr\bigl(\rho_j(x_j+bI)^{-1}y\bigr)
         & \leq \operatorname{Re}\tr\bigl(\rho_j(x_j+bI)^{-1}x_j\bigr).
        \label{eq:peps_minimum_density_variation}
    \end{align}
    Applying~\cite[``Clipped joint eigenvalues'']{QICLean2026Blueprint} to $x_j,\rho_j\ge0$ of trace $1$ and~(\ref{eq:peps_minimum_density_variation}) gives (\ref{eq:peps_minimum_clipped}) with $0<\lambda_j\le1$.
\end{proof}

\begin{lemma}[Logarithmic filter ratios at a patch minimum]
    \label{thm:peps_patch_minimum_filter_log_ratio}
    \lean{TNLean.PEPS.exists_clipped_filter_of_isMinOn_regularizedPatchObjective}
    \leanok
    \uses{def:peps_regularized_patch_objective,def:peps_patch_canonical_marginal}
    Under the hypotheses of Lemma~\ref{thm:peps_patch_minimum_clipped},
    for each $j$ the filter $K_j=(x_j+bI_{X_j})^{-a_j/2}$ satisfies $[K_j,\rho_j]=0$, and
    there exist a unitary $W_j$ and real tuples $l_j,p_j$ with $l_{j,i}>0$ such that
    \begin{align}
        K_j & =W_j\operatorname{diag}(l_j)W_j^*,\qquad
        \rho_j=W_j\operatorname{diag}(p_j)W_j^*, \notag           \\
        \bigl|\log l_{j,i}-\log l_{j,k}\bigr|
            & \le\frac{a_j}2\bigl|\log p_{j,i}-\log p_{j,k}\bigr|
        \label{eq:peps_patch_filter_log_ratio}
    \end{align}
    whenever $p_{j,i},p_{j,k}>0$. This is the logarithmic ratio estimate in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L201-L212}{proof of Proposition 4.1}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_patch_minimum_clipped,
        thm:peps_ordered_patch_normalized_coordinates,
        thm:peps_ordered_patch_marginal_coordinates}
    By (\ref{eq:peps_minimum_clipped}), $K_j$ is diagonal in the basis $W_j=U_j$ with eigenvalues
    $l_{j,i}=(s_{j,i}+b)^{-a_j/2}=\max\{p_{j,i}/\lambda_j,b\}^{-a_j/2}>0$, so $[K_j,\rho_j]=0$. Since $t\mapsto\log\max\{t/\lambda_j,b\}$ is $1$-Lipschitz in $\log t$ on $(0,\infty)$, multiplying by $a_j/2$ gives (\ref{eq:peps_patch_filter_log_ratio}).
\end{proof}

\begin{lemma}[Energy of a minimizing nested patch output]
    \label{thm:peps_nested_patch_minimizer_energy}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.nestedPatch_minimizer_energy_le}
    \leanok
    \uses{def:polynomial_peps_model,def:square_area_law_hamiltonian,
        def:square_area_law_state,def:peps_nested_patches,
        def:peps_regularized_patch_objective,def:peps_regularized_patch_minimum}
    Let $H$ be an original square Hamiltonian with interaction bound $J\ge0$,
    $\Delta>0$, and $\Omega$ a unit vector satisfying (\ref{eq:polynomial_peps_projector_gap}).
    Fix a site order on $\Lambda_L$, a centre $c$, $u\ge4$,
    $0<\kappa\le1$, and $b>0$, and use the nested patches $X_j=K(c,r_j)$ ($0\le j<m$) of Definition~\ref{def:peps_nested_patches}
    with weights $a_j=\kappa/m$ and input $U\Omega$.
    If $x\in\mathcal D$ minimizes $F$ on $\mathcal D$ and $\phi=\widehat\Psi(x)$, then
    \begin{align}
        \operatorname{Re}\langle\phi,H_{\mathrm{int}}\phi\rangle-E_0
         & \le36q^2J\kappa^2.
        \label{eq:peps_nested_patch_minimizer_energy}
    \end{align}
    This is the energy bound in the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L213-L336}{proof of Proposition 4.1}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regularized_patch_minimum,thm:peps_nested_patch_sizes,
        thm:peps_patch_minimum_filter_log_ratio,
        thm:peps_ordered_patch_normalized_coordinates,
        thm:peps_patch_factor_chain_list,
        thm:square_nearest_neighbor_terms,thm:square_area_law_ground_gap,
        thm:peps_nested_patch_crossing,thm:area_law_nearest_neighbor_budget,
        thm:area_law_crossing_support_count,
        thm:peps_nested_patch_quadratic_sum}
    Here $0<a_j\le1$ and $\|\phi\|=1$. With $d_{ij}=q^{|D_i\cap X_j|}$, (\ref{eq:peps_patch_filter_log_ratio}) and $a_j\le1$ yield
    \begin{align}
        \operatorname{Re}\langle\phi,H_{\mathrm{int}}\phi\rangle-E_0
         & \le\sum_{j=0}^{m-1}a_j^2

    \end{align}

\end{proof}

    $R_0(q,J,\Delta,B_0,p,C_A)$ be $R_0(C_0,B_0,p,\kappa(q,J,\Delta))$ from
    (\ref{eq:peps_patch_radius_exponent}), matching the constants in the

\end{definition}

\begin{lemma}[Nested cylinders from marginal closeness and entropy]
    \label{lem:peps_closed_square_cylinder_core}
    \lean{TNLean.PEPS.Approximation.exists_closedSquare_cylinders_of_marginal_closeness}
    \leanok
    \uses{def:peps_closed_square_sample,def:peps_regularized_patch_objective,
        def:peps_patch_canonical_marginal,def:peps_regional_cylinder,
        def:peps_patch_radius_exponent}
    Let $\Lambda$ be a finite induced lattice domain, $L\ge2$, $B_0,p,\kappa,C>0$,
    $\|\Psi\|=1$, $c$ a centre, $0\le u\le B_0\log L$, and
    $r_0\le\dots\le r_{m-1}$ ($m\ge1$). Set $X_j=K(c,r_j)$ and
    $a_j=\kappa/m$. If $S_\Psi(X_j)\le C(1+u)$ for all $j$ and, for every $b>0$, every minimizer of $F$ on $\mathcal D$ satisfies $\|\rho_{X_j}-\rho_{X_j}^\Psi\|_1\le\tfrac1{16}$, then there exist
    subspaces $S_j\subseteq\mathcal H_{X_j}$ with $\dim S_j\le L^{R_0}$ (where $R_0=R_0(16C+\log8,B_0,p,\kappa)$ from (\ref{eq:peps_patch_radius_exponent})) and $v\in\sum_j S_j\otimes\mathcal H_{X_j^c}$ such that $\|\Psi-v\|\le L^{-p}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_patch_radius_bounds,
        thm:peps_patch_minimum_growth_entropy_distance,
        thm:peps_regularized_patch_minimum,thm:peps_patch_inverse_cylinder}
    Set $R=R_0\log L$ and $b=e^{-R}$. By Lemma~\ref{thm:peps_patch_minimum_growth_entropy_distance}, $\log N(R)\le\frac\kappa2\bigl(C_0(1+u)+R/4\bigr)$ with $C_0=16C+\log8$. At a minimizer, Lemma~\ref{thm:peps_patch_inverse_cylinder} yields $S_j$ with $\dim S_j\le b^{-1}=L^{R_0}$ and $v\in\sum_j S_j\otimes\mathcal H_{X_j^c}$ satisfying $\|\Psi-v\|\le N(R)(2b)^{\kappa/2}\le L^{-p}$ by Lemma~\ref{thm:peps_patch_radius_bounds}.
\end{proof}
