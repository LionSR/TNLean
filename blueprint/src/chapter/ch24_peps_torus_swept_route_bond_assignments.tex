\begin{definition}[Successive bond assignments on the endpoint route]%
    \label{def:peps_torus_swept_route_bond_assignments}
    \lean{TNLean.PEPS.sweptPhysicalBRouteDirection,
        TNLean.PEPS.torusVacatingPlaquetteMove,
        TNLean.PEPS.torusSweptPhysicalRouteOperators}
    \leanok
    \uses{def:peps_torus_swept_physical_endpoint_route,
        def:peps_torus_directed_bond_update,def:peps_torus_swept_string_assignment}
    On a torus with periods $L_x\ge 8$ and $L_y\ge 7$, starting from the four-endpoint string assignment $u_0$ of Definition~\ref{def:peps_torus_swept_string_assignment}, define $u_n$ ($0\le n\le 12$) recursively along the $12$-step translated plaquette-base route
    \begin{align}
         & (1,1)\to(2,1)\to(2,2)\to(3,2)\to(4,2)\to(4,1)\to(4,0) \notag                         \\
         & \qquad\to(3,0)\to(2,0)\to(1,0)\to(0,0)\to(0,-1)\to(0,-2) \label{eq:swept_route_path}
    \end{align}
    by replacing the shared directed bond of the source plaquette (with bottom rightward $B$, right upward $V$, top rightward $T$, and left upward $L$) under a right, up, left, or down step by
    \begin{align}
        V\mapsto TL B^{-1},\qquad T\mapsto VB L^{-1},\qquad L\mapsto T^{-1}VB,\qquad B\mapsto V^{-1}TL, \label{eq:swept_route_updates}
    \end{align}
    while retaining all other ordered bonds (as auxiliary coefficient assignments for \cite[\S6.6, local lines 2340--2423]{Schuch2010PEPS}, without asserting a complete physical braid).
\end{definition}

\begin{lemma}[Derived vacancies and moving based fluxes]%
    \label{thm:peps_torus_swept_route_vacancies}
    \lean{TNLean.PEPS.regularWalkHolonomy_torusVacatingPlaquetteMove,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_target_vacancy,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_movingFlux}
    \leanok
    \uses{def:peps_torus_swept_route_bond_assignments,
        thm:peps_torus_directed_bond_update,thm:peps_torus_plaquette_holonomy}
    Each update in (\ref{eq:swept_route_updates}) sets the clockwise source plaquette holonomy to $1$.
    Along (\ref{eq:swept_route_path}), each step's target plaquette is vacant in $u_{n-1}$, and the moving based flux at stage $n$ is
    \begin{align}
        H_{\mathrm{mov}}(u_n) =
        \begin{cases}
            h,               & n\in\{0,1\},  \\
            ghg^{-1},        & 2\le n\le 8,  \\
            hghg^{-1}h^{-1}, & 9\le n\le 12.
        \end{cases}
        \label{eq:swept_route_moving_flux}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Substituting (\ref{eq:swept_route_updates}) into the clockwise holonomy $B^{-1}V^{-1}TL$ yields $1$.
    Evaluating $u_n$ recursively from $u_0$ along (\ref{eq:swept_route_path}) (using injectivity of the translated coordinate envelope across seams) gives $H_{\mathrm{target}}(u_{n-1})=1$ and (\ref{eq:swept_route_moving_flux}).
\end{proof}

\begin{definition}[Final endpoint and partner-comparison walks]%
    \label{def:peps_torus_swept_route_partner_walks}
    \lean{TNLean.PEPS.torusSweptPhysicalRouteFinalEndpoint,
        TNLean.PEPS.torusSweptPhysicalRouteAJointLoop,

    \leanok
    \uses{def:peps_torus_swept_route_bond_assignments,
        def:peps_torus_swept_patch_endpoint_walks}
    Relative to the translation vertex, define the final endpoint bases
    \begin{align}
        (A_2,A_1,B'_1,B_2) = \bigl((-1,1),\ (3,1),\ (0,-2),\ (1,-2)\bigr), \label{eq:final_endpoint_bases}
    \end{align}
    retaining the $A_2\to A_1$ connector of Definition~\ref{def:peps_torus_swept_patch_endpoint_walks} and connecting $B'_1\to B_2$ by the single rightward edge $(0,-2)\to(1,-2)$.
    Each joint loop traverses the partner loop conjugated along the connector $p$ followed by the base loop $\ell_0$, giving holonomy $H(\ell_0)\,H(p)^{-1}H(\ell_1)H(p)$.
\end{definition}

\begin{lemma}[Reunion holonomies of the endpoint route]%
    \label{thm:peps_torus_swept_route_reunion_holonomy}
    \lean{TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_finalEndpoints,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_finalConnectorA,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_transportedAPartner,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_AJointLoop,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_reunionConnector,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_BReunionLoop,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_BReunionLoop_conjClass,
        TNLean.PEPS.regularWalkHolonomy_torusSweptPhysicalRoute_BReunionLoop_eq_one_iff}
    \leanok
    \uses{def:peps_torus_swept_route_partner_walks,
        thm:peps_torus_swept_route_vacancies,thm:peps_regular_connector_holonomy,
        thm:peps_conjugacy_class_invariance}
    Under the final assignment $u_{12}$ of Definition~\ref{def:peps_torus_swept_route_bond_assignments}, the based endpoint fluxes at (\ref{eq:final_endpoint_bases}) are
    \begin{align}
        \bigl(H_{A_2},H_{A_1},H_{B'_1},H_{B_2}\bigr) = (g^{-1},\ g,\ hghg^{-1}h^{-1},\ h^{-1}). \label{eq:final_endpoint_fluxes}
    \end{align}
    The $A_2\to A_1$ connector has holonomy $H_A=h^{-1}g^{-1}$, so the transported $A_1$ loop has flux $H_A^{-1}gH_A=ghgh^{-1}g^{-1}$ and the $A$-joint loop has flux $g^{-1}(ghgh^{-1}g^{-1})=hgh^{-1}g^{-1}$.
    The $B'_1\to B_2$ reunion connector has holonomy $1$, and the partner-first $B$-reunion loop has flux
    \begin{align}
        H_{B,\mathrm{joint}} = h(ghg^{-1}h^{-1})h^{-1}, \label{eq:B_reunion_flux}
    \end{align}
    which lies in the conjugacy class of $ghg^{-1}h^{-1}$ and satisfies $H_{B,\mathrm{joint}}=1\iff gh=hg$ (as properties of $u_{12}$, separate from physical step implementation).
\end{lemma}

\begin{proof}\leanok
    Evaluating the four plaquette loops and both connectors on $u_{12}$ via Lemma~\ref{thm:peps_torus_swept_route_vacancies} and Lemma~\ref{thm:peps_regular_connector_holonomy} gives (\ref{eq:final_endpoint_fluxes}) and (\ref{eq:B_reunion_flux}); Lemma~\ref{thm:peps_conjugacy_class_invariance} gives the conjugacy-class and commutation equivalences.
\end{proof}
