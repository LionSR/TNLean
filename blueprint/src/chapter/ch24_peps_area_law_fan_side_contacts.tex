\section{Fan contacts along cell boundaries}
\label{sec:area_law_fan_side_contacts}

In \cite[Section~11]{OpenAI2026AreaLaw}, labels across cell sides are
assigned to elementary perimeter segments. The identities below show that
each fan triangle meets the sides of its own cell and any other closed
fine cell only along its elementary base.

% Source: 10-geometry.tex, prop:two-families, lines 299--323,
% revision adc7f1241b42e322a6451854ab7e4b4c146bf78a.
% These are boundary identities, not the full two-family proposition.

\begin{lemma}[A triangle meets its own cell sides along its base]
    \label{thm:area_law_fan_own_side_contact}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanPolygon_inter_dyadicCellSide}
    \leanok
    \uses{def:area_law_cell_fan,def:area_law_dyadic_cell_side}
    Fix a translated dyadic square, an exponent $\ell\in\mathbb N$,
    a cell index $z\in\mathbb Z^2$, and a side-midpoint subdivision.
    For any triangle $P_i$ of the resulting fan with elementary base
    $e_i=[a_i,b_i]$ and any full side $E$ of the square,
    \begin{align}
        P_i\cap E=e_i\cap E.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:area_law_cell_fan,thm:area_law_cell_fan_cover,
        thm:area_law_elementary_side_boundary_geometry}
    Let $Q$ be the half-open square and $c$ its center.
    Since $e_i\subseteq\overline Q$, $P_i$ is the convex join of $\{c\}$ and
    $e_i$, and $c\in\operatorname{int}Q$, every point of $P_i\setminus e_i$
    lies in $\operatorname{int}Q$. Any side $E$ of $\overline Q$ is disjoint
    from $\operatorname{int}Q$, so $P_i\cap E\subseteq e_i\cap E$; the reverse
    inclusion follows from $e_i\subseteq P_i$.
\end{proof}

\begin{lemma}[Contact with a distinct cell passes through the base]
    \label{thm:area_law_fan_other_cell_contact}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayer_cellFanPolygon_inter_closedCell_eq}
    \leanok
    \uses{def:area_law_cell_fan,def:area_law_fine_layer_indices}
    Fix an origin, a finite endpoint set, and a nonnegative integer
    neighborhood width. Let $z\in F_{k,\ell_k}$ and
    $w\in F_{h,\ell_h}$ be fine-cell indices with
    $(k,z)\ne(h,w)$. Choose any midpoint subdivision for $Q_{\ell_k}(z)$,
    and let $P_i$ be one of its fan triangles with base $e_i$. Then
    \begin{align}
        P_i\cap\overline{Q_{\ell_h}(w)}
        =e_i\cap\overline{Q_{\ell_h}(w)}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fine_cells_disjoint,thm:area_law_cell_fan_cover}
    The half-open fine cells $Q_{\ell_k}(z)$ and $Q_{\ell_h}(w)$ are disjoint,
    so $\operatorname{int}Q_{\ell_k}(z)$ is disjoint from
    $\overline{Q_{\ell_h}(w)}$. Every point of $P_i\setminus e_i$ lies in
    $\operatorname{int}Q_{\ell_k}(z)$ and therefore outside
    $\overline{Q_{\ell_h}(w)}$, giving
    $P_i\cap\overline{Q_{\ell_h}(w)}\subseteq e_i\cap\overline{Q_{\ell_h}(w)}$.
    The reverse inclusion follows from $e_i\subseteq P_i$.
\end{proof}
