\section{Physical contractions and Gaussian source estimates}

With the source positions, local branch labels, and Gaussian law of
Definition~\ref{def:peps_source_gaussian_law}, the estimates below concern
allowed words and isometric input frames following
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
(\texttt{04-compression.tex}, lines 5--19 and 383--558), choosing finite
orthonormal coordinates of unrestricted dimension whenever matrices or partial
traces are used.

\begin{definition}[Physical components of a word]
    \label{def:peps_physical_component}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalComponent}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalComponentCoordinates}
    \lean{TNLean.Lattice.PartyCircuit.Layout.physicalDiscardIso}
    \lean{TNLean.Lattice.PartyCircuit.Layout.physicalDiscardIso_tmul}
    \leanok
    For an evaluated word $V:H\to K$, a linear isometric equivalence
    $J:K\simeq\C^X\otimes D$, and $x\in X$, define
    \begin{align}
        C_{x,V}=(\langle x|\otimes I_D)JV:H\longrightarrow D,
        \label{eq:peps_physical_component}
    \end{align}
    with matrix $A_{x,V}$ in finite orthonormal bases of $H$ and $D$. When $K$
    consists of physical registers followed by discarded registers, the standard
    identification $J$ maps $s\otimes t\mapsto [s]_{\rm phys}\otimes t$.
\end{definition}

\begin{theorem}[Component contraction]
    \label{thm:peps_physical_component_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalComponent_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalComponentCoordinates_le_one}
    \leanok
    \uses{def:peps_physical_component}
    If $V$ is allowed, then $\|C_{x,V}\|\leq1$ and $\|A_{x,V}\|\leq1$.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_physical_component}
    By (\ref{eq:peps_physical_component}),
    $\|C_{x,V}\|\leq\|\langle x|\otimes I_D\|\,\|J\|\,\|V\|\leq1$, and
    orthonormal coordinates give $\|A_{x,V}\|=\|C_{x,V}\|\le 1$.
\end{proof}

\begin{definition}[Mixed physical Gram operator]
    \label{def:peps_physical_gram}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalGramMap}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalGramMap_inner}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalGramMatrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalGramMatrix_apply}
    \leanok
    For words $V:H\to K$ and $W:H'\to K$ with output identification $J$ and
    finite orthonormal bases $(e_i)$ of $H$ and $(e'_j)$ of $H'$, define
    $O:H\to H'$ and its matrix by
    \begin{align}
        \langle t,Os\rangle & =\langle C_{y,W}t,C_{x,V}s\rangle,      &
        O_{ji}              & =\langle C_{y,W}e'_j,C_{x,V}e_i\rangle.
        \label{eq:peps_physical_gram}
    \end{align}
\end{definition}

\begin{theorem}[Gram contraction]
    \label{thm:peps_physical_gram_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalGramMap_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalGramMatrix_le_one}
    \leanok
    \uses{def:peps_physical_gram,thm:peps_physical_component_contraction}
    For allowed $V,W$, the Gram operator and matrix satisfy $\|O\|\leq1$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_physical_component_contraction}
    By (\ref{eq:peps_physical_gram}), Cauchy--Schwarz, and
    Theorem~\ref{thm:peps_physical_component_contraction},
    $|\langle t,Os\rangle|\leq\|C_{y,W}t\|\,\|C_{x,V}s\|\leq\|t\|\,\|s\|$.
\end{proof}

\begin{definition}[Joint physical and discarded output coordinates]
    \label{def:peps_physical_output_matrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalOutputMatrix}
    \leanok
    \uses{def:peps_physical_component}
    In a finite orthonormal basis of $D$, let $K_V$ be the matrix of $JV$ in the
    product output basis on $\C^X\otimes D$ and the chosen input basis on $H$.
\end{definition}

\begin{theorem}[Partial trace and exterior contraction]
    \label{thm:peps_exterior_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.partialTrace_physicalOutputMatrix_mul_conjTranspose}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalOutputMatrix_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.rectangularTraceNorm_partialTrace_physicalOutputMatrix_le}
    \leanok
    \uses{def:peps_physical_output_matrix,def:peps_physical_gram}
    For any rectangular matrix $\rho:H'\to H$,
    \begin{align}
        \bigl[\operatorname{tr}_{D}(K_V\rho K_W^\dagger)\bigr]_{xy}
         & =\sum_{i,j}\rho_{ij}O_{ji}.
        \label{eq:peps_exterior_contraction_entries}
    \end{align}
    If $V,W$ are allowed, then $\|K_V\|,\|K_W\|\leq1$ and
    \begin{align}
        \|\operatorname{tr}_{D}(K_V\rho K_W^\dagger)\|_1\leq\|\rho\|_1.
        \label{eq:peps_exterior_contraction_norm}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Expanding $\operatorname{tr}_D(K_V\rho K_W^\dagger)$ in the orthonormal basis
    of $D$ and substituting (\ref{eq:peps_physical_gram}) yields
    (\ref{eq:peps_exterior_contraction_entries}); since $\|K_V\|=\|JV\|\le 1$
    and $\|K_W\|=\|JW\|\le 1$, trace-norm contractivity of the partial trace
    gives (\ref{eq:peps_exterior_contraction_norm}).
\end{proof}

\begin{definition}[Proper input frames]
    \label{def:peps_physical_frames}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalComponentFrameMatrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalComponentFrameMatrix_apply}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalFrameGramMatrix}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalFrameGramMatrix_apply}
    \lean{TNLean.Lattice.PartyCircuit.Word.physicalOutputFrameMatrix}
    \leanok
    \uses{def:peps_physical_component,def:peps_physical_gram}
    For linear isometries $F:\C^n\to H$ and $F':\C^{n'}\to H'$, define
    \begin{align}
        A_{x,V,F} & =[C_{x,V}F],                               &
        O_{ji}    & =\langle C_{y,W}F'e'_j,C_{x,V}Fe_i\rangle, &
        K_{V,F}   & =[JVF].
        \label{eq:peps_physical_frames}
    \end{align}
\end{definition}

\begin{theorem}[Contraction in proper frames]
    \label{thm:peps_physical_frame_contraction}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalComponentFrameMatrix_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalFrameGramMatrix_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.norm_physicalOutputFrameMatrix_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.rectangularTraceNorm_partialTrace_physicalOutputFrameMatrix_le}
    \leanok
    \uses{def:peps_physical_frames,thm:peps_physical_component_contraction}
    For allowed $V,W$, the matrices in (\ref{eq:peps_physical_frames}) satisfy
    $\|A_{x,V,F}\|,\|O\|,\|K_{V,F}\|\le 1$, and for any rectangular $Z$,
    \begin{align}
        \|\operatorname{tr}_{D}(K_{V,F}ZK_{W,F'}^\dagger)\|_1\leq\|Z\|_1.
        \label{eq:peps_physical_frame_contraction}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Composing $V,W$ with the isometries $F,F'$ reduces all bounds to
    Theorems~\ref{thm:peps_physical_component_contraction},
    \ref{thm:peps_physical_gram_contraction}, and~\ref{thm:peps_exterior_contraction}.
\end{proof}

\begin{definition}[Schmidt-weighted physical vectors]
    \label{def:peps_schmidt_physical_vector}
    \lean{TNLean.Lattice.PartyCircuit.Word.schmidtPhysicalVector}
    \lean{TNLean.Lattice.PartyCircuit.Word.schmidtPhysicalVector_repr}
    \leanok
    \uses{def:peps_physical_frames}
    For an isometry $F:\C^{I\times T}\to H$, weights $\tau_t\in\R$, and physical
    coordinate $x\in X$, define
    \begin{align}
        \psi_i=\sum_{t\in T}\sqrt{\tau_t}\,|t\rangle\otimes C_{x,V}F|i,t\rangle,
        \label{eq:peps_schmidt_physical_vector}
    \end{align}
    with $(t,d)$ coordinate $\sqrt{\tau_t}(A_{x,V,F})_{d,(i,t)}$, and define
    $\phi_l$ analogously for a bra word $W$, indices $L,U$, frame $F'$, and
    weights $\tau'_u$.
\end{definition}

\begin{theorem}[Weighted source matrix from Schmidt-weighted vectors]
    \label{thm:peps_weighted_physical_source_identity}
    \lean{TNLean.Lattice.PartyCircuit.Word.sum_schmidtPhysicalVector_outer_eq_weightedSourceError}
    \leanok
    \uses{def:peps_schmidt_physical_vector}
    For any $z_{il}\in\C$, tracing out $D$ in
    $\sum_{i,l}z_{il}|\psi_i\rangle\langle\phi_l|$ yields the matrix
    $W:T\leftarrow U$ with entries
    \begin{align}
        W_{tu}=\sqrt{\tau_t}\sqrt{\tau'_u}\sum_{i,l}z_{il}O_{(l,u),(i,t)}
        \label{eq:peps_weighted_physical_source_identity}
    \end{align}
    (equal to $\sqrt{\tau_t\tau'_u}\sum_{i,l}z_{il}O_{(l,u),(i,t)}$ when
    $\tau_t,\tau'_u\ge 0$).
\end{theorem}

\begin{proof}\leanok
    Expand $\psi_i,\phi_l$ by (\ref{eq:peps_schmidt_physical_vector}) and sum
    over the discard coordinate $d\in D$ using (\ref{eq:peps_physical_frames}).
\end{proof}

\begin{definition}[Separated output and its physical operator]
    \label{def:peps_separated_physical_density}
    \lean{TNLean.Lattice.PartyCircuit.Word.separatedSchmidtOutput}
    \lean{TNLean.Lattice.PartyCircuit.Word.separatedSchmidtOutput_repr}
    \lean{TNLean.Lattice.PartyCircuit.Word.separatedPhysicalCoordinates}
    \lean{TNLean.Lattice.PartyCircuit.Word.separatedPhysicalDensity}
    \leanok
    \uses{def:peps_schmidt_physical_vector}
    For an exterior word $V':H_E\to K_E$, output identification
    $J':K_E\simeq\C^Y\otimes D'$, and isometry $G:\C^T\to H_E$, define
    \begin{align}
        \Psi_i & =\sum_t\sqrt{\tau_t}\,(JVF|i,t\rangle)\otimes(J'V'G|t\rangle), &
        M(z)   & =\operatorname{tr}_{D\otimes D'}
        \sum_{i,l}z_{il}|\Psi_i\rangle\langle\Phi_l|,
        \label{eq:peps_separated_physical_density}
    \end{align}
    where $\Phi_l$ is defined analogously from a bra word $W'$, frame $G'$, and
    weights $\tau'_u$, and physical coordinates $X\times Y$ precede discard
    coordinates $D\times D'$.
\end{definition}

\begin{theorem}[Tracing both discarded factors]
    \label{thm:peps_separated_physical_density_blocks}
    \lean{TNLean.Lattice.PartyCircuit.Word.partialTrace_separatedSchmidtOutput_eq_weightedSourceError}
    \lean{TNLean.Lattice.PartyCircuit.Word.separatedPhysicalDensity_submatrix}
    \leanok
    \uses{def:peps_separated_physical_density,thm:peps_weighted_physical_source_identity}
    For each $(x,y)\in X\times Y$, the $Y\times Y$ block $M(z)_{xy}$ of $M(z)$
    satisfies
    \begin{align}
        M(z)_{xy}=\operatorname{tr}_{D'}(K_{V',G}W_{xy}(z)K_{W',G'}^\dagger).
        \label{eq:peps_separated_physical_density_block}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Tracing (\ref{eq:peps_separated_physical_density}) over $D$ at $(x,y)$ gives
    $W_{xy}(z)$ by Theorem~\ref{thm:peps_weighted_physical_source_identity}, and
    tracing over $D'$ yields (\ref{eq:peps_separated_physical_density_block}).
\end{proof}

\begin{theorem}[Expected exterior trace norm under diagonal covariance]
    \label{thm:peps_weighted_physical_source_bound}
    \lean{TNLean.Lattice.PartyCircuit.Word.integral_rectangularTraceNorm_weightedPhysicalSource_le}
    \leanok
    \uses{thm:peps_physical_frame_contraction,def:peps_separated_physical_density}
    If $V,W,V',W'$ are allowed, $p_i,p'_l,\tau_t,\tau'_u\ge 0$ sum to $1$,
    $\kappa\geq0$, and $z_{il}\overline{z_{i'l'}}$ is integrable with
    \begin{align}
        \mathbb E[z_{il}\overline{z_{i'l'}}]
        =\mathbf1_{i=i'}\mathbf1_{l=l'}\,
        \kappa\sqrt{p_i p'_l},
        \label{eq:peps_diagonal_covariance}
    \end{align}
    then for every $(x,y)\in X\times Y$,
    \begin{align}
        \mathbb E\|M(z)_{xy}\|_1\leq\sqrt\kappa.
        \label{eq:peps_weighted_physical_source_bound}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    By (\ref{eq:peps_separated_physical_density_block}) and
    Theorem~\ref{thm:peps_physical_frame_contraction},
    $\|M(z)_{xy}\|_1\le\|W_{xy}(z)\|_1$ with $\|O\|\le 1$; taking expectations
    under (\ref{eq:peps_diagonal_covariance}) bounds
    $\mathbb E\|W_{xy}(z)\|_1\le\sqrt\kappa$.
\end{proof}

\begin{theorem}[Gaussian estimates for separated words]
    \label{thm:peps_gaussian_separated_physical_bound}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integral_rectangularTraceNorm_gaussianPhysicalSource_le}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integrable_gaussianSeparatedPhysicalDensity_and_integral_le}
    \leanok
    \uses{def:peps_original_schmidt_coordinates,thm:peps_actual_gaussian_covariance,thm:peps_weighted_physical_source_bound}
    For the Gaussian coefficient product $Z_{u,v}$ on $S\subseteq\mathcal E$ at
    fixed local ket and bra labels, if $k>0$, the local Schmidt and crossing
    weights are probability weights, the input frames are isometries, and all
    four separated words are allowed, then $\|M(Z)\|_1$ is integrable and
    \begin{align}
        \mathbb E_k\|M(Z)_{xy}\|_1 & \leq k^{-|S|/2},       &
        \mathbb E_k\|M(Z)\|_1      & \leq |X|^2 k^{-|S|/2}.
        \label{eq:peps_gaussian_separated_physical_bound}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_actual_gaussian_covariance}, $Z_{u,v}$ satisfies
    (\ref{eq:peps_diagonal_covariance}) with $\kappa=k^{-|S|}$, so
    Theorem~\ref{thm:peps_weighted_physical_source_bound} gives the first bound
    in (\ref{eq:peps_gaussian_separated_physical_bound}), and summing
    $\|M(Z)\|_1\le\sum_{x,y\in X}\|M(Z)_{xy}\|_1$ over $|X|^2$ blocks gives the
    second.
\end{proof}

\begin{theorem}[Prepared matrices on mixed pure inputs]
    \label{thm:peps_prepared_vector_density}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_mulVec_repr}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_mul_vecMulVec}
    \lean{TNLean.Lattice.PartyCircuit.Word.preparedMatrix_mul_vecMulVec_preparedInput}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceBasisMatrix_mul_vecMulVec}
    \leanok
    For prepared words with linear maps $T,T'$ and matrices $K_T,K_{T'}$ in
    finite orthonormal bases, and any input vectors $\psi,\phi$,
    \begin{align}
        K_T[\psi] & =[T\psi],                      &
        K_T|\psi\rangle\langle\phi|K_{T'}^\dagger
                  & =|T\psi\rangle\langle T'\phi|.
        \label{eq:peps_prepared_vector_density}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    From $K_T[\psi]=[T\psi]$ and $K_{T'}[\phi]=[T'\phi]$, we have
    $K_T|\psi\rangle\langle\phi|K_{T'}^\dagger=(K_T|\psi\rangle)(K_{T'}|\phi\rangle)^\dagger
        =|T\psi\rangle\langle T'\phi|$.
\end{proof}

\begin{theorem}[Independence of physical output coordinates]
    \label{thm:peps_physical_corrected_basis_invariance}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.rectangularTraceNorm_physicalCorrectedSourceTerm_basis_eq}
    \leanok
    \uses{thm:peps_prepared_vector_density}
    For any mixed pure input $|\psi\rangle\langle\phi|$, the trace norm
    $\|\operatorname{tr}_{\mathrm{disc}}F_S(E;|\psi\rangle\langle\phi|)\|_1$ is
    invariant under changes of the orthonormal physical and discard output bases.
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_prepared_vector_density}, $F_S(E;|\psi\rangle\langle\phi|)$
    is a linear combination of outer products $|T_u\psi\rangle\langle T_v\phi|$,
    and both $\operatorname{tr}_{\mathrm{disc}}$ and $\|\cdot\|_1$ are unitarily
    invariant on orthonormal output coordinates.
\end{proof}

\begin{theorem}[Corrected operator on mixed pure inputs]
    \label{thm:peps_gaussian_vector_density}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedSourceTerm_gaussian_vecMulVec_eq_sum}
    \leanok
    \uses{def:peps_partial_schmidt_output,thm:peps_corrected_gaussian_sum,thm:peps_prepared_vector_density}
    If the affected set $\mathcal A$ meets every source in $S$ and the weights
    and frames represent every original source vector, then for any $\psi,\phi$,
    \begin{align}
        F_S(\Delta(\omega))(|\psi\rangle\langle\phi|)
        =\sum_{\xi,\zeta}c_{\mathcal A,\xi}\overline{c_{\mathcal A,\zeta}}
        \sum_{u,v}Z^{\xi,\zeta}_{u,v}(\omega)
        |\Psi_{\xi,u}(\psi)\rangle\langle\Psi_{\zeta,v}(\phi)|.
        \label{eq:peps_gaussian_vector_density}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Apply Theorem~\ref{thm:peps_prepared_vector_density} termwise to the matrix
    expansion of Theorem~\ref{thm:peps_corrected_gaussian_sum}.
\end{proof}

\begin{theorem}[Square integrability of the coefficient products]
    \label{thm:peps_gaussian_products_integrable}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceCorrection_product_memLp_two}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialSourceCorrection_product_memLp_two}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.integrable_rectangularTraceNorm_sourceCorrectionSum}
    \leanok
    \uses{thm:peps_actual_gaussian_covariance}
    For $k>0$ and nonnegative local weights, every Gaussian coefficient product
    $Z_{u,v}$ (and $Z^{\xi,\zeta}_{u,v}$ when $\mathcal A$ meets $S$) lies in
    $L^2$, so the trace norm of any finite linear combination of these products
    with fixed matrix coefficients is integrable.
\end{theorem}

\begin{proof}\leanok
    Theorem~\ref{thm:peps_actual_gaussian_covariance} gives finite second moments
    for $Z_{u,v}$ and $Z^{\xi,\zeta}_{u,v}$ on the probability space, which
    implies $L^1$ integrability of the trace norm of any finite matrix linear
    combination.
\end{proof}

\begin{theorem}[Integrability of corrected physical operators]

\begin{theorem}[A sample bounded by the average error sum]
    \label{thm:peps_sampled_error_integral_bound}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_sampledSourceMatrix_error_le_sum_integral}
    \leanok
    \uses{thm:peps_actual_physical_integrability,thm:peps_sampled_physical_error}
    For $k>0$, nonnegative representing source weights and frames, finite
    orthonormal input and physical/discard output bases, and any input matrix,
    there exists a sample $\omega$ such that
    \begin{align}
        \|\widehat\rho_{\rm phys}(\omega)-\rho_{\rm phys}\|_1
        \leq\sum_{\varnothing\ne S\subseteq\mathcal E}
        \mathbb E_k\|\operatorname{tr}_{\rm disc}F_S(\Delta)\|_1.
        \label{eq:peps_sampled_error_integral_bound}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_actual_physical_integrability}, the finite sum of
    nonnegative random variables $\sum_{\varnothing\ne S\subseteq\mathcal E}\|\operatorname{tr}_{\rm disc}F_S(\Delta(\omega))\|_1$
    is integrable and hence $\le$ its expectation at some $\omega$;
    Theorem~\ref{thm:peps_sampled_physical_error} then gives (\ref{eq:peps_sampled_error_integral_bound}).
\end{proof}

\begin{definition}[Affected parties and retained free sources]
    \label{def:peps_corrected_free_sources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedMask}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedFreeSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.correctedFreeSlots_eq_true_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_correctedFreeSlot}
    \leanok
    For $S\subseteq\mathcal E$, let $\mathcal A_S$ be the set of all endpoints
    of sources in $S$. A retained source slot is free if and only if its
    original position lies in $S$ or its two endpoints lie on opposite sides of
    $\mathcal A_S$; all other retained sources are prepared with their original
    vectors.
\end{definition}

\begin{theorem}[Lifetime and coefficient costs]
    \label{thm:peps_actual_source_lifetime_cost}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isTouched_correctedMask_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.corrected_slot_isTouched}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.card_touched_correctedMask_le}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient_physical_cost_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient_physical_cost_le}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_mul_conj_le_lifetime}
    \leanok
    \uses{def:peps_corrected_free_sources}
    A gate occurrence is touched if it has a participant in $\mathcal A_S$ (so
    every $e\in S$ belongs to a touched gate). If every party in $\mathcal A_S$
    participates in at most $b$ gate occurrences, at most $2b|S|$ occurrences
    are touched. If each touched gate has coefficient sum at most $B\geq1$ and
    $d_p\leq d$ ($d\ge 1$) on $\mathcal A_S$, then
    \begin{align}
        \sum_{\xi,\zeta}|c_{\mathcal A_S,\xi}\overline{c_{\mathcal A_S,\zeta}}|
         & \leq B^{4b|S|},         &
        \left(\sum_{\xi,\zeta}|c_{\mathcal A_S,\xi}
        \overline{c_{\mathcal A_S,\zeta}}|\right)
        \prod_{p\in\mathcal A_S}d_p^2
         & \leq (B^{4b}d^4)^{|S|}.
        \label{eq:peps_actual_source_lifetime_cost}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Since $|\mathcal A_S|\le 2|S|$ and each $p\in\mathcal A_S$ participates in
    at most $b$ gates, at most $2b|S|$ gates are touched; factoring the ket and
    bra coefficient sums over touched gates gives $\le (B^{2b|S|})^2=B^{4b|S|}$,
    and multiplying by $\prod_{p\in\mathcal A_S}d_p^2\le d^{2|\mathcal A_S|}\le d^{4|S|}$
    yields (\ref{eq:peps_actual_source_lifetime_cost}).
\end{proof}

\begin{definition}[Two contraction factors]
    \label{def:peps_actual_tensor_partition}
    \lean{TNLean.Lattice.PartyCircuit.Word.IsTensorPartitioned}
    \leanok
    A word $W$ admits a tensor partition for a Boolean owner map if there exist
    allowed source-free words $C,D$ with $\|C\|,\|D\|\leq1$ on the restricted
    layouts such that under the owner-partition isometries $P_{\rm in},P_{\rm out}$,
    \begin{align}
        P_{\rm out}W=(C\otimes D)P_{\rm in}.
        \label{eq:peps_actual_tensor_partition}
    \end{align}
\end{definition}

\begin{theorem}[Selective preparation gives two-side factors]
    \label{thm:peps_corrected_source_partition}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.exists_corrected_source_partition}
    \leanok
    \uses{def:peps_corrected_free_sources,def:peps_actual_tensor_partition}
    For any allowed chronological composition, any $S\subseteq\mathcal E$, and
    any partial branch for $\mathcal A_S$, the word obtained by preparing the
    nonfree sources of Definition~\ref{def:peps_corrected_free_sources} followed
    by the partial residual admits a tensor partition
    (\ref{eq:peps_actual_tensor_partition}) with free input spaces independent of
    the branch.
\end{theorem}

\begin{proof}\leanok
    Every nonfree source is normalized with both endpoints on the same side of
    $\mathcal A_S$, and the partial residual is allowed and source-free, so each
    operation factors across the $\mathcal A_S$ partition into allowed
    source-free words $C$ and $D$.
\end{proof}

\begin{definition}[An explicit positive sample count]
    \label{def:peps_source_sample_count}
    \lean{TNLean.Lattice.Compression.sourceSamplingCount}
    \lean{TNLean.Lattice.Compression.sourceSamplingCount_pos}
    \lean{TNLean.Lattice.Compression.sourceSamplingCount_lt}
    \leanok
    For $N\in\N$, $Q\in\R$, and $\epsilon>0$, define
    \begin{align}
        k(N,Q,\epsilon)=
        \left\lceil\left(\frac{8\max(1,N)Q}{\epsilon}\right)^2\right\rceil+1,
        \label{eq:peps_source_sample_count}
    \end{align}
    so that $0<k(N,Q,\epsilon)<(8\max(1,N)Q/\epsilon)^2+2$.
\end{definition}

\begin{theorem}[Finite subset summation and the sample-count estimate]
    \label{thm:peps_source_sample_count_bound}
    \lean{TNLean.Lattice.Compression.sum_nonemptyFinsets_pow_card}
    \lean{TNLean.Lattice.Compression.sum_sourceErrorWeights}
    \lean{TNLean.Lattice.Compression.div_sqrt_sourceSamplingCount_le}
    \lean{TNLean.Lattice.Compression.sum_sourceErrorWeights_sourceSamplingCount_le}
    \leanok
    \uses{def:peps_source_sample_count}
    For a finite set $\mathcal E$ with $|\mathcal E|=N$ and $k>0$,
    \begin{align}
        \sum_{\varnothing\ne S\subseteq\mathcal E}q^{|S|}
         & =(1+q)^N-1,                            &
        \sum_{\varnothing\ne S\subseteq\mathcal E}Q^{|S|}k^{-|S|/2}
         & =\left(1+\frac{Q}{\sqrt k}\right)^N-1.
        \label{eq:peps_source_subset_sums}
    \end{align}
    If $Q\geq0$ and $\epsilon>0$, then $k=k(N,Q,\epsilon)$ from
    (\ref{eq:peps_source_sample_count}) satisfies
    \begin{align}
        \frac{Q}{\sqrt{k(N,Q,\epsilon)}}
         & \leq\frac{\epsilon}{8\max(1,N)},
        \label{eq:peps_source_sample_ratio}
    \end{align}
    and if $0<\epsilon\leq1$, then
    $\sum_{\varnothing\ne S\subseteq\mathcal E}Q^{|S|}k^{-|S|/2}\le\epsilon/4$.
\end{theorem}

\begin{proof}\leanok
    Expanding $\prod_{e\in\mathcal E}(1+q)=1+\sum_{\varnothing\ne S\subseteq\mathcal E}q^{|S|}$
    and setting $q=Q/\sqrt k$ gives (\ref{eq:peps_source_subset_sums}). By
    (\ref{eq:peps_source_sample_count}), $\sqrt k\ge 8\max(1,N)Q/\epsilon$, giving
    (\ref{eq:peps_source_sample_ratio}); for $0<\epsilon\le 1$,
    $Nq\le\epsilon/8\le 1/8$ implies $(1+q)^N-1\le e^{Nq}-1\le\epsilon/4$.
\end{proof}
