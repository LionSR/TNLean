\section{Physical-site oscillation with a spectator}
\label{sec:qc_site_oscillation}

Let $I$ and $J$ be finite sets, let $q\in\N$, and put
$\mathcal H=(\C^q)^{\otimes I}$ and $\mathcal K=\C^J$.
The spectator space $\mathcal K$ has arbitrary finite dimension, including
zero. Write $\mathcal A_X$ for the physical operators acting on
$X\subseteq I$, as in Definition~\ref{def:qc_local_circuit}.
Operators $B,C$ on $\mathcal H\otimes\mathcal K$ are arbitrary complex
matrices. Their norm is the operator norm induced by the Euclidean norm:
$\|B\|=\sup_{\|\psi\|_2\le1}\|B\psi\|_2$.
For physical operators $U$, write $\widehat U=U\otimes\mathbb 1_{\mathcal K}$,
and use $[A,B]=AB-BA$.

\subsection{The on-site unitary supremum}

\begin{definition}[Oscillation at a physical site]
    \label{def:qc_site_oscillation}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.siteOscillation}
    \leanok
    Following \cite{OpenAI2026AreaLaw}, for $y\in I$, define the on-site
    unitary supremum
    \begin{align}
        d_y(B) & =\sup\bigl\{\|[\widehat U,B]\|:
        U\in\mathcal A_{\{y\}},\ U^\dagger U=UU^\dagger=\mathbb 1\bigr\},
        \label{eq:qc_osc_supremum}
    \end{align}
    where every test unitary acts as the identity on the spectator.
\end{definition}

\begin{lemma}[Bounds for the oscillation supremum]
    \label{thm:qc_site_oscillation_bounds}
    \uses{def:qc_site_oscillation}
    \lean{QuantumCircuit.siteOscillation_bddAbove,
        QuantumCircuit.siteOscillation_nonneg,
        QuantumCircuit.norm_commutator_le_siteOscillation,
        QuantumCircuit.norm_commutator_le_siteOscillation_right,
        QuantumCircuit.siteOscillation_le,
        QuantumCircuit.siteOscillation_le_two_mul_norm}
    \leanok
    The set in~(\ref{eq:qc_osc_supremum}) contains zero and is bounded
    above by $2\|B\|$. In particular, its supremum is finite and
    $0\le d_y(B)\le2\|B\|$. Every unitary $U\in\mathcal A_{\{y\}}$
    satisfies $\|[\widehat U,B]\|=\|[B,\widehat U]\|\le d_y(B)$.
    If these commutator norms are all at most a real number $r$, then
    $d_y(B)\le r$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_site_oscillation}
    The identity is an admissible unitary with zero commutator, and for each
    admissible $U$ the operator $\widehat U$ is unitary, so
    \begin{align}
        \|[\widehat U,B]\|
         & \le\|\widehat UB\|+\|B\widehat U\|=2\|B\|.
        \label{eq:qc_osc_commutator_bound}
    \end{align}
    The upper- and lower-bound properties of the supremum then yield the
    stated inequalities, since reversing a commutator changes only its sign.
\end{proof}

\begin{lemma}[Subadditivity and absolute homogeneity]
    \label{thm:qc_site_oscillation_seminorm}
    \uses{def:qc_site_oscillation}
    \lean{QuantumCircuit.siteOscillation_add_le,
        QuantumCircuit.siteOscillation_smul,
        QuantumCircuit.siteOscillation_neg,
        QuantumCircuit.siteOscillation_sub_le,
        QuantumCircuit.siteOscillation_sum_le}
    \leanok
    For every $c\in\C$,
    \begin{align}
        d_y(B+C) & \le d_y(B)+d_y(C),
        \label{eq:qc_osc_subadditivity}
    \end{align}
    \begin{align}
        d_y(cB) & =|c|d_y(B).
        \label{eq:qc_osc_homogeneity}
    \end{align}
    Consequently $d_y(-B)=d_y(B)$ and
    $d_y(B-C)\le d_y(B)+d_y(C)$.
    For any finite set $F$ and any family $(B_k)_{k\in F}$,
    $d_y(\sum_{k\in F}B_k)\le\sum_{k\in F}d_y(B_k)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:qc_site_oscillation_bounds,thm:qc_site_oscillation_commutant}
    The identities
    $[\widehat U,B+C]=[\widehat U,B]+[\widehat U,C]$ and
    $[\widehat U,cB]=c[\widehat U,B]$, combined with the norm inequalities
    and the supremum bound, give~(\ref{eq:qc_osc_subadditivity}) and
    $d_y(cB)\le|c|d_y(B)$.
    When $c\ne0$, applying this scalar bound to $c^{-1}$ and $cB$ yields
    the reverse inequality in~(\ref{eq:qc_osc_homogeneity}), whereas $c=0$
    reduces to $d_y(0)=0$.
    Setting $c=-1$ gives invariance under negation, which together with
    $B-C=B+(-C)$ yields the subtraction bound.
    For a finite sum, distributing the commutator gives
    \begin{align}
        \Bigl\|\Bigl[\widehat U,\sum_{k\in F}B_k\Bigr]\Bigr\|
         & \le\sum_{k\in F}\|[\widehat U,B_k]\|
        \le\sum_{k\in F}d_y(B_k),
    \end{align}
    and taking the supremum over $U$ completes the bound.
\end{proof}

\begin{lemma}[Vanishing on the physical commutant]
    \label{thm:qc_site_oscillation_commutant}
    \uses{def:qc_site_oscillation}
    \lean{QuantumCircuit.siteOscillation_eq_zero_of_commute,
        QuantumCircuit.siteOscillation_zero,
        QuantumCircuit.siteOscillation_one_kronecker}
    \leanok
    If $[\widehat U,B]=0$ for every $U\in\mathcal A_{\{y\}}$, then
    $d_y(B)=0$. In particular, $d_y(0)=0$, and every operator $S$ on
    the spectator satisfies $d_y(\mathbb 1_{\mathcal H}\otimes S)=0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:qc_site_oscillation_bounds}
    The commutation hypothesis bounds every norm in the supremum by
    zero, whereas $d_y(B)\ge0$. For a spectator operator,
    $(U\otimes\mathbb 1)(\mathbb 1\otimes S)=U\otimes S
        =(\mathbb 1\otimes S)(U\otimes\mathbb 1)$.
\end{proof}

\subsection{Averaging physical sites}
\label{sec:qc_spectator_expectation}

For $K\subseteq I$, let $E_K$ be the normalized partial-trace expectation
of Definition~\ref{def:qc_site_expectation}. The set $K$ consists of the
retained physical sites. Thus averaging over a set $Q\subseteq I$
means using $K=I\setminus Q$.

\begin{definition}[Site expectation with a spectator]
    \label{def:qc_spectator_expectation}
    \uses{def:qc_site_expectation}
    \lean{QuantumCircuit.spectatorSiteExpectation}
    \leanok
    For $a,b\in J$, let $B_{ab}$ be the physical matrix whose entries are
    $(B_{ab})_{\sigma,\tau}=B_{(\sigma,a),(\tau,b)}$.
    Define $E_K^{\mathcal K}$ by applying $E_K$ to each such block:
    \begin{align}
        \bigl(E_K^{\mathcal K}(B)\bigr)_{(\sigma,a),(\tau,b)}
         & =\bigl(E_K(B_{ab})\bigr)_{\sigma,\tau}.
        \label{eq:qc_spectator_blocks}
    \end{align}
    In particular, when $q>0$, set $Q=I\setminus K$ and write
    $u\sqcup\rho$ for the configuration equal to $u$ on $K$ and to
    $\rho$ on $Q$. Then
    \begin{align}
        \bigl(E_K^{\mathcal K}(B)\bigr)_{(\sigma,a),(\tau,b)}
         & =q^{-|Q|}\mathbf1_{\{\sigma|_Q=\tau|_Q\}}
        \sum_{\rho\in\{0,\ldots,q-1\}^{Q}}
        B_{(\sigma|_K\sqcup\rho,a),(\tau|_K\sqcup\rho,b)},
        \label{eq:qc_spectator_trace}
    \end{align}
    with both spectator indices left unchanged.
\end{definition}

\begin{lemma}[Preservation of spectator factors]
    \label{thm:qc_spectator_factors}
    \uses{def:qc_spectator_expectation}
    \lean{QuantumCircuit.spectatorSiteExpectation_kronecker,
        QuantumCircuit.spectatorSiteExpectation_one_kronecker,
        QuantumCircuit.spectatorSiteExpectation_univ}
    \leanok
    Suppose $q>0$. For physical operators $A$ and spectator operators $S$,
    \begin{align}
        E_K^{\mathcal K}(A\otimes S) & =E_K(A)\otimes S,
        \label{eq:qc_spectator_tensor}
    \end{align}
    \begin{align}
        E_K^{\mathcal K}(\mathbb 1\otimes S) & =\mathbb 1\otimes S.
        \label{eq:qc_spectator_fixed}
    \end{align}
    Retaining all physical sites gives $E_I^{\mathcal K}(B)=B$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_spectator_expectation,def:qc_site_expectation,
        lem:qc_site_expectation_properties}
    The $(a,b)$ block of $A\otimes S$ is $S_{ab}A$, so linearity of $E_K$
    gives $E_K(S_{ab}A)=S_{ab}E_K(A)$, proving~(\ref{eq:qc_spectator_tensor}).
    Since $E_K$ fixes the physical identity, this identity
    specializes to~(\ref{eq:qc_spectator_fixed}), and because every physical
    matrix is supported on $I$, $E_I(B_{ab})=B_{ab}$ on every block.
\end{proof}

\begin{lemma}[Commutation on the averaged sites]
    \label{thm:qc_spectator_commutant}
    \uses{def:qc_spectator_expectation}
    \lean{QuantumCircuit.spectatorSiteExpectation_commute}
    \leanok
    For every $U\in\mathcal A_{I\setminus K}$,
    $[E_K^{\mathcal K}(B),\widehat U]=0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_spectator_expectation,def:qc_site_expectation}
    Each block $E_K(B_{ab})$ is supported on $K$ by its partial-trace
    formula. Since operators on disjoint sets of physical sites commute,
    $E_K(B_{ab})U=UE_K(B_{ab})$, and matrix multiplication by $\widehat U$
    acts separately on each spectator block to give the commutation identity.
\end{proof}

\begin{definition}[Weyl product on a set of sites]
    \label{def:qc_partial_weyl}
    \uses{lem:qc_site_expectation_average}
    \lean{QuantumCircuit.partialWeyl}
    \leanok
    Suppose $q>0$, let $S\subseteq I$, and let
    $p\in(\Z/q\Z\times\Z/q\Z)^I$. For the on-site Weyl operators
    $W(p_x)$, put $M_x=W(p_x)$ when $x\in S$ and
    $M_x=\mathbb 1$ otherwise. Define
    \begin{align}
        P_S(p) & =\bigotimes_{x\in I}M_x.
        \label{eq:qc_partial_weyl}
    \end{align}
    In particular, $P_{\varnothing}(p)=\mathbb 1$.
\end{definition}

\begin{lemma}[Unitarity and site insertion for Weyl products]
    \label{thm:qc_partial_weyl_identities}
    \uses{def:qc_partial_weyl,def:qc_local_circuit}
    \lean{QuantumCircuit.partialWeyl_mem_unitary,
        QuantumCircuit.outsideWeyl_eq_partialWeyl,
        QuantumCircuit.partialWeyl_insert,
        QuantumCircuit.partialWeyl_singleton_mem}
    \leanok
    Suppose $q>0$. Every $P_S(p)$ is unitary. The Weyl product
    $W_K(p)$ with factors $W(p_x)$ off $K$ and the identity on $K$ satisfies
    $W_K(p)=P_{I\setminus K}(p)$, and
    $P_{\{y\}}(p)\in\mathcal A_{\{y\}}$ for every $y\in I$.
    If $y\notin S$, then
    \begin{align}
        P_{S\cup\{y\}}(p) & =P_{\{y\}}(p)P_S(p).
        \label{eq:qc_partial_weyl_insert}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_partial_weyl,def:qc_local_circuit}
    Since each tensor factor $M_x$ is unitary, taking adjoints and multiplying
    factor by factor gives $P_S(p)^\dagger P_S(p)=\mathbb 1$ and
    $P_S(p)P_S(p)^\dagger=\mathbb 1$.
    Both $W_K(p)$ and $P_{I\setminus K}(p)$ have factor $\mathbb 1$ on $K$ and
    $W(p_x)$ off $K$, yielding the complementary-set identity, while the
    singleton product $P_{\{y\}}(p)$ has identity factors off $\{y\}$ and
    therefore belongs to $\mathcal A_{\{y\}}$. In~(\ref{eq:qc_partial_weyl_insert}),
    the right-hand side has factor $W(p_y)\mathbb 1$ at $y$ and
    $\mathbb 1\,M_x$ at every other site, matching the tensor factors of
    $P_{S\cup\{y\}}(p)$.
\end{proof}

\begin{lemma}[Commutator with a product of unitaries]
    \label{thm:qc_unitary_product_commutator}
    \lean{QuantumCircuit.norm_commutator_mul_le}
    \leanok
    Let $R$ be a normed star ring satisfying the $C^*$-identity. For
    $B\in R$ and unitaries $U,V\in R$,
    \begin{align}
        \|[B,UV]\| & \le\|[B,U]\|+\|[B,V]\|.
        \label{eq:qc_unitary_product_commutator}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Expanding $[B,UV]=[B,U]V+U[B,V]$ and applying the triangle inequality
    together with unitary invariance of the norm gives
    $\|[B,UV]\|\le\|[B,U]V\|+\|U[B,V]\|
        =\|[B,U]\|+\|[B,V]\|$.
\end{proof}

\begin{lemma}[Amplified Weyl averaging formula]
    \label{thm:qc_spectator_average}
    \uses{def:qc_spectator_expectation,lem:qc_site_expectation_average}
    \lean{QuantumCircuit.spectatorSiteExpectation_eq_average}
    \leanok
    Suppose $q>0$. For $p\in(\Z/q\Z\times\Z/q\Z)^I$, let
    $W_K(p)$ be the product of the on-site Weyl operators $W(p_x)$ at
    $x\notin K$, with the identity at $x\in K$.
    Then
    \begin{align}
        E_K^{\mathcal K}(B)
         & =q^{-2|I|}\sum_{p\in(\Z/q\Z\times\Z/q\Z)^I}
        \widehat{W_K(p)}B\widehat{W_K(p)}^{\dagger}.
        \label{eq:qc_spectator_average}
    \end{align}
    Labels at retained sites are included in the sum; each choice of
    Weyl operators outside $K$ consequently occurs $q^{2|K|}$ times.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:qc_spectator_expectation,lem:qc_site_expectation_average}
    For every physical $U$, the $(a,b)$ block of
    $\widehat UB\widehat U^\dagger$ is $UB_{ab}U^\dagger$, so applying the
    physical Weyl averaging formula to each block $B_{ab}$ in
    (\ref{eq:qc_spectator_blocks}) yields the identity.
\end{proof}

\begin{lemma}[Contraction with an arbitrary spectator]
    \label{thm:qc_spectator_contraction}
    \uses{def:qc_spectator_expectation}
    \lean{QuantumCircuit.norm_spectatorSiteExpectation_le}
    \leanok
    If $q>0$, then $\|E_K^{\mathcal K}(B)\|\le\|B\|$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:qc_spectator_average,thm:qc_partial_weyl_identities}
    By Lemma~\ref{thm:qc_partial_weyl_identities}, every
    $\widehat{W_K(p)}$ is unitary, so each of the $q^{2|I|}$ summands in
    (\ref{eq:qc_spectator_average}) has norm $\|B\|$.
    Applying the triangle inequality with the normalization
    $q^{-2|I|}q^{2|I|}=1$ gives $\|E_K^{\mathcal K}(B)\|\le\|B\|$.
\end{proof}

\begin{lemma}[Averaging error from on-site commutator bounds]
    \label{thm:qc_spectator_error}
    \uses{def:qc_spectator_expectation}
    \lean{QuantumCircuit.norm_sub_spectatorSiteExpectation_le}
    \leanok
    Suppose $q>0$, and let $\varepsilon:I\to\R$ satisfy
    $\|[B,\widehat U]\|\le\varepsilon_y$ for each $y\notin K$ and
    every unitary $U\in\mathcal A_{\{y\}}$. Then
    \begin{align}
        \|B-E_K^{\mathcal K}(B)\|
         & \le\sum_{y\notin K}\varepsilon_y.
        \label{eq:qc_spectator_error}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:qc_spectator_average,def:qc_partial_weyl,
        thm:qc_partial_weyl_identities,thm:qc_unitary_product_commutator}
    For unitaries $V,W$ on $\mathcal H\otimes\mathcal K$, the identity
    $[B,VW]=[B,V]W+V[B,W]$ implies
    $\|[B,VW]\|\le\|[B,V]\|+\|[B,W]\|$ by
    Lemma~\ref{thm:qc_unitary_product_commutator}.
    Starting from $P_{\varnothing}(p)=\mathbb 1$, induction over the sites
    outside $K$ via the insertion identity~(\ref{eq:qc_partial_weyl_insert})
    and the single-site unitarity and support properties of
    Lemma~\ref{thm:qc_partial_weyl_identities} (with the identity on the
    spectator throughout) bounds the commutator with $\widehat{W_K(p)}$ by
    $\sum_{y\notin K}\varepsilon_y$.
    For the unitary $V=\widehat{W_K(p)}$, the identity
    \begin{align}
        B-VBV^\dagger & =[B,V]V^\dagger
        \label{eq:qc_spectator_difference}
    \end{align}
    therefore yields
    \begin{align}
        \|B-VBV^\dagger\| & =\|[B,V]\|
        \le\sum_{y\notin K}\varepsilon_y.
        \label{eq:qc_spectator_difference_bound}
    \end{align}
    Averaging~(\ref{eq:qc_spectator_difference_bound}) over all Weyl labels
    via~(\ref{eq:qc_spectator_average}) and $q^{-2|I|}\sum_p B=B$ gives
    (\ref{eq:qc_spectator_error}).
\end{proof}

For $q>0$, setting $K=I\setminus Q$ and $\varepsilon_y=d_y(B)$ in
Lemma~\ref{thm:qc_spectator_error} together with
Lemma~\ref{thm:qc_site_oscillation_bounds} yields the physical-site
averaging estimate of \cite{OpenAI2026AreaLaw}:
\begin{align}
    \|B-E_{I\setminus Q}^{\mathcal K}(B)\|
     & \le\sum_{y\in Q}d_y(B).
    \label{eq:qc_osc_averaging_error}
\end{align}
By Lemma~\ref{thm:qc_spectator_commutant}, the averaged operator
$E_{I\setminus Q}^{\mathcal K}(B)$ commutes with $U\otimes\mathbb 1_{\mathcal K}$
for every $U\in\mathcal A_Q$, with no factor depending on $\dim\mathcal K$.
