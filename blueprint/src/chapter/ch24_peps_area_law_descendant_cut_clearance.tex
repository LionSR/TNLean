\section{Cut clearance of belt and descendant birth templates}
\label{sec:area_law_descendant_cut_clearance}

The belt and replacement templates of \cite[Section~11]{OpenAI2026AreaLaw}
remain far from the cut endpoints $Z$ relative to their size. Write $r=2^k$ and
$t_k$ for the layer and fine sides (Definition~\ref{def:area_law_fine_belt_scales}),
$C_0\ge2$ for the layer width, and $d(S)$ for the removal half-side
(Definition~\ref{def:area_law_repair_square}). A belt tile has
$(u,s_0,n)=(t_k,u,\lceil C_Bu\rceil)$, and a replacement with removal half-side
$u$ has $(s_0,n)=(2u,\lceil C_Bu\rceil)$, with $k_0$ chosen after $D_0,C_B$.

\begin{theorem}[Scale inequalities of late layers]
    \label{thm:area_law_descendant_clearance_scales}
    \lean{TNLean.PEPS.AreaLaw.Geometry.two_pow_fineScaleIndex_le_rpow,
        TNLean.PEPS.AreaLaw.Geometry.eventually_descendant_cutClearance_scales}
    \leanok
    \uses{def:area_law_fine_belt_scales,thm:area_law_exponent_gaps,
        thm:area_law_repair_margins}
    For all $k$, $t_k\le r^\zeta$, and for any $D_0,C_B\in\mathbb R$, all
    sufficiently large $k$ satisfy
    \begin{align}
        2t_k^{\rho_0}              & \le r, &
        4D_0\cdot2t_k              & <r,    &
        2\lceil C_Bt_k\rceil^\beta & <r.
        \label{eq:area_law_descendant_clearance_scales}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{def:area_law_fine_belt_scales,thm:area_law_exponent_gaps,
        thm:area_law_repair_margins}
    Since $\lfloor\zeta k\rfloor\le\zeta k$, $t_k\le r^\zeta$. For any $M$ and
    $a\in\{\zeta\rho_0,\zeta,\zeta\beta\}\subset[0,1)$
    (Theorems~\ref{thm:area_law_exponent_gaps} and~\ref{thm:area_law_repair_margins}),
    $Mr^a=Mr^{-(1-a)}r<r$ once $r^{-(1-a)}<(\max\{M,0\}+1)^{-1}$; using
    $\lceil C_Bt_k\rceil\le(\max\{C_B,0\}+1)t_k$ for $t_k\ge1$ gives~(\ref{eq:area_law_descendant_clearance_scales}).
\end{proof}

\begin{theorem}[Template clearances from endpoint distance]
    \label{thm:area_law_template_clearance_of_endpoint_distance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.cutClearance_of_endpoint_dist_ge}
    \leanok
    \uses{def:area_law_template,thm:area_law_descendant_clearance_scales}
    If a template $T$ with parameters $(n,s_0)$ satisfies
    $\operatorname{dist}_\infty(T,Z)\ge(C_0-1)r$, $0\le u\le t_k$, $s_0\le2u$,
    $n\le\lceil C_Bu\rceil$, and the second and third bounds of~(\ref{eq:area_law_descendant_clearance_scales}),
    then
    \begin{align}
        \operatorname{dist}_\infty(T,Z) & >4D_0s_0,  &
        \operatorname{dist}_\infty(T,Z) & >2n^\beta.
        \label{eq:area_law_template_clearance}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Since $C_0\ge2$, $\operatorname{dist}_\infty(T,Z)\ge r$. By~(\ref{eq:area_law_descendant_clearance_scales})
    and $0\le u\le t_k$, $4D_0s_0\le\max\{0,4D_0\cdot2t_k\}<r$ and
    $2n^\beta\le2\lceil C_Bt_k\rceil^\beta<r$, yielding~(\ref{eq:area_law_template_clearance}).
\end{proof}

\begin{theorem}[Initial marks are far from the cut]
    \label{thm:area_law_initial_mark_endpoint_distance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_zero_endpoint_dist_ge}
    \leanok
    \uses{thm:area_law_initial_repair_mark_facts,
        thm:area_law_belt_marks_closed_cell,
        thm:area_law_fine_cell_containment,
        thm:area_law_dyadic_layer_distance}
    Every initial mark $v$ of scale $S_v$ marks a belt cell at some layer
    $k\ge k_0$ with $S_v=t_k$ and
    \begin{align}
        \operatorname{dist}_\infty(v,Z) & \ge C_0r.
        \label{eq:area_law_initial_mark_dist}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_initial_repair_mark_facts,
        thm:area_law_belt_marks_closed_cell,
        thm:area_law_fine_cell_containment,
        thm:area_law_dyadic_layer_distance}
    By Theorems~\ref{thm:area_law_initial_repair_mark_facts},~\ref{thm:area_law_belt_marks_closed_cell},
    and~\ref{thm:area_law_fine_cell_containment}, $v\in\overline{D_k}$ with
    $S_v=t_k$, so Theorem~\ref{thm:area_law_dyadic_layer_distance} gives~(\ref{eq:area_law_initial_mark_dist}).
\end{proof}

\begin{theorem}[Descendant birth regions are far from the cut]
    \label{thm:area_law_descendant_birth_endpoint_distance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_birth_endpoint_dist_ge}
    \leanok
    \uses{thm:area_law_initial_mark_endpoint_distance,
        def:area_law_uniform_repair_conclusion,
        thm:area_law_repair_descendant_scales}
    Under the descendant bound of Definition~\ref{def:area_law_uniform_repair_conclusion},
    if a generation-$(j+1)$ replacement has parent mark $m$ of scale $S_m$ and
    initial ancestor at layer $k\ge k_0$ of scale $t_k$, then $d(S_m)\le t_k$
    and every $y$ in its birth region satisfies
    \begin{align}
        \operatorname{dist}_\infty(y,Z) & \ge C_0r-2d(t_k).
        \label{eq:area_law_descendant_birth_dist}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_initial_mark_endpoint_distance,
        def:area_law_uniform_repair_conclusion,
        thm:area_law_repair_descendant_scales,thm:area_law_repair_margins}
    By Theorems~\ref{thm:area_law_repair_descendant_scales}
    and~\ref{thm:area_law_repair_margins}, $S_m=d^{(j)}(t_k)$ and
    $1\le d(S)\le S^{\rho_0}\le S$ give $d(S_m)\le t_k$. Since $y$ lies within
    $2d(t_k)$ of the initial ancestor mark $v_0$
    (Definition~\ref{def:area_law_uniform_repair_conclusion}, even when
    extending into the dummy region),~(\ref{eq:area_law_initial_mark_dist}) and
    the triangle inequality give~(\ref{eq:area_law_descendant_birth_dist}).
\end{proof}

\begin{theorem}[Cut clearance of belt birth templates]
    \label{thm:area_law_belt_birth_template_clearance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialBelt_birthTemplate_cutClearance}
    \leanok
    \uses{thm:area_law_template_clearance_of_endpoint_distance,
        thm:area_law_fan_run_cover,thm:area_law_fine_cell_containment,
        thm:area_law_dyadic_layer_distance,
        def:area_law_initial_birth_regions}
    For any $D_0,C_B\in\mathbb R$ and all sufficiently large $k_0$, every belt
    tile at layer $k\ge k_0$ has birth region $B$ satisfying
    $\operatorname{dist}_\infty(B,Z)\ge C_0r$, and every template $T\subseteq B$
    with $s_0\le2t_k$ and $n\le\lceil C_Bt_k\rceil$ (in particular
    $(s_0,n)=(t_k,\lceil C_Bt_k\rceil)$) satisfies~(\ref{eq:area_law_template_clearance}).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_fan_run_cover,thm:area_law_fine_cell_containment,
        thm:area_law_dyadic_layer_distance,
        thm:area_law_descendant_clearance_scales,
        thm:area_law_template_clearance_of_endpoint_distance}
    By Definition~\ref{def:area_law_initial_birth_regions} and
    Theorems~\ref{thm:area_law_fan_run_cover},~\ref{thm:area_law_fine_cell_containment},
    and~\ref{thm:area_law_dyadic_layer_distance}, $B\subseteq\overline{D_k}$ has
    $\operatorname{dist}_\infty(B,Z)\ge C_0r\ge(C_0-1)r$; applying
    Theorems~\ref{thm:area_law_descendant_clearance_scales}
    and~\ref{thm:area_law_template_clearance_of_endpoint_distance} with
    $u=t_k$ gives~(\ref{eq:area_law_template_clearance}).
\end{proof}

\begin{theorem}[Cut clearance of descendant birth templates]
    \label{thm:area_law_descendant_birth_template_clearance}
    \lean{TNLean.PEPS.AreaLaw.Geometry.descendant_birthTemplate_cutClearance}
    \leanok
    \uses{thm:area_law_uniform_repair_induction,
        thm:area_law_descendant_birth_endpoint_distance,
        thm:area_law_descendant_clearance_scales,
        thm:area_law_template_clearance_of_endpoint_distance}
    For all sufficiently large $S_*$, every $C_0\ge2$, all $D_0,C_B\in\mathbb R$,
    and all sufficiently large $k_0$, each replacement of any generation with
    removal half-side $u$ has an initial ancestor at layer $k\ge k_0$ with
    $u\le t_k$, its birth region $B$ satisfies
    $\operatorname{dist}_\infty(B,Z)\ge(C_0-1)r$ (including in the dummy
    region), and every template $T\subseteq B$ with $s_0\le2u$ and
    $n\le\lceil C_Bu\rceil$ (in particular $(s_0,n)=(2u,\lceil C_Bu\rceil)$)
    satisfies~(\ref{eq:area_law_template_clearance}).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_uniform_repair_induction,thm:area_law_repair_depth,
        thm:area_law_descendant_clearance_scales,
        thm:area_law_descendant_birth_endpoint_distance,
        thm:area_law_repair_margins,
        thm:area_law_template_clearance_of_endpoint_distance}
    Above the thresholds of Theorems~\ref{thm:area_law_uniform_repair_induction}
    and~\ref{thm:area_law_descendant_clearance_scales}, combining~(\ref{eq:area_law_descendant_birth_dist})
    with $2d(t_k)\le2t_k^{\rho_0}\le r$ (Theorem~\ref{thm:area_law_repair_margins}
    and~(\ref{eq:area_law_descendant_clearance_scales})) gives
    $\operatorname{dist}_\infty(B,Z)\ge(C_0-1)r$, and
    Theorem~\ref{thm:area_law_template_clearance_of_endpoint_distance} gives~(\ref{eq:area_law_template_clearance}).
\end{proof}
