\section{The bootstrap lemma in Hastings' area law}
\label{sec:hastings_bootstrap}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 280--411 (Lemma 2, label ble,
% and its proof).

The bootstrap lemma of \cite{Hastings2007AreaLaw} converts overlap of a ground
state with a mixture of bounded-Schmidt-rank states into a cut-entropy bound in
three steps: approximate factorization of the ground-state projector maps the
mixture to higher-rank mixtures with large overlap, overlap at each scale bounds
the Schmidt tail, and the entropy under geometric tail constraints is
maximized. Combining these steps with \cite[Lemma~1]{Hastings2007AreaLaw} yields
the chain bootstrap bound.

\begin{lemma}[Gibbs' inequality]
    \label{thm:hastings_gibbs_inequality}
    \lean{TNLean.MPS.AreaLaw.Hastings.probabilityEntropy_le_sum_mul_neg_log}
    \leanok
    For a probability vector $p$ and weights $w_z>0$ with $\sum_zw_z\le1$,
    \begin{align}
        H(p)\le-\sum_zp_z\ln w_z.
        \label{eq:hastings_gibbs_inequality}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Since $\ln t\le t-1$, $p_z\ln(w_z/p_z)\le w_z-p_z$ for $p_z>0$ (and $0\le w_z$
    for $p_z=0$); summing over $z$ gives
    $H(p)+\sum_zp_z\ln w_z\le\sum_zw_z-1\le0$.
\end{proof}

\begin{lemma}[Entropy under geometric tail constraints]
    \label{thm:hastings_geometric_tail_entropy}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.probabilityEntropy_le_of_geometric_tails,
        TNLean.MPS.AreaLaw.Hastings.probabilityEntropy_le_of_scale_tails
    }
    \leanok
    Let $p_0,\dots,p_{N-1}$ be a probability vector, $K\ge1$, $D\ge2$, and
    $0<x<1$. If $\sum_{a\ge KD^{2n}}p_a\le x^{n+1}$ for all $n\ge0$, then
    \begin{align}
        H(p)\le\ln K-\ln(1-x)+x\bigl(\ln(D^2-1)-\ln x\bigr)
        +\frac{x^2\,(2\ln D-\ln x)}{1-x}.
        \label{eq:hastings_geometric_tail_entropy}
    \end{align}
    For $x=e^{-2/\xi'}$ ($\xi'>0$) and $K=kD^{2m+2}$ ($k\ge1$, $m\in\mathbb N$),
    \begin{align}
        H(p)\le\ln k+(2m+2+\xi')\ln D+1+\ln(D^2-1)+\ln(\xi'/2+1)
        \label{eq:hastings_scale_tail_entropy}
    \end{align}
    \cite[Lemma~2, lines 357--390]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_gibbs_inequality}
    Apply (\ref{eq:hastings_gibbs_inequality}) with weights
    $w_a=(1-x)/K$ on $[0,K)$ and
    $w_a=(1-x)x^n/\bigl(KD^{2(n-1)}(D^2-1)\bigr)$ on $[KD^{2(n-1)},KD^{2n})$
    for $n\ge1$; these sum to $1-x^{N+1}\le1$ on $[0,KD^{2N})$, and bounding
    $\sum_{a\ge KD^{2i}}p_a\le x^{i+1}$ gives
    (\ref{eq:hastings_geometric_tail_entropy}). For $x=e^{-2/\xi'}$,
    $x/(1-x)\le\xi'/2$ yields (\ref{eq:hastings_scale_tail_entropy})
    \cite[lines 386--389]{Hastings2007AreaLaw}.
\end{proof}

\begin{lemma}[The component orthogonal to a unit vector]
    \label{thm:hastings_orthogonal_component}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_sq_sub_normSq_inner_eq}
    \leanok
    For $\|\Psi\|=1$ and any vector $v$,
    \begin{align}
        \|v\|^2-|\langle\Psi,v\rangle|^2=\bigl\|v-\langle\Psi,v\rangle\Psi\bigr\|^2.
        \label{eq:hastings_orthogonal_component}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Expand the right-hand side of (\ref{eq:hastings_orthogonal_component}) using
    $\|\Psi\|=1$.
\end{proof}

\begin{lemma}[Overlap with the ground state after an approximate projector]
    \label{thm:hastings_bootstrap_overlap}
    \lean{TNLean.MPS.AreaLaw.Hastings.overlap_ge_of_norm_sub_vecMulVec_le}
    \leanok
    Let $\|\Psi_0\|=1$, $\rho=\sum_iw_i|\psi_i\rangle\langle\psi_i|$ with
    $\operatorname{tr}\rho=1$ and $\langle\Psi_0,\rho\Psi_0\rangle=P>0$, and
    $\|O-|\Psi_0\rangle\langle\Psi_0|\|\le\epsilon$. Then
    \begin{align}
        \langle\Psi_0,O\rho O^\dagger\Psi_0\rangle
        \ge\Bigl(1-\frac{2\epsilon^2}{P}\Bigr)\operatorname{tr}(O\rho O^\dagger),
        \label{eq:hastings_bootstrap_overlap}
    \end{align}
    and $\operatorname{tr}(O\rho O^\dagger)>0$ when $\epsilon<\sqrt P$
    \cite[Lemma~2, eq.~(hg), lines 317--331]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_orthogonal_component}
    Set $P_0=|\Psi_0\rangle\langle\Psi_0|$,
    $a=\langle\Psi_0,O\rho O^\dagger\Psi_0\rangle$, and
    $b=\operatorname{tr}((1-P_0)O\rho O^\dagger)$. For $\epsilon<\sqrt P$,
    Minkowski's inequality gives $a\ge(\sqrt P-\epsilon)^2$ and
    $(1-P_0)O=(1-P_0)(O-P_0)$ gives $b\le\epsilon^2$, so $a+b>0$ and
    $b/(a+b)\le\epsilon^2/\bigl(2(\epsilon-\sqrt P/2)^2+P/2\bigr)\le2\epsilon^2/P$.
    For $\epsilon\ge\sqrt P$, $1-2\epsilon^2/P<0$.
\end{proof}

\begin{lemma}[Overlap with a vector of bounded Schmidt rank]
    \label{thm:hastings_schmidt_overlap_tail}
    \lean{TNLean.MPS.AreaLaw.Hastings.normSq_dotProduct_le_kyFanNorm}
    \leanok
    For a bipartite vector $\varphi$ with coefficient matrix $C_\varphi$ and
    reduced operator $\rho_\varphi=C_\varphi C_\varphi^\dagger$, any $\psi$ of
    Schmidt rank at most $K$ (below the first-factor dimension) satisfies
    $|\langle\varphi,\psi\rangle|^2\le\|\psi\|^2\,\|\rho_\varphi\|_{(K)}$, where
    $\|\cdot\|_{(K)}$ is the Ky Fan $K$-norm.
\end{lemma}
\begin{proof}\leanok
    For $\psi\ne0$, normalize $\psi$ and apply the Ky Fan variational bound for
    rank-$K$ overlaps.
\end{proof}

\begin{lemma}[The Schmidt tail from one mixture]
    \label{thm:hastings_mixture_tail}
    \lean{TNLean.MPS.AreaLaw.Hastings.one_sub_kyFanNorm_le_of_schmidtRank_mixture}
    \leanok
    Let $\|\Psi\|=1$ have reduced state $\rho_\Psi$, let $K\in\mathbb N$,
    $\eta\ge0$, and let $\psi_1,\dots,\psi_r$ have Schmidt rank at most $K$ with
    weights $w_i\ge0$ and $W=\sum_iw_i\|\psi_i\|^2>0$. If
    $(1-\eta)W\le\sum_iw_i|\langle\Psi,\psi_i\rangle|^2$, then
    $1-\|\rho_\Psi\|_{(K)}\le\eta$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_schmidt_overlap_tail}
    If $K$ is at least the first-factor dimension,
    $\|\rho_\Psi\|_{(K)}=\operatorname{tr}\rho_\Psi=1$; otherwise
    Lemma~\ref{thm:hastings_schmidt_overlap_tail} gives
    $(1-\eta)W\le\sum_iw_i|\langle\Psi,\psi_i\rangle|^2\le W\|\rho_\Psi\|_{(K)}$.
\end{proof}

\begin{lemma}[The cut rank under supported operators]
    \label{thm:hastings_supported_cut_rank}
    \lean{
        TNLean.MPS.AreaLaw.rank_cutMatrix_toEuclideanLin_le_pow,
        TNLean.MPS.AreaLaw.rank_cutMatrix_toEuclideanLin_le_of_subset_leftBlock,
        TNLean.MPS.AreaLaw.rank_cutMatrix_toEuclideanLin_le_of_disjoint_leftBlock,
        TNLean.MPS.AreaLaw.rank_cutMatrix_toEuclideanLin_le_of_isOneSided,
        TNLean.MPS.AreaLaw.rank_schmidtCoeffMatrix_mulVec_le_of_isOneSided,
        TNLean.MPS.AreaLaw.cutMatrix_toEuclideanLin
    }
    \leanok
    \uses{def:chain_model}
    On $\mathcal H_1\otimes\mathcal H_2$, $X\otimes1$ and $1\otimes X$ do not
    increase the coefficient-matrix rank. On an $L$-site chain of local
    dimension $q\ge1$, let $r_k(\psi)$ be the Schmidt rank across cut $k$. If
    $X$ acts on $\{0,\dots,k-1\}$ or $\{k,\dots,L-1\}$, then
    $r_k(X\psi)\le r_k(\psi)$; if $X$ acts on $\{0,\dots,k+m-1\}$ ($k+m\le L$),
    then $r_k(X\psi)\le q^{2m}r_k(\psi)$
    \cite[Lemma~2, lines 321--322]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_cut_rank_growth}
    Since
    \begin{align}
        C_{(X\otimes1)\psi}=XC_\psi,\qquad C_{(1\otimes X)\psi}=C_\psi X^{\mathsf T},
        \label{eq:hastings_supported_cut_rank_one_sided}
    \end{align}
    one-sided operators do not increase rank. By
    Lemma~\ref{thm:chain_cut_rank_growth}, shifting a cut by $m$ sites changes
    $r_k$ by at most $q^m$, and $X$ is one-sided across cut $k+m$, so
    \begin{align}
        r_k(X\psi)\le q^m\,r_{k+m}(X\psi)\le q^m\,r_{k+m}(\psi)
        \le q^{2m}\,r_k(\psi).
        \label{eq:hastings_supported_cut_rank_shift}
    \end{align}
\end{proof}

\begin{definition}[The remainder of the bootstrap bound]
    \label{def:hastings_bootstrap_remainder}
    \lean{TNLean.MPS.AreaLaw.Hastings.bootstrapRemainder}
    \leanok
    For $\xi'>0$ and $D\in\mathbb N$ with $D\ge2$, set
    \begin{align}
        F(\xi',D)=(\xi'+4)\ln D+1+\ln(D^2-1)+\ln(\xi'/2+1)
        \label{eq:hastings_bootstrap_remainder}
    \end{align}
    \cite[Lemma~2, lines 304--309]{Hastings2007AreaLaw}.
\end{definition}

\begin{lemma}[The Schmidt tail from overlaps with mixtures of bounded Schmidt rank]
    \label{thm:hastings_bootstrap_schmidt_tail}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_eigenvalues₀_tail_le_of_schmidtRank_mixtures,
        TNLean.MPS.AreaLaw.Hastings.eigenvalues₀_nonneg_and_sum_eq_one}
    \leanok
    Let $\|\Psi\|=1$ have reduced state $\rho_\Psi$ with eigenvalues
    $p_1\ge p_2\ge\dots$, and let $k,D\in\mathbb N$, $\xi'>0$, and $c\ge1$. If
    for every $s\in\mathbb N$ with $ce^{-2s/\xi'}<1$ some mixture of states of
    Schmidt rank at most $kD^{2s}$ has normalized overlap with $\Psi$ at least
    $1-ce^{-2s/\xi'}$, then $(p_a)$ is a probability vector and there exists
    $m'\in\mathbb N$ with $m'<\tfrac{\xi'}{2}\ln c+1$ such that for all
    $n\ge0$,
    \begin{align}
        \sum_{a>kD^{2(m'+1+n)}}p_a\le e^{-2(n+1)/\xi'}
        \label{eq:hastings_bootstrap_schmidt_tail}
    \end{align}
    \cite[Lemma~2, eqs.~(constraint) and (constraint2), lines 333--356]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_mixture_tail}
    Since $\rho_\Psi\ge0$ and $\operatorname{tr}\rho_\Psi=\|\Psi\|^2=1$, $(p_a)$
    is a probability vector, and Lemma~\ref{thm:hastings_mixture_tail} gives
    $\sum_{a>kD^{2s}}p_a\le ce^{-2s/\xi'}$ whenever $ce^{-2s/\xi'}<1$. Taking
    the least $m'\in\mathbb N$ with $ce^{-2m'/\xi'}\le1$ gives
    $m'<\tfrac{\xi'}2\ln c+1$ and $ce^{-2(m'+1+n)/\xi'}\le e^{-2(n+1)/\xi'}<1$.
\end{proof}

\begin{lemma}[Entropy from overlaps with mixtures of bounded Schmidt rank]
    \label{thm:hastings_bootstrap_stage}
    \lean{TNLean.MPS.AreaLaw.Hastings.vonNeumannEntropy_le_of_schmidtRank_mixtures}
    \leanok
    \uses{def:hastings_bootstrap_remainder}
    Under the hypotheses of Lemma~\ref{thm:hastings_bootstrap_schmidt_tail}
    with $k\ge1$ and $D\ge2$,
    \begin{align}
        S(\rho_\Psi)\le\ln k+\xi'\ln(c)\ln D+F(\xi',D)
        \label{eq:hastings_bootstrap_stage}
    \end{align}
    \cite[Lemma~2, eq.~(bootstrap)]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_geometric_tail_entropy,
        thm:hastings_bootstrap_schmidt_tail}
    Combine (\ref{eq:hastings_bootstrap_schmidt_tail}) and
    $2m'+2\le\xi'\ln c+4$ from Lemma~\ref{thm:hastings_bootstrap_schmidt_tail}
    with (\ref{eq:hastings_scale_tail_entropy}).
\end{proof}

\begin{lemma}[Cut coordinates preserve inner products]
    \label{thm:hastings_cut_coordinates}
    \lean{TNLean.MPS.AreaLaw.Hastings.star_cutVector_dotProduct}
    \leanok
    \uses{def:chain_model}
    For chain vectors $x,y$ and their bipartite coordinates $\hat x,\hat y$
    across cut $j$, $\langle\hat x,\hat y\rangle=\langle x,y\rangle$.
\end{lemma}
\begin{proof}\leanok
    Bipartite cut coordinates reindex the chain configuration basis.
\end{proof}

\begin{lemma}[The entropy of a cut from its coefficient matrix]
    \label{thm:hastings_cut_entropy}
    \lean{TNLean.MPS.AreaLaw.Hastings.regionEntropy_interval_eq}
    \leanok
    \uses{def:chain_model}
    For a chain vector $\Omega$ with coefficient matrix $C_\Omega$ across cut
    $j$, the reduced operator of $|\Omega\rangle\langle\Omega|$ on
    $\{0,\dots,j-1\}$ is $C_\Omega C_\Omega^\dagger$, so its regional entropy
    equals $S(C_\Omega C_\Omega^\dagger)$.
\end{lemma}
\begin{proof}\leanok
    In cut coordinates across $\{0,\dots,j-1\}$, the partial trace over the
    complement of $|\Omega\rangle\langle\Omega|$ is $C_\Omega C_\Omega^\dagger$.
\end{proof}

\begin{lemma}[The cut rank past the end of the chain]
    \label{thm:hastings_terminal_cut_rank}
    \lean{TNLean.MPS.AreaLaw.Hastings.rank_cutMatrix_le_pow_of_lt}
    \leanok
    \uses{def:chain_model}
    On an $L$-site chain of local dimension $q\ge1$, if $L<j+s$ then every
    vector has Schmidt rank at most $q^{2s}$ across cut $j$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_cut_rank_growth}
    By Lemma~\ref{thm:chain_cut_rank_growth}, shifting cut $j\le L$ to $L$
    (where the rank is at most $1$) changes the rank by at most
    $q^{L-j}\le q^{2s}$; for $j>L$ the rank is already at most $1$.
\end{proof}

\begin{lemma}[One scale of the bootstrap on the chain]
    \label{thm:hastings_bootstrap_scale}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_mixture_rank_cutMatrix_le_overlap_of_factorization}
    \leanok
    \uses{def:chain_model}
    Let $\|\Omega\|=1$ on an $L$-site chain of local dimension $q\ge1$, $j$ be a
    cut, and $\xi',C_1\in\mathbb R$. Suppose that for every $m\in\mathbb N$ with
    $j+m\le L$ there exist $O_B,O_L,O_R$ supported on $\{0,\dots,j+m-1\}$,
    $\{0,\dots,j-1\}$, and $\{j,\dots,L-1\}$ such that
    $\|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|\le C_1e^{-m/\xi'}$. Let
    $\psi_1,\dots,\psi_r$ have Schmidt rank at most $k\ge1$ across cut $j$ with
    $w_i\ge0$, $\sum_iw_i\|\psi_i\|^2=1$, and
    $P=\sum_iw_i|\langle\psi_i,\Omega\rangle|^2>0$. Set $c=2C_1^2/P$. For every
    $s\in\mathbb N$ with $ce^{-2s/\xi'}<1$ (reading $-s/\xi'=0$ if $\xi'=0$),
    some mixture of vectors of Schmidt rank at most $kq^{2s}$ across cut $j$ has
    normalized overlap at least $1-ce^{-2s/\xi'}$ with $\Omega$
    \cite[Lemma~2, eq.~(hg), lines 315--331]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_bootstrap_overlap, thm:hastings_supported_cut_rank,
        thm:hastings_terminal_cut_rank}
    For $j+s\le L$, set $O=O_BO_LO_R$ at $m=s$: each $O\psi_i$ has Schmidt rank
    at most $kq^{2s}$ by Lemma~\ref{thm:hastings_supported_cut_rank}, and
    Lemma~\ref{thm:hastings_bootstrap_overlap} with
    $\epsilon=C_1e^{-s/\xi'}<\sqrt P$ gives the overlap for
    $\sum_iw_i|O\psi_i\rangle\langle O\psi_i|$. For $j+s>L$, take $\Omega$
    itself by Lemma~\ref{thm:hastings_terminal_cut_rank}.
\end{proof}

\begin{lemma}[The bootstrap bound on the chain, stage form]
    \label{thm:hastings_bootstrap_chain_stage}
    \lean{TNLean.MPS.AreaLaw.Hastings.entropy_le_of_overlap_schmidtRank_mixture_of_factorization}
    \leanok
    \uses{def:chain_model, def:hastings_bootstrap_remainder}
    In the setting of Lemma~\ref{thm:hastings_bootstrap_scale} with $q\ge2$,
    $\xi'>0$, $C_1\ge1$, and
    \begin{align}
        \|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|\le C_1e^{-m/\xi'}
        \label{eq:hastings_bootstrap_chain_stage_hyp}
    \end{align}
    for all $j+m\le L$, the entropy $S_j$ of $\Omega$ on $\{0,\dots,j-1\}$
    satisfies
    \begin{align}
        S_j\le\ln k+\xi'\ln\bigl(2C_1^2/P\bigr)\ln q+F(\xi',q)
        \label{eq:hastings_bootstrap_chain_stage}
    \end{align}
    \cite[Lemma~2]{Hastings2007AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_bootstrap_stage, thm:hastings_bootstrap_scale,
        thm:hastings_cut_coordinates, thm:hastings_cut_entropy}
    Combine Lemmas~\ref{thm:hastings_cut_coordinates},
    \ref{thm:hastings_cut_entropy}, \ref{thm:hastings_bootstrap_scale}, and
    \ref{thm:hastings_bootstrap_stage} with $c=2C_1^2/P\ge1$ and $D=q$.
\end{proof}

\begin{lemma}[The bootstrap bound from the factorization at one cut]
    \label{thm:hastings_bootstrap_factorization_at}
    \lean{TNLean.MPS.AreaLaw.Hastings.entropy_le_of_overlap_schmidtRank_mixture_of_isFactorizationAt}
    \leanok
    \uses{def:hastings_factorization_at, def:hastings_bootstrap_remainder}
    If $\|\Omega\|=1$ on an $L$-site chain of local dimension $q\ge2$ admits an
    approximate factorization (Definition~\ref{def:hastings_factorization_at})
    of $|\Omega\rangle\langle\Omega|$ at cut $j$ with error $C_1e^{-m/\xi'}$
    ($\xi'>0$, $C_1\ge1$) for every $m\in\mathbb N$, then
    (\ref{eq:hastings_bootstrap_chain_stage}) holds for every mixture as in
    Lemma~\ref{thm:hastings_bootstrap_chain_stage}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_bootstrap_chain_stage, thm:hastings_factorization_bootstrap_form}
    By Lemma~\ref{thm:hastings_factorization_bootstrap_form}, $O_B$ supported
    on $\{j-m,\dots,j+m-1\}$ is supported on $\{0,\dots,j+m-1\}$, so
    Lemma~\ref{thm:hastings_bootstrap_chain_stage} applies.
\end{proof}

\begin{lemma}[The bootstrap lemma]
    \label{thm:hastings_bootstrap}
    \lean{TNLean.MPS.AreaLaw.exists_entropy_le_of_overlap_schmidtRank_mixture}
    \leanok
    \uses{def:chain_model, def:hastings_bootstrap_remainder,
        thm:hastings_lemma1_endpoint}
    For every range $R$ there exists $K=K(R)\ge1$ such that, for any Hamiltonian
    $H$ of range $R$, term norm bound $J\ge0$, unique ground state $\Psi_0$, and
    gap $\Delta E>0$ on $L$ sites of local dimension $D$, with $\tilde\xi$ from
    Lemma~\ref{thm:hastings_lemma1_endpoint}, $\xi'=6\tilde\xi$, and
    $C_1=K\tilde\xi^3$, any unit-trace mixture $\rho$ of pure states of Schmidt
    rank at most $k\ge1$ across cut $j$ with
    $P=\langle\Psi_0,\rho\Psi_0\rangle>0$ satisfies
    \begin{align}
        S(\rho^0_{1,j})\le\ln k+\xi'\ln\bigl(2C_1^2/P\bigr)\ln D+F(\xi',D),
        \label{eq:hastings_bootstrap}
    \end{align}
    where $\rho^0_{1,j}=\operatorname{tr}_{X_{j+1,N}}|\Psi_0\rangle\langle\Psi_0|$.
    This is \cite[Lemma~2, lines 280--311]{Hastings2007AreaLaw} at
    $\xi'=6\tilde\xi$ with $C_1$ polynomial in $\tilde\xi$, inheriting the
    raised-velocity chain localization length $\tilde\xi$ of
    Lemma~\ref{thm:hastings_lemma1_endpoint}
    \cite{gap:hastings07_lemma1_decay_length,
        gap:hastings07_gaussian_average_exponent}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint,
        thm:hastings_bootstrap_factorization_at, thm:hastings_factorization_mono,
        lem:chain_entropy_inequalities}
    Lemma~\ref{thm:hastings_lemma1_endpoint} gives $K_0(R)\ge0$; set
    $K=\max(K_0,1)$ so $C_1=K\tilde\xi^3\ge1$ (since $\tilde\xi\ge20$). For
    $D\ge2$, Lemma~\ref{thm:hastings_bootstrap_factorization_at} gives
    (\ref{eq:hastings_bootstrap}); for $D=1$, $S(\rho^0_{1,j})=0$ and the
    right-hand side is nonnegative under $\ln0=0$.
\end{proof}
