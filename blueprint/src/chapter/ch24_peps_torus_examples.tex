\chapter{Examples of PEPS on the torus}%
\label{ch:peps_examples}

The simplest PEPS place one four-leg tensor at every site of a square lattice
on the torus. This chapter fixes that setting and computes the states of the
standard examples of \cite[Appendix~A]{Cirac2021Matrix}: the GHZ state, the
cluster state, the CZX model, the AKLT state, the resonating valence bond
state, and the PEPS whose coefficients are the Boltzmann weights of classical
models. The toric code and the quantum double models are treated with the
theory of $G$-injective tensors in Chapter~\ref{ch:peps_quantum_double}.

\section{Torus PEPS of a four-leg tensor}%
\label{sec:peps_torus_site_tensor}

Two-dimensional PEPS on the discrete torus are defined by a four-leg tensor
$A^i_{\alpha\beta\gamma\delta}$ placed at every site of a square lattice, with
virtual indices ordered top, right, down, left
\cite[Appendix~A, ``Two dimensions: PEPS'']{Cirac2021Matrix}. On a torus, the
four legs of site $v=(x,y)$ correspond to the edges from $v$ to $(x,y+1)$
(top), from $v$ to $(x+1,y)$ (right), from $(x,y-1)$ to $v$ (down), and from
$(x-1,y)$ to $v$ (left).

\begin{definition}[Torus PEPS of a four-leg tensor]%
    \label{def:peps_torus_site_tensor}
    \lean{TNLean.PEPS.torusDownEdge, TNLean.PEPS.torusLeftEdge,
        TNLean.PEPS.torusTopLeg, TNLean.PEPS.torusRightLeg,
        TNLean.PEPS.torusDownLeg, TNLean.PEPS.torusLeftLeg,
        TNLean.PEPS.torusLeftEdge_add_one, TNLean.PEPS.torusDownEdge_add_one,
        TNLean.PEPS.torusSiteTensor}
    \leanok
    \uses{def:peps_tensor, def:peps_torus_uniform_bond_dim}
    For a four-leg tensor $a^i_{\alpha\beta\gamma\delta}$ with physical
    dimension $d$ and bond dimension $D$, its torus PEPS on the $w\times h$
    torus has bond dimension $D$ on every edge and local tensor $a$ at every
    site $v$, with $(\alpha,\beta,\gamma,\delta)$ read on the top, right, down,
    and left legs of $v$
    \cite[Appendix~A, ``Two dimensions: PEPS'']{Cirac2021Matrix}, where
    $\mathrm{right}(v)=\mathrm{left}(v+(1,0))$ and
    $\mathrm{top}(v)=\mathrm{down}(v+(0,1))$.
\end{definition}

\begin{lemma}[Shift-invariant functions on the torus are constant]%
    \label{lem:peps_torus_shift_const}
    \lean{TNLean.PEPS.torusVertex_apply_eq_apply_zero_of_shift}
    \leanok
    \uses{def:peps_torus_geometry_translation}
    If $f(x+1,y)=f(x,y)$ and $f(x,y+1)=f(x,y)$ for all $(x,y)$ on the $w\times h$
    torus, then $f(x,y)=f(0,0)$ for all $(x,y)$.
\end{lemma}

\begin{proof}\leanok
    Induct on $0\le x<w$ and $0\le y<h$.
\end{proof}

\begin{lemma}[Contraction of a torus PEPS]%
    \label{lem:peps_torus_site_tensor_contraction}
    \lean{TNLean.PEPS.stateCoeff_torusSiteTensor_eq_sum_edge,
        TNLean.PEPS.isHorizontalTorusEdge_torusRightEdge,
        TNLean.PEPS.isVerticalTorusEdge_torusUpEdge,
        TNLean.PEPS.torusRightEdge_injective, TNLean.PEPS.torusUpEdge_injective,
        TNLean.PEPS.torusRightEdge_ne_torusUpEdge, TNLean.PEPS.torusEdgeEquiv,
        TNLean.PEPS.stateCoeff_torusSiteTensor}
    \leanok
    \uses{def:peps_torus_site_tensor, def:peps_stateCoeff}
    For $w,h\ge 3$, the right edges $e_{\mathrm R}(x,y)$ and up edges
    $e_{\mathrm U}(x,y)$ are pairwise distinct and partition $E$, and the torus
    PEPS of $a$ (Definition~\ref{def:peps_torus_site_tensor}) has coefficient
    \begin{align}
        \Coeff(\sigma)=\sum_{h_\bullet,v_\bullet}\ \prod_{(x,y)}
        a^{\sigma(x,y)}_{v_{(x,y)}\,h_{(x,y)}\,v_{(x,y-1)}\,h_{(x-1,y)}},
        \label{eq:peps_torus_site_contraction}
    \end{align}
    where $h_{(x,y)}$ labels $e_{\mathrm R}(x,y)$ and $v_{(x,y)}$ labels
    $e_{\mathrm U}(x,y)$.
\end{lemma}

\begin{proof}\leanok
    For $w,h\ge 3$, $p=q+(1,0)$ and $q=p+(1,0)$ would imply $2\equiv 0\pmod w$,
    so $e_{\mathrm R}$ and $e_{\mathrm U}$ are injective with disjoint images
    whose union is $E$. Reindexing the edge sum along
    $(x,y)\mapsto(e_{\mathrm R}(x,y),e_{\mathrm U}(x,y))$ gives
    (\ref{eq:peps_torus_site_contraction}).
\end{proof}

For $w=2$ or $h=2$, opposite legs of a site share the same graph edge, so
(\ref{eq:peps_torus_site_contraction}) and the cluster, CZX, AKLT, RVB, and
classical-model statements below require $w,h\ge 3$
\cite{gap:rmp_peps_examples_small_torus}.

\begin{lemma}[Pulling a virtual string through a rectangle]%
    \label{lem:peps_torus_string_pull_through}
    \lean{TNLean.PEPS.gridPath_column_prod_eq, TNLean.PEPS.gridPath_prod_eq,
        TNLean.PEPS.torusWestSouthString, TNLean.PEPS.torusNorthEastString,
        TNLean.PEPS.torusWestSouthString_eq_torusNorthEastString,
        TNLean.PEPS.sum_smul_torusWestSouthString_eq}
    \leanok
    \uses{def:peps_torus_site_tensor}
    Let $a$ be a four-leg tensor and $\alpha\mapsto X_\alpha\in M$ a map into a
    monoid $M$ such that $X_tX_r=X_lX_b$ whenever $a^{s}_{trbl}\neq0$. On an
    $m\times n$ rectangle of sites with left-leg labels $P_{ij}$ and top-leg
    labels $Q_{ij}$ ($0\le i<m$, $0\le j<n$), let
    \begin{align}
        S_{\mathrm{WS}} & = P_{0,0}\cdots P_{0,n-1}\,Q_{0,n}\cdots Q_{m-1,n}, &
        S_{\mathrm{NE}} & = Q_{0,0}\cdots Q_{m-1,0}\,P_{m,0}\cdots P_{m,n-1}
        \notag
    \end{align}
    be the west--south and north--east boundary strings. Then
    $S_{\mathrm{WS}}=S_{\mathrm{NE}}$ on every bond labelling with nonzero PEPS weight,
    and for any $\C$-algebra $M$ the contracted network with $S_{\mathrm{WS}}$
    inserted equals that with $S_{\mathrm{NE}}$ inserted
    \cite[Section~``MPO and PEPO'']{Cirac2021Matrix}.
\end{lemma}

\begin{proof}\leanok
    Since $\mathrm{right}(i,j)=\mathrm{left}(i+1,j)$ and
    $\mathrm{down}(i,j)=\mathrm{top}(i,j+1)$, $a^s_{trbl}\neq 0$ gives
    $Q_{ij}P_{i+1,j}=P_{ij}Q_{i,j+1}$. Induction on $n$ gives
    $P_{i,0}\cdots P_{i,n-1}Q_{i,n}=Q_{i,0}P_{i+1,0}\cdots P_{i+1,n-1}$, and
    induction on $m$ gives $S_{\mathrm{WS}}=S_{\mathrm{NE}}$ on every nonzero
    summand.
\end{proof}

\section{The GHZ state}%
\label{sec:peps_ghz}

The two-dimensional GHZ state is $\sum_{i=0}^{d-1}\ket{i,i,\ldots,i}$
\cite[Appendix~A, ``The GHZ state'']{Cirac2021Matrix}; the simple-graph
representation requires $w,h\ge 2$ \cite{gap:rmp_peps_examples_small_torus}.

\begin{definition}[Two-dimensional GHZ tensor]%
    \label{def:peps_ghz_tensor}
    \lean{TNLean.PEPS.ghzSiteTensor, TNLean.PEPS.ghzPEPS}
    \leanok
    \uses{def:peps_torus_site_tensor}
    The GHZ tensor has $D=d$ and
    $A^i_{\alpha\beta\gamma\delta}=\delta_{i=\alpha=\beta=\gamma=\delta}$
    \cite[Appendix~A, ``The GHZ state'']{Cirac2021Matrix}.
\end{definition}

\begin{lemma}[The GHZ PEPS is the GHZ state]%
    \label{thm:peps_ghz_state}
    \lean{TNLean.PEPS.sum_prod_ghzSiteTensor, TNLean.PEPS.stateCoeff_ghzPEPS}
    \leanok
    \uses{def:peps_ghz_tensor, def:peps_stateCoeff,
        lem:peps_torus_site_tensor_contraction}
    For $w,h\ge1$ and the GHZ tensor $A$ (Definition~\ref{def:peps_ghz_tensor}),
    \begin{align}
        \sum_{h,v}\prod_{(x,y)}
        A^{\sigma(x,y)}_{v_{(x,y)},h_{(x,y)},v_{(x,y-1)},h_{(x-1,y)}}
         & = \sum_{i=0}^{d-1}\prod_{(x,y)}\delta_{\sigma(x,y),i}
        \label{eq:peps_ghz_periodic_contraction}
    \end{align}
    \cite[Appendix~A, ``The GHZ state'']{Cirac2021Matrix}, and for $w,h\ge2$
    the simple-graph GHZ PEPS has coefficient (\ref{eq:peps_ghz_periodic_contraction}).
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_shift_const}
    A summand in (\ref{eq:peps_ghz_periodic_contraction}) is nonzero iff
    $v_{(x,y)}=h_{(x,y)}=v_{(x,y-1)}=h_{(x-1,y)}=\sigma(x,y)$ for all $(x,y)$,
    which by Lemma~\ref{lem:peps_torus_shift_const} holds iff
    $\sigma(x,y)=h_{(x,y)}=v_{(x,y)}=i$ for a single constant $i$.
\end{proof}

\begin{lemma}[Non-injectivity of the GHZ PEPS]
    \label{thm:peps_ghz_not_injective}
    \lean{TNLean.PEPS.ghzSiteTensor_not_linearIndependent,
        TNLean.PEPS.ghzPEPS_not_isVertexInjective}
    \leanok
    \uses{def:peps_ghz_tensor, def:IsVertexInjective}
    For $d\ge2$, the four-leg GHZ tensor is not injective, and for $d\ge2$,
    $w,h\ge2$, the simple-graph GHZ PEPS is not vertex-injective.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction}
    Since $0\neq 1$, $A^i_{0100}=0$ for all $i$; on the simple graph ($w,h\ge 2$),
    the top and right edges at a vertex are distinct, so the virtual basis
    vector with label $1$ on the right bond and $0$ elsewhere lies in $\ker A_v$.
\end{proof}

\begin{lemma}[On-site permutations and their virtual action]
    \label{thm:peps_ghz_permutation}
    \lean{TNLean.PEPS.ghzSiteTensor_perm, TNLean.PEPS.ghzSiteTensor_add,
        TNLean.PEPS.sum_prod_ghzSiteTensor_perm,
        TNLean.PEPS.stateCoeff_ghzPEPS_perm, TNLean.PEPS.stateCoeff_ghzPEPS_add}
    \leanok
    \uses{def:peps_ghz_tensor, def:peps_stateCoeff, thm:peps_ghz_state}
    Every permutation $\pi\in S_d$ satisfies
    \begin{align}
        A^{\pi(i)}_{\pi(\alpha)\pi(\beta)\pi(\gamma)\pi(\delta)}
         & = A^i_{\alpha\beta\gamma\delta},
        \label{eq:peps_ghz_perm_cov}
    \end{align}
    leaves (\ref{eq:peps_ghz_periodic_contraction}) invariant under
    $\sigma\mapsto\pi\circ\sigma$ for $w,h\ge 1$, and gives
    $\Coeff(\pi\circ\sigma)=\Coeff(\sigma)$ for $w,h\ge 2$ (including the
    $\Z_d$ shift $\pi(i)=i+g$).
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_ghz_state}
    Bijectivity of $\pi$ gives (\ref{eq:peps_ghz_perm_cov}); reindexing $i\mapsto\pi(i)$
    in (\ref{eq:peps_ghz_periodic_contraction}) of Lemma~\ref{thm:peps_ghz_state}
    gives $\Coeff(\pi\circ\sigma)=\Coeff(\sigma)$.
\end{proof}

\section{The cluster state}%
\label{sec:peps_cluster}

\begin{definition}[Two-dimensional cluster tensor]%
    \label{def:peps_cluster_tensor}
    \lean{TNLean.PEPS.clusterSiteTensor, TNLean.PEPS.clusterPEPS}
    \leanok
    \uses{def:peps_torus_site_tensor}
    The cluster tensor is $A=\ket{0}(00{+}{+}|+\ket{1}(11{-}{-}|$ with
    $(\pm|=[(0|\pm(1|]/\sqrt{2}$, so
    $A^s_{\alpha\beta\gamma\delta}=\tfrac12\delta_{s,\alpha}\delta_{s,\beta}(-1)^{s(\gamma+\delta)}$
    \cite[Appendix~A, ``The cluster state'']{Cirac2021Matrix}.
\end{definition}

\begin{lemma}[The cluster PEPS]%
    \label{thm:peps_cluster_state}
    \lean{TNLean.PEPS.stateCoeff_clusterPEPS}
    \leanok
    \uses{def:peps_cluster_tensor, def:peps_stateCoeff}
    For $w,h\ge 3$ and $N=wh$, the cluster PEPS
    (Definition~\ref{def:peps_cluster_tensor}) has coefficient
    \begin{align}
        \Coeff(\sigma)=2^{-N}\prod_{v}(-1)^{\sigma_v\sigma_{v+(1,0)}
                +\sigma_v\sigma_{v+(0,1)}}.
        \label{eq:peps_cluster_coeff}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction}
    In (\ref{eq:peps_torus_site_contraction}) of
    Lemma~\ref{lem:peps_torus_site_tensor_contraction}, $\delta_{s,\alpha}\delta_{s,\beta}$
    pins $v_v=h_v=\sigma_v$, leaving
    $\prod_v\tfrac12(-1)^{\sigma_v\sigma_{v-(0,1)}+\sigma_v\sigma_{v-(1,0)}}$;
    reindexing $v$ gives (\ref{eq:peps_cluster_coeff}).
\end{proof}

\begin{definition}[Graph-state circuit and stabilizers]
    \label{def:peps_graph_stabilizers}
    \lean{TNLean.PEPS.graphCircuitSign, TNLean.PEPS.graphFlip,
        TNLean.PEPS.graphNeighborSign, TNLean.PEPS.graphStabilizer,
        TNLean.PEPS.clusterBondTarget, TNLean.PEPS.clusterCircuitState,
        TNLean.PEPS.clusterStabilizer}
    \leanok
    \uses{def:peps_torus_geometry_translation}
    For a finite family of loop-free bonds $e$ with endpoints $a(e)\neq b(e)\in V$,
    set $g(\sigma)=\prod_e(-1)^{\sigma_{a(e)}\sigma_{b(e)}}$ and define the
    stabilizer $K_v$ by
    \begin{align}
        (K_v\psi)(\sigma)
         & =\prod_{e:a(e)=v}(-1)^{\sigma_{b(e)}}
        \prod_{e:b(e)=v}(-1)^{\sigma_{a(e)}}
        \psi(\sigma\mathbin{\oplus}\mathbf{1}_v).
        \label{eq:peps_graph_stabilizer_def}
    \end{align}
    On the $w\times h$ torus ($N=wh$), the normalized circuit state has
    coefficients $2^{-N/2}g(\sigma)$
    \cite[Appendix~A, ``The cluster state'']{Cirac2021Matrix}.
\end{definition}

\begin{lemma}[Common graph-stabilizer eigenspace]
    \label{thm:peps_graph_stabilizer_line}
    \lean{TNLean.PEPS.graphStabilizer_graphCircuitSign,
        TNLean.PEPS.graphStabilizer_fixed_iff}
    \leanok
    \uses{def:peps_graph_stabilizers}
    For the loop-free bond family of Definition~\ref{def:peps_graph_stabilizers},
    $K_v\psi=\psi$ for all $v\in V$ if and only if $\psi=cg$ for some $c\in\C$.
\end{lemma}

\begin{proof}\leanok
    With $s_v(\sigma)=\prod_{e:a(e)=v}(-1)^{\sigma_{b(e)}}\prod_{e:b(e)=v}(-1)^{\sigma_{a(e)}}$,
    $g(\sigma\oplus\mathbf{1}_v)=s_v(\sigma)g(\sigma)$ and $s_v(\sigma)^2=1$, so
    $K_v\psi=\psi$ is equivalent to
    $g(\sigma\oplus\mathbf{1}_v)\psi(\sigma\oplus\mathbf{1}_v)=g(\sigma)\psi(\sigma)$.
    Inducting on the Hamming weight of $\sigma$ gives
    $g(\sigma)\psi(\sigma)=\psi(0)$, i.e.\ $\psi=\psi(0)g$.
\end{proof}

\begin{lemma}[Square-torus cluster circuit and uniqueness]
    \label{thm:peps_cluster_stabilizer_line}
    \lean{TNLean.PEPS.stateCoeff_clusterPEPS_eq_smul_graphCircuitSign,
        TNLean.PEPS.stateCoeff_clusterPEPS_eq_smul_clusterCircuitState,
        TNLean.PEPS.clusterStabilizer_apply,
        TNLean.PEPS.clusterStabilizer_fixed_iff,
        TNLean.PEPS.stateCoeff_clusterPEPS_ne_zero}
    \leanok
    \uses{def:peps_cluster_tensor, def:peps_graph_stabilizers}
    On a torus with $w,h\ge3$ and $N=wh$, the cluster PEPS satisfies
    $\Coeff(\sigma)=2^{-N}g(\sigma)\neq 0$, and $\C\cdot\Coeff$ is the common
    $+1$ eigenspace of
    \begin{align}
        K_v=X_v Z_{v+(1,0)}Z_{v-(1,0)}Z_{v+(0,1)}Z_{v-(0,1)}.
        \label{eq:peps_cluster_stabilizer_pauli}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_cluster_state, thm:peps_graph_stabilizer_line}
    Combine Lemma~\ref{thm:peps_cluster_state} ($\Coeff(\sigma)=2^{-N}g(\sigma)$,
    $\Coeff(0)=2^{-N}\neq 0$) with Lemma~\ref{thm:peps_graph_stabilizer_line}.
\end{proof}

\section{The CZX model}%
\label{sec:peps_czx}

In the CZX model \cite{ChenLiuWen2011CZX}, each site carries four corner qubits
$\ket{ijkl}$ (top-left, top-right, bottom-right, bottom-left) and the ground
state is the product over plaquettes of $\ket{0000}+\ket{1111}$, written as a
bond-dimension-four PEPS in \cite[Appendix~A, ``The CZX model'']{Cirac2021Matrix}.

\begin{definition}[CZX tensor]%
    \label{def:peps_czx_tensor}
    \lean{TNLean.PEPS.czxBond, TNLean.PEPS.czxQubits, TNLean.PEPS.czxTopLeft,
        TNLean.PEPS.czxTopRight, TNLean.PEPS.czxBottomRight,
        TNLean.PEPS.czxBottomLeft, TNLean.PEPS.czxSiteTensor,
        TNLean.PEPS.czxBondSwap, TNLean.PEPS.czxBondSwap_czxBond,
        TNLean.PEPS.czxPEPS}
    \leanok
    \uses{def:peps_torus_site_tensor}
    The CZX site tensor is
    \begin{align}
        \sum_{i,j,k,l=0}^{1}\ket{ijkl}\bigl((i,j),(j,k),(k,l),(l,i)\bigr|
        \notag
    \end{align}
    \cite[Appendix~A, ``The CZX model'']{Cirac2021Matrix}. The CZX PEPS places
    this tensor at every site with the down and left bond pairs reversed
    ($(a,b)\leftrightarrow(b,a)$) so that each bond matches the two corner
    qubits of its shared side \cite{gap:rmp_peps_czx_bond_orientation}.
\end{definition}

\begin{lemma}[The CZX PEPS is a product of plaquette GHZ states]%
    \label{thm:peps_czx_plaquette_product}
    \lean{TNLean.PEPS.czx_bond_iff_plaquette, TNLean.PEPS.stateCoeff_czxPEPS}
    \leanok
    \uses{def:peps_czx_tensor, def:peps_stateCoeff}
    For $w,h\ge 3$ and $\sigma(v)=(i_v,j_v,k_v,l_v)$, the CZX PEPS
    (Definition~\ref{def:peps_czx_tensor}) has coefficient
    \begin{align}
        \Coeff(\sigma)=\prod_{v}\delta\bigl[j_v=i_{v+(1,0)}=k_{v+(0,1)}
            =l_{v+(1,1)}\bigr]
        \label{eq:peps_czx_plaquette_coeff}
    \end{align}
    \cite[Appendix~A, ``The CZX model'']{Cirac2021Matrix},
    \cite{ChenLiuWen2011CZX}.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction}
    In (\ref{eq:peps_torus_site_contraction}) of
    Lemma~\ref{lem:peps_torus_site_tensor_contraction}, the top and right legs
    pin $v_v=(i_v,j_v)$ and $h_v=(j_v,k_v)$, and the reversed down and left legs
    impose $(i_{v-(0,1)},j_{v-(0,1)})=(l_v,k_v)$ and
    $(j_{v-(1,0)},k_{v-(1,0)})=(i_v,l_v)$, which reindex to
    (\ref{eq:peps_czx_plaquette_coeff}).
\end{proof}

\begin{definition}[CZX on-site symmetry]%
    \label{def:peps_czx_on_site}
    \lean{TNLean.PEPS.onSiteOperator, TNLean.PEPS.onSiteOperator_apply,
        TNLean.PEPS.onSiteOperator_monomial, TNLean.PEPS.czxSiteFlip,
        TNLean.PEPS.czxTopLeft_czxSiteFlip, TNLean.PEPS.czxTopRight_czxSiteFlip,
        TNLean.PEPS.czxBottomRight_czxSiteFlip,
        TNLean.PEPS.czxBottomLeft_czxSiteFlip, TNLean.PEPS.czxSiteFlip_czxQubits,
        TNLean.PEPS.czxSiteFlip_czxSiteFlip, TNLean.PEPS.czxSiteFlip_symm,
        TNLean.PEPS.czxCZExponent, TNLean.PEPS.czxOnSite,
        TNLean.PEPS.czxOnSite_eq_monomial}
    \leanok
    \uses{def:peps_czx_tensor}
    The CZX on-site $\Z_2$ generator $U_{CZX}=U_XU_{CZ}$ with
    $U_X=X_1X_2X_3X_4$ and $U_{CZ}=CZ_{12}CZ_{23}CZ_{34}CZ_{41}$ acts by
    $U_{CZX}\ket{ijkl}=(-1)^{ij+jk+kl+li}\ket{\bar\imath\bar\jmath\bar k\bar l}$,
    and $U_{CZX}^{\otimes V}$ has entries
    $(U_{CZX}^{\otimes V})_{\sigma',\sigma}=\prod_{v\in V}(U_{CZX})_{\sigma'_v\sigma_v}$
    \cite{ChenLiuWen2011CZX}.
\end{definition}

\begin{theorem}[The CZX PEPS is invariant under the on-site symmetry]%
    \label{thm:peps_czx_invariant}
    \lean{TNLean.PEPS.prod_neg_one_pow_czxCZExponent,
        TNLean.PEPS.onSiteOperator_czxOnSite_mulVec_stateCoeff_czxPEPS}
    \leanok
    \uses{def:peps_czx_on_site, thm:peps_czx_plaquette_product}
    For $w,h\ge 3$, $U_{CZX}^{\otimes V}\Coeff=\Coeff$
    \cite[Appendix~A, ``The CZX model'']{Cirac2021Matrix},
    \cite{ChenLiuWen2011CZX}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_czx_plaquette_product}
    Bitwise negation preserves (\ref{eq:peps_czx_plaquette_coeff}). On any
    configuration with plaquette values $P_v=j_v=i_{v+(1,0)}=k_{v+(0,1)}=l_{v+(1,1)}$,
    the phase exponent at $v$ is
    $e_v=P_{v-(1,0)}P_v+P_vP_{v-(0,1)}+P_{v-(0,1)}P_{v-(1,1)}+P_{v-(1,1)}P_{v-(1,0)}$;
    shifting $v$ pairs the third and fourth sums with the first and second, so
    $\sum_v e_v\equiv 0\pmod 2$.
\end{proof}

\begin{definition}[CZX leg operator]%
    \label{def:peps_czx_leg_operator}
    \lean{TNLean.PEPS.czxLegFlip, TNLean.PEPS.czxLegFlip_czxBond,
        TNLean.PEPS.czxLegPhase, TNLean.PEPS.czxLegPhase_czxBond,
        TNLean.PEPS.czxLegOperator}
    \leanok
    \uses{def:peps_czx_tensor}
    On a two-qubit virtual leg, $V=(X\otimes X)\,CZ$ acts by
    $V\ket{ab}=(-1)^{ab}\ket{\bar a\bar b}$.
\end{definition}

\begin{lemma}[Pulling the symmetry through a site]%
    \label{thm:peps_czx_pull_through}
    \lean{TNLean.PEPS.czxOnSite_mul_czxSiteTensor}
    \leanok
    \uses{def:peps_czx_on_site, def:peps_czx_leg_operator}
    For the CZX site tensor $A^s_{trbl}$
    \cite[Section~III.B, ``Non-injective PEPS: SPT phases'']{Cirac2021Matrix},
    \begin{align}
        \sum_s(U_{CZX})_{s's}\,A^s_{trbl}
         & =\sum_{t',r',b',l'}A^{s'}_{t'r'b'l'}\,V_{t't}V_{r'r}V_{b'b}V_{l'l}.
        \label{eq:peps_czx_pull_through}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    For $s=(i,j,k,l)$, both sides are supported on the bitwise-negated indices
    and carry the sign $(-1)^{ij+jk+kl+li}=(-1)^{ij}(-1)^{jk}(-1)^{kl}(-1)^{li}$.
\end{proof}

\begin{definition}[Boundary legs of the CZX tensor network]%
    \label{def:peps_czx_boundary_embedding}
    \lean{TNLean.PEPS.czxBoundaryLegs, TNLean.PEPS.czxBoundaryEmbedding}
    \leanok
    \uses{def:peps_czx_tensor}
    For $N>0$, the boundary embedding $\iota:(\C^2)^{\otimes N}\to(\C^4)^{\otimes N}$
    is defined on $c\in\{0,1\}^N$ (indices modulo $N$) by
    \begin{align}
        \iota\ket{c} & =\ket{(c_0,c_1),(c_1,c_2),\dots,(c_{N-1},c_0)}.
        \label{eq:peps_czx_boundary_embedding}
    \end{align}
\end{definition}

\begin{lemma}[The boundary action is the CZX matrix product unitary]%
    \label{thm:peps_czx_boundary_mpu}
    \lean{TNLean.PEPS.czxBoundaryLegs_injective,
        TNLean.PEPS.czxBoundaryEmbedding_conjTranspose_mul_self,
        TNLean.PEPS.onSiteOperator_czxLegOperator_mul_czxBoundaryEmbedding,
        TNLean.PEPS.czxBoundaryEmbedding_conjTranspose_mul_onSiteOperator_mul,
        TNLean.PEPS.onSiteOperator_czxLegOperator_mul_czxBoundaryEmbedding_review}
    \leanok
    \uses{def:peps_czx_leg_operator, def:peps_czx_boundary_embedding,
        thm:asymex_czx_kernel, def:asymex_czx_review_tensor}
    For every $N>0$ and $U_N\ket{c}=(-1)^{\sum_m c_m c_{m+1}}\ket{\bar c}$
    (Lemma~\ref{thm:asymex_czx_kernel}), $\iota$ and $V$
    (Definitions~\ref{def:peps_czx_leg_operator}--\ref{def:peps_czx_boundary_embedding})
    satisfy
    \begin{align}
        \iota^\dagger\iota               & = \Id,                            &
        V^{\otimes N}\iota               & = \iota\,U_N = (-1)^N\iota\,O(A), &
        \iota^\dagger V^{\otimes N}\iota & = U_N
        \notag
    \end{align}
    \cite[Appendix~A, ``The CZX MPU'']{Cirac2021Matrix},
    \cite{ChenLiuWen2011CZX}, where $O(A)$ is the review operator of
    Definition~\ref{def:asymex_czx_review_tensor} (see
    Section~\ref{sec:peps_czx_rectangle_boundary} and
    \cite{gap:rmp_peps_czx_boundary_chain} for the rectangular-region boundary
    reduction).
\end{lemma}

\begin{proof}\leanok
    \uses{thm:asymex_czx_kernel, thm:asymex_czx_review_kernel}
    Injectivity of $c\mapsto\bigl((c_m,c_{m+1})\bigr)_{m=0}^{N-1}$ gives
    $\iota^\dagger\iota=\Id$; applying $V$ to each pair $(c_m,c_{m+1})$ produces
    $\prod_m(-1)^{c_mc_{m+1}}\iota\ket{\bar c}=\iota U_N\ket{c}$
    (Lemma~\ref{thm:asymex_czx_kernel}), and $U_N=(-1)^N O(A)$ by
    Lemma~\ref{thm:asymex_czx_review_kernel}.
\end{proof}

\section{The AKLT state}%
\label{sec:peps_aklt}

The two-dimensional AKLT state places singlets on the lattice edges and projects
the four spin-$\tfrac12$ at each site onto their symmetric subspace
\cite[Appendix~A, ``The AKLT model'']{Cirac2021Matrix}. For
$s=(s_0,s_1,s_2,s_3)\in\{0,1\}^4$, write $|s|=\sum_{j=0}^3 s_j$ and let
$P_{\pi}$ permute the four factors by $\pi\in S_4$.

\begin{definition}[Symmetric projector on four spin-$\tfrac12$]%
    \label{def:peps_aklt_sym_projector}
    \lean{TNLean.PEPS.legKronecker, TNLean.PEPS.legPermMatrix,
        TNLean.PEPS.symProjector}
    \leanok
    The projector onto the symmetric subspace of $(\C^2)^{\otimes 4}$ is
    $\Pi_{\mathrm{sym}}=\frac{1}{24}\sum_{\pi\in S_4}P_{\pi}$.
\end{definition}

\begin{lemma}[The symmetric subspace has dimension five]%
    \label{lem:peps_aklt_sym_projector}
    \lean{TNLean.PEPS.legKronecker_mul, TNLean.PEPS.legKronecker_one,
        TNLean.PEPS.legKronecker_transpose,
        TNLean.PEPS.legPermMatrix_mul, TNLean.PEPS.legPermMatrix_conjTranspose,
        TNLean.PEPS.legPermMatrix_mul_legKronecker,
        TNLean.PEPS.symProjector_mul_self, TNLean.PEPS.symProjector_conjTranspose,
        TNLean.PEPS.symProjector_mul_legKronecker, TNLean.PEPS.legWeight,
        TNLean.PEPS.card_perm_comp_eq, TNLean.PEPS.symProjector_apply,
        TNLean.PEPS.choose_legWeight_dvd, TNLean.PEPS.sum_div_choose_legWeight,
        TNLean.PEPS.trace_symProjector, TNLean.PEPS.finrank_range_symProjector}
    \leanok
    \uses{def:peps_aklt_sym_projector}
    $\Pi_{\mathrm{sym}}$ satisfies $\Pi_{\mathrm{sym}}^2=\Pi_{\mathrm{sym}}=\Pi_{\mathrm{sym}}^\dagger$,
    $\Pi_{\mathrm{sym}}U^{\otimes 4}=U^{\otimes 4}\Pi_{\mathrm{sym}}$ for all
    $U\in\MN{2}$, $(\Pi_{\mathrm{sym}})_{s,t}=\binom{4}{|s|}^{-1}\delta_{|s|,|t|}$,
    and $\dim\operatorname{im}\Pi_{\mathrm{sym}}=\tr\Pi_{\mathrm{sym}}=5$.
\end{lemma}

\begin{proof}\leanok
    Since $P_{\pi}P_{\rho}=P_{\rho\pi}$, $P_{\pi}^{\dagger}=P_{\pi^{-1}}$, and
    $P_{\pi}U^{\otimes 4}=U^{\otimes 4}P_{\pi}$, the average $\Pi_{\mathrm{sym}}$
    is a Hermitian idempotent commuting with $U^{\otimes 4}$. Moreover,
    $|\{\pi\in S_4:s=t\circ\pi\}|=24\binom{4}{|s|}^{-1}\delta_{|s|,|t|}$, and
    $\dim\operatorname{im}\Pi_{\mathrm{sym}}=\tr\Pi_{\mathrm{sym}}=\sum_{k=0}^{4}\binom4k\binom4k^{-1}=5$.
\end{proof}

\begin{definition}[AKLT tensor]%
    \label{def:peps_aklt_tensor}
    \lean{TNLean.PEPS.akltLegs,
        TNLean.PEPS.akltSiteTensorFun, TNLean.PEPS.akltSiteTensor,
        TNLean.PEPS.akltPEPS}
    \leanok
    \uses{def:peps_aklt_sym_projector, def:peps_torus_site_tensor,
        def:aklt_tensor_review}
    With $Y=\left(\begin{smallmatrix}0&-1\\1&0\end{smallmatrix}\right)$ and
    $L=\mathbb{1}\otimes\mathbb{1}\otimes Y\otimes Y$, the AKLT tensor on the
    square lattice is $A=\Pi_{\mathrm{sym}}L$, with physical index
    $s\in\{0,1\}^4$ and virtual indices ordered top, right, down, left
    \cite[Appendix~A, ``The AKLT model'']{Cirac2021Matrix}.
\end{definition}

\begin{lemma}[Entries and physical span of the AKLT tensor]%
    \label{lem:peps_aklt_entries}
    \lean{Matrix.singletY_apply, Matrix.singletY_mul_transpose,
        TNLean.PEPS.akltLegs_apply,
        TNLean.PEPS.legWeight_vec, TNLean.PEPS.akltSiteTensorFun_apply,
        TNLean.PEPS.akltLegs_mul_transpose, TNLean.PEPS.span_akltSiteTensorFun}
    \leanok
    \uses{def:peps_aklt_tensor, lem:peps_aklt_sym_projector}
    The AKLT tensor (Definition~\ref{def:peps_aklt_tensor}) has entries
    \begin{align}
        A^{s}_{\alpha\beta\gamma\delta}=
        \frac{(-1)^{\gamma+\delta}}{\binom{4}{k}}\,\delta_{|s|,k},\qquad
        k=\alpha+\beta+(1-\gamma)+(1-\delta),
        \notag
    \end{align}
    and $\spn\{s\mapsto A^s_{\alpha\beta\gamma\delta}\}_{\alpha,\beta,\gamma,\delta}
        =\operatorname{im}\Pi_{\mathrm{sym}}$.
\end{lemma}

\begin{proof}\leanok
    Since $Y\ket{b}=(-1)^b\ket{1-b}$, $L\ket{\alpha\beta\gamma\delta}
        =(-1)^{\gamma+\delta}\ket{\alpha,\beta,1-\gamma,1-\delta}$, and since
    $LL^T=\mathbb{1}$, $\operatorname{im}(\Pi_{\mathrm{sym}}L)=\operatorname{im}\Pi_{\mathrm{sym}}$.
\end{proof}

\begin{lemma}[$SU(2)$ covariance of the AKLT tensor]%
    \label{thm:peps_aklt_su2}
    \lean{Matrix.mul_singletY, TNLean.PEPS.akltSiteTensor_covariant,
        Matrix.adjugate_eq_star_of_mem_specialUnitaryGroup,
        TNLean.PEPS.akltSiteTensor_su2_covariant}
    \leanok
    \uses{def:peps_aklt_tensor, lem:peps_aklt_sym_projector}
    For every $U\in\MN{2}$ and $V=(\operatorname{adj}U)^T$ (with $V=\bar U$ when
    $U\in SU(2)$),
    \begin{align}
        U^{\otimes 4}A=A\,(U\otimes U\otimes V\otimes V).
        \notag
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $UY=YV$ for $V=(\operatorname{adj}U)^T$ and
    $U^{\otimes 4}\Pi_{\mathrm{sym}}=\Pi_{\mathrm{sym}}U^{\otimes 4}$
    (Lemma~\ref{lem:peps_aklt_sym_projector}),
    $U^{\otimes 4}\Pi_{\mathrm{sym}}L=\Pi_{\mathrm{sym}}L(U\otimes U\otimes V\otimes V)$;
    for $U\in SU(2)$, $\operatorname{adj}U=U^\dagger$, so $V=\bar U$.
\end{proof}

\begin{lemma}[The AKLT PEPS as projected singlets]%
    \label{thm:peps_aklt_state}
    \lean{TNLean.PEPS.akltLegs_vec, TNLean.PEPS.stateCoeff_akltPEPS}
    \leanok
    \uses{def:peps_aklt_tensor, lem:peps_torus_site_tensor_contraction}
    For $w,h\ge 3$, the AKLT PEPS (Definition~\ref{def:peps_aklt_tensor}) has
    coefficient
    \begin{align}
        \Coeff(\sigma)=\sum_{\tau:V\to\{0,1\}^4}\prod_{v}(\Pi_{\mathrm{sym}})_{\sigma_v,\tau_v}
        \prod_{v}Y_{\tau_{v+(1,0),3},\,\tau_{v,1}}\,Y_{\tau_{v+(0,1),2},\,\tau_{v,0}},
        \notag
    \end{align}
    representing $\bigotimes_v\Pi_{\mathrm{sym}}$ applied to a singlet
    $\ket{01}-\ket{10}$ on every edge
    \cite[Appendix~A, ``The AKLT model'']{Cirac2021Matrix}.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction}
    In (\ref{eq:peps_torus_site_contraction}) of
    Lemma~\ref{lem:peps_torus_site_tensor_contraction}, expanding
    $A=\Pi_{\mathrm{sym}}L$ over $\tau_v\in\{0,1\}^4$ pins $v_v=\tau_{v,0}$ and
    $h_v=\tau_{v,1}$, leaving
    $Y_{\tau_{v,2},\tau_{v-(0,1),0}}Y_{\tau_{v,3},\tau_{v-(1,0),1}}$; shifting
    $v$ gives the result.
\end{proof}

\section{The RVB state}%
\label{sec:peps_rvb}

The nearest-neighbour RVB PEPS of bond dimension $3$ combines
\begin{align}
    P=\ket{0}\bigl[(0222|+(2022|+(2202|+(2220|\bigr]
    +\ket{1}\bigl[(1222|+(2122|+(2212|+(2221|\bigr]
    \notag
\end{align}
with $Y$ extended by $1$ on the vacuum state $2$ (so the bond state is
$\ket{01}-\ket{10}+\ket{22}$; extending by $0$ would make $A=0$,
\cite{gap:rmp_peps_rvb_bond_vacuum})
\cite[Appendix~A, ``The RVB state'']{Cirac2021Matrix}.

\begin{definition}[RVB tensor]%
    \label{def:peps_rvb_tensor}
    \lean{TNLean.PEPS.rvbSingletY, TNLean.PEPS.rvbProjector,
        TNLean.PEPS.rvbSiteTensorInt, TNLean.PEPS.rvbSiteTensor,
        TNLean.PEPS.rvbPEPS}
    \leanok
    \uses{def:peps_torus_site_tensor}
    The RVB tensor is $A=P(\mathbb{1}\otimes\mathbb{1}\otimes Y\otimes Y)$ with
    $Y=\left(\begin{smallmatrix}0&-1&0\\1&0&0\\0&0&1\end{smallmatrix}\right)$,
    virtual indices ordered top, right, down, left.
\end{definition}

\begin{lemma}[Local dimer constraint]%
    \label{lem:peps_rvb_local}
    \lean{TNLean.PEPS.rvbFlip, TNLean.PEPS.rvbSiteTensorInt_eq,
        TNLean.PEPS.rvbSiteTensor_apply, TNLean.PEPS.rvbSiteTensor_ne_zero_iff,
        TNLean.PEPS.rvbBondLabel, TNLean.PEPS.rvbBondLabel_ne_two_iff,
        TNLean.PEPS.rvbSiteTensorInt_ne_zero,
        TNLean.PEPS.rvbSiteTensorInt_bondLabel}
    \leanok
    \uses{def:peps_rvb_tensor}
    $A^{s}_{urdl}\neq 0$ if and only if $(u,r,d,l)$ has a unique entry $\neq 2$,
    equal to $s$ if in $\{u,r\}$ and to $1-s$ if in $\{d,l\}$; when nonzero,
    $A^s_{urdl}=-1$ if $d=1$ or $l=1$ and $+1$ otherwise.
\end{lemma}

\begin{proof}\leanok
    Follows from $Y\ket{b}=(-1)^b\ket{1-b}$ ($b\in\{0,1\}$), $Y\ket{2}=\ket{2}$,
    and finite case analysis over $\{0,1\}\times\{0,1,2\}^4$.
\end{proof}

\begin{lemma}[The RVB PEPS is the superposition of singlet coverings]%
    \label{thm:peps_rvb_state}
    \lean{TNLean.PEPS.IsTorusDimerCovering, TNLean.PEPS.rvbCoveringBonds,
        TNLean.PEPS.stateCoeff_rvbPEPS}
    \leanok
    \uses{def:peps_rvb_tensor, def:peps_stateCoeff}
    For $w,h\ge3$, let $\mathcal D$ be the set of dimer coverings $c\subseteq E$
    (every site $v$ lies on a unique edge $(v,v+e)\in c$, $e\in\{(1,0),(0,1)\}$).
    The RVB PEPS (Definition~\ref{def:peps_rvb_tensor}) has coefficient
    \begin{align}
        \Coeff(\sigma)=\sum_{c\in\mathcal D}\ \prod_{(v,v+e)\in c}Y_{\sigma_{v+e},\sigma_v}
        \notag
    \end{align}
    \cite[Appendix~A, ``The RVB state'']{Cirac2021Matrix}.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction, lem:peps_rvb_local}
    In (\ref{eq:peps_torus_site_contraction}) of
    Lemma~\ref{lem:peps_torus_site_tensor_contraction},
    Lemma~\ref{lem:peps_rvb_local} bijects nonzero bond labellings $(h_\bullet,v_\bullet)$
    with dimer coverings $c\in\mathcal D$ via the edges with label $\neq 2$,
    where $(v,v+e)\in c$ carries label $\sigma_v$ and contributes
    $Y_{\sigma_{v+e},\sigma_v}$.
\end{proof}

\section{PEPS from classical models}%
\label{sec:peps_classical_models}

Let $q\ge 1$, $h\colon\{0,\ldots,q-1\}^{2}\to\mathbb{R}$, and $\beta\in\mathbb{R}$.

\begin{definition}[Classical energy]%
    \label{def:peps_classical_energy}
    \lean{TNLean.PEPS.classicalEnergy}
    \leanok
    On the torus, a configuration $\sigma:V\to\{0,\ldots,q-1\}$ has energy
    \begin{align}
        H(\sigma)=\sum_{v}\bigl(h(\sigma_{v},\sigma_{v-(1,0)})
        +h(\sigma_{v},\sigma_{v-(0,1)})\bigr).
        \label{eq:peps_classical_energy_def}
    \end{align}
\end{definition}

\begin{definition}[Classical-model tensor]%
    \label{def:peps_classical_tensor}
    \lean{TNLean.PEPS.classicalBondMatrix, TNLean.PEPS.classicalSiteTensor,
        TNLean.PEPS.classicalPEPS}
    \leanok
    \uses{def:peps_torus_site_tensor}
    With $M=\sum_{i,j}e^{-\beta h(i,j)/2}|i)(j|$, the classical-model tensor is
    \begin{align}
        A=\Bigl[\sum_{i}\ket{i}(i\,i\,i\,i|\Bigr]
        (\mathbb{1}\otimes\mathbb{1}\otimes M\otimes M),
        \qquad
        A^s_{\alpha\beta\gamma\delta}=\delta_{s,\alpha}\delta_{s,\beta}M_{s\gamma}M_{s\delta}
        \notag
    \end{align}
    \cite[Appendix~A, ``PEPS from classical models'']{Cirac2021Matrix}.
\end{definition}

\begin{lemma}[Coefficients of the classical-model PEPS]%
    \label{thm:peps_classical_coeff}
    \lean{TNLean.PEPS.stateCoeff_classicalPEPS,
        TNLean.PEPS.conj_mul_stateCoeff_classicalPEPS,
        TNLean.PEPS.normSq_stateCoeff_classicalPEPS}
    \leanok
    \uses{def:peps_classical_tensor, def:peps_classical_energy,
        def:peps_stateCoeff}
    For $w,h\ge 3$, the classical-model PEPS
    (Definition~\ref{def:peps_classical_tensor}) has coefficient
    $\Coeff(\sigma)=e^{-\beta H(\sigma)/2}$ and Gibbs weight
    $|\Coeff(\sigma)|^{2}=e^{-\beta H(\sigma)}$
    \cite{gap:rmp_peps_examples_small_torus}.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_torus_site_tensor_contraction}
    In (\ref{eq:peps_torus_site_contraction}) of
    Lemma~\ref{lem:peps_torus_site_tensor_contraction},
    $\delta_{s,\alpha}\delta_{s,\beta}$ pins $v_v=h_v=\sigma_v$, leaving
    $\prod_v M_{\sigma_v\sigma_{v-(0,1)}}M_{\sigma_v\sigma_{v-(1,0)}}=e^{-\beta H(\sigma)/2}$
    by (\ref{eq:peps_classical_energy_def}).
\end{proof}

\begin{lemma}[Diagonal expectation values are Gibbs averages]%
    \label{thm:peps_classical_expectation}
    \lean{TNLean.PEPS.expectation_classicalPEPS}
    \leanok
    \uses{def:peps_classical_tensor, def:peps_classical_energy,
        def:peps_stateCoeff}
    For $w,h\ge 3$, a diagonal observable
    $O=\sum_{\sigma}O(\sigma)\ket{\sigma}\bra{\sigma}$, and
    $\ket{\psi}=\sum_{\sigma}\Coeff(\sigma)\ket{\sigma}$,
    \begin{align}
        \frac{\sum_{\sigma}\overline{\Coeff(\sigma)}\,O(\sigma)\Coeff(\sigma)}
        {\sum_{\sigma}\overline{\Coeff(\sigma)}\Coeff(\sigma)}
        =\frac{\sum_{\sigma}O(\sigma)e^{-\beta H(\sigma)}}
        {\sum_{\sigma}e^{-\beta H(\sigma)}}
        \notag
    \end{align}
    \cite[Appendix~A, ``PEPS from classical models'']{Cirac2021Matrix}.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_classical_coeff}
    Substitute $\overline{\Coeff(\sigma)}\Coeff(\sigma)=e^{-\beta H(\sigma)}$
    from Lemma~\ref{thm:peps_classical_coeff}.
\end{proof}

At the critical inverse temperature $\beta_c$, power-law decay of classical
correlations implies that any local parent Hamiltonian of $\ket{\psi}$ is
gapless \cite[Appendix~A, ``PEPS from classical models'']{Cirac2021Matrix},
\cite{Hastings2006Spectral,Nachtergaele2006LiebRobinson}.
