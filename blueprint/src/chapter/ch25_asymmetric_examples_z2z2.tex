\section{The anomalous $\Z_2\times\Z_2$ symmetry with a mixed three-cocycle}%
\label{sec:asymex_z2z2}

On a periodic chain of $L$ sites, each carrying two qubits with label
$a=(a_1,a_2)$, four commuting-square matrix product operators
$U_e=\id$, $U_x$, $U_y$, $U_{xy}$ represent the group $\Z_2\times\Z_2$ up
to phases: $U_gU_h=\lambda(g,h)^LU_{gh}$ with
$\lambda(g,h)=(-1)^{g_1h_2}$. The tensor of $U_g$ is
\begin{align}
    M_g^{(o,i)}[\alpha,\beta]
     & =\delta_{o,g+i}\,\delta_{\beta,\pi_g(i)}\,f_g(\alpha,i),
    \label{eq:asymex_z2z2_tensor}
\end{align}
with bond label $\pi_g(i)=i_2$ and phase $f_g(\alpha,i)=(-1)^{\alpha
            i_2}$ for $g\in\{x,xy\}$, and with bond dimension one and no phase for
$g\in\{e,y\}$. Every entry lies in $\{0,\pm1\}$, so each tensor is the
entrywise coercion of an integer matrix family. Restricted to the second
qubit, $M_x$ is exactly the bond-two symmetry tensor of
Section~\ref{sec:asymex_czx}, and $M_{xy}$ is $M_x$ after the letter
relabelling $o\mapsto o+y$.

This presentation carries the mixed class of $H^3(\Z_2\times\Z_2,U(1))$:
its restriction to $x$ and to $y$ is trivial and its restriction to the
diagonal element $xy$ is the class of Section~\ref{sec:asymex_czx}. No
cocycle class is computed in this development. What is formalized below is
the normality of the four tensors, the periodic fusion table of the four
operators, the length-dependent operator identity for their sum, and the
compression data of the sixteen pair blocks of the stacked square of that
sum; the three-cocycle itself, and the classification of which of the
sixteen pair blocks is not split, are read off from these compression data
without invoking any cohomological computation
(\texttt{Notes/OpenProblemsTN/checks/asym\_z2z2\_typeIII\_data.md},
Sections~1--3).

\begin{definition}[The four symmetry tensors]%
    \label{def:asymex_z2z2_tensors}
    \lean{Z2Z2Condensation.eTensor, Z2Z2Condensation.yTensor,
        Z2Z2Condensation.xTensor, Z2Z2Condensation.xyTensor,

        Z2Z2Condensation.xyMPS}
    \leanok
    \uses{def:mpo_tensor, def:mpo_to_mps}
    The tensors of $U_e$ and $U_y$ have bond dimension one and the four
    nonzero letters $\delta_{o,i}$ and $\delta_{o,y+i}$; the tensors of
    $U_x$ and $U_{xy}$ have bond dimension two and the four nonzero letters
    of (\ref{eq:asymex_z2z2_tensor}), each equal to
    $\begin{pmatrix}1&0\\1&0\end{pmatrix}$ or
    $\begin{pmatrix}0&1\\0&-1\end{pmatrix}$ according to $i_2$. Every
    tensor is also recorded over the sixteen-letter pair alphabet
    $a=4o+i$.
\end{definition}

\begin{theorem}[Normality of the four symmetry tensors]%
    \label{thm:asymex_z2z2_normal}
    \lean{Z2Z2Condensation.eMPS_isNormal, Z2Z2Condensation.yMPS_isNormal,
        Z2Z2Condensation.xMPS_isNormal, Z2Z2Condensation.xyMPS_isNormal,
        Z2Z2Condensation.xMPS_isNBlkInjective_two,
        Z2Z2Condensation.xyMPS_isNBlkInjective_two,
        Z2Z2Condensation.xMPS_not_isNBlkInjective_one,
        Z2Z2Condensation.xyMPS_not_isNBlkInjective_one}
    \leanok
    \uses{def:asymex_z2z2_tensors, def:normal}
    The bond-one tensors of $U_e$ and $U_y$ are normal at blocking length
    one: their single nonzero letter at the diagonal or antidiagonal
    coordinate already acts by the nonzero scalar $1$. The bond-two
    tensors of $U_x$ and $U_{xy}$ are not normal at length one, since their
    four nonzero letters span only a two-dimensional space of matrices,
    but are normal at blocking length two: their length-two words span the
    full two-by-two matrix algebra.
\end{theorem}

\begin{proof}
    \leanok
    For the bond-one tensors, a single nonzero scalar letter already
    witnesses normality at length one. For the bond-two tensors, an
    explicit integer linear combination of four length-two words gives
    each of the four matrix units, a finite verification over integer
    matrices; a linear functional vanishing on both nonzero letters at
    length one shows the length-one span is only two-dimensional.
\end{proof}

\section{The condensation defect and its periodic fusion table}%
\label{sec:asymex_z2z2_defect}

The \emph{condensation defect} of the anomalous symmetry is the sum of the
four periodic operators, $A=U_e+U_x+U_y+U_{xy}$. It is generated by the
direct-sum tensor $N=M_e\oplus M_x\oplus M_y\oplus M_{xy}$ of
Definition~\ref{def:mpo_direct_sum}, of bond dimension six.

\begin{definition}[The condensation defect tensor]%
    \label{def:asymex_z2z2_defect}
    \lean{Z2Z2Condensation.condensationTensor,
        Z2Z2Condensation.mpo_condensationTensor}
    \leanok
    \uses{def:mpo_direct_sum, def:asymex_z2z2_tensors}
    The condensation defect tensor is the iterated direct sum
    $N=((M_e\oplus M_x)\oplus M_y)\oplus M_{xy}$ of bond dimension six.
    Its periodic operator is the sum of the four periodic operators of
    Definition~\ref{def:asymex_z2z2_tensors}: $\rho^{(L)}(N)=\rho^{(L)}(M_e)
        +\rho^{(L)}(M_x)+\rho^{(L)}(M_y)+\rho^{(L)}(M_{xy})$ at every system
    size.
\end{definition}

\begin{proof}
    \leanok
    \uses{thm:mpo_direct_sum_additive}
    Three applications of the additivity of
    Theorem~\ref{thm:mpo_direct_sum_additive}.
\end{proof}

\begin{theorem}[The periodic fusion table]%
    \label{thm:asymex_z2z2_fusion}
    \lean{Z2Z2Condensation.bondDim, Z2Z2Condensation.symTensor,
        Z2Z2Condensation.gmul, Z2Z2Condensation.fusionSign,
        Z2Z2Condensation.mpo_symTensor_mul,
        Z2Z2Condensation.mpo_x_mul_y_eq_neg_of_odd}
    \leanok
    \uses{def:asymex_z2z2_tensors, lem:asym_mpo_trace_eq}
    Writing $T_g$ for the tensor of $U_g$ indexed by $g\in\Z_2\times\Z_2$
    with group law $gh$, and $\lambda(g,h)=(-1)^{g_1h_2}$, the periodic
    operators satisfy $\rho^{(L)}(T_g)\rho^{(L)}(T_h)=\lambda(g,h)^L
        \rho^{(L)}(T_{gh})$ for every $L\geq1$ and every pair $g,h$, with
    \begin{align}
        \lambda(g,h)
         & =\begin{cases}
                -1 & \text{if }g\in\{x,xy\}\text{ and }h\in\{y,xy\}, \\
                +1 & \text{otherwise.}
            \end{cases}
        \notag
    \end{align}
    On odd rings, $\lambda(x,y)\lambda(y,x)=-1$, so
    $\rho^{(L)}(T_x)\rho^{(L)}(T_y)=-\rho^{(L)}(T_y)\rho^{(L)}(T_x)$: the
    four operators form a projective representation of
    $\Z_2\times\Z_2$; on even rings the representation is linear.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:asym_mpo_trace_eq}
    For each of the sixteen pairs, an explicit finite check (over integer
    matrices, after clearing the pair-alphabet stacked tensor to a
    lookup table) identifies the positive-length word traces of the
    stacked product $T_g\cdot T_h$ with $\lambda(g,h)^{|w|}$ times those
    of $T_{gh}$; Lemma~\ref{lem:asym_mpo_trace_eq} converts this into the
    stated operator identity for every $L\geq1$. The odd-ring
    anticommutation follows by substituting $\lambda(x,y)=-1$ and
    $\lambda(y,x)=-1$ and using $(-1)^L=-1$.
\end{proof}

\begin{theorem}[The condensation defect squares to a length-dependent
        multiple of itself]%
    \label{thm:asymex_z2z2_defect_sq}
    \lean{Z2Z2Condensation.mpo_condensationTensor_sq,
        Z2Z2Condensation.mpo_condensationTensor_apply_zero,
        Z2Z2Condensation.mpo_condensationTensor_ne_zero,
        Z2Z2Condensation.condensation_quarter_sq_eq_self_iff_even}
    \leanok
    \uses{def:asymex_z2z2_defect, thm:asymex_z2z2_fusion}
    For every $L\geq1$, $\rho^{(L)}(N)^2=\bigl(3+(-1)^L\bigr)\rho^{(L)}(N)$,
    and $\rho^{(L)}(N)\neq0$. Consequently $\tfrac14\rho^{(L)}(N)$ is a
    projector if and only if $L$ is even: on odd rings the anomalous
    symmetry cannot be condensed by its defect.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_z2z2_fusion}
    Expanding the square by Theorem~\ref{thm:asymex_z2z2_fusion}, each of
    the four operators $\rho^{(L)}(T_k)$ receives the four weights
    $\lambda(g,h)^L$ from the four pairs $(g,h)$ with $gh=k$, three of
    which are $+1$ and one of which is $(-1)^L$, giving the stated
    coefficient. The all-zero diagonal entry of $\rho^{(L)}(N)$ equals
    $1$, since only the identity letter contributes to it, so
    $\rho^{(L)}(N)\neq0$. The projector equivalence follows by comparing
    the coefficient $\tfrac1{16}\bigl(3+(-1)^L\bigr)$ against $\tfrac14$
    on the nonzero operator.
\end{proof}

\section{The sixteen pair blocks of the stacked defect}%
\label{sec:asymex_z2z2_pairs}

itself, of bond dimension thirty-six, is block diagonal over the sixteen
pairs $(g,h)\in(\Z_2\times\Z_2)^2$, the block of $(g,h)$ being the stacked
tensor $T_g\cdot T_h$ of the product $U_gU_h$.

\begin{definition}[The stacked defect tensor and its sixteen targets]%
    \label{def:asymex_z2z2_pairs}
    \lean{Z2Z2Condensation.condensationSquare, Z2Z2Condensation.pairSlots,
        Z2Z2Condensation.pairDim, Z2Z2Condensation.pairTarget}
    \leanok
    \uses{def:asymex_z2z2_defect, def:mpdo_operator_product}
    The stacked defect tensor is the doubled-index view of $N\cdot N$,
    over the pair alphabet on $16$ letters. The sixteen slots are the
    pairs $(g,h)$, and the target of the slot $(g,h)$ is the tensor of
    $U_{gh}$ carried with the fusion weight $\lambda(g,h)$.
\end{definition}

Twelve of the sixteen pairs \emph{split}: the stacked tensor is
literally conjugate, by an integer gauge with no zero slots, to its
target. Ten of them, with at most one factor from $\{x,xy\}$ and with
$\lambda=+1$, are equal to their targets outright; the two pairs
$(x,y)$ and $(xy,y)$ carry the weight $-1$ and are conjugate to
$-M_{xy}$ and $-M_x$ by the sign gauge
$\begin{pmatrix}0&-1\\1&0\end{pmatrix}$.

    entire bond space with no zero slots, the remainder of the resulting
    one-slot compression datum is identically zero.
\end{proof}

\begin{theorem}[The four non-split pair blocks]%
    \label{thm:asymex_z2z2_nonsplit}
    \lean{Z2Z2Condensation.xXCompression, Z2Z2Condensation.xXyCompression,
        Z2Z2Condensation.xyXCompression, Z2Z2Condensation.xyXyCompression,
        Z2Z2Condensation.xX_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xXy_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xyX_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xyXy_evalWord_remainder_eq_zero}
    \leanok
    \uses{def:asymex_z2z2_pairs, thm:asymex_explicit_gauge, thm:asym_multiblock}
    The four pairs $(g,h)$ with both $g,h\in\{x,xy\}$ -- the elements
    whose tensors carry the controlled-phase dressing -- do not split.
    Each has bond dimension four and admits a one-slot compression datum
    onto its one-dimensional target $\lambda(g,h)M_{gh}$ with $z=3$ zero
    slots, an explicit unimodular integer gauge, and a remainder that
    vanishes on every word of length at least four (the bound $1+z=4$ of
    Theorem~\ref{thm:asym_multiblock}); no shorter uniform bound is
    claimed or needed. For the block $(x,x)$, whose target is
    $\lambda(x,x)M_e=\delta$, the compression pair is
    \begin{align}
        W & =(0,0,0,1),
          &
        V & =(1,-1,-1,1)^{\mathsf T},
        \notag
    \end{align}
    with $WV=1$ and $WB^{ij}V=\delta^{ij}$ for every letter, and the
    remainder $R^{ij}=B^{ij}-V\delta^{ij}W$ satisfies
    $R^{i_1j_1}R^{i_2j_2}R^{i_3j_3}R^{i_4j_4}=0$; the other three blocks
    are worked out by the same construction, each with its own gauge.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    For each of the four pairs, an explicit integer gauge, verified
    mutually inverse to its recorded inverse by a finite check, conjugates
    every letter of the stacked tensor into a matrix that is block upper
    triangular for the flag with the target in the third diagonal position
    and zero elsewhere off that position; Theorem~\ref{thm:asymex_explicit_gauge}
    reads off the compression pair from the gauge, and the nilpotency
    clause of Theorem~\ref{thm:asym_multiblock} with $|S|+z=4$ gives the
    stated vanishing.
\end{proof}

\begin{theorem}[Both sitewise intertwiner spaces vanish for the four
        non-split blocks]%
    \label{thm:asymex_z2z2_no_intertwiner}
    \lean{Z2Z2Condensation.xX_right_intertwiner_eq_zero,
        Z2Z2Condensation.xX_left_intertwiner_eq_zero,
        Z2Z2Condensation.xXy_right_intertwiner_eq_zero,
        Z2Z2Condensation.xXy_left_intertwiner_eq_zero,
        Z2Z2Condensation.xyX_right_intertwiner_eq_zero,
        Z2Z2Condensation.xyX_left_intertwiner_eq_zero,
        Z2Z2Condensation.xyXy_right_intertwiner_eq_zero,
        Z2Z2Condensation.xyXy_left_intertwiner_eq_zero}
    \leanok
    \uses{thm:asymex_z2z2_nonsplit}
    For each of the four non-split pairs $(g,h)$, if $v$ satisfies
    $B^{oi}v=\lambda(g,h)(M_{gh})_{00}v$ for every letter of the block $B$
    of $(g,h)$, then $v=0$; likewise if $u$ satisfies
    $uB^{oi}=\lambda(g,h)(M_{gh})_{00}u$ for every letter, then $u=0$.
\end{theorem}

\begin{proof}
    \leanok
    Only two of the sixteen letters of each block are nonzero, and each is
    a rank-one matrix with an explicit integer kernel and image;
    comparing the four coordinates of the two equations at those two
    letters forces every coordinate of $v$, respectively of $u$, to
    vanish. For the block $(x,x)$, whose nonzero letters are $(0,0)$ and
    $(1,1)$, comparing the four coordinates of $B^{ij}v=v$ at $(0,0)$
    gives $v_0=v_1=v_2=v_3$, and at $(1,1)$ gives $v_3=v_0$ and
    $v_1=-v_3$, forcing $v_0=v_1=v_2=v_3=0$; the transposed comparison at
    $(0,0)$ gives $u_1=u_2=u_3=0$, and at $(1,1)$ gives $u_0=0$. The
    other three blocks eliminate coordinates in the same way, through
    their own rank-one letters.
\end{proof}

\begin{remark}[The anomaly as a failure of conjugate closure, four more
        instances]%
    \label{rem:asymex_z2z2_anomaly}
    \uses{cor:asym_anomaly_not_star, thm:asymex_z2z2_no_intertwiner}
    Theorem~\ref{thm:asymex_z2z2_no_intertwiner} exhibits four further
    instances of the phenomenon named in
    Corollary~\ref{cor:asym_anomaly_not_star}: the stacked product of the
    anomalous symmetry with itself, at each of the four pairs drawn from
    $\{x,xy\}$, generates an algebra that is not closed under conjugate
    transposition, exactly as for the single squared CZX tensor of
    Section~\ref{sec:asymex_czx}. Physically, the obstruction to a
    sitewise fusion pair shows precisely at the four fusions involving two
    of the controlled-phase-dressed elements, while the twelve split
    pairs of Theorem~\ref{thm:asymex_z2z2_split} have one-dimensional
    sitewise intertwiner spaces in both directions.
\end{remark}

\begin{theorem}[The assembled compression datum of the stacked defect]%
    \label{thm:asymex_z2z2_assembled}
    \lean{Z2Z2Condensation.condensationSquare_trace_evalWord,
        Z2Z2Condensation.exists_condensationSquare_compression,
        Z2Z2Condensation.condensationSquare_compression_z}
    \leanok
    \uses{def:asymex_z2z2_pairs, thm:asym_multiblock,
        thm:asymex_z2z2_split, thm:asymex_z2z2_nonsplit}
    The trace of every nonempty word of the stacked defect tensor is the
    sum over the sixteen pairs of $\lambda(g,h)^{|w|}$ times the trace of
    the word of $M_{gh}$. Consequently the stacked defect tensor admits a
    multi-block compression, in the sense of
    Theorem~\ref{thm:asym_multiblock}, onto the sixteen weighted targets
    $\lambda(g,h)M_{gh}$, with exactly $z=12$ zero slots: the bond
    dimension count is $36=4(1+2+1+2)+12$, all twelve zero slots coming
    from the four non-split blocks of
    Theorem~\ref{thm:asymex_z2z2_nonsplit}, which alone need $z=3$ each.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asym_multiblock}
    Summing the word-trace identity of Theorem~\ref{thm:asymex_z2z2_fusion}
    over the sixteen pairs and applying
    Lemma~\ref{lem:asym_mpo_trace_eq} identifies the word traces of the
    stacked defect tensor with the stated sum; Theorem~\ref{thm:asym_multiblock}
    then supplies the compression datum, and the dimension count follows
    from clause (vii) together with the bond dimensions recorded in
    Theorems~\ref{thm:asymex_z2z2_split} and
    \ref{thm:asymex_z2z2_nonsplit}.
\end{proof}

\begin{remark}[What is and is not proved here]%
    \label{rem:asymex_z2z2_scope}
    No element of $H^3(\Z_2\times\Z_2,U(1))$ is computed anywhere in this
    development: the class attached to this presentation is external
    data, recorded only in
    \texttt{Notes/OpenProblemsTN/checks/asym\_z2z2\_typeIII\_data.md}. What
    is formalized is exactly the statements above: normality of the four
    tensors, the periodic fusion table, the length-dependent identity for
    the condensation defect, and, for the stacked defect tensor, that the
    remainders of the four non-split pair blocks vanish on every word of
    length at least four while the remainders of the twelve split pair
    blocks vanish identically. Neither a shorter nilpotency bound for the
    non-split blocks nor any cohomological invariant is claimed.
\end{remark}
