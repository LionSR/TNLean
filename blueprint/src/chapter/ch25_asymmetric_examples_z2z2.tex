\section{The anomalous \texorpdfstring{$\Z_2\times\Z_2$}{Z2 x Z2} symmetry and its condensation defect}%
\label{sec:asymex_z2z2}

The operators $U_{g,L}$ on a periodic chain of $L$ two-qubit sites are
defined by~(\ref{eq:asymex_z2z2_action}). The construction targets the mixed
three-cocycle of type II whose exponent $(-1)^{a_1b_2c_2}$ couples the first
factor of the group to the second, motivated by anomalous
$\Z_2\times\Z_2$ symmetry-protected topological phases~\cite{ChenGuWen2011}.
The four operators do not form an exact representation. After an on-site
dressing they do (Theorem~\ref{thm:asymex_z2z2_exact}), and the anomaly
class of the dressed representation has the detector values of the type-II
class (Theorem~\ref{thm:asymex_z2z2_anomaly_class}); in particular it is
nontrivial.
Matrix product operator symmetries of this kind are reviewed
in~\cite[Section~III.A.6]{Cirac2021Matrix}. The local tensors are assembled from
the letters of the CZX matrix product
unitary~\cite[Appendix~A, ``The CZX MPU'']{Cirac2021Matrix}, whose
symmetry-protected context is~\cite[Section~III.B.1]{Cirac2021Matrix}. The
group matrix product operators of~\cite{GarreRubio2023MPOSym} and the
anomalous representations of~\cite{GarreRubioSchuch2024DomainWall} are exact representations, whereas the
tensors defined below compose only projectively on odd rings, built from the CZX
letters of Section~\ref{sec:asymex_czx}. The sum
$\mathcal C_L = \sum_{g\in\Z_2\times\Z_2} U_{g,L}$ over the group is a
condensation defect in the sense of~\cite{Roumpedakis2023Condensation}.
Throughout this section the chain length is written $L$ rather than $N$,
because $N$ denotes the direct-sum tensor that generates the condensation defect.

On a periodic chain of $L>0$ two-qubit sites, define
\begin{align}
    U_{g,L}\ket{t}
     & =(-1)^{g_1\sum_{n=0}^{L-1}t_{n,2}t_{n+1,2}}
    \ket{g\oplus t}.
    \label{eq:asymex_z2z2_action}
\end{align}
Here $g=(g_1,g_2)\in G=\Z_2\times\Z_2$,
$t=(t_0,\ldots,t_{L-1})$ with $t_n=(t_{n,1},t_{n,2})\in\{0,1\}^2$,
and $(g\oplus t)_n=g\oplus t_n$ is componentwise addition modulo two.
Site indices are cyclic, so $t_L=t_0$.
The vectors $\ket{t}$ form the computational basis of
$(\C^2\otimes\C^2)^{\otimes L}$.
Write $e=(0,0)$, $y=(0,1)$, $x=(1,0)$, and $xy=(1,1)$.
The product $gh$ denotes $g\oplus h$.
The phase exponent in~(\ref{eq:asymex_z2z2_action}) is evaluated
on the input bits.

Let $X_n^{(r)}$ flip qubit $r$ at site $n$, and define the diagonal
operator $D_{2,L}$ by
\begin{align}
    D_{2,L}\ket{t} & =(-1)^{\sum_n t_{n,2}t_{n+1,2}}\ket{t}.
    \notag
\end{align}
Then
\begin{align}
    U_{g,L}
     & =\left(\prod_{n=0}^{L-1}
    (X_n^{(1)})^{g_1}(X_n^{(2)})^{g_2}\right)D_{2,L}^{\,g_1}.
    \label{eq:asymex_z2z2_gates}
\end{align}
For $L\geq3$, $D_{2,L}$ is the product of controlled-$Z$ gates
between neighboring second qubits.
Thus $U_{x,L}$ flips the first bits and applies a controlled phase
to the second bits; it does not flip the second bits and is not
the CZX operator on those qubits.
The operator $U_{xy,L}$ flips both bits, with the diagonal phase
acting first. Operator products act from right to left.

The cyclic convention in~(\ref{eq:asymex_z2z2_action}) also specifies
the short rings. For $L=1$, the phase exponent is
$t_{0,2}^2=t_{0,2}$, so $D_{2,1}=Z^{(2)}$ and
$U_{xy,1}=X^{(1)}X^{(2)}Z^{(2)}$, where
$Z\ket{b}=(-1)^b\ket{b}$.
For $L=2$, the exponent is $2t_{0,2}t_{1,2}$, so $D_{2,2}=\Id$.
There is one self-loop at length one and two cyclic bonds at length
two; these are not replaced by the edges of a simple graph.

The local tensors are matrix product operator tensors whose first
physical index is the output, or matrix row, and whose second is the
input, or matrix column.

\begin{definition}[The four symmetry tensors]%
    \label{def:asymex_z2z2_tensors}
    \lean{Z2Z2Condensation.eTensor, Z2Z2Condensation.yTensor,
        Z2Z2Condensation.xTensor, Z2Z2Condensation.xyTensor,
        Z2Z2Condensation.eMPS, Z2Z2Condensation.yMPS, Z2Z2Condensation.xMPS,
        Z2Z2Condensation.xyMPS}
    \leanok
    \uses{def:mpo_tensor, def:mpo_to_mps}
    Encode a two-qubit label $i=(i_1,i_2)$ by $2i_1+i_2$.
    For $g\in G$, let $D_g=1$ when $g_1=0$ and $D_g=2$ when
    $g_1=1$. Define $M_g^{(o,i)}\in\MN{D_g}$ by
    \begin{align}
        M_g^{(o,i)}[\alpha,\beta]
         & =\delta_{o,g\oplus i}\delta_{\beta,\pi_g(i)}f_g(\alpha,i),
        \label{eq:asymex_z2z2_tensor}
    \end{align}
    where $\delta$ is the Kronecker delta and
    $\alpha,\beta\in\{0,\ldots,D_g-1\}$.
    For $g\in\{e,y\}$, set $\pi_g(i)=0$ and $f_g(\alpha,i)=1$.
    For $g\in\{x,xy\}$, set $\pi_g(i)=i_2$ and
    $f_g(\alpha,i)=(-1)^{\alpha i_2}$.
    Thus the bond-one entries are $\delta_{o,i}$ and
    $\delta_{o,y\oplus i}$, respectively.
    Each tensor has four nonzero physical-index pairs.
    For the bond-two tensors, the nonzero matrix is $T_{i_2}$, where
    \begin{align}
        T_0 & =\begin{pmatrix}1&0\\1&0\end{pmatrix}, \notag \\
        T_1 & =\begin{pmatrix}0&1\\0&-1\end{pmatrix}.
        \label{eq:asymex_z2z2_local_matrices}
    \end{align}
    Pairing the physical indices gives an MPS tensor on the
    sixteen-letter alphabet indexed by $4o+i$, using the integer
    encodings of $o$ and $i$.
\end{definition}

Every entry lies in $\{0,\pm1\}$.
The matrices $T_0,T_1$ are those of the bond-two symmetry tensor in
Section~\ref{sec:asymex_czx}, but the physical output constraints
are different.
The tensor $M_{xy}$ is obtained from $M_x$ by relabeling the outgoing
physical index as $o\mapsto o\oplus y$.
The boundary convention is the ordinary, unnormalized trace:
\begin{align}
    \rho^{(L)}(M_g)_{s,t}
     & =\tr\left(M_g^{(s_0,t_0)}\cdots
    M_g^{(s_{L-1},t_{L-1})}\right).
    \label{eq:asymex_z2z2_trace_boundary}
\end{align}
There are no open boundary vectors.

Theorem~\ref{thm:asymex_z2z2_normal} establishes the word spans, with
one-site injectivity for the bond-one tensors and two-site injectivity
for the bond-two tensors. The multiplication phases are retained in
Theorem~\ref{thm:asymex_z2z2_fusion}. The dressed exact representation
is given in Theorem~\ref{thm:asymex_z2z2_exact}, and its nontrivial
anomaly detectors in Theorem~\ref{thm:asymex_z2z2_anomaly_class}.
These detector values concern the dressed family; they do not identify
the full cohomology class of the original family.

The first property is that the periodic contraction of each tensor
reproduces the operator of~(\ref{eq:asymex_z2z2_action}).

\begin{theorem}[Periodic contraction and unitarity]%
    \label{thm:asymex_z2z2_physical}
    \lean{Z2Z2Condensation.mpo_symTensor_apply,
        Z2Z2Condensation.mpo_symTensor_mulVec_single,
        Z2Z2Condensation.mpo_symTensor_mem_unitaryGroup,
        Z2Z2Condensation.symTensor_isMPUPos,
        Z2Z2Condensation.mpo_symTensor_mul_self,
        Z2Z2Condensation.mpo_symTensor_inv,
        Z2Z2Condensation.mpo_symTensor_conjTranspose}
    \leanok
    \uses{def:asymex_z2z2_tensors}

    For every $L>0$, $\rho^{(L)}(M_g)=U_{g,L}$.
    Each $U_{g,L}$ is unitary, and
    \begin{align}
        U_{g,L}^{-1}=U_{g,L}^{\dagger}
         & =(-1)^{Lg_1g_2}U_{g,L}.
        \label{eq:asymex_z2z2_inverse}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    Expand the trace in~(\ref{eq:asymex_z2z2_trace_boundary})
    over bond indices $b_n\in\{0,\ldots,D_g-1\}$, with $b_L=b_0$:
    \begin{align}
        \rho^{(L)}(M_g)_{s,t}
         & =\sum_{b_0,\ldots,b_{L-1}}
        \prod_{n=0}^{L-1}
        M_g^{(s_n,t_n)}[b_n,b_{n+1}].
        \label{eq:asymex_z2z2_cyclic}
    \end{align}
    If $s\neq g\oplus t$, the physical constraints make every summand
    zero. Otherwise the bond constraints force
    $b_{n+1}=\pi_g(t_n)$, leaving exactly one bond configuration.
    For $g_1=0$ its weight is one. For $g_1=1$ its weight is
    $\prod_n(-1)^{t_{n-1,2}t_{n,2}}$.
    Reindexing the cyclic sum therefore yields
    \begin{align}
        \rho^{(L)}(M_g)_{s,t}
         & =\delta_{s,g\oplus t}
        (-1)^{g_1\sum_n t_{n,2}t_{n+1,2}}.
        \label{eq:asymex_z2z2_kernel}
    \end{align}
    This is~(\ref{eq:asymex_z2z2_action}), including its self-loop
    convention at $L=1$.

    The map $t\mapsto g\oplus t$ permutes the computational basis,
    and every coefficient in~(\ref{eq:asymex_z2z2_action}) has
    modulus one. Hence the images form an orthonormal basis, proving
    unitarity.

    Set $E(t_2)=\sum_n t_{n,2}t_{n+1,2}$. Complementing all second
    bits gives
    \begin{align}
        E(1\oplus t_2)
         & =\sum_n(1-t_{n,2})(1-t_{n+1,2})
        =L-2\sum_n t_{n,2}+E(t_2).
        \label{eq:asymex_z2z2_flip_phase}
    \end{align}
    Thus $E(g_2\oplus t_2)\equiv E(t_2)+Lg_2\pmod 2$.
    Applying~(\ref{eq:asymex_z2z2_action}) twice gives
    $U_{g,L}^2=(-1)^{Lg_1g_2}\Id$.
    Together with unitarity, this proves
    (\ref{eq:asymex_z2z2_inverse}).
\end{proof}

Equation~(\ref{eq:asymex_z2z2_flip_phase}) also fixes the gate order:
$D_{2,L}\prod_nX_n^{(1)}X_n^{(2)}
    =(-1)^LU_{xy,L}$.
At length zero, the trace of the empty product is $D_g$.
The bond-one tensors therefore give the scalar $1$, while the
bond-two tensors give $2$, not a unitary operator.
All unitarity and group-multiplication assertions here concern
positive lengths.

The basis action also determines multiplication. For $g,h\in G$,
\begin{align}
    U_{g,L}U_{h,L}\ket{t}
     & =(-1)^{h_1E(t_2)+g_1E(h_2\oplus t_2)}
    \ket{g\oplus h\oplus t}
    =(-1)^{Lg_1h_2}U_{gh,L}\ket{t}.
    \label{eq:asymex_z2z2_basis_product}
\end{align}
The last equality uses~(\ref{eq:asymex_z2z2_flip_phase}) and the
fact that the parity of $g_1+h_1$ is the first bit of $gh$.
Thus $\lambda(g,h)=(-1)^{g_1h_2}$ supplies a length-dependent
scalar factor, reflecting the non-trivial projective 2-cocycle class of
$\Z_2\times\Z_2$~\cite{ChenGuWen2011,GarreRubio2023MPOSym}.
The tensor proof of this law, through word traces,
is given in Theorem~\ref{thm:asymex_z2z2_fusion}.

The second property concerns the word spans of the four tensors,
which the compression results of
Section~\ref{sec:asymex_z2z2_pairs} require.

\begin{theorem}[Injectivity after blocking of the four symmetry tensors]%
    \label{thm:asymex_z2z2_normal}
    \lean{Z2Z2Condensation.eMPS_isNormal, Z2Z2Condensation.yMPS_isNormal,
        Z2Z2Condensation.xMPS_isNormal, Z2Z2Condensation.xyMPS_isNormal,
        Z2Z2Condensation.xMPS_isNBlkInjective_two,
        Z2Z2Condensation.xyMPS_isNBlkInjective_two,
        Z2Z2Condensation.xMPS_not_isNBlkInjective_one,
        Z2Z2Condensation.xyMPS_not_isNBlkInjective_one}
    \leanok
    \uses{def:asymex_z2z2_tensors, def:normal}
    The bond-one tensors of $U_e$ and $U_y$ are injective without
    blocking: a nonzero matrix is the scalar $1$ and spans
    $\MN{1}$. The bond-two tensors of $U_x$ and $U_{xy}$ are not
    injective without blocking, since their four nonzero matrices span
    only a two-dimensional space. Their length-two words span
    $\MN{2}$, so they are injective after blocking two sites.
    These assertions concern word spans. The tensors are proportional
    to normal tensors (NT); the spectral-radius normalization required
    for NT is imposed separately.
\end{theorem}

\begin{proof}
    \leanok
    For each bond-one tensor, a nonzero scalar matrix spans $\MN{1}$.
    For each bond-two tensor, all four products $T_aT_b$ occur as
    length-two words. If $E_{ab}$ denotes a matrix unit, direct
    multiplication gives
    \begin{align}
        2E_{00} & =T_0T_0+T_1T_0, \notag \\
        2E_{10} & =T_0T_0-T_1T_0, \notag \\
        2E_{01} & =T_0T_1-T_1T_1, \notag \\
        2E_{11} & =T_0T_1+T_1T_1.
        \notag
    \end{align}
    These integer linear combinations span $\MN{2}$.
    At length one, $T_0,T_1$ are linearly independent, but the
    functional $Q\mapsto Q_{00}-Q_{10}$ vanishes on both and takes
    the value $1$ on $E_{00}$. Their span is therefore proper.
\end{proof}

\subsection{The condensation defect and its periodic fusion table}%
\label{sec:asymex_z2z2_defect}

The condensation defect is the operator
$A_L=U_{e,L}+U_{x,L}+U_{y,L}+U_{xy,L}$.
It is generated by the direct-sum tensor
$N=M_e\oplus M_x\oplus M_y\oplus M_{xy}$ of
Definition~\ref{def:mpo_direct_sum}, with bond dimension six.

\begin{definition}[The condensation defect tensor]%
    \label{def:asymex_z2z2_defect}
    \lean{Z2Z2Condensation.condensationTensor,
        Z2Z2Condensation.mpo_condensationTensor}
    \leanok
    \uses{def:mpo_direct_sum, def:asymex_z2z2_tensors}
    The condensation defect tensor is the iterated direct sum
    $N=((M_e\oplus M_x)\oplus M_y)\oplus M_{xy}$ of bond dimension six.
    Its periodic operator is the sum of the four periodic operators of
    Definition~\ref{def:asymex_z2z2_tensors}:
    \begin{align}
        \rho^{(L)}(N)
         & =\rho^{(L)}(M_e)+\rho^{(L)}(M_x)
        +\rho^{(L)}(M_y)+\rho^{(L)}(M_{xy}).
        \label{eq:asymex_z2z2_defect_sum}
    \end{align}
    This identity holds at every system size, including length zero.
    For $L>0$, it reads $\rho^{(L)}(N)=A_L$.
\end{definition}

\begin{proof}
    \leanok
    \uses{thm:mpo_direct_sum_additive}
    Apply the additivity of
    Theorem~\ref{thm:mpo_direct_sum_additive} three times.
\end{proof}

The multiplication law~(\ref{eq:asymex_z2z2_basis_product}) of the
four operators is now derived from the tensors themselves, through
word traces of their stacked products.

\begin{theorem}[The periodic fusion table]%
    \label{thm:asymex_z2z2_fusion}
    \lean{Z2Z2Condensation.bondDim, Z2Z2Condensation.symTensor,
        Z2Z2Condensation.gmul, Z2Z2Condensation.fusionSign,
        Z2Z2Condensation.mpo_symTensor_mul,
        Z2Z2Condensation.mpo_x_mul_y_eq_neg_of_odd}
    \leanok
    \uses{def:asymex_z2z2_tensors, lem:asym_mpo_trace_eq}
    For every $L\geq1$ and every $g,h\in\Z_2\times\Z_2$, the periodic
    operators satisfy
    \begin{align}
        \rho^{(L)}(M_g)\rho^{(L)}(M_h)
         & =\lambda(g,h)^L\rho^{(L)}(M_{gh}),
        \label{eq:asymex_z2z2_fusion}
    \end{align}
    where
    \begin{align}
        \lambda(g,h)
         & =\begin{cases}
                -1 & \text{if }g\in\{x,xy\}\text{ and }h\in\{y,xy\}, \\
                +1 & \text{otherwise}.
            \end{cases}
        \notag
    \end{align}
    On odd rings, $\lambda(x,y)\lambda(y,x)=-1$, so
    $\rho^{(L)}(M_x)\rho^{(L)}(M_y)=-\rho^{(L)}(M_y)\rho^{(L)}(M_x)$.
    The four operators form a projective representation of
    $\Z_2\times\Z_2$ on odd rings and a linear representation on even
    rings.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:asym_mpo_trace_eq}
    The stacked tensor is defined by
    $(M_g\cdot M_h)^{(o,i)}
        =\sum_{j=0}^3M_g^{(o,j)}\otimes M_h^{(j,i)}$.
    For each of the sixteen pairs, the pair-block compression gives
    \begin{align}
        \tr((M_g\cdot M_h)^w)
         & =\lambda(g,h)^{|w|}\tr(M_{gh}^w)
        \label{eq:asymex_z2z2_pair_trace}
    \end{align}
    for every nonempty word $w$ on the paired physical alphabet.
    The explicit split and non-split compressions are described in
    Theorems~\ref{thm:asymex_z2z2_split}
    and~\ref{thm:asymex_z2z2_nonsplit}.
    Lemma~\ref{lem:asym_mpo_trace_eq} converts
    (\ref{eq:asymex_z2z2_pair_trace}) into the operator identity
    (\ref{eq:asymex_z2z2_fusion}).
    The odd-ring anticommutation follows by substituting
    $\lambda(x,y)=-1$ and $\lambda(y,x)=+1$ and using $(-1)^L=-1$.
\end{proof}

The four operators are related to the group matrix product operators of
Section~\ref{sec:asymex_group_cocycle_mpo} for the mixed cocycle of
Definition~\ref{def:asymex_group_cocycle_klein} by an on-site dressing.
Identify the two-qubit label $2i_1+i_2$ with the group element
$(i_1,i_2)$, so that each site carries $\C[\Z_2\times\Z_2]$, and write
$U^{\omega}_{g,L}$ for the periodic operator of that construction on a
ring of $L$ sites.

\begin{theorem}[Comparison with the cocycle construction]%
    \label{thm:asymex_z2z2_group_cocycle}
    \lean{Z2Z2Condensation.mpo_symTensor_eq_groupCocycle,
        Z2Z2Condensation.secondZDressing,
        Z2Z2Condensation.gmul_kleinEquiv}
    \leanok
    \uses{thm:asymex_z2z2_physical, thm:asymex_group_cocycle_klein}
    Define the diagonal operator $Z_{2,L}$ by
    $Z_{2,L}\ket{t}=(-1)^{\sum_nt_{n,2}}\ket{t}$, the product of the
    Pauli $Z$ on the second qubit of every site. For every
    $g\in\Z_2\times\Z_2$ and every $L>0$,
    \begin{align}
        U_{g,L}
         & =U^{\omega}_{g,L}\,Z_{2,L}^{\,g_1}.
        \label{eq:asymex_z2z2_group_cocycle}
    \end{align}
    The dressing is trivial for $g\in\{e,y\}$ and is a product of
    single-site unitaries. The cocycle operators form an exact
    representation, whereas the operators $U_{g,L}$ compose only up to
    the signs of Theorem~\ref{thm:asymex_z2z2_fusion}.
\end{theorem}

\begin{proof}
    \leanok
    Both sides vanish at $(s,t)$ unless $s=g\oplus t$, since the table
    of the labels $2i_1+i_2$ is the group law of $\Z_2\times\Z_2$. On
    that support, Theorem~\ref{thm:asymex_group_cocycle_klein} gives the
    sign of the right side as $(-1)^e$ with
    \begin{align}
        e
         & =g_1\sum_n\bigl(t_{n+1,2}(t_{n,2}-t_{n+1,2})+t_{n,2}\bigr).
        \notag
    \end{align}
    Reindexing $\sum_nt_{n,2}=\sum_nt_{n+1,2}$ cyclically and using
    $t_{n+1,2}^2=t_{n+1,2}$ gives
    $e\equiv g_1\sum_nt_{n,2}t_{n+1,2}\pmod2$, which is the sign
    of~(\ref{eq:asymex_z2z2_action}).
\end{proof}

\begin{theorem}[An exact normal representation by dressed tensors]%
    \label{thm:asymex_z2z2_exact}
    \lean{Z2Z2Condensation.secondZSign,
        Z2Z2Condensation.secondZSign_ne_zero,
        Z2Z2Condensation.diagonal_prod_secondZSign,
        Z2Z2Condensation.secondZDressing_mul_self,
        Z2Z2Condensation.kleinTensor,
        Z2Z2Condensation.mpo_kleinTensor,
        Z2Z2Condensation.kleinTensor_isNormal,
        Z2Z2Condensation.kleinFamily,
        Z2Z2Condensation.kleinFamily_isNormalRepresentation}
    \leanok
    \uses{def:asymex_z2z2_tensors, def:asymex_group_cocycle_klein,
        def:mpug_anomaly_three_cochain}
    For $g\in\Z_2\times\Z_2$ let $M'_g$ be the tensor $M_g$ of
    Definition~\ref{def:asymex_z2z2_tensors} with the letter of input label
    $i$ multiplied by $(-1)^{g_1i_2}$. It has the bond dimension of $M_g$,
    its doubled-index tensor is normal, and its periodic operator on every
    ring of $L>0$ sites is $U^{\omega}_{g,L}$. In particular the tensors
    $M'_g$ form an exact representation of $\Z_2\times\Z_2$ by normal
    tensors.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_z2z2_group_cocycle, thm:asymex_z2z2_normal,
        thm:asymex_group_cocycle_klein, thm:mpo_diagonal_dressing}
    By Theorem~\ref{thm:mpo_diagonal_dressing} the periodic operator
    of $M'_g$ is $U_{g,L}Z_{2,L}^{\,g_1}$, which is $U^{\omega}_{g,L}$ by
    (\ref{eq:asymex_z2z2_group_cocycle}) and $Z_{2,L}^2=\Id$. The
    operators $U^{\omega}_{g,L}$ multiply as the group by
    Theorem~\ref{thm:asymex_group_cocycle_klein}. The letters of $M'_g$
    are those of the normal tensor $M_g$ of
    Theorem~\ref{thm:asymex_z2z2_normal} multiplied by signs, so
    Theorem~\ref{thm:mpo_diagonal_dressing} gives normality.
\end{proof}

\begin{theorem}[The condensation defect squares to a length-dependent
        multiple of itself]%
    \label{thm:asymex_z2z2_defect_sq}
    \lean{Z2Z2Condensation.mpo_condensationTensor_sq,
        Z2Z2Condensation.mpo_condensationTensor_apply_zero,
        Z2Z2Condensation.mpo_condensationTensor_ne_zero,
        Z2Z2Condensation.condensation_quarter_sq_eq_self_iff_even}
    \leanok
    \uses{def:asymex_z2z2_defect, thm:asymex_z2z2_fusion}
    For every $L\geq1$,
    \begin{align}
        A_L^2 & =(3+(-1)^L)A_L.
        \label{eq:asymex_z2z2_defect_square}
    \end{align}
    Its all-zero diagonal entry is $1$, so $A_L\neq0$.
    Consequently $(A_L/4)^2=A_L/4$ if and only if $L$ is even.
    Thus the normalization by $\tfrac14$ is not idempotent on odd
    rings.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_z2z2_fusion}
    Expand the square using Theorem~\ref{thm:asymex_z2z2_fusion}.
    For each $k\in G$, the four pairs satisfying $gh=k$ contribute
    three weights $+1$ and one weight $(-1)^L$. Hence
    \begin{align}
        A_L^2
         & =\sum_{k\in G}
        \left(\sum_{\substack{g,h\in G \\gh=k}}
        \lambda(g,h)^L\right)U_{k,L}
        =(3+(-1)^L)A_L.
        \notag
    \end{align}
    At the all-zero physical-index pair, $M_e^{(0,0)}=1$ and
    $M_x^{(0,0)}=M_y^{(0,0)}=M_{xy}^{(0,0)}=0$.
    Since $L>0$, only the identity tensor contributes to the all-zero
    diagonal entry, giving $1$ and proving $A_L\neq0$.
    Finally,
    \begin{align}
        (A_L/4)^2-A_L/4
         & =\frac{(-1)^L-1}{16}A_L.
        \label{eq:asymex_z2z2_quarter_difference}
    \end{align}
    Nonvanishing of $A_L$ reduces idempotence to $(-1)^L=1$.
\end{proof}

\begin{lemma}[Orthogonal projection on even rings and idempotence on odd rings]%
    \label{lem:asymex_z2z2_defect_adjoint}
    \lean{Z2Z2Condensation.mpo_condensationTensor_conjTranspose,
        Z2Z2Condensation.mpo_xyTensor_ne_zero,
        Z2Z2Condensation.isStarProjection_quarter_condensation_of_even,
        Z2Z2Condensation.isIdempotentElem_half_condensation_of_odd,
        Z2Z2Condensation.conjTranspose_sub_mpo_condensationTensor_of_odd,
        Z2Z2Condensation.not_isSelfAdjoint_half_condensation_of_odd}
    \leanok
    \uses{def:asymex_z2z2_defect}
    On every positive even ring, $A_L/4$ is an orthogonal projector.
    On every odd ring, $A_L/2$ is an idempotent that is not
    self-adjoint.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:asymex_z2z2_physical, thm:asymex_z2z2_defect_sq}
    Equation~(\ref{eq:asymex_z2z2_inverse}) gives
    \begin{align}
        A_L^\dagger
         & =\Id+U_{x,L}+U_{y,L}+(-1)^LU_{xy,L}.
        \label{eq:asymex_z2z2_defect_adjoint}
    \end{align}
    For even $L$, this equals $A_L$, and
    (\ref{eq:asymex_z2z2_defect_square}) gives $(A_L/4)^2=A_L/4$.
    A self-adjoint idempotent is an orthogonal projector.
    For odd $L$, (\ref{eq:asymex_z2z2_defect_square}) instead gives
    $(A_L/2)^2=A_L/2$, while
    $A_L^\dagger-A_L=-2U_{xy,L}\neq0$ because $U_{xy,L}$ is unitary.
\end{proof}

\subsection{The sixteen pair blocks of the stacked defect}%
\label{sec:asymex_z2z2_pairs}

The operator identities of the fusion table have word-trace
counterparts, to which the multi-block compression theorem applies.
The stacked product $B=N\cdot N$ has local matrices
\begin{align}
    B^{(o,i)}

for $L>0$.
After grouping bond indices by $(g,h)\in G^2$, it is block diagonal

    \label{thm:asymex_z2z2_split}
    \lean{Z2Z2Condensation.eEStacked_eq, Z2Z2Condensation.eYStacked_eq,

        Z2Z2Condensation.eY_remainder_eq_zero,
        Z2Z2Condensation.eX_remainder_eq_zero,
        Z2Z2Condensation.eXy_remainder_eq_zero,
        Z2Z2Condensation.yE_remainder_eq_zero,
        Z2Z2Condensation.yY_remainder_eq_zero,
        Z2Z2Condensation.yX_remainder_eq_zero,
        Z2Z2Condensation.yXy_remainder_eq_zero,
        Z2Z2Condensation.xE_remainder_eq_zero,

        Z2Z2Condensation.xY_remainder_eq_zero,

    \leanok
    \uses{def:asymex_z2z2_pairs, def:asymex_one_slot, thm:asymex_explicit_gauge}
    For each of the twelve split pairs $(g,h)$, the corresponding block
    of the stacked defect tensor admits a compression onto the single
    target $\lambda(g,h)M_{gh}$ with $z=0$.
    An invertible bond-space gauge conjugates the block exactly to this
    target, with vanishing remainder. The gauge is the identity for ten
    pairs and $\begin{pmatrix}0&-1\\1&0\end{pmatrix}$ for $(x,y)$ and
    $(xy,y)$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:asymex_one_slot, thm:asymex_one_slot, thm:asymex_explicit_gauge}
    Substitution of~(\ref{eq:asymex_z2z2_tensor}) in the stacked
    product shows that the ten positive-weight blocks equal their
    targets at every physical-index pair.
    For the other two, set
    $S=\begin{pmatrix}0&-1\\1&0\end{pmatrix}$.
    Direct multiplication gives $S^{-1}T_{1-b}S=-T_b$ for
    $b\in\{0,1\}$. This proves the conjugation identities for
    $(x,y)$ and $(xy,y)$.
    Theorem~\ref{thm:asymex_explicit_gauge} gives the compression
    maps $W=S^{-1}$ and $V=S$ in these two cases, and identity maps
    in the other ten.
    In every case $B^{(o,i)}=VC_s^{(o,i)}W$ on the entire bond
    space. Thus there are no zero diagonal blocks and the remainder
    vanishes identically.
\end{proof}

The remaining four pairs are the ones with both factors of bond
dimension two. Their stacked blocks have bond dimension four but
targets of bond dimension one, so the compression must discard three
bond directions.

\begin{theorem}[The four non-split pair blocks]%
    \label{thm:asymex_z2z2_nonsplit}
    \lean{Z2Z2Condensation.xXCompression, Z2Z2Condensation.xXyCompression,
        Z2Z2Condensation.xyXCompression, Z2Z2Condensation.xyXyCompression,
        Z2Z2Condensation.xX_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xXy_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xyX_evalWord_remainder_eq_zero,
        Z2Z2Condensation.xyXy_evalWord_remainder_eq_zero}
    \leanok
    \uses{def:asymex_z2z2_pairs, def:asymex_one_slot, thm:asymex_explicit_gauge,
        thm:asym_multiblock}
    The four pairs $(g,h)$ with both $g,h\in\{x,xy\}$ do not split.
    Their targets, in the order $(x,x)$, $(x,xy)$, $(xy,x)$,
    $(xy,xy)$, are $M_e$, $-M_y$, $M_y$, $-M_e$.
    Each block has bond dimension four and admits a compression onto
    its one-dimensional target $\lambda(g,h)M_{gh}$ with $z=3$ zero
    diagonal entries, an explicit unimodular integer gauge, and a
    remainder that vanishes on every word of length at least four.
    This is the bound $1+z=4$ of
    Theorem~\ref{thm:asym_multiblock}.
    For the block $(x,x)$, write
    $\delta^{ij}=\delta_{i,j}$ for the scalar entries of $M_e$.
    Its compression pair is
    \begin{align}
        W & =(0,0,0,1),
        \notag                        \\
        V & =(1,-1,-1,1)^{\mathsf T},
        \notag
    \end{align}
    with $WV=1$ and $WB^{ij}V=\delta^{ij}$ for every physical-index
    pair. The remainder $R^{ij}=B^{ij}-V\delta^{ij}W$ satisfies
    $R^{i_1j_1}R^{i_2j_2}R^{i_3j_3}R^{i_4j_4}=0$.
    The other three blocks admit the same construction, each with its
    own gauge.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_explicit_gauge}
    For each of the four pairs, a finite calculation verifies an
    integer gauge and its inverse. This gauge puts every matrix of
    the stacked tensor in upper triangular form, with the target
    in the third diagonal position and zero in the other diagonal
    positions.
    Theorem~\ref{thm:asymex_explicit_gauge} gives the compression pair
    from this gauge. Subtracting $VC^{ij}W$ removes the only
    potentially nonzero diagonal entry. The remaining matrices are
    strictly upper triangular in a common four-dimensional basis,
    so every product of four remainders is zero.
    Equivalently, this is the nilpotency clause of
    Theorem~\ref{thm:asym_multiblock}, with $|S|+z=4$.
    The absence of a splitting follows from the intertwiner
    calculation in Theorem~\ref{thm:asymex_z2z2_no_intertwiner}.
\end{proof}

\begin{theorem}[Both sitewise intertwiner spaces vanish for the four
        non-split blocks]%
    \label{thm:asymex_z2z2_no_intertwiner}
    \lean{Z2Z2Condensation.xX_right_intertwiner_eq_zero,
        Z2Z2Condensation.xX_left_intertwiner_eq_zero,
        Z2Z2Condensation.xXy_right_intertwiner_eq_zero,
        Z2Z2Condensation.xXy_left_intertwiner_eq_zero,
        Z2Z2Condensation.xyX_right_intertwiner_eq_zero,
        Z2Z2Condensation.xyX_left_intertwiner_eq_zero,
        Z2Z2Condensation.xyXy_right_intertwiner_eq_zero,
        Z2Z2Condensation.xyXy_left_intertwiner_eq_zero}
    \leanok
    \uses{thm:asymex_z2z2_nonsplit}
    For each of the four non-split pairs $(g,h)$, let $B$ denote its
    stacked block and set $C^{oi}=\lambda(g,h)(M_{gh}^{oi})_{00}$.
    If a column vector $v\in\C^4$ satisfies $B^{oi}v=C^{oi}v$ for
    every physical-index pair, then $v=0$.
    Likewise, if a row vector $u$ satisfies $uB^{oi}=C^{oi}u$ for
    every physical-index pair, then $u=0$.
\end{theorem}

\begin{proof}
    \leanok
    Four of the sixteen matrices of each block are nonzero, taking
    only two distinct values. Each distinct value has rank one, with
    an explicit integer kernel and image.
    Comparing the four coordinates of the two independent equations
    forces every coordinate of $v$, respectively of $u$, to vanish.

    For the block $(x,x)$, the nonzero physical-index pairs are
    $(0,0)$ and $(1,1)$, repeated at $(2,2)$ and $(3,3)$.
    Its two matrices are $T_0\otimes T_0$ and $T_1\otimes T_1$.
    The equation $(T_0\otimes T_0)v=v$ gives
    $v_0=v_1=v_2=v_3$, while $(T_1\otimes T_1)v=v$ gives
    $v_0=v_3$ and $v_1=-v_3$. Hence all coordinates vanish.
    The row equation $u(T_0\otimes T_0)=u$ gives
    $u_1=u_2=u_3=0$, and $u(T_1\otimes T_1)=u$ then gives $u_0=0$.

    The block $(xy,x)$ has the same two matrices and target value
    $+1$ at its nonzero physical-index pairs.
    For $(x,xy)$ and $(xy,xy)$, the matrices are
    $T_1\otimes T_0$ and $T_0\otimes T_1$, with target value $-1$.
    The equation $(T_1\otimes T_0)v=-v$ gives
    $v=(-v_2,-v_2,v_2,v_2)^{\mathsf T}$.
    Substituting this into $(T_0\otimes T_1)v=-v$ forces $v_2=0$.
    On the left, $u(T_1\otimes T_0)=-u$ forces
    $u_0=u_1=u_3=0$, and $u(T_0\otimes T_1)=-u$ then forces
    $u_2=0$.
\end{proof}

The sixteen pair-block compressions assemble into one compression of
the full stacked tensor.

\begin{theorem}[Compression of the stacked defect tensor]%
    \label{thm:asymex_z2z2_assembled}
    \lean{Z2Z2Condensation.condensationSquare_trace_evalWord,
        Z2Z2Condensation.exists_condensationSquare_compression,
        Z2Z2Condensation.condensationSquare_compression_z}
    \leanok
    \uses{def:asymex_z2z2_pairs, thm:asym_multiblock,
        thm:asymex_z2z2_split, thm:asymex_z2z2_nonsplit}
    For every nonempty word $w$, the stacked defect tensor $B=N\cdot N$
    satisfies
    \begin{align}
        \tr(B^w)
         & =\sum_{g,h\in\Z_2\times\Z_2}
        \lambda(g,h)^{|w|}\tr(M_{gh}^w).
        \label{eq:asymex_z2z2_assembled_trace}
    \end{align}
    Consequently $B$ admits a multi-block compression, in the sense of
    Theorem~\ref{thm:asym_multiblock}, onto the sixteen weighted
    targets $\lambda(g,h)M_{gh}$.
    Every such compression has exactly $z=12$ zero diagonal entries,
    since $36=4(1+2+1+2)+12$.
    In the pair-block construction, these twelve entries come from
    the four non-split blocks of
    Theorem~\ref{thm:asymex_z2z2_nonsplit}, each of which has $z=3$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asym_multiblock}
    Decode $w$ as a pair of physical configurations of length $|w|$.
    The periodic operator of $B$ is the square of the sum in
    (\ref{eq:asymex_z2z2_defect_sum}). Expanding it and applying
    Theorem~\ref{thm:asymex_z2z2_fusion} gives the sum of the
    corresponding matrix entries of
    $\lambda(g,h)^{|w|}\rho^{(|w|)}(M_{gh})$.
    Lemma~\ref{lem:asym_mpo_trace_eq} identifies these entries with
    word traces. Since
    $\tr(C_{(g,h)}^w)=\lambda(g,h)^{|w|}\tr(M_{gh}^w)$,
    this proves~(\ref{eq:asymex_z2z2_assembled_trace}) in the form
    required for compression.

    The targets have positive bond dimensions and are normal in the
    word-span sense by Theorem~\ref{thm:asymex_z2z2_normal}:
    multiplication by the nonzero scalar $\lambda(g,h)$ does not
    change any fixed-length word span.
    Theorem~\ref{thm:asym_multiblock} therefore supplies the
    compression from~(\ref{eq:asymex_z2z2_assembled_trace}).
    Each $k\in G$ occurs as $gh$ for four pairs, so the sum of target
    dimensions is $4(1+1+2+2)=24$.
    The dimension identity in clause (vii) gives $36=24+z$, hence
    $z=12$.
    The explicit pair-block construction allocates three of these
    zero diagonal entries to each non-split block and none to a
    split block.
\end{proof}

    The letters of $B=N\cdot N$ are block diagonal over the sixteen pairs
    $(g,h)$, with the stacked tensor $M_g\cdot M_h$ as the block of $(g,h)$.

    of $B$ onto the weighted targets $\lambda(g,h)M_{gh}$ with $z=12$, and

\begin{proof}
    \leanok

    so Theorem~\ref{thm:asymex_mul_block_diagonal} makes $B$ block diagonal

    remainder of the direct sum is the direct sum of the pair remainders.
    The twelve split pairs have zero remainder. On each of the four
    non-split pairs, the remainder letters vanish outside two physical
    letters, and every product of three of these remainder matrices
    vanishes; on the pair $(x,x)$ the square of the remainder at the
    physical letter $0$ is nonzero.

\begin{remark}[Failure of closure under conjugate transposition]%
    \label{rem:asymex_z2z2_anomaly}
    \uses{cor:asym_anomaly_not_star, thm:asymex_z2z2_no_intertwiner}
    By Theorem~\ref{thm:asymex_z2z2_no_intertwiner} and
    Corollary~\ref{cor:asym_anomaly_not_star}, the algebra generated
    by the matrices of each stacked pair block with
    $g,h\in\{x,xy\}$ is not closed under conjugate transposition.
    This is the same algebraic obstruction as for the squared CZX
    tensor in Section~\ref{sec:asymex_czx}.
    For each of these four blocks, there is no nonzero sitewise
    intertwiner with the specified target in either direction.
    In contrast, the twelve split pairs of
    Theorem~\ref{thm:asymex_z2z2_split} have one-dimensional
    sitewise intertwiner spaces in both directions.
    Indeed, conjugation identifies these spaces with the commutant
    of the target matrices, and injectivity after blocking makes
    that commutant consist only of scalar matrices.
\end{remark}

\begin{remark}[Nilpotency of the stacked defect remainders]%
    \label{rem:asymex_z2z2_scope}
    The four non-split pair blocks have remainders that vanish on
    every word of length at least four; the twelve split pair blocks
    have identically vanishing remainders.
\end{remark}

\subsection{The anomaly class of the full group}%
\label{sec:asymex_z2z2_anomaly}

detector values for the dressed representation.

products distinguish all cohomology classes. The distinction between the dressed
representation and the original odd-ring projective family remains as described
in~\cite{gap:glm23_z2z2_anomaly_detector_scope}.

\subsection{The eight scalar representatives}

\begin{definition}[Klein-four cocycle representatives]
    \label{def:asymex_klein_cocycle_family}
    \lean{TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily}
    \leanok
    \uses{def:mpug_scalar_cochains}
    For $p,q,r\in\Z_2$ and $a,b,c\in\Z_2^2$, set
    \begin{align}
        \omega_{pqr}(a,b,c)
        =(-1)^{p a_1b_1c_1+q a_2b_2c_2+r a_1b_2c_2}.
    \end{align}
    These are the representatives in the $\Z_2\times\Z_2$ table
    of~\cite[Section~6]{GarreRubio2023MPOSym}. The exponent is evaluated in $\Z_2$.
\end{definition}

\begin{theorem}[Cocycle equation and restriction signs]
    \label{thm:asymex_klein_cocycle_restrictions}
    \lean{TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_isCocycle,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_isNormalized,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_signs,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_restriction_iff,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_one_subgroup_iff,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_zero_zero_one}
    \leanok
    \uses{def:asymex_klein_cocycle_family,def:mpug_scalar_cochains,def:mpug_three_cocycle,
        def:asymex_group_cocycle_klein}
    Each $\omega_{pqr}$ is a normalized three-cocycle. At
    $a=(1,0)$, $b=(0,1)$, and $a+b=(1,1)$ its diagonal values are
    \begin{align}
        \bigl(\omega_{pqr}(a,a,a),\omega_{pqr}(b,b,b),
        \omega_{pqr}(a+b,a+b,a+b)\bigr)
        =\bigl((-1)^p,(-1)^q,(-1)^{p+q+r}\bigr).
    \end{align}
    Its restriction to the subgroup generated by $(x,y)$ is identically one
    if and only if $px+qy+rxy=0$ in $\Z_2$. In particular, for every subgroup
    $H\leq\Z_2\times\Z_2$, $\omega_{111}|_H=1$ if and only if $H=\{0\}$.
    The representative $\omega_{001}$ is the cocycle
    $(-1)^{a_1b_2c_2}$ of Definition~\ref{def:asymex_group_cocycle_klein}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:asymex_klein_cocycle_family}
    The exponent is additive in each argument, which gives the cocycle equation,
    and vanishes if any argument is zero. Substitution gives the three signs.
    On the subgroup generated by $g=(x,y)$, normalization leaves only the value
    at $(g,g,g)$ to check. Since $x^2=x$ and $y^2=y$, this value is
    $(-1)^{px+qy+rxy}$. For $p=q=r=1$, the exponent is nonzero at every
    nonzero $g$. Hence any subgroup on which the restriction is one contains
    only zero.
\end{proof}


\begin{theorem}[Distinct Klein-four classes and the last-row restriction]
    \label{thm:asymex_klein_distinct_classes}
    \lean{TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_diagonal_eq_of_cohomologousTo,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_cohomologousTo_iff,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_one_isTrivialGaugeClass_comap_iff}
    \leanok
    \uses{def:asymex_klein_cocycle_family,thm:asymex_klein_cocycle_restrictions,
        def:mpug_cohomologous_to,def:mpug_trivial_gauge_class,def:mpug_cochain_restriction}
    The representatives $\omega_{pqr}$ and $\omega_{p'q'r'}$ are cohomologous
    if and only if $(p,q,r)=(p',q',r')$. For every subgroup $H\leq\Z_2^2$,
    the class of $\omega_{111}|_H$ is trivial if and only if $H=\{0\}$.
    These are the distinction of the eight representatives and the last-row
    subgroup assertion in~\cite[Section~6]{GarreRubio2023MPOSym}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_cocycle_restrictions,thm:asymex_klein_subgroup_table,
        thm:mpug_cyclic_invariant_gauge,thm:mpug_cohomologous_to_equivalence}
    For an element $g$ of order dividing two, the cyclic product
    $\omega(g,0,g)\omega(g,g,g)$ is invariant under multiplication by a
    three-coboundary. Normalization reduces it to $\omega(g,g,g)$.
    The three signs therefore agree for cohomologous representatives;
    they recover $p$, $q$, and then $r$. Conversely, equal parameters give
    the same cocycle. If the restriction of $\omega_{111}$ to $H$ has
    trivial class, Theorem~\ref{thm:asymex_klein_subgroup_table} gives
    diagonal value $1$ at every $g\in H$.
    The last row has diagonal value $-1$ at every nonzero element, so
    $H=\{0\}$. On the trivial subgroup the cocycle is identically one.
\end{proof}


\begin{theorem}[Four-block orbits for the last Klein-four class]
    \label{thm:asymex_klein_four_blocks}
    \lean{TNLean.Algebra.LSymbol.stabilizer_eq_bot_of_kleinCocycleFamily_one,

    and $|Gx|=4$. In particular, if $X$ is nonempty and the action is
    transitive, then $|X|=4$. This gives the block count for the last
    class in~\cite[Section~6]{GarreRubio2023MPOSym}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:mpug_fixed_block_trivializes,thm:asymex_klein_distinct_classes}

    Thus the restricted class is trivial and $G_x=\{0\}$.
    The orbit--stabilizer identity gives $|Gx|\,|G_x|=|G|=4$.
    For a transitive action, $Gx=X$.
\end{proof}


\begin{theorem}[L-symbols on a free transitive action]
    \label{thm:asymex_free_action_l_symbols}
    \lean{TNLean.Algebra.LSymbol.actionGaugeEquiv_of_bijective_smul,
        TNLean.Algebra.LSymbol.isCompatible_regular}
    \leanok
    \uses{def:mpug_l_symbols,def:mpug_l_gauge,def:mpug_three_cocycle}
    Let $G$ act freely and transitively on a nonempty set $X$.
    Any two L-symbols $L_1,L_2$ compatible with the same three-cochain
    $\omega$ are related by an action-tensor gauge, with the fusion gauge
    fixed to one. If $\omega$ is a three-cocycle, the regular action of $G$
    on itself admits the compatible solution $L^x_{g,h}=\omega(g,h,x)^{-1}$.
    This is the trivial-stabilizer case of
    \cite[Section~4.1]{GarreRubio2023MPOSym}, in the compatibility
    convention of Definition~\ref{def:mpug_l_symbols}.
\end{theorem}

\begin{proof}
    \leanok
    Fix $x_0\in X$, and write each $x\in X$ uniquely as $k\cdot x_0$.
    Set $\gamma_{g,k\cdot x_0}=L_2^{x_0}(g,k)/L_1^{x_0}(g,k)$.
    Compatibility gives, for $i=1,2$,
    \begin{align}
        \label{eq:asymex_free_gauge}
        L_i^{k\cdot x_0}(g,h)
        =\frac{L_i^{x_0}(g,hk)L_i^{x_0}(h,k)}
        {\omega(g,h,k)L_i^{x_0}(gh,k)}.
    \end{align}
    Taking the ratio in~(\ref{eq:asymex_free_gauge}) yields
    $L_1^x(g,h)=\gamma_{gh,x}L_2^x(g,h)/
        (\gamma_{g,h\cdot x}\gamma_{h,x})$.
    For the regular action, the cocycle equation directly verifies
    compatibility of $\omega(g,h,x)^{-1}$.
\end{proof}

\begin{theorem}[Unique four-block L-symbol class]
    \label{thm:asymex_klein_four_block_uniqueness}

    relabeling and action-tensor gauge, and it has four blocks.
    This is the L-symbol uniqueness assertion of
    \cite[Section~6]{GarreRubio2023MPOSym} for the $(1,1,1)$ class.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_four_blocks,thm:asymex_free_action_l_symbols,

\begin{theorem}[Normalization and involution signs]
    \label{thm:asymex_l_symbol_normalization}
    \lean{TNLean.Algebra.LSymbol.IsCompatible.isNormalized_gauge,

    action of $G$ on $X$. The action-tensor gauge
    $\gamma_{g,x}=L^x_{e,e}$, with fusion gauge equal to one, makes $L$
    normalized. For every three-cochain $\omega$, every normalized $L$
    compatible with $\omega$, every $x\in X$, and every $g\in G$ with
    $g^2=e$, one has
    $L^x_{g,g}=\omega(g,g,g)L^{g\cdot x}_{g,g}$.

    \cite[Section~6]{GarreRubio2023MPOSym}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:mpug_l_gauge_preservation,thm:mpug_block_factorization,

    $L^x_{g,e}=L^x_{e,e}/\omega(g,e,e)$ and
    $L^x_{e,g}=\omega(e,e,g)L^{g\cdot x}_{e,e}$. Since $\omega$ is normalized,
    these are $L^x_{g,e}=L^x_{e,e}$ and $L^x_{e,g}=L^{g\cdot x}_{e,e}$.

    Compatibility at $(g,g,g)$ reduces to the stated sign relation.
\end{proof}

\begin{theorem}[Diagonal L-symbols for the Klein-four representatives]

\begin{theorem}[Trivial restrictions of all Klein-four representatives]
    \label{thm:asymex_klein_subgroup_table}
    \lean{TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_isTrivialGaugeClass_comap_iff,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_isTrivialGaugeClass_comap_iff_generators}
    \leanok
    \uses{def:asymex_klein_cocycle_family,def:mpug_cochain_restriction,
        def:mpug_trivial_gauge_class}
    Let $H\leq\Z_2^2$, with $a=(1,0)$ and $b=(0,1)$.
    The restriction of $\omega_{pqr}$ to $H$ represents the trivial
    cohomology class if and only if $pu+qv+ruv=0$ for every $(u,v)\in H$.
    Equivalently, $H$ must avoid precisely the elements in the right column:
    \begin{center}
        \begin{tabular}{c@{\qquad}c}
            $(p,q,r)$ & forbidden elements of $H$ \\[2pt]
            $(0,0,0)$ & none                      \\
            $(1,0,0)$ & $a,ab$                    \\
            $(0,1,0)$ & $b,ab$                    \\
            $(1,1,0)$ & $a,b$                     \\
            $(0,0,1)$ & $ab$                      \\
            $(0,1,1)$ & $b$                       \\
            $(1,0,1)$ & $a$                       \\
            $(1,1,1)$ & $a,b,ab$
        \end{tabular}
    \end{center}
    This gives the subgroup restrictions in
    \cite[Section~6]{GarreRubio2023MPOSym}, including arbitrary changes
    of fusion gauge.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_cocycle_restrictions,thm:mpug_cyclic_invariant_gauge}
    If the restricted class is trivial, its cyclic invariant at every
    element of $H$ is one. Since each element has square equal to the
    identity and the representative is normalized, this invariant is
    $\omega_{pqr}(g,g,g)$; hence $pu+qv+ruv=0$.
    Conversely, write $Q(x)=px_1+qx_2+rx_1x_2$. A check over the finitely
    many $p,q,r\in\Z_2$ and $g,h,k\in\Z_2^2$ gives
    \begin{align}
        Q(g)=Q(h)=Q(k)=Q(g+h)=Q(g+k)=Q(h+k)=0
        \implies pg_1h_1k_1+qg_2h_2k_2+rg_1h_2k_2=0.
        \notag
    \end{align}
    Subgroup closure therefore makes
    the entire restriction identically one.
    Testing the identity and the three nonidentity elements gives the table.
\end{proof}

\begin{definition}[The one-qubit Klein-four symmetry]

        U_b\ket{t} & =\ket{\bar t}.\notag
    \end{align}
    Thus $U_a=\prod_i CZ_{i,i+1}Z_i$ and $U_b=\prod_i X_i$, as in
    \cite[Section~6]{GarreRubio2023MPOSym}. Define
    $U_{a^p b^q}=U_a^pU_b^q$ for $p,q\in\{0,1\}$.
    If $\widehat T^{ij}$ is the decorated CZX tensor, the tensors
    $A^{ij}=\widehat T^{i,1-j}$ and $B^{ij}=\delta_{i,1-j}$ have bond dimensions

\end{definition}

\begin{theorem}[The printed Klein-four representation]
    \label{thm:asymex_klein_spin_representation}
    \lean{KleinSymmetry.generatorA_mul_self,KleinSymmetry.generatorB_mul_self,
        KleinSymmetry.generators_commute,KleinSymmetry.generatorA_mul_generatorB,
        KleinSymmetry.operator_one,KleinSymmetry.operator_mul,
        KleinSymmetry.generatorA_mem_unitaryGroup,KleinSymmetry.generatorB_mem_unitaryGroup,
        KleinSymmetry.operator_mem_unitaryGroup,KleinSymmetry.mpo_tensorA,KleinSymmetry.mpo_tensorB,
        KleinSymmetry.representation}
    \leanok
    \uses{def:asymex_klein_spin_symmetry}
    At every positive length, $U_a^2=U_b^2=I$ and $U_aU_b=U_bU_a$.
    Every $U_g$ is unitary, $U_e=I$, and $U_gU_h=U_{gh}$, so $g\mapsto U_g$
    is a representation of $\Z_2\times\Z_2$ on the state space.
    The periodic operators of $A$ and $B$ are $U_a$ and $U_b$.
    Their product is the decorated CZX operator.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_czx_decorated_operator}
    The diagonal entries of $U_a$ are signs, and the spin flip is an
    involution, so both generators square to the identity and are unitary.
    By Theorem~\ref{thm:asymex_czx_decorated_operator},
    $O_N(A)(s,t)=O_N(\widehat T)(s,\bar t)
        =\delta_{s,t}(-1)^{\sum_i(s_is_{i+1}+s_i)}=U_a(s,t)$, and
    $O_N(B)(s,t)=\Id(s,\bar t)=\delta_{s,\bar t}=U_b(s,t)$. By (\ref{eq:asymex_klein_spin_generators}), $U_aU_b$ is the
    decorated CZX operator, whose square is $I$. Multiplying
    $(U_aU_b)^2=I$ on the left by $U_a$ and on the right by $U_b$ proves
    commutation. The multiplication law follows by adding exponents
    modulo two.
\end{proof}

\begin{definition}[The two invariant state spaces]
    \label{def:asymex_klein_spin_spaces}
    \lean{KleinSymmetry.constantSpinSpace,KleinSymmetry.clusterPairSpace}
    \leanok
    \uses{def:asymex_klein_spin_symmetry}
    Put $z_0=\ket{0}^{\otimes N}$, $z_1=\ket{1}^{\otimes N}$,
    $u=\sum_t\ket{t}$, and $v=U_au$. The two spaces of
    \cite[Section~6]{GarreRubio2023MPOSym} are
    $S_z=\operatorname{span}\{z_0,z_1\}$ and
    $S_c=\operatorname{span}\{u,v\}$.
    Multiplying $u$ and $v$ by $2^{-N/2}$ gives the normalized uniform
    and cluster-like vectors used there, without changing $S_c$.
\end{definition}

\begin{theorem}[Action on the two state pairs]
    \label{thm:asymex_klein_spin_spaces}
    \lean{KleinSymmetry.generatorA_mulVec_constantBasis,
        KleinSymmetry.generatorB_mulVec_constantBasis,KleinSymmetry.generatorB_mulVec_const,
        KleinSymmetry.generatorA_mulVec_cluster,KleinSymmetry.generatorB_mulVec_cluster,
        KleinSymmetry.generators_preserve_constantSpinSpace,
        KleinSymmetry.generators_preserve_clusterPairSpace}
    \leanok
    \uses{def:asymex_klein_spin_spaces}
    Both $S_z$ and $S_c$ are invariant under $U_a$ and $U_b$. More precisely,
    for $j\in\{0,1\}$,
    \begin{align}
        \label{eq:asymex_klein_spin_actions}
        U_a z_j & =z_j,           \\
        U_b z_j & =z_{1-j},\notag \\
        U_a v   & =u,\notag       \\
        U_b u   & =u,\notag       \\
        U_b v   & =v.\notag
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_spin_representation}
    The exponent in (\ref{eq:asymex_klein_spin_generators}) is zero on
    the all-zero configuration and $2N$ on the all-one configuration.
    Since $U_b$ permutes configurations by the global flip with no sign, it
    fixes the uniform vector directly: $U_bu=u$. The same flip exchanges the
    all-zero and all-one configurations, giving $U_bz_j=z_{1-j}$.
    Since $v=U_au$, the identities $U_a^2=I$ and $U_bU_a=U_aU_b$ then give $U_av=u$ and $U_bv=v$.
    Linearity proves invariance of both spans.
\end{proof}

\begin{theorem}[Normalization of scalar three-cocycles]
    \label{thm:asymex_three_cocycle_normalization}
    \lean{TNLean.Algebra.ScalarThreeCochain.exists_isNormalized_cohomologousTo}
    \leanok
    \uses{def:mpug_three_cocycle,def:mpug_scalar_cochains,def:mpug_fusion_gauge,
        def:mpug_cohomologous_to}
    Let $G$ be a small group, whose underlying set lies in a fixed
    set-theoretic universe. Every three-cocycle $\omega:G^3\to\C^\times$ is cohomologous to a
    normalized one. One may use the two-cochain
    $\beta(g,h)=\omega(g,e,e)/\omega(e,e,h)$ and replace $\omega$ by
    $(d\beta)\omega$, where
    $d\beta(g,h,k)=\beta(g,hk)\beta(h,k)/(\beta(g,h)\beta(gh,k))$.
\end{theorem}

\begin{proof}
    \leanok
    The cocycle equation gives $\omega(e,e,e)=1$ and
    \begin{align}
        \label{eq:asymex_three_cocycle_axes}
        \omega(g,h,e) & =\frac{\omega(gh,e,e)}{\omega(h,e,e)},       \\
        \omega(e,g,h) & =\frac{\omega(e,e,gh)}{\omega(e,e,g)},\notag \\
        \omega(g,e,h) & =\omega(g,e,e)\omega(e,e,h).\notag
    \end{align}
    Substitution of (\ref{eq:asymex_three_cocycle_axes}) into $d\beta$
    makes each identity-axis value of $(d\beta)\omega$ equal to one.
\end{proof}

\begin{theorem}[Triviality from the Klein diagonal signs]
    \label{thm:asymex_klein_trivial_diagonal}
    \lean{TNLean.Algebra.ScalarThreeCochain.isTrivialGaugeClass_of_klein_diagonal_eq_one}
    \leanok
    \uses{def:mpug_three_cocycle,def:mpug_scalar_cochains,def:mpug_trivial_gauge_class}
    A normalized three-cocycle on $G=\{e,a,b,c=ab\}\cong\Z_2^2$ whose
    diagonal values $\omega(g,g,g)$ are all one is a coboundary.
    This is the trivial-class case of the classification in
    \cite[Section~6]{GarreRubio2023MPOSym}.
\end{theorem}

\begin{proof}
    \leanok
    Set $(s_1,s_2,s_3,s_4,s_5,s_6)=
        (\omega(a,a,b),\omega(a,b,a),\omega(a,b,b),\omega(a,b,c),
        \omega(b,a,a),\omega(b,a,b))$.
    Choose $t\in\C^\times$ with $t^2=s_1s_5/s_2$ and define a normalized
    two-cochain by its values on $\{a,b,c\}^2$:
    \begin{align}
        \label{eq:asymex_klein_trivial_gauge}
        (\beta(g,h))_{g,h\in\{a,b,c\}}
         & =\begin{pmatrix}
                ts_4/(s_3s_6)   & 1/s_3 & ts_1s_4/s_6 \\
                1/(ts_3)        & 1     & s_6/t       \\
                s_1s_4/(s_2s_6) & 1     & 1
            \end{pmatrix}.
    \end{align}
    The six corresponding values of $d\beta$ are $s_1,\ldots,s_6$.
    Hence $\eta=\omega/(d\beta)$ has those six values equal to one.
    Its diagonal values remain one because $g^2=e$ and $\beta$ is normalized.
    Applying the cocycle equation successively to the remaining triples
    gives one for all eighteen other nonidentity entries. Thus $\eta=1$
    and $\omega=d\beta$.
\end{proof}

\begin{theorem}[Complete classification of Klein three-cocycles]
    \label{thm:asymex_klein_complete_classification}
    \lean{TNLean.Algebra.ScalarThreeCochain.cohomologousTo_of_klein_diagonal_eq,
        TNLean.Algebra.ScalarThreeCochain.cohomologousTo_iff_klein_cyclicInvariant_eq,
        TNLean.Algebra.ScalarThreeCochain.cohomologousTo_iff_klein_three_cyclicInvariants,
        TNLean.Algebra.ScalarThreeCochain.exists_cohomologousTo_kleinCocycleFamily,
        TNLean.Algebra.ScalarThreeCochain.existsUnique_cohomologousTo_kleinCocycleFamily,
        TNLean.Algebra.ScalarThreeCochain.kleinCocycleFamily_add,
        TNLean.Algebra.ScalarThreeCochain.kleinAnomalyClass,
        TNLean.Algebra.ScalarThreeCochain.kleinAnomalyClass_bijective,
        TNLean.Algebra.ScalarThreeCochain.kleinH3Equiv}
    \leanok
    \uses{def:asymex_klein_cocycle_family,def:mpug_cohomologous_to,
        def:mpug_h3_group_cohomology,def:mpug_three_cocycle,def:mpug_scalar_cochains}
    Every three-cocycle on $\Z_2^2$ is cohomologous to exactly one
    $\omega_{pqr}$. The correspondence $[\omega_{pqr}]\mapsto(p,q,r)$ is
    an isomorphism $H^3(\Z_2\times\Z_2,\C^\times_{\mathrm{triv}})\cong\Z_2^3$.
    Two arbitrary three-cocycles $\omega,\nu$ are cohomologous if and only if
    \begin{align}
        \label{eq:asymex_klein_complete_detectors}
        \omega(g,e,g)\omega(g,g,g) & =\nu(g,e,g)\nu(g,g,g)
    \end{align}
    at $g=a,b,c$. For normalized cocycles these products reduce to the
    diagonal values. This is the classification and detector assertion
    of \cite[Section~6]{GarreRubio2023MPOSym} with $\C^\times$ coefficients;
    the source states it for $U(1)$, and the degree-three comparison of the
    two coefficient groups is not formalized
    here~\cite{gap:glm23_klein_h3_circle_coefficients}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_three_cocycle_normalization,
        thm:asymex_klein_trivial_diagonal,thm:asymex_klein_distinct_classes,
        thm:mpug_cyclic_invariant_gauge,thm:mpug_h3_group_cohomology}
    Normalize both cocycles. Their quotient has trivial diagonal precisely
    when their three signs agree, so
    Theorem~\ref{thm:asymex_klein_trivial_diagonal} proves sufficiency of
    (\ref{eq:asymex_klein_complete_detectors}). Necessity follows from
    invariance of the cyclic products under coboundaries.
    The cocycle equation at $(g,g,g,g)$ makes each normalized diagonal
    value a sign. If the signs are $(-1)^p,(-1)^q,(-1)^t$, the matching
    representative is $\omega_{p,q,p+q+t}$.
    Theorem~\ref{thm:asymex_klein_distinct_classes} proves uniqueness.
    Finally $\omega_{pqr}\omega_{p'q'r'}=\omega_{p+p',q+q',r+r'}$, so
    this bijection of cohomology classes is a group isomorphism.
\end{proof}

\begin{definition}[The four printed tensors]
    \label{def:asymex_klein_printed_family}
    \lean{KleinSymmetry.printedBond,KleinSymmetry.printedTensor,KleinSymmetry.printedFamily}
    \leanok
    \uses{def:asymex_klein_spin_symmetry}
    For $G=\{e,a,b,ab\}$, let $T_e$ be the bond-one identity tensor,
    $T_a=A$ the diagonal tensor, $T_b=B$ the bond-one spin-flip tensor,
    and $T_{ab}=\widehat T$ the decorated CZX tensor.
\end{definition}

\begin{theorem}[Normality and anomaly of the printed symmetry]
    \label{thm:asymex_klein_printed_anomaly}
    \lean{KleinSymmetry.tensorA_isNormal,KleinSymmetry.mpo_printedFamily,
        KleinSymmetry.printedFamily_isNormalRepresentation,
        KleinSymmetry.diagonalHom,KleinSymmetry.printedFamily_comap_diagonal,
        KleinSymmetry.cyclicInvariant_omega_printedFamily,
        KleinSymmetry.not_isTrivialGaugeClass_omega_printedFamily,
        KleinSymmetry.exists_normalized_omega_printedFamily,
        KleinSymmetry.anomalyClass_omega_printedFamily_ne_zero}
    \leanok
    \uses{def:asymex_klein_printed_family,def:mpug_anomaly_three_cochain,
        def:mpug_trivial_gauge_class,def:mpug_h3_group_cohomology}
    For every $N>0$ and $g\in G$, the periodic operator of $T_g$ is the
    printed operator, $O_N(T_g)=U_g$. The four tensors are normal and form
    an exact representation on every positive-length periodic chain.
    For every choice of fusion tensors,
    the anomaly satisfies
    \begin{align}
        \label{eq:asymex_klein_printed_anomaly}
        \omega(ab,e,ab)\omega(ab,ab,ab) & =-1.
    \end{align}
    Its class $[\omega]\in H^3(\Z_2\times\Z_2,\C^\times_{\mathrm{triv}})$
    is nonzero and admits a normalized representative $\nu$
    with $\nu(ab,ab,ab)=-1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_spin_representation,
        thm:asymex_czx_decorated_normal,
        thm:asymex_czx_decorated_anomaly_class,
        thm:mpug_cyclic_invariant_gauge,
        thm:asymex_three_cocycle_normalization,
        thm:mpug_h3_group_cohomology,thm:mpug_family_restriction_omega}
    The equalities $O_N(T_g)=U_g$ are
    Theorem~\ref{thm:asymex_klein_spin_representation} for $g=a,b,ab$ and
    the identity tensor for $g=e$. On the physical alphabet
    $\{0,1\}^2$ let $\varphi(i,j)=(i,1-j)$. Then
    $A^{\varphi(i,j)}=\widehat T^{ij}$, so $A^{\varphi\circ w}=\widehat T^{w}$ for every
    word $w$ and
    $\operatorname{span}\{A^{v}:|v|=n\}\supseteq\operatorname{span}\{\widehat T^{w}:|w|=n\}$.
    Normality of $\widehat T$ therefore gives normality of the diagonal tensor. The identity and spin-flip
    tensors have bond dimension one. The operator multiplication law
    follows from the commuting involutions.
    Restriction to $\{e,ab\}$ gives exactly the decorated CZX family; let
    $\omega_{\mathrm{CZX}}$ be the anomaly of its fusion tensors and write
    $ab$ also for the generator of $\{e,ab\}$. Restricting the fusion
    tensors (Theorem~\ref{thm:mpug_family_restriction_omega}) and using
    invariance of the cyclic product under a change of fusion tensors
    (Theorem~\ref{thm:mpug_cyclic_invariant_gauge}) gives
    \begin{align}
        \omega(ab,e,ab)\,\omega(ab,ab,ab)
         & =\omega_{\mathrm{CZX}}(ab,e,ab)\,\omega_{\mathrm{CZX}}(ab,ab,ab)=-1,
        \notag
    \end{align}
    the last value by Theorem~\ref{thm:asymex_czx_decorated_anomaly_class}.
    This proves (\ref{eq:asymex_klein_printed_anomaly}). A trivial class would have
    cyclic product one, so the gauge class is nontrivial; by
    Theorem~\ref{thm:mpug_h3_group_cohomology} the class
    $[\omega]\in H^3(\Z_2\times\Z_2,\C^\times_{\mathrm{triv}})$ is nonzero. Finally normalization makes $\nu(ab,e,ab)=1$.
\end{proof}

\begin{theorem}[The anomaly class of the printed symmetry]
    \label{thm:asymex_klein_printed_anomaly_class}
    \lean{KleinSymmetry.aHom,KleinSymmetry.printedFamily_comap_aHom,
        KleinSymmetry.tensorA_isReduction,KleinSymmetry.aFusionData,
        KleinSymmetry.aFusionData_isAssociator,
        KleinSymmetry.bHom,KleinSymmetry.printedFamily_comap_bHom,
        KleinSymmetry.bFusionData,
        KleinSymmetry.cyclicInvariant_omega_printedFamily_a,
        KleinSymmetry.cyclicInvariant_omega_printedFamily_b,
        KleinSymmetry.omega_printedFamily_cohomologousTo,
        KleinSymmetry.diagonal_of_cohomologousTo_omega_printedFamily,
        KleinSymmetry.kleinH3Equiv_anomalyClass_omega_printedFamily}
    \leanok
    \uses{def:asymex_klein_printed_family,def:asymex_klein_cocycle_family,
        def:mpug_anomaly_three_cochain,def:mpug_cohomologous_to,def:mpug_h3_group_cohomology}
    For every choice of fusion tensors of the four printed tensors,
    \begin{align}
        \label{eq:asymex_klein_printed_cyclic}
        \omega(a,e,a)\,\omega(a,a,a) & =1, &
        \omega(b,e,b)\,\omega(b,b,b) & =1.
    \end{align}
    Hence $\omega$ is cohomologous to $\omega_{001}$, its class corresponds
    to $(0,0,1)$ under
    $H^3(\Z_2\times\Z_2,\C^\times_{\mathrm{triv}})\cong\Z_2^3$, and every
    normalized cocycle $\nu$ cohomologous to $\omega$ has
    \begin{align}
        \label{eq:asymex_klein_printed_diagonal}
        \bigl(\nu(a,a,a),\nu(b,b,b),\nu(ab,ab,ab)\bigr) & =(1,1,-1).
    \end{align}
    This is the statement of \cite[Section~6]{GarreRubio2023MPOSym} that in
    this realization ``the only non-trivial $3$-cocycle element is
    $\omega(ab,ab,ab)=-1$'', read through the diagonal values that label the
    classes; read entrywise, the printed sentence describes no
    cocycle~\cite{gap:glm23_klein_printed_anomaly_scope}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_printed_anomaly,thm:mpug_family_restriction_omega,
        thm:mpug_cyclic_invariant_gauge,thm:asymex_klein_complete_classification}
    As in the proof of Theorem~\ref{thm:asymex_klein_printed_anomaly},
    each cyclic product is computed on a restriction of the family with
    chosen fusion tensors. On $\{e,b\}$ all tensors have bond dimension one,
    $B$ squares to the identity tensor letter by letter, and the one-by-one
    identity fusion tensors make both fusion trees equal, so the anomaly is
    one. On $\{e,a\}$ the letters of $A$ are $A^{00}=u_0e_0^{\mathsf T}$ and
    $A^{11}=u_1e_1^{\mathsf T}$ with $u_0=(1,1)$ and $u_1=(-1,1)$, the other
    letters vanishing. Take the fusion tensors of $(a,a)$ to be
    $\langle 00|$ and $|00\rangle+|11\rangle$, and the identity whenever a
    factor is $e$. Every nonzero letter of the stacked square of $A$ is
    $(u_i\otimes u_i)(e_i\otimes e_i)^{\mathsf T}$, and
    $\langle 00|u_i\otimes u_i\rangle=(e_i\otimes e_i)^{\mathsf T}(|00\rangle+|11\rangle)=1$,
    so these fusion tensors reduce the stacked square to the identity tensor,
    and the two fusion trees $\langle 00|\otimes 1$ and $1\otimes\langle 00|$
    agree on every letter $u_i\otimes u_i\otimes u_i$ of the triple product.
    Hence $\omega(a,e,a)\,\omega(a,a,a)=1$. Together with
    (\ref{eq:asymex_klein_printed_anomaly}), the three cyclic products are
    those of $\omega_{001}$, and
    Theorem~\ref{thm:asymex_klein_complete_classification} identifies the
    class. For normalized $\nu$ the cyclic products are the diagonal values.
\end{proof}

\begin{theorem}[L-symbol relations of the printed symmetry]
    \label{thm:asymex_klein_printed_l_symbols}
    \lean{KleinSymmetry.lSymbol_signs_printedFamily}
    \leanok
    \uses{def:asymex_klein_printed_family,def:mpug_anomaly_three_cochain,
        def:mpug_l_symbols,def:mpug_l_normalized}
    Let $\nu$ be a normalized cocycle cohomologous to the anomaly of the
    printed symmetry, and let $L$ be normalized L-symbols compatible with
    $\nu$ for an action of $\Z_2\times\Z_2$ on a set of blocks. For every
    block $x$,
    \begin{align}
        \label{eq:asymex_klein_printed_l_signs}
        L^x_{ab,ab} & =-L^{ab\cdot x}_{ab,ab}, &
        L^x_{a,a}   & =L^{a\cdot x}_{a,a},     &
        L^x_{b,b}   & =L^{b\cdot x}_{b,b}.
    \end{align}
    In the two-block phases with unbroken subgroup $H_a$ or $H_b$, the element
    $ab$ exchanges the blocks, which is the relation
    $L^x_{ab,ab}=-L^y_{ab,ab}$ of \cite[Section~6]{GarreRubio2023MPOSym}. The
    relation for $a$ relates the two blocks in the phase $H_b$, and that for
    $b$ in the phase $H_a$; the source attaches them to $H_a$ and $H_b$
    respectively~\cite{gap:glm23_klein_printed_anomaly_scope}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_klein_printed_anomaly_class,thm:asymex_l_symbol_normalization}
    Each nonidentity element is an involution, so
    Theorem~\ref{thm:asymex_l_symbol_normalization} gives
    $L^x_{g,g}=\nu(g,g,g)L^{g\cdot x}_{g,g}$. Substitute the diagonal values
    (\ref{eq:asymex_klein_printed_diagonal}).
\end{proof}
