\section{The interaction picture in Hastings' approximate ground-state projector}
\label{sec:hastings_interaction_picture}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 843--901 (step five of the
% proof of Lemma 1). Finite complex matrices; no lattice content.

The last step of the proof of Lemma~1 of \cite{Hastings2007AreaLaw} writes the
evolution generated by $M_L+M_B+M_R$ as a time-ordered exponential of the
interaction-picture boundary term, multiplied by the evolution generated by
$M_L+M_R$. This section states the ingredients for finite complex matrices
$A$ and $B$, with $A$ in the role of $M_L+M_R$ and $B$ in the role of $M_B$.
Norms are operator norms.

\begin{definition}[Interaction-picture propagator]
    \label{def:hastings_interaction_propagator}
    \lean{TNLean.MPS.AreaLaw.Hastings.interactionGenerator,
        TNLean.MPS.AreaLaw.Hastings.interactionPropagator}
    \leanok
    For complex square matrices $A$ and $B$ and a real number $t$, put
    \begin{align*}
        B(t) & =e^{iAt}Be^{-iAt}, & U(t) & =e^{i(A+B)t}e^{-iAt}.
    \end{align*}
\end{definition}

\begin{theorem}[Interaction-picture factorization]
    \label{thm:hastings_interaction_picture}
    \lean{TNLean.MPS.AreaLaw.Hastings.interactionPropagator_hasDerivAt,
        TNLean.MPS.AreaLaw.Hastings.eq_interactionPropagator_of_hasDerivAt,
        TNLean.MPS.AreaLaw.Hastings.interactionPropagator_mem_unitaryGroup,
        TNLean.MPS.AreaLaw.Hastings.hermitianUnitaryPath_add_eq_interactionPropagator_mul}
    \leanok
    \uses{def:hastings_interaction_propagator}
    Let $A$ and $B$ be complex square matrices. The path $U$ is differentiable,
    $U(0)=1$ and
    \begin{align*}
        U'(t)=U(t)\,iB(t)
    \end{align*}
    for every real $t$, and every differentiable path $W$ with $W(0)=1$ and
    $W'(t)=W(t)\,iB(t)$ for all $t$ equals $U$. Moreover
    \begin{align*}
        e^{i(A+B)t}=U(t)e^{iAt},
    \end{align*}
    and if $A$ and $B$ are Hermitian then $U(t)$ is unitary. Thus $U(t)$ is the
    time-ordered exponential of $i\int_0^tB(t')\,dt'$ of
    \cite[lines 853--864]{Hastings2007AreaLaw}, with later times to the right.
\end{theorem}
\begin{proof}\leanok
    Differentiating the product gives
    $U'(t)=e^{i(A+B)t}\,iB\,e^{-iAt}=U(t)\,iB(t)$, and $U(0)=1$. Put
    $V(t)=e^{iAt}e^{-i(A+B)t}$, so that $V'(t)=-iB(t)V(t)$ and $V(t)U(t)=1$.
    For a solution $W$, the derivative of $W(t)V(t)$ is
    $W(t)\,iB(t)V(t)-W(t)\,iB(t)V(t)=0$, so $W(t)V(t)=W(0)V(0)=1$ and
    $W(t)=W(t)V(t)U(t)=U(t)$. The factorization follows from
    $e^{-iAt}e^{iAt}=1$, and $U(t)$ is a product of two unitaries when $A$ and
    $A+B$ are Hermitian.
\end{proof}

\begin{lemma}[Cut-off commutation estimate]
    \label{lem:hastings_cutoff_commutation}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_hermitianUnitaryPath_mul_sub_le}
    \leanok
    Let $K$ be a Hermitian matrix and $X$ a matrix of the same size. For every
    real $t$,
    \begin{align*}
        \|e^{iKt}X-X\|\leq|t|\,\|KX\|.
    \end{align*}
    In \cite[lines 878--880]{Hastings2007AreaLaw} this is applied with $K=M_L+M_R$
    and $X=O_LO_R$.
\end{lemma}
\begin{proof}\leanok
    The derivative of $s\mapsto e^{iKs}X$ is $e^{iKs}\,iKX$, whose norm is
    $\|KX\|$ because $e^{iKs}$ is unitary. The mean value inequality on the
    interval between $0$ and $t$ gives the bound.
\end{proof}

\begin{definition}[Gaussian average of the propagator]
    \label{def:hastings_interaction_average}
    \lean{TNLean.MPS.AreaLaw.Hastings.interactionAverage,
        TNLean.MPS.AreaLaw.Hastings.interactionAverage_eq_integral}
    \leanok
    \uses{def:hastings_interaction_propagator}
    For a variance $h\geq0$, let $\gamma_h$ be the centered Gaussian measure
    of variance $h$ on the real line, and put
    \begin{align*}
        P_B=\int U(t)\,d\gamma_h(t).
    \end{align*}
    With $h=q/\Delta E^2$, the density of $\gamma_h$ is
    $(\Delta E/\sqrt{2\pi q})e^{-(t\Delta E)^2/2q}$, and $P_B$ is the operator
    of \cite[lines 891--897]{Hastings2007AreaLaw}.
\end{definition}

\begin{lemma}[The Gaussian average is a contraction]
    \label{lem:hastings_interaction_average_norm}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_interactionAverage_le}
    \leanok
    \uses{def:hastings_interaction_average,thm:hastings_interaction_picture}
    If $A$ and $B$ are Hermitian, then $\|P_B\|\leq1$ for every variance.
\end{lemma}
\begin{proof}\leanok
    Each $U(t)$ is unitary, and $\gamma_h$ is a probability measure.
\end{proof}

\begin{theorem}[Moving the cut-off past the boundary evolution]
    \label{thm:hastings_interaction_average_last}
    \lean{TNLean.MPS.AreaLaw.Hastings.norm_interactionAverage_mul_sub_le,
        TNLean.MPS.AreaLaw.Hastings.norm_interactionAverage_mul_mul_sub_le}
    \leanok
    \uses{def:hastings_interaction_average,thm:hastings_interaction_picture,
        lem:hastings_cutoff_commutation,lem:hastings_gaussian_absolute_moment}
    Let $A$ and $B$ be Hermitian, let $X$ and $P_0$ be matrices of the same
    size, and let $h\geq0$. Then
    \begin{align*}
        \Bigl\|P_BX-P_0\Bigr\|
        \leq\Bigl\|\int e^{i(A+B)t}\,d\gamma_h(t)\,X-P_0\Bigr\|
        +\sqrt h\,\|AX\|.
    \end{align*}
    With $A=M_L+M_R$, $B=M_B$, $X=O_LO_R$ and $h=q/\Delta E^2$, this is the
    step from the estimate with $e^{i(M_L+M_B+M_R)t}$ to eq.~(last) of
    \cite[lines 881--890]{Hastings2007AreaLaw}: if
    $\|(M_L+M_R)O_LO_R\|\leq\delta$ and
    $\|\int e^{i(M_L+M_B+M_R)t}\,d\gamma_h(t)\,O_LO_R-P_0\|\leq\varepsilon$,
    then $\|P_BO_LO_R-P_0\|\leq\varepsilon+\sqrt h\,\delta$.
\end{theorem}
\begin{proof}\leanok
    By the factorization,
    $e^{i(A+B)t}X-U(t)X=U(t)\bigl(e^{iAt}X-X\bigr)$, whose norm is at most
    $|t|\,\|AX\|$ by the cut-off commutation estimate, since $U(t)$ is
    unitary. Integrating against $\gamma_h$ and using the first absolute
    moment bound gives
    $\bigl\|\int e^{i(A+B)t}\,d\gamma_h\,X-P_BX\bigr\|\leq\sqrt h\,\|AX\|$,
    and the triangle inequality concludes.
\end{proof}
