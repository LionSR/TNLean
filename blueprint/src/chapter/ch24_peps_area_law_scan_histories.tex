\section{Finite collar histories and good-history sampling}
\label{sec:al_finite_scan_histories}

Fix a finite domain $\Lambda\subset\mathbb Z^2$, a color $A\subset\Lambda$,
and a nonempty finite target $T\subset\mathbb Z^2$. Write
$d(x)=\min_{t\in T}\|x-t\|_\infty$ and $T_j=T+[-j,j]^2_{\mathbb Z}$, with
$d_\Lambda$ the induced nearest-neighbor graph metric on $\Lambda$, following
\cite[Section~9, moving partitions]{OpenAI2026AreaLaw}.

\begin{definition}[Physical partitions and padded fill slots]
    \label{def:al_scan_physical_partition}
    \lean{TNLean.PEPS.AreaLaw.Scan.PhysicalPartition,
        TNLean.PEPS.AreaLaw.Scan.middle,
        TNLean.PEPS.AreaLaw.Scan.receiving,
        TNLean.PEPS.AreaLaw.Scan.assign,
        TNLean.PEPS.AreaLaw.Scan.initialPartition,
        TNLean.PEPS.AreaLaw.Scan.orientedDepth,
        TNLean.PEPS.AreaLaw.Scan.initialFront,
        TNLean.PEPS.AreaLaw.Scan.fillCount,
        TNLean.PEPS.AreaLaw.Scan.fillSide,
        TNLean.PEPS.AreaLaw.Scan.nominalFront,
        TNLean.PEPS.AreaLaw.Scan.depthRow,
        TNLean.PEPS.AreaLaw.Scan.orientedRow,
        TNLean.PEPS.AreaLaw.Scan.fillSlot,
        TNLean.PEPS.AreaLaw.Scan.fill,
        TNLean.PEPS.AreaLaw.Scan.charge}
    \leanok
    A physical partition $(P_0,Y,V)$ of $\Lambda$ into near, middle, and far
    sides starts at cutoffs $u<v$ with
    \begin{align}
        P_0 & =\{x\in A:d(x)\le u\},                      &
        V   & =(\Lambda\setminus A)\cup\{x\in A:d(x)>v\}, &
        Y   & =\Lambda\setminus(P_0\cup V). \notag
    \end{align}
    Order each positive-depth row once and pad it to $n$ slots (entries outside
    $A$ or absent are blank); the far side visits rows in decreasing depth.
    Fills alternate near and far starting near: after $k$ fills,
    \begin{align}
        t_P & =\lfloor(k+1)/2\rfloor,    &
        t_F & =\lfloor k/2\rfloor,       &
        j_P & =u+1+\lfloor t_P/n\rfloor, &
        j_F & =-v+\lfloor t_F/n\rfloor.
        \label{eq:al_scan_nominal_fronts}
    \end{align}
    Moving $B\subseteq\Lambda$ to receiving side $S\in\{P_0,V\}$ transfers
    $B\cap Y$ to $S$; a charge moves $B\cap Y$ to $S$ only if $B\cap S\ne\varnothing$
    and $B\cap Y\ne\varnothing$.
\end{definition}

    side's consumed count by $1$. Assigned sites never change sides, $Y$ is

\end{proof}

    $L+2r_0$, contradicting $\operatorname{dist}_\infty(T,Z)>2L+10r_0$.
\end{proof}

    Lemma~\ref{thm:al_scan_charge_capacity}.

\begin{lemma}[Uniform rough upper bound for replica labels]
    \label{thm:al_scan_actual_rough_upper}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.eventually_actualScan_suppliedPrevector_rough_upper}
    \leanok
    \uses{def:al_scan_comparison_prefixes,def:al_scan_actual_fill_transport,
        def:al_scan_actual_charge_data}
    Suppose $n,m,M>0$, $r_0\le D$, $1\le D$, $4D\le m$, $8Km\le L$,
    $|T_j\setminus T_{j-1}|\le n$ ($1\le j\le L$), $q\ge1$, $a>0$, and a unit
    physical vector $\widetilde\Omega$ satisfies
    $S_{\widetilde\Omega}(Q_j)\le B_{\mathrm{sh}}$ ($0\le j\le L$) for
    $Q_j=A\cap(T_j\setminus T)$. Let $E$ be a set of Schmidt indices at
    $X=A\cap T$ of mass $z>0$ and size $d$, with normalized Schmidt truncation
    $\psi$ and Young labels $\lambda_k$ satisfying
    \begin{align}
        \|\pi_R^{\lambda_k}\psi^{\otimes k}\|^2
         & \ge\frac{1}{2(k+1)^{d^2}}
        \label{eq:al_scan_rough_upper_isotypic_floor}
    \end{align}
    for large $k$, and set
    $\xi_k=\pi_R^{\lambda_k}(\widetilde\Omega\otimes\mathrm{Bell}_{CR})^{\otimes k}$
    with band exponent $a/2$. Then there exists $c\ge0$ such that for all
    sufficiently large $k$, every round $s+1\le nm$, $p\in\mathbb R$, and fill
    or charge root $\mathcal M_{s,k}(p)$ (clamped to $[0,1]$),
    \begin{align}
        N_{s,k}(p)^2
         & :=\|\mathcal M_{s,k}(p)^{-1/4}\xi_k\|^2>0, \notag                        \\
        -\frac{\log N_{s,k}(p)^2}{k}
         & \le 2\log d-\log z
        +aK\left(\frac{B_{\mathrm{sh}}}{z}+1+3nD\log q\right)+\varepsilon_k, \notag \\
        \varepsilon_k
         & :=\frac{\log4+(d^2+cK)\log(k+2)}{k}\longrightarrow0.
        \label{eq:al_scan_actual_rough_upper}
    \end{align}
\end{lemma}
\begin{proof}\leanok

    comparator bound across the finite family of terminal partitions yields
    (\ref{eq:al_scan_actual_rough_upper}).
\end{proof}
