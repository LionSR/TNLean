\section{Finite collar histories and good-history sampling}
\label{sec:al_finite_scan_histories}

Fix a finite domain $\Lambda\subset\mathbb Z^2$, a color $A\subset\Lambda$,
and a nonempty finite target $T\subset\mathbb Z^2$. Write
$d(x)=\min_{t\in T}\|x-t\|_\infty$ and $T_j=T+[-j,j]^2_{\mathbb Z}$.
The graph metric is that of the induced nearest-neighbor graph on $\Lambda$.
The following finite construction follows
\cite[Section~9, moving partitions]{OpenAI2026AreaLaw}.

\begin{definition}[Physical partitions and padded fill slots]
    \label{def:al_scan_physical_partition}
    \lean{TNLean.PEPS.AreaLaw.Scan.PhysicalPartition,
        TNLean.PEPS.AreaLaw.Scan.middle,
        TNLean.PEPS.AreaLaw.Scan.receiving,
        TNLean.PEPS.AreaLaw.Scan.assign,
        TNLean.PEPS.AreaLaw.Scan.initialPartition,
        TNLean.PEPS.AreaLaw.Scan.orientedDepth,
        TNLean.PEPS.AreaLaw.Scan.initialFront,
        TNLean.PEPS.AreaLaw.Scan.fillCount,
        TNLean.PEPS.AreaLaw.Scan.fillSide,
        TNLean.PEPS.AreaLaw.Scan.nominalFront,
        TNLean.PEPS.AreaLaw.Scan.depthRow,
        TNLean.PEPS.AreaLaw.Scan.orientedRow,
        TNLean.PEPS.AreaLaw.Scan.fillSlot,
        TNLean.PEPS.AreaLaw.Scan.fill,
        TNLean.PEPS.AreaLaw.Scan.charge}
    \leanok
    A physical partition assigns each site to the near side $P_0$, the middle
    $Y$, or the far side $V$. For cutoffs $u<v$, initially
    $P_0=\{x\in A:d(x)\le u\}$ and
    $V=(\Lambda\setminus A)\cup\{x\in A:d(x)>v\}$.
    Order each positive-depth row once and pad it to $n$ slots. Sites outside
    $A$ and absent entries are blank. The far side visits rows in decreasing
    depth, using the same fixed order within each row.
    Fills alternate near and far, starting near. After $k$ fills the two
    consumed slot counts are $t_P=\lfloor(k+1)/2\rfloor$ and
    $t_F=\lfloor k/2\rfloor$; the oriented fronts are
    $j_P=u+1+\lfloor t_P/n\rfloor$ and $j_F=-v+\lfloor t_F/n\rfloor$.
    A move of a set $B$ transfers only $B\cap Y$ to its receiving side.
    A charge transfers this set only if $B$ meets both that side and $Y$.
\end{definition}

\begin{theorem}[Consumed slots and the middle region]
    \label{thm:al_scan_consumed_slots}
    \lean{TNLean.PEPS.AreaLaw.Scan.assign_eq_none,

        TNLean.PEPS.AreaLaw.Scan.fill_at_side_index,

        TNLean.PEPS.AreaLaw.Scan.initial_middle_depth,
        TNLean.PEPS.AreaLaw.Scan.nominalFront_le_of_unconsumed}
    \leanok
    \uses{def:al_scan_physical_partition}
    A blank or already assigned fill slot makes no physical move but increases
    its receiving side's consumed count by one. Assignments never change sides,
    and the middle only decreases. For a selected set $B$, the removed middle
    is $B\cap Y$. If all rows have at most $n$ sites, a site remaining in the
    middle after $k$ fills has oriented depth at least the corresponding
    nominal front.
\end{theorem}
\begin{proof}\leanok
    If a site's row is completed, its fixed index is less than $n$, so it occurs
    in a consumed slot. A site initially outside the middle never enters it,
    and consuming its slot removes it from the middle regardless of earlier
    charges. Near slot $t$ is consumed at fill $2t$, and far slot $t$ at fill
    $2t+1$.
\end{proof}

        TNLean.PEPS.AreaLaw.Scan.CollarScan.oldChargeState_front_le}
    \leanok
    \uses{def:al_scan_actual_histories,thm:al_scan_consumed_slots}
    For $m,M>0$, the history weights are nonnegative and sum to one.
    Extending a history multiplies its weight by $(2M)^{-K}$.
    Each band's offset has marginal probability $1/m$, even after subsequent
    charges; each side--slot pair at the next charge has conditional probability
    $1/(2M)$. Charges leave the nominal fronts unchanged.

    oriented nominal fronts, provided its row has at most $n$ sites.
\end{theorem}
\begin{proof}\leanok
    There are $m^K(2M)^{Kk}$ histories and $(2M)^K$ possible new charges.
    Fixing one coordinate divides the corresponding count by $m$ or $2M$.
    Induction on the choices shows that every surviving middle site was
    initially unassigned and avoids every consumed fill slot. Apply
    Theorem~\ref{thm:al_scan_consumed_slots}.
\end{proof}

    \leanok
    For positive $d$, the ambient depth row is $T_d\setminus T_{d-1}$, and its
    physical part has no larger cardinality. Graph distance bounds depth
    variation: $|d(x)-d(y)|\le d_\Lambda(x,y)$. Let $Z$ be the set of endpoints

    graph $r_0$-ball meeting $A\cap T_L$ is contained in $A$.
\end{theorem}
\begin{proof}\leanok
    Minimize the integer sup-norm distance over $T$. Each lattice edge changes
    that minimum by at most one; summing along a shortest graph walk gives the
    lipchitz bound. If a ball meeting $A\cap T_L$ leaves $A$, a path of length
    at most $2r_0$ within the ball crosses an edge with an endpoint of depth at
    most $L+2r_0$, contradicting the clearance.
\end{proof}

    \leanok
    \uses{def:al_scan_actual_histories,thm:al_scan_ambient_rows}
    Suppose $n,m\ge2$, $D\ge1$, $r_0\le D$, $4D\le m$, $8Km\le L$, and at
    most $nm$ fills are consumed. Assume $|T_d\setminus T_{d-1}|\le n$ for
    $1\le d\le L$, and at most $\mu$ labels have any fixed anchor.
    Each charge interval $[j-r_0,j+D]$ visits only unsigned depths in $[1,L]$.
    Its number of labels is at most $(D+r_0+1)n\mu\le3\mu nD$.
    Thus $C_1\ge3\mu$ permits padding every list to $M=\lceil C_1nD\rceil$,
    with a unique slot for every label, including labels with equal anchors.
    If $C_1>0$, then $M\le(C_1+1)nD$ and
    $1/(2M)\ge[2(C_1+1)]^{-1}/(nD)$.
\end{theorem}
\begin{proof}\leanok
    Sum the row cardinalities, then sum the anchor fibers. Either nominal front
    advances at most $m$ rows; the band cutoffs and $4D\le m$ keep the entire
    candidate interval in the collar. Ordering distinct labels gives distinct
    slots. Finally $\lceil C_1nD\rceil\le C_1nD+1\le(C_1+1)nD$.
\end{proof}

\begin{theorem}[Ball sampling on a good history]
    \label{thm:al_scan_good_ball_sampling}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.IsGoodOldHistory,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.good_split_anchor_bounds,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.existsUnique_selected_of_bounds,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.selected_split_sampling,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.oldChargeState_depth_bounds,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.good_split_sampling_domainGraph}
    \leanok
    \uses{thm:al_scan_history_weights,thm:al_scan_ambient_rows,thm:al_scan_charge_capacity}
    Retain the capacity and clearance assumptions, and suppose $2r_0\le D$.
    A pre-charge history is good when every assigned site of $A$ has oriented
    depth at most $j+D/2$ on its receiving side. For every labelled graph
    $r_0$-ball meeting that side and the middle, the anchor lies in
    $[j-r_0,j+D]$ and its labelled slot is unique. Choosing this side and slot
    has probability $1/(2M)\ge c/(nD)$ for $c=1/[2(C_1+1)]>0$, and the extended
    physical history removes the ball's intersection with $Y$ from $Y$.
\end{theorem}
\begin{proof}\leanok
    An unassigned site has oriented depth at least $j$, so the anchor has depth
    at least $j-r_0$. The ball lies in $A$ by clearance; an assigned site of
    the ball therefore has depth at most $j+D/2$, giving anchor depth at most
    $j+D/2+r_0\le j+D$. The unique-slot and marginal probability formulas give
    the probability, and the charge transition transfers the selected ball
    intersected with the middle.
\end{proof}

\begin{theorem}[Expected cost of the charge moves]
    \label{thm:al_scan_expected_move_cost}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.splitIncidences,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.mem_splitIncidences,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.chargeMoveCost,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.splitIncidenceCost,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.splitIncidenceCost_band_le_expected_chargeMoveCost,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.splitIncidenceCost_le_expected_chargeMoveCost,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.good_splitIncidenceCost_le_expected_chargeMoveCost_domainGraph}
    \leanok
    \uses{thm:al_scan_good_ball_sampling}
    Under the good-history sampling assumptions, let $F_{g,s}(B)\ge0$ be a
    nonnegative cost of a finite set of physical sites moved toward side $s$
    in band $g$. Sum $F_{g,s}(B_i\cap Y_g)$ over every splitting incidence
    $(g,s,i)$, where $s$ is the receiving side and $B_i$ is the labelled charge
    ball. If $\Delta_g(c)$ is the set removed from $Y_g$ by the simultaneous
    charge choice $c$ and $s_g(c)$ is its selected side, then
    \begin{align}
        \frac{1}{2M}\sum_{(g,s,i)\ \mathrm{split}} F_{g,s}(B_i\cap Y_g)
         & \le \sum_c q_c\sum_g F_{g,s_g(c)}(\Delta_g(c)).
        \label{eq:al_scan_expected_move_cost}
    \end{align}
    For $C_1>0$, the left coefficient may be replaced by
    $1/[2(C_1+1)nD]$.
\end{theorem}
\begin{proof}\leanok
    For each band and realized side--slot pair, at most one labelled incidence
    is selected, with cost equal to that of the removed middle set, while all
    other contributions are nonnegative. Summing this pointwise inequality
    against the choice weights $q_c$, using the $1/(2M)$ marginal for each
    available labelled slot, and interchanging the finite sums over choices and
    bands gives the stated bound.
\end{proof}

\begin{theorem}[Finite uniform-offset counting]
    \label{thm:al_scan_offset_count}
    \lean{TNLean.PEPS.AreaLaw.Scan.sum_historyWeight_offset_mem,
        TNLean.PEPS.AreaLaw.Scan.sum_historyWeight_mul_eq_sum_extendHistory,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.front_injective_offset,

        TNLean.PEPS.AreaLaw.Scan.CollarScan.historyWeight_mixture_event_le,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.split_interval_band_unique}
    \leanok
    \uses{def:al_scan_actual_histories,thm:al_scan_history_weights}
    At fixed band, side, and front time, at most $a+b+1$ offsets place a fixed
    depth in $[j-a,j+b]$. Every history event contained in that interval event
    therefore has probability at most $(a+b+1)/m$. If $a+b\le m$, the spacing
    $8m$ between band centers allows at most one band on each side to reach the
    fixed depth across all offsets, so the union over both sides and all bands
    has probability at most $2(a+b+1)/m$, and convex mixtures preserve this
    bound.
\end{theorem}
\begin{proof}\leanok
    The front is affine with coefficient $1$ or $-1$ in its offset, so it
    injects into the integer interval $[d-b,d+a]$, and the marginal probability
    of an offset set is its cardinality divided by $m$. Reindexing an extended
    history into its prefix and last charge factors the leaf weights as
    $w_hq_c$. For a fixed side, at most one band center can reach $d$ across all
    offsets; summing the two side bounds gives $2(a+b+1)/m$.
\end{proof}

\begin{theorem}[Physical cut boundaries and changed sites]
    \label{thm:al_physical_cut_calculus}
    \lean{TNLean.PEPS.AreaLaw.mem_edgeBoundary_pair_iff,
        TNLean.PEPS.AreaLaw.edgeBoundary_compl,
        TNLean.PEPS.AreaLaw.edgeBoundary_union_subset,
        TNLean.PEPS.AreaLaw.edgeBoundary_sdiff_subset,
        TNLean.PEPS.AreaLaw.card_edgeBoundary_union_le,
        TNLean.PEPS.AreaLaw.card_edgeBoundary_sdiff_le,
        TNLean.PEPS.AreaLaw.card_edgeBoundary_le_of_endpoint_cover,
        TNLean.PEPS.AreaLaw.card_edgeBoundary_le_four_mul_card,

    Let $\Lambda\subset\mathbb Z^2$ be finite and let $B,C\subseteq\Lambda$.
    An unordered pair belongs to $\partial_\Lambda B$ when its sites are
    adjacent in the induced graph and one belongs to $B$. Complements are taken
    relative to $\Lambda$, and
    \begin{align}
        \partial_\Lambda(B^c)          & = \partial_\Lambda B, \\
        \partial_\Lambda(B\cup C)      & \subseteq
        \partial_\Lambda B\cup\partial_\Lambda C,              \\
        \partial_\Lambda(B\setminus C) & \subseteq
        \partial_\Lambda B\cup\partial_\Lambda C,              \\
        \partial_\Lambda B             & \subseteq
        \partial_\Lambda C\cup\partial_\Lambda(B\mathbin{\triangle} C).
    \end{align}
    Consequently the boundary cardinalities of $B\cup C$ and
    $B\setminus C$ are at most
    $|\partial_\Lambda B|+|\partial_\Lambda C|$.
    If $R\subseteq\Lambda$ contains at least one site of each edge of
    $\partial_\Lambda B$, then $|\partial_\Lambda B|\le4|R|$.
    In particular,
    \begin{align}
        |\partial_\Lambda B| & \le4|B|, \\
        |\partial_\Lambda B| & \le
        |\partial_\Lambda C|+4|B\mathbin{\triangle} C|.
    \end{align}
    The last estimate is the changed-site bound in the proof of
    \cite[Lemma~9.1(2)]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:area_law_ambient_boundary_crossing}
    Checking membership of the two endpoints of each edge gives the complement
    equality and the three inclusions. Covering the crossing edges by the at
    most four incident lattice edges at each site of $R$ gives
    $|\partial_\Lambda B|\le4|R|$. Taking $R=B$ yields the first inequality, and
    applying it to $B\mathbin{\triangle} C$ in the last inclusion yields the
    second.
\end{proof}

\begin{theorem}[Physical boundaries of depth prefixes]
    \label{thm:al_scan_prefix_boundary}

    and let $Z$ consist of both sites of every edge of $\partial_\Lambda A$.
    Fix integers $L\ge1$ and $n,r_0\ge0$, and suppose
    $\|t-z\|_\infty>2L+10r_0$ for every $t\in T$ and $z\in Z$.
    Every ambient dilation $T_j$ with $0\le j\le L$ is disjoint from $Z$.
    Suppose further that $T$ is nonempty and that
    $|T_d\setminus T_{d-1}|\le n$ for each integer $1\le d\le L$.
    Set $\delta(x)=\min_{t\in T}\|x-t\|_\infty$.
    For every integer $j\le L$, define
    $P_j=\{x\in A:\delta(x)\le j\}$ and
    $Q_j=\{x\in A:0<\delta(x)\le j\}$. Then
    \begin{align}
        Q_j                    & = P_j\setminus P_0, \\
        |\partial_\Lambda P_j| & \le4n,              \\
        |\partial_\Lambda Q_j| & \le8n,
    \end{align}
    which are the full-row edge estimates used in
    \cite[Lemma~9.1(2)]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_physical_ambient_boundary,
        lem:area_law_ambient_boundary_layers,thm:al_physical_cut_calculus}

    $j\le L$ from $T$, contradicting the clearance.
    For $j\ge0$, comparison with the ambient boundary gives
    $|\partial_\Lambda P_j|\le|\partial T_j|$.
    When $1\le j\le L$, each edge leaving $T_j$ has its inside site in
    $T_j\setminus T_{j-1}$; at $j=0$, its outside site lies in
    $T_1\setminus T$. In either case the relevant row has at most $n$

\end{proof}

\begin{theorem}[Entropy of a valid middle-to-side move]

\begin{theorem}[Finite-leaf charge energy inequality]
    \label{thm:al_scan_finite_charge_energy}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScanRound.goodOld_le_capacity,
        TNLean.PEPS.AreaLaw.Scan.ScanRound.energySum_le_of_leaf_bounds}
    \leanok

    Suppose at most $N\ge0$ labels split at each leaf and each label's
    terminal split weight is at most $d\ge0$. For $0\le p\le1$, set
    \begin{align}
        \mathcal Q & =\sum_{h\text{ good}}w_h
        \sum_{i\text{ split in }h}\int_\Theta
        \eta_{i,h}(\theta)\,d\mu_h(\theta),        \\

    \end{align}

        \label{eq:al_scan_finite_charge_energy}
    \end{align}
\end{theorem}
\begin{proof}\leanok

    $\int\eta^{1/8}\le(\int\eta)^{1/8}$. Applying finite weighted
    H\"older to the good-history and split-label pairs, whose total
    weight is at most $N$, bounds their contribution by
    $N^{7/8}\mathcal Q^{1/8}$. On each bad-old or new leaf,
    $\int\eta^{1/8}\le B^{1/8}/2$; counting labels and summing
    probabilities gives $NB^{1/8}b/2$ and $NB^{1/8}/2$, respectively.
    Multiplying by $1-p$ and $p$ and bounding each label's terminal
    split weight by $d$ yields the stated inequality.
\end{proof}

    Theorem~\ref{thm:al_scan_actual_charge_sum_bound}, with universal constants
    $c_0,C_{\mathrm{ent}},e_{\mathrm{ent}}$ ($c_0>0$), constant $C_\beta\ge0$,
    and sequence $0\le\beta_k\le C_\beta\log(k+2)$, for every $k\ge1$,
    every nonzero symmetric vector $\mathrm{pre}$, and every $0\le p_0<p_1\le1$,
    set
    \begin{align}
        \rho & = \frac{ka}{2(C_1+1)nD},                        \\
        B    & = \log (N^{\mathrm{fill}}_{0,k,\mathrm{pre}})^2
        -\log (N^{\mathrm{fill}}_{nm,k,\mathrm{pre}})^2
        +2nm\bigl(C_{\mathrm{ent}}kaKa^{1/4}\ell^{e_{\mathrm{ent}}}+\beta_k\bigr),
    \end{align}
    where $(N^{\mathrm{fill}}_{j,k,\mathrm{pre}})^2
        =\|\mathcal M^{\mathrm{fill}}_{j,k}(0)^{-1/4}\mathrm{pre}\|^2$ for the
    fill root $\mathcal M^{\mathrm{fill}}_{j,k}(0)$ at stage $j$ and parameter
    zero with metric exponent $a/2$. Then there exist a charge stage
    $0\le s<nm$ and a parameter $p\in[p_0,p_1]$ such that
    $\mathcal Q_{s,k,\mathrm{pre}}(p)\le B/(\rho nm(p_1-p_0))$,
    giving the finite-averaging selection of \cite[Section~9]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_scan_actual_charge_sum_bound,thm:al_scan_actual_charge_selection}
    By~(\ref{eq:al_scan_actual_charge_sum}),
    $\rho\sum_s\int_{p_0}^{p_1}\mathcal Q_{s,k,\mathrm{pre}}(p)\,dp\le B$.
    Since $k\ge1$, $a>0$, $C_1>0$, $n\ge2$, $D\ge1$, and $4D\le m$, both
    $\rho$ and $nm$ are positive, so
    Theorem~\ref{thm:al_scan_actual_charge_selection} yields the desired
    stage $s$ and parameter $p\in[p_0,p_1]$.
\end{proof}
