\section{Trace contraction of a site-dependent operator chain}

\begin{definition}[Trace contraction with a closing matrix]
    \label{def:trace_chain_operator}
    \lean{MPOTensor.traceChainOperator}
    Let $n,q,\chi$ be nonnegative integers. Supply matrices
    $A_v(i,j)\in\C^{\chi\times\chi}$ for $0\le v<n$ and $0\le i,j<q$,
    together with $C\in\C^{\chi\times\chi}$. Their trace contraction is
    the operator with entries
    \begin{align}
        \mathcal T(A,C)(\sigma,\tau)
         & =\operatorname{tr}\left(C\prod_{v=0}^{n-1}A_v(\sigma_v,\tau_v)\right).
        \label{eq:trace_chain_operator}
    \end{align}
    The product follows increasing site order. Empty chains and zero
    dimensions are allowed; at length zero the unique entry is
    $\operatorname{tr}(C)$.
\end{definition}

\begin{theorem}[Absorption of a closing matrix on a nonempty chain]
    \label{thm:trace_chain_closing_absorption}
    \lean{MPOTensor.exists_traceChainOperator_one}
    Let $n,q,\chi$ be nonnegative integers with $n\ge1$.
    Supply matrices $A_v(x,y)\in\C^{\chi\times\chi}$ for
    $0\le v<n$ and $0\le x,y<q$, and a matrix
    $C\in\C^{\chi\times\chi}$.
    There are matrices $A'_v(x,y)$ of the same size such that, for all
    $\sigma,\tau\in\{0,\ldots,q-1\}^{\{0,\ldots,n-1\}}$,
    \begin{align}
        \operatorname{tr}\left(C\prod_{v=0}^{n-1}A_v(\sigma_v,\tau_v)\right)
         & =\operatorname{tr}\left(\prod_{v=0}^{n-1}A'_v(\sigma_v,\tau_v)\right).
        \label{eq:trace_chain_closing_absorption}
    \end{align}
    The dimensions $q$ and $\chi$ may vanish.
\end{theorem}
\begin{proof}
    Set $A'_0(x,y)=CA_0(x,y)$ and $A'_v(x,y)=A_v(x,y)$ for $1\le v<n$.
    Associativity gives (\ref{eq:trace_chain_closing_absorption}).
\end{proof}

\begin{theorem}[A trace-chain representation of any finite matrix]
    \label{thm:finite_matrix_trace_chain}
    \lean{MPOTensor.exists_traceChainOperator_of_finite_matrix}
    \uses{def:trace_chain_operator}
    For every matrix $M$ on $(\C^q)^{\otimes n}$ there are a natural number
    $B$, site-dependent tensors $A_v(i,j)\in\C^{B\times B}$ and a closing
    matrix $C\in\C^{B\times B}$ with $M=\mathcal T(A,C)$.
    The empty chain and zero local dimension are included. No quantitative
    bound on $B$ or procedure for finding the tensors is asserted.
\end{theorem}
\begin{proof}
    Index the bond by pairs of full configurations. Diagonal entries of
    each site tensor select their two physical coordinates; put the
    coefficient of $M$ in a diagonal closing matrix. The trace selects the
    unique matching pair. The same construction works for $n=0$, when
    there is one empty configuration pair and bond dimension one.
\end{proof}
