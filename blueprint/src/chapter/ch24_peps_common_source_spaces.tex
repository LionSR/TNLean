\section{Common private source spaces}
\label{sec:peps_common_source_spaces}

\begin{definition}[Sources in prescribed pair slots]
    \label{def:peps_source_slots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.ofSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.ofSlots_nil}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.ofSlots_cons}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.isNormalized_ofSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.layout_ofSlots_eq}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.partyPairs_ofSlots}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.slotLayout}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.prepareSlots}
    \leanok
    \uses{def:peps_source_inventory}
    Following \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 253--267), an ordered list of distinct
    party pairs $(p_e,q_e)$ ($p_e\ne q_e$) with complex inner product halfspaces
    $U_e,V_e$ and vectors $\eta_e\in U_e\otimes V_e$ defines an ordered source
    list (normalized when $\|\eta_e\|=1$ for all $e$). For any original memory
    $\mathcal H$, its slot preparation map $Q_{\eta,\mathcal H}$ satisfies
    $Q_{\eta,\mathcal H}x=(\bigotimes_e\eta_e)\otimes x$.
\end{definition}

\begin{lemma}[A common order and endpoint orientation]
    \label{thm:peps_source_ordering}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.of_perm}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_ofSlots_expands}
    \leanok
    \uses{def:peps_source_slots,def:peps_source_expansion}
    For any reference source list $R$ with distinct unordered pairs and any
    normalized source list $S$ with the same unordered pair set, there exist
    normalized sources in the pair order and endpoint orientations of $R$ whose
    preparation transforms into $P_{S,\mathcal H}$ by allowed source-free
    operations for every spectator memory $\mathcal H$
    (Definition~\ref{def:peps_source_expansion}); permuting whole source blocks
    also defines such a transformation.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_source_expansion,thm:peps_source_expansion_operations}
    By induction on $R$ using Lemma~\ref{thm:peps_source_expansion_operations},
    move the unique source in $S$ matching the first unordered pair of $R$ to
    the front, apply the swap $U\otimes V\to V\otimes U$ if its endpoint
    orientation is reversed, and apply the induction hypothesis to the remaining
    pairs.
\end{proof}

\begin{definition}[Orthogonal sums of branch spaces]
    \label{def:peps_common_halfspaces}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.commonSpace}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.commonInclusion}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.commonProjection}
    \leanok
    For a finite index set $\Xi$ and complex inner product spaces $U_\xi$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 253--267), define the orthogonal sum
    $\widehat U=\bigoplus_{\xi\in\Xi}U_\xi$ with norm
    $\|(u_\xi)_{\xi\in\Xi}\|^2=\sum_{\xi\in\Xi}\|u_\xi\|^2$, coordinate
    inclusions $j_\xi:U_\xi\to\widehat U$, and coordinate projections
    $\pi_\xi:\widehat U\to U_\xi$.
\end{definition}

\begin{lemma}[Embedding and recovering a pair source]
    \label{thm:peps_common_pair_source}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.commonProjection_commonInclusion}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.norm_commonProjection_le}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.commonVector}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.norm_commonVector}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.mapL_commonProjection_commonVector}
    \leanok
    \uses{def:peps_common_halfspaces}
    Each $j_\xi$ is an isometry with $\|\pi_\xi\|\le1$ and
    $\pi_\xi j_\xi=\Id_{U_\xi}$. For a second family $V_\xi$ with inclusions
    $k_\xi:V_\xi\to\widehat V$ and projections $\rho_\xi:\widehat V\to V_\xi$,
    setting $\widehat\eta_\xi=(j_\xi\otimes k_\xi)\eta_\xi$ for
    $\eta_\xi\in U_\xi\otimes V_\xi$ gives
    \begin{align}
        \|\widehat\eta_\xi\|                     & =\|\eta_\xi\|, &
        (\pi_\xi\otimes\rho_\xi)\widehat\eta_\xi & =\eta_\xi.
        \label{eq:peps_common_pair_recovery}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    From $\|j_\xi u\|=\|u\|$, $\|\pi_\xi u\|^2\le\sum_{\zeta}\|u_\zeta\|^2=\|u\|^2$,
    $\pi_\xi j_\xi=\Id_{U_\xi}$, and $\rho_\xi k_\xi=\Id_{V_\xi}$, the tensor
    map $j_\xi\otimes k_\xi$ is isometric and
    $(\pi_\xi\otimes\rho_\xi)(j_\xi\otimes k_\xi)=\Id_{U_\xi\otimes V_\xi}$.
\end{proof}

\begin{lemma}[Local maps on a prepared pair]
    \label{thm:peps_map_source_pair}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapPair}
    \lean{TNLean.Lattice.PartyCircuit.Word.mapPair_spec}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_mapPair_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_mapPair_source}
    \leanok
    \uses{def:peps_party_layout,def:peps_source_inventory}
    For continuous linear maps $A:U\to U'$ and $B:V\to V'$ at the endpoints of
    a pair source $\eta\in U\otimes V$ and any spectator memory $\mathcal H$,
    applying $A$ and $B$ locally sends $u\otimes(v\otimes x)\mapsto Au\otimes(Bv\otimes x)$
    and satisfies
    \begin{align}
        (A\otimes B\otimes\Id)P_{\eta,\mathcal H}
         & =P_{(A\otimes B)\eta,\mathcal H},
        \label{eq:peps_map_source_pair}
    \end{align}
    forming an allowed source-free composition when $\|A\|,\|B\|\le 1$.
\end{lemma}

\begin{proof}\leanok
    Verify (\ref{eq:peps_map_source_pair}) on elementary tensors
    $\eta=u\otimes v$ and extend by linearity; $\|A\|,\|B\|\le 1$ gives
    allowedness.
\end{proof}

\begin{lemma}[Local contractions with a prepared tail]
    \label{thm:peps_source_cons_mapL}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.Expands.cons_mapL}
    \leanok
    \uses{def:peps_source_inventory,def:peps_source_expansion}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 253--267), for parties $p\ne q$,
    contractions $A:U\to U'$ and $B:V\to V'$, $\eta\in U\otimes V$, a source
    list $T$, and any spectator memory $\mathcal H$, adjoining $s=(p,q,U,V,\eta)$
    and $s'=(p,q,U',V',(A\otimes B)\eta)$ before $T$ to form $S$ and $S'$ gives
    an allowed source-free composition $D$ satisfying
    $D P_{S,\mathcal H}=P_{S',\mathcal H}$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_map_source_pair,def:peps_source_inventory}
    Apply Lemma~\ref{thm:peps_map_source_pair} with spectator memory
    $\mathcal H_T\otimes\mathcal H$ to obtain
    $D P_{s,\mathcal H_T\otimes\mathcal H}=P_{s',\mathcal H_T\otimes\mathcal H}$,
    and compose on the right with $P_{T,\mathcal H}$.
\end{proof}

\begin{lemma}[Common spaces for normalized source lists]
    \label{thm:peps_common_source_lists}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.common_ofSlots_expands}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_common_expansions}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_prepareSlots_recovery}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_prepareSlots_expands}
    \leanok
    \uses{def:peps_source_slots,def:peps_common_halfspaces,def:peps_source_expansion}
    For a finite family $(S_\xi)_{\xi\in\Xi}$ of normalized source lists sharing
    the distinct unordered pair set of a reference list, there exist common
    halfspaces in the reference pair order, normalized vectors
    $\widehat\eta_{\xi,e}$, and allowed source-free compositions $D_\xi$ such
    that for every spectator memory $\mathcal H$,
    \begin{align}
        D_\xi Q_{\widehat\eta_\xi,\mathcal H}
         & =P_{S_\xi,\mathcal H}.
        \label{eq:peps_common_list_recovery}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_source_ordering,thm:peps_common_pair_source,thm:peps_source_cons_mapL,
        thm:peps_source_expansion_operations}
    Reorder each $S_\xi$ by Lemma~\ref{thm:peps_source_ordering}, embed each
    slot into the orthogonal sums $\widehat U_e,\widehat V_e$ via
    Lemma~\ref{thm:peps_common_pair_source}, and recover each slot by
    (\ref{eq:peps_common_pair_recovery}), Lemma~\ref{thm:peps_source_cons_mapL},
    and Lemma~\ref{thm:peps_source_expansion_operations}.
\end{proof}

\begin{lemma}[Local factorization in common private spaces]
    \label{thm:peps_common_party_maps}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_common_source_preparation}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_common_prepared_tensorPartyMaps}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_registers}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 253--267), for a finite ordered party
    set $P$ and a finite family $(F_\xi)_{\xi\in\Xi}$ of allowed compositions
    sharing input and output register layouts, there exist fixed halfspaces for
    every unordered pair in $P$, normalized vectors $\widehat\eta_{\xi,e}$, and
    local contractions $B_{\xi,p}$ with $\xi$-independent domains and codomains
    such that under the party-grouping identifications $U_{\rm in},U_{\rm out}$,
    \begin{align}
        U_{\rm out}F_\xi
         & =\left(\bigotimes_{p\in P}B_{\xi,p}\right)
        U_{\rm in}Q_{\widehat\eta_\xi,\mathcal H_{\rm in}}.
        \label{eq:peps_common_party_maps}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_complete_pair_sources,thm:peps_complete_prepared_party_maps,
        thm:peps_common_source_lists,thm:peps_party_factorization}
    Complete the sources of each $F_\xi$ to all unordered pairs in $P$ by
    Lemmas~\ref{thm:peps_complete_pair_sources}
    and~\ref{thm:peps_complete_prepared_party_maps}, embed into common
    halfspaces by Lemma~\ref{thm:peps_common_source_lists}, and factor the
    resulting source-free composition by party via
    Lemma~\ref{thm:peps_party_factorization}.
\end{proof}

\begin{lemma}[The weighted gate in common private spaces]
    \label{thm:peps_common_source_gate}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_common_source_gate}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_registers}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 233--267), for $F=\sum_{\xi\in\Xi}c_\xi F_\xi$
    with $(F_\xi)_{\xi\in\Xi}$ as in Lemma~\ref{thm:peps_common_party_maps},
    \begin{align}
        U_{\rm out}F
         & =\sum_{\xi\in\Xi}c_\xi
        \left(\bigotimes_{p\in P}B_{\xi,p}\right)
        U_{\rm in}Q_{\widehat\eta_\xi,\mathcal H_{\rm in}}.
        \label{eq:peps_common_source_gate}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_common_party_maps}
    Multiply (\ref{eq:peps_common_party_maps}) by $c_\xi$ and sum over
    $\xi\in\Xi$.
\end{proof}


\begin{lemma}[Finite coordinates for a finite family of source vectors]
    \label{thm:peps_finite_pair_coordinates}
    \lean{TNLean.Lattice.PartyCircuit.PairSource.exists_finite_coordinates}
    \leanok
    For complex inner product spaces $U,V$, finite $\Xi$, and algebraic tensors
    $\eta_\xi\in U\otimes V$ in \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 279--299), there exist $a,b\in\N$,
    linear isometries $f:\C^a\to U$ and $g:\C^b\to V$, and vectors
    $\eta^0_\xi\in\C^a\otimes\C^b$ such that $(f\otimes g)\eta^0_\xi=\eta_\xi$
    and $\|\eta^0_\xi\|=\|\eta_\xi\|$ for all $\xi\in\Xi$.
\end{lemma}

\begin{proof}\leanok
    Span the finitely many elementary tensor factors of $\{\eta_\xi\}_{\xi\in\Xi}$
    by finite-dimensional subspaces $U_0\subseteq U$ and $V_0\subseteq V$, and
    choose orthonormal coordinate isometries $f:\C^a\simeq U_0\hookrightarrow U$
    and $g:\C^b\simeq V_0\hookrightarrow V$.
\end{proof}

\begin{lemma}[Source preparations in finite coordinates]
    \label{thm:peps_finite_source_coordinates}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.ofSlots_mapIsometry_expands}
    \lean{TNLean.Lattice.PartyCircuit.SourceInventory.exists_finite_coordinate_expansions}
    \leanok
    \uses{def:peps_source_slots,def:peps_source_expansion}
    For a finite source list with halfspaces $U_e,V_e$ and a finite family
    $\eta_{\xi,e}\in U_e\otimes V_e$, there exist $a_e,b_e\in\N$, isometries
    $f_e:\C^{a_e}\to U_e$ and $g_e:\C^{b_e}\to V_e$, vectors
    $\eta^0_{\xi,e}\in\C^{a_e}\otimes\C^{b_e}$ with
    $(f_e\otimes g_e)\eta^0_{\xi,e}=\eta_{\xi,e}$ and
    $\|\eta^0_{\xi,e}\|=\|\eta_{\xi,e}\|$, and allowed source-free compositions
    $D_\xi$ satisfying $D_\xi Q_{\eta^0_\xi,\mathcal H}=Q_{\eta_\xi,\mathcal H}$
    for every spectator memory $\mathcal H$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_finite_pair_coordinates,thm:peps_source_cons_mapL,
        thm:peps_source_expansion_operations}
    Apply Lemma~\ref{thm:peps_finite_pair_coordinates} at each slot $e$, and
    compose the local isometries $f_e,g_e$ across the slots by
    Lemmas~\ref{thm:peps_source_cons_mapL}
    and~\ref{thm:peps_source_expansion_operations}.
\end{proof}

\begin{lemma}[The source-gate expansion in finite coordinates]
    \label{thm:peps_finite_source_gate}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_finite_source_preparation}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_finite_prepared_tensorPartyMaps}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_finite_source_gate}
    \leanok
    \uses{def:peps_source_slots,def:peps_party_registers}
    In \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}
    (\texttt{04-compression.tex}, lines 233--299), for $F=\sum_{\xi\in\Xi}c_\xi F_\xi$
    with $(F_\xi)_{\xi\in\Xi}$ as in Lemma~\ref{thm:peps_common_party_maps},
    there exist common dimensions $a_e,b_e\in\N$ for every unordered pair in
    $P$, normalized vectors $\eta^0_{\xi,e}\in\C^{a_e}\otimes\C^{b_e}$, and
    local contractions $B_{\xi,p}$ such that
    \begin{align}
        U_{\rm out}F_\xi
         & =\left(\bigotimes_{p\in P}B_{\xi,p}\right)
        U_{\rm in}Q_{\eta^0_\xi,\mathcal H_{\rm in}}, \notag \\
        U_{\rm out}F
         & =\sum_{\xi\in\Xi}c_\xi
        \left(\bigotimes_{p\in P}B_{\xi,p}\right)
        U_{\rm in}Q_{\eta^0_\xi,\mathcal H_{\rm in}}.
        \label{eq:peps_finite_source_gate}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_common_party_maps,thm:peps_common_source_lists,
        thm:peps_finite_source_coordinates,thm:peps_party_factorization}
    Combine Lemmas~\ref{thm:peps_common_party_maps}
    and~\ref{thm:peps_common_source_lists} with
    Lemma~\ref{thm:peps_finite_source_coordinates}, factor the resulting
    source-free operations by party via Lemma~\ref{thm:peps_party_factorization},
    and sum over $\xi\in\Xi$ with weights $c_\xi$.
\end{proof}
