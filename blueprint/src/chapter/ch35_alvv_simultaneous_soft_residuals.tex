\section{Simultaneous soft replacements with estimated energies}
\label{sec:alvv_simultaneous_soft_residuals}

The following estimates use one common interior after several regional
replacements. The number of crossing bonds is retained throughout the
argument. In particular, replacing three separated regions requires no
nearest-neighbour representation of any already softened term.

\begin{lemma}[Residuals for a common regional interior]
    \label{lem:alvv_soft_family_residuals}
    \lean{TNLean.MPS.AreaLaw.ALVV.norm_simultaneous_soft_original_residual,
        TNLean.MPS.AreaLaw.ALVV.norm_simultaneous_soft_reverse_residual}
    \leanok
    \uses{lem:regional_barrier_soft_residual}
    Let consecutive site intervals have internal bond sums $A_j$, and
    let $Y=\sum_{i\in J}h_i$ be a sum of actual boundary bonds, where
    $C=|J|$. Suppose $0\leq h_i\leq I$ with their designated
    nearest-neighbour supports. Let $e_jI\leq A_j$, and choose
    $e_j-10\leq p_j\leq e_j$ on a finite selected set $S$.
    For $t>0$ and $m\geq2$, put
    \begin{align}
        H_0&=\sum_j(A_j-e_jI)+Y,\\
        G&=\sum_j\bigl(\mathbf1_{j\in S}
          (p_jI+h_{m,t}(A_j-p_jI))+
          \mathbf1_{j\notin S}A_j-e_jI\bigr)+Y.
    \end{align}
    Fix orthonormal eigenbases with real eigenvalues for the actual
    matrices whose spectral spaces occur below. If $E\leq C$,
    $\eta\geq0$ and $T\geq\eta+3C+12$, then
    \begin{align}
        \|(H_0-G)|_{E_{H_0}(E+\eta)}\|
        \leq |S|\left(8Ce^{-(T-\eta-3C-12)/8}
          +\frac{t}{m}\left(\frac Tt\right)^m\right).
    \end{align}
    If $\mu\leq C$, $\eta\geq0$, $D=10|S|$ and
    $t,T\geq U=2(\eta+D+3C+2)$, then
    \begin{align}
        \|(H_0-G)|_{E_G(\mu+\eta)}\|
        \leq |S|\left(8Ce^{-(T-U)/8}
          +\frac{t}{m}\left(\frac Tt\right)^m\right).
    \end{align}
    The first inequality also holds at order $m=1$.
\end{lemma}
\begin{proof}\leanok
    The raw centered interior is positive and commutes with each
    regional sum. The estimated-center soft inequalities give the
    same commutation for the soft interior and restore positivity
    after the correction $DI$. Apply
    Lemma~\ref{lem:regional_barrier_soft_residual} separately to each
    selected region, using the actual boundary $J$ throughout.
    In the first case use the projection floor
    $u=\eta+3C+2$ above $e_j+u$; in the second case use the soft floor
    $U-(t/m)(U/t)^m$ above $p_j+U$, after shifting the spectral
    coordinates and cutoff by $D$. The respective denominators are
    at least two. The equality $H_0-G=\sum_{j\in S}(A_j-F_j)$
    and the triangle inequality give the factor $|S|$.
\end{proof}

\begin{lemma}[The actual common soft interior]
    \label{lem:alvv_five_soft_structure}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_five_region_soft_structure}
    \leanok
    Let $q\geq1$, let four ordered cuts divide an $L$-site chain into
    five consecutive regions, and let $0\leq h_i\leq I$ have their
    designated nearest-neighbour supports. Write $A_j$ for the five
    internal regional sums, $J$ for the actual crossing set, $Y$ for
    its sum and $C=|J|\leq4$.
    There are attained regional minima $e_j$, with a common unit
    ground vector, and an attained ground energy $0\leq E\leq C$ of
    $H=\sum_i h_i-(\sum_j e_j)I$.
    These can be chosen before every set $S$ of replaced regions,
    estimates $e_j-10\leq p_j\leq e_j$ for $j\in S$, scale $t>0$
    and order $m\geq1$. Define
    \begin{align}
        F_j&=p_jI+h_{m,t}(A_j-p_jI),\qquad
        B=\sum_j\bigl(\mathbf1_{j\in S}F_j+
          \mathbf1_{j\notin S}A_j-e_jI\bigr),\\
        D&=10|S|,\qquad G=B+Y.
    \end{align}
    Then $B+DI\geq0$, $G\leq H$, and $B$ commutes with every $A_j$.
    The actual ground energy $\mu$ of $G$ is attained and satisfies
    $-D\leq\mu\leq E\leq C$. For $j\in S$ and $U\geq0$,
    \begin{align}
        \left(U-\frac{t}{m}\left(\frac Ut\right)^m\right)
        \mathbf1_{[p_j+U,\infty)}(A_j)\leq B+DI.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Disjoint regional supports give a common unit vector attaining
    every regional minimum. The raw centered interior is positive
    and annihilates this vector. Testing the original sum with this
    vector bounds its ground energy by $C$.
    Apply the estimated-center soft order inequalities to each
    selected block and sum their ten-unit positive corrections.
    Functions of the commuting regional matrices still commute with
    all those matrices. The single-block projection floor lies below
    the common corrected interior. Finally, construct a unit ground
    vector of $G$. Positivity of $G+DI$ and testing $G\leq H$ on an
    original ground vector give $-D\leq\mu\leq E$.
\end{proof}

\begin{lemma}[Two residuals for the same simultaneous soft operator]
    \label{lem:alvv_five_soft_residuals}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_five_region_soft_residuals}
    \leanok
    \uses{lem:alvv_five_soft_structure,lem:alvv_soft_family_residuals}
    The choices in Lemma~\ref{lem:alvv_five_soft_structure} can be made
    so that the following estimates hold for the same $G$ and $\mu$.
    Suppose $m\geq2$ and $\eta\geq0$. If $T\geq\eta+3C+12$, then
    \begin{align}
        \|(H-G)|_{E_H(E+\eta)}\|
        \leq |S|\left(8C e^{-(T-\eta-3C-12)/8}
          +\frac{t}{m}\left(\frac Tt\right)^m\right).
    \end{align}
    Put $U=2(\eta+D+3C+2)$. If $t\geq U$ and $T\geq U$, then
    \begin{align}
        \|(H-G)|_{E_G(\mu+\eta)}\|
        \leq |S|\left(8C e^{-(T-U)/8}
          +\frac{t}{m}\left(\frac Tt\right)^m\right).
    \end{align}
    Orthonormal eigenbases for $H$ and $G$ are constructed with the
    corresponding ground data, before every $\eta,T$. Empty regional intervals,
    repeated cuts, an empty selected set and $L=0$ are included.
\end{lemma}
\begin{proof}\leanok
    For the original-side estimate, use the raw centered interior
    and the floor $u=\eta+3C+2$ above $e_j+u$.
    Since $E\leq C$, the weighted restriction denominator is at
    least two. The center estimate gives
    $e_j+u\leq p_j+T$ and
    $p_j+T-e_j-u\geq T-\eta-3C-12$.

    For the reverse estimate, shift the same soft Hamiltonian by
    $DI$ and use its actual shifted eigenvalues and cutoff.
    At order at least two and scale $t\geq U$, the projection floor
    is at least $U/2=\eta+D+3C+2$. Therefore
    \begin{align}
        U-\frac{t}{m}\left(\frac Ut\right)^m
          -(\mu+\eta+D)-2C\geq2.
    \end{align}
    Lemma~\ref{lem:regional_barrier_soft_residual} applies to the
    original local boundary in both cases. Each denominator is at
    least two, so the exponential coefficient is at most $8C$.
    The discrepancy $H-G$ is the sum of the selected residuals;
    summing their restricted norms gives the factor $|S|$.
\end{proof}

The estimates are corrected auxiliary bounds for
\cite[lines 1178--1255 and 1373--1392]{Arad2016RigorousRG}.
They do not assert the printed uniform energy estimate or the printed
spectral enlargement. The same-operator spectral comparison and MPO representation are proved
in Lemma~\ref{lem:alvv_same_soft_representation}. Final filter parameter
choices require further arguments. The false printed
cluster estimate and its counterexamples remain recorded in
\cite{gap:alvv16_cluster_approximation}.
