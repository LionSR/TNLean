\section{Evolution on disjoint regions}
\label{sec:qc_disjoint_relative_evolution}

The approximation of one-dimensional propagation by two layers of block
unitaries uses localized interaction-picture evolutions
\cite{Osborne2006Dynamics,Huang2015GroundState}. The following auxiliary
statements assemble the evolutions on disjoint regions. They do not choose
the chain partition or establish its Lieb--Robinson estimates.

\begin{lemma}[Product of commuting evolution paths]
    \label{thm:qc_disjoint_evolution_product_rule}
    \lean{MatrixEvolution.hasDerivAt_ordered_prod_of_commute}
    \leanok
    Let $U_j(t)$ and $G_j(t)$ be square complex matrices, indexed by
    $0\le j<m$. Fix a real time $t$. Suppose that
    $U_j'(t)=iG_j(t)U_j(t)$ and that $U_j(t)$ commutes with
    $G_k(t)$ whenever $j\ne k$. Then the ordered product
    $W(t)=U_0(t)\cdots U_{m-1}(t)$ satisfies
    \begin{align}
        W'(t)=i\Bigl(\sum_{j=0}^{m-1}G_j(t)\Bigr)W(t).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Apply the product rule, preserving the order of the factors. In the
    summand obtained by differentiating $U_j$, commute its generator
    through the preceding factors. Summing gives the assertion.
\end{proof}

\begin{lemma}[Approximation by projected evolutions on disjoint regions]
    \label{thm:qc_disjoint_projected_propagator}
    \lean{QuantumCircuit.exists_disjoint_projected_propagator_approximation}
    \leanok
    \uses{thm:qc_disjoint_evolution_product_rule,
        thm:qc_supported_projected_relative_evolution}
    Let the sites form a finite set, let $q\ge1$, and let $K_j$,
    $0\le j<m$, be pairwise disjoint sets of sites. Let $A,B_j$ be
    Hermitian operators and put $H=A-\sum_jB_j$. Write $E_{K_j}$ for
    normalized partial-trace expectation onto $K_j$, and set
    \begin{align}
        G_j(s)=E_{K_j}\bigl(e^{isA}B_je^{-isA}\bigr).
    \end{align}
    There exist unitary paths $U_j(s)$ acting on $K_j$, with
    $U_j(0)=1$ and $U_j'(s)=iG_j(s)U_j(s)$, having the following
    property. For every real $t$ and every real family $\varepsilon_j$
    such that
    \begin{align}
        \bigl\|e^{isA}B_je^{-isA}-G_j(s)\bigr\|
        \le\varepsilon_j
    \end{align}
    for all $j$ and all $s$ between $0$ and $t$, one has
    \begin{align}
        \bigl\|e^{-itH}-e^{-itA}U_0(t)\cdots U_{m-1}(t)\bigr\|
        \le |t|\sum_j\varepsilon_j.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Construct each path by the supported projected-evolution theorem,
    applied to $A$ and $A-B_j$. Disjointness of the regions gives the
    commutations in Lemma~\ref{thm:qc_disjoint_evolution_product_rule}.
    Thus their product is generated by $\sum_jG_j$.
    The exact relative path $e^{isA}e^{-isH}$ is generated by
    $\sum_je^{isA}B_je^{-isA}$. The generator difference has norm at
    most $\sum_j\varepsilon_j$ on the interval. Duhamel comparison
    gives the bound for the relative paths. Multiplication by
    $e^{-itA}$ preserves the operator norm and gives the stated bound.
\end{proof}
