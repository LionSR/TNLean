\section{Regional states under relabelling}
\label{sec:area_law_regional_relabelling}

\begin{definition}[Relabelling physical configurations]
    \label{def:area_law_regional_relabelling}
    \lean{TNLean.PEPS.AreaLaw.regionConfigurationEquiv,
        TNLean.PEPS.AreaLaw.relabelState}
    \leanok
    \uses{def:area_law_domain}
    Let $e:\Lambda\to\Lambda'$ be a bijection of finite domains, and let
    $A\subseteq\Lambda$ and $A'\subseteq\Lambda'$ satisfy $e(A)=A'$,
    inducing bijections $e_A:A\to A'$ and
    $e_B:\Lambda\setminus A\to\Lambda'\setminus A'$.
    For any vector $\Omega$, define $\Omega'$ by
    \begin{align}
        \Omega'(\sigma') & =\Omega(\sigma'\circ e),
    \end{align}
    and transform regional configurations by $e_A$ and $e_B$.
\end{definition}

\begin{lemma}[Covariance of reductions and invariance of entropy]
    \label{thm:area_law_regional_relabelling}
    \lean{TNLean.PEPS.AreaLaw.norm_relabelState,
        TNLean.PEPS.AreaLaw.reducedState_relabel,
        TNLean.PEPS.AreaLaw.regionalEntropy_relabel}
    \leanok
    \uses{def:area_law_regional_relabelling,def:area_law_ground_entropy}
    For configurations $\sigma,\tau:A\to\{0,\ldots,q-1\}$, write
    $\rho_A^\Omega(\sigma,\tau)$ for the $(\sigma,\tau)$ entry of the
    reduced state $\rho_A$ of Definition~\ref{def:area_law_ground_entropy}.
    For every vector $\Omega$, every bijection $e:\Lambda\to\Lambda'$ with
    $e(A)=A'$, and all $\sigma',\tau':A'\to\{0,\ldots,q-1\}$,
    \begin{align}
        \|\Omega'\|                        & =\|\Omega\|,
        \label{eq:area_law_relabel_norm}                                                      \\
        \rho_{A'}^{\Omega'}(\sigma',\tau') & =\rho_A^\Omega(\sigma'\circ e_A,\tau'\circ e_A),
        \label{eq:area_law_relabel_reduction}                                                 \\
        S_{\Omega'}(A')                    & =S_\Omega(A).
        \label{eq:area_law_relabel_entropy}
    \end{align}
    These identities express the tensor-factor conventions of
    \cite[Section~2]{OpenAI2026AreaLaw} in induced-domain coordinates.
\end{lemma}
\begin{proof}\leanok
    For $\sigma:A\to\{0,\ldots,q-1\}$ and
    $\upsilon:\Lambda\setminus A\to\{0,\ldots,q-1\}$, write $\sigma\sqcup\upsilon$
    for the configuration on $\Lambda$ restricting to $\sigma$ on $A$ and
    $\upsilon$ on $\Lambda\setminus A$.
    Since $\sigma'\mapsto\sigma'\circ e$ is a bijection of configurations,
    \begin{align}
        \|\Omega'\|^2 & =\sum_{\sigma'}|\Omega(\sigma'\circ e)|^2
        =\sum_\sigma|\Omega(\sigma)|^2=\|\Omega\|^2,
    \end{align}
    which proves (\ref{eq:area_law_relabel_norm}). Because $e$ maps $A$ onto $A'$ and
    $\Lambda\setminus A$ onto $\Lambda'\setminus A'$,
    \begin{align}
        (\sigma'\sqcup\upsilon')\circ e
         & =(\sigma'\circ e_A)\sqcup(\upsilon'\circ e_B).
        \label{eq:area_law_relabel_split}
    \end{align}
    Substituting (\ref{eq:area_law_relabel_split}) into the partial trace and
    reindexing by the bijection $\upsilon'\mapsto\upsilon'\circ e_B$ gives
    \begin{align}
        \rho_{A'}^{\Omega'}(\sigma',\tau')
         & =\sum_{\upsilon'}\Omega'(\sigma'\sqcup\upsilon')\,
        \overline{\Omega'(\tau'\sqcup\upsilon')}                                           \\
         & =\sum_{\upsilon'}\Omega\big((\sigma'\circ e_A)\sqcup(\upsilon'\circ e_B)\big)\,
        \overline{\Omega\big((\tau'\circ e_A)\sqcup(\upsilon'\circ e_B)\big)}              \\
         & =\sum_{\upsilon}\Omega\big((\sigma'\circ e_A)\sqcup\upsilon\big)\,
        \overline{\Omega\big((\tau'\circ e_A)\sqcup\upsilon\big)}
        =\rho_A^\Omega(\sigma'\circ e_A,\tau'\circ e_A),
    \end{align}
    which proves (\ref{eq:area_law_relabel_reduction}). Thus
    $\rho_{A'}^{\Omega'}=P\rho_A^\Omega P^{-1}$ for the permutation matrix $P$ of
    $\sigma'\mapsto\sigma'\circ e_A$. Since conjugation by $P$ preserves the
    spectrum, (\ref{eq:area_law_relabel_entropy}) follows.
\end{proof}
