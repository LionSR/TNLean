\section{The physical boundary symmetry of a CZX rectangle}%
\label{sec:peps_czx_rectangle_boundary}

Let $R=\lbrack x_0,x_0+w)\times\lbrack y_0,y_0+h)$ be a coordinate rectangle in a
$W\times H$ torus with $W,H\ge3$, $0<w<W$, $0<h<H$, $x_0+w\le W$, and
$y_0+h\le H$, and write $P=2w+2h$. Using the open-region map $A_R$ (summing
only internal bonds and leaving crossing bonds free), we identify the
plaquette spins in \cite[lines~330--345]{ChenLiuWen2011CZX} with $\operatorname{im}A_R$.

\begin{definition}[Clockwise native rectangle boundary]%
    \label{def:czx_rectangle_perimeter}
    \lean{TNLean.PEPS.rectanglePerimeterCorner,
        TNLean.PEPS.rectanglePerimeterEnd,
        TNLean.PEPS.rectanglePerimeterReversed,
        TNLean.PEPS.torusRectanglePerimeterCode,
        TNLean.PEPS.torusRectanglePerimeterEdge,
        TNLean.PEPS.torusRectanglePerimeterSite,
        TNLean.PEPS.torusRectanglePerimeterLeg,
        TNLean.PEPS.torusRectanglePerimeterCorner}
    \leanok
    \uses{def:peps_torus_rectangular_grid}
    Number the crossing bonds of $R$ clockwise (top, right, bottom, left),
    with first plaquette corners $(i,h)$, $(w,h-j)$, $(w-k,0)$, and $(0,\ell)$
    on the four sides, and reverse the native pair order on the bottom and left.
\end{definition}

\begin{theorem}[Native perimeter and distinct plaquette corners]%
    \label{thm:czx_rectangle_perimeter}
    \lean{TNLean.PEPS.even_rectanglePerimeter,
        TNLean.PEPS.rectanglePerimeterEnd_eq_corner_add_one,
        TNLean.PEPS.torusRectanglePerimeterEdge_boundary,
        TNLean.PEPS.torusRectanglePerimeterEdge_injective,
        TNLean.PEPS.torusRectanglePerimeterEquiv,
        TNLean.PEPS.torusRectanglePerimeterEquiv_apply_val,
        TNLean.PEPS.torusRectanglePerimeterSite_mem,
        TNLean.PEPS.torusRectanglePerimeterLeg_val,
        TNLean.PEPS.rectanglePerimeterCorner_le,
        TNLean.PEPS.rectanglePerimeterCorner_injective,
        TNLean.PEPS.torusRectanglePerimeterCorner_injective}
    \leanok
    \uses{def:czx_rectangle_perimeter}
    The numbering of Definition~\ref{def:czx_rectangle_perimeter} is a
    bijection $\{0,\ldots,P-1\}\to\partial R$ (including rectangles meeting a
    coordinate seam) whose cyclic successor is addition in $\Z/P\Z$; every bond
    has its specified endpoint in $R$, the perimeter corners are distinct in
    the torus, and $P$ is even.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:peps_torus_rectangle_boundary_card}
    The four native edge families are pairwise disjoint and injective,
    Lemma~\ref{lem:peps_torus_rectangle_boundary_card} gives surjectivity, and
    $w<W$, $h<H$ make the closed corner grid injective modulo $(W,H)$.
\end{proof}

\begin{theorem}[Exact coefficients and disjoint boundary columns]%
    \label{thm:czx_open_region_coefficients}
    \lean{TNLean.PEPS.czx_regionIncidentConfig_unique,
        TNLean.PEPS.openRegionWeight_czxPEPS,
        TNLean.PEPS.openRegionWeight_czxPEPS_disjoint,
        TNLean.PEPS.openRegionCoefficientMatrix_czxPEPS_gram_offDiagonal}
    \leanok
    \uses{def:peps_open_region_contraction,def:peps_czx_tensor}
    For any finite region $R$, a physical configuration $\sigma$ determines at
    most one nonzero incident-bond assignment (with no sum over edges wholly
    outside $R$), and $(A_R)_{\sigma,\mu}=1$ iff this assignment exists and
    has boundary $\mu$, and $0$ otherwise. Distinct boundary columns $\mu$
    have disjoint physical supports, so $A_R^\dagger A_R$ is diagonal in the
    native boundary basis.
\end{theorem}
\begin{proof}\leanok
    By Definition~\ref{def:peps_czx_tensor}, the four physical qubits at each
    $v\in R$ fix all four incident bond labels, uniquely determining the
    incident-bond assignment in Definition~\ref{def:peps_open_region_contraction}
    with weight $1$ when compatible with $\mu$ and $0$ otherwise.
\end{proof}

\begin{definition}[Crossing-bond symmetry]%
    \label{def:czx_region_boundary_symmetry}
    \lean{TNLean.PEPS.czxRegionBoundaryFlip,
        TNLean.PEPS.czxRegionBoundaryPhase}
    \leanok
    \uses{def:peps_czx_leg_operator,def:peps_open_region_contraction}
    For a native boundary configuration $\mu=(\mu_e)_{e\in\partial R}$ with
    $\mu_e=(a_e,b_e)$, let $f_R\mu$ flip both bits on each $e\in\partial R$
    and put $\phi_R(\mu)=\prod_{e\in\partial R}(-1)^{a_eb_e}$.
\end{definition}

\begin{theorem}[Physical region pull-through]%
    \label{thm:czx_region_physical_pullthrough}
    \lean{TNLean.PEPS.czxLegPhase_swap,
        TNLean.PEPS.czxLegFlip_swap,
        TNLean.PEPS.regionPhysicalMap_czxOnSite_openRegionWeight,
        TNLean.PEPS.regionPhysicalProductMatrix_czxOnSite_intertwines}
    \leanok
    \uses{def:czx_region_boundary_symmetry,def:peps_czx_on_site}
    For any finite region $R$, let $U_R$ apply the CZX on-site symmetry
    (Definition~\ref{def:peps_czx_on_site}) at each site of $R$. Then reversal
    of a native bond pair preserves its phase and commutes with simultaneous
    bit flip, and
    \begin{align}
        U_RA_R|\mu\rangle & =\phi_R(\mu)A_R|f_R\mu\rangle.
        \label{eq:czx_region_physical_pullthrough}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_czx_pull_through,thm:peps_region_physical_maps}
    Applying Theorem~\ref{thm:peps_czx_pull_through} and
    Theorem~\ref{thm:peps_region_physical_maps} inside the finite contraction,
    each internal bond contributes $((-1)^{ab})^2=1$ while each $e\in\partial R$
    contributes $(-1)^{a_eb_e}$; reindexing the incident-bond sum by simultaneous
    bit flips gives (\ref{eq:czx_region_physical_pullthrough}).
\end{proof}

\begin{definition}[Oriented rectangle boundary labels]%
    \label{def:czx_rectangle_boundary_labels}
    \lean{TNLean.PEPS.czxRectangleBoundaryLabels,
        TNLean.PEPS.czxRectangleCoordinatesEquiv,
        TNLean.PEPS.czxRectangleCoordinatesEquiv_apply,
        TNLean.PEPS.czxRectangleEffectiveBoundaryConfig,
        TNLean.PEPS.czxRectangleBoundaryMatrix}
    \leanok
    \uses{def:czx_rectangle_perimeter,def:peps_czx_boundary_embedding,
        def:peps_open_region_contraction}
    Let $E_R\mu$ be the bijective clockwise label sequence of
    Definition~\ref{def:czx_rectangle_perimeter} with bottom and left pairs
    reversed. For $c\in\{0,1\}^P$, set $\iota(c)_i=(c_i,c_{i+1})$,
    $\mu(c)=E_R^{-1}\iota(c)$, and $F_R|c\rangle=A_R|\mu(c)\rangle$.
\end{definition}

\begin{theorem}[The contraction imposes the cyclic plaquette constraint]%
    \label{thm:czx_rectangle_boundary_support}
    \lean{TNLean.PEPS.czxRectangleBoundaryLabels_mem_range_of_openRegionWeight_ne_zero}
    \leanok
    \uses{def:czx_rectangle_boundary_labels}
    If $A_R|\mu\rangle\ne0$, then $E_R\mu=\iota(c)$ for some $c\in\{0,1\}^P$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_rectangle_perimeter,thm:czx_open_region_coefficients}
    By Theorems~\ref{thm:czx_rectangle_perimeter} and
    \ref{thm:czx_open_region_coefficients}, the unique nonzero incident-bond
    assignment equates adjacent physical qubits across each shared internal
    bond along straight sides and within one site tensor at each corner,
    enforcing $E_R\mu=\iota(c)$.
\end{proof}

\begin{definition}[Plaquette-bit configurations]%
    \label{def:czx_plaquette_witness}
    \lean{TNLean.PEPS.czxPlaquettePhysical,TNLean.PEPS.czxPlaquetteBondConfig}
    \leanok
    \uses{def:peps_czx_tensor}
    Given plaquette-corner bits $q$, assign site $(x,y)$ the physical qubits
    $(q_{x,y+1},q_{x+1,y+1},q_{x+1,y},q_{x,y})$ and each native bond the
    corresponding adjacent pair.
\end{definition}

\begin{theorem}[Plaquette-bit configurations and a faithful effective boundary]%
    \label{thm:czx_rectangle_boundary_faithful}
    \lean{TNLean.PEPS.czxPlaquetteBondConfig_right,
        TNLean.PEPS.czxPlaquetteBondConfig_up,
        TNLean.PEPS.component_czxPlaquettePhysical,
        TNLean.PEPS.openRegionWeight_czxPlaquettePhysical,
        TNLean.PEPS.czxRectangleBoundaryLabels_plaquette,
        TNLean.PEPS.czxRectangleEffectiveBoundaryConfig_injective,
        TNLean.PEPS.exists_czxRectanglePlaquetteBoundary,
        TNLean.PEPS.exists_czxRectangleBoundaryMatrix_selector,
        TNLean.PEPS.czxRectangleBoundaryMatrix_mulVec_injective}
    \leanok
    \uses{def:czx_rectangle_boundary_labels,def:czx_plaquette_witness}
    For Definition~\ref{def:czx_plaquette_witness}, each local tensor and
    region configuration coefficient equals $1$, and for every $c\in\{0,1\}^P$
    there exists $\sigma_c$ with $F_R(\sigma_c,c')=\delta_{c,c'}$; in
    particular, $F_R$ is injective.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_rectangle_perimeter,thm:czx_open_region_coefficients}
    Extending $c\in\{0,1\}^P$ from the distinct perimeter corners
    (Theorem~\ref{thm:czx_rectangle_perimeter}) by $0$ elsewhere produces
    $\sigma_c$ with $F_R(\sigma_c,c')=\delta_{c,c'}$ by
    Theorem~\ref{thm:czx_open_region_coefficients}.
\end{proof}

\begin{theorem}[Exact physical image and its dimension]%
    \label{thm:czx_rectangle_physical_image}
    \lean{TNLean.PEPS.range_openRegionMap_czxRectangle,
        TNLean.PEPS.finrank_range_openRegionMap_czxRectangle}
    \leanok
    \uses{def:czx_rectangle_boundary_labels}
    One has $\operatorname{im}A_R=\operatorname{im}F_R$ and
    $\dim\operatorname{im}A_R=2^P$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_rectangle_boundary_support,thm:czx_rectangle_boundary_faithful}
    Theorem~\ref{thm:czx_rectangle_boundary_support} gives
    $\operatorname{im}A_R=\operatorname{im}F_R$, and the selectors $\sigma_c$
    of Theorem~\ref{thm:czx_rectangle_boundary_faithful} show that the $2^P$
    columns of $F_R$ are linearly independent.
\end{proof}

\begin{theorem}[Physical action on the effective plaquette spins]%
    \label{thm:czx_rectangle_effective_symmetry}
    \lean{TNLean.PEPS.regionPhysicalMap_czxRectangleBoundaryMatrix_column,
        TNLean.PEPS.regionPhysicalProductMatrix_czxRectangleBoundaryMatrix_monomial}
    \leanok
    \uses{def:czx_rectangle_boundary_labels}
    On the effective boundary coordinates of
    Definition~\ref{def:czx_rectangle_boundary_labels},
    \begin{align}
        U_RF_R|c\rangle
         & =(-1)^{\sum_i c_ic_{i+1}}F_R|\bar c\rangle.
        \label{eq:czx_rectangle_effective_symmetry}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_region_physical_pullthrough,thm:czx_rectangle_boundary_faithful}
    Applying (\ref{eq:czx_region_physical_pullthrough}) of
    Theorem~\ref{thm:czx_region_physical_pullthrough} to $\mu(c)$ and using
    $\phi_R(\mu(c))=(-1)^{\sum_i c_ic_{i+1}}$ and $f_R\mu(c)=\mu(\bar c)$ gives
    (\ref{eq:czx_rectangle_effective_symmetry}).
\end{proof}

\begin{theorem}[The printed matrix product operator on the physical rectangle image]%
    \label{thm:czx_rectangle_review_mpo}
    \lean{TNLean.PEPS.regionPhysicalProductMatrix_czxRectangleBoundaryMatrix_review}
    \leanok
    \uses{def:czx_rectangle_boundary_labels,def:asymex_czx_review_tensor}
    The printed CZX matrix product operator $O(A)$ of
    Definition~\ref{def:asymex_czx_review_tensor} satisfies $U_RF_R=F_RO(A)$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_rectangle_effective_symmetry,thm:czx_rectangle_perimeter,
        thm:asymex_czx_kernel,thm:asymex_czx_review_kernel}
    By Theorems~\ref{thm:asymex_czx_kernel} and
    \ref{thm:asymex_czx_review_kernel}, $D_PX^{\otimes P}=(-1)^PX^{\otimes P}D_P$;
    since $P$ is even by Theorem~\ref{thm:czx_rectangle_perimeter}, combining
    with (\ref{eq:czx_rectangle_effective_symmetry}) gives $U_RF_R=F_RO(A)$
    (without substituting the normalized bra-ket display for the printed matrices).
\end{proof}

\begin{theorem}[Closed-state slices and reduced-density support]%
    \label{thm:czx_rectangle_density_support}
    \lean{TNLean.PEPS.openRegionWeight_czxPlaquetteComplement,
        TNLean.PEPS.stateCoeff_czxPlaquetteComplement,
        TNLean.PEPS.range_regionReducedDensity_czxRectangle,
        TNLean.PEPS.rank_regionReducedDensity_czxRectangle}
    \leanok
    \uses{def:czx_rectangle_boundary_labels,def:czx_plaquette_witness,
        def:peps_region_reduced_density}
    For every $c\in\{0,1\}^P$ there exists a complement physical configuration
    whose closed-state slice equals $F_R|c\rangle$. Consequently the
    unnormalized reduced density $\rho_R$ of
    Definition~\ref{def:peps_region_reduced_density} satisfies
    \begin{align}
        \operatorname{ran}\rho_R
         & =\operatorname{im}F_R=\operatorname{ran}\Gamma_R,
         & \operatorname{rank}\rho_R                         & =2^P.
        \label{eq:czx_rectangle_density_support}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_open_region_coefficients,thm:czx_rectangle_boundary_faithful,
        thm:czx_rectangle_physical_image,thm:peps_open_region_cut_unconditional,
        thm:peps_region_reduced_density_support}
    Extending $c\in\{0,1\}^P$ to global plaquette bits
    (Definition~\ref{def:czx_plaquette_witness}), the complement configuration
    has weight $1$ on $\mu(c)$ and $0$ on all $\mu\ne\mu(c)$ by
    Theorem~\ref{thm:czx_open_region_coefficients}, so
    Theorem~\ref{thm:peps_open_region_cut_unconditional} reduces the cut
    contraction to $F_R|c\rangle$. Combining with
    Theorems~\ref{thm:czx_rectangle_boundary_faithful},
    \ref{thm:czx_rectangle_physical_image}, and
    \ref{thm:peps_region_reduced_density_support} yields
    (\ref{eq:czx_rectangle_density_support}).
\end{proof}

\begin{theorem}[Trace normalization preserves the physical support]%
    \label{thm:czx_rectangle_density_normalization}
    \lean{TNLean.PEPS.trace_regionReducedDensity_czxRectangle_ne_zero,
        TNLean.PEPS.range_normalizedRegionReducedDensity_czxRectangle,
        TNLean.PEPS.rank_normalizedRegionReducedDensity_czxRectangle}
    \leanok
    \uses{def:peps_region_reduced_density,def:czx_rectangle_boundary_labels}
    One has $\operatorname{Tr}\rho_R\ne0$, and $\rho_R/\operatorname{Tr}\rho_R$
    has range $\operatorname{im}F_R$ and rank $2^P$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:czx_rectangle_density_support,thm:peps_region_reduced_density_support}
    By Theorems~\ref{thm:czx_rectangle_density_support} and
    \ref{thm:peps_region_reduced_density_support}, $\rho_R\ge0$ has rank
    $2^P>0$, so $\operatorname{Tr}\rho_R\ne0$ and rescaling by
    $(\operatorname{Tr}\rho_R)^{-1}$ preserves its range and rank.
\end{proof}

No formula for the nonzero eigenvalues, flatness, or entropy is asserted here,
nor is $F_R$ asserted to be an isometry; nonrectangular regions remain outside
the perimeter construction.
