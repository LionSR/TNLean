\begin{theorem}[Regular torus parent constraints force plaquette flatness]%
    \label{thm:peps_regular_torus_parent_flatness}
    \lean{TNLean.PEPS.IsGInjective.torusPlaquetteHolonomy_eq_one_of_parent_annihilates,
        TNLean.PEPS.IsGInjective.regionParent_not_annihilates_torusPlaquetteFlux,
        TNLean.PEPS.IsGInjective.torusPlaquetteHolonomy_eq_one_of_parentHamiltonian_annihilates,
        TNLean.PEPS.torusPlaquetteHolonomy_eq_one_iff_squareTransport,
        TNLean.PEPS.IsGInjective.isTorusFlat_of_parentHamiltonian_annihilates_twistedState}
    \leanok
    \uses{thm:peps_regular_region_parent_flatness,
        def:peps_region_parent_hamiltonian}
    Place a native four-leg regular $G$-injective tensor on a square torus
    $\Z_{L_x}\times\Z_{L_y}$ with $L_x,L_y\ge 3$, and insert regular group
    operators $u$ on its ordered bonds so that the physical state satisfies
    $\Psi(A^u)\ne 0$. If an original plaquette parent term $h_P^{\mathrm{global}}$
    satisfies $h_P^{\mathrm{global}}\Psi(A^u)=0$, then the plaquette holonomy
    is $\operatorname{hol}_u(P)=1$.

    If the parent Hamiltonian $H=\sum_P h_P^{\mathrm{global}}$
    (Definition~\ref{def:peps_region_parent_hamiltonian}) satisfies
    $H\Psi(A^u)=0$, then $\operatorname{hol}_u(P)=1$ for every elementary
    plaquette $P$; equivalently, with rightward and upward bond operators
    $A_v,B_v\in G$, every vertex $v$ obeys
    \begin{align}
        B_{v+(1,0)}A_v & = A_{v+(0,1)}B_v.
        \label{eq:peps_torus_parent_flatness_transport}
    \end{align}
    In particular, a nonzero one-bond flux state $\Psi_g$ with $g\ne 1$
    satisfies $h_P^{\mathrm{global}}\Psi_g\ne 0$ on the plaquette $P$
    surrounding its endpoint.

    No $G$-isometry is assumed. This is a necessary local step in
    \cite[Theorem~5.5]{Schuch2010PEPS}; it does not claim that
    arbitrary parent-kernel vectors have inserted-bond descriptions,
    or that flatness alone gives the full torus ground-space classification.
    The regular native-torus scope and the remaining global representation step are recorded in
    \cite{gap:rmp_peps_quantum_double_g_isometry}.
\end{theorem}

\begin{proof}\leanok
    Each native plaquette $P$ is the support of a four-step closed walk,
    so its induced subgraph is connected; Theorem~\ref{thm:peps_regular_region_parent_flatness}
    applied to the induced walk gives $h_P^{\mathrm{global}}\Psi(A^u)=0\implies\operatorname{hol}_u(P)=1$.
    Since $h_P^{\mathrm{global}}\ge 0$, $H\Psi(A^u)=0$ implies
    $h_P^{\mathrm{global}}\Psi(A^u)=0$ for every $P$.
    The ordered plaquette holonomy
    $\operatorname{hol}_u(P)=B_v^{-1}A_{v+(0,1)}^{-1}B_{v+(1,0)}A_v$ equals
    $1$ if and only if~(\ref{eq:peps_torus_parent_flatness_transport}) holds.
    A one-bond insertion of $g\ne 1$ has $\operatorname{hol}_u(P)=g\ne 1$,
    so $h_P^{\mathrm{global}}\Psi_g\ne 0$.
\end{proof}
