\section{Exact operator contraction for distributed PEPS compression}
\label{sec:peps_distributed_operator}
This is the finite operator representation used in
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{04-compression.tex}, lines 565--579.
Let $P$ be a finite party set, $G$ a finite set of gate occurrences,
$P_g\subseteq P$ the participant set of $g\in G$, $r_g\in P_g$ its chosen root,
$\Lambda_g$ a finite branch alphabet (possibly empty), $J_k=\{0,\ldots,k-1\}$ for $k\in\mathbb N$,
and $Y_p,X_p$ the physical input and output alphabets at $p\in P$.
% This chapter proves representation algebra for supplied party-local matrices.
% Identifying an actual chronological circuit with those matrices and the global
% branch/sample sum, and proving uniform polynomial label bounds, remain open.

\begin{definition}[Active gate occurrences]
    \label{def:compression_operator_active_gates}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.ActiveGate}
    \leanok
    The set of active gate occurrences is the finite set $G$ (with distinct elements for repeated gate applications).
\end{definition}

\begin{definition}[Tagged star and sample links]
    \label{def:compression_operator_links}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.Link}
    \leanok
    The tagged link family is the disjoint union
    \begin{align}
        \mathcal L := \bigsqcup_{g\in G}\Bigl(\{(g,\mathrm{star},q):q\in P_g\setminus\{r_g\}\}\sqcup\{(g,\mathrm{sample},\{p,q\}):\{p,q\}\subseteq P_g,\ p\ne q\}\Bigr). \notag
    \end{align}
\end{definition}

\begin{definition}[Gate occurrences at a party]
    \label{def:compression_operator_party_gates}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.GateAt}
    \leanok
    For $p\in P$, the set of incident gate occurrences is $G_p:=\{g\in G:p\in P_g\}$.
\end{definition}

\begin{definition}[Pair-sample positions]
    \label{def:compression_operator_samples}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.Sample}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.SampleAt}
    \leanok
    The pair-sample position set and its party restrictions are
    \begin{align}
        S   & := \bigsqcup_{g\in G}\{\{p,q\}\subseteq P_g:p\ne q\}, &
        S_p & := \{s=(g,\{q,q'\})\in S:p\in\{q,q'\}\}. \notag
    \end{align}
\end{definition}

\begin{definition}[Star and sample edges]
    \label{def:compression_operator_edges}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.starLink}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.sampleLink}
    \leanok
    For $g\in G$ and $q\in P_g\setminus\{r_g\}$, the star edge is $\ell_{\mathrm{star}}(g,q):=(g,\mathrm{star},q)\in\mathcal L$. For $s=(g,\{p,q\})\in S$, the sample edge is $\ell_{\mathrm{sample}}(s):=(g,\mathrm{sample},\{p,q\})\in\mathcal L$.
\end{definition}

\begin{definition}[Oriented link endpoints]
    \label{def:compression_operator_endpoints}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.src}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.tgt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.endpoints_eq}
    \leanok
    Each $\ell\in\mathcal L$ has oriented endpoints $(u_\ell,v_\ell)$ given by $(r_g,q)$ for $\ell=(g,\mathrm{star},q)$ and by a fixed ordering $(s^-,s^+)$ of $\{p,q\}$ for $\ell=(g,\mathrm{sample},\{p,q\})$, so $\{u_\ell,v_\ell\}$ is the endpoint set of $\ell$.
\end{definition}

\begin{definition}[Party incidences]
    \label{def:compression_operator_incidences}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.LinkAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.starLinkAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.sampleLinkAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.starGateAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.rootGateAt}
    \leanok
    For $p\in P$, let $I_p:=\{\ell\in\mathcal L:p\in\{u_\ell,v_\ell\}\}$. Then $r_g\in P_g$ gives $g\in G_{r_g}$, $(g,\mathrm{star},q)\in I_p$ whenever $p\in\{r_g,q\}$ (with $g\in G_p$), and $s\in S_p$ gives $\ell_{\mathrm{sample}}(s)\in I_p$.
\end{definition}

\begin{definition}[Virtual alphabets]
    \label{def:compression_operator_alphabets}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.alphabet}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.alphabetFintype}
    \leanok
    The finite virtual alphabet of $\ell\in\mathcal L$ is $D_\ell:=\Lambda_g$ for $\ell=(g,\mathrm{star},q)$ and $D_\ell:=J_k$ for $\ell=(g,\mathrm{sample},s)$.
\end{definition}

\begin{definition}[Gate-label assignments]
    \label{def:compression_operator_gate_labels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.GateLabels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.GateLabelsAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.restrictGates}
    \leanok
    Global and party-local gate assignments are elements $\kappa\in\prod_{g\in G}\Lambda_g$ and $\xi_p\in\prod_{g\in G_p}\Lambda_g$, with restriction $\kappa_p:=\kappa|_{G_p}$.
\end{definition}

\begin{definition}[Virtual-label assignments]
    \label{def:compression_operator_link_labels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.LinkLabels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.LinkLabelsAt}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.restrictLinks}
    \leanok
    Global and party-local virtual assignments are elements $\eta\in\prod_{\ell\in\mathcal L}D_\ell$ and $\eta_p\in\prod_{\ell\in I_p}D_\ell$, with restriction $\eta_p:=\eta|_{I_p}$.
\end{definition}

\begin{definition}[Local sample assignments]
    \label{def:compression_operator_sample_labels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.restrictSamples}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.localSamples}
    \leanok
    A global sample assignment $j\in J_k^S$ restricts to $j_p:=j|_{S_p}\in J_k^{S_p}$, and any $\eta_p\in\prod_{\ell\in I_p}D_\ell$ induces $j(\eta_p)\in J_k^{S_p}$ via $j(\eta_p)_s:=\eta_p(g,\mathrm{sample},s)$ for $s\in S_p$.
\end{definition}

\begin{definition}[Shared virtual labels]
    \label{def:compression_operator_shared_labels}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.sharedLinkLabels}
    \leanok
    For $\kappa\in\prod_{g\in G}\Lambda_g$ and $j\in J_k^S$, define $\eta(\kappa,j)\in\prod_{\ell\in\mathcal L}D_\ell$ by
    \begin{align}
        \eta(\kappa,j)_{(g,\mathrm{star},q)}   & := \kappa_g,   &
        \eta(\kappa,j)_{(g,\mathrm{sample},s)} & := j_s. \notag
    \end{align}
\end{definition}

\begin{definition}[Local star consistency]
    \label{def:compression_operator_consistency}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.StarConsistent}
    \leanok
    A pair $(\eta_p,\xi_p)\in\bigl(\prod_{\ell\in I_p}D_\ell\bigr)\times\bigl(\prod_{g\in G_p}\Lambda_g\bigr)$ is star-consistent if $\eta_p(g,\mathrm{star},q)=\xi_p(g)$ for every $(g,\mathrm{star},q)\in I_p$.
\end{definition}

\begin{lemma}[Consistency of shared assignments]
    \label{thm:compression_operator_shared_consistency}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.sharedLinkLabels_consistent}
    \leanok
    For every $(\kappa,j)\in\bigl(\prod_{g\in G}\Lambda_g\bigr)\times J_k^S$ and $p\in P$, the pair $(\eta(\kappa,j)_p,\kappa_p)$ is star-consistent.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_shared_labels,def:compression_operator_consistency}
    By Definitions~\ref{def:compression_operator_shared_labels} and~\ref{def:compression_operator_consistency}, $\eta(\kappa,j)_p(g,\mathrm{star},q)=\kappa_g=\kappa_p(g)$ for every $(g,\mathrm{star},q)\in I_p$.
\end{proof}

\begin{definition}[Root-owned scalar coefficients]
    \label{def:compression_operator_root_coefficients}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.rootCoefficient}
    \leanok
    For coefficients $c_g:\Lambda_g\to\mathbb C$ and $\xi_p\in\prod_{g\in G_p}\Lambda_g$, define $C_p(\xi_p):=\prod_{g\in G:r_g=p}c_g(\xi_p(g))$.
\end{definition}

\begin{definition}[Local operator tensor]
    \label{def:compression_operator_local_tensor}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.localTensor}
    \leanok
    Given matrices $T_p(\xi_p,j_p)\in\mathbb C^{X_p\times Y_p}$, define the local tensor at $p\in P$ by
    \begin{align}
        A_p(\eta_p;x_p,y_p)
         & =\sum_{\xi_p\in\prod_{g\in G_p}\Lambda_g}
        \mathbf1_{\{(\eta_p,\xi_p)\ {\rm consistent}\}}
        C_p(\xi_p)\,T_p(\xi_p,j(\eta_p))_{x_p,y_p}.
        \label{eq:compression_operator_local_tensor}
    \end{align}
\end{definition}

\begin{definition}[Globally consistent configurations]
    \label{def:compression_operator_consistent_configs}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.ConsistentConfig}
    \leanok
    A globally consistent configuration is a pair $(\eta,(\xi_p)_{p\in P})$ such that $(\eta_p,\xi_p)$ is star-consistent (Definition~\ref{def:compression_operator_consistency}) for every $p\in P$.
\end{definition}

\begin{definition}[Recovery at gate roots and sample links]
    \label{def:compression_operator_recovery}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.gateLabelsOfConfig}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.samplesOfConfig}
    \leanok
    From a globally consistent configuration $(\eta,(\xi_p)_{p\in P})$, recover $\kappa_g:=\xi_{r_g}(g)$ for $g\in G$ and $j_s:=\eta_{(g,\mathrm{sample},s)}$ for $s\in S$.
\end{definition}

\begin{lemma}[Every participant agrees with the root]
    \label{thm:compression_operator_root_agreement}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.gateLabel_eq_root}
    \leanok
    In any globally consistent configuration $(\eta,(\xi_p)_{p\in P})$, one has $\xi_p(g)=\kappa_g$ for all $p\in P$ and $g\in G_p$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_consistency,def:compression_operator_recovery}
    For $p=r_g$, $\xi_{r_g}(g)=\kappa_g$ by Definition~\ref{def:compression_operator_recovery}; for $p\in P_g\setminus\{r_g\}$, Definition~\ref{def:compression_operator_consistency} at $p$ and $r_g$ gives $\xi_p(g)=\eta_{(g,\mathrm{star},p)}=\xi_{r_g}(g)=\kappa_g$.
\end{proof}

\begin{lemma}[Recovery of every virtual label]
    \label{thm:compression_operator_link_recovery}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.sharedLinkLabels_recover}
    \leanok
    If $(\kappa,j)$ is recovered from a globally consistent configuration $(\eta,(\xi_p)_{p\in P})$ via Definition~\ref{def:compression_operator_recovery}, then $\eta(\kappa,j)=\eta$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_consistency,def:compression_operator_recovery,def:compression_operator_shared_labels}
    By Definitions~\ref{def:compression_operator_shared_labels}, \ref{def:compression_operator_consistency}, and~\ref{def:compression_operator_recovery}, $\eta(\kappa,j)_{(g,\mathrm{star},q)}=\xi_{r_g}(g)=\eta_{(g,\mathrm{star},q)}$ and $\eta(\kappa,j)_{(g,\mathrm{sample},s)}=j_s=\eta_{(g,\mathrm{sample},s)}$.
\end{proof}

\begin{definition}[Bijection of consistent configurations]
    \label{def:compression_operator_configuration_bijection}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.consistentConfigEquiv}
    \leanok
    By Lemmas~\ref{thm:compression_operator_root_agreement} and~\ref{thm:compression_operator_link_recovery}, the map $(\kappa,j)\mapsto(\eta(\kappa,j),(\kappa_p)_{p\in P})$ is a bijection from $(\prod_{g\in G}\Lambda_g)\times J_k^S$ onto the set of globally consistent configurations, with inverse given by Definition~\ref{def:compression_operator_recovery}.
\end{definition}

\begin{definition}[Reading endpoint configurations]
    \label{def:compression_operator_endpoint_reader}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.endpointLabels}
    \leanok
    Given endpoint label pairs $(\eta_\ell^-,\eta_\ell^+)\in D_\ell^2$ for $\ell\in\mathcal L$, the endpoint reader at $p\in P$ assigns to $\ell\in I_p$ the label $\eta_\ell^-$ if $u_\ell=p$ and $\eta_\ell^+$ otherwise.
\end{definition}

\begin{lemma}[Reading diagonal endpoint labels]
    \label{thm:compression_operator_diagonal_reader}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.endpointLabels_diagonal}
    \leanok
    For diagonal endpoint labels $(\eta_\ell,\eta_\ell)_{\ell\in\mathcal L}$, the endpoint reader of Definition~\ref{def:compression_operator_endpoint_reader} at $p\in P$ returns $\eta_p=\eta|_{I_p}$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_endpoint_reader,def:compression_operator_link_labels}
    Both branches of Definition~\ref{def:compression_operator_endpoint_reader} return $\eta_\ell$ for $\ell\in I_p$.
\end{proof}

\begin{lemma}[Samples of a shared assignment]
    \label{thm:compression_operator_shared_samples}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.localSamples_sharedLinkLabels}
    \leanok
    For every $(\kappa,j)$ and $p\in P$, one has $j(\eta(\kappa,j)_p)=j_p$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_shared_labels,def:compression_operator_sample_labels}
    By Definitions~\ref{def:compression_operator_sample_labels} and~\ref{def:compression_operator_shared_labels}, $j(\eta(\kappa,j)_p)_s=\eta(\kappa,j)_{(g,\mathrm{sample},s)}=j_s=(j_p)_s$ for $s\in S_p$.
\end{proof}

\begin{lemma}[Each scalar coefficient appears once]
    \label{thm:compression_operator_coefficient_product}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.prod_rootCoefficient}
    \leanok
    For every $\kappa\in\prod_{g\in G}\Lambda_g$, one has $\prod_{p\in P}C_p(\kappa_p)=\prod_{g\in G}c_g(\kappa_g)$.
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_root_coefficients}
    Partitioning $G=\bigsqcup_{p\in P}\{g\in G:r_g=p\}$ in Definition~\ref{def:compression_operator_root_coefficients} gives the product identity.
\end{proof}

\begin{definition}[Endpoint form of the local tensor]
    \label{def:compression_operator_endpoint_tensor}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.endpointTensor}
    \leanok
    The endpoint form of $A_p$ evaluates (\ref{eq:compression_operator_local_tensor}) on $(x_p,y_p)$ and the local virtual assignment read at $p$ by Definition~\ref{def:compression_operator_endpoint_reader}.
\end{definition}

\begin{definition}[Contracted operator]
    \label{def:compression_operator_operator}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.contractedOperator}
    \leanok
    Contracting the endpoint tensors of Definition~\ref{def:compression_operator_endpoint_tensor} with identity bond matrices $I_{D_\ell}$ on all $\ell\in\mathcal L$ defines the operator $\sigma:\prod_{p\in P}Y_p\to\prod_{p\in P}X_p$.
\end{definition}

\begin{lemma}[Reduction to one label per link]
    \label{thm:compression_operator_link_sum}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.contractedOperator_eq_linkSum}
    \leanok
    The contracted operator $\sigma$ of Definition~\ref{def:compression_operator_operator} satisfies
    \begin{align}
        \sigma(x,y)
         & =\sum_{\eta\in\prod_{\ell\in\mathcal L}D_\ell}
        \prod_{p\in P}A_p(\eta_p;x_p,y_p).
        \label{eq:compression_operator_link_sum}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{def:compression_operator_operator,thm:compression_operator_diagonal_reader}
    Identity bond matrices in Definition~\ref{def:compression_operator_operator} restrict the endpoint sum to diagonal labels $(\eta_\ell,\eta_\ell)$, where Lemma~\ref{thm:compression_operator_diagonal_reader} gives the local argument $\eta_p$.
\end{proof}

\begin{lemma}[Identity-bond factorization across a vertex partition]
    \label{thm:peps_identity_cut_factorization}
    \lean{TNLean.Lattice.DependentBondNetwork.flattening_network_one_eq_mul,
        TNLean.Lattice.DependentBondNetwork.rank_flattening_network_one_le}
    \leanok
    \uses{def:peps_dependent_network,def:peps_group_contraction}
    Let $(V,E)$ be a finite directed multigraph with finite virtual alphabets $D_e$, local tensors $A_v$ on physical alphabets $P_v$, and identity-bond contraction $Z$.
    For $S\subseteq V$, let $\partial S\subseteq E$ be the edges crossing between $S$ and $S^c$, set $K_S:=\prod_{e\in\partial S}D_e$, let $U_S,U_{S^c}$ be the partial contractions on $S,S^c$ with open boundary indices in $K_S$, and let $F_S$ be the $\bigl(\prod_{v\in S}P_v\bigr)\times\bigl(\prod_{v\in S^c}P_v\bigr)$ flattening of $Z$. Then
    \begin{align}
        F_S & =U_S U_{S^c}^{\mathsf T},
        \label{eq:peps_identity_cut_factor}
    \end{align}
    and if every $P_v$ is finite, then (as in \cite[Part~II, proof of Theorem~B]{TNLean2026OneDimensional})
    \begin{align}
        \operatorname{rank}F_S
         & \le |K_S|=\prod_{e\in\partial S}|D_e|.
        \label{eq:peps_identity_cut_rank}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{cor:peps_group_tensor_network,thm:peps_group_contraction_nuclear}
    With identity bond matrices, Corollary~\ref{cor:peps_group_tensor_network} and Lemma~\ref{thm:peps_group_contraction_nuclear} factor $F_S$ over $\eta\in K_S$ as (\ref{eq:peps_identity_cut_factor}), and the column count $|K_S|$ of $U_S$ gives (\ref{eq:peps_identity_cut_rank}).
\end{proof}

\begin{lemma}[Cut ranks of distributed product-basis columns]
    \label{thm:compression_operator_column_cut_rank}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.flattening_contractedOperator_column_eq_mul,
        TNLean.Lattice.Compression.DistributedOperatorContraction.rank_flattening_contractedOperator_column_le}
    \leanok
    \uses{def:compression_operator_operator,def:compression_operator_links,
        def:compression_operator_alphabets,def:peps_ket_column_dependent_fix_input}
    For $\sigma$ as in Definition~\ref{def:compression_operator_operator}, $z\in\prod_{p\in P}Y_p$, and $S\subseteq P$, let $F_{S,z}$ be the flattening across $S\mid S^c$ of $x\mapsto\sigma(x,z)$, and let $U_{S,z},U_{S^c,z}$ be the partial contractions on $S,S^c$ of the input-fixed local tensors $A_{p,z}(\eta_p,x_p):=A_p(\eta_p;x_p,z_p)$ (Definition~\ref{def:peps_ket_column_dependent_fix_input}). Then
    \begin{align}
        F_{S,z} & =U_{S,z}U_{S^c,z}^{\mathsf T},
        \label{eq:compression_column_cut_factor}
    \end{align}
    and if every $X_p$ is finite, then (as in \cite[Lemma~8.1]{OpenAI2026PolynomialPEPS})
    \begin{align}
        \operatorname{rank}F_{S,z}
         & \le\prod_{\ell\in\partial S}|D_\ell|.
        \label{eq:compression_column_cut_rank}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_identity_cut_factorization,
        def:peps_ket_column_dependent_fix_input,def:compression_operator_operator}
    By Definitions~\ref{def:compression_operator_operator} and~\ref{def:peps_ket_column_dependent_fix_input}, the identity-bond contraction of $(A_{p,z})_{p\in P}$ on $\mathcal L$ is $x\mapsto\sigma(x,z)$, so Lemma~\ref{thm:peps_identity_cut_factorization} gives (\ref{eq:compression_column_cut_factor}) and (\ref{eq:compression_column_cut_rank}).
\end{proof}

\begin{lemma}[Exact branch and sample evaluation]
    \label{thm:compression_operator_branch_sum}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.contractedOperator_eq_branchSampleSum}
    \leanok
    The contracted operator $\sigma$ of Definition~\ref{def:compression_operator_operator} has entries (including when $\Lambda_g=\varnothing$, $\mathcal L=\varnothing$, $|P_g|=1$, or $k=0$)
    \begin{align}
        \sigma(x,y)
         & =\sum_{\kappa\in\prod_{g\in G}\Lambda_g}
        \left(\prod_{g\in G}c_g(\kappa_g)\right)
        \sum_{j\in J_k^S}\prod_{p\in P}T_p(\kappa_p,j_p)_{x_p,y_p}.
        \label{eq:compression_operator_branch_sum}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_operator_link_sum,def:compression_operator_local_tensor,def:compression_operator_configuration_bijection,thm:compression_operator_shared_samples,thm:compression_operator_coefficient_product}
    Substituting (\ref{eq:compression_operator_local_tensor}) into (\ref{eq:compression_operator_link_sum}), reindexing globally consistent configurations by the bijection of Definition~\ref{def:compression_operator_configuration_bijection}, and applying Lemmas~\ref{thm:compression_operator_shared_samples} and~\ref{thm:compression_operator_coefficient_product} yields (\ref{eq:compression_operator_branch_sum}).
\end{proof}

\begin{lemma}[Rectangular matrix form of exact evaluation]
    \label{thm:compression_operator_matrix_sum}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.contractedOperator_eq_sum_productMatrix}
    \leanok
    In matrix form, (\ref{eq:compression_operator_branch_sum}) reads
    \begin{align}
        \sigma
         & =\sum_{\kappa\in\prod_{g\in G}\Lambda_g}
        \left(\prod_{g\in G}c_g(\kappa_g)\right)
        \sum_{j\in J_k^S}\bigotimes_{p\in P}T_p(\kappa_p,j_p).
        \label{eq:compression_operator_matrix_sum}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:compression_operator_branch_sum}
    Evaluating the $(x,y)$ entry of the right-hand side of (\ref{eq:compression_operator_matrix_sum}) gives (\ref{eq:compression_operator_branch_sum}).
\end{proof}

\begin{definition}[Density-branch coefficient]
    \label{def:compression_operator_density_coefficient}
    \lean{TNLean.Lattice.Compression.DistributedOperatorContraction.densityPairCoefficient}

\subsection{Finite coordinates for the distributed network}

\begin{definition}[Enumerated virtual alphabets]
    \label{def:compression_operator_finite_coordinates}
    \lean{TNLean.PEPS.Approximation.DistributedOperatorContraction.toPartyNetwork}
    \leanok
    \uses{def:compression_operator_local_tensor,def:peps_party_network_routing}
    Following \cite[proof of Theorem~5.2 and Lemma~8.2]{OpenAI2026PolynomialPEPS}, given the links and finite alphabets $A_e$ of Definition~\ref{def:compression_operator_operator}, bijections $\vartheta_e:A_e\to[|A_e|]$ ($[m]=\{0,\ldots,m-1\}$), $q\in\mathbb N$, and physical readouts $u_p:[q]\to X_p$, $v_p:[q]\to Y_p$, the enumerated network has virtual dimensions $|A_e|$ and local tensors (with $(s,t)\in[q]^2\simeq[q^2]$)
    \begin{align}
        \widetilde A_p(\eta_p;s,t)
         & = A_p\bigl((\vartheta_e^{-1}(\eta_{p,e}))_{e\ni p};
        u_p(s),v_p(t)\bigr).
        \label{eq:compression_finite_coordinates}
    \end{align}
\end{definition}

\begin{lemma}[Contraction in enumerated coordinates]
    \label{thm:compression_operator_finite_coordinates}
    \lean{TNLean.PEPS.Approximation.DistributedOperatorContraction.operatorCoeff_toPartyNetwork}
    \leanok
    \uses{def:compression_operator_finite_coordinates,def:compression_operator_operator}
    For any site map $p\mapsto a_p\in V$ and $x,y:V\to[q]$, the contracted operator $\widetilde\sigma$ of Definition~\ref{def:compression_operator_finite_coordinates} satisfies
    \begin{align}
        \widetilde\sigma(x,y)
         & = \sigma\bigl((u_p(x(a_p)))_p,(v_p(y(a_p)))_p\bigr).
        \label{eq:compression_coordinate_operator}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:compression_operator_finite_coordinates,thm:compression_operator_link_sum}
    Reindexing the contraction sum (\ref{eq:compression_operator_link_sum}) along the bijection $\prod_e\vartheta_e$ and substituting (\ref{eq:compression_finite_coordinates}) gives (\ref{eq:compression_coordinate_operator}).
\end{proof}
