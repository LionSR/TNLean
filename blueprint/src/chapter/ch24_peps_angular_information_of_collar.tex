\begin{definition}[Constants of the cell information estimate]
    \label{def:peps_cell_information_constants}
    \lean{TNLean.PEPS.Approximation.cellInformationScale,
        TNLean.PEPS.Approximation.cellInformationConstant,
        TNLean.PEPS.Approximation.cellInformationExponent}
    \leanok
    \uses{def:peps_coarse_information_constants,def:peps_cell_partition_constants,
        thm:peps_cell_generation_count}
    Fix $q\in\mathbb N$, $c>0$, $C_{\rm tpl}$, $N$, $C_a$, $N_a$, and recall $\beta_0<1$. Let
    $u_0=u_0(C_{\rm tpl},66^2,1,N,C_a,N_a)$ from
    Definition~\ref{def:peps_coarse_information_constants},
    $u_1=U(2/c,\beta_0,1)$ from Lemma~\ref{thm:peps_cell_generation_count} (so $2u^{\beta_0}\le cu$ for $u\ge u_1$), and
    $u_{\rm cell}=\max\{u_0,u_1\}$. Using $C(\beta_0,u_{\rm cell})$ and $K(\beta_0,u_{\rm cell})$ from
    Definition~\ref{def:peps_cell_partition_constants}, define
    \begin{align}
        C_{\rm cell}             & =\max\{1,2\log q,\max\{C_a,0\}\}\,
        C(\beta_0,u_{\rm cell}), &
        K_{\rm cell}             & =K(\beta_0,u_{\rm cell}).
        \label{eq:peps_cell_information_constants}
    \end{align}
\end{definition}

\begin{lemma}[Positivity of the cell constants]
    \label{thm:peps_cell_information_constants_pos}
    \lean{TNLean.PEPS.Approximation.cellInformationConstant_pos,
        TNLean.PEPS.Approximation.cellInformationExponent_pos}
    \leanok
    \uses{def:peps_cell_information_constants}
    Always $C_{\rm cell}>0$ and $K_{\rm cell}>0$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_cell_partition_constants_bounds}
    By Lemma~\ref{thm:peps_cell_partition_constants_bounds}, $C(\beta_0,u_{\rm cell})>0$ and $K(\beta_0,u_{\rm cell})>0$, while $\max\{1,2\log q,\max\{C_a,0\}\}\ge1$.
\end{proof}

\begin{lemma}[Cell information with explicit constants]
    \label{thm:peps_cell_conditional_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:area_law_ground_entropy,def:peps_cell_information_constants}
    In the conditional form of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}
    (with the explicit collar hypothesis discussed in
    \cite{gap:polypeps_angular_annular_geometry_scope}), fix
    $q,R\in\mathbb N$, $J,\Delta,D_0,C_{\rm tpl}\in\mathbb R$, and $c>0$.
    Assume the range-$R$ sublinear collar bound at $(C,N)$ (Definition~\ref{def:area_law_collar_bound})
    and the range-$R$ amplification bound at $(\max\{C,1\},1)$ with $(C_a,N_a)$ (Definition~\ref{def:area_law_amplification_bound}),
    and let $C_{\rm cell},K_{\rm cell}$ be given by (\ref{eq:peps_cell_information_constants}).
    For $s\ge1$, $o\in\mathbb R^2$, $Q_i=o+si+[0,s)^2$ ($i\in\mathbb Z^2$),
    $i_0\notin A\subseteq\mathbb Z^2$, a finite integer domain $\Lambda$, a range-$R$ Hamiltonian $H$ with bound $J$, and a unit vector $\Omega$ satisfying (\ref{eq:area_law_projector_gap}),
    set $G=\Lambda\cap Q_{i_0}$ and $B=\Lambda\cap\bigcup_{i\in A}Q_i$.
    If $E\subseteq\Lambda\setminus(G\cup B)$ satisfies $\operatorname{dist}_\infty(\overline{Q_{i_0}},E)\ge cs$, then
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_{\rm cell}(1+\log s)^{K_{\rm cell}}.
        \label{eq:peps_cell_cmi_explicit}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_cell_information_constants,
        thm:peps_coarse_information_of_collar_explicit,
        thm:peps_cell_two_family_partition_explicit,
        def:peps_cell_partition_constants,thm:peps_cell_generation_count,
        lem:peps_coarse_exponent_lt_one,thm:peps_cell_partition_constants_bounds,
        thm:peps_two_family_partition_inverse_image,
        cor:peps_cell_information_counting_form}
    Lemma~\ref{thm:peps_coarse_information_of_collar_explicit} (for at most $66^2$ boxes of diameter $\le u$) bounds each tile by $C_2=\max\{C_a,0\}$ for $u\ge u_0$ and separation $u^{\beta_0}$. The partition of Lemma~\ref{thm:peps_cell_two_family_partition_explicit} at $(\beta_0,u_{\rm cell})$ has all tile scales $u\ge u_{\rm cell}$, so $u\ge u_0$ and $2u^{\beta_0}\le cu$ (giving $u^{\beta_0}<cs\le\operatorname{dist}_\infty(\overline{Q_{i_0}},E)$). Restricting to $\Lambda$ via Lemma~\ref{thm:peps_two_family_partition_inverse_image} and applying Lemma~\ref{cor:peps_cell_information_counting_form} with $N_{\rm t}$ tiles, residual $D$, and $M=\max\{1,2\log q,C_2\}$ yields
    \begin{align}
        I_\Omega(G:E\mid B)
         & \le 2|D|\log q+N_{\rm t}C_2
        \le M(N_{\rm t}+|D|)
        \le MC(\beta_0,u_{\rm cell})(1+\log s)^{K(\beta_0,u_{\rm cell})}.
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Information gained by adjoining selected cells]
    \label{thm:peps_selected_region_information_of_cell_bounds}
    \lean{TNLean.PEPS.Approximation.mutualInformation_selectedRegion_le_of_cell_bound}
    \leanok
    \uses{def:peps_angular_grid,def:area_law_ground_entropy}
    Fix $o\in\mathbb R^2$, $s>0$, $r\in\mathbb R$, $c>0$, and $Q_i=o+si+[0,s)^2$ ($i\in\mathbb Z^2$).
    For $A_{\rm old}\subseteq A\subseteq\mathbb R^2$, let $S$ be the finite set of cells meeting $A^{+r}$ but not $A_{\rm old}^{+r}$, and write $R_{\rm old},R$ for the physical samples of $U_{s,r}(A_{\rm old}),U_{s,r}(A)$ and $G_i$ for that of $Q_i$.
    Let $\Omega$ be a vector on $\Lambda_L$ and $E\subseteq\Lambda_L$ satisfy $\operatorname{dist}_\infty(U_{s,r}(A),E)\ge cs$.
    If $I_\Omega(G_i:E\mid B)\le k$ whenever $i\in S$, $B$ is a union of $G_j$ ($j\ne i$), $E\subseteq(G_i\cup B)^c$, and $\operatorname{dist}_\infty(\overline{Q_i},E)\ge cs$, then $I_\Omega(R_{\rm old}:E)\le b_0$ implies
    \begin{align}
        I_\Omega(R:E) & \le b_0+|S|k.
        \label{eq:peps_selected_region_information}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_selected_cell_owner,thm:peps_selected_cell_closure_clearance}
    Here $R=R_{\rm old}\cup\bigcup_{i\in S}G_i$. By Lemmas~\ref{thm:peps_selected_cell_owner} and~\ref{thm:peps_selected_cell_closure_clearance}, each intermediate conditioning set is a union of cells $G_j$ ($j\ne i$) disjoint from $E$, with $\operatorname{dist}_\infty(\overline{Q_i},E)\ge cs$. Applying the chain rule for mutual information across $i\in S$ adds at most $|S|k$.
\end{proof}

\begin{lemma}[Finite annular information induction]
    \label{thm:peps_annular_information_induction}
    \lean{TNLean.PEPS.Approximation.mutualInformation_le_of_annular_reset}
    \leanok
    \uses{def:peps_annular_layers,def:peps_angular_grid,
        thm:peps_annular_cutoff,def:peps_angular_separation}
    Let $\Omega$ be a unit vector on a finite physical square $\Lambda$,
    $X,E\subseteq\Lambda$, $n>0$, $a,\varepsilon\ge0$,
    $0<a'<a/100$, and $0<d<a'/4$. Suppose $\operatorname{diam}_\infty X\le A_0n$, $|\mathcal V|\le m_0$, and angular separation holds with $(n,t,a)$ (Definition~\ref{def:peps_angular_separation}).
    Let $m\in\mathbb N$ satisfy $t\le s_m<2t$ with $s_j=2^{-j}n$,
    let $A_j$ be the cumulative annular layers (Definition~\ref{def:peps_annular_layers}, with $A_0=X$ when $m=0$), and set
    \begin{align}
        N & =(m_0+1)\left(2\left\lceil
        \frac{\max\{A_0,2\}+3a'/2}{d}\right\rceil_++3\right)^2.
        \label{eq:peps_annular_induction_cell_budget}
    \end{align}
    Using the origin-$0$ grid at each scale, suppose that for every $j\le m$ and $A_{\mathrm{old}}\subseteq A_j$ whose new selected cell set $S$ (meeting $A_j^{+3a's_j/2}$ but not $A_{\mathrm{old}}^{+3a's_j/2}$) satisfies $|S|\le N$,
    $I_\Omega(\Lambda\cap U_{ds_j,3a's_j/2}(A_{\mathrm{old}}):\Lambda\cap E^{+3a's_j/2})\le\varepsilon$ implies
    \begin{align}
        I_\Omega(\Lambda\cap A_j^{+a's_j}:
        \Lambda\cap E^{+a's_j})\leq\varepsilon.
        \label{eq:peps_annular_reset_step}
    \end{align}
    Then $I_\Omega(X:E)\le\varepsilon$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_cumulative_layer_identities,thm:peps_annular_separation,
        thm:peps_dyadic_scale_bounds,thm:peps_annular_cell_count,
        thm:peps_annular_cell_clearance}
    By Lemmas~\ref{thm:peps_annular_separation} and~\ref{thm:peps_annular_cell_count}, the padded regions are disjoint and the new-cell count at each scale $j$ is at most $N$. At $j=0$, $A_{\mathrm{old}}=\varnothing$ gives mutual information $0\le\varepsilon$, so (\ref{eq:peps_annular_reset_step}) holds. For $j>0$, the old selected union and $\Lambda\cap E^{+3a's_j/2}$ lie in the scale-$(j-1)$ padded regions, so monotonicity and (\ref{eq:peps_annular_reset_step}) propagate the bound to $j=m$, where $X\subseteq\Lambda\cap A_m^{+a's_m}$ and $E\subseteq\Lambda\cap E^{+a's_m}$ by Lemma~\ref{thm:peps_cumulative_layer_identities}.
\end{proof}

\begin{definition}[Clearance and threshold of the one-scale reset]
    \label{def:peps_angular_single_scale_constants}
    \lean{TNLean.PEPS.Approximation.angularClearance,
        TNLean.PEPS.Approximation.angularSingleScaleThreshold,
        TNLean.PEPS.Approximation.exp_one_le_angularSingleScaleThreshold}
    \leanok
    \uses{def:peps_reset_width,def:peps_cell_information_constants,
        def:peps_angular_cell_reset_constants}
    For $a,a',d\in\mathbb R$, set $c_{\rm ang}(a,a',d)=(a/2-3a'-d)/d$.
    For $q,N,N_a,M\in\mathbb N$ and $J,\Delta,p,C_{\rm tpl},C_a\in\mathbb R$, let $C_{\rm cell},K_{\rm cell}$ be given by (\ref{eq:peps_cell_information_constants}) at $c=c_{\rm ang}(a,a',d)$, and set
    \begin{align}
        L_{\rm one} & =\max\bigl\{L_{\rm cell}\bigl(a',d,C_p(J,\Delta,p),C_{\rm cell},M,
        K_{\rm cell}\bigr),L_0(q,J,\Delta,p)\bigr\},
        \label{eq:peps_angular_single_scale_threshold}
    \end{align}
    using Definitions~\ref{def:peps_reset_width} and~\ref{def:peps_angular_cell_reset_constants}. Then $L_{\rm one}\ge e$.
\end{definition}

\begin{lemma}[Uniform information reset at one scale, with explicit constants]
    \label{thm:peps_angular_single_scale_reset_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.angular_single_scale_reset_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:polynomial_peps_model,def:peps_angular_grid,
        def:peps_angular_single_scale_constants,def:peps_angular_cell_reset_constants,
        def:peps_reset_width}
    Fix $q\ge1$, $J\ge0$, $\Delta>0$, and $D_0,C_{\rm tpl}$ such that the range-$1$ sublinear collar
    bound holds at $(C,N)$ and the range-$1$ amplification bound at $(\max\{C,1\},1)$ holds at $(C_a,N_a)$.
    For $p>0$, $M\in\mathbb N$, $0<a'<a/100$, $0<d<\min\{a'/4,1\}$,
    $D\ge D_{\rm cell}(C_p(J,\Delta,p),a',d)$, and $L\ge L_{\rm one}$ from (\ref{eq:peps_angular_single_scale_threshold}),
    the conclusion of Lemma~\ref{thm:peps_angular_single_scale_reset_of_collar} (with cell budget $N=M$) holds for every unit ground vector $\Omega$ on $\Lambda_L$ satisfying (\ref{eq:polynomial_peps_projector_gap}), $s\ge D\log L$, grid origin, and $A_{\rm old}\subseteq A,E\subseteq\mathbb R^2$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_scaled_cell_clearance,
        thm:peps_selected_region_information_of_collar_explicit,
        thm:peps_angular_reset_width_of_cell_information_explicit,
        thm:peps_angular_information_reset_explicit,
        def:peps_angular_single_scale_constants,def:peps_cell_information_constants,
        def:peps_reset_width}
    With $c=c_{\rm ang}(a,a',d)>0$, $C_1=C_{\rm cell}$, $K=K_{\rm cell}$, and $C_p=C_p(J,\Delta,p)$, $L\ge L_{\rm one}$ gives $ds\ge1$, (\ref{eq:peps_single_scale_width}) with $N=M$ (by Lemma~\ref{thm:peps_angular_reset_width_of_cell_information_explicit}), and $L\ge L_0(q,J,\Delta,p)$. By Lemma~\ref{thm:peps_scaled_cell_clearance}, $\operatorname{dist}_\infty(U_{ds,3a's/2}(A),C)\ge c\,ds>0$, so $R\cap C=\varnothing$ and (\ref{eq:peps_selected_region_collar_explicit}) gives $I_\Omega(R:C)\le b=L^{-p}+MC_1(1+\log(ds))^K$. Since $C_p(1+b+\log L)<a's/3\le a's/2$, Lemma~\ref{thm:peps_angular_information_reset_explicit} yields $I_\Omega(R_0:C_0)\le L^{-p}$.
\end{proof}

\begin{lemma}[Uniform information reset at one scale]
    \label{thm:peps_angular_single_scale_reset_of_collar}
    \lean{TNLean.PEPS.Approximation.exists_angular_single_scale_reset_of_sublinearCollarBound}
    \leanok
    \uses{def:area_law_collar_bound,def:polynomial_peps_model,def:peps_angular_grid}
    Fix $q\ge1$, $J\ge0$, $\Delta>0$, template constants satisfying the range-$1$ sublinear collar bound (\cite{gap:polypeps_angular_annular_geometry_scope}), $p>0$, $N\in\mathbb N$, $0<a'<a/100$, and $0<d<\min\{a'/4,1\}$.
    There exist $D\ge1$ and $L_0\ge e$ (preceding $L,H,\Omega,s$, origin, and $A_{\rm old},A,E$) such that for $L\ge L_0$, a unit ground vector $\Omega$ on $\Lambda_L$ satisfying (\ref{eq:polynomial_peps_projector_gap}), $s\ge D\log L$, any origin for the side-$ds$ grid, and $A_{\rm old}\subseteq A,E\subseteq\mathbb R^2$ with $\operatorname{dist}_\infty(A,E)\ge as/2$ and $|S|\le N$ (where $S$ is the set of cells meeting $A^{+3a's/2}$ but not $A_{\rm old}^{+3a's/2}$), setting
    \begin{align}
        R_{\rm old} & =\Lambda_L\cap U_{ds,3a's/2}(A_{\rm old}), &
        C           & =\Lambda_L\cap E^{+3a's/2}, \notag           \\
        R_0         & =\Lambda_L\cap A^{+a's},                   &
        C_0         & =\Lambda_L\cap E^{+a's}, \notag
    \end{align}
    the bound $I_\Omega(R_{\rm old}:C)\le L^{-p}$ implies $I_\Omega(R_0:C_0)\le L^{-p}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_angular_single_scale_reset_of_collar_explicit,
        thm:area_law_amplification,def:peps_angular_single_scale_constants,
        def:peps_angular_cell_reset_constants}
    Choose $(C,N_{\rm col})$ for the collar bound and $(C_a,N_a)$ from Theorem~\ref{thm:area_law_amplification} at $(\max\{C,1\},1)$, and apply Lemma~\ref{thm:peps_angular_single_scale_reset_of_collar_explicit} with $D=D_{\rm cell}(C_p(J,\Delta,p),a',d)$ and $L_0=L_{\rm one}$ (with $M=N$), which enforces $ds\ge1$ and
    \begin{align}
        C_p\bigl(1+L^{-p}+NC_1(1+\log(ds))^K+\log L\bigr)
         & <a's/3.
        \label{eq:peps_single_scale_width}
    \end{align}
\end{proof}

\begin{definition}[Slope and threshold of angular information]
    \label{def:peps_angular_information_constants}
    \lean{TNLean.PEPS.Approximation.angularMargin,

        TNLean.PEPS.Approximation.angularCellCount,
        TNLean.PEPS.Approximation.angularScale,

        TNLean.PEPS.Approximation.angularScale_pos,

    \leanok
    \uses{def:peps_reset_width,def:peps_angular_cell_reset_constants,
        def:peps_angular_single_scale_constants}
    For $a,A_0\in\mathbb R$ and $m_0\in\mathbb N$, set
    \begin{align}
        a'(a)                   & =\frac{\min\{a,1\}}{200}, \qquad d(a)=\frac{a'(a)}8, \notag \\
        N_{\rm cell}(a,A_0,m_0) & =(m_0+1)\left(2\left\lceil
        \frac{\max\{A_0,2\}+3a'(a)/2}{d(a)}\right\rceil_++3\right)^2,
        \notag
    \end{align}
    and define the slope and threshold of \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS} via (\ref{eq:peps_angular_cell_reset_constants}) and (\ref{eq:peps_angular_single_scale_threshold}) by
    \begin{align}
        D_{\rm ang}(J,\Delta,a,p) & =D_{\rm cell}\bigl(C_p(J,\Delta,p),a'(a),d(a)\bigr),
        \notag                                                                                         \\
        L_{\rm ang}               & =L_{\rm one}\big|_{a'=a'(a),\,d=d(a),\,M=N_{\rm cell}(a,A_0,m_0)}.
        \label{eq:peps_angular_information_constants}
    \end{align}
    Then $D_{\rm ang}\ge1$ and $L_{\rm ang}\ge e$; if $J\ge0$ and $\Delta,a,p>0$, then $C_p(J,\Delta,p)\ge52$ gives
    $D_{\rm ang}(J,\Delta,a,p)=2400\,C_p(J,\Delta,p)/\min\{a,1\}$.
\end{definition}
