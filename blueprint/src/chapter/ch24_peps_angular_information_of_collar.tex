\begin{definition}[Constants of the cell information estimate]
    \label{def:peps_cell_information_constants}
    \lean{TNLean.PEPS.Approximation.cellInformationScale,
        TNLean.PEPS.Approximation.cellInformationConstant,
        TNLean.PEPS.Approximation.cellInformationExponent}
    \leanok
    \uses{def:peps_coarse_information_constants,def:peps_cell_partition_constants,
        thm:peps_cell_generation_count}
    Fix $q\in\mathbb N$, $c>0$, a template constant $C_{\rm tpl}$, a collar
    threshold $N$, an amplification constant $C_a$ and an amplification
    threshold $N_a$, and recall $\beta_0<1$. Let
    $u_0=u_0(C_{\rm tpl},66^2,1,N,C_a,N_a)$ be the coarse threshold of
    Definition~\ref{def:peps_coarse_information_constants} and
    $u_1=U(2/c,\beta_0,1)$ the power threshold of
    Theorem~\ref{thm:peps_cell_generation_count}, so that
    $2u^{\beta_0}\le cu$ for $u\ge u_1$. The lower tile scale is
    $u_{\rm cell}=\max\{u_0,u_1\}$. With the partition constants of
    Definition~\ref{def:peps_cell_partition_constants}, the cell prefactor and
    exponent are
    \begin{align}
        C_{\rm cell}             & =\max\{1,2\log q,\max\{C_a,0\}\}\,
        C(\beta_0,u_{\rm cell}), &
        K_{\rm cell}             & =K(\beta_0,u_{\rm cell}).
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Positivity of the cell constants]
    \label{thm:peps_cell_information_constants_pos}
    \lean{TNLean.PEPS.Approximation.cellInformationConstant_pos,
        TNLean.PEPS.Approximation.cellInformationExponent_pos}
    \leanok
    \uses{def:peps_cell_information_constants}
    For all values of the parameters, the cell
    prefactor $C_{\rm cell}$ and the cell exponent $K_{\rm cell}$ are positive.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_cell_partition_constants_bounds}
    The partition constants are positive by
    Theorem~\ref{thm:peps_cell_partition_constants_bounds}, and
    $\max\{1,2\log q,\max\{C_a,0\}\}\ge1$.
\end{proof}

\begin{theorem}[Cell information with explicit constants]
    \label{thm:peps_cell_conditional_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:area_law_ground_entropy,def:peps_cell_information_constants}
    In the conditional form of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}
    (with the explicit collar hypothesis discussed in
    \cite{gap:polypeps_angular_annular_geometry_scope}), fix
    $q\in\mathbb N$, $J,\Delta\in\mathbb R$ and template constants
    $D_0,C_{\rm tpl}$. Suppose that the range-one sublinear collar bound holds
    with constant $C$ and threshold $N$, and that the range-one amplification
    bound holds at $(\max\{C,1\},1)$ with constant $C_a$ and threshold $N_a$.
    Fix $c>0$, and let $C_{\rm cell},K_{\rm cell}$ be the constants of
    Definition~\ref{def:peps_cell_information_constants}.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, and set
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Choose an index $i_0$ and any set of indices $A$ with $i_0\notin A$.
    On any finite integer domain $\Lambda$, let $H$ be a range-one Hamiltonian
    with interaction bound $J$, and let $\Omega$ be a unit ground vector with
    full-system gap $\Delta$.
    Put $G=\Lambda\cap Q_{i_0}$ and
    $B=\Lambda\cap\bigcup_{i\in A}Q_i$.
    If $E\subseteq\Lambda\setminus(G\cup B)$ and
    $\operatorname{dist}_\infty(x,y)\ge cs$ for every
    $x\in\overline{Q_{i_0}}$ and $y\in E$, then
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_{\rm cell}(1+\log s)^{K_{\rm cell}}.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{def:peps_cell_information_constants,
        thm:peps_coarse_information_of_collar_explicit,
        thm:peps_cell_two_family_partition_explicit,
        def:peps_cell_partition_constants,thm:peps_cell_generation_count,
        thm:peps_two_family_partition_inverse_image,
        cor:peps_cell_information_counting_form}
    The coarse information estimate with explicit constants, for unions of at
    most $66^2$ boxes of diameter at most the scale, gives the tile bound
    $C_2=\max\{C_a,0\}$ at every scale $u\ge u_0$ and separation
    $u^{\beta_0}$. The partition with explicit constants at the exponent
    $\beta_0$ and the lower bound $u_{\rm cell}$ has all tile scales at least
    $u_{\rm cell}$, so both $u\ge u_0$ and $2u^{\beta_0}\le cu$ hold at every
    tile scale. Giving the cells indexed by $A$ colour zero and the other
    exterior cells colour one, and restricting the ordered partition to
    $\Lambda$, retains its order and families without increasing its residual
    cardinality. The separation of tiles of one family from their predecessors
    and from the corresponding exterior cells gives the two coarse
    information bounds. For the first family, the additional separation from
    $E$ follows from $u^{\beta_0}<cs$ and the closed-cell hypothesis.
    If $N_{\rm t}$ is the number of tiles and $D$ the residual, the counting
    form of the two-family inequality gives
    \begin{align}
        I_\Omega(G:E\mid B)
         & \le 2|D|\log q+N_{\rm t}C_2
        \le M(N_{\rm t}+|D|)
        \le MC(\beta_0,u_{\rm cell})(1+\log s)^{K(\beta_0,u_{\rm cell})},
        \notag
    \end{align}
    where $M=\max\{1,2\log q,C_2\}$, which is the stated bound.
\end{proof}

\begin{theorem}[Cell information under the sublinear collar bound]
    \label{thm:peps_cell_conditional_information_of_collar}
    \lean{TNLean.PEPS.Approximation.exists_cell_conditionalMutualInformation_le_of_sublinearCollarBound}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_ground_entropy}
    In the conditional form of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}
    (with the explicit collar hypothesis discussed in
    \cite{gap:polypeps_angular_annular_geometry_scope}), fix
    $q\in\mathbb N$, $J\in\mathbb R$, $\Delta>0$, and template constants
    for which the range-one sublinear collar bound holds, and fix $c>0$.
    There exist $C_1>0$ and $K\ge0$ with the following property.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, and set
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Choose an index $i_0$ and any set of indices $A$ with $i_0\notin A$.
    On any finite integer domain $\Lambda$, let $H$ be a range-one Hamiltonian
    with interaction bound $J$, and let $\Omega$ be a unit ground vector with
    full-system gap $\Delta$.
    Put $G=\Lambda\cap Q_{i_0}$ and
    $B=\Lambda\cap\bigcup_{i\in A}Q_i$.
    If $E\subseteq\Lambda\setminus(G\cup B)$ and
    $\operatorname{dist}_\infty(x,y)\ge cs$ for every
    $x\in\overline{Q_{i_0}}$ and $y\in E$, then
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_1(1+\log s)^K.
        \notag
    \end{align}
    The constants precede the scale, grid, finite domain, Hamiltonian and
    ground vector.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_cell_conditional_information_of_collar_explicit,
        thm:area_law_amplification,def:peps_cell_information_constants}
    Choose a constant $C$ and a threshold $N$ for the collar bound, and by
    Theorem~\ref{thm:area_law_amplification} a constant $C_a$ and a threshold
    $N_a$ for the range-one amplification bound at $(\max\{C,1\},1)$.
    Theorem~\ref{thm:peps_cell_conditional_information_of_collar_explicit}
    gives the claim with $C_1=C_{\rm cell}>0$ and $K=K_{\rm cell}>0$.
\end{proof}

\begin{theorem}[Square-grid cell information with explicit constants]
    \label{thm:peps_square_cell_conditional_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_square_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:polynomial_peps_model,def:peps_cell_information_constants}
    Fix $q\in\mathbb N$, $J\ge0$, $\Delta\in\mathbb R$ and template constants
    $D_0,C_{\rm tpl}$. Suppose that the range-one sublinear collar bound holds
    with constant $C$ and threshold $N$, and that the range-one amplification
    bound holds at $(\max\{C,1\},1)$ with constant $C_a$ and threshold $N_a$.
    Fix $c>0$. The conclusion of
    Theorem~\ref{thm:peps_cell_conditional_information_of_collar_explicit}
    holds with the same constants $C_{\rm cell},K_{\rm cell}$ of
    Definition~\ref{def:peps_cell_information_constants} for the original open
    square $\Lambda_L=\{1,\ldots,L\}^2$, every Hamiltonian of the square-grid
    model with interaction bound $J$, and every unit ground vector with
    full-system gap $\Delta$. The regions $G,B,E$ are the same physical cell,
    conditioning union and exterior region, and the separation hypothesis
    concerns the full closed ambient cell.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_cell_conditional_information_of_collar_explicit,
        thm:finite_product_entropy_relabel,thm:square_area_law_ground_gap}
    Relabeling the square sites by the integer rectangle
    $\{0,\ldots,L-1\}^2$ and replacing the grid origin $o$ by $o-(1,1)$
    sends the chosen half-open cell and its closure
    to the original cell and closure under translation by $(1,1)$, while
    preserving distances.
    The relabeling preserves the Hamiltonian, ground-state equation and
    full-system gap, and carries both cell samples and the conditioning
    union to the original physical regions.
    Theorem~\ref{thm:peps_cell_conditional_information_of_collar_explicit} then
    applies, and because relabeling preserves each of the four regional
    entropies in $I_\Omega(G:E\mid B)$, the stated bound holds for the same
    vector $\Omega$.
\end{proof}

\begin{theorem}[Information gained by adjoining selected cells]
    \label{thm:peps_selected_region_information_of_cell_bounds}
    \lean{TNLean.PEPS.Approximation.mutualInformation_selectedRegion_le_of_cell_bound}
    \leanok
    \uses{def:peps_angular_grid,def:area_law_ground_entropy}
    For the finite cell addition in the proof of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, fix a grid origin $o$,
    a side length $s>0$, an enlargement radius $r$, and $c>0$.
    Write $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Let $A_{\rm old}\subseteq A$ be ambient sets and let $S$ be the
    finite set of cells selected by $A$ but not by $A_{\rm old}$.
    Write $R_{\rm old},R$ for the physical samples of their selected unions,
    and $G_i$ for the physical sample of cell $i$.
    Let $\Omega$ be a vector on the square lattice and let $E$ be a physical
    region at distance at least $cs$ from every point of the selected union
    for $A$.

    Suppose that, for every $i\in S$ and every union $B$ of physical samples
    of cells other than $i$, the conditions $E\subseteq(G_i\cup B)^c$ and
    $\operatorname{dist}_\infty(\overline{Q_i},E)\ge cs$ imply
    $I_\Omega(G_i:E\mid B)\le k$. If $I_\Omega(R_{\rm old}:E)\le b_0$, then
    \begin{align}
        I_\Omega(R:E) & \le b_0+|S|k.
        \notag
    \end{align}
    No normalization or sign assumption on $k$ is required.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_selected_cell_owner,thm:peps_selected_cell_closure_clearance}
    The selected region is exactly
    $R_{\rm old}\cup\bigcup_{i\in S}G_i$.
    The unique cell index of each physical site identifies every intermediate
    conditioning region with a union of whole cells.
    Positive clearance makes $E$ disjoint from these regions, and continuity
    of distance extends the clearance to each closed cell.
    The assumed conditional bound therefore applies at every addition.
    The entropy chain identity bounds the total increase by
    $\sum_{i\in S}k=|S|k$.
\end{proof}

\begin{theorem}[Information reset for padded annular regions]
    \label{thm:peps_angular_information_reset}
    \lean{TNLean.PEPS.Approximation.exists_angular_information_reset}
    \leanok
    \uses{def:peps_angular_grid,def:area_law_ground_entropy}
    For the regions used in \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS},
    fix a local dimension $q\ge1$, an interaction bound $J\ge0$, a gap
    $\Delta>0$, and $p>0$. There are constants $C_p>0$ and $L_0$ with the
    following property. Let $L\ge L_0$ and let $\Omega$ be a normalized
    ground vector of the square Hamiltonian with full-system gap $\Delta$.
    Choose ambient sets $A,E$, a grid origin, and parameters $a'\ge0$,
    $d>0$, and $s>0$.
    Let $R$ be the physical sample of all cells of side $ds$ meeting
    $A^{+3a's/2}$, and let $C=\Lambda_L\cap E^{+3a's/2}$.
    Put $R_0=\Lambda_L\cap A^{+a's}$ and $C_0=\Lambda_L\cap E^{+a's}$.
    Suppose $R\cap C=\varnothing$, $b\ge0$, and
    \begin{align}
        I_\Omega(R:C)   & \le b,\notag \\
        C_p(1+b+\log L) & \le a's/2.
        \notag
    \end{align}
    Then $I_\Omega(R_0:C_0)\le L^{-p}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_annular_cell_clearance,thm:peps_physical_integer_sup_distance,
        thm:peps_information_reset}
    Each core lies in its corresponding larger region and has distance
    strictly greater than $a's/2$ from the corresponding physical complement.
    The common translation of physical lattice coordinates preserves
    the sup distance. The information reset theorem applies to the
    original Hamiltonian and ground vector with these two buffers.
\end{proof}

\begin{theorem}[Uniform reset width for bounded cell costs]
    \label{thm:peps_angular_reset_width_of_cell_information}
    \lean{TNLean.PEPS.Approximation.exists_angular_reset_width_lt_of_cell_information}
    \leanok
    For the uniform choice of scales in
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, fix $a',C_p,C_1,p>0$,
    $0<d\le1$, and real numbers $N,K\ge0$.
    There are $D\ge1$ and $L_0\ge e$ such that, for all real $L,s$ with
    $L\ge L_0$ and $s\ge D\log L$, one has $ds\ge1$ and
    \begin{align}
        C_p\left(1+L^{-p}+NC_1(1+\log(ds))^K+\log L\right)
         & <a's/3.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_annular_reset_width}
    With $D=\max\{1,4C_p/(a'/3),1/d\}$, the fixed-slope width
    estimate applies with coefficients $C_3=C_p(2+NC_1)$ and $C_4=C_p$,
    where the resulting threshold is increased to at least $e$.
    Then $\log L\ge1$ implies $ds\ge1$, while $L^{-p}\le1$ and
    $(1+\log(ds))^K\le(1+\log s)^K$.
    Since $(1+\log s)^K\ge1$, the required width is at most
    $C_3(1+\log s)^K+C_4\log L$, which is strictly less than $a's/3$.
\end{proof}

\begin{theorem}[Selected-cell information under the collar bound]
    \label{thm:peps_selected_region_information_of_collar}
    \lean{TNLean.PEPS.Approximation.exists_mutualInformation_selectedRegion_le_square_of_sublinearCollarBound}
    \leanok
    \uses{def:area_law_collar_bound,def:polynomial_peps_model,def:peps_angular_grid}
    In the conditional step of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS} (with the collar
    hypothesis remaining explicit, as recorded in
    \cite{gap:polypeps_angular_annular_geometry_scope}), fix $q\in\mathbb N$,
    $J\ge0$, $\Delta>0$ and fixed template constants
    for which the range-one sublinear collar bound holds, and fix $c>0$.
    There exist $C_1>0$ and $K\ge0$ with the following property.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, let $r\in\mathbb R$,
    and let $A_{\rm old}\subseteq A$ be ambient sets.
    Suppose that the set $S$ of cells selected by $A$ but not by
    $A_{\rm old}$ is finite.
    On the original open square $\Lambda_L=\{1,\ldots,L\}^2$, let $H$ have
    interaction bound $J$ and let $\Omega$ be a unit ground vector with
    full-system gap $\Delta$.
    Write $R_{\rm old}$ and $R$ for the physical samples of the two selected
    unions. Let $E\subseteq\Lambda_L$ have distance at least $cs$ from every
    point of the ambient selected union for $A$.
    If $I_\Omega(R_{\rm old}:E)\le b_0$, then
    \begin{align}
        I_\Omega(R:E) & \le b_0+|S|C_1(1+\log s)^K.
        \notag
    \end{align}
    The constants precede the grid, ambient sets, square size, Hamiltonian,
    ground vector and exterior region.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_square_cell_conditional_information_of_collar,
        thm:peps_selected_region_information_of_cell_bounds}
    Let $C_1,K$ be the constants from the square-grid cell information theorem.
    Its bound applies to every new cell conditioned on any union of the other
    whole cells, with the same vector $\Omega$ and exterior $E$.
    The selected-cell summation theorem therefore gives
    $I_\Omega(R:E)\le b_0+\sum_{i\in S}C_1(1+\log s)^K$,
    which is the stated estimate.
\end{proof}

\begin{theorem}[Finite annular information induction]
    \label{thm:peps_annular_information_induction}
    \lean{TNLean.PEPS.Approximation.mutualInformation_le_of_annular_reset}
    \leanok
    \uses{def:peps_annular_layers,def:peps_angular_grid,
        thm:peps_annular_cutoff,def:peps_angular_separation}
    Let $\Omega$ be a unit vector on a finite physical square $\Lambda$,
    and let $X,E\subseteq\Lambda$. Identify the sites with their integer
    positions in the plane, and take all enlargements in the ambient
    $\ell_\infty$ metric before intersecting with $\Lambda$.
    Let $n>0$, $a\geq0$, $\varepsilon\geq0$,
    $0<a'<a/100$, and $0<d<a'/4$. Suppose that $X$ has diameter at most
    $A_0n$, that the marked set has at most $m_0$ points, and that
    angular separation holds with parameters $n,t,a$.
    Let $m\in\mathbb N$ satisfy $t\leq s_m<2t$, where $s_j=2^{-j}n$,
    and let $A_j$ be the cumulative annular layers. When $m=0$, the sole
    layer and its cumulative set are both $X$. Put
    \begin{align}
        N & =(m_0+1)\left(2\left\lceil
        \frac{\max\{A_0,2\}+3a'/2}{d}\right\rceil_{\mathbb N}+3\right)^2.
        \notag
    \end{align}
    Use the grid with origin zero at each scale. Assume the following
    reset implication for every $j\leq m$ and every $A_{\mathrm{old}}\subseteq A_j$.
    Let $S$ be exactly the cells touching $A_j^{+3a's_j/2}$ that do not
    touch $A_{\mathrm{old}}^{+3a's_j/2}$, and suppose $|S|\leq N$.
    If the mutual information between the physical sample of the old
    selected cells and $\Lambda\cap E^{+3a's_j/2}$ is at most
    $\varepsilon$, then
    \begin{align}
        I_\Omega(\Lambda\cap A_j^{+a's_j}:
        \Lambda\cap E^{+a's_j})\leq\varepsilon.
        \notag
    \end{align}
    Then $I_\Omega(X:E)\leq\varepsilon$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_cumulative_layer_identities,thm:peps_annular_separation,
        thm:peps_dyadic_scale_bounds,thm:peps_annular_cell_count,
        thm:peps_annular_cell_clearance}
    Angular separation makes the two padded regions disjoint at every
    scale. At the first scale the old group is empty, whose entropy is
    zero because the state is normalized. The initial selected cells
    satisfy the stated bound $N$, so the reset implication applies.
    At each later scale, the old selected group and the enlarged
    exterior lie inside the two preceding padded regions. Monotonicity
    of mutual information therefore bounds their information by
    $\varepsilon$. The exact new-cell count is again at most $N$, and
    the reset implication proves the invariant at the new scale.
    In the finite induction of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, $A_m=X$, and removing
    the padding from both regions gives the desired estimate.
\end{proof}

\begin{theorem}[Uniform information reset at one scale]
    \label{thm:peps_angular_single_scale_reset_of_collar}
    \lean{TNLean.PEPS.Approximation.exists_angular_single_scale_reset_of_sublinearCollarBound}
    \leanok
    \uses{def:area_law_collar_bound,def:polynomial_peps_model,def:peps_angular_grid}
    Fix $q\in\mathbb N$ with $q\ge1$, $J\ge0$, $\Delta>0$, and template constants for which the
    range-one sublinear collar bound holds. Fix $p>0$, $N\in\mathbb N$ and
    real parameters satisfying $0<a'<a/100$, $0<d<a'/4$, and $d\le1$.
    There exist $D\ge1$ and $L_0\ge e$ with the following property.
    Let $L\in\mathbb N$ satisfy $L\ge L_0$, and let $\Omega$ be a unit
    ground vector of an open-square Hamiltonian on
    $\Lambda_L=\{1,\ldots,L\}^2$, with interaction bound $J$ and
    full-system gap $\Delta$.
    Let $s\ge D\log L$, choose any origin for the square grid of side $ds$,
    and let $A_{\rm old}\subseteq A$ and $E$ be ambient subsets of
    $\mathbb R^2$. Suppose that
    $\operatorname{dist}_{\infty}(x,y)\ge as/2$ for every $x\in A$ and
    $y\in E$. Let $S$ be the set of cells meeting $A^{+3a's/2}$ but
    not meeting $A_{\rm old}^{+3a's/2}$, and suppose that $|S|\le N$.
    Define the physical regions
    \begin{align}
        R_{\rm old} & =\Lambda_L\cap U_{ds,3a's/2}(A_{\rm old}), &
        C           & =\Lambda_L\cap E^{+3a's/2}, \notag           \\
        R_0         & =\Lambda_L\cap A^{+a's},                   &
        C_0         & =\Lambda_L\cap E^{+a's}. \notag
    \end{align}
    If $I_\Omega(R_{\rm old}:C)\le L^{-p}$, then
    $I_\Omega(R_0:C_0)\le L^{-p}$.
    All constants precede the square size, Hamiltonian, ground vector, scale,
    grid origin and ambient regions. The sublinear collar hypothesis remains
    explicit in this conditional step of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}; see
    \cite{gap:polypeps_angular_annular_geometry_scope}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_scaled_cell_clearance,
        thm:peps_selected_region_information_of_collar,
        thm:peps_angular_reset_width_of_cell_information,
        thm:peps_angular_information_reset}
    With $c=(a/2-3a'-d)/d>0$, let $C_1,K$ be the cell-information constants
    for this clearance coefficient and $C_p$ the information-reset
    constant. Choosing $D,L_0$ so that $ds\ge1$ and
    \begin{align}
        C_p\bigl(1+L^{-p}+NC_1(1+\log(ds))^K+\log L\bigr)
         & <a's/3
        \label{eq:peps_single_scale_width}
    \end{align}
    whenever $L\ge L_0$ and $s\ge D\log L$ (increasing $L_0$ to the
    reset threshold if necessary) ensures that
    the selected union $U_{ds,3a's/2}(A)$ has distance at least
    $(a/2-3a'-d)s=c\,ds>0$ from the exterior enlargement.
    Thus its physical sample $R$ is disjoint from $C$. The selected-cell
    estimate and $|S|\le N$ give
    $I_\Omega(R:C)\le b$, where
    $b=L^{-p}+NC_1(1+\log(ds))^K\ge0$.
    By (\ref{eq:peps_single_scale_width}),
    $C_p(1+b+\log L)<a's/3\le a's/2$.
    The information reset for the padded regions therefore yields
    $I_\Omega(R_0:C_0)\le L^{-p}$.
\end{proof}

\begin{theorem}[Angular information under the sublinear collar bound]
    \label{thm:peps_angular_information_of_collar}
    \lean{TNLean.PEPS.Approximation.angularInformationBound_of_sublinearCollarBound}
    \leanok
    \uses{def:area_law_collar_bound,def:polynomial_peps_model}
    In the conditional form of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS} (with the collar
    premise remaining separate from the model and gap assumptions, as
    recorded in \cite{gap:polypeps_angular_annular_geometry_scope}), fix
    $q\in\mathbb N$ with $q\ge1$, an interaction bound $J\ge0$, and
    a full-system gap $\Delta>0$. Assume the range-one sublinear collar
    bound for these parameters and fixed template constants.
    For every $a>0$, $A_0\ge1$, $m_0\in\mathbb N$ and $p>0$, there are
    constants $D,L_0\in\mathbb R$ with the following property.
    Let $L\ge L_0$ be an integer and let $H$ be a Hamiltonian of the
    square-grid model on $\Lambda_L=\{1,\ldots,L\}^2$, with interaction
    bound $J$. Let $\Omega$ be a unit ground vector of $H$ with
    full-system gap $\Delta$.
    Choose at most $m_0$ marked points $\mathcal V\subset\mathbb R^2$,
    disjoint physical regions $X,E\subseteq\Lambda_L$, and a real scale
    $n$ satisfying $D\log L\le n\le A_0L$ and
    $\operatorname{diam}_\infty X\le A_0n$.
    Put $d_{\mathcal V}(x)=\min_{v\in\mathcal V}|x-v|_\infty$, with
    $d_{\varnothing}=+\infty$. If every $x\in X$ and $y\in E$ satisfy
    \begin{align}
        |x-y|_\infty
         & \ge a\max\{D\log L,\min(n,d_{\mathcal V}(x))\},
        \notag
    \end{align}
    then $I_\Omega(X:E)\le L^{-p}$.
    The constants precede the Hamiltonian, ground vector, marked points,
    physical regions and scale.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_angular_single_scale_reset_of_collar,
        thm:peps_annular_information_induction,thm:peps_annular_cutoff,
        thm:peps_annular_separation}
    Set $a'=\min\{a,1\}/200$ and $d=a'/8$, and put
    \begin{align}
        N & =(m_0+1)
        \left(2\left\lceil\frac{\max\{A_0,2\}+3a'/2}{d}\right\rceil_{+}
        +3\right)^2.
        \notag
    \end{align}
    The one-scale reset estimate chooses $D\ge1$ and $L_0\ge e$
    for these fixed geometric data, this cell-count bound and the accuracy
    exponent $p$. In particular, $t=D\log L>0$ for $L\ge L_0$.
    The inequality $t\le n$ yields a dyadic cutoff with
    $s_j=2^{-j}n$ and $t\le s_m<2t$. Every processed scale $s_j$, with
    $j\le m$, satisfies $s_j\ge t$, so the same reset constants apply
    at each scale. Process the layers from farthest to nearest the
    marked points, and write $A_j$ for the cumulative processed set.

    At scale $s_j$, the old selected cell group and the enlarged exterior
    lie inside the two cores controlled at the preceding scale.
    The first old group is empty. The number of new cells is at most $N$,
    and cumulative angular separation gives
    $\operatorname{dist}_\infty(A_j,E)\ge as_j/2$.
    The one-scale reset estimate therefore restores
    \begin{align}
        I_\Omega\bigl(\Lambda_L\cap A_j^{+a's_j}:
        \Lambda_L\cap E^{+a's_j}\bigr) & \le L^{-p}.
        \notag
    \end{align}
    Here $F^{+r}$ denotes the ambient $\ell_\infty$ enlargement of $F$.
    The same bound is restored at each step; it is not summed over the
    scales. The last cumulative set is $A_m=X$.
    Monotonicity of mutual information on both regions removes the final
    padding and gives the result. The empty marked set and the case of one
    layer are included in the same induction.
\end{proof}
