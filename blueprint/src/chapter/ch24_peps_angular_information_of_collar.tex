\begin{definition}[Constants of the cell information estimate]
    \label{def:peps_cell_information_constants}
    \lean{TNLean.PEPS.Approximation.cellInformationScale,
        TNLean.PEPS.Approximation.cellInformationConstant,
        TNLean.PEPS.Approximation.cellInformationExponent}
    \leanok
    \uses{def:peps_coarse_information_constants,def:peps_cell_partition_constants,
        thm:peps_cell_generation_count}
    Fix $q\in\mathbb N$, $c>0$, a template constant $C_{\rm tpl}$, a collar
    threshold $N$, an amplification constant $C_a$ and an amplification
    threshold $N_a$, and recall $\beta_0<1$. Let
    $u_0=u_0(C_{\rm tpl},66^2,1,N,C_a,N_a)$ be the coarse threshold of
    Definition~\ref{def:peps_coarse_information_constants} and
    $u_1=U(2/c,\beta_0,1)$ the power threshold of
    Theorem~\ref{thm:peps_cell_generation_count}, so that
    $2u^{\beta_0}\le cu$ for $u\ge u_1$. The lower tile scale is
    $u_{\rm cell}=\max\{u_0,u_1\}$. With the partition constants of
    Definition~\ref{def:peps_cell_partition_constants}, the cell prefactor and
    exponent are
    \begin{align}
        C_{\rm cell}             & =\max\{1,2\log q,\max\{C_a,0\}\}\,
        C(\beta_0,u_{\rm cell}), &
        K_{\rm cell}             & =K(\beta_0,u_{\rm cell}).
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Positivity of the cell constants]
    \label{thm:peps_cell_information_constants_pos}
    \lean{TNLean.PEPS.Approximation.cellInformationConstant_pos,
        TNLean.PEPS.Approximation.cellInformationExponent_pos}
    \leanok
    \uses{def:peps_cell_information_constants}
    For all values of the parameters, the cell
    prefactor $C_{\rm cell}$ and the cell exponent $K_{\rm cell}$ are positive.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_cell_partition_constants_bounds}
    The partition constants are positive by
    Theorem~\ref{thm:peps_cell_partition_constants_bounds}, and
    $\max\{1,2\log q,\max\{C_a,0\}\}\ge1$.
\end{proof}

\begin{theorem}[Cell information with explicit constants]
    \label{thm:peps_cell_conditional_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.cell_conditionalMutualInformation_le_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:area_law_ground_entropy,def:peps_cell_information_constants}
    In the conditional form of \cite[Lemma~3.2]{OpenAI2026PolynomialPEPS}
    (with the explicit collar hypothesis discussed in
    \cite{gap:polypeps_angular_annular_geometry_scope}), fix
    $q,R\in\mathbb N$, $J,\Delta\in\mathbb R$ and template constants
    $D_0,C_{\rm tpl}$. Suppose that the sublinear collar bound at range $R$ holds
    with constant $C$ and threshold $N$, and that the amplification bound at
    range $R$ holds at $(\max\{C,1\},1)$ with constant $C_a$ and threshold $N_a$.
    The source has only site and nearest-neighbour terms, which is range one;
    ranges $R>1$ extend it.
    Fix $c>0$, and let $C_{\rm cell},K_{\rm cell}$ be the constants of
    Definition~\ref{def:peps_cell_information_constants}.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, and set
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Choose an index $i_0$ and any set of indices $A$ with $i_0\notin A$.
    On any finite integer domain $\Lambda$, let $H$ be a Hamiltonian of range $R$
    with interaction bound $J$, and let $\Omega$ be a unit ground vector with
    full-system gap $\Delta$.
    Put $G=\Lambda\cap Q_{i_0}$ and
    $B=\Lambda\cap\bigcup_{i\in A}Q_i$.
    If $E\subseteq\Lambda\setminus(G\cup B)$ and
    $\operatorname{dist}_\infty(x,y)\ge cs$ for every
    $x\in\overline{Q_{i_0}}$ and $y\in E$, then
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_{\rm cell}(1+\log s)^{K_{\rm cell}}.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{def:peps_cell_information_constants,
        thm:peps_coarse_information_of_collar_explicit,
        thm:peps_cell_two_family_partition_explicit,
        def:peps_cell_partition_constants,thm:peps_cell_generation_count,
        lem:peps_coarse_exponent_lt_one,thm:peps_cell_partition_constants_bounds,
        thm:peps_two_family_partition_inverse_image,
        cor:peps_cell_information_counting_form}
    The coarse information estimate with explicit constants, for unions of at
    most $66^2$ boxes of diameter at most the scale, gives the tile bound
    $C_2=\max\{C_a,0\}$ at every scale $u\ge u_0$ and separation
    $u^{\beta_0}$. The partition with explicit constants at the exponent
    $\beta_0$ and the lower bound $u_{\rm cell}$ has all tile scales at least
    $u_{\rm cell}$, so both $u\ge u_0$ and $2u^{\beta_0}\le cu$ hold at every
    tile scale. Giving the cells indexed by $A$ colour zero and the other
    exterior cells colour one, and restricting the ordered partition to
    $\Lambda$, retains its order and families without increasing its residual
    cardinality. The separation of tiles of one family from their predecessors
    and from the corresponding exterior cells gives the two coarse
    information bounds. For the first family, the additional separation from
    $E$ follows from $u^{\beta_0}<cs$ and the closed-cell hypothesis.
    If $N_{\rm t}$ is the number of tiles and $D$ the residual, the counting
    form of the two-family inequality gives
    \begin{align}
        I_\Omega(G:E\mid B)
         & \le 2|D|\log q+N_{\rm t}C_2
        \le M(N_{\rm t}+|D|)
        \le MC(\beta_0,u_{\rm cell})(1+\log s)^{K(\beta_0,u_{\rm cell})},
        \notag
    \end{align}
    where $M=\max\{1,2\log q,C_2\}$, which is the stated bound.
\end{proof}

    $q,R\in\mathbb N$, $J\in\mathbb R$, $\Delta>0$, and template constants

    The source has only site and nearest-neighbour terms, which is range one;
    ranges $R>1$ extend it.
    There exist $C_1>0$ and $K\ge0$ with the following property.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, and set
    $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Choose an index $i_0$ and any set of indices $A$ with $i_0\notin A$.
    On any finite integer domain $\Lambda$, let $H$ be a Hamiltonian of range $R$

    $B=\Lambda\cap\bigcup_{i\in A}Q_i$.

    $\operatorname{dist}_\infty(x,y)\ge cs$ for every
    $x\in\overline{Q_{i_0}}$ and $y\in E$, then
    \begin{align}
        I_\Omega(G:E\mid B) & \le C_1(1+\log s)^K.
        \notag
    \end{align}

\end{theorem}
\begin{proof}\leanok

\end{proof}

    $\{0,\ldots,L-1\}^2$ and replacing the grid origin $o$ by $o-(1,1)$

    full-system gap, and carries both cell samples and the conditioning

\begin{theorem}[Information gained by adjoining selected cells]
    \label{thm:peps_selected_region_information_of_cell_bounds}
    \lean{TNLean.PEPS.Approximation.mutualInformation_selectedRegion_le_of_cell_bound}
    \leanok
    \uses{def:peps_angular_grid,def:area_law_ground_entropy}
    For the finite cell addition in the proof of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, fix a grid origin $o$,
    a side length $s>0$, an enlargement radius $r$, and $c>0$.
    Write $Q_i=o+si+[0,s)^2$ for $i\in\mathbb Z^2$.
    Let $A_{\rm old}\subseteq A$ be ambient sets and let $S$ be the
    finite set of cells selected by $A$ but not by $A_{\rm old}$.
    Write $R_{\rm old},R$ for the physical samples of their selected unions,
    and $G_i$ for the physical sample of cell $i$.
    Let $\Omega$ be a vector on the square lattice and let $E$ be a physical
    region at distance at least $cs$ from every point of the selected union
    for $A$.

    Suppose that, for every $i\in S$ and every union $B$ of physical samples
    of cells other than $i$, the conditions $E\subseteq(G_i\cup B)^c$ and
    $\operatorname{dist}_\infty(\overline{Q_i},E)\ge cs$ imply
    $I_\Omega(G_i:E\mid B)\le k$. If $I_\Omega(R_{\rm old}:E)\le b_0$, then
    \begin{align}
        I_\Omega(R:E) & \le b_0+|S|k.
        \notag
    \end{align}
    No normalization or sign assumption on $k$ is required.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_selected_cell_owner,thm:peps_selected_cell_closure_clearance}
    The selected region is exactly
    $R_{\rm old}\cup\bigcup_{i\in S}G_i$.
    The unique cell index of each physical site identifies every intermediate
    conditioning region with a union of whole cells.
    Positive clearance makes $E$ disjoint from these regions, and continuity
    of distance extends the clearance to each closed cell.
    The assumed conditional bound therefore applies at every addition.
    The entropy chain identity bounds the total increase by
    $\sum_{i\in S}k=|S|k$.
\end{proof}

\begin{theorem}[Information reset for padded annular regions, with explicit constants]
    \label{thm:peps_angular_information_reset_explicit}
    \lean{TNLean.PEPS.Approximation.angular_information_reset_of_resetThreshold_le}
    \leanok
    \uses{def:peps_angular_grid,def:area_law_ground_entropy,def:peps_reset_width}
    Fix a local dimension $q\ge1$, an interaction bound $J\ge0$, a gap
    $\Delta>0$, and $p>0$, and let $C_p=C_p(J,\Delta,p)$ and
    $L_0=L_0(q,J,\Delta,p)$ be the reset width and size threshold of
    Definition~\ref{def:peps_reset_width}. Let $L\ge L_0$ and let $\Omega$ be a
    normalized ground vector of the square Hamiltonian with interaction bound $J$
    and full-system gap $\Delta$.
    Choose ambient sets $A,E$, a grid origin, and parameters $a'\ge0$,
    $d>0$, and $s>0$.
    Let $R$ be the physical sample of all cells of side $ds$ meeting
    $A^{+3a's/2}$, and let $C=\Lambda_L\cap E^{+3a's/2}$.
    Put $R_0=\Lambda_L\cap A^{+a's}$ and $C_0=\Lambda_L\cap E^{+a's}$.
    Suppose $R\cap C=\varnothing$, $b\ge0$, $I_\Omega(R:C)\le b$ and
    $C_p(1+b+\log L)\le a's/2$.
    Then $I_\Omega(R_0:C_0)\le L^{-p}$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_annular_cell_clearance,thm:peps_physical_integer_sup_distance,
        thm:peps_information_reset_explicit}
    Each core lies in its corresponding larger region and has distance
    strictly greater than $a's/2$ from the corresponding physical complement.
    The common translation of physical lattice coordinates preserves
    the sup distance. The information reset theorem with explicit constants
    applies to the original Hamiltonian and ground vector with these two buffers.
\end{proof}

\begin{definition}[Slope and threshold of the cell reset]
    \label{def:peps_angular_cell_reset_constants}
    \lean{TNLean.PEPS.Approximation.angularCellResetSlope,
        TNLean.PEPS.Approximation.one_le_angularCellResetSlope,

    \leanok
    \uses{def:peps_angular_reset_width_constants}
    For real numbers $C_p,a',d,C_1,N,K$, put
    \begin{align}
        D_{\rm cell}(C_p,a',d)         & =\max\Bigl\{1,\frac{12C_p}{a'},\frac1d\Bigr\},
        \notag                                                                          \\
        L_{\rm cell}(a',d,C_p,C_1,N,K) & =\max\bigl\{L_{\rm w}\bigl(K,C_p(2+NC_1),a'/3,
        D_{\rm cell}(C_p,a',d)\bigr),e\bigr\},
        \notag
    \end{align}
    with $L_{\rm w}$ the size threshold of
    Definition~\ref{def:peps_angular_reset_width_constants}. Then
    $D_{\rm cell}\ge1$ and $L_{\rm cell}\ge e$. The slope satisfies
    $D_{\rm cell}\ge4C_p/c_*$ for the available width $c_*=a'/3$ of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}.
\end{definition}

\begin{theorem}[Uniform reset width for bounded cell costs, with explicit constants]
    \label{thm:peps_angular_reset_width_of_cell_information_explicit}
    \lean{TNLean.PEPS.Approximation.angular_reset_width_lt_of_cell_information}
    \leanok
    \uses{def:peps_angular_cell_reset_constants}
    Fix $a',C_p,C_1,p>0$, $0<d\le1$, and real numbers $N,K\ge0$, and let
    $D\ge D_{\rm cell}(C_p,a',d)$. For all real $L,s$ with
    $L\ge L_{\rm cell}(a',d,C_p,C_1,N,K)$ and $s\ge D\log L$, one has $ds\ge1$ and
    \begin{align}
        C_p\left(1+L^{-p}+NC_1(1+\log(ds))^K+\log L\right)
         & <a's/3.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{cor:peps_angular_reset_width_fits_explicit,
        def:peps_angular_reset_width_constants,def:peps_angular_cell_reset_constants}
    Since $D\ge D_{\rm cell}\ge1$, the threshold $L_{\rm w}$ at the slope $D$ is at
    most the one at $D_{\rm cell}$, hence at most $L$, and $D\ge4C_p/(a'/3)$.
    The explicit fitting estimate applies with coefficients $C_3=C_p(2+NC_1)$ and
    $C_4=C_p$ and width $c=a'/3$.
    Since $L\ge e$, $\log L\ge1$, so $s\ge D\ge1/d$ gives $ds\ge1$, while
    $L^{-p}\le1$ and $(1+\log(ds))^K\le(1+\log s)^K$.
    Since $(1+\log s)^K\ge1$, the required width is at most
    $C_3(1+\log s)^K+C_4\log L$, which is strictly less than $a's/3$.
\end{proof}

\begin{theorem}[Selected-cell information with explicit constants]
    \label{thm:peps_selected_region_information_of_collar_explicit}
    \lean{TNLean.PEPS.Approximation.mutualInformation_selectedRegion_le_square_of_sublinearCollarBoundAt}
    \leanok
    \uses{def:area_law_collar_bound,def:area_law_amplification_bound,
        def:polynomial_peps_model,def:peps_angular_grid,def:peps_cell_information_constants}
    Fix $q\in\mathbb N$, $J\ge0$, $\Delta\in\mathbb R$ and template constants
    $D_0,C_{\rm tpl}$. Suppose that the range-one sublinear collar bound holds
    with constant $C$ and threshold $N$, and that the range-one amplification
    bound holds at $(\max\{C,1\},1)$ with constant $C_a$ and threshold $N_a$.
    Fix $c>0$, and let $C_{\rm cell},K_{\rm cell}$ be the constants of
    Definition~\ref{def:peps_cell_information_constants} at this $c$.
    Let $s\ge1$, let $o\in\mathbb R^2$ be a grid origin, let $r\in\mathbb R$,
    and let $A_{\rm old}\subseteq A$ be ambient sets. Let $S$ be a finite set
    of cell indices consisting exactly of the cells of side $s$ meeting $A^{+r}$
    but not $A_{\rm old}^{+r}$.
    On the original open square $\Lambda_L=\{1,\ldots,L\}^2$, let $H$ have
    interaction bound $J$ and let $\Omega$ be a unit ground vector with
    full-system gap $\Delta$.
    Write $R_{\rm old}$ and $R$ for the physical samples of the two selected
    unions. Let $E\subseteq\Lambda_L$ have distance at least $cs$ from every
    point of the ambient selected union for $A$.
    If $I_\Omega(R_{\rm old}:E)\le b_0$, then
    \begin{align}
        I_\Omega(R:E) & \le b_0+|S|C_{\rm cell}(1+\log s)^{K_{\rm cell}}.
        \notag
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_square_cell_conditional_information_of_collar_explicit,
        thm:peps_selected_region_information_of_cell_bounds}
    The square-grid cell estimate with explicit constants applies to every new
    cell conditioned on any union of the other whole cells, with the same

    \end{align}

        I_\Omega(\Lambda\cap A_j^{+a's_j}:
        \Lambda\cap E^{+a's_j})\leq\varepsilon.
        \notag
    \end{align}

\end{theorem}

    scale. At the first scale the old group is empty, whose entropy is
    zero because the state is normalized. The initial selected cells

    At each later scale, the old selected group and the enlarged
    exterior lie inside the two preceding padded regions. Monotonicity
    of mutual information therefore bounds their information by
    $\varepsilon$. The exact new-cell count is again at most $N$, and
    the reset implication proves the invariant at the new scale.
    In the finite induction of
    \cite[Proposition~3.4]{OpenAI2026PolynomialPEPS}, $A_m=X$, and removing
    the padding from both regions gives the desired estimate.
\end{proof}

    disjoint physical regions $X,E\subseteq\Lambda_L$, and a real scale
    $n$ satisfying $D\log L\le n\le A_0L$ and
    $\operatorname{diam}_\infty X\le A_0n$.
    Put $d_{\mathcal V}(x)=\min_{v\in\mathcal V}|x-v|_\infty$, with
    $d_{\varnothing}=+\infty$. If every $x\in X$ and $y\in E$ satisfy
    \begin{align}
        |x-y|_\infty
         & \ge a\max\{D\log L,\min(n,d_{\mathcal V}(x))\},
        \notag
    \end{align}
    then $I_\Omega(X:E)\le L^{-p}$.
    The constants precede the Hamiltonian, ground vector, marked points,
    physical regions and scale.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_angular_information_of_collar_explicit,
        thm:area_law_amplification,def:peps_angular_information_constants}
    The collar bound holds with some constant $C$ and threshold $N_{\rm col}$,
    and by Theorem~\ref{thm:area_law_amplification} the amplification bound at
    $(\max\{C,1\},1)$ holds with some constant $C_a$ and threshold $N_a$.
    Theorem~\ref{thm:peps_angular_information_of_collar_explicit} then applies
    with $D=D_{\rm ang}(J,\Delta,a,p)$ and $L_0=L_{\rm ang}$ at these
    constants.
\end{proof}
