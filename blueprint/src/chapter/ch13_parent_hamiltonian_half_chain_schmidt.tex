\section{Half-chain Schmidt spectrum and the dual fixed point}%
\label{sec:half_chain_schmidt}

On a ring of $2L$ sites split into two halves of $L$ sites each, the matrix
product vector splits at the two boundaries into a pair of half-chain
families whose overlaps are matrix elements of the $L$-fold transfer
map~\cite[Theorem~6]{PerezGarcia2007Matrix}.
The unnormalized half-chain reduced density matrix factors through the
half-chain Gram matrix. Away from zero, its spectrum matches that of a
boundary comparison matrix of size $D^2$, which, under the spectral gap
condition $\rho_{\mathrm{spec}}(\E_A - P_\Lambda) < 1$, converges entrywise,
geometrically in $L$, to the diagonal matrix with entries
$\lambda_\alpha\lambda_\beta/\tr(\Lambda)^2$ determined by the fixed point $\Lambda$.

The source states its canonical form with a trace-preserving dual map and a
positive diagonal fixed point $\Lambda$ of that dual. Here the trace-preserving
direction is placed on the transfer map itself and $\Lambda$ is its fixed
point; the two conventions are exchanged by passing to the
conjugate-transposed tensor.

\begin{definition}[Half-chain boundary families]%
    \label{def:half_chain_family}
    \lean{MPSTensor.halfChainMatrix}
    \lean{MPSTensor.halfChainMatrix_apply}
    \lean{MPSTensor.halfChainSwapMatrix}
    \lean{MPSTensor.halfChainSwapMatrix_apply}
    \leanok
    \uses{def:mps_tensor, def:eval_word}
    For a tensor $A$ and a half-chain length $L$, the \emph{half-chain family}
    is the matrix $\Psi$ with rows indexed by configurations
    $\sigma\in\Fin{d}^L$ and columns indexed by boundary pairs
    $(\alpha,\beta)$, whose column at $(\alpha,\beta)$ is the half-chain
    vector
    \begin{align}
        \ket{\Psi_{\alpha,\beta}}
         & =\sum_{\sigma}
        \bra{\alpha}A^{\sigma}\ket{\beta}\,\ket{\sigma}.
        \label{eq:half_chain_vector}
    \end{align}
    The \emph{complementary family} $\Phi$ is the matrix with rows indexed by
    boundary pairs and columns by configurations, whose row at
    $(\alpha,\beta)$ is the vector $\ket{\Psi_{\beta,\alpha}}$ with the two
    boundary indices swapped.
\end{definition}

\begin{definition}[Half-chain coefficient kernel]%
    \label{def:half_chain_kernel}
    \lean{MPSTensor.halfChainKernel}
    \lean{MPSTensor.halfChainKernel_apply}
    \leanok
    \uses{def:mpv}
    The \emph{coefficient kernel} of the ring of $2L$ sites is the matrix over
    the two half-chain configuration spaces whose entry at $(\sigma,\tau)$ is
    the matrix product vector component
    $V^{(2L)}(A)_{\sigma\tau}=\tr(A^{\sigma}A^{\tau})$ of the joined
    configuration.
\end{definition}

\begin{lemma}[Splitting of the ring state at the two boundaries]%
    \label{lem:half_chain_kernel_split}
    \lean{MPSTensor.halfChainKernel_eq_mul}
    \leanok
    \uses{def:half_chain_family, def:half_chain_kernel}
    The coefficient kernel factors as the product $\Psi\Phi$ of the
    half-chain family with its complementary family. Equivalently,
    \begin{align}
        \ket{\psi}
         & =\sum_{\alpha,\beta}
        \ket{\Psi_{\alpha,\beta}}\ket{\Psi_{\beta,\alpha}}.
        \label{eq:half_chain_resolution}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    Inserting the resolution of the identity
    $\sum_{\beta}\ket{\beta}\bra{\beta}$ at the two cut bonds of the ring
    trace yields $\tr(A^{\sigma}A^{\tau})
        =\sum_{\alpha,\beta}\bra{\alpha}A^{\sigma}\ket{\beta}
        \bra{\beta}A^{\tau}\ket{\alpha}$.
\end{proof}

\begin{definition}[Half-chain reduced density matrix]%
    \label{def:half_chain_reduced}
    \lean{MPSTensor.halfChainReducedMatrix}
    \lean{MPSTensor.halfChainReducedMatrix_apply}
    \lean{MPSTensor.posSemidef_halfChainReducedMatrix}
    \leanok
    \uses{def:half_chain_kernel}
    The \emph{unnormalized half-chain reduced density matrix} of the ring
    state on $2L$ sites is $\rho_L=KK^{\dagger}$ with $K$ the coefficient
    kernel. Entrywise this is the partial trace of the unnormalized ring state
    over the complementary half,
    \begin{align}
        (\rho_L)_{\sigma\sigma'}
         & =\sum_{\tau}
        V^{(2L)}(A)_{\sigma\tau}\,
        \overline{V^{(2L)}(A)_{\sigma'\tau}},
        \label{eq:half_chain_reduced_entry}
    \end{align}
    and $\rho_L$ is positive semidefinite.
\end{definition}

\begin{definition}[Half-chain Gram matrix]%
    \label{def:half_chain_gram}
    \lean{MPSTensor.halfChainGram}
    \leanok
    \uses{def:half_chain_family}
    The \emph{half-chain Gram matrix} is $G=\Psi^{\dagger}\Psi$, the matrix of
    overlaps
    $G_{(\alpha,\beta),(\alpha',\beta')}
        =\braket{\Psi_{\alpha,\beta}}{\Psi_{\alpha',\beta'}}$ of the half-chain
    vectors.
\end{definition}

\begin{lemma}[Half-chain overlaps through the transfer map]%
    \label{lem:half_chain_gram_transfer}
    \lean{MPSTensor.halfChainGram_apply_eq_transferMap_pow}
    \leanok
    \uses{def:half_chain_gram, def:transfer_map}
    The half-chain overlaps are matrix elements of the $L$-fold transfer map
    on matrix units:
    \begin{align}
        \braket{\Psi_{\alpha,\beta}}{\Psi_{\alpha',\beta'}}
         & =\bra{\alpha'}\E_A^{L}
        \bigl(\ket{\beta'}\bra{\beta}\bigr)\ket{\alpha}.
        \label{eq:half_chain_gram_transfer}
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    Expanding the overlap as a sum over configurations of products
    $\bra{\alpha'}A^{\sigma}\ket{\beta'}
        \overline{\bra{\alpha}A^{\sigma}\ket{\beta}}$ yields the $(\alpha',\alpha)$ entry of
    $\sum_{\sigma}A^{\sigma}\ket{\beta'}\bra{\beta}(A^{\sigma})^{\dagger}$,
    which is the $L$-fold transfer map applied to the matrix unit.
\end{proof}

\begin{lemma}[Half-chain Schmidt factorization]%
    \label{thm:half_chain_schmidt_factorization}
    \lean{MPSTensor.halfChainReducedMatrix_eq_mul_swap_gram}
    \lean{MPSTensor.halfChainSwapMatrix_mul_conjTranspose}
    \leanok
    \uses{def:half_chain_reduced, def:half_chain_gram}
    Let $S$ denote the swap of the two boundary indices. The unnormalized
    half-chain reduced density matrix factors exactly as
    \begin{align}
        \rho_L
         & =\Psi\,\bigl(SG^{\mathsf T}S\bigr)\,\Psi^{\dagger},
        \label{eq:half_chain_factorization}
    \end{align}
    where $SG^{\mathsf T}S$ is the Gram matrix of the complementary family.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:half_chain_kernel_split}
    By the splitting $K=\Psi\Phi$ of
    Lemma~\ref{lem:half_chain_kernel_split},
    $\rho_L=\Psi(\Phi\Phi^{\dagger})\Psi^{\dagger}$, and the Gram matrix
    $\Phi\Phi^{\dagger}$ of the complementary family is the transpose of the
    boundary-swapped half-chain Gram matrix.
\end{proof}

\begin{lemma}[Boundary comparison of the half-chain spectrum]%
    \label{thm:half_chain_charpoly}
    \lean{MPSTensor.charpoly_halfChainReducedMatrix}
    \leanok
    \uses{def:half_chain_reduced, def:half_chain_gram}
    The characteristic polynomials of the half-chain reduced density matrix
    and of the boundary comparison matrix $(SG^{\mathsf T}S)G$ of size $D^2$
    agree after matching the trivial kernel factors:
    \begin{align}
        X^{D^2}\,\chi_{\rho_L}(X)
         & =X^{d^L}\,\chi_{(SG^{\mathsf T}S)G}(X).
        \label{eq:half_chain_charpoly}
    \end{align}
    In particular the nonzero spectrum of $\rho_L$, with multiplicities, is
    that of the boundary comparison matrix.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:half_chain_schmidt_factorization, lem:matrix_rectangular_charpoly_cyclicity}
    Applying rectangular cyclicity of the characteristic polynomial to the
    factorization (\ref{eq:half_chain_factorization}) after regrouping the product
    as $\Psi$ times $(SG^{\mathsf T}S)\Psi^{\dagger}$ replaces $\rho_L$ by the reversed product
    $(SG^{\mathsf T}S)\Psi^{\dagger}\Psi=(SG^{\mathsf T}S)G$.
\end{proof}

\begin{lemma}[Geometric transfer error of the half-chain overlaps]%
    \label{thm:half_chain_gram_geometric}
    \lean{MPSTensor.exists_geometric_bound_halfChainGram_sub_fixedPointProj}
    \leanok
    \uses{def:half_chain_gram, def:transfer_map}
    Let $A$ have trace-preserving transfer map $\E_A$ with fixed point
    $\Lambda$, $\tr(\Lambda)\neq0$, and suppose
    \begin{align}
        \rho_{\mathrm{spec}}\bigl(\E_A-P_{\Lambda}\bigr) & <1,
        \label{eq:half_chain_c2_gap}
    \end{align}
    where $P_{\Lambda}$ is the fixed-point projection. Then there are
    constants $C>0$ and $0<r<1$ such that for every $L\geq1$ and all boundary
    pairs,
    \begin{align}
        \Bigl|
        G_{(\alpha,\beta),(\alpha',\beta')}
        -\bra{\alpha'}P_{\Lambda}
        \bigl(\ket{\beta'}\bra{\beta}\bigr)\ket{\alpha}
        \Bigr|
         & \leq C r^{L}.
        \label{eq:half_chain_gram_geometric}
    \end{align}
    The hypothesis~(\ref{eq:half_chain_c2_gap}) is the spectral gap condition
    of \cite[Theorem~6]{PerezGarcia2007Matrix} in the trace-preserving gauge:
    apart from the simple eigenvalue one, the spectrum of $\E_A$ lies
    strictly inside the unit disc. The rate $r$ is chosen strictly between the
    spectral radius of $\E_A-P_{\Lambda}$ and one.
\end{lemma}

\begin{proof}
    \leanok
    \uses{lem:half_chain_gram_transfer, thm:geometric_bound_of_spectral_radius_lt_one}
    For $L\geq1$, the power decomposition gives
    $\E_A^{L}=P_{\Lambda}+(\E_A-P_{\Lambda})^{L}$. By (\ref{eq:half_chain_gram_transfer}),
    the deviation of each Gram entry from the corresponding fixed-point projection
    entry is a matrix element of $(\E_A-P_{\Lambda})^{L}$ applied to a matrix unit,
    which is bounded by the operator norm of $(\E_A-P_{\Lambda})^{L}$ and therefore
    decays geometrically below unit spectral radius.
\end{proof}

\begin{lemma}[Diagonal limit of the half-chain overlaps]%
    \label{lem:half_chain_gram_limit_diagonal}
    \lean{MPSTensor.fixedPointProj_single_entry_diagonal}
    \leanok
    For a diagonal fixed point
    $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_D)$ with
    $\tr(\Lambda)\neq0$, the limiting overlaps form a diagonal matrix in the
    boundary pairs:
    \begin{align}
        \bra{\alpha'}P_{\Lambda}
        \bigl(\ket{\beta'}\bra{\beta}\bigr)\ket{\alpha}
         & =\frac{\lambda_\alpha}{\tr(\Lambda)}\,
        \delta_{\alpha,\alpha'}\delta_{\beta,\beta'}.
        \label{eq:half_chain_limit_diagonal}
    \end{align}
    When $\tr(\Lambda)=1$, as in \cite{PerezGarcia2007Matrix}, this formula reduces
    to $\lambda_\alpha\delta_{\alpha,\alpha'}\delta_{\beta,\beta'}$.
\end{lemma}

\begin{proof}
    \leanok
    Evaluating the fixed-point projection on $\ket{\beta'}\bra{\beta}$ gives
    $\tr(\ket{\beta'}\bra{\beta})=\delta_{\beta,\beta'}$, while the diagonal entry
    of $\Lambda$ at $(\alpha',\alpha)$ is $\lambda_\alpha\delta_{\alpha,\alpha'}$.
\end{proof}

\begin{lemma}[Geometric convergence to the doubled fixed-point spectrum]%
    \label{thm:half_chain_comparison_geometric}
    \lean{MPSTensor.exists_geometric_bound_halfChainComparison_sub_diagonal}
    \leanok
    \uses{def:half_chain_gram, def:transfer_map}
    Let $A$ have trace-preserving transfer map with diagonal fixed point
    $\Lambda=\operatorname{diag}(\lambda_1,\ldots,\lambda_D)$,
    $\tr(\Lambda)\neq0$, satisfying the spectral condition
    (\ref{eq:half_chain_c2_gap}). Then there are constants $C>0$ and $0<r<1$
    such that for every $L\geq1$ the boundary comparison matrix satisfies,
    entrywise,
    \begin{align}
        \Bigl|
        \bigl((SG^{\mathsf T}S)G\bigr)_{(\alpha,\beta),(\alpha',\beta')}
        -\frac{\lambda_\alpha\lambda_\beta}{\tr(\Lambda)^2}\,
        \delta_{\alpha,\alpha'}\delta_{\beta,\beta'}
        \Bigr|
         & \leq C r^{L}.
        \label{eq:half_chain_comparison_geometric}
    \end{align}
    The limiting matrix is the diagonal matrix $\Lambda\otimes\Lambda$ up to
    the trace normalization. Together with
    Lemma~\ref{thm:half_chain_charpoly}, the nonzero spectrum of the
    half-chain reduced density matrix on the ring of $2L$ sites is that of a
    $D^2\times D^2$ matrix converging entrywise, geometrically in $L$, to
    $\Lambda\otimes\Lambda/\tr(\Lambda)^2$. Lemma~\ref{thm:half_chain_ordered_limit}
    proves convergence of the normalized physical eigenvalues.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:half_chain_gram_geometric, lem:half_chain_gram_limit_diagonal}
    By Lemma~\ref{thm:half_chain_gram_geometric} and
    Lemma~\ref{lem:half_chain_gram_limit_diagonal}, both factors of
    $(SG^{\mathsf T}S)G$ lie entrywise within $Cr^{L}$ of their diagonal
    limits, since swapping and transposing $G$ permute its entries and exchange
    the two diagonal weights. Because the limiting entries are bounded by some
    $B$, the matrix product deviates from the product of the limits by at most
    $\bigl(D^2(C(B+C)+BC)+1\bigr)r^{L}$.
\end{proof}

\subsection{Ordered eigenvalues and finite-size normalization}

\begin{definition}[Zero-extended ordered spectrum]%
    \label{def:half_chain_ordered_spectrum}
    \lean{Matrix.IsHermitian.paddedEigenvalues}
    \lean{Matrix.IsHermitian.paddedEigenvalues_fin}
    \lean{Matrix.IsHermitian.paddedEigenvalues_eq_zero}
    \lean{Matrix.PosSemidef.eigenvalues₀_nonneg}
    \leanok
    For a Hermitian matrix $M$ of size $n$, let $s_k(M)$ be its $k$th
    largest eigenvalue, counted with multiplicity, for $0\leq k<n$, and
    set $s_k(M)=0$ for $k\geq n$. If $M\geq0$, all these values are
    nonnegative.
\end{definition}

\begin{lemma}[Ordered functional calculus and perturbations]%
    \label{thm:half_chain_weyl}
    \lean{Matrix.IsHermitian.eigenvalues₀_cfc_monotone}
    \lean{Matrix.IsHermitian.eigenvalues₀_add_algebraMap}
    \lean{Matrix.IsHermitian.eigenvalues₀_smul}
    \lean{Matrix.IsHermitian.abs_eigenvalues₀_sub_le}
    \lean{Matrix.IsHermitian.eigenvalues₀_tendsto}
    \lean{Matrix.IsHermitian.paddedEigenvalues_smul}
    \lean{Matrix.IsHermitian.paddedEigenvalues_tendsto}
    \leanok
    An increasing real function acts termwise on the ordered eigenvalues
    of a Hermitian matrix. In particular, adding $tI$ adds $t$ to each
    eigenvalue, and multiplication by $t\geq0$ multiplies each eigenvalue
    by $t$. If $M$ and $N$ are Hermitian matrices of the same size, then
    $|s_k(M)-s_k(N)|\leq\lVert M-N\rVert_{\mathrm{op}}$ for every $k$.
    Consequently, convergence of Hermitian matrices of fixed size implies
    convergence of every zero-extended ordered eigenvalue.
\end{lemma}

\begin{proof}
    \leanok
    Applying the increasing scalar function to the spectral decomposition and
    sorting the characteristic roots in decreasing order gives the
    functional-calculus identities. The operator inequality
    $-\lVert M-N\rVert I\leq M-N\leq\lVert M-N\rVert I$, combined with Weyl
    monotonicity and scalar translation, then yields the eigenvalue
    perturbation bound, from which convergence of the zero-extended ordered
    eigenvalues follows.
\end{proof}

\begin{lemma}[Characteristic roots with zero padding]%
    \label{thm:half_chain_padded_charpoly}
    \lean{Matrix.PosSemidef.paddedEigenvalues_eq_of_charpoly}
    \leanok
    \uses{def:half_chain_ordered_spectrum}
    Let $M\geq0$ have size $n$ and $N\geq0$ have size $m$.
    If $X^m\chi_M(X)=X^n\chi_N(X)$, then $s_k(M)=s_k(N)$ for every
    $k\geq0$.
\end{lemma}

\begin{proof}
    \leanok
    Sorting the real characteristic roots in decreasing order places the
    additional $m$ and $n$ zeros of the nonnegative matrices $M$ and $N$ at
    the ends of their respective ordered lists, so extending either list by
    further zeros produces the same sequence.
\end{proof}

\begin{definition}[Positive boundary density and ring normalization]%
    \label{def:half_chain_positive_density}
    \lean{MPSTensor.halfChainSpectralMatrix}
    \lean{MPSTensor.halfChainNormSq}
    \lean{MPSTensor.normalizedHalfChainReducedMatrix}
    \lean{MPSTensor.normalizedHalfChainSpectralMatrix}
    \lean{MPSTensor.halfChainEigenvalues}
    \leanok
    \uses{def:half_chain_reduced, def:half_chain_gram, def:half_chain_ordered_spectrum}
    Write $W_L=\Phi_L\Phi_L^\dagger=SG_L^{\mathsf T}S$ and define
    \begin{align}
        H_L            & = \sqrt{G_L}\,W_L\sqrt{G_L},
                       & Z_L                          & = \operatorname{tr}\rho_L,
        \label{eq:half_chain_positive_density}                                     \\
        \widehat\rho_L & = Z_L^{-1}\rho_L,
                       & \widehat H_L                 & = Z_L^{-1}H_L.
        \label{eq:half_chain_normalized_density}
    \end{align}
    Here $\rho_L$ is the unnormalized physical reduced matrix.
    Both normalized matrices are defined as zero when $Z_L=0$.
    The physical entanglement spectrum is $s_k(\widehat\rho_L)$.
\end{definition}

\begin{lemma}[Exact physical spectral comparison]%
    \label{thm:half_chain_positive_comparison}
    \lean{MPSTensor.posSemidef_halfChainSpectralMatrix}
    \lean{MPSTensor.charpoly_halfChainSpectralMatrix}
    \lean{MPSTensor.halfChainNormSq_eq_sum}
    \lean{MPSTensor.halfChainNormSq_nonneg}
    \lean{MPSTensor.trace_halfChainSpectralMatrix}
    \lean{MPSTensor.posSemidef_normalizedHalfChainReducedMatrix}
    \lean{MPSTensor.posSemidef_normalizedHalfChainSpectralMatrix}
    \lean{MPSTensor.charpoly_normalizedHalfChainSpectralMatrix}
    \lean{MPSTensor.trace_normalizedHalfChainReducedMatrix}
    \lean{MPSTensor.halfChainEigenvalues_eq_boundary}
    \lean{MPSTensor.halfChainEigenvalues_eq_zero}
    \leanok
    \uses{def:half_chain_positive_density}
    For every finite half-chain length $L$, $H_L\geq0$ and
    \begin{align}
        Z_L & = \sum_{\sigma,\tau}
        |V^{(2L)}(A)_{\sigma\tau}|^2
        =\operatorname{tr}H_L\geq0,
        \label{eq:half_chain_actual_norm}     \\
        X^{D^2}\chi_{\widehat\rho_L}(X)
            & =X^{d^L}\chi_{\widehat H_L}(X).
        \label{eq:half_chain_normalized_charpoly}
    \end{align}
    The same characteristic-polynomial identity holds before normalization.
    Both normalized matrices are positive semidefinite, and
    $s_k(\widehat\rho_L)=s_k(\widehat H_L)$ for every $k$.
    In particular, the physical spectrum vanishes for $k\geq D^2$.
    These statements include $d^L<D^2$, singular Gram matrices, and
    zero-norm lengths. If $Z_L>0$, then
    $\operatorname{tr}\widehat\rho_L=1$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:half_chain_charpoly, thm:half_chain_padded_charpoly}
    Because $H_L$ is a positive congruence of $W_L$, it is positive
    semidefinite, and cyclicity of trace together with $\sqrt{G_L}\sqrt{G_L}=G_L$
    yields (\ref{eq:half_chain_actual_norm}). Applying the rectangular-product
    characteristic-polynomial identity to $\rho_L=\Psi_L W_L\Psi_L^\dagger$ and
    the square-product identity to $H_L$, then scaling both products by
    $Z_L^{-1}$, gives (\ref{eq:half_chain_normalized_charpoly}), from which
    Lemma~\ref{thm:half_chain_padded_charpoly} equates the ordered
    eigenvalues.
\end{proof}

\begin{definition}[Two-boundary limiting density]%
    \label{def:half_chain_limiting_density}
    \lean{MPSTensor.halfChainGramLimit}
    \lean{MPSTensor.halfChainSpectrumLimit}
    \lean{MPSTensor.halfChainSpectrumLimit_eq_kronecker}
    \lean{MPSTensor.posSemidef_halfChainSpectrumLimit}
    \lean{MPSTensor.trace_halfChainSpectrumLimit}
    \leanok
    For $\lambda_\alpha\geq0$ with $\sum_\alpha\lambda_\alpha=1$, let
    $\Lambda=\operatorname{diag}(\lambda_\alpha)$,
    $G_\infty=\operatorname{diag}_{\alpha\beta}(\lambda_\alpha)$, and
    $H_\infty=\Lambda\otimes\Lambda
        =\operatorname{diag}_{\alpha\beta}(\lambda_\alpha\lambda_\beta)$.
    The matrix $H_\infty$ is positive semidefinite and has trace one.
\end{definition}

\begin{lemma}[Normalized physical eigenvalue convergence]%
    \label{thm:half_chain_ordered_limit}
    \lean{MPSTensor.tendsto_halfChainGram}
    \lean{MPSTensor.tendsto_halfChainSpectralMatrix_of_diagonal}
    \lean{MPSTensor.tendsto_halfChainSpectralMatrix}
    \lean{MPSTensor.tendsto_halfChainNormSq}
    \lean{MPSTensor.eventually_halfChainNormSq_pos}
    \lean{MPSTensor.tendsto_normalizedHalfChainSpectralMatrix}
    \lean{MPSTensor.tendsto_halfChainEigenvalues_of_spectralMatrix}
    \lean{MPSTensor.tendsto_halfChainEigenvalues}
    \leanok
    \uses{def:half_chain_positive_density, def:half_chain_limiting_density}
    Suppose $\E_A$ is trace preserving, fixes $\Lambda$, and satisfies
    (\ref{eq:half_chain_c2_gap}). Then $Z_L\to1$, so $Z_L>0$ for all
    sufficiently large $L$. The normalized physical eigenvalues obey
    \begin{align}
        \lim_{L\to\infty}s_k(\widehat\rho_L)
         & =s_k(\Lambda\otimes\Lambda)
        .
        \label{eq:half_chain_ordered_limit}
    \end{align}
    Repeated eigenvalues are counted with multiplicity, and both lists use
    the zero-extension convention of Definition~\ref{def:half_chain_ordered_spectrum}.
    No invertibility of $\Lambda$ or finite-length normalization is assumed.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:half_chain_gram_geometric, thm:half_chain_positive_comparison,
        thm:half_chain_weyl}
    The geometric transfer estimate $Cr^L$ gives $G_L\to G_\infty$.
    More generally, whenever $G_L\to\operatorname{diag}(g_p)$ with $g_p\geq0$,
    continuity of the positive square root implies
    $H_L\to\operatorname{diag}(g_p g_{p^{\mathrm{swap}}})$, so specializing to
    $g_{\alpha\beta}=\lambda_\alpha$ yields
    $H_L\to\Lambda\otimes\Lambda$.
    Taking traces gives $Z_L\to1$ and hence
    $\widehat H_L\to\Lambda\otimes\Lambda$, from which Weyl's bound and the
    exact physical comparison yield (\ref{eq:half_chain_ordered_limit})
    (here square-root continuity establishes qualitative convergence rather
    than a sharper quantitative rate).
\end{proof}

\subsection{The unital canonical convention}

\begin{lemma}[Half-chain Gram matrices under adjoints]%
    \label{thm:half_chain_adjoint_gram}
    \lean{MPSTensor.halfChainGram_conjTranspose}
    \leanok
    For $B_i=A_i^\dagger$, the half-chain Gram matrices satisfy
    $G_L(B)_{(\beta',\alpha'),(\beta,\alpha)}
        =G_L(A)_{(\alpha,\beta),(\alpha',\beta')}$ at every length.
\end{lemma}

\begin{proof}
    \leanok
    Because the transfer map of $B$ is the trace-pairing adjoint of that of
    $A$, its $L$-th power is the corresponding power of the adjoint, and
    evaluating this relation on the matrix units in the overlap formula
    exchanges the boundary pairs.
\end{proof}

\begin{lemma}[Half-chain limit for normal unital tensors]%
    \label{thm:half_chain_unital_limit}
    \lean{MPSTensor.tendsto_halfChainGram_unital}
    \lean{MPSTensor.tendsto_halfChainSpectralMatrix_unital}
    \lean{MPSTensor.tendsto_halfChainEigenvalues_unital}
    \lean{MPSTensor.halfChainNormSq_unital}
    \leanok
    Let $A$ be a normal tensor in the unital canonical convention,
    $\sum_i A_iA_i^\dagger=I$, and let
    $\Lambda=\operatorname{diag}(\lambda_\alpha)\geq0$ have trace one and
    satisfy $\sum_i A_i^\dagger\Lambda A_i=\Lambda$.
    Then $Z_L\to1$ and $Z_L>0$ for all sufficiently large $L$.
    For every $k\in\mathbb N$, the physical reduced density obeys
    $s_k(\widehat\rho_L)\to s_k(\Lambda\otimes\Lambda)$.
    Here normality is assumed explicitly, whereas
    Theorem~\ref{thm:half_chain_source_limit} derives it from the source
    canonical conditions and Condition C2.
    Under the index and decay conventions detailed in
    \cite{gap:pgvwc07_half_chain_spectrum_conventions}, the intermediate Gram
    limit is $\operatorname{diag}_{\alpha\beta}(\lambda_\beta)$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:half_chain_adjoint_gram, thm:half_chain_ordered_limit}
    Normality gives irreducibility and a unique peripheral eigenvalue, both of
    which pass to the adjoint family $B_i=A_i^\dagger$ because the dual channel
    has the same eigenvalues under the trace pairing. Its complementary
    fixed-point map therefore has spectral radius strictly below one, so the
    geometric Gram estimate applied to $B_i=A_i^\dagger$ and
    Lemma~\ref{thm:half_chain_adjoint_gram} give the Gram limit for the
    original tensor $A$. With $g_{\alpha\beta}=\lambda_\beta$, the positive
    boundary limit has entries $g_{\alpha\beta}g_{\beta\alpha}
        =\lambda_\alpha\lambda_\beta$, and the norm and ordered-eigenvalue
    arguments of Lemma~\ref{thm:half_chain_ordered_limit} then yield the
    eigenvalue convergence of the physical reduced matrices.
\end{proof}


\begin{theorem}[Half-chain interpretation from canonical form and Condition C2]%
    \label{thm:half_chain_source_limit}
    \lean{MPSTensor.isNormalTensor_of_pgvwc07_canonical_c2}
    \lean{MPSTensor.tendsto_halfChainEigenvalues_of_pgvwc07_canonical_c2}
    \leanok
    Let $A$ satisfy the three one-block canonical conditions of
    \cite[Theorem~4]{PerezGarcia2007Matrix}: $\sum_i A_iA_i^\dagger=I$,
    a faithful diagonal dual fixed point $\Lambda=\operatorname{diag}
        (\lambda_\alpha)>0$, and
    $\{X:\E_A(X)=X\}=\mathbb C I$.
    Normalize $\operatorname{tr}\Lambda=1$ and the transfer spectral radius
    to one. Assume Condition C2: every unit-modulus transfer eigenvalue is one.
    Then $A$ is normal. For every $k\in\mathbb N$, the ordered physical
    half-chain eigenvalues, with multiplicities and zero extension, satisfy
    $s_k(\widehat\rho_L)\to s_k(\Lambda\otimes\Lambda)$.
    This gives the one-block canonical half-chain conclusion of
    \cite[Theorem~6]{PerezGarcia2007Matrix} without an additional normality
    or complementary-gap assumption. The hypothesis that this is a single
    canonical block is explicit; reduction of a general weighted canonical
    representation is not asserted here.
\end{theorem}

\begin{proof}
    \leanok
    \uses{lem:pgvwc07_block_irreducible, thm:half_chain_unital_limit}
    The faithful dual fixed point and scalar fixed-point condition imply
    irreducibility: an invariant projection has a positive defect with
    zero pairing against $\Lambda$, hence is fixed, scalar, and trivial.
    Together with the identity eigenvector from unitality, the exclusion of
    other peripheral eigenvalues by Condition C2, and the spectral-radius
    normalization, this irreducibility establishes that $A$ is normal, so
    Lemma~\ref{thm:half_chain_unital_limit} yields the half-chain eigenvalue
    limit.
\end{proof}


\begin{lemma}[Uniform periodic norm of a unital tensor]
    \label{lem:half_chain_unital_periodic_norm}
    \lean{MPSTensor.norm_mpvState_le_bondDimension_of_unital}
    \notready
    A unital tensor of positive bond dimension $D$ satisfies
    $\|\psi_N\|\leq D$ at every length $N$, without primitivity or
    a spectral gap.
\end{lemma}
\begin{proof}
    Positivity and unitality place every transfer eigenvalue in the
    closed unit disk. The squared periodic norm is the trace of the
    transfer power, so its modulus is at most the matrix-space
    dimension $D^2$, counting algebraic multiplicities.
\end{proof}

\begin{lemma}[Vanishing physical contribution of subunit canonical weights]
    \label{lem:half_chain_weighted_remainder}
    \lean{MPSTensor.PGVWC07CanonicalFormData.norm_mpvState_sub_unitWeight_sum_le,
        MPSTensor.PGVWC07CanonicalFormData.tendsto_norm_mpvState_sub_unitWeight_sum}
    \notready
    \uses{lem:half_chain_unital_periodic_norm}
    Let $A_i=\bigoplus_j w_j A_i^j$ be a TI canonical representation with
    $0<w_j\leq1$, unital blocks $A^j$, and block dimensions $D_j$.
    Its actual periodic vector satisfies
    \begin{align}
        \left\|\psi_N(A)-\sum_{w_j=1}\psi_N(A^j)\right\|
         & \leq\sum_{w_j<1}w_j^N D_j\longrightarrow0.
        \label{eq:half_chain_weighted_remainder}
    \end{align}
    No condition on the peripheral transfer spectrum is required for
    this estimate.
\end{lemma}
\begin{proof}
    The exact canonical decomposition is
    $\psi_N(A)=\sum_jw_j^N\psi_N(A^j)$.
    Subtract the unit-weight terms and apply the triangle inequality
    and Lemma~\ref{lem:half_chain_unital_periodic_norm}. Each remaining
    geometric sequence tends to zero, and there are finitely many blocks.
\end{proof}

\begin{lemma}[Normalized diagonal dual fixed point of a canonical block]
    \label{lem:half_chain_canonical_normalized_dual}
    \lean{MPSTensor.PGVWC07CanonicalFormData.exists_normalized_diagonal_dualFixedPoint}
    \notready
    Each canonical block has a strictly positive diagonal dual fixed point
    $\operatorname{diag}(\lambda_a)$ with $\sum_a\lambda_a=1$.
\end{lemma}
\begin{proof}
    The canonical dual fixed point $\Lambda$ is diagonal and positive
    definite, so its diagonal entries are real and strictly positive.
    Divide them by their positive sum. Linearity preserves the dual
    fixed-point equation.
\end{proof}

\begin{lemma}[Transfer eigenvectors of canonical sectors]
    \label{lem:half_chain_canonical_sector_eigenvalues}
    \lean{MPSTensor.PGVWC07CanonicalFormData.transferMap_reindex_blockDiagonal,
        MPSTensor.PGVWC07CanonicalFormData.sectorProjection,
        MPSTensor.PGVWC07CanonicalFormData.sectorProjection_ne_zero,
        MPSTensor.PGVWC07CanonicalFormData.transferMap_sectorProjection,
        MPSTensor.PGVWC07CanonicalFormData.hasEigenvalue_weight_sq,
        MPSTensor.PGVWC07CanonicalFormData.hasEigenvalue_of_block}
    \notready
    For a weighted canonical tensor, let $P_j$ project onto the $j$th
    bond sector. Then $P_j\ne0$ and $\mathcal E_A(P_j)=w_j^2P_j$.
    More generally, every eigenvalue $\mu$ of the transfer map of the
    unweighted block $A^j$ gives the eigenvalue $w_j^2\mu$ of
    $\mathcal E_A$. Block-diagonal observables evolve separately,
    with the $j$th block acted on by $w_j^2\mathcal E_{A^j}$.
\end{lemma}
\begin{proof}
    Multiply the actual canonical block matrices. The factors from a
    block and its adjoint give $w_j^2$; unitality gives the projection
    eigenvector. Embedding an arbitrary block eigenvector in that
    sector gives the remaining assertion.
\end{proof}

\begin{lemma}[Unique dominant sector from normalized generic C2]
    \label{lem:half_chain_c2_dominant_sector}
    \lean{MPSTensor.PGVWC07CanonicalFormData.exists_weight_eq_one,
        MPSTensor.PGVWC07CanonicalFormData.weight_eq_one_unique_of_simple_fixedEigenvalue,
        MPSTensor.PGVWC07CanonicalFormData.existsUnique_weight_eq_one_of_c2}
    \notready
    \uses{lem:half_chain_canonical_sector_eigenvalues}
    Suppose the actual canonical transfer map has spectral radius one
    and satisfies generic Condition C2, counting algebraic multiplicity:
    its only peripheral eigenvalue is $1$, and that root is simple.
    There is exactly one block $j$ with $w_j=1$.
\end{lemma}
\begin{proof}
    If every weight were less than one, the canonical diagonal formula
    would give $\|\mathcal E_A(I)\|<1$. Positivity implies
    $r(\mathcal E_A)\leq\|\mathcal E_A(I)\|$, contradicting the
    radius normalization. Two distinct unit-weight sector projections
    would be independent fixed vectors, contradicting algebraic
    simplicity of the root $1$.
\end{proof}

\begin{lemma}[Actual physical approximation by the dominant normal block]
    \label{lem:half_chain_c2_dominant_physical_limit}
    \lean{MPSTensor.PGVWC07CanonicalFormData.exists_dominantBlock_physical_limit_of_c2,
        MPSTensor.PGVWC07CanonicalFormData.isNormalTensor_of_weight_eq_one_of_c2,
        MPSTensor.PGVWC07CanonicalFormData.exists_dominantNormalBlock_physical_limit_of_c2}
    \notready
    \uses{lem:half_chain_c2_dominant_sector, lem:half_chain_weighted_remainder}
    Under the preceding canonical and normalized C2 hypotheses, the
    unique unit-weight block $A^j$ is normal and
    $\|\psi_N(A)-\psi_N(A^j)\|\to0$.
\end{lemma}
\begin{proof}
    Canonical unitality, the faithful dual fixed point, and scalar fixed
    points give irreducibility of the block. Its peripheral eigenvalues
    lift to those of the actual tensor, so C2 excludes every peripheral
    value other than one. Thus the block is normal. The unit-weight sum
    in (\ref{eq:half_chain_weighted_remainder}) is precisely this block.
\end{proof}

\begin{lemma}[General canonical C2 half-chain limit]%
    \label{thm:half_chain_general_c2_source}
    \notready
    For a TI MPS satisfying Condition C2, \cite[Theorem~6]{PerezGarcia2007Matrix}
    asserts convergence of the half-chain reduced spectrum to the doubled
    canonical fixed-point spectrum. As recorded in
    \cite{gap:pgvwc07_half_chain_spectrum_conventions}, extending

\end{lemma}
