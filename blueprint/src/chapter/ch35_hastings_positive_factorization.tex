\section{The positive version of Hastings' approximate factorization}
\label{sec:hastings_positive_factorization}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 246--267 (Lemma 1) and
% lines 912--943 (note added).

The note added to \cite[lines 912--943]{Hastings2007AreaLaw} replaces the
boundary operator $O_B$ of \cite[Lemma~1]{Hastings2007AreaLaw} by the positive
contraction $O_B^\dagger O_B$, with $P_0=|\Omega\rangle\langle\Omega|$ for a
unit vector $\Omega$ on the chain of $L$ sites.

\begin{definition}[Approximate factorization at a cut]
    \label{def:hastings_factorization_at}
    \lean{TNLean.MPS.AreaLaw.Hastings.IsFactorizationAt}
    \leanok
    \uses{def:chain_model}
    For $c,l\in\N$ and $\epsilon\in\R$, $(O_B,O_L,O_R)$ is an \emph{approximate
        factorization} at $(c,l)$ with error $\epsilon$ if $O_L,O_R$ are commuting
    orthogonal projections supported on $\lbrack0,c)$ and $\lbrack c,L)$, $O_B$ is supported
    on $\lbrack c-l,c+l)\cap\lbrack0,L)$ with $\Vert O_B\Vert\le1$, and
    \begin{align}
        \Vert O_BO_LO_R-P_0\Vert & \le\epsilon.
        \label{eq:hastings_factorization_at}
    \end{align}
\end{definition}

\begin{lemma}[Enlarging the error]
    \label{thm:hastings_factorization_mono}
    \lean{TNLean.MPS.AreaLaw.Hastings.IsFactorizationAt.mono}
    \leanok
    \uses{def:hastings_factorization_at}
    An approximate factorization at $(c,l)$ with error $\epsilon$ is also one
    with any error $\epsilon'\ge\epsilon$.
\end{lemma}
\begin{proof}\leanok
    Only~(\ref{eq:hastings_factorization_at}) depends on $\epsilon$.
\end{proof}

\begin{lemma}[Supports are closed under adjoints]
    \label{thm:hastings_supported_conj_transpose}
    \lean{Entropy.IsSupportedOn.conjTranspose}
    \leanok
    \uses{def:chain_model}
    If $A$ is supported on $S$, then $A^\dagger$ is supported on $S$.
\end{lemma}
\begin{proof}\leanok
    $(A^\dagger)_{x,y}=\overline{A_{y,x}}$ preserves the support condition on
    $x,y$.
\end{proof}

\begin{lemma}[The rank-one projection onto a unit vector]
    \label{thm:hastings_rank_one_projection}
    \lean{TNLean.MPS.AreaLaw.Hastings.isStarProjection_vecMulVec_of_norm_eq_one}
    \leanok
    For $\|\Omega\|=1$, $P_0=|\Omega\rangle\langle\Omega|$ satisfies
    $P_0^2=P_0=P_0^\dagger$.
\end{lemma}
\begin{proof}\leanok
    $P_0^\dagger=P_0$ and $P_0^2=\langle\Omega,\Omega\rangle P_0=P_0$.
\end{proof}

\begin{lemma}[An elementary bound]
    \label{thm:hastings_positivity_constant}
    \lean{TNLean.MPS.AreaLaw.Hastings.sqrt_one_sub_sq_add_le}
    \leanok
    For $0\le\epsilon\le1$,
    \begin{align}
        \sqrt{1-(1-\epsilon)^2}+3\epsilon+\epsilon^2 & \le(\sqrt2+4)\sqrt\epsilon.
        \label{eq:hastings_positivity_constant}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Use $1-(1-\epsilon)^2\le2\epsilon$ and $\epsilon^2\le\epsilon\le\sqrt\epsilon$.
\end{proof}

\begin{lemma}[The positive repair at one cut]
    \label{thm:hastings_positive_repair}
    \lean{TNLean.MPS.AreaLaw.Hastings.IsFactorizationAt.conjTranspose_mul_self}
    \leanok
    \uses{def:hastings_factorization_at}
    If $(O_B,O_L,O_R)$ is an approximate factorization at $(c,l)$ with error
    $\epsilon$, then $O_B^\dagger O_B\ge0$ and $(O_B^\dagger O_B,O_L,O_R)$ is
    an approximate factorization at $(c,l)$ with error
    $(\sqrt2+4)\sqrt\epsilon$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_supported_conj_transpose,
        thm:hastings_rank_one_projection, thm:hastings_positivity_constant}
    By Lemma~\ref{thm:hastings_supported_conj_transpose}, $O_B^\dagger O_B\ge0$
    has the same support as $O_B$ and $\Vert O_B^\dagger O_B\Vert=\Vert O_B\Vert^2\le1$.
    For $\epsilon\le1$, the note-added bound with $Q=O_LO_R$ and
    (\ref{eq:hastings_positivity_constant}) give
    \begin{align}
        \Vert O_B^\dagger O_BO_LO_R-P_0\Vert
        \le\sqrt{1-(1-\epsilon)^2}+3\epsilon+\epsilon^2
        \le(\sqrt2+4)\sqrt\epsilon,
        \notag
    \end{align}
    while for $\epsilon>1$,
    $\Vert O_B^\dagger O_BO_LO_R-P_0\Vert\le\Vert O_B^\dagger O_BO_LO_R\Vert+\Vert P_0\Vert\le2<(\sqrt2+4)\sqrt\epsilon$.
\end{proof}

\begin{lemma}[The positive version of Lemma~1]
    \label{thm:hastings_positive_factorization}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_posSemidef_factorization}
    \leanok
    \uses{def:hastings_factorization_at}
    If $\|\Omega\|=1$, $C_1\ge0$, $\xi'\in\R$, and there is an approximate
    factorization at $(c,l)$ with error $C_1e^{-l/\xi'}$, then there is one at
    $(c,l)$ with $O_B\ge0$ and error
    \begin{align}
        (\sqrt2+4)\sqrt{C_1}\,e^{-l/(2\xi')}.
        \label{eq:hastings_positive_factorization}
    \end{align}
    This is $O_B^+$ of \cite[lines 912--943]{Hastings2007AreaLaw}, halving the
    rate of Lemma~1~\cite{gap:hastings07_lemma1_positivity_rate}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:hastings_positive_repair}
    Apply Lemma~\ref{thm:hastings_positive_repair} with
    $\sqrt{C_1e^{-l/\xi'}}=\sqrt{C_1}\,e^{-l/(2\xi')}$.
\end{proof}

\begin{lemma}[The factorization in the form used by the bootstrap]
    \label{thm:hastings_factorization_bootstrap_form}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_factorization_of_isFactorizationAt}
    \leanok
    \uses{def:hastings_factorization_at}
    An approximate factorization at $(j,m)$ with error $\epsilon$ has $O_B$
    supported on $\lbrack0,j+m)$, $O_L$ on $\lbrack0,j)$, and $O_R$ on $\lbrack j,L)$
    satisfying~(\ref{eq:hastings_factorization_at}), supplying the scale-$m$
    hypothesis of Lemma~\ref{thm:hastings_bootstrap_scale}.
\end{lemma}
\begin{proof}\leanok
    Since $\lbrack j-m,j+m)\cap\lbrack0,L)\subseteq\lbrack0,j+m)\cap\lbrack0,L)$, support on
    $\lbrack j-m,j+m)\cap\lbrack0,L)$ implies support on $\lbrack0,j+m)\cap\lbrack0,L)$.
\end{proof}
