\section{The non-anomalous \texorpdfstring{$\Z_3$}{Z3} clock symmetry}%
\label{sec:asymex_z3clock}

The anomalous symmetry of Section~\ref{sec:asymex_z3anomalous} is an exact
representation of $\Z_3$, and its condensation defect satisfies
$A^2=3A$ at every length
(Theorem~\ref{thm:asymex_z3anomalous_defect_square}). This section
treats a second family of $\Z_3$ symmetry operators whose representatives
multiply only up to phases depending on the length, and shows that the
structure constants of its condensation defect then genuinely depend on the
length. The symmetry is not anomalous: its Else--Nayak class is trivial, a
fact checked in exact arithmetic and not formalized here. The length
dependence is therefore not a signature of an anomaly; it comes from the
per-site phases of the chosen representatives.

As in Section~\ref{sec:asymex_z3anomalous}, let
$\omega=\exp(2\pi\mathrm{i}/3)$, let $X\ket{s}=\ket{s+1}$ on a qutrit, and
let $CZ\ket{a,b}=\omega^{ab}\ket{a,b}$ and $Z\ket{a}=\omega^a\ket{a}$. On a
periodic chain of $N>0$ qutrits the clock operators are
\begin{align}
    U_N & =X^{\otimes N}\prod_{k}CZ_{k,k+1},                    &
    V_N & =(X^{2})^{\otimes N}\prod_{k}CZ^{2}_{k,k+1}Z^{2}_{k}.
\end{align}

\begin{definition}[The clock tensors]
    \label{def:asymex_z3clock_tensors}
    \lean{Z3Clock.phaseChar, Z3Clock.clockTensor, Z3Clock.uTensor, Z3Clock.vTensor}
    \leanok
    For $g\in\Z_3$ and a weight $f\colon\Z_3\times\Z_3\to\Z_3$, the clock
    tensor of bond dimension three has the entries
    \begin{align}
        \bigl(T_{g,f}^{o,i}\bigr)_{l,r} & =
        \begin{cases}
            \omega^{f(l,i)} & \text{if } o=i+g \text{ and } r=i, \\
            0               & \text{otherwise},
        \end{cases}
    \end{align}
    with output label $o$, input label $i$ and bond labels $l,r$. The
    tensor $U$ is $T_{1,f_U}$ with $f_U(l,i)=li$, and the tensor $V$ is
    $T_{2,f_V}$ with $f_V(l,i)=2li+2i$.
\end{definition}

\begin{theorem}[Kernels and the trivial cocycle]
    \label{thm:asymex_z3clock_kernels}
    \lean{Z3Clock.mpo_clockTensor, Z3Clock.mpo_uTensor_apply, Z3Clock.mpo_vTensor_apply,
        Z3Clock.clockTensor_apply_eq_cocycle, Z3Clock.mpo_clockTensor_eq_cocycle,
        Z3Clock.mpo_uTensor_eq_cocycle, Z3Clock.mpo_vTensor_eq_cocycle,
        Z3Clock.clockTensor_isMPUPos}
    \leanok
    \uses{def:asymex_z3clock_tensors, def:asymex_group_cocycle_tensor,
        def:asymex_group_cocycle_cyclic, thm:asymex_group_cocycle_kernel}
    On a periodic chain of $N>0$ qutrits, the periodic operator of
    $T_{g,f}$ is monomial:
    \begin{align}
        \bra{s}O_N(T_{g,f})\ket{t} & =
        \prod_k\delta_{s_k,t_k+g}\,\omega^{f(t_k,t_{k+1})}.
    \end{align}
    In particular the periodic operators of $U$ and $V$ are $U_N$ and
    $V_N$. The tensor $T_{g,f}$ is, entrywise, the group tensor of
    Definition~\ref{def:asymex_group_cocycle_tensor} for the trivial
    cocycle $\omega_0=1$ of $\Z_3$ times the bond weight $\omega^{f(l,i)}$,
    so that $O_N(T_{g,f})$ is the plain shift by $g$ followed by the
    diagonal circuit with phase $\omega^{\sum_k f(t_k,t_{k+1})}$ on the input
    configuration $t$. Every clock tensor is a matrix product unitary.
\end{theorem}

\begin{proof}
    \leanok
    The tensor $T_{g,f}$ forces its right bond to carry the input label,
    so on a closed chain the only surviving bond configuration is
    $b_k=t_{k-1}$, which gives the kernel. With the trivial cocycle the
    group tensor has entry one exactly where $T_{g,f}$ is nonzero, and its
    periodic operator is the shift by $g$. A permutation matrix times a
    diagonal matrix of unit-modulus phases is unitary.
\end{proof}

\begin{theorem}[Multiplication with length-dependent phases]
    \label{thm:asymex_z3clock_table}
    \lean{Z3Clock.mpo_clockTensor_mul, Z3Clock.mpo_u_mul_u, Z3Clock.mpo_u_mul_v,
        Z3Clock.mpo_v_mul_u, Z3Clock.mpo_v_mul_v, Z3Clock.mpo_u_mul_u_mul_u}
    \leanok
    \uses{thm:asymex_z3clock_kernels}
    On every periodic chain of $N>0$ qutrits,
    \begin{align}
        U_N^2  & =\omega^N V_N,       &
        U_NV_N & =V_NU_N=\omega^N\Id, &
        V_N^2  & =U_N,                &
        U_N^3  & =\omega^{2N}\Id.
    \end{align}
    When $3\mid N$ these are the relations of $\Z_3$; otherwise the
    representatives multiply only up to the phases $\omega^N$.
\end{theorem}

\begin{proof}
    \leanok
    The product of $O_N(T_{g,f})$ and $O_N(T_{h,f'})$ is the shift by
    $g+h$ with phase
    $\omega^{\sum_k (f(t_k+h,t_{k+1}+h)+f'(t_k,t_{k+1}))}$. If
    $f(a+h,b+h)+f'(a,b)=f''(a,b)+c+\kappa(a)-\kappa(b)$ for all $a,b$, the
    terms $\kappa(t_k)-\kappa(t_{k+1})$ cancel around the chain and the
    product is $\omega^{cN}O_N(T_{g+h,f''})$. For $U\cdot U$ one has
    $(a+1)(b+1)+ab=(2ab+2b)+1+(a-b)$; for $U\cdot V$ and $V\cdot U$ the
    left side is $2a+b+1=0+1+(2a-2b)$ modulo three; for $V\cdot V$ it is
    $ab+(a-b)$. These are checked on the nine pairs $(a,b)$.
\end{proof}

\begin{theorem}[The condensation defect of an exact group representation]
    \label{thm:asymex_group_cocycle_defect}
    \lean{MPOTensor.GroupCocycle.isMPOFusionAlgebra, MPOTensor.GroupCocycle.sum_mpo_mul_sum_mpo,
        Z3Clock.cocycleDefect, Z3Clock.cocycleDefect_mul_self}
    \leanok
    \uses{thm:asymex_group_cocycle_mul, def:mpo_fusion_algebra}
    For every finite group $G$ and every three-cocycle, the periodic
    operators $U_g$ of Definition~\ref{def:asymex_group_cocycle_tensor}
    form a matrix product operator fusion algebra with structure constants
    $N_{gh}^k=\delta_{k,gh}$, and $A_0=\sum_g U_g$ satisfies $A_0^2=\lvert G\rvert A_0$
    at every positive length. For $\Z_3$ and the trivial cocycle,
    $A_0^2=3A_0$.
\end{theorem}

\begin{proof}
    \leanok
    The first claim restates $U_gU_h=U_{gh}$. For the second,
    $A_0^2=\sum_g\sum_h U_{gh}$, and for each $g$ the map $h\mapsto gh$
    is a bijection of $G$, so every term $U_k$ occurs $\lvert G\rvert$ times.
\end{proof}

\begin{definition}[The clock condensation defect]
    \label{def:asymex_z3clock_defect}
    \lean{Z3Clock.defectTensor, Z3Clock.mpo_defectTensor}
    \leanok
    \uses{def:asymex_z3clock_tensors}
    The defect $A=\delta\oplus U\oplus V$ is the block-diagonal tensor of
    bond dimension $1+3+3$ whose blocks are the identity tensor $\delta$ and
    the clock tensors $U$ and $V$. Its periodic operator is
    $O_N(A)=\Id+U_N+V_N$.
\end{definition}

\begin{theorem}[Length-dependent fusion of the clock defect]
    \label{thm:asymex_z3clock_defect_square}
    \lean{Z3Clock.mpo_defect_mul_defect, Z3Clock.eq_of_smul_add_smul_add_smul_eq,
        Z3Clock.defect_coeff_eq, Z3Clock.mpo_defect_mul_defect_eq_three_smul_iff,
        Z3Clock.not_exists_const_defect_coeff}
    \leanok
    \uses{def:asymex_z3clock_defect, thm:asymex_z3clock_table}
    On every periodic chain of length $L>0$,
    \begin{align}
        O_L(A)^2 & =(1+2\omega^L)\,\Id+3\,U_L+(2+\omega^L)\,V_L,
    \end{align}
    and the operators $\Id$, $U_L$ and $V_L$ are linearly independent, so
    these coefficients are the unique ones. The coefficients are the power
    sums of the weight multisets $\{1,\omega,\omega\}$, $\{1,1,1\}$ and
    $\{1,1,\omega\}$. In particular $O_L(A)^2=3\,O_L(A)$ if and only if
    $3\mid L$, and there are no coefficients independent of $L$ with
    $O_L(A)^2=a\,\Id+b\,U_L+c\,V_L$ for every $L>0$.
\end{theorem}

\begin{proof}
    \leanok
    Expanding $(\Id+U_L+V_L)^2$ with
    Theorem~\ref{thm:asymex_z3clock_table} gives
    $\Id+3U_L+2V_L+\omega^LV_L+2\omega^L\Id$. The three operators shift the
    constant configuration $0$ to the constant configurations $0$, $1$ and
    $2$ respectively, each with phase one, so reading a combination of them
    in these three entries recovers its coefficients. The coefficient of
    $V_L$ is $2+\omega^L$, which equals $3$ exactly when $\omega^L=1$, that
    is, when $3\mid L$; comparing the lengths $1$ and $3$ would force
    $\omega=1$.
\end{proof}

\begin{corollary}[The clock family is not a fusion algebra]
    \label{cor:asymex_z3clock_not_fusion_algebra}
    \lean{Z3Clock.repBondDim, Z3Clock.repTensor, Z3Clock.not_isMPOFusionAlgebra}
    \leanok
    \uses{thm:asymex_z3clock_defect_square, def:mpo_fusion_algebra}
    For no nonnegative integers $N_{ab}^c$, independent of the length, do
    the periodic operators of $\delta$, $U$ and $V$ satisfy
    $O_L(T_a)O_L(T_b)=\sum_cN_{ab}^cO_L(T_c)$ at every positive length.
    In contrast, the group operators of the trivial cocycle form such an
    algebra (Theorem~\ref{thm:asymex_group_cocycle_defect}).
\end{corollary}

\begin{proof}
    \leanok
    Such constants would give $\omega^LV_L=U_L^2=N_{11}^0\Id+N_{11}^1U_L+N_{11}^2V_L$,
    hence $\omega^L=N_{11}^2$ for every $L>0$ by linear independence, and
    the lengths $1$ and $3$ give $\omega=1$.
\end{proof}

\begin{remark}[What remains]
    \label{rem:asymex_z3clock_scope}
    The stacked square $A\cdot A$ has bond dimension $49$ and, by the
    verification record, the block triangular form of
    Theorem~\ref{thm:asym_multiblock} with nine targets and $28$
    one-dimensional zero diagonal blocks. That compression is not
    formalized; the fusion identity above is proved directly at the level
    of periodic operators.
\end{remark}
