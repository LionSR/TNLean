\section{The energy of the final state of Hastings' hierarchical construction}
\label{sec:hastings05_hierarchical_energy}

% Hastings, arXiv:cond-mat/0508554v3, proj3new.tex, lines 705--720 (eq. hchy).

At the end of the hierarchical construction,
\cite[eq.~(hchy), lines 709--720]{Hastings2006Solving} bounds the energy of a
final state from the final deficiency $b_h$. Below we first bound the energy of
the normalized projection of the ground state from its norm deficiency
(Section~\ref{sec:hastings05_hierarchical_projection}), and then prove
eq.~(hchy) for every final state directly from the spectral windows, both for
the open chain and (with one additional window on the closing link) for the
periodic chain \cite{gap:hastings05_hierarchical_energy}.

\begin{theorem}[Energies of nearby unit vectors]
    \label{thm:hastings05_energy_nearby}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_sub_re_inner_le}
    \leanok
    For $\|\phi\|=\|\psi\|=1$ and any operator $M$,
    $|\operatorname{Re}\langle\phi,M\phi\rangle-\operatorname{Re}\langle\psi,M\psi\rangle|
        \le2\|M\|\,\|\phi-\psi\|$.
\end{theorem}
\begin{proof}
    \leanok
    Apply Cauchy--Schwarz to
    $\langle\phi,M\phi\rangle-\langle\psi,M\psi\rangle
        =\langle\phi-\psi,M\phi\rangle+\langle\psi,M(\phi-\psi)\rangle$.
\end{proof}

\begin{theorem}[Normalizing the image under one projection]
    \label{thm:hastings05_normalization_one}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.norm_inv_norm_smul_toEuclideanLin_sub_le_two_mul}
    \leanok
    For an orthogonal projection $Q$ and $\|\psi\|=1$, if
    $\|Q\psi-\psi\|\le b$, then $\|Q\psi/\|Q\psi\|-\psi\|\le2b$.
\end{theorem}
\begin{proof}
    \leanok
    Use $|1-\|Q\psi\||\le\|Q\psi-\psi\|\le b$.
\end{proof}

\begin{theorem}[A final state of nearly the ground-state energy]
    \label{thm:hastings05_hchy}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.exists_mem_range_re_inner_le_add,
        TNLean.MPS.AreaLaw.LocalSolution.exists_mem_range_re_inner_sum_le_add}
    \leanok
    \uses{thm:hastings05_energy_nearby, thm:hastings05_normalization_one}
    If an orthogonal projection $Q$ and $\|\Psi\|=1$ satisfy
    $\|Q\Psi-\Psi\|\le b<1$, then $\phi=Q\Psi/\|Q\Psi\|$ satisfies $\|\phi\|=1$,
    $Q\phi=\phi$, $\|\phi-\Psi\|\le2b$, and for every $M$,
    \begin{align}
        \operatorname{Re}\langle\phi,M\phi\rangle
        \le\operatorname{Re}\langle\Psi,M\Psi\rangle+4\|M\|b
        \label{eq:hastings05_hchy}
    \end{align}
    (at most $\operatorname{Re}\langle\Psi,M\Psi\rangle+4NJb$ when
    $M=\sum_{i<N}M_i$ with $\|M_i\|\le J$)
    \cite[eq.~(hchy), lines 709--720]{Hastings2006Solving},
    \cite{gap:hastings05_hierarchical_energy}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:hastings05_energy_nearby, thm:hastings05_normalization_one}
    Since $\|Q\Psi\|\ge1-b>0$, combine
    Theorems~\ref{thm:hastings05_normalization_one}
    and~\ref{thm:hastings05_energy_nearby}.
\end{proof}

\subsection*{Every final state}

\begin{theorem}[A vector kept by a window]
    \label{thm:hastings05_window_expectation}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_sub_le_of_windowProjection,
        TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_sub_le_of_compressedWindowProjection}
    \leanok
    \uses{def:hastings05_compressed_window}
    Let $c\in\mathbb R$ and $x\ge0$. If $A=A^\dagger$ and
    $P_x(A,c)\phi=\phi$, then
    $|\operatorname{Re}\langle\phi,A\phi\rangle-c\|\phi\|^2|\le x\|\phi\|^2$.
    If $M=M^\dagger$, $Q$ is an orthogonal projection, and
    $QP_x(QMQ,c)\phi=\phi$, then $Q\phi=\phi$ and
    $|\operatorname{Re}\langle\phi,M\phi\rangle-c\|\phi\|^2|\le x\|\phi\|^2$.
\end{theorem}
\begin{proof}\leanok
    For $P_x(A,c)\phi=\phi$,
    $|\langle\phi,(A-c)\phi\rangle|\le\|\phi\|\,\|(A-c)\phi\|\le x\|\phi\|^2$;
    for $QP_x(QMQ,c)\phi=\phi$, $Q\phi=\phi$ and
    $\langle\phi,QMQ\phi\rangle=\langle\phi,M\phi\rangle$.
\end{proof}

\begin{definition}[Sums over the tree of blocks]
    \label{def:hastings05_hierarchy_sum}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.hierarchySum}
    \leanok
    For a family $f_k^m$ indexed by level $m$ and block $k$, define
    $T_k^0(f)=0$ and $T_k^{m+1}(f)=f_k^m+T_{2k}^m(f)+T_{2k+1}^m(f)$.
\end{definition}

\begin{theorem}[The joining operators below a block]
    \label{thm:hastings05_hierarchy_sum_expectation}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_hierarchySum_sub_le}
    \leanok
    \uses{thm:hastings05_window_expectation, def:hastings05_hierarchy_sum,
        thm:hastings05_hierarchy_levels}
    In the setting of Theorem~\ref{thm:hastings05_hierarchy_levels} with radii
    $x_m\ge0$, every $P_k^m$ is an orthogonal projection, and any $\phi$ with
    $P_k^m\phi=\phi$ satisfies
    \begin{align}
        \bigl|\operatorname{Re}\langle\phi,T_k^m(M)\phi\rangle
        -T_k^m(c)\|\phi\|^2\bigr|\le T_k^m(x)\|\phi\|^2.
        \label{eq:hastings05_hierarchy_sum_expectation}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Induct on $m$: $P_k^{m+1}\phi=\phi$ implies $P_{2k}^m\phi=P_{2k+1}^m\phi=\phi$,
    and Theorem~\ref{thm:hastings05_window_expectation} bounds the $M_k^m$ term.
\end{proof}

\begin{theorem}[Eq.~(hchy) for every final state]
    \label{thm:hastings05_hchy_every_state}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.re_inner_hierarchySum_le_add}
    \leanok
    \uses{thm:hastings05_hierarchy_sum_expectation}
    In the setting of Theorem~\ref{thm:hastings05_hierarchy_sum_expectation}
    with $c_k^m=\operatorname{Re}\langle\Psi,M_k^m\Psi\rangle$ for $\|\Psi\|=1$,
    every unit vector $\phi$ with $P_k^m\phi=\phi$ satisfies
    \begin{align}
        \operatorname{Re}\langle\phi,T_k^m(M)\phi\rangle
        \le\operatorname{Re}\langle\Psi,T_k^m(M)\Psi\rangle+T_k^m(x)
        \label{eq:hastings05_hchy_every_state}
    \end{align}
    \cite[eq.~(hchy), lines 709--720]{Hastings2006Solving},
    \cite{gap:hastings05_hierarchical_energy}.
\end{theorem}
\begin{proof}\leanok
    Since $T_k^m(c)=\operatorname{Re}\langle\Psi,T_k^m(M)\Psi\rangle$, apply
    (\ref{eq:hastings05_hierarchy_sum_expectation}) with $\|\phi\|=1$.
\end{proof}

\begin{theorem}[At most $2n$ final states]
    \label{thm:hastings05_final_state_count_hierarchy}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.card_le_two_mul_re_trace_of_hierarchy}
    \leanok
    \uses{thm:hastings05_hchy_every_state, thm:hastings_final_state_count}
    In the setting of Theorem~\ref{thm:hastings05_hchy_every_state}, if
    $H=T_0^h(M)$ satisfies $H-E_0\ge\Delta E(1-P)$ for an orthogonal projection
    $P$ and $\Delta E>0$, and
    $\operatorname{Re}\langle\Psi,H\Psi\rangle+T_0^h(x)\le E_0+\Delta E/2$,
    then any orthonormal family kept by $P_0^h$ has size at most
    $2\operatorname{Re}\operatorname{tr}P$
    \cite[lines 722--730]{Hastings2006Solving}.
\end{theorem}
\begin{proof}\leanok
    Combine (\ref{eq:hastings05_hchy_every_state}) with
    Theorem~\ref{thm:hastings_final_state_count}.
\end{proof}

For the periodic chain of supersites, one additional spectral window controls
the closing link $(N,1)$ \cite{gap:hastings05_hierarchical_energy}.

\begin{theorem}[The window of the closing link]
    \label{thm:hastings05_closing_link_window}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.norm_sub_closingLinkWindow_le}
    \leanok
    \uses{thm:hastings05_hierarchy_levels, thm:hastings05_level_against_ground_state}
    In the setting of Theorem~\ref{thm:hastings05_hierarchy_levels}, if
    $W=W^\dagger$ satisfies $\|(W-c_W)\psi\|\le\epsilon$ and $\|W-c_W\|\le K$,
    then $R=P_0^hP_{x_h}(P_0^hWP_0^h,c_W)$ is an orthogonal projection with
    $\|\psi-R\psi\|\le D_{h+1}$.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:hastings05_level_against_ground_state} to the pair of
    commuting projections $(P_0^h,1)$ with deficiency $D_h$.
\end{proof}

\begin{theorem}[Eq.~(hchy) for the periodic chain]
    \label{thm:hastings05_hchy_periodic}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.re_inner_hierarchySum_add_le_add}
    \leanok
    \uses{thm:hastings05_hchy_every_state, thm:hastings05_window_expectation}
    In the setting of Theorem~\ref{thm:hastings05_hchy_every_state}, for
    $W=W^\dagger$, $x_W\ge0$, and
    $R=P_0^hP_{x_W}(P_0^hWP_0^h,\operatorname{Re}\langle\Psi,W\Psi\rangle)$,
    every unit vector $\phi$ with $R\phi=\phi$ satisfies
    \begin{align}
        \operatorname{Re}\langle\phi,(T_0^h(M)+W)\phi\rangle
        \le\operatorname{Re}\langle\Psi,(T_0^h(M)+W)\Psi\rangle+T_0^h(x)+x_W
        \label{eq:hastings05_hchy_periodic}
    \end{align}
    \cite[eq.~(hchy), lines 709--720]{Hastings2006Solving},
    \cite{gap:hastings05_hierarchical_energy}.
\end{theorem}
\begin{proof}\leanok
    Combine Theorem~\ref{thm:hastings05_window_expectation} for $R$ (giving
    $P_0^h\phi=\phi$ and
    $\operatorname{Re}\langle\phi,W\phi\rangle\le\operatorname{Re}\langle\Psi,W\Psi\rangle+x_W$)
    with (\ref{eq:hastings05_hchy_every_state}).
\end{proof}

\begin{theorem}[At most $2n$ final states of the periodic chain]

\subsection*{Constructing the projections and using the physical gap}

\begin{definition}[The projections determined by the pair operators]
    \label{def:hastings05_constructed_hierarchy}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.hierarchyProjection,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchyProjection_zero,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchyProjection_succ,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchyClosingProjection}
    \leanok
    \uses{def:hastings05_compressed_window}
    Given Hermitian operators $M_k^m$, real centres $c_k^m$ and radii $x_m$,
    set $P_k^0=1$. Define $P_k^{m+1}$ as the spectral window of
    $M_k^m$ at $c_k^m$ and radius $x_m$, inside the range of
    $P_{2k}^mP_{2k+1}^m$. For a closing operator $W$, let $R$ be its
    window at $c_W$ and radius $x_h$ inside the range of $P_0^h$.
\end{definition}

\begin{theorem}[Deficiency of the constructed projections]
    \label{thm:hastings05_constructed_deficiency}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.norm_sub_hierarchyProjection_le,
        TNLean.MPS.AreaLaw.LocalSolution.exists_unit_mem_hierarchyProjection,
        TNLean.MPS.AreaLaw.LocalSolution.norm_sub_hierarchyClosingProjection_le,
        TNLean.MPS.AreaLaw.LocalSolution.exists_unit_mem_hierarchyClosingProjection}
    \leanok
    \uses{def:hastings05_constructed_hierarchy, thm:hastings05_supersite_hierarchy,
        thm:hastings05_closing_link_window, thm:hastings05_hchy}
    Suppose $M_k^m$ acts on its block of $2^{m+1}$ supersites, all radii are
    positive, and $\psi$ is a unit vector with
    $\|(M_k^m-c_k^m)\psi\|\le\epsilon$ and $\|M_k^m-c_k^m\|\le K$.
    Then $P_k^m$ is an orthogonal projection and
    $\|\psi-P_k^m\psi\|\le D_m$. If $W$ satisfies the same two estimates,
    $R$ is an orthogonal projection and $\|\psi-R\psi\|\le D_{h+1}$.
    Whenever the relevant deficiency $D$ is less than one, that projection
    retains a unit vector $\phi$ with $\|\phi-\psi\|\le2D$.
\end{theorem}
\begin{proof}\leanok
    The projections of disjoint blocks commute. Apply the deficiency
    recurrence to the recursively defined windows, and normalize the image
    of $\psi$ under the final projection.
\end{proof}

\begin{definition}[A state retained throughout a tree of blocks]
    \label{def:hastings05_retained_tree}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.IsRetainedHierarchy}
    \leanok
    A vector is retained throughout a block of level zero if its projection
    keeps it. It is retained throughout a block of level $m+1$ if its
    projection keeps it and it is retained throughout both child blocks.
    This describes membership in the nested local spectral spaces of the
    hierarchical construction in~\cite[lines 668--704]{Hastings2006Solving}.
\end{definition}

\begin{theorem}[The closing window preserves the hierarchy]
    \label{thm:hastings05_retained_tree}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.toEuclideanLin_eq_of_mul_eq,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchyProjection_children,
        TNLean.MPS.AreaLaw.LocalSolution.isRetainedHierarchy_of_hierarchyProjection,
        TNLean.MPS.AreaLaw.LocalSolution.isRetainedHierarchy_of_hierarchyClosingProjection}
    \leanok
    \uses{def:hastings05_retained_tree, def:hastings05_constructed_hierarchy,
        thm:hastings05_supersite_hierarchy}
    For the constructed hierarchy, a vector kept by a block projection is
    retained throughout that block. Every vector kept by the additional
    closing window is retained throughout the original tree.
    Thus the periodic correction changes the retained space at the root
    while preserving all retained spaces below it. It has the tree-shaped
    representation called a generalized matrix product state in
    \cite[lines 596--607 and 732--743]{Hastings2006Solving}. No bound on the
    dimensions of the intermediate retained spaces is asserted there.
\end{theorem}
\begin{proof}\leanok
    If $PQ=Q$ and $Q\phi=\phi$, then $P\phi=PQ\phi=\phi$.
    The parent window is contained in the range of the product of its
    commuting child projections, so both children keep the vector.
    Induction gives every node below the parent. The closing window is
    contained in the final root projection, so the same induction applies.
\end{proof}

\begin{definition}[The joining edge of a block]
    \label{def:hastings05_hierarchy_edge}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.hierarchyEdge}
    \leanok
    For edge operators $a_j$, where $a_j$ joins supersites $j-1$ and $j$,
    set $M_k^m=a_{(2k+1)2^m}$. Edge zero is reserved for the closing link.
\end{definition}

\begin{theorem}[Each internal edge occurs once]
    \label{thm:hastings05_hierarchy_edges}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.hierarchySum_hierarchyEdge,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchySum_hierarchyEdge_add_zero,
        TNLean.MPS.AreaLaw.LocalSolution.hierarchyEdge_mem_supportedOperators}
    \leanok
    \uses{def:hastings05_hierarchy_edge, def:hastings05_hierarchy_sum}
    The sum below block $k$ of level $m$ is
    $T_k^m(M)=\sum_{k2^m<j<(k+1)2^m}a_j$. In particular,
    $T_0^h(M)+a_0=\sum_{0\le j<2^h}a_j$.
    If each positive-index edge acts on its two adjacent supersites,
    $M_k^m$ acts on block $k$ of level $m+1$.
\end{theorem}
\begin{proof}\leanok
    Split the interval of internal edges into its two halves and their
    joining edge. Both endpoints of that edge belong to the parent block.
\end{proof}

\begin{theorem}[The physical Hamiltonian controls the count]
    \label{thm:hastings05_physical_gap_count}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_sub_le_norm_sub,
        TNLean.MPS.AreaLaw.LocalSolution.abs_re_inner_sub_le_of_norm_sub_le,
        TNLean.MPS.AreaLaw.LocalSolution.card_le_two_mul_re_trace_of_approximation}
    \leanok
    \uses{thm:hastings_final_state_count}
    Let $H-E_0\ge\Delta(1-P_0)$ with $\Delta>0$ and $P_0$ an orthogonal
    projection. Let $\psi$ be a unit vector of energy $E_0$ under $H$.
    Suppose $\|H-T\|\le\delta$ and an orthonormal family $\phi_j$ satisfies
    $\operatorname{Re}\langle\phi_j,T\phi_j\rangle
        \le\operatorname{Re}\langle\psi,T\psi\rangle+\eta$.
    If $2\delta+\eta\le\Delta/2$, the family has at most
    $2\operatorname{Re}\operatorname{tr}P_0$ members.
\end{theorem}
\begin{proof}\leanok
    For every vector $v$,
    $|\operatorname{Re}\langle v,(H-T)v\rangle|
        \le\|H-T\|\|v\|^2$. Apply this to $\psi$ and each $\phi_j$ to obtain
    $\operatorname{Re}\langle\phi_j,H\phi_j\rangle
        \le E_0+2\delta+\eta\le E_0+\Delta/2$.
    Theorem~\ref{thm:hastings_final_state_count} gives the count.
\end{proof}

\begin{theorem}[The quantitative periodic construction]
    \label{thm:hastings05_quantitative_periodic}
    \lean{TNLean.MPS.AreaLaw.LocalSolution.exists_unit_mem_periodicHierarchyProjection,
        TNLean.MPS.AreaLaw.LocalSolution.card_le_two_mul_re_trace_of_periodicHierarchyProjection,
        TNLean.MPS.AreaLaw.LocalSolution.exists_unit_mem_periodicHierarchyProjection_of_finite,
        TNLean.MPS.AreaLaw.LocalSolution.card_le_two_mul_re_trace_of_periodicHierarchyProjection_of_finite}
    \leanok
    \uses{def:hastings05_periodic_projection,
        thm:hastings05_constructed_deficiency, thm:hastings05_physical_gap_count,
        thm:hastings05_supersite_final_states, thm:hastings05_explicit_projection_length}
    Let $h\ge2$, $J,\Delta,K>0$, and
    $r\ge512h^2(1+\max\{0,\log(J/\Delta)\})$.
    Suppose the positive-index Hermitian edges satisfy the two-supersite support condition,
    $\psi$ is a unit vector, and, with
    $c_j=\operatorname{Re}\langle\psi,a_j\psi\rangle$,
    $\|(a_j-c_j)\psi\|\le K e^{-r}$ and $\|a_j-c_j\|\le K$.
    Then $R$ retains a unit vector $\phi$ satisfying
    $\|\phi-\psi\|\le2(3^{h+1}-1)e^{-r/8}$.
    A finite family of $N\le2^h$ edges satisfies the same assertion after
    extension by zero.
    If in addition $K\le Jr$,
    $\|H-\sum_{0\le j<2^h}a_j\|\le2^hJe^{-r}$,
    $\psi$ has ground energy $E_0$ under $H$, and
    $H-E_0\ge\Delta(1-P_0)$ for an orthogonal projection $P_0$,
    every orthonormal family in the range of $R$
    has at most $2\operatorname{Re}\operatorname{tr}P_0$ members.
\end{theorem}
\begin{proof}\leanok
    The chosen length makes $D_{h+1}<1$ and bounds it by
    $(3^{h+1}-1)e^{-r/8}$. Normalize $R\psi$ for the first assertion.
    The total radius is at most $\Delta/4$, and the truncation error is at
    most $\Delta/8$. Apply Theorem~\ref{thm:hastings05_physical_gap_count}
    to the energy estimate for every vector kept by $R$.
\end{proof}
