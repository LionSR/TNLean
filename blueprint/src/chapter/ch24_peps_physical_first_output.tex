\section{Physical output of the replacement}
\label{sec:peps_physical_first_output}

The source replacement of
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS} has at every party
% Source: 04-compression.tex, lines 199--227.
$p$ the output space $\mathcal A_p\otimes\mathcal Y_p$, where $\mathcal A_p$
contains the retained auxiliary registers of $p$ and $\mathcal Y_p$ is the
original output space of $p$. The original output is read out by a
contraction $K$ into physical coordinates $\mathcal H$ and discarded
coordinates $\mathcal E$. The auxiliary registers are read out together with
the discarded coordinates; no tensor factor is exchanged.

\begin{definition}[Readout of the replacement]
    \label{def:peps_physical_first_replacement}
    \uses{thm:peps_circuit_effect_error}
    \lean{QuantumCircuit.Party.extendedReadout,
        QuantumCircuit.Party.EffectCircuit.replacementPhysicalDensity}
    \leanok
    If $R$ is the constructed source-only replacement, with output space
    $\bigotimes_p(\mathcal A_p\otimes\mathcal Y_p)=\mathcal A\otimes\mathcal Y$,
    the extended readout $K\otimes\id_{\mathcal A}$ maps it to
    $\mathcal H\otimes(\mathcal E\otimes\mathcal A)$, and
    \begin{align}
        R_{\rm phys} & =(K\otimes\id_{\mathcal A})\,R.
        \label{eq:peps_physical_first_exchange}
    \end{align}
    For an input $\psi$, the physical state of the replacement is the partial
    trace over $\mathcal E\otimes\mathcal A$ of
    $\ket{R_{\rm phys}\psi}\bra{R_{\rm phys}\psi}$.
\end{definition}

\begin{lemma}[Exact sources recover the replacement physical state]
    \label{thm:peps_physical_first_exact_sources}
    \lean{QuantumCircuit.Party.SourceCircuit.physicalDensity_pure_input}
    \lean{QuantumCircuit.Party.SourceCircuit.physicalSourceReplacedDensity_exact}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le_half_of_count_le}
    \leanok
    \uses{thm:peps_source_only_density_error}
    Let $W$ be an original circuit with at most $M$ nonprivate gates,
    satisfying the original expansion bounds. Set
    $\delta=\varepsilon/(8\max\{1,M\})$ for $\varepsilon>0$, and
    construct its replacement $R$ with accuracy parameter $\delta$. For a pure input
    $\psi$ with $\|\psi\|\le1$, let $\rho_W$ be the original physical
    output state. In the source-replacement expression for $R_{\rm phys}$,
    insert the exact operator of each original pair source. The resulting
    physical operator is the replacement state $\rho_R$, and
    $\|\rho_R-\rho_W\|_1\le\varepsilon/2$.
    The physical coordinates may be the labelled tensor coordinates with
    an arbitrary original local dimension $d_p$ at each party $p$.
\end{lemma}

\begin{proof}\leanok
    Exact source operators recover the output operator of the source-only
    circuit. The physical density of a pure input is the partial trace of the
    outer product of its output vector, so the physical partial trace of
    this operator is $\rho_R$, and applying
    Lemma~\ref{thm:peps_source_only_density_error} at the original
    gate-count bound $M$ gives the trace-norm bound.
\end{proof}
