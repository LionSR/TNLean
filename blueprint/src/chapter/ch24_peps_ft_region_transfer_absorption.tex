\begin{lemma}[Region vertex product across an inserted site]%
    \label{thm:peps_prod_region_insert_split}
    \lean{TNLean.PEPS.prod_region_insert_split}
    \leanok
    For $v\notin R$, any global virtual configuration $\zeta$, and any physical configuration $\sigma$ on $R\cup\{v\}$,
    \begin{align}
        \prod_{w\in R\cup\{v\}}A_w(\zeta|_{\inc(w)},\sigma_w)
         & =
        A_v(\zeta|_{\inc(v)},\sigma_v)
        \prod_{w\in R}A_w(\zeta|_{\inc(w)},\sigma|_R{}_w).
        \label{eq:peps_prod_region_insert_split}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Split off the factor at $v\notin R$ and reindex the remaining product over $R$.
\end{proof}

\begin{lemma}[One-site split of the blocked-region weight]%
    \label{thm:peps_regionBlockedWeight_insert_eq_sum_split}
    \lean{TNLean.PEPS.regionBlockedWeight_insert_eq_sum_split}
    \leanok
    \uses{thm:peps_prod_region_insert_split}
    For $v\notin R$, boundary configuration $\mu$ on $\partial(R\cup\{v\})$, and physical configuration $\sigma$ on $R\cup\{v\}$,
    \begin{align}
        T_{A,R\cup\{v\}}^{\mu}(\sigma)
         & =
        \sum_{\zeta:\,\zeta|_{\partial(R\cup\{v\})}=\mu}
        A_v(\zeta|_{\inc(v)},\sigma_v)
        \prod_{w\in R}A_w(\zeta|_{\inc(w)},\sigma|_R{}_w).
        \label{eq:peps_region_blocked_weight_insert}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Apply (\ref{eq:peps_prod_region_insert_split}) of Lemma~\ref{thm:peps_prod_region_insert_split} to each summand of $T_{A,R\cup\{v\}}^{\mu}(\sigma)$.
\end{proof}

\begin{lemma}[A non-incident crossing edge survives the insertion]%
    \label{thm:peps_isRegionBoundaryEdge_insert_of_not_incident}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_insert_of_not_incident}
    \leanok
    If $v\notin R$ and $e\in\partial R$ is not incident to $v$, then $e\in\partial(R\cup\{v\})$.
\end{lemma}
\begin{proof}\leanok
    Since $v\notin e$, adjoining $v$ to $R$ leaves the membership of both endpoints of $e$ in $R\cup\{v\}$ unchanged.
\end{proof}

\begin{definition}[Boundary label of the smaller region across an insertion]%
    \label{def:peps_boundaryLabelOfInsert}
    \lean{TNLean.PEPS.boundaryLabelOfInsert}
    \leanok
    \uses{thm:peps_isRegionBoundaryEdge_insert_of_not_incident}
    For $v\notin R$, a crossing-edge label $\mu$ on $\partial(R\cup\{v\})$, and a local virtual configuration $\eta$ on $\inc(v)$, define the crossing-edge label $\beta(\mu,\eta)$ on $\partial R$ by
    \begin{align}
        \beta(\mu,\eta)_e & =
        \begin{cases}
            \eta_e, & e\in\partial R\cap\inc(v),      \\
            \mu_e,  & e\in\partial R\setminus\inc(v).
        \end{cases}
        \label{eq:peps_boundary_label_of_insert}
    \end{align}
\end{definition}

\begin{lemma}[Boundary-label fiber identity across an inserted site]%
    \label{thm:peps_regionBoundaryLabel_eq_boundaryLabelOfInsert}
    \lean{TNLean.PEPS.regionBoundaryLabel_eq_boundaryLabelOfInsert}
    \leanok
    \uses{def:peps_boundaryLabelOfInsert}
    For $v\notin R$, if $\zeta|_{\partial(R\cup\{v\})}=\mu$ and $\zeta|_{\inc(v)}=\eta$, then $\zeta|_{\partial R}=\beta(\mu,\eta)$.
\end{lemma}
\begin{proof}\leanok
    Compare $\zeta|_{\partial R}$ and (\ref{eq:peps_boundary_label_of_insert}) edge by edge on $\partial R\cap\inc(v)$ and $\partial R\setminus\inc(v)$.
\end{proof}

\begin{lemma}[Region gauge absorption]%
    \label{thm:peps_regionInsertedCoeff_applyGauge}
    \lean{TNLean.PEPS.regionInsertedCoeff_applyGauge}
    \leanok
    \uses{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff, thm:peps_edge_gauge_absorption, def:applyGauge}
    For an edge-gauge family $X$, gauged tensor $B^X$, and boundary edge $f\in\partial R$ with orientation map $\mathcal O_{R,f}$,
    \begin{align}
        C_{B^X}(R,f,M;\sigma,\tau)
         & =
        C_B\bigl(R,f,
        \mathcal O_{R,f}\bigl(X_f^{\mathsf T}
        \mathcal O_{R,f}(M)\,(X_f^{-1})^{\mathsf T}\bigr);
        \sigma,\tau\bigr).
        \label{eq:peps_region_inserted_coeff_apply_gauge}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    With $p_B(R)=\prod_{g\notin\partial R}D_B(g)$ and $N=\mathcal O_{R,f}(M)$, Lemma~\ref{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff} and Lemma~\ref{thm:peps_edge_gauge_absorption} give
    \begin{align}
        C_{B^X}(R,f,M;\sigma,\tau)
         & =p_B(R)\,E_{B^X}(f,\omega_{R,\sigma,\tau},N)
        =p_B(R)\,E_B\bigl(f,\omega_{R,\sigma,\tau},X_f^{\mathsf T}\,N\,(X_f^{-1})^{\mathsf T}\bigr),
        \notag
    \end{align}
    and applying Lemma~\ref{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff} on $B$ with $\mathcal O_{R,f}\circ\mathcal O_{R,f}=\id$ yields (\ref{eq:peps_region_inserted_coeff_apply_gauge}).
\end{proof}

\subsection{From the conjugation identity to the absorbed equality}%
\label{subsec:peps_absorbed_equality}

% Companion to docs/paper-gaps/peps_normal_ft_section3_route.tex.
At $f\in\partial R$, the per-edge gauge identity takes the conjugation form $C_A(R,f,M;\sigma,\tau)=C_B(R,f,Z\,\phi_{h_f}(M)\,Z^{-1};\sigma,\tau)$, which the region comparison of \cite[Section~3]{Molnar2018NormalPEPS} converts into the absorbed form $C_A(R,f,M;\sigma,\tau)=C_{B^X}(R,f,\phi_{h_f}(M);\sigma,\tau)$.

\begin{definition}[Orientation-adapted absorbing gauge]%
    \label{def:peps_absorbedBoundaryGauge}
    \lean{TNLean.PEPS.absorbedBoundaryGauge}
    \leanok
    For $f\in\partial R$ and $Z\in\GL(D_B(f))$, the orientation-adapted absorbing gauge is
    \begin{align}
        X_f
         & =
        \begin{cases}
            Z^{\mathsf T} & \text{if the left endpoint of $f$ lies in $R$,} \\
            Z^{-1}        & \text{otherwise.}
        \end{cases}
        \label{eq:peps_absorbed_boundary_gauge}
    \end{align}
\end{definition}

\begin{lemma}[Matrix reconciliation for the absorbing gauge]%
    \label{thm:peps_regionEdgeOrient_absorbedBoundaryGauge}
    \lean{TNLean.PEPS.regionEdgeOrient_absorbedBoundaryGauge}
    \leanok
    \uses{def:peps_absorbedBoundaryGauge}
    For $X_f$ as in (\ref{eq:peps_absorbed_boundary_gauge}) and every $N\in\MN{D_B(f)}$,
    \begin{align}
        \mathcal O_{R,f}\bigl(X_f^{\mathsf T} \mathcal O_{R,f}(N)
        (X_f^{-1})^{\mathsf T}\bigr)
         & =
        Z\,N\,Z^{-1}.
        \label{eq:peps_region_edge_orient_absorbed}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    If the left endpoint of $f$ lies in $R$, then $\mathcal O_{R,f}=\id$ and $X_f^{\mathsf T}=Z$. Otherwise $\mathcal O_{R,f}(N)=N^{\mathsf T}$ and $X_f=Z^{-1}$, so $X_f^{\mathsf T}\mathcal O_{R,f}(N)(X_f^{-1})^{\mathsf T}=(ZNZ^{-1})^{\mathsf T}$, and applying $\mathcal O_{R,f}$ again gives $ZNZ^{-1}$.
\end{proof}

\begin{lemma}[Region absorbed equality from the conjugation identity]%
    \label{thm:peps_regionInsertedCoeff_eq_applyGauge_of_conjIdentity}
    \lean{TNLean.PEPS.regionInsertedCoeff_eq_applyGauge_of_conjIdentity}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:applyGauge, def:peps_absorbedBoundaryGauge, thm:peps_regionEdgeOrient_absorbedBoundaryGauge, thm:peps_regionInsertedCoeff_applyGauge}
    Let $f\in\partial R$, assume $h:D_A=D_B$, and suppose $Z\in\GL(D_B(f))$ satisfies $C_A(R,f,M;\sigma,\tau)=C_B(R,f,Z\,\phi_{h_f}(M)\,Z^{-1};\sigma,\tau)$ for all $M,\sigma,\tau$. If $X$ is an edge-gauge family with $X_f$ given by (\ref{eq:peps_absorbed_boundary_gauge}), then
    \begin{align}
        C_A(R,f,M;\sigma,\tau)
         & =
        C_{B^X}(R,f,\phi_{h_f}(M);\sigma,\tau)
        \label{eq:peps_region_absorbed_equality}
    \end{align}
    (the region analog of Lemma~\ref{thm:peps_postAbsorptionEdgeInsertionEquality}).
\end{lemma}
\begin{proof}\leanok
    Combine (\ref{eq:peps_region_inserted_coeff_apply_gauge}) at $\phi_{h_f}(M)$ with (\ref{eq:peps_region_edge_orient_absorbed}) and the conjugation hypothesis.
\end{proof}

\begin{lemma}[Edge absorbed equality from the region absorbed equality]%
    \label{thm:peps_edgeInsertedCoeff_eq_applyGauge_of_region}
    \lean{TNLean.PEPS.edgeInsertedCoeff_eq_applyGauge_of_region}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_edgeInsertedCoeff, def:applyGauge, thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff}
    Let $f\in\partial R$ and assume $h:D_A=D_B>0$. If (\ref{eq:peps_region_absorbed_equality}) holds for all $M,\sigma,\tau$, then for every global physical configuration $\sigma$ and $N\in\MN{D_A(f)}$,
    \begin{align}
        E_A(f,\sigma,N)
         & =
        E_{B^X}(f,\sigma,\phi_{h_f}(N)).
        \label{eq:peps_edge_absorbed_equality}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    With $p_A(R)=\prod_{g\notin\partial R}D_A(g)>0$, Lemma~\ref{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff} and $\mathcal O_{R,f}\circ\mathcal O_{R,f}=\id$ give
    \begin{align}
        p_A(R)\,E_A(f,\sigma,N)
         & =C_A\bigl(R,f,\mathcal O_{R,f}(N);\sigma|_R,\sigma|_{V\setminus R}\bigr) \notag          \\
         & =C_{B^X}\bigl(R,f,\phi_{h_f}(\mathcal O_{R,f}(N));\sigma|_R,\sigma|_{V\setminus R}\bigr)
        =p_A(R)\,E_{B^X}(f,\sigma,\phi_{h_f}(N)),
        \notag
    \end{align}
    and cancelling $p_A(R)>0$ yields (\ref{eq:peps_edge_absorbed_equality}).
\end{proof}

\begin{lemma}[Region absorbed equality from the edge absorbed equality]%
    \label{thm:peps_regionInsertedCoeff_eq_applyGauge_of_edge}
    \lean{TNLean.PEPS.regionInsertedCoeff_eq_applyGauge_of_edge}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_edgeInsertedCoeff, def:applyGauge, thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff}
    Assume $h:D_A=D_B$. If (\ref{eq:peps_edge_absorbed_equality}) holds at an edge $e$ for all $\sigma$ and $N$, then (\ref{eq:peps_region_absorbed_equality}) holds at every region $R$ with $e\in\partial R$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff} and $\mathcal O_{R,e}\circ\phi_{h_e}=\phi_{h_e}\circ\mathcal O_{R,e}$, both sides of (\ref{eq:peps_region_absorbed_equality}) equal $p_A(R)\,E_A(e,\omega_{R,\sigma,\tau},\mathcal O_{R,e}(M))=p_A(R)\,E_{B^X}(e,\omega_{R,\sigma,\tau},\phi_{h_e}(\mathcal O_{R,e}(M)))$.
\end{proof}
