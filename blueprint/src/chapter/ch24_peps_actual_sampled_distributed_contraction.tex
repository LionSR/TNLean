\section{Original occurrences and common sample assignments}
\label{sec:peps_original_gate_sample_reindexing}

Following \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, let
$\mathcal W:a\to b$ be a chronological source circuit on an arbitrary
party set $P$, with original gate occurrences $g\in G$ having nonempty
participant sets $C_g\subseteq P$ and finite branch sets $\Xi_g$.
Write $\mathcal E(C_g):=\{\{p,q\}\subseteq C_g:p\ne q\}$,
$\mathcal S:=\bigsqcup_{g\in G}\mathcal E(C_g)$, and
$\mathcal S_p:=\{s\in\mathcal S:p\in s\}$.
Let $N$ and $n_p$ be the lengths of the ordered full and incident source lists,
write $[r]=\{0,\ldots,r-1\}$ for $r\in\N$, and let
$\tau:[N]\to\mathcal S$ and $\kappa_p:[n_p]\to\mathcal S_p$ be the canonical position bijections.

\begin{lemma}[Labels on the full selected gate set]

    \begin{align}

For the next three theorems, let $G,K_g,B_g$ be finite, set $\mathcal K:=\prod_{g\in G} K_g$ and $\mathcal B:=\prod_{g\in G} B_g$, and for $z\in\prod_{g\in G}(K_g\times B_g)$ write $z^K_g:=(z_g)_1$ and $z^B_g:=(z_g)_2$.

\begin{lemma}[Pairing the ket and bra label sums]
    \label{thm:peps_ordered_gate_label_pair_sum}
    \lean{TNLean.Lattice.Compression.sum_gateLabels_eq_sum_orderedGateLabelPairs}

    \begin{align}
        \sum_{\eta\in\mathcal K}\sum_{\theta\in\mathcal B}F(\eta,\theta)
         & =\sum_{z\in\prod_g(K_g\times B_g)}F(z^K,z^B).

    \end{align}

\section{The sampled physical density as a distributed contraction}
\label{sec:peps_actual_sampled_distributed_contraction}

Let $P$ be finite with a repetition-free ordering, local physical dimensions $d:P\to\N$, physical register list $h_d$, private output list $c$, and a chronological circuit $\mathcal W:a\to h_dc$ with product input $\psi$.
Retain the endpoint dimensions $u_e,v_e$, $r_e=\min(u_e,v_e)$, branch coefficients $a_g:\Xi_g\to\C$, sample count $k\in\N$, ordered-pair arrays $\omega$, and local orthonormal bases of Definition~\ref{def:peps_incident_sampled_physical_factor}.
Let $\widehat\rho_{\mathrm{phys},\psi}(\widehat X)$ be the sampled physical operator, let $\gamma=\lambda_W\overline{\lambda_W}$ be the squared modulus of the empty-participant gate scalar, and for incident gate labels $\alpha,\beta$ and $s:\mathcal S_p\to[k]$ define
\begin{align}
    T_p(\alpha,\beta,s)
     & =M^{\rm inc}_{p,\alpha,\beta,s\circ\kappa_p,\omega}.
    \label{eq:peps_distributed_incident_matrix}
\end{align}

\begin{lemma}[The sampled physical sum in original gate and sample positions]
    \label{thm:peps_original_gate_sample_physical_sum}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.physicalSourceReplacedDensity_pure_sampled_eq_scalar_smul_originalGateSampleSum}
    \leanok
    \uses{def:peps_incident_sampled_physical_factor,def:peps_source_sample_assignment_indices}
    With $T_p$ from (\ref{eq:peps_distributed_incident_matrix}) (including empty sets, zero dimensions, and $k=0$ with $0^{-1}=0$ and $0^0=1$),
    \begin{align}
        \widehat\rho_{\mathrm{phys},\psi}(\widehat X)
         & =(k^{-1})^N\gamma
        \sum_{\eta,\theta\in\prod_{g\in G}\Xi_g}
        \left(\prod_{g\in G}a_g(\eta_g)\overline{a_g(\theta_g)}\right)
        \sum_{j:\mathcal S\to[k]}
        \bigotimes_{p\in P}T_p(\eta|_p,\theta|_p,j|_p).
        \label{eq:peps_original_gate_sample_physical_sum}
    \end{align}
    % Source: 04-compression.tex, lines 351--381 and 565--588.
\end{lemma}
