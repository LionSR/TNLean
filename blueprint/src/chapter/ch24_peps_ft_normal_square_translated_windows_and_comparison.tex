\begin{definition}[Comparison square, comparison region, and interior window]%
    \label{def:peps_normalSquareComparisonRegions}
    \lean{TNLean.PEPS.normalSquareComparisonSquare}
    \lean{TNLean.PEPS.normalSquareComparisonRegion}
    \lean{TNLean.PEPS.IsNormalSquareComparisonWindow}
    \leanok
    \uses{def:peps_squareLatticeCoordinateRegions}
    At offset $(a,b)$ on the $n\times m$ lattice
    (Definition~\ref{def:peps_squareLatticeCoordinateRegions}), the
    \emph{comparison square} and \emph{comparison region} are
    \begin{align}
        S_{a,b} & = R(a,b;3,3), \notag                                       \\
        R_{a,b} & = R(a+1,b;2,3)\cup R(a,b+1;3,2)=S_{a,b}\setminus\{(a,b)\},
        \notag
    \end{align}
    and the \emph{interior comparison window} is defined by the bounds
    $4\le a\le n-9$ and $4\le b\le m-9$, under which every boundary edge of
    $R_{a,b}$ and $S_{a,b}$ carries the interior edge-blocking margins.
\end{definition}

\begin{lemma}[Membership in the comparison region]%
    \label{lem:peps_normalSquareComparisonRegion_membership}
    \lean{TNLean.PEPS.mem_normalSquareComparisonRegion}
    \lean{TNLean.PEPS.notMem_normalSquareComparisonRegion}
    \lean{TNLean.PEPS.insert_normalSquareComparisonRegion}
    \leanok
    \uses{def:peps_normalSquareComparisonRegions, lem:peps_squareLatticeCoordinateRegion_identities}
    For any $(a,b)$, a vertex $(x,y)$ satisfies $(x,y)\in R_{a,b}$ if and only
    if
    \begin{align}
         & (a+1\le x<a+3\wedge b\le y<b+3)
        \lor
        (a\le x<a+3\wedge b+1\le y<b+3).
        \label{eq:peps_comp_region_mem}
    \end{align}
    In particular, for $v=(v_1,v_2)\in V$, $v\notin R_{v_1,v_2}$ and
    $R_{v_1,v_2}\cup\{v\}=S_{v_1,v_2}$.
\end{lemma}
\begin{proof}\leanok
    Unfolding $R_{a,b}=R(a+1,b;2,3)\cup R(a,b+1;3,2)$ via
    Lemma~\ref{lem:peps_squareLatticeCoordinateRegion_identities} gives
    (\ref{eq:peps_comp_region_mem}). At $(a,b)=(x,y)=(v_1,v_2)$, both
    $v_1+1\le v_1$ and $v_2+1\le v_2$ are false, so $v\notin R_{v_1,v_2}$, and
    adjoining $\{(v_1,v_2)\}$ to (\ref{eq:peps_comp_region_mem}) yields
    $v_1\le x<v_1+3\wedge v_2\le y<v_2+3$, which is $S_{v_1,v_2}$.
\end{proof}

\begin{lemma}[Band decomposition of the comparison square complement]%
    \label{lem:peps_normalSquareComparisonSquare_complementBands}
    \lean{TNLean.PEPS.compl_normalSquareComparisonSquare_eq_bands}
    \leanok
    \uses{def:peps_normalSquareComparisonRegions}
    On the $n\times m$ lattice with $a+3\le n$ and $b+3\le m$,
    \begin{align}
        V\setminus S_{a,b}
         & =
        \{(x,y):0\le x<a,\ 0\le y<m\}
        \cup
        \{(x,y):a+3\le x<n,\ 0\le y<m\}
        \notag        \\
         & \qquad\cup
        \{(x,y):a\le x<a+3,\ 0\le y<b\}
        \cup
        \{(x,y):a\le x<a+3,\ b+3\le y<m\}.
        \label{eq:peps_comp_square_bands}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    For $(x,y)\in V$ ($0\le x<n$, $0\le y<m$), negating
    $(x,y)\in S_{a,b}\Longleftrightarrow a\le x<a+3\wedge b\le y<b+3$ yields the
    four disjuncts of (\ref{eq:peps_comp_square_bands}).
\end{proof}

\begin{lemma}[Band decomposition of the comparison region complement]%
    \label{lem:peps_normalSquareComparisonRegion_complementBands}
    \lean{TNLean.PEPS.compl_normalSquareComparisonRegion_eq_bands}
    \leanok
    \uses{def:peps_normalSquareComparisonRegions, lem:peps_normalSquareComparisonRegion_membership}
    On the $n\times m$ lattice with $2\le a$, $1\le b$, $a+3\le n$, and
    $b+3\le m$, the complement $V\setminus R_{a,b}$ is the union of the four
    bands of (\ref{eq:peps_comp_square_bands})
    (Lemma~\ref{lem:peps_normalSquareComparisonSquare_complementBands}) and the
    $3\times 2$ corner rectangle
    $R(a-2,b-1;3,2)=\{(x,y):a-2\le x<a+1,\ b-1\le y<b+1\}$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{lem:peps_normalSquareComparisonRegion_membership},
    $S_{a,b}=R_{a,b}\cup\{(a,b)\}$, and $(a,b)\in R(a-2,b-1;3,2)\subseteq
        V\setminus R_{a,b}$ since $2\le a$ and $1\le b$; hence
    \begin{align}
        V\setminus R_{a,b}
         & =
        R(a-2,b-1;3,2)\cup(V\setminus S_{a,b}).
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Union closure of the boundary countermodel]%

\end{lemma}
\begin{proof}\leanok
    Since $u\in R_e$ and $\inj_{\partial}(R_e)$ holds, $v\in R_e$, contradicting
    $v\in B_e$ and $R_e\cap B_e=\varnothing$.
\end{proof}

\begin{lemma}[Rectangular injectivity does not force an open-boundary
        blocking]%
    \label{thm:peps_rectangularInjectivity_not_boundaryBlockingData}
    \lean{TNLean.PEPS.exists_rectangularInjectivity_not_boundaryBlockingData}
    \leanok
    \uses{thm:peps_leftBoundaryRightEdgeInjectivityCountermodel_unionClosure,
        thm:peps_leftBoundaryRightEdgeInjectivityCountermodel_rectangular,
        thm:peps_leftBoundaryRightEdge_not_boundaryBlockingData}
    On the open $7\times 7$ rectangle, there exists a union-closed region
    assignment $\inj_{\partial}$ satisfying all $2\times 3$ and $3\times 2$
    rectangular conditions that admits no three-partite injective blocking
    around the boundary edge $u$--$v$ (this abstract region countermodel does
    not concern the periodic torus setting of
    \cite[Theorem~3]{Molnar2018NormalPEPS}).
\end{lemma}
\begin{proof}\leanok
    Combine
    Lemmas~\ref{thm:peps_leftBoundaryRightEdgeInjectivityCountermodel_unionClosure},
    \ref{thm:peps_leftBoundaryRightEdgeInjectivityCountermodel_rectangular},
    and~\ref{thm:peps_leftBoundaryRightEdge_not_boundaryBlockingData}.
\end{proof}

\begin{lemma}[Injectivity from normal edge blocking]%
    \label{thm:peps_normalEdgeBlockingHypotheses_injectiveChain}
    \lean{TNLean.PEPS.NormalEdgeBlockingData.injective_chain}
    \lean{TNLean.PEPS.NormalEdgeBlockingData.endpoint_disjoint_cover}
    \lean{TNLean.PEPS.NormalEdgeBlockingData.endpoint_injective_disjoint_cover}
    \lean{TNLean.PEPS.NormalEdgeBlockingHypotheses.ofBlockingData}
    \lean{TNLean.PEPS.NormalEdgeBlockingHypotheses.ofBlockingData_blockingData}
    \lean{TNLean.PEPS.NormalEdgeBlockingHypotheses.injective_chain_at_edge}
    \lean{TNLean.PEPS.NormalEdgeBlockingHypotheses.endpoint_disjoint_cover_at_edge}
    \lean{TNLean.PEPS.%
        NormalEdgeBlockingHypotheses.endpoint_injective_disjoint_cover_at_edge}
    \leanok
    \uses{def:peps_normalEdgeBlockingHypotheses}
    Under Definition~\ref{def:peps_normalEdgeBlockingHypotheses}, the one-edge
    datum $d_e=(R_e,B_e,C_e)$ at each edge $e=\{u_e,v_e\}$ ($u_e<v_e$) satisfies
    $\inj(R_e)$, $\inj(B_e)$, $\inj(C_e)$, and
    \begin{align}
        u_e                 & \in R_e,
        \notag                                                      \\
        v_e                 & \in B_e,
        \notag                                                      \\
        R_e\cap B_e         & =R_e\cap C_e=B_e\cap C_e=\varnothing,
        \notag                                                      \\
        R_e\cup B_e\cup C_e & =V.
        \label{eq:peps_edge_blocking_cover}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Immediate by projecting the fields of $d_e$ in
    Definition~\ref{def:peps_normalEdgeBlockingHypotheses}.
\end{proof}
