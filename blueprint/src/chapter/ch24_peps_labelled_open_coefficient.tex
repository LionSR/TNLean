\begin{definition}[Open coefficients on labelled bonds]
    \label{def:peps_labelled_open_coefficient}
    \lean{TNLean.PEPS.DependentBondNetwork.RegionBoundaryEndpoint,
        TNLean.PEPS.DependentBondNetwork.RegionSiteConfig,
        TNLean.PEPS.DependentBondNetwork.joinRegionConfig,
        TNLean.PEPS.DependentBondNetwork.openBondFactor,
        TNLean.PEPS.DependentBondNetwork.openCoefficient,
        TNLean.PEPS.DependentBondNetwork.openBoundaryContraction,
        TNLean.PEPS.DependentBondNetwork.deltaExteriorConfig,
        TNLean.PEPS.DependentBondNetwork.deltaCompletedTensor,
        TNLean.PEPS.DependentBondNetwork.internalBondMatrices}
    \leanok
    For a region $R$ in a finite directed multigraph, assign an independent
    virtual index to every incidence at every vertex of $R$. Fix the indices
    on region-side incidences of crossing bonds to boundary data $\theta$.
    Multiply the region site coefficients and the prescribed matrix entries
    on internal bonds, and sum the remaining indices. Pair this coefficient
    with any joint boundary tensor by summing over $\theta$.
\end{definition}
\begin{theorem}[Exact delta-exterior completion]
    \label{thm:peps_labelled_open_delta}
    \lean{TNLean.PEPS.DependentBondNetwork.network_deltaCompletedTensor}
    \leanok
    \uses{def:peps_labelled_open_coefficient}
    Fix a virtual label $v_0$. Replace every exterior site by a delta tensor
    fixing each incidence opposite $R$ to its prescribed boundary label,
    and each other exterior incidence to $v_0$. Retain the internal bond
    matrices and insert identity matrices on all other bonds. The closed
    contraction equals the open coefficient exactly, with no scalar factor.
\end{theorem}
\begin{proof}\leanok
    Split the site assignments into region and exterior assignments.
    The exterior deltas leave one assignment. Identity matrices on crossing
    bonds impose the boundary constraints, while those on exterior bonds
    contribute one. The remaining sum is the defining open coefficient.
\end{proof}

\begin{definition}[Native coordinates and an embedded one-site collar]
    \label{def:peps_dual_collar}
    \lean{TNLean.PEPS.torusLocalTupleEquiv,TNLean.PEPS.torusEndpointTupleEquiv,
        TNLean.PEPS.torusOpenCoefficient,TNLean.PEPS.torusDeltaCompletedTensor,
        TNLean.PEPS.TorusDualCollar,TNLean.PEPS.TorusDualCollar.region,
        TNLean.PEPS.TorusDualCollar.swept,TNLean.PEPS.torusDualFluxBondMatrices}
    \leanok
    \uses{def:peps_labelled_open_coefficient,def:peps_dual_rectangle_comb}
    Native horizontal and vertical bond labels retain their direction and
    endpoint polarity. The four incidence labels at a site are ordered
    top, right, down, left. Around a rectangular sweep of side lengths
    $W,H$, take the translated collar with lifted coordinates
    $0\leq i\leq W+1$, $0\leq j\leq H+1$, requiring
    $W+1<w$ and $H+1<h$.
\end{definition}
\begin{theorem}[Native delta completion and collar geometry]
    \label{thm:peps_dual_collar_geometry}
    \lean{TNLean.PEPS.labelledNetwork_eq_torusBondNetwork,
        TNLean.PEPS.torusBondNetwork_deltaCompletedTensor,
        TNLean.PEPS.TorusDualCollar.origin_mem,TNLean.PEPS.TorusDualCollar.injective,
        TNLean.PEPS.TorusDualCollar.neighbors_mem,
        TNLean.PEPS.TorusDualCollar.incident_internal,
        TNLean.PEPS.TorusDualCollar.noninternal_outside,
        TNLean.PEPS.torusOpenCoefficient_eq_zero_of_isEmpty}
    \leanok
    \uses{def:peps_dual_collar,thm:peps_labelled_open_delta}
    The labelled contraction equals the native four-leg contraction, so the
    delta-completion identity holds in native coordinates as well.
    The collar is nonempty and embedded. Every swept site and its four
    neighbors lie in it; hence each noninternal bond has both endpoints
    outside the swept support. For an empty virtual alphabet the open
    coefficient is zero, since the collar contains an incidence.
\end{theorem}
\begin{proof}\leanok
    Regroup the four anchored bond endpoints at each vertex and read their
    incident site coordinates. The collar bounds follow directly from the
    lifted coordinate ranges. A noninternal bond incident to a swept site
    would have its other endpoint among that site's neighbors, a contradiction.
\end{proof}
\begin{theorem}[Collared open deformation with a correlated boundary]
    \label{thm:peps_dual_collar_open}
    \lean{TNLean.PEPS.TorusDualCollar.exists_internalFluxGauge,
        TNLean.PEPS.TorusDualCollar.openCoefficient_eq,
        TNLean.PEPS.TorusDualCollar.correlatedBoundary_eq}
    \leanok
    \uses{thm:peps_dual_collar_geometry,thm:peps_dual_rectangle,
        thm:peps_dual_flux_deformation}
    Two labelled unit-step paths with the same lifted endpoints in the
    embedded patch have equal open coefficients on its collar, provided
    the site tensors are virtually invariant on the swept interior.
    The group and its matrix representation are arbitrary. Equality persists
    after contraction against any joint boundary tensor, without a
    factorization hypothesis or a nonempty virtual-alphabet assumption.
\end{theorem}
\begin{proof}\leanok
    Derive the supported gauge from the rectangular homotopy. Both endpoints
    of crossing bonds are outside its support, so their gauge values are
    identity. Complete the fixed physical coefficient by exterior delta
    tensors and apply the closed gauge-contraction identity. The exact
    completion formula gives equality of open coefficients. Sum it against
    the joint boundary tensor. If the virtual alphabet is empty, both open
    sums vanish.
\end{proof}
