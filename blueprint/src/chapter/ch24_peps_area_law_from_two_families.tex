\section{The area law from the two-family decomposition}
\label{sec:area_law_from_two_families}

\begin{theorem}[Uniform area law from the two-family decomposition]
    \label{thm:area_law_conditional_two_families}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw_of_twoFamilies}
    \leanok
    \uses{def:area_law_two_family_decomposition,def:area_law_ground_entropy}
    Suppose that every integer $R\ge0$ admits constants $D_0>2R+10$ and $C_{\rm tpl}\ge24$ satisfying the two-family decomposition of Definition~\ref{def:area_law_two_family_decomposition} (supplied by Theorem~\ref{thm:area_law_two_family_decomposition}).
    Then for all integers $q\ge1$, $R\ge0$ and reals $J,\Delta>0$, there exists $C\ge0$ such that for every finite induced domain $\Lambda$, admissible Hamiltonian of range $R$ and interaction bound $J$ with unit ground vector $\Omega$ and global gap $\ge\Delta$, and cut $A\subseteq\Lambda$ (\cite[Theorem~1.1, Section~11]{OpenAI2026AreaLaw}),
    \begin{align}
        S_\Omega(A) & \le C|\partial_\Lambda A|.\notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_sublinear_collar,
        thm:area_law_conditional_collar_two_families}
    Choosing $D_0>2R+10$ and $C_{\rm tpl}\ge24$ for $R$, Theorem~\ref{thm:area_law_sublinear_collar} establishes the sublinear collar bound at $(D_0,C_{\rm tpl})$, and Theorem~\ref{thm:area_law_conditional_collar_two_families} yields $S_\Omega(A)\le C|\partial_\Lambda A|$.
\end{proof}

\subsection{Safe rectangles as consequences of the area law}

The uniform area law implies a linear entropy bound for safe rectangles, which together with the buffered-rectangle estimate yields the recurrence of~\cite[Proposition~3.3]{OpenAI2026AreaLaw} uniformly in the real clearance $d$.

\begin{theorem}[Boundary of a rectangle avoiding cut endpoints]
    \label{thm:area_law_safe_rectangle_boundary}
    \lean{TNLean.PEPS.AreaLaw.card_edgeBoundary_rectRegion_le_of_disjoint}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_cell_counts}
    For finite $\Lambda\subset\mathbb Z^2$, $A\subseteq\Lambda$ with cut-edge endpoint set $Z_A$, and any integer rectangle $Q$ with $Q\cap Z_A=\varnothing$,
    \begin{align}
        |\partial_\Lambda(A\cap Q)| \le 48\operatorname{size}(Q). \label{eq:safe_rect_boundary}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_physical_ambient_boundary,
        lem:area_law_ambient_boundary_layers,thm:area_law_buffer_contours}
    By Theorem~\ref{thm:area_law_physical_ambient_boundary}, Lemma~\ref{lem:area_law_ambient_boundary_layers}, and Theorem~\ref{thm:area_law_buffer_contours} (\cite[Lemma~9.4]{OpenAI2026AreaLaw}) with $\operatorname{size}(Q)\ge1$,
    \begin{align}
        |\partial_\Lambda(A\cap Q)|
         & \le |\partial Q|
        \le4|Q^{+1}\setminus Q|
        \le16\bigl(\operatorname{size}(Q)+2\bigr)
        \le48\operatorname{size}(Q).\notag
    \end{align}
\end{proof}

\begin{theorem}[Linear entropy for rectangles with real clearance]
    \label{thm:area_law_safe_rectangle_linear_entropy}
    \lean{TNLean.PEPS.AreaLaw.exists_regionalEntropy_safe_rect_le_linear}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_ground_entropy}
    For integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$, there exists $K\ge0$ such that for all $d\ge0$, finite induced domains $\Lambda$, range-$R$ interaction-$J$ Hamiltonians $H$, unit eigenvectors $\Omega$ satisfying the projector gap (\ref{eq:area_law_projector_gap}), cuts $A\subseteq\Lambda$, and integer rectangles $Q$ with $d\operatorname{size}(Q)<\|t-z\|_\infty$ ($\forall t\in Q,z\in Z_A$; \cite[Theorem~1.1, Section~3]{OpenAI2026AreaLaw}),
    \begin{align}
        S_\Omega(A\cap Q)\le K\operatorname{size}(Q). \label{eq:safe_rect_linear_entropy}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:openai_uniform_area_law,thm:area_law_safe_rectangle_boundary}
    Let $C\ge0$ be the constant of Theorem~\ref{thm:openai_uniform_area_law} at interaction bound $J+1>0$, and set $K=48C$.
    Since $d\ge0$ and $d\operatorname{size}(Q)<\|t-z\|_\infty$ force $Q\cap Z_A=\varnothing$, Theorem~\ref{thm:openai_uniform_area_law} and (\ref{eq:safe_rect_boundary}) give $S_\Omega(A\cap Q)\le C|\partial_\Lambda(A\cap Q)|\le48C\operatorname{size}(Q)$.
\end{proof}

\begin{theorem}[Joint buffer and fixed-clearance entropy recurrence]
    \label{thm:area_law_real_box_recurrence_after_area_law}
    \lean{TNLean.PEPS.AreaLaw.exists_initialBuffer_and_real_boxEntropy_recurrence}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_ground_entropy}
    For integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$, there exist an integer $c\ge1$ and real $b\ge0$ such that for every $d>2R+10$:
    \begin{enumerate}
        \item For any instance $(\Lambda,H,\Omega,A,Q)$ as in Theorem~\ref{thm:area_law_safe_rectangle_linear_entropy}, integer $r\ge1$, and rectangle $Q_0$ with $\operatorname{size}(Q_0)\le r$ and $Q_0^{+cr}\subseteq Q$, setting $X=A\cap Q_0$ and $T=(A\cap Q_0^{+cr})\setminus X$ (\cite[Lemma~3.2]{OpenAI2026AreaLaw}),
              \begin{align}
                  S_\Omega(X\cup T)-S_\Omega(T)
                  \le\tfrac12S_\Omega(X)+br.\notag
              \end{align}
        \item Defining the supremum over all such instances (with $\sup\varnothing=0$)
              \begin{align}
                  F_d(r) & =\sup\bigl\{S_\Omega(A\cap Q):
                  \operatorname{size}(Q)\le r,\ d\operatorname{size}(Q)<\|t-z\|_\infty
                  \ \forall t\in Q,z\in Z_A\bigr\}, \label{eq:Fd_def}
              \end{align}
              and $p_0=(2c+5)^{-2}$, for all integers $M,r\ge1$ (\cite[Proposition~3.3]{OpenAI2026AreaLaw}),
              \begin{align}
                  F_d(Mr)
                   & \le\bigl[(1-p_0/2)M^2+(p_0/2)\,4(c+2)M\bigr]F_d(r)
                  +bM^2r.\notag
              \end{align}
    \end{enumerate}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_initial_buffer,
        thm:area_law_safe_rectangle_linear_entropy,
        thm:area_law_regional_state_identities}
    Choose $(c,b_0)$ from Theorem~\ref{thm:area_law_initial_buffer} and $K$ from Theorem~\ref{thm:area_law_safe_rectangle_linear_entropy}, and set $b=\max\{b_0,K\}$.
    By (\ref{eq:safe_rect_linear_entropy}) and Theorem~\ref{thm:area_law_regional_state_identities}, $0\le F_d(t)\le Kt$ for all $t\ge0$; since $\alpha=(1-p_0/2)M^2+2p_0(c+2)M\ge0$ and $M\le M^2$,
    \begin{align}
        F_d(Mr)\le KMr\le bM^2r \le\alpha F_d(r)+bM^2r.\notag
    \end{align}
\end{proof}

\begin{theorem}[Improvement of the exponent at fixed real clearance]
    \label{thm:area_law_real_box_improvement_after_area_law}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_uniform_boxEntropy_improvement_real}
    \leanok
    \uses{thm:area_law_real_box_recurrence_after_area_law,
        def:area_law_bootstrap_parameters}
    Fix integers $q\ge1$, $R\ge0$, reals $J\ge0$, $\Delta>0$, $C_{\rm box}\ge0$, and $e_0\in(e_*,1)$ with $e_*=2\cdot10^{-6}$ and $\ell=\ell(e_0)$ from Definition~\ref{def:area_law_bootstrap_parameters}.
    There exists $C>0$ (chosen before $e,d$ and all instances) such that for all $e\in(e_*,e_0]$ and $d>2R+10$, if $F_d(r)$ in (\ref{eq:Fd_def}) satisfies $F_d(r)\le C_{\rm box}r^{1+e}$ for all $r\ge1$, then $F_d(r)\le Cr^{1+(1-\ell)e}$ for all $r\ge1$ (\cite[Proposition~9.5]{OpenAI2026AreaLaw}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_safe_rectangle_linear_entropy,
        thm:area_law_bootstrap_margins}
    Take $K\ge0$ from Theorem~\ref{thm:area_law_safe_rectangle_linear_entropy} and set $C=\max\{K,1\}>0$, so (\ref{eq:safe_rect_linear_entropy}) implies $F_d(r)\le Kr$.
    By Theorem~\ref{thm:area_law_bootstrap_margins}, $0<\ell<1$, so $e'=(1-\ell)e>0$ and for all $r\ge1$,
    \begin{align}
        F_d(r)\le Kr\le Cr\le Cr^{1+e'}.\notag
    \end{align}
\end{proof}
