\section{The area law from the two-family decomposition}
\label{sec:area_law_from_two_families}

\begin{theorem}[Uniform area law from the two-family decomposition]
    \label{thm:area_law_conditional_two_families}
    \lean{TNLean.PEPS.AreaLaw.uniformAreaLaw_of_twoFamilies}
    \leanok
    \uses{def:area_law_two_family_decomposition,def:area_law_ground_entropy}
    Suppose that for every integer $R\ge0$ there are real constants
    $D_0>2R+10$ and $C_{\rm tpl}\ge24$ for which the two-family decomposition
    of Definition~\ref{def:area_law_two_family_decomposition} holds.
    Then, for every pair of integers $q\ge1$, $R\ge0$ and every
    $J,\Delta>0$, there is a constant $C\ge0$ with the following property.
    For every finite induced domain $\Lambda$, every admissible Hamiltonian
    of range $R$ and interaction bound $J$ with a unit ground vector $\Omega$
    and global spectral gap at least $\Delta$, and every $A\subseteq\Lambda$,
    \begin{align}
        S_\Omega(A) & \le C|\partial_\Lambda A|.\notag
    \end{align}
    The constant is independent of the domain, Hamiltonian, ground vector,
    and cut. This is a conditional consequence of the argument in
    \cite[Theorem~1.1, Section~11]{OpenAI2026AreaLaw}; the geometric
    decomposition remains a hypothesis.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_sublinear_collar,
        thm:area_law_conditional_collar_two_families}
    Fix $q,R,J,\Delta$ and choose $D_0,C_{\rm tpl}$ from the hypothesis
    for this range $R$. Theorem~\ref{thm:area_law_sublinear_collar}
    proves the collar bound at this same pair, since $D_0>2R+10$ and
    $C_{\rm tpl}\ge24$. Theorem~\ref{thm:area_law_conditional_collar_two_families}
    then applies to this collar bound and the assumed two-family
    decomposition, and gives the asserted constant.
\end{proof}

\subsection{Safe rectangles as consequences of the area law}

The uniform area law gives a linear entropy bound for safe rectangles.
Together with the buffered-rectangle estimate, this yields the recurrence
in the proof of~\cite[Proposition~3.3]{OpenAI2026AreaLaw} with one pair of
constants for all admissible real clearances.

\begin{theorem}[Boundary of a rectangle avoiding cut endpoints]
    \label{thm:area_law_safe_rectangle_boundary}
    \lean{TNLean.PEPS.AreaLaw.card_edgeBoundary_rectRegion_le_of_disjoint}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_cell_counts}
    Let $\Lambda\subset\mathbb Z^2$ be finite, let $A\subseteq\Lambda$, and
    let $Z_A$ be the endpoints of the edges crossing $A$. If an integer
    rectangle $Q$ satisfies $Q\cap Z_A=\varnothing$, then
    $|\partial_\Lambda(A\cap Q)|\le48\operatorname{size}(Q)$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_physical_ambient_boundary,
        lem:area_law_ambient_boundary_layers,thm:area_law_buffer_contours}
    The physical-to-ambient comparison used in
    \cite[Lemma~9.4]{OpenAI2026AreaLaw} and the first outer layer give
    \begin{align}
        |\partial_\Lambda(A\cap Q)|
         & \le |\partial Q|
        \le4|Q^{+1}\setminus Q|
        \le16\bigl(\operatorname{size}(Q)+2\bigr)
        \le48\operatorname{size}(Q).\notag
    \end{align}
    The layer estimate is the rectangle dilation bound with both radii
    equal to one, and every integer rectangle has size at least one.
\end{proof}

\begin{theorem}[Linear entropy for rectangles with real clearance]
    \label{thm:area_law_safe_rectangle_linear_entropy}
    \lean{TNLean.PEPS.AreaLaw.exists_regionalEntropy_safe_rect_le_linear}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_ground_entropy}
    Fix integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$.
    There is a constant $K\ge0$ with the following property for every real
    $d\ge0$. Let $\Lambda$ be a finite induced domain, let $H$ be an
    admissible Hamiltonian of range $R$ and interaction bound $J$, and let
    $\Omega$ be a unit vector with eigenvalue $E_0$ satisfying the
    full-system projector gap (\ref{eq:area_law_projector_gap}).
    For any cut $A\subseteq\Lambda$ and integer rectangle $Q$ such that
    $d\operatorname{size}(Q)<\|t-z\|_\infty$ for every $t\in Q$ and
    $z\in Z_A$,
    \begin{align}
        S_\Omega(A\cap Q)\le K\operatorname{size}(Q).\notag
    \end{align}
    The constant is independent of $d$ and all physical instances.
    This is a consequence of~\cite[Theorem~1.1]{OpenAI2026AreaLaw} for
    the rectangles defined in Section~3 of that paper.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:openai_uniform_area_law,thm:area_law_safe_rectangle_boundary}
    Apply the uniform area law at interaction bound $J+1>0$, keeping every
    interaction term, the total Hamiltonian and the vector $\Omega$
    unchanged. Let $C\ge0$ be its constant and put $K=48C$.
    A point $z\in Q\cap Z_A$ would give
    $d\operatorname{size}(Q)<\|z-z\|_\infty=0$, contradicting $d\ge0$.
    Thus $Q\cap Z_A=\varnothing$, and
    \begin{align}
        S_\Omega(A\cap Q)
        \le C|\partial_\Lambda(A\cap Q)|
        \le48C\operatorname{size}(Q).\notag
    \end{align}
\end{proof}

\begin{theorem}[Joint buffer and fixed-clearance entropy recurrence]
    \label{thm:area_law_real_box_recurrence_after_area_law}
    \lean{TNLean.PEPS.AreaLaw.exists_initialBuffer_and_real_boxEntropy_recurrence}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_ground_entropy}
    Fix integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$.
    There are an integer $c\ge1$ and a real $b\ge0$ such that both assertions
    below hold for every real $d>2R+10$.

    Let $\Lambda$ be any finite induced domain, let $H$ be an admissible
    Hamiltonian of range $R$ and interaction bound $J$, and let $\Omega$
    be a unit eigenvector with eigenvalue $E_0$ satisfying the full-system
    projector gap. Let $A\subseteq\Lambda$ and let $Q$ satisfy
    $d\operatorname{size}(Q)<\|t-z\|_\infty$ for every $t\in Q$ and
    $z\in Z_A$. For every integer $r\ge1$ and integer rectangle $Q_0$ with
    $\operatorname{size}(Q_0)\le r$ and $Q_0^{+cr}\subseteq Q$, put
    $X=A\cap Q_0$ and $T=(A\cap Q_0^{+cr})\setminus X$. Then
    \begin{align}
        S_\Omega(X\cup T)-S_\Omega(T)
        \le\tfrac12S_\Omega(X)+br.\notag
    \end{align}

    For each fixed $d$, let
    \begin{align}
        F_d(r) & =\sup\bigl\{S_\Omega(A\cap Q):
        \operatorname{size}(Q)\le r,\notag                          \\
               & \hspace{2em}d\operatorname{size}(Q)<\|t-z\|_\infty
        \quad\forall t\in Q,\ \forall z\in Z_A\bigr\}.\notag
    \end{align}
    This supremum ranges over all finite domains, admissible Hamiltonians,
    eigenvalues, unit eigenvectors with the full-system projector gap, cuts and
    integer rectangles at the fixed $q,R,J,\Delta,d$; the supremum of an
    empty collection is taken to be zero. Put $p_0=(2c+5)^{-2}$.
    For all integers $M,r\ge1$,
    \begin{align}
        F_d(Mr)
         & \le\bigl[(1-p_0/2)M^2+(p_0/2)\,4(c+2)M\bigr]F_d(r)
        +bM^2r.\notag
    \end{align}
    The buffer estimate is that of~\cite[Lemma~3.2]{OpenAI2026AreaLaw};
    the multiplicative coefficient is the one in the recurrence in the
    proof of~\cite[Proposition~3.3]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_initial_buffer,
        thm:area_law_safe_rectangle_linear_entropy,
        thm:area_law_regional_state_identities}
    Choose $c\ge1$ and $b_0\ge0$ from the real-clearance buffer estimate,
    and $K\ge0$ from the linear entropy bound. Set $b=\max\{b_0,K\}$.
    Since $b_0r\le br$, the buffer conclusion holds with these same $c,b$.

    Fix $d>2R+10$. Entropy nonnegativity and the linear bound give
    $0\le F_d(t)\le Kt$ for every integer $t\ge0$, including an empty
    collection in the supremum. Write
    $\alpha=(1-p_0/2)M^2+(p_0/2)\,4(c+2)M$.
    Since $0\le p_0\le1$, one has $\alpha\ge0$; and $M\ge1$ gives
    $M\le M^2$. Consequently,
    \begin{align}
        F_d(Mr)\le KMr\le bM^2r
        \le\alpha F_d(r)+bM^2r.\notag
    \end{align}
    All constants were chosen before $d$ and the physical data.
\end{proof}

\begin{theorem}[Improvement of the exponent at fixed real clearance]
    \label{thm:area_law_real_box_improvement_after_area_law}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_uniform_boxEntropy_improvement_real}
    \leanok
    \uses{thm:area_law_real_box_recurrence_after_area_law,
        def:area_law_bootstrap_parameters}
    Fix integers $q\ge1$, $R\ge0$ and reals $J\ge0$, $\Delta>0$.
    Fix $e_0$ with $e_*<e_0<1$, where $e_*=2\cdot10^{-6}$, and fix
    $C_{\rm box}\ge0$. Put $\ell=\ell(e_0)$ as in
    Definition~\ref{def:area_law_bootstrap_parameters}.
    There is a constant $C>0$ such that the following implication holds
    for every $e$ with $e_*<e\le e_0$ and every real $d>2R+10$.

    Let $F_d(r)$ be the supremum over the physical instances and safe
    rectangles at fixed $q,R,J,\Delta,d$ specified in
    Theorem~\ref{thm:area_law_real_box_recurrence_after_area_law}.
    If $F_d(r)\le C_{\rm box}r^{1+e}$ for every integer $r\ge1$, then
    $F_d(r)\le Cr^{1+(1-\ell)e}$ for every integer $r\ge1$.
    The constant $C$ is chosen before $e,d$ and all physical instances.
    This gives the exponent improvement in the proof of
    \cite[Proposition~9.5]{OpenAI2026AreaLaw} as a consequence of the
    uniform area law.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_safe_rectangle_linear_entropy,
        thm:area_law_bootstrap_margins}
    Choose $K\ge0$ from the linear safe-rectangle estimate and set
    $C=\max\{K,1\}>0$. These constants precede $e$ and $d$.
    Fix admissible $e,d$ and an integer $r\ge1$.
    Every entropy in the defining collection for $F_d(r)$ is at most
    $Ks\le Kr$, where $s\le r$ is the size of its rectangle.
    Since $Kr\ge0$, this also bounds an empty collection under the
    stated supremum convention. Thus $F_d(r)\le Kr$.

    The parameter bounds give $0<\ell<1$; hence
    $e'=(1-\ell)e>0$. Since $r\ge1$, one has $r\le r^{1+e'}$,
    and therefore
    \begin{align}
        F_d(r)\le Kr\le Cr\le Cr^{1+e'}.\notag
    \end{align}
    The linear bound proves the conclusion uniformly, including every
    instance satisfying the assumed bound with coefficient $C_{\rm box}$.
\end{proof}
