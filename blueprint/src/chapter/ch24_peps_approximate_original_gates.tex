\section{Approximate original gate expansions}
\label{sec:peps_approximate_original_gates}

The final paragraph of the proof of
\cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} permits approximate
monomial expansions: we retain the original gate occurrences and rescale
only their scalar coefficients, so the resulting operator and physical
trace-norm estimates do not involve private dimensions.

\begin{definition}[Original operators on a fixed chronology]
    \label{def:peps_approximate_original_operators}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.GateMaps}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.evalWithGateMaps}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.IsGateApproximation}
    \leanok
    Fix a chronological circuit whose gate occurrence $g$ carries an
    ordered expansion $T_g=\sum_j c_{g,j}M_{g,j}$, and assign an original
    operator $G_g$ on the same local input and output memories to every $g$.
    Let $W_G$ be their chronological composition with the prescribed private
    operations and spectator identities. For $\delta\ge0$, require all private
    operators and all $G_g$ to be contractions, every listed $M_{g,j}$ to be
    allowed, and $\|T_g-G_g\|\le\delta$. Each nonprivate occurrence has a
    participating set of cardinality different from $1$ (possibly $\varnothing$).
\end{definition}

\begin{lemma}[Contractivity of the original chronology]
    \label{thm:peps_original_operator_contraction}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.norm_evalWithGateMaps_le_one}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.norm_eval_le_one}
    \leanok
    \uses{def:peps_approximate_original_operators}
    Under the hypotheses of Definition~\ref{def:peps_approximate_original_operators},
    $\|W_G\|\le1$. The same bound holds for every original contraction circuit.
\end{lemma}
\begin{proof}
    \leanok
    By induction on the chronology, composition is submultiplicative in operator
    norm, tensoring with a spectator identity preserves the norm, and the
    prescribed owner and register identifications are isometries.
\end{proof}

\begin{definition}[Rescaling an approximate expansion]
    \label{def:peps_rescaled_original}
    \lean{TNLean.Lattice.PartyCircuit.rescalePartyGate}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rescaledOriginal}
    \leanok
    \uses{def:peps_approximate_original_operators}
    Replacing each $c_{g,j}$ by $c_{g,j}/(1+\delta)$ defines the chronological
    circuit $\widetilde W_\delta$ with the original private operations and
    local operator $\widetilde T_g=T_g/(1+\delta)$ at $g$.
\end{definition}
\begin{proof}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.monomials_rescalePartyGate}
    \lean{TNLean.Lattice.PartyCircuit.gate_rescalePartyGate}
    \lean{TNLean.Lattice.PartyCircuit.sum_norm_rescalePartyGate_le}
    \lean{TNLean.Lattice.PartyCircuit.isAllowed_rescalePartyGate}
    The ordered monomial list is unchanged, linearity gives
    $\widetilde T_g=T_g/(1+\delta)$, and $\|T_g\|\le\|G_g\|+\delta\le1+\delta$
    gives $\|\widetilde T_g\|\le1$. Each monomial remains allowed, while
    \begin{align}
        \sum_j\left|\frac{c_{g,j}}{1+\delta}\right|
        \le\sum_j|c_{g,j}|.
        \label{eq:peps_rescaled_coefficient_sum}
    \end{align}
\end{proof}

\begin{definition}[Occurrences and monomial counts after rescaling]
    \label{def:peps_rescaled_occurrences}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rescaledLocationsEquiv}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.monomialCount}
    \leanok
    \uses{def:peps_rescaled_original}
    Matching every gate of $\widetilde W_\delta$ to the same original
    occurrence while keeping repeated occurrences distinct, let $k_g$ be
    the length of its prescribed monomial list.
\end{definition}

\begin{lemma}[Preservation of the original resource bounds]
    \label{thm:peps_rescaled_resources}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.nonprivateCount_rescaledOriginal}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.participants_rescaledOriginal}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.participationCount_rescaledOriginal}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.isExpansionBounded_rescaledOriginal}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.isMonomialBounded_rescaledOriginal}
    \leanok
    \uses{def:peps_rescaled_occurrences}
    Rescaling preserves the number $N$ of nonprivate occurrences, their
    participating sets, and each party's participation count. If each monomial
    has at most $r$ pair effects, $\sum_j|c_{g,j}|\le S$, and $k_g\le K$, the
    same three bounds hold for $\widetilde W_\delta$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:peps_rescaled_original,def:peps_rescaled_occurrences}
    The occurrence correspondence of Definition~\ref{def:peps_rescaled_occurrences}
    preserves each participating set; the monomials and their pair effects are
    identical, and (\ref{eq:peps_rescaled_coefficient_sum}) bounds the rescaled
    coefficient sums.
\end{proof}

\begin{lemma}[Two-factor telescoping estimate]
    \label{thm:peps_contraction_composition_error}
    \lean{TNLean.Lattice.PartyCircuit.norm_comp_sub_comp_le_one}
    \leanok
    Let $A,C:E\to F$ and $B,D:F\to G$ be bounded linear maps between
    complex inner-product spaces. If $\|B\|,\|C\|\le1$, then
    $\|BA-DC\|\le\|A-C\|+\|B-D\|$.
\end{lemma}
\begin{proof}
    \leanok
    Combine $BA-DC=B(A-C)+(B-D)C$ with the triangle inequality and
    submultiplicativity of the operator norm.
\end{proof}

\begin{lemma}[Chronological operator error]
    \label{thm:peps_approximate_chronological_error}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.norm_rescaledOriginal_sub_evalWithGateMaps_le}
    \leanok
    \uses{def:peps_rescaled_original}
    If $\|T_g-G_g\|\le\delta$ for $\delta\ge0$ and the original chronology has
    $N$ nonprivate occurrences, then $\|\widetilde W_\delta-W_G\|\le2\delta N$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_original_operator_contraction,thm:peps_contraction_composition_error}
    The local error satisfies
    \begin{align}
        \|\widetilde T_g-G_g\|
        \le \frac{\|T_g-G_g\|+\delta\|G_g\|}{1+\delta}
        \le2\delta.
        \label{eq:peps_rescaled_local_error}
    \end{align}
    By Lemmas~\ref{thm:peps_original_operator_contraction} and
    \ref{thm:peps_contraction_composition_error}, private maps and register
    permutations add no error while spectator identities preserve
    (\ref{eq:peps_rescaled_local_error}), so induction adds $2\delta$ per gate.
\end{proof}

\begin{definition}[Physical output of the original operators]
    \label{def:peps_original_gate_density}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.physicalDensityWithGateMaps}
    \leanok
    \uses{def:peps_approximate_original_operators}
    Given an isometric identification $K$ of the final memory with
    finite physical and discarded coordinate spaces and an input $x$, define
    the unnormalized physical outputs
    $\rho_G(x)=\operatorname{Tr}_{\rm discard}|KW_Gx\rangle\langle KW_Gx|$
    and $\widetilde\rho_\delta(x)=\operatorname{Tr}_{\rm discard}|K\widetilde W_\delta x\rangle\langle K\widetilde W_\delta x|$.
\end{definition}

\begin{lemma}[Physical error of approximate original gates]
    \label{thm:peps_approximate_physical_error}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rectangularTraceNorm_rescaledOriginal_density_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rectangularTraceNorm_rescaledOriginal_density_sub_le_budget}
    \leanok
    \uses{def:peps_original_gate_density}
    If $\|x\|\le1$, then
    $\|\widetilde\rho_\delta(x)-\rho_G(x)\|_1\le4\delta N$.
    In particular, for $\varepsilon>0$ and $N\le M$, taking
    $\delta=\varepsilon/(8\max\{1,M\})$ gives
    $\|\widetilde\rho_\delta(x)-\rho_G(x)\|_1\le\varepsilon/2$, and replacing
    $\varepsilon$ by $\varepsilon/2$ gives error at most $\varepsilon/4$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_approximate_chronological_error,thm:peps_original_operator_contraction}
    For $u=K\widetilde W_\delta x$ and $v=KW_Gx$,
    Lemma~\ref{thm:peps_original_operator_contraction} and the isometry $K$
    give $\|u\|,\|v\|\le1$. By Lemma~\ref{thm:peps_approximate_chronological_error},
    \begin{align}
        \|\widetilde\rho_\delta(x)-\rho_G(x)\|_1
         & =\bigl\|\operatorname{Tr}_{\rm discard}
        (|u\rangle\langle u|-|v\rangle\langle v|)\bigr\|_1
        \le2\|u-v\|                                \notag \\
         & \le2\|\widetilde W_\delta-W_G\|\,\|x\|
        \le4\delta N.
        \label{eq:peps_approximate_density_error}
    \end{align}
    Substituting $\delta$ into (\ref{eq:peps_approximate_density_error}) yields
    the remaining claims.
\end{proof}

\begin{lemma}[Natural inverse-power accuracy]
    \label{thm:peps_approximate_natural_accuracy}
    \lean{TNLean.Lattice.PartyCircuit.gateApproximation_error_inv_pow_le}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rectangularTraceNorm_rescaledOriginal_density_sub_le_inv_pow}
    \uses{def:peps_original_gate_density}
    \leanok
    Let $N\le CL^n$ with $n,p\in\N$ and $L\ge\max\{1,16C\}$. Local accuracy
    $\delta=L^{-(n+p+1)}$ gives
    $\|\widetilde\rho_\delta(x)-\rho_G(x)\|_1\le L^{-p}/4$
    for $\|x\|\le1$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_approximate_physical_error}
    Lemma~\ref{thm:peps_approximate_physical_error} gives
    $\|\widetilde\rho_\delta(x)-\rho_G(x)\|_1\le4L^{-(n+p+1)}N\le(16C/L)L^{-p}/4\le L^{-p}/4$.
\end{proof}

\begin{lemma}[One fixed accuracy exponent for approximate gates]
    \label{thm:peps_approximate_gate_real_accuracy}
    \lean{TNLean.Lattice.PartyCircuit.gateApproximation_error_real_pow_le}
    \leanok
    Let $L\ge2$, $N,n\in\N$, and $N\le C L^n$. For $p\in\R$ and
    $\lceil x\rceil_+=\max(0,\lceil x\rceil)$, set
    $q=n+\lceil p\rceil_++\lceil16C\rceil_+$; then $4L^{-q}N\le L^{-p}/4$,
    with $q$ independent of $L$.
\end{lemma}
\begin{proof}
    \leanok
    With $s=\lceil16C\rceil_+$, the bounds $16C\le s\le2^s\le L^s$ give
    $4L^{-q}N\le(16C/L^s)L^{-\lceil p\rceil_+}/4\le L^{-p}/4$.
\end{proof}

\begin{lemma}[Approximate original gates at real inverse-power accuracy]
    \label{thm:peps_approximate_original_real_accuracy}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.rectangularTraceNorm_rescaledOriginal_density_sub_le_real_pow}
    \leanok
    Suppose a chronological contraction circuit has $N\le C L^n$ nonprivate
    gate occurrences, where $L\ge2$. Fix $p\in\R$, choose $q$ as in
    Lemma~\ref{thm:peps_approximate_gate_real_accuracy}, let each original
    gate $G_g$ be approximated in operator norm to accuracy $\delta=L^{-q}$ by
    an allowed expansion $T_g$, and rescale its coefficients by $(1+L^{-q})^{-1}$.
    Then for $\|x\|\le1$,
    $\|\widetilde\rho_{L^{-q}}(x)-\rho_G(x)\|_1\le L^{-p}/4$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_approximate_gate_real_accuracy,thm:peps_approximate_physical_error}
    Lemmas~\ref{thm:peps_approximate_physical_error} and
    \ref{thm:peps_approximate_gate_real_accuracy} give
    $\|\widetilde\rho_{L^{-q}}(x)-\rho_G(x)\|_1\le4L^{-q}N\le L^{-p}/4$.
\end{proof}

\begin{definition}[Termwise bounds on the original gate expansions]
    \label{def:peps_original_termwise_bounds}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.IsTermwiseBounded}
    \leanok
    An original contraction circuit has termwise bounds $(r,C)$ if, at
    every nonprivate gate occurrence $g$, each monomial $M_{g,j}$ in its
    prescribed expansion contains at most $r$ pair effects and satisfies
    $|c_{g,j}|\le C$.
\end{definition}

\begin{lemma}[Coefficient sums of gate expansions]
    \label{thm:peps_gate_coefficient_sum}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.sum_norm_coefficients_le}
    \lean{TNLean.Lattice.PartyCircuit.PartyGate.sum_norm_coefficients_le_of_power_bounds}
    \leanok
    For a prescribed gate expansion $\sum_{j=1}^{k}c_jM_j$ with $|c_j|\le C$
    for all $j$, $\sum_{j=1}^{k}|c_j|\le kC$. In particular, if $L,C_C\ge0$,
    $u,v\in\N$, $k\le C_KL^u$, and $|c_j|\le C_CL^v$ for all $j$, then the
    coefficient estimate used in assumption~3 and the opening reduction of
    \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} gives
    \begin{align}
        \sum_{j=1}^{k}|c_j|\leq C_KC_CL^{u+v}.
        \label{eq:peps_termwise_polynomial_sum}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Summing $|c_j|\le C$ over $1\le j\le k$ gives $kC$, and multiplying
    $k\le C_KL^u$ by $C_CL^v\ge0$ yields (\ref{eq:peps_termwise_polynomial_sum}).
\end{proof}

\begin{lemma}[Uniform coefficient sums for the original circuit]
    \label{thm:peps_original_coefficient_sum}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isExpansionBounded_of_isTermwiseBounded}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isExpansionBounded_of_termwise_power_bounds}
    \uses{def:peps_original_termwise_bounds}
    \leanok
    Suppose an original circuit has $k_g\le K$ at each nonprivate gate
    occurrence $g$ and termwise bounds $(r,C)$ with $C\ge0$. Then every such
    gate satisfies $\sum_j|c_{g,j}|\le KC\le S$ for any $S\ge KC$, and each
    monomial has at most $r$ pair effects. Consequently, if $K\le C_KL^u$ and
    the termwise bounds are $(r,C_CL^v)$ with $L,C_C\ge0$, the uniform
    coefficient-sum bound is $C_KC_CL^{u+v}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:peps_original_termwise_bounds,thm:peps_gate_coefficient_sum}
    Lemma~\ref{thm:peps_gate_coefficient_sum} bounds $\sum_j|c_{g,j}|\le KC$
    at each nonprivate gate occurrence $g$, and (\ref{eq:peps_termwise_polynomial_sum})
    gives $C_KC_CL^{u+v}$ under the power bounds; composition, spectator
    registers, private operations, and register permutations preserve these
    local bounds.
\end{proof}
