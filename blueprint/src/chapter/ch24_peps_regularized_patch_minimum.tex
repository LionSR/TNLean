\section{Minimum norm of regularized regional filters}
\label{sec:peps_regularized_patch_minimum}

Let $V$ be a finite ordered set of sites, with finite-dimensional coordinate
Hilbert spaces $\mathcal H_v$, and set
$\mathcal H_X=\bigotimes_{v\in X}\mathcal H_v$.
The empty tensor product is $\C$.
For $X\subseteq V$, use the regional extension $\widehat K_X$ of
Definition~\ref{def:peps_regional_operator_extension}.
Fix regions $X_0,\ldots,X_{m-1}$ and a unit vector $\Omega\in\mathcal H_V$.
The following argument proves attainment and the elementary norm bounds in the
\href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L68-L99}{regularized variational problem}
of the September 24, 2026 polynomial-PEPS manuscript.
% Source: 03-patches.tex, lines 51--114, commit
% adc7f1241b42e322a6451854ab7e4b4c146bf78a.
% These are the finite-dimensional minimum and elementary bounds only, not
% stationarity, energy, slow regulator growth, retained rank, or Proposition 4.1.

\begin{definition}[Feasible densities and ordered filters]
    \label{def:peps_regularized_patch_objective}
    \lean{
        TNLean.PEPS.regularizedPatchDomain,
        TNLean.PEPS.regularizedPatchFilter,
        TNLean.PEPS.regularizedPatchOperator,
        TNLean.PEPS.regularizedPatchOutput,
        TNLean.PEPS.regularizedPatchObjective
    }
    \leanok
    \uses{def:peps_regional_operator_extension}
    For $b>0$ and $a_0,\ldots,a_{m-1}\in\R$ with $A=\sum_j a_j$, define
    \begin{align}
        \mathcal D & =\prod_{j=0}^{m-1}
        \{x_j\in\operatorname{End}(\mathcal H_{X_j}):x_j\ge0,\ \tr x_j=1\},         \notag \\
        K_j(x)     & =(x_j+bI_{X_j})^{-a_j/2},\qquad L_j(x)=\widehat{K_j(x)}_{X_j}, \notag \\
        M(x)       & =L_{m-1}(x)\cdots L_0(x),                                      \notag \\
        \Psi(x)    & =M(x)\Omega,\qquad F(x)=\|\Psi(x)\|
        \label{eq:peps_regularized_patch_objective}
    \end{align}
    (with $M(x)=I$ and $F(x)=1$ when $m=0$).

    For two occurrences $X_0=X_1=X$ with $K_j=(x_j+bI_X)^{-a_j/2}$, the upper
    wire below carries $\mathcal H_X$ and the lower wire carries
    $\mathcal H_{V\setminus X}$:
    % Formula: ((K_1 tensor I)(K_0 tensor I) Omega)_{i,alpha}
    % = sum_{j,k} (K_1)_{i,j}(K_0)_{j,k} Omega_{k,alpha}.
    % Source: 03-patches.tex lines 73--90 at adc7f1241b42e322a6451854ab7e4b4c146bf78a.
    % Ink-to-index map: upper row is the repeated regional physical space;
    % lower row is its complement. Omega has those two physical output legs.
    % Each K_j acts only on the upper row; the aligned lower I is the
    % complementary identity in that factor. Action starts at Omega on the right.
    % Contractions: the four internal physical wires; no swaps or crossings.
    % Open legs: i on the upper left and alpha on the lower left.
    % Boundary signature: phys:w, phys:w; output lies in H_X tensor H_(V\X).
    % This repeated-region example assumes no commutation and asserts no nesting
    % condition for the general family of regions.
    \begin{center}
        \tenkzkernel
        \begin{tenkz}[rows={wire,wire}, cols=5, bonds=none]
            \tn[at=(1,1), name=filterone, skin=box,
                ports={180:physical, 0:physical}]{K_1}
            \tn[at=(1,3), name=filterzero, skin=box,
                ports={180:physical, 0:physical}]{K_0}
            \tn[at=(2,1), name=identityone, skin=box,
                ports={180:physical, 0:physical}]{I}
            \tn[at=(2,3), name=identityzero, skin=box,
                ports={180:physical, 0:physical}]{I}
            \tn[at=(1,5), name=reference, skin=box, wires=2,
                ports={180@1:physical, 180@2:physical}]{\Omega}
            \tnwire[name=insideoutput]{open w}{filterone.180}
            \tnwire{filterone.0}{filterzero.180}
            \tnwire{filterzero.0}{reference.180@1}
            \tnwire[name=outsideoutput]{open w}{identityone.180}
            \tnwire{identityone.0}{identityzero.180}
            \tnwire{identityzero.0}{reference.180@2}
            \tnmark[form=label, label pos=n]{on insideoutput 0.5}{$i$}
            \tnmark[form=label, label pos=s]{on outsideoutput 0.5}{$\alpha$}
        \end{tenkz}
    \end{center}
    Reading right to left from $\Omega$, $\Psi(x)=(K_1\otimes I)(K_0\otimes I)\Omega$.
\end{definition}

For a single-qubit example ($X=V$, $b=1$, $a_0=a_1=2$,
$x_0=\ketbra{0}{0}$, $x_1=\ketbra{+}{+}$ with
$\ket{+}=(\ket{0}+\ket{1})/\sqrt2$),
\begin{align}
    K_0 & =\begin{pmatrix}1/2&0\\0&1\end{pmatrix},                &
    K_1 & =\begin{pmatrix}3/4&-1/4\\-1/4&3/4\end{pmatrix}, \notag
\end{align}
so $(K_1K_0)_{01}=-1/4\ne-1/8=(K_0K_1)_{01}$.

\begin{lemma}[Compactness and continuity]
    \label{lem:peps_regularized_patch_compact_continuous}
    \lean{
        TNLean.PEPS.isCompact_regularizedPatchDomain,
        TNLean.PEPS.regularizedPatchDomain_nonempty,
        TNLean.PEPS.continuousOn_regularizedPatchFilter,
        TNLean.PEPS.continuousOn_regularizedPatchObjective
    }
    \leanok
    \uses{def:peps_regularized_patch_objective}
    The set $\mathcal D$ is compact and nonempty, and each $L_j$ and $F$ are
    continuous on $\mathcal D$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regional_extension_norm}
    Each factor of $\mathcal D$ is compact in finite dimension and contains
    $I_{X_j}/\dim\mathcal H_{X_j}$ (since $\|\Omega\|=1$ implies
    $\dim\mathcal H_{X_j}>0$). Since $x_j+bI_{X_j}\ge bI_{X_j}>0$ on $\mathcal D$,
    $K_j$, $L_j$, $M$, and $F$ are continuous.
\end{proof}

\begin{lemma}[Bounds for every feasible tuple]
    \label{thm:peps_regularized_patch_bounds}
    \lean{
        TNLean.PEPS.regularizedPatchFilter_norm_bounds,
        TNLean.PEPS.regularizedPatchObjective_bounds,
        TNLean.PEPS.regularizedPatchObjective_pos
    }
    \leanok
    \uses{def:peps_regularized_patch_objective}
    If $a_j\ge0$ for all $j$, then for every $x\in\mathcal D$, $j$, and
    $\xi\in\mathcal H_V$,
    \begin{align}
        (1+b)^{-a_j/2}\|\xi\|
         & \le\|L_j(x)\xi\|\le b^{-a_j/2}\|\xi\|,
        \label{eq:peps_regularized_patch_filter_bounds} \\
        0<(1+b)^{-A/2}
         & \le F(x)\le b^{-A/2}.
        \label{eq:peps_regularized_patch_objective_bounds}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regional_extension_norm}
    Since $0\le x_j\le I_{X_j}$, the spectrum of $x_j+bI_{X_j}$ lies in
    $[b,1+b]$, so $\|K_j(x)\|\le b^{-a_j/2}$ and $\|K_j(x)^{-1}\|\le(1+b)^{a_j/2}$.
    Lemma~\ref{thm:peps_regional_extension_norm} gives
    (\ref{eq:peps_regularized_patch_filter_bounds}), and iterating across
    $L_0,\ldots,L_{m-1}$ on $\|\Omega\|=1$ gives
    (\ref{eq:peps_regularized_patch_objective_bounds}).
\end{proof}

\begin{definition}[Minimum and normalization]
    \label{def:peps_regularized_patch_minimum}
    \lean{
        TNLean.PEPS.regularizedPatchMinimum,
        TNLean.PEPS.normalizedRegularizedPatchOutput
    }
    \leanok
    \uses{def:peps_regularized_patch_objective}
    Define
    \begin{align}
        N               & =\inf_{x\in\mathcal D}F(x), &
        \widehat\Psi(x) & =F(x)^{-1}\Psi(x).
        \label{eq:peps_regularized_patch_minimum_def}
    \end{align}
\end{definition}

\begin{lemma}[Attainment and common value]
    \label{thm:peps_regularized_patch_minimum}
    \lean{
        TNLean.PEPS.exists_isMinOn_regularizedPatchObjective,
        TNLean.PEPS.regularizedPatchObjective_eq_minimum,
        TNLean.PEPS.regularizedPatchMinimum_bounds,
        TNLean.PEPS.regularizedPatchMinimum_pos,
        TNLean.PEPS.norm_normalizedRegularizedPatchOutput,
        TNLean.PEPS.normalizedRegularizedPatchOutput_eq_minimum_smul
    }
    \leanok
    \uses{def:peps_regularized_patch_minimum}
    There exists $x_*\in\mathcal D$ attaining $F(x_*)=N=\min_{x\in\mathcal D}F(x)$,
    where $\widehat\Psi(x_*)=N^{-1}M(x_*)\Omega$. If $a_j\ge0$ for all $j$, then
    \begin{align}
        0<(1+b)^{-A/2}      & \le N\le b^{-A/2},       &
        \|\widehat\Psi(x)\| & =1\quad(x\in\mathcal D).
        \label{eq:peps_regularized_patch_minimum_bounds}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:peps_regularized_patch_compact_continuous,thm:peps_regularized_patch_bounds}
    Combine Lemma~\ref{lem:peps_regularized_patch_compact_continuous} with
    (\ref{eq:peps_regularized_patch_objective_bounds}).
\end{proof}

\begin{corollary}[Exponential regularization]
    \label{cor:peps_regularized_patch_exp_bounds}
    \lean{
        TNLean.PEPS.regularizedPatchObjective_exp_bounds,
        TNLean.PEPS.regularizedPatchMinimum_exp_bounds
    }
    \leanok
    \uses{def:peps_regularized_patch_minimum}
    If $a_j\ge0$ for all $j$ and $b=e^{-R}$ ($R\in\R$), then every
    $x\in\mathcal D$ satisfies
    \begin{align}
        (1+e^{-R})^{-A/2} & \le F(x)\le e^{RA/2},          &
        2^{-A/2}          & \le N\le e^{RA/2}\quad(R\ge0).
        \label{eq:peps_regularized_patch_exp_bounds}
    \end{align}
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:peps_regularized_patch_bounds,thm:peps_regularized_patch_minimum}
    Set $b=e^{-R}$ in (\ref{eq:peps_regularized_patch_objective_bounds}) and
    (\ref{eq:peps_regularized_patch_minimum_bounds}), using $1+e^{-R}\le2$ for
    $R\ge0$.
\end{proof}

\begin{lemma}[The logarithmic minimum]
    \label{thm:peps_regularized_patch_log_minimum}
    \lean{TNLean.PEPS.isLeast_log_regularizedPatchObjective}
    \leanok
    \uses{def:peps_regularized_patch_minimum}
    If $a_j\ge0$ for all $j$, then $\log N=\min_{x\in\mathcal D}\log F(x)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regularized_patch_minimum}
    Apply monotonicity of $\log$ on $(0,\infty)$ to
    Lemma~\ref{thm:peps_regularized_patch_minimum}.
\end{proof}

\begin{lemma}[Cylinder approximation from ordered inverse filters]
    \label{thm:peps_patch_inverse_cylinder}
    \lean{TNLean.PEPS.exists_mem_iSup_dependentRegionCylinder_norm_sub_le}
    \leanok
    \uses{def:peps_regularized_patch_objective,def:peps_regional_cylinder}
    In the first-head expansion of the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/03-patches.tex\#L484-L532}{proof of Proposition 4.1},
    let $X_0\subseteq\cdots\subseteq X_{m-1}$, $a_j\ge0$, $b>0$, $A=\sum_ja_j$,
    and $\|\Omega\|=1$. Every $x\in\mathcal D$ admits subspaces
    $S_j\subseteq\mathcal H_{X_j}$ with $\dim S_j\le1/b$ and
    $v\in\sum_j\mathcal C_{X_j}(S_j)$ satisfying (including non-minimizers $x$,
    repeated regions, $a_j=0$, singular $x_j$, and $m=0$)
    \begin{align}
        \|\Omega-v\| & \le F(x)\,(2b)^{A/2}.
        \label{eq:peps_patch_inverse_cylinder}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_regional_extension_algebra,
        thm:peps_regional_extension_norm,thm:peps_cylinder_projector}
    Split $G_j=(x_j+bI)^{a_j/2}=H_j+T_j$ into the spectral part $H_j$ on
    eigenvalues of $x_j$ in $[b,1]$ ($\operatorname{rank}H_j\le1/b$ since
    $\tr x_j=1$) and the tail $T_j$ ($\|T_j\|\le(2b)^{a_j/2}$). Since
    $G_0\cdots G_{m-1}=M(x)^{-1}$, we have $\Omega=G_0\cdots G_{m-1}\Psi(x)$ and
    \begin{align}
        G_0G_1\cdots G_{m-1}
         & =\sum_{j=0}^{m-1}T_0\cdots T_{j-1}\,H_j\,G_{j+1}\cdots G_{m-1}
        +T_0\cdots T_{m-1}.
        \label{eq:peps_patch_first_head_expansion}
    \end{align}
    Since $X_0\subseteq\cdots\subseteq X_{m-1}$, the $j$th summand of
    (\ref{eq:peps_patch_first_head_expansion}) applied to $\Psi(x)$ lies in
    $\mathcal C_{X_j}(S_j)$ for $S_j=\operatorname{ran}(T_0\cdots T_{j-1}H_j|_{X_j})$
    ($\dim S_j\le\operatorname{rank}H_j\le1/b$), while
    $\|T_0\cdots T_{m-1}\Psi(x)\|\le(2b)^{A/2}F(x)$, giving
    (\ref{eq:peps_patch_inverse_cylinder}).
\end{proof}
