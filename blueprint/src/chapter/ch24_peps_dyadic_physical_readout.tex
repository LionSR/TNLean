\section{Physical coordinates after folding}
\label{sec:peps_dyadic_physical_readout}

The physical registers of a dyadic protocol are indexed by its parties,
whereas the output operator is indexed by configurations of the original
square. The following identities give the precise change of coordinates
used in \cite[Section~8]{OpenAI2026PolynomialPEPS}.

\begin{definition}[Local physical readout]
    \label{def:peps_dyadic_local_readout}
    \leanok
    \uses{def:peps_dyadic_protocol}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.physicalReadout}
    Let $s(v)$ be the party carrying a genuine site $v$, let $d_p$ be the
    physical dimension at party $p$, and let
    $E:\prod_p\{0,\ldots,d_p-1\}\to\{0,\ldots,q-1\}^{[L]^2}$
    be the protocol's configuration bijection. Define
    $j_p(i)=(E^{-1}(v\mapsto i))_p$.
\end{definition}

\begin{theorem}[Physical coordinates]
    \label{thm:peps_dyadic_physical_coordinates}
    \leanok
    \uses{def:peps_dyadic_protocol}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv_apply_val}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv_symm_apply_site_val}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv_symm_apply_eq_of_not_mem_range}
    For every party configuration $\sigma$, square configuration $x$ and
    genuine site $v$, $(E\sigma)_v=\sigma_{s(v)}$ and
    $(E^{-1}x)_{s(v)}=x_v$, with the physical coordinate identified by
    $d_{s(v)}=q$. At a party outside the image of $s$, the inverse coordinate
    is independent of $x$.
\end{theorem}
\begin{proof}\leanok
    The forward configuration bijection reads the physical coordinate of
    $s(v)$ at $v$, which gives the forward identity. Applying its inverse
    gives the inverse-coordinate identity.
    Outside the image of $s$, the physical dimension is one.
\end{proof}

\begin{theorem}[Folded readout]
    \label{thm:peps_dyadic_folded_readout}
    \leanok
    \uses{def:peps_dyadic_local_readout}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.foldVertex_anchor_site}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.physicalReadout_fold_eq_symm}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.bijective_physicalReadout_fold}
    Suppose $L>0$, and let $f(p)$ be the folded dyadic anchor of party $p$.
    Then $f(s(v))=v$ and $j_p(x_{f(p)})=(E^{-1}x)_p$ for every party $p$.
    Consequently $x\mapsto(j_p(x_{f(p)}))_p$ is a bijection.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_dyadic_physical_coordinates}
    A genuine site's party has scale zero and that site's address, so
    folding its anchor returns the site. At such a party the inverse-coordinate
    identity gives $j_{s(v)}(x_v)=(E^{-1}x)_{s(v)}$. At every remaining party
    both coordinates are equal because $d_p=1$. The full map is therefore
    $E^{-1}$ and is bijective.
\end{proof}
