\chapter{Positive local constraints from the spectral gap}%
\label{ch:peps_area_law_constraints}

The second stage of the proof of the area law replaces the Hamiltonian by a
sum of positive operators, each of which annihilates the ground state, that
keep a spectral gap and whose tails decay rapidly in the graph distance. This
chapter constructs these constraints and proves the two conditional operator
estimates of \cite[Section~5]{OpenAI2026AreaLaw}.

This chapter follows Section~4 of the two-dimensional area-law manuscript
(Propositions~4.3 and~4.5 and Lemma~4.4), first on a general finite graph $G$
and then, in Section~\ref{sec:al_lattice_constraints}, on induced lattice
domains. Let $H=\sum_i h_i$ with $h_i=h_i^*$ acting on $X_i\ni a_i$, where
$\operatorname{diam}_G(X_i)\le R$, $|X_i|\le v_R$,
$\sum_{i:x\in X_i}\|h_i\|\le b_0$, $\|h_i\|\le J$, and spheres in $G$ grow at
most polynomially. Assume $H\Omega=E_0\Omega$ for $\|\Omega\|=1$ and
$H-E_0I\ge\Delta(I-P_\Omega)$ with $\Delta>0$ and
$P_\Omega=|\Omega\rangle\langle\Omega|$. Write $E_{i,l}$ for the normalized
partial-trace expectation onto $N_l(a_i)$
(Definition~\ref{def:qc_site_expectation}) and $\alpha=p/(p+1)$ for a fixed
integer $p\ge1$.

Filtering each $h_i$ by a smooth spectral filter of width $\Delta/2$ with time
kernel $f(t)=O(e^{-c|t|^\alpha})$ gives
\begin{align}
    M_i=\int_{\mathbb R}f(t)e^{itH}h_ie^{-itH}\,dt-\langle\Omega,h_i\Omega\rangle I,
    \qquad k_i=|M_i|/c_*,\qquad c_*=\max\{1,J(\|f\|_1+1)\}.
    \label{eq:al_filtered_constraint}
\end{align}



\begin{theorem}[Positive replacement of the Hamiltonian]
    \label{thm:al_positive_replacement}
    \lean{TNLean.PEPS.AreaLaw.commute_filterIntegral_of_forall_commute,
        TNLean.PEPS.AreaLaw.positiveConstraint_mem_supportedOperators_component,
        TNLean.PEPS.AreaLaw.localizationPrefactor,
        TNLean.PEPS.AreaLaw.localizationPrefactor_nonneg,
        TNLean.PEPS.AreaLaw.exists_positiveQuasilocalConstraints}
    \leanok
    \uses{thm:qc_graph_ball_localization, lem:al_site_expectation_order,
        lem:qc_lr_closed_region}
    There are constants $C\ge0$ and $c>0$, depending only on
    $p,R,v_R,b_0,J,\Delta$ and the sphere-growth constants of $G$, such that the
    operators $k_i$ of~(\ref{eq:al_filtered_constraint}) and
    $k_{i,l}=E_{i,l}(k_i)$ ($l\in\mathbb N$) satisfy
    \begin{align}
        0\le k_i\le I,\qquad k_i\Omega=0,\qquad
        \sum_ik_i\ge c_*^{-1}(H-E_0I)\ge\frac{\Delta}{c_*}(I-P_\Omega),
        \label{eq:al_positive_replacement_gap} \\
        0\le k_{i,l}\le I,\qquad
        \|k_i-k_{i,l}\|\le Ce^{-cl^\alpha},
        \label{eq:al_positive_replacement_tail}
    \end{align}
    and each $k_i$ acts on the connected component of $a_i$.
\end{theorem}
\begin{proof}\leanok
    Since the filter multiplier $\chi(E-E_0)$ in~(\ref{eq:al_filtered_constraint})
    vanishes for $|E-E_0|\ge\Delta$ and equals $1$ at $E=E_0$, we have
    $M_i\Omega=0$, $\sum_iM_i=H-E_0I$, and $\|M_i\|\le c_*$. Because
    $M_i\le|M_i|$ and $\ker M_i\subseteq\ker|M_i|$,~(\ref{eq:al_positive_replacement_gap})
    follows. Integrating the localization bound of
    Theorem~\ref{thm:qc_graph_ball_localization} against $|f(t)|$ (split at
    $|t|\asymp l$) and applying
    $\||A|-|B|\|\le((\|A\|+\|B\|)\|A-B\|)^{1/2}$ with
    $E_{i,l}(|E_{i,l}M_i|)=|E_{i,l}M_i|$ (Lemma~\ref{lem:al_site_expectation_order})
    yields~(\ref{eq:al_positive_replacement_tail}). By
    Lemma~\ref{lem:qc_lr_closed_region}, $e^{itH}h_ie^{-itH}$ commutes with all
    operators outside the connected component of $a_i$, hence so do $M_i$ and
    $k_i$.
\end{proof}

\begin{lemma}[Square roots and local channels]
    \label{lem:al_quasilocal_roots}
    \lean{TNLean.PEPS.AreaLaw.sqrt_mul_exp_neg, TNLean.PEPS.AreaLaw.quasilocalRoots}
    \leanok
    \uses{thm:al_positive_replacement, lem:al_site_expectation_order,
        thm:al_root_channel_commutant}
    Set $G_i=(I-k_i)^{1/2}$, $K_i=k_i^{1/2}$, $G_{i,l}=(I-k_{i,l})^{1/2}$, and
    $K_{i,l}=k_{i,l}^{1/2}$. These lie in $[0,I]$ and satisfy
    $G_i\Omega=\Omega$, $K_i\Omega=0$, $G_{i,l}^2+K_{i,l}^2=I$, and
    \begin{align}
        \|G_i-G_{i,l}\|+\|K_i-K_{i,l}\|
        \le2\|k_i-k_{i,l}\|^{1/2}\le2C^{1/2}e^{-(c/2)l^\alpha}.
        \label{eq:al_root_tail}
    \end{align}
    The unital completely positive maps $\mathcal E_i(B)=G_iBG_i+K_iBK_i$ and
    $\mathcal E_{i,l}(B)=G_{i,l}BG_{i,l}+K_{i,l}BK_{i,l}$ are completely
    contractive, satisfy
    $\|(\mathcal E_i-\mathcal E_{i,l})\otimes\mathrm{id}\|\le4\|k_i-k_{i,l}\|^{1/2}$,
    and satisfy $\mathcal E_{i,l}(B)=B$ (respectively
    $(\mathcal E_{i,l}\otimes\mathrm{id})(B)=B$) whenever $[B,A]=0$ (respectively
    $[B,A\otimes I]=0$) for all $A$ acting on $N_l(a_i)$
    (Theorem~\ref{thm:al_root_channel_commutant}).
\end{lemma}
\begin{proof}\leanok
    The bound $\|U^{1/2}-V^{1/2}\|\le\|U-V\|^{1/2}$ and~(\ref{eq:al_positive_replacement_tail})
    yield~(\ref{eq:al_root_tail}). Since $v\mapsto Gv\oplus Kv$ is an isometry,
    $\mathcal E(B)$ (and $\mathcal E\otimes\mathrm{id}$) is a compression of
    $B\oplus B$, while $ABA-A'BA'=(A-A')BA+A'B(A-A')$ bounds
    $(\mathcal E_i-\mathcal E_{i,l})\otimes\mathrm{id}$. By
    Lemma~\ref{lem:al_site_expectation_order}, $G_{i,l}$ and $K_{i,l}$ act on
    $N_l(a_i)$, so any $B$ commuting with $A$ (or $A\otimes I$) for all $A$ on
    $N_l(a_i)$ commutes with both Kraus operators and is fixed by
    $G_{i,l}^2+K_{i,l}^2=I$.
\end{proof}

\section{Truncation near a finite set}
\label{sec:al_truncation}

Fix a site set $S_0$ and an integer $r_0\ge0$. For each label $i$ with
$d_i^0=d(a_i,S_0)<\infty$, set $r_i=\max\{r_0,\lfloor d_i^0/2\rfloor\}$,
$\widetilde h_i=k_{i,r_i}$, and $\widetilde X_i=N_{r_i}(a_i)$; if
$d_i^0=\infty$, set $\widetilde h_i=k_i$ and let $\widetilde X_i$ be the
connected component of $a_i$. Assume $|N_d(x)|\le K_b(d+1)^2$ for all $x,d$ and
at most $\mu$ labels share any anchor.

\begin{lemma}[Volume-free truncation error]
    \label{lem:al_truncation_error}
    \lean{TNLean.PEPS.AreaLaw.exists_one_add_pow_mul_exp_neg_rpow_le,
        TNLean.PEPS.AreaLaw.exists_sum_shell_le, TNLean.PEPS.AreaLaw.setDist,
        TNLean.PEPS.AreaLaw.exists_mem_edist_eq_setDist,
        TNLean.PEPS.AreaLaw.truncationRadius, TNLean.PEPS.AreaLaw.truncatedConstraint,
        TNLean.PEPS.AreaLaw.card_label_shell_le,
        TNLean.PEPS.AreaLaw.norm_sum_truncatedConstraint_sub_le}
    \leanok
    \uses{thm:al_positive_replacement}
    If $\|k_i-k_{i,l}\|\le Ce^{-cl^\alpha}$ for all $l$, then
    $\|\sum_i\widetilde h_i-\sum_ik_i\|\le C'|S_0|e^{-(c/2)r_0^\alpha}$ with $C'$
    depending only on $C,c,\alpha,K_b,\mu$.
\end{lemma}
\begin{proof}\leanok
    At distance $d<\infty$ from $S_0$ there are at most $\mu|S_0|K_b(d+1)^2$
    labels, while $d_i^0=\infty$ gives $\widetilde h_i-k_i=0$. Using
    $\max\{r_0,u\}^\alpha\ge(r_0^\alpha+u^\alpha)/2$,
    \begin{align}
        \sum_{d\ge0}(d+1)^2e^{-c\max\{r_0,\lfloor d/2\rfloor\}^\alpha}
        \le e^{-(c/2)r_0^\alpha}\sum_{d\ge0}(d+1)^2e^{-(c/2)\lfloor d/2\rfloor^\alpha}
        <\infty.
        \notag
    \end{align}
\end{proof}

\begin{theorem}[Truncation near a finite set]
    \label{thm:al_truncation}
    \lean{TNLean.PEPS.AreaLaw.half_le_two_rpow,
        TNLean.PEPS.AreaLaw.exists_truncationConstant,
        TNLean.PEPS.AreaLaw.truncatedConstraint_mem_Icc,
        TNLean.PEPS.AreaLaw.exists_finiteSetTruncation,
        TNLean.PEPS.AreaLaw.kernelExponent_one,
        TNLean.PEPS.AreaLaw.exists_finiteSetTruncation_of_gap,
        Matrix.PosSemidef.eq_inner_smul_of_gap,
        Matrix.PosSemidef.eigenspace_eq_span_of_gap}
    \leanok
    \uses{lem:al_truncation_error, thm:al_positive_replacement,
        thm:area_law_ground_eigenspace}
    Take $p=1$ ($\alpha=1/2$) in Theorem~\ref{thm:al_positive_replacement}, set
    $H_F=\sum_i k_i$, $\widetilde H=\sum_i\widetilde h_i$, and $g=\Delta/c_*$,
    and fix $C_0>0$. There is $C_1$ such that for all $n\ge2$ and $S_0$ with
    $|S_0|\le C_0n^2$ (the source assumes $S_0\ne\varnothing$; for
    $S_0=\varnothing$, $\widetilde h_i=k_i$), setting
    $r_0=\lceil C_1(\log n)^2\rceil$ and $\epsilon_n=\min\{n^{-1000},g/4\}$
    gives $0\le\widetilde h_i\le I$, $\|\widetilde H-H_F\|\le\epsilon_n$, and a
    unique ground state $\widetilde\Omega$ of $\widetilde H$ with ground energy
    $e_0\in[0,\epsilon_n]$, gap at least $g/2$, and (after a phase choice)
    \begin{align}
        \|\widetilde\Omega-\Omega\|\le2(\epsilon_n/g)^{1/2},\qquad
        \frac12\||\widetilde\Omega\rangle\langle\widetilde\Omega|-P_\Omega\|_1
        \le(2\epsilon_n/g)^{1/2}.
        \label{eq:al_truncation_state_closeness}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    By Lemma~\ref{lem:al_truncation_error},
    $\|\widetilde H-H_F\|\le C'C_0n^2e^{-(c/2)\sqrt{r_0}}\le C'C_0n^{2-(c/2)\sqrt{C_1}}\le\epsilon_n$
    for $C_1$ large enough. Since $\epsilon_n\le g/4$, a ground minimizer
    $\widetilde\Omega$ of $\widetilde H$ has $e_0\in[0,\epsilon_n]$, orthogonal
    excitation energy at least $g-\epsilon_n\ge g/2$, and
    $g(1-|\langle\Omega,\widetilde\Omega\rangle|^2)\le2\epsilon_n$, yielding~(\ref{eq:al_truncation_state_closeness}).
    By Theorem~\ref{thm:area_law_ground_eigenspace}, any $\psi$ with
    $\widetilde H\psi=e_0\psi$ satisfies, for
    $\varphi=\psi-\langle\widetilde\Omega,\psi\rangle\widetilde\Omega$,
    \begin{align}
        0=\langle\varphi,(\widetilde H-e_0I)\varphi\rangle
        \ge\frac g2\|\varphi\|^2,
        \notag
    \end{align}
    so $\psi=\langle\widetilde\Omega,\psi\rangle\widetilde\Omega$.
\end{proof}

\begin{theorem}[A common auxiliary label in physical coordinates]
    \label{thm:al_global_schmidt_bell_prevector}
    \lean{TNLean.PEPS.AreaLaw.exists_global_schmidtBellPrevector}
    \leanok
    Let $V$ be a finite site set of local dimension $q>0$, and let $\Omega$ be a
    unit vector with $H\Omega=E\Omega$. Fix $X\subseteq V$, select a subset $F$
    of a spectral basis of the reduced density on $X$ with total mass $z>0$ and
    $d=|F|$, and let $\psi\in\mathcal H_{X^c}\otimes\mathbb C^d$ be the
    normalized selected Schmidt vector with marginal $\rho$ on $\mathbb C^d$.
    For each $k>0$ and irreducible representation $\lambda$ of $S_k$, let
    $P_\lambda$ be the central $\lambda$-isotypic projection on
    $(\mathbb C^d)^{\otimes k}$ \cite[Section~6]{OpenAI2026AreaLaw}. Following
    \cite[Section~8 and Section~9, lines~50--59]{OpenAI2026AreaLaw}, there is a
    sequence $\lambda_k$ such that, setting
    \begin{align}
        \omega_d                                         & =d^{-1/2}\sum_{a=0}^{d-1}|a,a\rangle,        &
        L_k                                              & =P_{\lambda_k},                       \notag   \\
        \xi_k                                            & =(I\otimes I\otimes L_k)
        (\Omega^{\otimes k}\otimes\omega_d^{\otimes k}), &
        \overline H_k                                    & =k^{-1}\sum_{j=1}^k H_j,
        \label{eq:al_schmidt_bell_defs}
    \end{align}
    identified on $V$ and two auxiliary sites, with copy-symmetrizer
    $P_{\mathrm{sym}}$, every $k>0$ satisfies
    \begin{align}
        \xi_k\ne0,\qquad \|\xi_k\|\le1,\qquad
        (\overline H_k\otimes I)\xi_k=E\xi_k,
        \label{eq:al_schmidt_bell_energy} \\
        P_{\mathrm{sym}}\xi_k=\xi_k,\qquad
        (I\otimes L_k\otimes I)\xi_k=\xi_k,\qquad
        (I\otimes I\otimes L_k)\xi_k=\xi_k,
        \label{eq:al_schmidt_bell_invariance}
    \end{align}
    and eventually in $k$,
    \begin{align}
        \|(I\otimes L_k)\psi^{\otimes k}\|^2
        \ge\frac1{2(k+1)^{d^2}},\qquad
        \log\dim\lambda_k-kS(\rho)=o(k).
        \label{eq:al_schmidt_bell_asymptotics}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Since $z>0$, $\|\psi\|=1$, and common-label selection for $\rho$ gives~(\ref{eq:al_schmidt_bell_asymptotics}).
    In replica coordinates, $u_k=(I\otimes I\otimes L_k)(\Omega^{\otimes k}\otimes\omega_d^{\otimes k})$
    satisfies $0<\|u_k\|\le1$, $(\overline H_k\otimes I)u_k=Eu_k$,
    $(I\otimes L_k\otimes I)u_k=(I\otimes I\otimes L_k)u_k=u_k$, and
    simultaneous-permutation invariance. If $U$ is the configuration isometry
    with $\xi_k=Uu_k$, then
    \begin{align}
        Ku_k=cu_k\ \Longrightarrow\ (UKU^{-1})\xi_k=c\,\xi_k,
        \label{eq:al_schmidt_bell_transport}
    \end{align}
    which transports these four identities to~(\ref{eq:al_schmidt_bell_energy})--(\ref{eq:al_schmidt_bell_invariance}).
\end{proof}

\begin{theorem}[One physical label sequence for the rough lower comparison]
    \label{thm:al_global_schmidt_bell_rough_lower}
    \lean{TNLean.PEPS.AreaLaw.exists_global_schmidtBellPrevector_rough_lower}
    \leanok
    \uses{thm:al_global_schmidt_bell_prevector}
    In the setting of Theorem~\ref{thm:al_global_schmidt_bell_prevector},
    suppose $H-EI\ge\gamma(I-|\Omega\rangle\langle\Omega|)$ with $\gamma>0$, and
    let $\lambda_k,\xi_k$ be given by
    Theorem~\ref{thm:al_global_schmidt_bell_prevector}. With $S'=S(\rho)$ and
    $\varepsilon_k=-2(\log\dim\lambda_k-kS')$, we have
    $2kS'-\varepsilon_k=2\log\dim\lambda_k$ and $\varepsilon_k/k\to0$.

    Fix $K\ge8$, $a=4/K$, and $t=2/K$. There are constants $c,C_f\ge0$ such that
    for any isometry $Z$ with $Z^*Z=I$ and $ZZ^*=P_{\mathrm{sym}}$, any regions
    $Q_{jg},Y_{jg}\subseteq V$ ($0\le g<K$) satisfying
    $Q_{jg}\cap Y_{jg}=\varnothing$ and $Q_{jg}\cup Y_{jg}\subseteq Q_{j'h}$ for
    $g<h$, with $L_{jg}=Q_{jg}\cup\{C\}$,
    $F_{jg}=(V\setminus(Q_{jg}\cup Y_{jg}))\cup\{R\}$, replica metric $W$ at
    parameter $t$, and any iterated weighted geometric mean $T$, setting
    \begin{align}
        A_{jg}    & =(W_{L_{jg}}^{-1}W_{F_{jg}}^{-1}W_{Y_{jg}})^2,
                  & B_{jg}                                         & =Z^*A_{jg}Z,
                  & M_k                                            & =\prod_{g=0}^{K-1}T(B_{\cdot g}),
                  & w_k                                            & =M_k^{-1/4}Z^*\xi_k,
        \label{eq:al_rough_lower_ops}                                                                                                  \\
        \varphi_k & =\frac{w_k}{\|w_k\|},\qquad v_k=Z\varphi_k,
                  & \delta_k                                       & =\frac{\langle v_k,(\overline H_k\otimes I)v_k\rangle-E}{\gamma},
        \notag                                                                                                                         \\
        r_k       & =\log(k+1)+c\log(k+2)+a\varepsilon_k,
                  & R_k                                            & =\frac K{2k}\bigl(2|r_k|+C_f\log(k+2)+2\log2+\log(3/2)\bigr),
        \label{eq:al_rough_lower_remainder}
    \end{align}
    gives $R_k\to0$ (independently of $Z,Q_{jg},Y_{jg},T$), and for every $k>0$
    and $0<\tau\le1/2$, $w_k\ne0$, $\|v_k\|=1$, $\delta_k\ge0$, and
    \begin{align}
        4S'-6\left(\tau\log d+\frac{h(\tau)}a\right)
        -3\frac{\delta_k}{\tau}\max(0,2S')-R_k
        \le-\frac{\log\|w_k\|^2}{k}
        \label{eq:al_rough_lower_bound}
    \end{align}
    \cite[Section~8]{OpenAI2026AreaLaw}. Choosing a scan point and identifying
    its regions and geometric mean remain separate assertions.
\end{theorem}
\begin{proof}\leanok
    Take $\lambda_k$ and $\xi_k$ from
    Theorem~\ref{thm:al_global_schmidt_bell_prevector}. Since $z>0$, $d>0$. The
    identities~(\ref{eq:al_schmidt_bell_invariance}) verify the rough lower
    comparison hypotheses at $t=2/K$ and exponent $1/4$, and the coefficient-$3$
    estimate applied to $v_k$ and $\delta_k$ gives~(\ref{eq:al_rough_lower_bound}).
    Finally, $\varepsilon_k/k\to0$ by~(\ref{eq:al_schmidt_bell_asymptotics}), so
    $R_k\to0$ by~(\ref{eq:al_rough_lower_remainder}).
\end{proof}

    $\mathcal C_{V\setminus X}=\mathcal C_X$,
    $\mathcal B_{V\setminus X}=\mathcal B_X\ge1$, and
    $\mathcal B_\varnothing=\mathcal B_V=1$.
\end{theorem}
\begin{proof}\leanok
    Using $\prod_{v\in\widetilde X_i}q=q^{|\widetilde X_i|}$, invariance of
    $\mathcal C_X$ under $X\leftrightarrow V\setminus X$, nonnegativity of
    squares, and $\mathcal C_\varnothing=\mathcal C_V=\varnothing$ gives all
    claims.
\end{proof}

\section{The induced lattice instance}
\label{sec:al_lattice_constraints}

Let $\Lambda\subset\Z^2$ be a finite domain with its induced nearest-neighbour
graph, and let $H=\sum_ih_i$ be an interaction of range $R$ and strength $J$ as
in Definition~\ref{def:area_law_hamiltonian}, with one term for every nonempty
support of induced-graph diameter at most $R$. Choose an anchor $a_i\in X_i$
for every support, and let $\Omega$ be a unit vector with $H\Omega=E_0\Omega$
and $H-E_0I\ge\Delta(I-P_\Omega)$, $\Delta>0$. Put $v_R=1+2R(R+1)$ and
$\mu_R=2^{v_R-1}$. In this setting the counting hypotheses of the previous
sections are consequences of the lattice geometry, and the constants depend
only on the parameters named below.

\begin{theorem}[Counting on an induced lattice domain]
    \label{thm:al_lattice_counts}
    \lean{TNLean.PEPS.AreaLaw.card_graphBall_domainGraph_le,
        TNLean.PEPS.AreaLaw.card_anchor_fiber_le,
        TNLean.PEPS.AreaLaw.card_cutEdges_le_card_edgeBoundary}
    \leanok
    \uses{def:area_law_hamiltonian, thm:area_diamond_count, thm:area_diamond_budget,
        lem:area_domain_graph_counts, thm:area_law_crossing_supports}
    In $\Lambda$, $|N_d(x)|\le2(d+1)^2$, at most $\mu_R$ supports contain a
    given site (Theorem~\ref{thm:area_law_crossing_supports}), at most $\mu_R$
    labels share an anchor, and
    $|\{(u,v):u\sim v,\,u\in X,\,v\notin X\}|\le|\partial_\Lambda X|$. The
    sphere, support-diameter, support-size, and per-site norm bounds are given
    in Lemma~\ref{lem:area_domain_graph_counts}.
\end{theorem}
\begin{proof}\leanok
    Since $u\sim v\implies\|u-v\|_1=1$, $N_d(x)$ lies in the $\ell^1$-ball of
    radius $d$ (Theorem~\ref{thm:area_diamond_count}). Supports and anchors are
    bounded by Theorem~\ref{thm:area_diamond_budget} since $a_i\in X_i$, and
    $(u,v)\mapsto\{u,v\}$ injects crossing pairs into $\partial_\Lambda X$.
\end{proof}

\begin{theorem}[Positive replacement on an induced lattice domain]
    \label{thm:al_lattice_positive_replacement}
    \lean{TNLean.PEPS.AreaLaw.exists_latticePositiveConstraints}
    \leanok
    \uses{thm:al_positive_replacement, lem:area_domain_graph_counts, def:area_law_ground_entropy}
    Fix $p\in\mathbb N_{\ge1}$, $\alpha=p/(p+1)$, $R,J\ge0$, and $\Delta>0$.
    There are $C\ge0$ and $c>0$, depending only on $p,R,J,\Delta$, such that for
    every $q\ge1$, finite domain $\Lambda$, interaction, anchor choice, and
    gapped ground vector $\Omega$, the constraints $k_i=|M_i|/c_*$ satisfy
    $c_*>0$ and
    \begin{align}
        0\le k_i\le I,\qquad k_i\Omega=0,\qquad
        H_F=\sum_ik_i\ge c_*^{-1}(H-E_0I)\ge g(I-P_\Omega)\quad(g=\Delta/c_*),
        \label{eq:al_lattice_positive_replacement_gap} \\
        0\le k_{i,l}=E_{i,l}(k_i)\le I,\qquad
        \|k_i-k_{i,l}\|\le Ce^{-cl^\alpha}\quad(l\in\mathbb N),
        \label{eq:al_lattice_positive_replacement_tail}
    \end{align}
    with $H_F\Omega=0$ and each $k_i$ acting on the connected component of
    $a_i$ \cite[Proposition~4.3]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:al_positive_replacement} with $v_R$, $b_0=\mu_RJ$,
    and sphere growth $2(d+1)^2$ from Lemma~\ref{lem:area_domain_graph_counts}.
\end{proof}

\begin{theorem}[Square roots and local channels on an induced lattice domain]
    \label{thm:al_lattice_quasilocal_roots}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeQuasilocalRoots}
    \leanok
    \uses{lem:al_quasilocal_roots, thm:al_lattice_positive_replacement,
        thm:al_root_channel_commutant}
    With $p,R,J,\Delta$ as in Theorem~\ref{thm:al_lattice_positive_replacement},
    there are $C\ge0$ and $c>0$ depending only on $p,R,J,\Delta$ such that the
    roots $G_i,K_i,G_{i,l},K_{i,l}$ and channels $\mathcal E_i,\mathcal E_{i,l}$
    of Lemma~\ref{lem:al_quasilocal_roots}, built from the constraints of
    Theorem~\ref{thm:al_lattice_positive_replacement}, satisfy all conclusions
    of Lemma~\ref{lem:al_quasilocal_roots} with
    \begin{align}
        \|G_i-G_{i,l}\|+\|K_i-K_{i,l}\|\le Ce^{-cl^\alpha},\qquad
        \|(\mathcal E_i-\mathcal E_{i,l})\otimes\mathrm{id}\|\le Ce^{-cl^\alpha}
        \notag
    \end{align}
    for every auxiliary system \cite[Lemma~4.4]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Substitute~(\ref{eq:al_lattice_positive_replacement_tail}) into
    Lemma~\ref{lem:al_quasilocal_roots} using
    $(Ce^{-cl^\alpha})^{1/2}=C^{1/2}e^{-(c/2)l^\alpha}$.
\end{proof}

\begin{theorem}[Truncation near a finite set on an induced lattice domain]
    \label{thm:al_lattice_truncation}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeFiniteSetTruncation,
        TNLean.PEPS.AreaLaw.exists_latticeFiniteSetTruncation_eigenspace_eq_span}
    \leanok
    \uses{thm:al_truncation, thm:al_lattice_counts, lem:area_domain_graph_counts,
        def:area_law_ground_entropy, thm:area_law_ground_eigenspace}
    Take $p=1$ ($\alpha=1/2$) in
    Theorem~\ref{thm:al_lattice_positive_replacement} and fix $C_0\ge0$. There
    is $C_1>0$, depending only on $R,J,\Delta,C_0$, such that for every lattice
    instance, $n\ge2$, and $S_0\subseteq\Lambda$ with $|S_0|\le C_0n^2$, setting
    $r_0=\lceil C_1(\log n)^2\rceil$, $g=\Delta/c_*$, and
    $\epsilon_n=\min\{n^{-1000},g/4\}$ gives $0\le\widetilde h_i\le I$ acting on
    $\widetilde X_i$, $\|\widetilde H-H_F\|\le\epsilon_n$, and a unit ground
    vector $\widetilde\Omega$ of $\widetilde H$ with ground energy
    $e_0\in[0,\epsilon_n]$, gap at least $g/2$,
    $\ker(\widetilde H-e_0I)=\operatorname{span}\{\widetilde\Omega\}$, and (after
    a phase choice) the bounds~(\ref{eq:al_truncation_state_closeness}). This is
    the truncation part of \cite[Proposition~4.5]{OpenAI2026AreaLaw} (stated
    there for $S_0\ne\varnothing$ and $C_0>0$).
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:al_truncation} with $v_R$, $b_0=\mu_RJ$, and sphere
    growth $2(d+1)^2$ (Lemma~\ref{lem:area_domain_graph_counts}), and
    $K_b=2,\mu=\mu_R$ (Theorem~\ref{thm:al_lattice_counts}). If
    $\widetilde H\psi=e_0\psi$ and
    $\varphi=\psi-\langle\widetilde\Omega,\psi\rangle\widetilde\Omega$, then
    $0=\langle\varphi,(\widetilde H-e_0I)\varphi\rangle\ge(g/2)\|\varphi\|^2$,
    so $\psi\in\operatorname{span}\{\widetilde\Omega\}$ by
    Theorem~\ref{thm:area_law_ground_eigenspace}.
\end{proof}

\begin{theorem}[Crossing count and cut parameter on an induced lattice domain]
    \label{thm:al_lattice_crossing_budget}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeCrossingBudget,
        TNLean.PEPS.AreaLaw.exists_latticeCutBudget_le_log}
    \leanok
    \uses{thm:al_crossing_budget, thm:al_lattice_counts}
    For $q\ge1$ there is $C\ge0$, depending only on $q,R$, such that for every
    finite domain $\Lambda$, anchor choice, $S_0,r_0$, and $X\subseteq S_0$ with
    $b_X=|\partial_\Lambda X|$, every $i\in\mathcal C_X$ has
    $\widetilde X_i=N_{r_0}(a_i)$ and
    \begin{align}
        |\mathcal C_X|\le Cb_X(r_0+1)^2,\qquad
        \log d_i\le C(r_0+1)^2\quad(i\in\mathcal C_X),\qquad
        \mathcal B_X\le C(1+b_X)(r_0+1)^6.
        \label{eq:al_lattice_crossing_bounds}
    \end{align}
    For $C_1,C_2\ge0$ there is $C'$, depending only on $q,R,C_1,C_2$, such that
    $r_0=\lceil C_1(\log n)^2\rceil$, $n\ge2$, $D\ge1$, and $b_X\le C_2nD$
    imply $\mathcal B_X\le C'nD(\log n)^{12}$
    \cite[Proposition~4.5]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:al_crossing_budget} with $K_b=2$, $\mu=\mu_R$, and
    cut-edge bound $|\partial_\Lambda X|$ from Theorem~\ref{thm:al_lattice_counts};
    substituting $r_0=\lceil C_1(\log n)^2\rceil$ into~(\ref{eq:al_lattice_crossing_bounds})
    gives the second bound.
\end{proof}
