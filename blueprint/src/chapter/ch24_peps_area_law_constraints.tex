\chapter{Positive quasi-local constraints from a global gap}%
\label{ch:peps_area_law_constraints}

This chapter follows Section~4 of the two-dimensional area-law manuscript
(Propositions 4.3 and 4.5 and Lemma 4.4), first on a general finite graph and
then, in Section~\ref{sec:al_lattice_constraints}, on the induced lattice
domains of the source. Let $G$ be a finite graph and
$H=\sum_i h_i$ a sum of Hermitian terms, where $h_i$ acts on a support $X_i$
with anchor $a_i\in X_i$, every support has graph diameter at most $R$ and at
most $v_R$ sites, the norms of the terms meeting a site sum to at most $b_0$,
$\|h_i\|\le J$, and the spheres of $G$ grow at most polynomially. Suppose that
$H\Omega=E_0\Omega$ for a unit vector $\Omega$ and that
$H-E_0I\ge\Delta(I-P_\Omega)$ with $\Delta>0$, where
$P_\Omega=|\Omega\rangle\langle\Omega|$. Write $E_{i,l}$ for the normalized
partial-trace expectation onto the graph ball $N_l(a_i)$
(Definition~\ref{def:qc_site_expectation}) and $\alpha=p/(p+1)$ for a fixed
integer $p\ge1$.

The construction filters each term with a smooth spectral filter of width
$\Delta/2$ whose time kernel $f$ decays like $e^{-c|t|^\alpha}$, removes its
ground expectation, and takes the absolute value:
\[
    M_i=\int_{\mathbb R}f(t)e^{itH}h_ie^{-itH}\,dt-\langle\Omega,h_i\Omega\rangle I,
    \qquad k_i=|M_i|/c_*,\qquad c_*=\max\{1,J(\|f\|_1+1)\}.
\]

\begin{lemma}[Order and commutant of the ball expectation]
    \label{lem:al_site_expectation_order}
    \lean{QuantumCircuit.heisenbergEvolution_eq_hermitianUnitaryPath,
        QuantumCircuit.siteExpectation_conjTranspose,
        QuantumCircuit.isHermitian_siteExpectation,
        QuantumCircuit.siteExpectation_nonneg, QuantumCircuit.siteExpectation_mono,
        QuantumCircuit.siteExpectation_le_one,
        QuantumCircuit.siteExpectation_eq_self_of_forall_commute,
        QuantumCircuit.mem_supportedOperators_of_forall_commute,
        Commute.cfcAbs_of_isHermitian,
        QuantumCircuit.siteExpectation_abs_of_mem_supportedOperators,
        QuantumCircuit.sqrt_mem_supportedOperators}
    \leanok
    \uses{lem:qc_site_expectation_average, lem:qc_site_expectation_properties}
    The expectation $E_K$ onto a set of sites $K$ preserves adjoints and the
    order, maps $[0,I]$ into $[0,I]$, and fixes every operator commuting with
    all operators acting outside $K$; such an operator acts on $K$. In
    particular $E_K$ fixes $|Y|$ for every Hermitian $Y$ acting on $K$, and
    $Y^{1/2}$ acts on $K$ for every $Y\ge0$ acting on $K$.
\end{lemma}
\begin{proof}\leanok
    $E_K$ is the uniform average of the conjugations by products of on-site
    Weyl operators outside $K$, and its values act on $K$.
\end{proof}

\begin{theorem}[Positive replacement of the Hamiltonian]
    \label{thm:al_positive_replacement}
    \lean{TNLean.PEPS.AreaLaw.commute_filterIntegral_of_forall_commute,
        TNLean.PEPS.AreaLaw.positiveConstraint_mem_supportedOperators_component,
        TNLean.PEPS.AreaLaw.localizationPrefactor,
        TNLean.PEPS.AreaLaw.localizationPrefactor_nonneg,
        TNLean.PEPS.AreaLaw.exists_positiveQuasilocalConstraints}
    \leanok
    \uses{thm:qc_graph_ball_localization, lem:al_site_expectation_order,
        lem:qc_lr_closed_region}
    There are constants $C\ge0$ and $c>0$, depending only on
    $p,R,v_R,b_0,J,\Delta$ and the growth of the spheres, such that
    \begin{align}
        0\le k_i\le I,\qquad k_i\Omega=0,\qquad
        \sum_ik_i\ge c_*^{-1}(H-E_0I)\ge\frac{\Delta}{c_*}(I-P_\Omega),
    \end{align}
    and for every integer $l\ge0$ the operator $k_{i,l}=E_{i,l}(k_i)$ lies in
    $[0,I]$ and satisfies $\|k_i-k_{i,l}\|\le Ce^{-cl^\alpha}$. Each $k_i$ acts
    on the connected component of $a_i$.
\end{theorem}
\begin{proof}\leanok
    The spectral filter multiplies the matrix element of $h_i$ between energies
    $E$ and $E_0$ by $\chi(E-E_0)$, which vanishes across the gap and equals
    $1$ at $E=E_0$; hence $M_i\Omega=0$, $\sum_iM_i=H-E_0I$ and
    $\|M_i\|\le c_*$. Since $M_i\le|M_i|$ and the kernel of $M_i$ lies in that
    of $|M_i|$, the first display follows. For the tail, the graph-ball
    localization of the dynamics (Theorem~\ref{thm:qc_graph_ball_localization})
    is integrated against $|f|$, splitting the time integral at $|t|$ of order
    $l$; the square-root estimate
    $\||A|-|B|\|\le((\|A\|+\|B\|)\|A-B\|)^{1/2}$ transfers the bound to the
    absolute value, because $E_{i,l}$ fixes $|E_{i,l}M_i|$. Finally the
    Heisenberg evolution of $h_i$ commutes with every operator acting outside
    the component of $a_i$ (Lemma~\ref{lem:qc_lr_closed_region}), hence so do
    the filtered term, its centering and its absolute value.
\end{proof}

\begin{lemma}[Square roots and local channels]
    \label{lem:al_quasilocal_roots}
    \lean{TNLean.PEPS.AreaLaw.sqrt_mul_exp_neg, TNLean.PEPS.AreaLaw.quasilocalRoots}
    \leanok
    \uses{thm:al_positive_replacement, lem:al_site_expectation_order,
        thm:al_root_channel_commutant}
    Let $G_i=(I-k_i)^{1/2}$, $K_i=k_i^{1/2}$, $G_{i,l}=(I-k_{i,l})^{1/2}$ and
    $K_{i,l}=k_{i,l}^{1/2}$. These are positive contractions, $G_i\Omega=\Omega$,
    $K_i\Omega=0$, $G_{i,l}^2+K_{i,l}^2=I$, and
    $\|G_i-G_{i,l}\|+\|K_i-K_{i,l}\|\le2\|k_i-k_{i,l}\|^{1/2}
        \le2C^{1/2}e^{-(c/2)l^\alpha}$. The maps
    $\mathcal E_i(B)=G_iBG_i+K_iBK_i$ and
    $\mathcal E_{i,l}(B)=G_{i,l}BG_{i,l}+K_{i,l}BK_{i,l}$ are unital and
    completely positive, contractive after adjoining any auxiliary system, and
    $\|(\mathcal E_i-\mathcal E_{i,l})\otimes\mathrm{id}\|\le4\|k_i-k_{i,l}\|^{1/2}$
    uniformly in the auxiliary dimension. The map $\mathcal E_{i,l}$ fixes every
    operator commuting with all operators acting on $N_l(a_i)$, and
    $\mathcal E_{i,l}\otimes\mathrm{id}$ fixes every operator commuting with
    $A\otimes I$ for all operators $A$ acting on $N_l(a_i)$
    (Theorem~\ref{thm:al_root_channel_commutant}).
\end{lemma}
\begin{proof}\leanok
    The square-root estimate $\|U^{1/2}-V^{1/2}\|\le\|U-V\|^{1/2}$ gives the
    root tails. The map $v\mapsto Gv\oplus Kv$ is an isometry, and
    $\mathcal E(B)$ is the compression of $B\oplus B$ by it, also after
    tensoring with an identity. The difference bound follows from
    $ABA-A'BA'=(A-A')BA+A'B(A-A')$. An operator commuting with the ball algebra
    commutes with $k_{i,l}$, hence with both local roots. Both local roots
    commute with every operator acting outside the ball, so they act on the
    ball; with an auxiliary system, an operator commuting with $A\otimes I$ for
    all such $A$ therefore commutes with both Kraus operators of
    $\mathcal E_{i,l}\otimes\mathrm{id}$, whose squares sum to the identity.
\end{proof}

\section{Truncation near a finite set}
\label{sec:al_truncation}

Fix a set $S_0$ of sites and an integer $r_0\ge0$. For a label whose
anchor has finite distance $d_i^0=d(a_i,S_0)$ from $S_0$ put
$r_i=\max\{r_0,\lfloor d_i^0/2\rfloor\}$, $\widetilde h_i=k_{i,r_i}$ and
$\widetilde X_i=N_{r_i}(a_i)$; if $d_i^0=\infty$ put $\widetilde h_i=k_i$ and let
$\widetilde X_i$ be the connected component of $a_i$. Suppose every graph ball
of radius $d$ has at most $K_b(d+1)^2$ sites and every site anchors at most
$\mu$ labels.

\begin{lemma}[Volume-free truncation error]
    \label{lem:al_truncation_error}
    \lean{TNLean.PEPS.AreaLaw.exists_one_add_pow_mul_exp_neg_rpow_le,
        TNLean.PEPS.AreaLaw.exists_sum_shell_le, TNLean.PEPS.AreaLaw.setDist,
        TNLean.PEPS.AreaLaw.exists_mem_edist_eq_setDist,
        TNLean.PEPS.AreaLaw.truncationRadius, TNLean.PEPS.AreaLaw.truncatedConstraint,
        TNLean.PEPS.AreaLaw.card_label_shell_le,
        TNLean.PEPS.AreaLaw.norm_sum_truncatedConstraint_sub_le}
    \leanok
    \uses{thm:al_positive_replacement}
    If $\|k_i-k_{i,l}\|\le Ce^{-cl^\alpha}$ for all $l$, then
    $\|\sum_i\widetilde h_i-\sum_ik_i\|\le C'|S_0|e^{-(c/2)r_0^\alpha}$ with $C'$
    depending only on $C,c,\alpha,K_b,\mu$.
\end{lemma}
\begin{proof}\leanok
    At distance $d$ from $S_0$ there are at most $\mu|S_0|K_b(d+1)^2$ labels,
    and labels at infinite distance have no error. Since
    $\max\{r_0,u\}^\alpha\ge(r_0^\alpha+u^\alpha)/2$, the series
    $\sum_d(d+1)^2e^{-c\max\{r_0,\lfloor d/2\rfloor\}^\alpha}$ is at most
    $e^{-(c/2)r_0^\alpha}\sum_d(d+1)^2e^{-(c/2)\lfloor d/2\rfloor^\alpha}$, and
    the last series converges because stretched exponentials beat polynomials.
\end{proof}

\begin{theorem}[Truncation near a finite set]
    \label{thm:al_truncation}
    \lean{TNLean.PEPS.AreaLaw.half_le_two_rpow,
        TNLean.PEPS.AreaLaw.exists_truncationConstant,
        TNLean.PEPS.AreaLaw.truncatedConstraint_mem_Icc,
        TNLean.PEPS.AreaLaw.exists_finiteSetTruncation,
        TNLean.PEPS.AreaLaw.kernelExponent_one,
        TNLean.PEPS.AreaLaw.exists_finiteSetTruncation_of_gap}
    \leanok
    \uses{lem:al_truncation_error, thm:al_positive_replacement}
    Use Theorem~\ref{thm:al_positive_replacement} with $p=1$, so $\alpha=1/2$,
    and fix $C_0>0$. There is $C_1$ such that for every real $n\ge2$ and every
    set $S_0$ with $|S_0|\le C_0n^2$ (the source states this for nonempty $S_0$; for
    $S_0=\varnothing$ every constraint is retained), with
    $r_0=\lceil C_1(\log n)^2\rceil$ and
    $\epsilon_n=\min\{n^{-1000},g/4\}$: the truncated terms are positive
    contractions, $\|\widetilde H-H_F\|\le\epsilon_n$, the ground energy of
    $\widetilde H$ lies in $[0,\epsilon_n]$, its ground vector
    $\widetilde\Omega$ is unique, its gap is at least $g/2$, and after a choice
    of phase $\|\widetilde\Omega-\Omega\|\le2(\epsilon_n/g)^{1/2}$ and
    $\frac12\||\widetilde\Omega\rangle\langle\widetilde\Omega|-P_\Omega\|_1
        \le(2\epsilon_n/g)^{1/2}$.
\end{theorem}
\begin{proof}\leanok
    Lemma~\ref{lem:al_truncation_error} bounds the error by
    $C'C_0n^2e^{-(c/2)\sqrt{r_0}}\le C'C_0n^{2-(c/2)\sqrt{C_1}}$, and $C_1$ is
    chosen once to make this at most $\epsilon_n$ for all $n\ge2$. The
    remaining claims follow from the stability of a global gap under a
    perturbation of norm at most a quarter of the gap: a minimizer of the
    energy is an eigenvector, every vector orthogonal to it has energy at
    least $g-\epsilon_n$, and the gap of $H_F$ at $\widetilde\Omega$ gives
    $g(1-|\langle\Omega,\widetilde\Omega\rangle|^2)\le2\epsilon_n$.
\end{proof}

For $X\subseteq S_0$ let $b_X$ be the number of edges with exactly one endpoint
in $X$, let $\mathcal C_X$ be the labels whose designated support meets both
$X$ and its complement, and put $d_i=q^{|\widetilde X_i|}$ and
$\mathcal B_X=1+\sum_{i\in\mathcal C_X}\log^2(ed_i)$.

\begin{theorem}[Crossing count and cut budget]
    \label{thm:al_crossing_budget}
    \lean{TNLean.PEPS.AreaLaw.cutEdges, TNLean.PEPS.AreaLaw.componentFinset,
        TNLean.PEPS.AreaLaw.designatedSupport, TNLean.PEPS.AreaLaw.crossingLabels,
        TNLean.PEPS.AreaLaw.exists_cut_edge_near,
        TNLean.PEPS.AreaLaw.truncationRadius_eq_of_mem_crossingLabels,
        TNLean.PEPS.AreaLaw.card_crossingLabels_le, TNLean.PEPS.AreaLaw.cutBudget,
        TNLean.PEPS.AreaLaw.log_dim_le_of_mem_crossingLabels,
        TNLean.PEPS.AreaLaw.cutBudget_le, TNLean.PEPS.AreaLaw.le_of_cutBudget_le,
        TNLean.PEPS.AreaLaw.truncatedConstraint_empty,
        TNLean.PEPS.AreaLaw.crossingLabels_empty,
        TNLean.PEPS.AreaLaw.truncatedConstraint_of_setDist_eq_top,
        TNLean.PEPS.AreaLaw.truncatedConstraint_mem_supportedOperators}
    \leanok
    \uses{thm:al_truncation}
    Each truncated term $\widetilde h_i$ acts on its designated support
    $\widetilde X_i$. Every support in $\mathcal C_X$ is a ball of radius exactly $r_0$,
    $|\mathcal C_X|\le2\mu K_b(r_0+1)^2b_X$, $\log d_i\le K_b(r_0+1)^2\log q$
    for $i\in\mathcal C_X$, and $\mathcal B_X\le C(1+b_X)(r_0+1)^6$. If moreover
    $r_0=\lceil C_1(\log n)^2\rceil$, $n\ge2$, $D\ge1$ and $b_X\le C_2nD$, then
    $\mathcal B_X\le C'nD(\log n)^{12}$. If $S_0$ is empty every constraint is
    retained and no label crosses a cut contained in $S_0$; constraints whose
    anchors cannot reach $S_0$ are always retained exactly.
\end{theorem}
\begin{proof}\leanok
    A support meeting $X\subseteq S_0$ has an anchor at finite distance; if
    $r_i>r_0$ then $r_i=\lfloor d_i^0/2\rfloor<d_i^0$ and the ball cannot meet
    $S_0$. A shortest path from the anchor to a point of the ball on the other
    side of the cut stays in the ball and contains a cut edge, so the anchor is
    within $r_0$ of one of the at most $2b_X$ endpoints of cut edges. Each such
    ball has at most $K_b(r_0+1)^2$ sites and each anchor carries at most $\mu$
    labels.
\end{proof}

% Exact-budget identities preserve the original finite label family.
% These identities do not assert the lattice-specialized budget producer.
\begin{theorem}[Exact cut budget and complementary cuts]
    \label{thm:al_exact_cut_budget}
    \lean{TNLean.PEPS.AreaLaw.cutBudget_eq_cutLogBudget,
        TNLean.PEPS.AreaLaw.crossingLabels_compl,
        TNLean.PEPS.AreaLaw.cutBudget_compl,
        TNLean.PEPS.AreaLaw.one_le_cutBudget,
        TNLean.PEPS.AreaLaw.crossingLabels_univ,
        TNLean.PEPS.AreaLaw.cutBudget_empty,
        TNLean.PEPS.AreaLaw.cutBudget_univ}
    \leanok
    \uses{thm:al_crossing_budget}
    Let $V$ be the vertex set of a finite graph, let $I$ be a finite label
    set with anchors $a_i\in V$, and let $q,r_0\in\mathbb N$ and
    $S_0\subseteq V$. Use the designated supports $\widetilde X_i$ of
    Theorem~\ref{thm:al_crossing_budget}. For every cut $X\subseteq V$, let
    $\mathcal C_X$ consist of the original labels whose designated support
    meets both $X$ and $V\setminus X$. Then the geometric budget is exactly
    the supported-family budget in \cite[Lemma~3.1]{OpenAI2026AreaLaw}:
    \begin{align}
        \mathcal B_X
         & =1+\sum_{i\in\mathcal C_X}
        \log^2\Bigl(\mathrm e\prod_{v\in\widetilde X_i}q\Bigr)
        =1+\sum_{i\in\mathcal C_X}\log^2(\mathrm e\,q^{|\widetilde X_i|}).
    \end{align}
    Here $\log0=0$ when $q=0$. Also,
    $\mathcal C_{V\setminus X}=\mathcal C_X$,
    $\mathcal B_{V\setminus X}=\mathcal B_X\ge1$, and
    $\mathcal B_\varnothing=\mathcal B_V=1$.
    No containment $X\subseteq S_0$ is required. Repeated designated
    supports and zero terms keep their original labels.
\end{theorem}
\begin{proof}\leanok
    The crossing conditions on each original label agree, and
    $\prod_{v\in\widetilde X_i}q=q^{|\widetilde X_i|}$. Exchanging the two
    sides of the cut exchanges the two witnesses of crossing. Every
    summand is a square, and the empty and full cuts have no crossing labels.
\end{proof}

\section{The induced lattice instance}
\label{sec:al_lattice_constraints}

Let $\Lambda\subset\Z^2$ be a finite domain with its induced nearest-neighbour
graph, and let $H=\sum_ih_i$ be an interaction of range $R$ and strength $J$ as
in Definition~\ref{def:area_law_hamiltonian}, with one term for every nonempty
support of induced-graph diameter at most $R$. Choose an anchor $a_i\in X_i$
for every support, and let $\Omega$ be a unit vector with $H\Omega=E_0\Omega$
and $H-E_0I\ge\Delta(I-P_\Omega)$, $\Delta>0$. Put $v_R=1+2R(R+1)$ and
$\mu_R=2^{v_R-1}$. In this setting the counting hypotheses of the previous
sections are consequences of the lattice geometry, and the constants depend
only on the parameters named below.

\begin{theorem}[Counting on an induced lattice domain]
    \label{thm:al_lattice_counts}
    \lean{TNLean.PEPS.AreaLaw.card_graphBall_domainGraph_le,
        TNLean.PEPS.AreaLaw.card_anchor_fiber_le,
        TNLean.PEPS.AreaLaw.card_cutEdges_le_card_edgeBoundary}
    \leanok
    \uses{def:area_law_hamiltonian, thm:area_diamond_count, thm:area_diamond_budget,
        lem:area_domain_graph_counts, thm:area_law_crossing_supports}
    Graph balls of radius $d$ in $\Lambda$ have at most $2(d+1)^2$ sites, at
    most $\mu_R$ supports contain a given site
    (Theorem~\ref{thm:area_law_crossing_supports}), at most $\mu_R$ labels share an
    anchor, and the pairs $(u,v)$ of adjacent sites with $u\in X$ and
    $v\notin X$ number at most $|\partial_\Lambda X|$. The sphere bound, the
    diameter and size bounds for supports and the site budget are
    Lemma~\ref{lem:area_domain_graph_counts}.
\end{theorem}
\begin{proof}\leanok
    Adjacent sites are at ambient $\ell^1$ distance one, so a graph ball lies in
    an ambient lattice ball and the exact diamond count applies. Supports and
    anchors are counted
    by the interaction budgets of Theorem~\ref{thm:area_diamond_budget}, and
    every label anchored at $x$ has a support containing $x$. The map
    $(u,v)\mapsto\{u,v\}$ is injective on the pairs crossing $X$.
\end{proof}

\begin{theorem}[Positive replacement on an induced lattice domain]
    \label{thm:al_lattice_positive_replacement}
    \lean{TNLean.PEPS.AreaLaw.exists_latticePositiveConstraints}
    \leanok
    \uses{thm:al_positive_replacement, lem:area_domain_graph_counts, def:area_law_ground_entropy}
    Fix an integer $p\ge1$, put $\alpha=p/(p+1)$, and fix $R\ge0$, $J\ge0$ and
    $\Delta>0$. There are $C\ge0$ and $c>0$, depending only on $p,R,J,\Delta$,
    such that for every $q\ge1$, every finite domain, interaction, choice of
    anchors and gapped ground vector as above, the constraints $k_i=|M_i|/c_*$
    satisfy $c_*>0$,
    \begin{align}
        0\le k_i\le I,\qquad k_i\Omega=0,\qquad
        \sum_ik_i\ge c_*^{-1}(H-E_0I)\ge\frac{\Delta}{c_*}(I-P_\Omega),
    \end{align}
    so $H_F=\sum_ik_i$ satisfies $H_F\ge g(I-P_\Omega)$ with $g=\Delta/c_*$ and
    $H_F\Omega=0$. For every integer $l\ge0$ the operator
    $k_{i,l}=E_{i,l}(k_i)$ lies in $[0,I]$ and satisfies
    $\|k_i-k_{i,l}\|\le Ce^{-cl^\alpha}$. Each $k_i$ acts on
    the connected component of $a_i$. This is Proposition~4.3 of
    \cite{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:al_positive_replacement} with $v_R$, $b_0=\mu_RJ$ and
    sphere growth $2(d+1)^2$, whose hypotheses on the graph are supplied by
    Lemma~\ref{lem:area_domain_graph_counts}.
\end{proof}

\begin{theorem}[Square roots and local channels on an induced lattice domain]
    \label{thm:al_lattice_quasilocal_roots}

\begin{theorem}[Truncation near a finite set on an induced lattice domain]
    \label{thm:al_lattice_truncation}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeFiniteSetTruncation}
    \leanok
    \uses{thm:al_truncation, thm:al_lattice_counts, lem:area_domain_graph_counts, def:area_law_ground_entropy}
    Take $p=1$ in Theorem~\ref{thm:al_lattice_positive_replacement}, so
    $\alpha=1/2$, and fix $C_0\ge0$. There is $C_1>0$, depending only on
    $R,J,\Delta,C_0$, such that for every lattice instance as above, every real
    $n\ge2$ and every set $S_0$ of sites with $|S_0|\le C_0n^2$, the truncation
    with $r_0=\lceil C_1(\log n)^2\rceil$ has the conclusions of
    Theorem~\ref{thm:al_truncation} with $g=\Delta/c_*$ other than uniqueness:
    it has a unit ground vector $\widetilde\Omega$ satisfying the stated gap,

    Proposition~4.5 of \cite{OpenAI2026AreaLaw}, which assumes $S_0$ nonempty

\end{theorem}
\begin{proof}\leanok

\end{proof}

\begin{theorem}[Crossing count and cut budget on an induced lattice domain]
    \label{thm:al_lattice_crossing_budget}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeCrossingBudget,
        TNLean.PEPS.AreaLaw.exists_latticeCutBudget_le_log}
    \leanok
    \uses{thm:al_crossing_budget, thm:al_lattice_counts}
    For $q\ge1$ there is $C\ge0$, depending only on $q$ and $R$, such that for
    every finite domain, every choice of anchors, every $S_0$, every $r_0$ and every
    $X\subseteq S_0$, with $b_X=|\partial_\Lambda X|$, every designated support
    meeting both $X$ and its complement is the ball of radius $r_0$ about its
    anchor and
    \begin{align}
        |\mathcal C_X|\le Cb_X(r_0+1)^2,\qquad
        \log d_i\le C(r_0+1)^2\quad(i\in\mathcal C_X),\qquad
        \mathcal B_X\le C(1+b_X)(r_0+1)^6.
    \end{align}
    For $C_1,C_2\ge0$ there is $C$, depending only on
    $q,R,C_1,C_2$, such that $r_0=\lceil C_1(\log n)^2\rceil$, $n\ge2$,
    $D\ge1$ and $b_X\le C_2nD$ imply $\mathcal B_X\le CnD(\log n)^{12}$. This
    is the crossing count and cut budget of Proposition~4.5 of
    \cite{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    Apply Theorem~\ref{thm:al_crossing_budget} with $K_b=2$ and $\mu=\mu_R$,
    whose hypotheses are supplied by Theorem~\ref{thm:al_lattice_counts}; the
    ordered cut edges are counted by $|\partial_\Lambda X|$. The second bound
    inserts $r_0=\lceil C_1(\log n)^2\rceil$ into the third inequality.
\end{proof}

\section{The scanner estimate from its inputs}
\label{sec:al_scanner}

Section~9 of the manuscript scans the collar of a finite target in rounds and
selects one charge round and one interpolation parameter at which the
transported, symmetrized vector has small defect energy (Proposition~9.2). Fix
exponents $0<\ell<1$, $0<\kappa<\mu<1-\ell$, $0<e<1$ with $\kappa\le e$,
$\nu>0$, let the total metric weight $W\ge1$ of the scan be arbitrary, possibly
depending on $n$, and put $L=\lfloor n^{1-\ell}\rfloor$,
$D=\lceil n^\kappa\rceil$, $m=\lfloor n^\mu\rfloor$, $\epsilon=n^{-\nu}$ and
$\delta_n=C\epsilon^{-1}(n^{e-\mu}+a^{1/4}(\log n)^{C})$. The statement below
takes as hypotheses, for one scan at scale $n$, the conclusions of the scan
histories (Lemma~9.1), of transport (Proposition~7.4), of the comparators
(Proposition~8.1), of continuity (Lemma~2.1) and of tails (Lemma~2.3); it is
therefore Proposition~9.2 restricted to those conclusions, not the source
proposition itself, which remains open until the scan data are constructed.
The finite transported measures are required to have mass $1/2$ only for
$0<p<1$, where the source's terminal weights are positive. The logarithmic norm is
continuous on $[0,1]$, and its values at $0$ and $1$ retain their matching
and comparison properties. No mass equality is required at an inactive
terminal leaf.

\begin{theorem}[Scanner estimate from the conclusions of its inputs]
    \label{thm:al_scanner_from_inputs}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScanData,
        TNLean.PEPS.AreaLaw.Scan.eventually_scaleFacts,
        TNLean.PEPS.AreaLaw.Scan.exists_budgets,
        TNLean.PEPS.AreaLaw.Scan.ScanData.integratedChargeBound,
        TNLean.PEPS.AreaLaw.Scan.exists_selected_density,
        TNLean.PEPS.AreaLaw.Scan.ScanData.energySum_le,
        TNLean.PEPS.AreaLaw.Scan.exists_defectEnergy_le,
        TNLean.PEPS.AreaLaw.Scan.scanner_estimate}
    \leanok
    There are constants, independent of $W$ and depending only on the exponents and on the constants of
    the inputs, such that for all sufficiently large $n$ and every scan at scale
    $n$ satisfying the conclusions of Lemmas~9.1, 2.1, 2.3 and
    Propositions~7.4, 8.1, the comparator budgets
    $B_{\mathrm{sh}}\le C(nL^e+n)$, $B_{\mathrm{exc}}\le CnD$ and
    $\mathcal B\le CnD(\log n)^{C}$ hold, and there is a remainder $\rho_k\to0$, uniform over the selected
    round and parameter, such that for every $k\ge1$ some charge round and
    some $p_k\in[\epsilon/2,\epsilon]$ satisfy
    $\mathcal Q(p_k)/(KnD)\le\delta_n+\rho_k$ and
    \begin{align}
        E_{\mathrm{def},k}\le Cn^\ell D^2W^2(\log n)^{C}
        \bigl(\delta_n^{1/8}+\epsilon+n^{-100}\bigr)+Cn^{-1000}+\rho_k.
        \label{eq:al_scanner_energy}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Integrate the logarithmic norm derivative over $[0,1]$ in every round,
    retaining the nonnegative charge contribution on
    $[\epsilon/2,\epsilon]$. Bounds and continuity on $(0,1)$ make the charge
    defect integrable; changing its values at $0$ and $1$ does not change
    the integral. Continuity of the logarithmic norm and the matching
    values at successive rounds give the telescoped entropy budget. Averaging over
    the $nm$ charge rounds and the parameter interval selects a round and a
    parameter with $\mathcal Q(p_k)/(KnD)\le\delta_n+\rho_k$. At that point
    H\"older's inequality with exponents $8/7$ and $8$ bounds the leaf
    energies of the old good leaves by $N(\mathcal Q/N)^{1/8}$, the new and
    bad leaves contribute at most $\epsilon$ and $n^{-200}$ times the total
    weight, and the gap of the copied Hamiltonian converts the energy into the
    defect energy. The budgets follow from the entropy input, the continuity
    error $O(n^{-247})$ and the crossing count, which gives the logarithmic
    power $12$ for the marginal parameter.
\end{proof}
