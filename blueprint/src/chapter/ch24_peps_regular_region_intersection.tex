\subsection{Regular intersection at one shared vertex}

Let $\Gamma=(V,E)$ be a finite simple graph and $G$ a finite group.
For $R\subseteq V$, let $\widehat{\mathcal S}_R^0$ be the regular
regional range on the ambient half-edge space (where every complementary
slice belongs to the open-region range $\mathcal S_R^0$, retaining all
half-edges crossing $\partial R$). This gives the regular-coordinate
intersection step in \cite[Theorem~5.4]{Schuch2010PEPS}.
% This regular-coordinate result does not complete Theorem 5.4 for arbitrary
% semi-regular representations or arbitrary G-injective physical site maps.

\begin{theorem}[Restriction and gluing of flat configurations]%
    \label{thm:peps_regular_flat_union}
    \lean{TNLean.PEPS.IsRegularRegionHalfEdgeFlat.of_subset,
        TNLean.PEPS.internalEdge_union_of_no_cross,
        TNLean.PEPS.isRegularRegionHalfEdgeFlat_union_iff}
    \leanok
    \uses{def:peps_regional_half_edge_flat}
    Flatness restricts to subregions. If $R\cap S=\{b\}$ and no edge
    joins $R\setminus S$ to $S\setminus R$, then every internal edge of
    $R\cup S$ is internal to $R$ or $S$, and a half-edge configuration is
    flat on $R\cup S$ if and only if it is flat on both $R$ and $S$.
\end{theorem}
\begin{proof}\leanok
    Restriction is immediate from Definition~\ref{def:peps_regional_half_edge_flat}.
    Conversely, for flatness potentials $k_R,k_S$ on $R,S$, setting
    $t=k_S(b)^{-1}k_R(b)$ and replacing $k_S(v)$ by $k_S(v)t$ makes
    the potentials agree at $b$, and every internal edge $e=(v,w)$ of $S$
    satisfies
    \begin{align}
        (k_S(v)t)^{-1}\alpha_{v,e}
         & =t^{-1}k_S(v)^{-1}\alpha_{v,e}
        =t^{-1}k_S(w)^{-1}\alpha_{w,e}
        =(k_S(w)t)^{-1}\alpha_{w,e}.
        \label{eq:peps_flat_potential_gluing}
    \end{align}
    Since no edge joins $R\setminus S$ to $S\setminus R$, every internal
    edge of $R\cup S$ lies in $R$ or $S$, so the glued potential is flat
    on $R\cup S$.
\end{proof}

\begin{theorem}[Atomic characterization and nesting of regular ranges]%
    \label{thm:peps_regular_range_nested}
    \lean{TNLean.PEPS.mem_regularGlobalRegionRange_iff_fixed_flat,
        TNLean.PEPS.regularGlobalRegionRange_antitone}
    \leanok
    \uses{def:peps_ambient_regular_range,def:peps_ambient_regional_constraints}
    A vector $x$ lies in $\widehat{\mathcal S}_R^0$ if and only if
    $A_vx=x$ for all $v\in R$, $B_ex=x$ for all internal edges $e$ of $R$,
    and $x(\alpha)=0$ whenever $\alpha|_R$ is not flat; hence
    $R\subseteq S$ implies
    $\widehat{\mathcal S}_S^0\subseteq\widehat{\mathcal S}_R^0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regular_actual_projector,thm:peps_regular_constraint_fixed_spaces,
        thm:peps_regular_flat_union}
    Theorems~\ref{thm:peps_regular_actual_projector}
    and~\ref{thm:peps_regular_constraint_fixed_spaces} replace simultaneous
    vertex and edge invariance by individual fixedness $A_vx=x$ and
    $B_ex=x$. Monotonicity follows from Theorem~\ref{thm:peps_regular_flat_union}.
\end{proof}

\begin{theorem}[Intersection of regular regional ranges]%
    \label{thm:peps_regular_region_intersection}
    \lean{TNLean.PEPS.regularGlobalRegionRange_inf_eq_union}
    \leanok
    \uses{def:peps_ambient_regular_range}
    For any finite group $G$, if $R\cap S=\{b\}$ and no edge joins
    $R\setminus S$ to $S\setminus R$, then
    \begin{align}
        \widehat{\mathcal S}_R^0\cap\widehat{\mathcal S}_S^0
         & =\widehat{\mathcal S}_{R\cup S}^0.
        \label{eq:peps_regular_region_intersection}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regular_range_nested,thm:peps_regular_flat_union}
    By Theorems~\ref{thm:peps_regular_flat_union}
    and~\ref{thm:peps_regular_range_nested}, any
    $x\in\widehat{\mathcal S}_R^0\cap\widehat{\mathcal S}_S^0$ is fixed by
    all $A_v$ ($v\in R\cup S$) and $B_e$ ($e$ internal to $R\cup S$) and
    vanishes when $\alpha|_{R\cup S}$ is not flat, giving
    $\widehat{\mathcal S}_R^0\cap\widehat{\mathcal S}_S^0\subseteq\widehat{\mathcal S}_{R\cup S}^0$;
    Theorem~\ref{thm:peps_regular_range_nested} gives the reverse inclusion.
\end{proof}

\begin{definition}[Regional conditions on the core physical space]%
    \label{def:peps_regular_core_slice_range}
    \lean{TNLean.PEPS.regularSubregionSlice,
        TNLean.PEPS.regularRegionRangeWithin,
        TNLean.PEPS.regularRegionConstantExtension}
    \leanok
    \uses{def:peps_ambient_regular_range}
    For $R\subseteq T$, let $\mathcal S_{R;T}^0$ be the space of vectors
    $x$ on the physical half-edge factors of $T$ whose slices at fixed
    coordinates on $T\setminus R$ lie in $\mathcal S_R^0$, and define
    $J_Tx$ on $\Gamma$ by $(J_Tx)(\alpha)=x(\alpha|_T)$.
\end{definition}

\begin{theorem}[Intersection on only the core physical factors]%
    \label{thm:peps_regular_core_intersection}
    \lean{TNLean.PEPS.regularRegionRangeWithin_eq_comap,
        TNLean.PEPS.regularRegionRangeWithin_self,
        TNLean.PEPS.regularRegionRangeWithin_inf_eq_range}
    \leanok
    \uses{def:peps_regular_core_slice_range}
    One has $\mathcal S_{R;T}^0=J_T^{-1}(\widehat{\mathcal S}_R^0)$ and
    $\mathcal S_{T;T}^0=\mathcal S_T^0=\operatorname{range}\Gamma_T$.
    If $R\cap S=\{b\}$, no edge joins $R\setminus S$ to $S\setminus R$,
    and $T=R\cup S$, then on the physical factors of $T$ alone (with
    arbitrary correlated boundary tensors on $\partial T$),
    \begin{align}
        \mathcal S_{R;T}^0\cap\mathcal S_{S;T}^0
         & =\operatorname{range}\Gamma_T.
        \label{eq:peps_regular_core_intersection}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regular_region_intersection}
    Complementary slices of $J_Tx$ depend only on coordinates in
    $T\setminus R$ (extending to $V\setminus R$ by $1$ outside $T$),
    giving $\mathcal S_{R;T}^0=J_T^{-1}(\widehat{\mathcal S}_R^0)$ and
    $\mathcal S_{T;T}^0=\mathcal S_T^0=\operatorname{range}\Gamma_T$.
    Pulling (\ref{eq:peps_regular_region_intersection}) back along $J_T$
    gives (\ref{eq:peps_regular_core_intersection}).
\end{proof}

\begin{definition}[Three four-legged blocks]%
    \label{def:peps_regular_three_block_geometry}
    \lean{TNLean.PEPS.threeBlockFourLegGraph,
        TNLean.PEPS.threeBlockLeftRegion,TNLean.PEPS.threeBlockRightRegion,
        TNLean.PEPS.threeBlockRegion}
    \leanok
    On vertices $0,\ldots,10$, set $R=\{0,1\}$, $S=\{1,2\}$,
    $T=\{0,1,2\}$, and
    \begin{align}
        E & =\{\{0,j\}:j\in\{1,3,4,5\}\}\cup
        \{\{1,j\}:j\in\{2,6,7\}\}\cup
        \{\{2,j\}:j\in\{8,9,10\}\},
        \label{eq:peps_three_block_edges}
    \end{align}
    so the core path $0$--$1$--$2$ has eight distinct exterior boundary
    vertices $3,\ldots,10$.
\end{definition}

\begin{theorem}[Four-legged regular intersection]%
    \label{thm:peps_regular_three_block_intersection}
    \lean{TNLean.PEPS.threeBlockFourLegGraph_core_degree,
        TNLean.PEPS.threeBlockFourLegGraph_boundary_card,
        TNLean.PEPS.threeBlock_regularGlobalRegionRange_intersection,
        TNLean.PEPS.threeBlock_regularRegionRange_intersection}
    \leanok
    \uses{def:peps_regular_three_block_geometry,def:peps_regular_core_slice_range}
    In Definition~\ref{def:peps_regular_three_block_geometry}, each core
    vertex has degree four, $T$ has eight boundary edges, and for every
    finite group $G$,
    $\widehat{\mathcal S}_R^0\cap\widehat{\mathcal S}_S^0
        =\widehat{\mathcal S}_T^0$ and
    $\mathcal S_{R;T}^0\cap\mathcal S_{S;T}^0
        =\operatorname{range}\Gamma_T$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_regular_region_intersection,thm:peps_regular_core_intersection}
    Inspection of (\ref{eq:peps_three_block_edges}) gives the degree and
    boundary counts, $R\cap S=\{1\}$, $R\cup S=T$, and $\{0,2\}\notin E$;
    Theorems~\ref{thm:peps_regular_region_intersection}
    and~\ref{thm:peps_regular_core_intersection} then yield both equalities.
\end{proof}
