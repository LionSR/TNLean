\section{Attaching the canonical parents}
\label{sec:mpo_endpoint_parent_attachment}

Let $A_p$ be one-site injective tensors of positive bond dimension $D_p$
and arbitrary physical dimension $d_p$, for $p\in\{0,1\}$. Write
$A_p=U_pB_p$ for their physical polar decompositions, where $B_p$ has
physical dimension $D_p^2$, and put $D=D_0+D_1$. The common physical
space is
\begin{align}
    \mathcal H=\C^{D^2}\oplus\C^{d_0}\oplus\C^{d_1}.
\end{align}
Let $J:\C^{D^2}\to\mathcal H$ be the inclusion of the first summand.
Let $E_p:\C^{d_p}\to\mathcal H$ be the full physical isometries
satisfying $E_pA_p=J\widehat B_p$, where $\widehat B_p$ extends $B_p$
by zero outside the $p$th diagonal physical sector.

For the unweighted mixed tensor $B$ and the virtual summand projections
$P_p$, define
\begin{align}
    S'_p=\left\{\sum_{i,j}\tr(B^iP_pB^jX)\ket{ij}:
    X\in\MN D\right\},
    \qquad h'_p=I-\Pi_{S'_p}.
\end{align}
The actual limiting interaction on $\mathcal H^{\otimes2}$ is
\begin{align}
    h_p=I-(J\otimes J)(I-h'_p)(J\otimes J)^\dagger.
\end{align}
Its distinction from the canonical parent
$k_p=I-\Pi_{\mathcal S^2_{E_pA_p}}$ is essential: their local kernels
need not have the same dimension. Their periodic kernels do agree.
This section proves the ordered attachment underlying the affine
parent-Hamiltonian comparison in
\cite[Section~5]{GarreRubio2023MPOSym}. It does not impose or establish
commutation with a common matrix product operator.

\begin{theorem}[Local comparison]
    \label{thm:mpo_arbitrary_endpoint_parent_order}
    \lean{MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_zero_le_parent,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_one_le_parent}
    For each $p\in\{0,1\}$,
    \begin{align}
        \mathcal S^2_{E_pA_p}\subseteq(J\otimes J)S'_p,
        \qquad 0\leq h_p\leq k_p\leq I.
    \end{align}
\end{theorem}
\begin{proof}
    Let $V_p:\C^{D_p}\to\C^D$ include the virtual summand, so
    $P_p=V_pV_p^\dagger$ and $V_p^\dagger B^iV_p=\widehat B_p^i$.
    For every boundary matrix $X\in\MN{D_p}$, cyclicity of the trace gives
    \begin{align}
        \tr\bigl(B^iP_pB^jV_pXV_p^\dagger\bigr)
        =\tr\bigl(\widehat B_p^i\widehat B_p^jX\bigr).
    \end{align}
    Hence $\mathcal S^2_{\widehat B_p}\subseteq S'_p$.
    Orthogonal projection onto a smaller subspace is smaller in operator
    order, so taking complements gives $h'_p\leq
        I-\Pi_{\mathcal S^2_{\widehat B_p}}$. Isometric inclusion with the
    unit penalty on the unused space preserves this inequality. The identity
    $E_pA_p=J\widehat B_p$ identifies its right side with $k_p$.
    Both operators are orthogonal projections, which gives the remaining bounds.
\end{proof}

\begin{definition}[Affine attachments]
    \label{def:mpo_arbitrary_endpoint_attachment}
    \lean{MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointParentInteraction,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_zero,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_one,
        MPSTensor.MPOSymmetry.continuous_arbitraryPhysicalEndpointAttachment,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_posSemidef,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_norm_le_one}
    \uses{thm:mpo_arbitrary_endpoint_parent_order}
    For $p\in\{0,1\}$ and $0\leq t\leq1$, set
    \begin{align}
        c_p(t)=h_p+t(k_p-h_p)=(1-t)h_p+tk_p.
    \end{align}
    Then $c_p(0)=h_p$, $c_p(1)=k_p$, and $c_p$ is continuous.
    Every $c_p(t)$ is positive and has operator norm at most one.
\end{definition}

\begin{theorem}[Exact periodic ground states and uniform gap]
    \label{thm:mpo_arbitrary_endpoint_attachment_gap}
    \lean{MPSTensor.interactionHamiltonian_affine_norm_gap_of_order,
        MPSTensor.MPOSymmetry.arbitraryPhysicalMixedInteraction_endpoint_ker_eq_parent,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_ker_eq_parent,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_groundSpace_eq_span_mpv,
        MPSTensor.MPOSymmetry.arbitraryPhysicalEndpointAttachment_groundSpace_finrank,
        MPSTensor.MPOSymmetry.exists_uniform_arbitraryPhysicalEndpointAttachment_gap}
    \uses{def:mpo_arbitrary_endpoint_attachment}
    Write $H_{p,N}$, $K_{p,N}$, and $C_{p,N}(t)$ for the periodic sums
    of $h_p$, $k_p$, and $c_p(t)$, respectively. For every $N\geq2$,
    \begin{align}
        \ker C_{p,N}(t)=\ker H_{p,N}=\ker K_{p,N}
        =\C\ket{\psi_{E_pA_p,N}},
        \qquad \dim\ker C_{p,N}(t)=1.
    \end{align}
    There exists $0<\delta\leq1$, independent of $p,t,N$, such that,
    for every $v\perp\ker C_{p,N}(t)$,
    \begin{align}
        \delta\|v\|\leq\|C_{p,N}(t)v\|.
    \end{align}
    The same lower gap bound that applies to the mixed path may be used
    for both attachments.
\end{theorem}
\begin{proof}
    The mixed path at parameter $p$ has exactly the nonzero periodic vector
    $\ket{\psi_{E_pA_p,N}}$ as its ground-state representative. Injectivity
    of $E_pA_p$ identifies the kernel of its canonical parent with the same
    line. Thus $\ker H_{p,N}=\ker K_{p,N}$.

    The local order gives $H_{p,N}\leq K_{p,N}$ and
    $H_{p,N}\leq C_{p,N}(t)$. Every common zero vector is killed by the
    affine sum; positivity gives the opposite kernel inclusion. Hence all
    three kernels agree, including for the two-site ring.

    Let $\delta$ be a uniform gap of the actual mixed path. In finite
    dimension, positivity gives, for every $v\perp\ker H_{p,N}$,
    \begin{align}
        \delta\|v\|^2\leq\langle v,H_{p,N}v\rangle.
    \end{align}
    Operator order and Cauchy--Schwarz then give
    \begin{align}
        \delta\|v\|^2
        \leq\langle v,C_{p,N}(t)v\rangle
        \leq\|v\|\,\|C_{p,N}(t)v\|.
    \end{align}
    Cancel $\|v\|$ when $v\ne0$; the case $v=0$ is immediate.
    No new loss in the lower gap bound occurs.
\end{proof}

The reverse of the first attachment and the second attachment join the
mixed path to the two canonical parents. The common MPO covariance and
fusion requirements in the definition of symmetric phase remain separate;
these attachments alone do not establish the symmetric classification.
