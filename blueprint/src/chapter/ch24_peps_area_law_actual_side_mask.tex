\section{Subdivisions determined by opposing corners}
\label{sec:area_law_actual_side_mask}

Opposing fine-cell corners determine which sides are subdivided in the cell
fans of \cite[Section~11]{OpenAI2026AreaLaw}. Fix the origin $o$, the finite
endpoint set, the dyadic layers, and an initial layer index $k_0$. Every fine
cell in a layer $k\ge k_0$ participates in this subdivision rule.

\begin{theorem}[Indexed affine parameters of an elementary side]
    \label{thm:area_law_indexed_side_interpolation}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFan_elementary_endpoints_lineMap}
    \leanok
    \uses{def:area_law_cell_fan}
    Fix an origin $o\in\mathbb R^2$, a natural exponent
    $\ell$, and a cell index $z\in\mathbb Z^2$.
    Choose an optional midpoint subdivision of each side of the cell.
    Write $A_s,B_s$ for the counterclockwise endpoints of side $s$, and set
    $L_s(u)=(1-u)A_s+uB_s$.
    Let $\varepsilon_s\in\{0,1\}$ record whether side $s$ is subdivided,
    and index its elementary segments by $h=0$ when $\varepsilon_s=0$
    and by $h\in\{0,1\}$ when $\varepsilon_s=1$.
    The initial and final endpoints of segment $(s,h)$ are
    \begin{align}
        a_{s,h}      & =L_s(\alpha_{s,h}), &
        b_{s,h}      & =L_s(\beta_{s,h}),      \\
        \alpha_{s,h} & =
        \begin{cases}
            0,   & \varepsilon_s=0, \\
            h/2, & \varepsilon_s=1,
        \end{cases}
                     & \beta_{s,h}         & =
        \begin{cases}
            1,       & \varepsilon_s=0, \\
            (h+1)/2, & \varepsilon_s=1.
        \end{cases}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{def:area_law_cell_fan}
    Parametrize side $s$ by a signed coordinate $u\in[-1,1]$, with affine
    parameter $(u+1)/2\in[0,1]$.
    An undivided side has initial and final signed coordinates $-1$ and $1$,
    giving affine parameters $0$ and $1$.
    Half $h\in\{0,1\}$ of a subdivided side has signed coordinates $h-1$
    and $h$, giving affine parameters $h/2$ and $(h+1)/2$.
\end{proof}

\begin{theorem}[Geometry of an elementary side]
    \label{thm:area_law_elementary_side_geometry}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFan_elementary_endpoints_cases,
        TNLean.PEPS.AreaLaw.Geometry.cellFan_elementary_segment_subset_whole}
    \leanok
    \uses{def:area_law_cell_fan}
    Fix a dyadic cell and an optional midpoint subdivision of each side.
    Let $[a,b]$ be a side ordered counterclockwise and $[u,v]$ one of its
    elementary segments. Either the side is undivided and $(u,v)=(a,b)$,
    or the side is subdivided and $(u,v)$ is $(a,(a+b)/2)$ or
    $((a+b)/2,b)$. In all cases $[u,v]\subseteq[a,b]$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_indexed_side_interpolation}
    In the affine parametrization of $[a,b]$ by $[0,1]$, the endpoints
    $(u,v)$ have parameters $(0,1)$ for an undivided side,
    $(0,1/2)$ for the first half, and $(1/2,1)$ for the second half.
    Each interval is contained in $[0,1]$.
\end{proof}

\begin{definition}[Midpoint subdivision by opposing corners]
    \label{def:area_law_actual_side_mask}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayerSplitMask}
    \leanok
    \uses{def:area_law_cell_fan,def:area_law_cell_corner,
        def:area_law_fine_layer_indices}
    For the fine cell indexed by $z$ in layer $k$, write $[a_s,b_s]$
    for side $s$. Set $\varepsilon_s=1$ if the midpoint
    $(a_s+b_s)/2$ is a corner of some fine cell in a layer $h\ge k_0$,
    and $\varepsilon_s=0$ otherwise. The elementary segments of side $s$
    are $[a_s,b_s]$ when $\varepsilon_s=0$, and its two halves
    $[a_s,(a_s+b_s)/2]$ and $[(a_s+b_s)/2,b_s]$ when $\varepsilon_s=1$.
\end{definition}

\begin{theorem}[Criterion for a divided side]
    \label{thm:area_law_actual_side_mask_iff}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayerSplitMask_eq_true_iff}
    \leanok
    \uses{def:area_law_actual_side_mask}
    Side $s$ of the fine cell indexed by $z$ in layer $k$ is subdivided if
    and only if there exist a layer $h\ge k_0$, an index $w\in F_{h,\ell_h}$,
    and a corner index $\varepsilon\in\{0,1\}^2$ such that corner
    $\varepsilon$ of the fine cell $w$ equals $(a_s+b_s)/2$.
\end{theorem}
\begin{proof}\leanok
    \uses{def:area_law_actual_side_mask}
    This is the defining condition for $\varepsilon_s=1$.
\end{proof}

\begin{theorem}[Opposing corners are elementary endpoints]
    \label{thm:area_law_elementary_corner_endpoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayer_corner_on_elementarySide}
    \leanok
    \uses{def:area_law_actual_side_mask}
    Suppose that $C_0\ge2$, $k\ge50\,000\,000$, and $z\in F_{k,\ell_k}$
    indexes a fine cell of layer $k$. Let $w\in F_{h,\ell_h}$ index a fine
    cell of a layer $h\ge k_0$. If a corner of the cell indexed by $w$ lies
    on an elementary segment of the cell indexed by $z$ under the midpoint
    subdivision, then that corner is an endpoint of the elementary segment.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_corner_side_subdivision,def:area_law_actual_side_mask,
        thm:area_law_elementary_side_geometry}
    Every elementary segment lies on its full side $[a_s,b_s]$, so any corner
    of cell $w$ on $[a_s,b_s]$ is $a_s$, $(a_s+b_s)/2$, or $b_s$. If side $s$
    is undivided, the midpoint $(a_s+b_s)/2$ cannot be a corner of cell $w$ by
    definition of the subdivision rule. If side $s$ is subdivided, each
    half-side excludes the opposite endpoint of $[a_s,b_s]$ and has the
    remaining two points as its endpoints.
\end{proof}

The paper takes the initial layer index $k_0$ sufficiently large without
fixing a numerical value \cite[Section~11]{OpenAI2026AreaLaw}; the threshold
$k\ge50\,000\,000$ is an explicit choice.
