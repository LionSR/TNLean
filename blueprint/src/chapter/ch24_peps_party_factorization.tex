\begin{definition}[The registers of a party]
    \label{def:peps_party_registers}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties}
    \lean{TNLean.Lattice.PartyCircuit.Word.owners_mem_parties}
    \lean{TNLean.Lattice.PartyCircuit.Layout.atParty}
    \lean{TNLean.Lattice.PartyCircuit.partyLayout}
    \lean{TNLean.Lattice.PartyCircuit.tensorPartyMaps}
    \lean{TNLean.Lattice.PartyCircuit.tensorPartyMaps_cons_tmul}
    \lean{TNLean.Lattice.PartyCircuit.partyLayout_eq_of_atParty_eq}
    \lean{TNLean.Lattice.PartyCircuit.tensorPartyMaps_heq}
    \lean{TNLean.Lattice.PartyCircuit.Layout.conj_memCongr_heq}
    \lean{TNLean.Lattice.PartyCircuit.Layout.norm_conj_memCongr}
    \lean{TNLean.Lattice.PartyCircuit.Layout.eq_conj_memCongr_of_heq}
    \leanok
    \uses{def:peps_party_layout}
    For an ordered register list $\ell$ and party $p$, let $\ell_p$ be the sublist of registers owned by $p$ in their original order, with memory $\mathcal H_{\ell,p}$ ($\mathbb C$ when $\ell_p=\emptyset$). Distinct parties $p_1,\ldots,p_s$ define the grouped memory $\bigotimes_{j=1}^s\mathcal H_{\ell,p_j}$, on which party-local maps act by ordered tensor product ($\operatorname{id}_{\mathbb C}$ for $s=0$). In \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS} (\texttt{04-compression.tex}, lines 233--251), the party set of a composition includes all register owners and all parties assigned to local operations (including scalar maps $\mathbb C\to\mathbb C$).
\end{definition}

\begin{definition}[Register partitions and grouping]

    \lean{TNLean.Lattice.PartyCircuit.Layout.partitionIso_append_false_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Layout.partitionIso_cons_cons_true_true_tmul}

    \lean{TNLean.Lattice.PartyCircuit.Layout.partitionIso_cons_cons_false_true_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Layout.partitionIso_cons_cons_false_false_tmul}
    \lean{TNLean.Lattice.PartyCircuit.Layout.withoutParty}
    \lean{TNLean.Lattice.PartyCircuit.Layout.atParty_withoutParty}
    \lean{TNLean.Lattice.PartyCircuit.Layout.mem_withoutParty}
    \lean{TNLean.Lattice.PartyCircuit.Layout.owners_withoutParty}
    \lean{TNLean.Lattice.PartyCircuit.partyLayout_withoutParty}
    \lean{TNLean.Lattice.PartyCircuit.groupByPartyIso}
    \lean{TNLean.Lattice.PartyCircuit.groupByPartyIso_cons}
    \leanok
    \uses{def:peps_party_registers}
    For any party subset $A$, associativity, tensor-factor exchanges, and unit identifications with $\mathbb C$ define an order-preserving unitary
    \begin{align}
        U_{A,\ell}:\mathcal H_\ell
         & \longrightarrow \mathcal H_{\ell|_A}\otimes
        \mathcal H_{\ell|_{A^c}}.
        \label{eq:peps_party_partition}
    \end{align}
    For distinct parties $p_1,\ldots,p_s$ containing all owners in $\ell$, iterating (\ref{eq:peps_party_partition}) defines the grouping unitary $U_\ell:\mathcal H_\ell\to\bigotimes_{j=1}^s\mathcal H_{\ell,p_j}$.
\end{definition}

\begin{proof}\leanok
    Set $U_{A,\emptyset}(z)=1\otimes z$ on $\mathbb C$, and for $U_{A,\ell}(y)=a\otimes b\in\mathcal H_{\ell|_A}\otimes\mathcal H_{\ell|_{A^c}}$ define $U_{A,(r,\ell)}$ inductively by
    \begin{align}
        U_{A,(r,\ell)}(x\otimes y)
        =\begin{cases}
             (x\otimes a)\otimes b & \text{if }\operatorname{owner}(r)\in A, \\
             a\otimes(x\otimes b)  & \text{otherwise}.
         \end{cases}
        \notag
    \end{align}
    Iterating over distinct $p_1,\ldots,p_s$ covering the owners of $\ell$ extracts each $\mathcal H_{\ell,p_j}$ once and leaves an empty remainder $\mathcal H_\emptyset=\mathbb C$.
\end{proof}

\begin{theorem}[Restriction to selected parties]
    \label{thm:peps_party_restriction}

    \lean{TNLean.Lattice.PartyCircuit.Word.eval_castLayouts}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict}
    \lean{TNLean.Lattice.PartyCircuit.Word.sources_restrict}
    \lean{TNLean.Lattice.PartyCircuit.Word.isAllowed_restrict}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_localMap_true_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_localMap_false_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_swap_true_true_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_swap_true_false_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_swap_false_true_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_swap_false_false_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_frame_true_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.restrict_frame_false_eval_heq}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties_castLayouts}
    \lean{TNLean.Lattice.PartyCircuit.Word.parties_restrict}
    \leanok
    \uses{def:peps_party_registers,def:peps_party_partition}
    For any source-free composition $F$ and party set $A$, retaining the registers and local operations with party in $A$ yields a source-free composition $F_A$ in the same chronological order with party set $\mathcal P(F)\cap A$, and $F_A$ is allowed whenever $F$ is allowed.
\end{theorem}

\begin{proof}\leanok
    Retaining a local operation precisely when its assigned party lies in $A$ and restricting exchanges and spectator registers recursively preserves the local contraction bounds and gives $\mathcal P(F_A)=\mathcal P(F)\cap A$ by structural induction on $F$.
\end{proof}

\begin{theorem}[Factorization across a partition of the parties]
    \label{thm:peps_party_binary_factorization}
    \lean{TNLean.Lattice.PartyCircuit.Word.partitionIso_eval}
    \leanok
    \uses{def:peps_party_partition,thm:peps_party_restriction}
    For any source-free composition $F:\mathcal H_\ell\to\mathcal H_{\ell'}$ and party set $A$,
    \begin{align}
        U_{A,\ell'}F & =(F_A\otimes F_{A^c})U_{A,\ell}.
        \label{eq:peps_party_binary_factorization}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_party_partition,thm:peps_party_restriction}
    By induction on $F$, if $L$ acts at party $p$ on registers $a$ owned by $p$ in front of $\ell$, and $J_{a,\ell}:\mathcal H_{(a,\ell)}\xrightarrow{\cong}\mathcal H_a\otimes\mathcal H_\ell$, then for $s\in\mathcal H_a$, $x\in\mathcal H_{\ell|_A}$, and $y\in\mathcal H_{\ell|_{A^c}}$,
    \begin{align}
        U_{A,(a,\ell)}J_{a,\ell}^{-1}
        \bigl(s\otimes U_{A,\ell}^{-1}(x\otimes y)\bigr)
        =\begin{cases}
             J_{a,\ell|_A}^{-1}(s\otimes x)\otimes y
              & \text{if $p\in A$},    \\
             x\otimes J_{a,\ell|_{A^c}}^{-1}(s\otimes y)
              & \text{if $p\notin A$},
         \end{cases}
        \notag
    \end{align}
    so $L$ acts on the $A$-factor when $p\in A$ and on the $A^c$-factor when $p\notin A$. For an exchange $S_{r,t,\ell}$ of registers $r,t$ with owners $p_r,p_t$ and $w=u\otimes(v\otimes U_{A,\ell}^{-1}(x\otimes y))$,
    \begin{align}
        U_{A,(t,r,\ell)}S_{r,t,\ell}w
         & =(S_A\otimes S_{A^c})U_{A,(r,t,\ell)}w   \\
         & =\begin{cases}
                (v\otimes(u\otimes x))\otimes y
                 & \text{if $p_r,p_t\in A$},            \\
                (u\otimes x)\otimes(v\otimes y)
                 & \text{if $p_r\in A$, $p_t\notin A$}, \\
                (v\otimes x)\otimes(u\otimes y)
                 & \text{if $p_r\notin A$, $p_t\in A$}, \\
                x\otimes(v\otimes(u\otimes y))
                 & \text{if $p_r,p_t\notin A$}.
            \end{cases}
        \label{eq:peps_party_exchange}
    \end{align}
    Since elementary tensors span and (\ref{eq:peps_party_binary_factorization}) is preserved under composition and tensoring with untouched registers, the result follows.
\end{proof}

\begin{theorem}[Tensor product of the local maps]
    \label{thm:peps_party_factorization}
    \lean{TNLean.Lattice.PartyCircuit.norm_tensorPartyMaps_le_one}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_tensorPartyMaps}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_tensorPartyMaps_parties}
    \lean{TNLean.Lattice.PartyCircuit.Word.eq_nil_and_eval_eq_id_of_parties_eq_empty}
    \leanok
    \uses{def:peps_party_registers,def:peps_party_partition}
    Let $F:\mathcal H_\ell\to\mathcal H_{\ell'}$ be an allowed source-free composition (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 233--251), and let $p_1,\ldots,p_s$ be distinct parties with $\mathcal P(F)\subseteq\{p_1,\ldots,p_s\}$. There exist contractions $B_{p_j}:\mathcal H_{\ell,p_j}\to\mathcal H_{\ell',p_j}$ such that
    \begin{align}
        U_{\ell'}F
         & =\left(\bigotimes_{j=1}^s B_{p_j}\right)U_\ell,
        \label{eq:peps_party_factorization}
    \end{align}
    with $\bigl\|\bigotimes_{j=1}^s B_{p_j}\bigr\|\le1$ (and $F=\operatorname{id}_{\mathbb C}$ when $\mathcal P(F)=\emptyset$, $s=0$).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_binary_factorization,thm:peps_party_restriction}
    Iterating (\ref{eq:peps_party_binary_factorization}) with $A=\{p_j\}$ ($j=1,\ldots,s$) leaves a composition with empty party set equal to $\operatorname{id}_{\mathbb C}$, and Theorem~\ref{thm:peps_party_restriction} ensures each factor $B_{p_j}=F_{\{p_j\}}$ is a contraction.
\end{proof}

\begin{theorem}[Prepared pair sources and local contractions]
    \label{thm:peps_prepared_party_maps}
    \lean{TNLean.Lattice.PartyCircuit.Word.exists_prepared_tensorPartyMaps}
    \leanok
    \uses{def:peps_source_inventory,def:peps_party_registers,def:peps_party_partition}
    Let $P$ be an ordered finite party set and $F:\mathcal H_{\rm in}\to\mathcal H_{\rm out}$ an allowed composition with $\mathcal P(F)\subseteq P$. There exist a normalized source list $S$ with one source per unordered pair in $F$ and contractions $B_p$ ($p\in P$) such that
    \begin{align}
        U_{\rm out}F
         & =\left(\bigotimes_{p\in P}B_p\right)
        U_{S,\rm in}P_{S,\mathcal H_{\rm in}},
        \label{eq:peps_prepared_party_maps}
    \end{align}
    where $U_{S,\rm in}$ and $U_{\rm out}$ group the prepared-source plus input registers and the output registers in the order of $P$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_grouped_sources,thm:peps_party_factorization}
    Theorem~\ref{thm:peps_grouped_sources} factors $F=V\circ P_{S,\mathcal H_{\rm in}}$ with $V$ allowed, source-free, and $\mathcal P(V)\subseteq P$; applying (\ref{eq:peps_party_factorization}) to $V$ yields (\ref{eq:peps_prepared_party_maps}).
\end{proof}

\begin{theorem}[Factorization by singleton restrictions]
    \label{thm:peps_actual_singleton_factorization}
    \lean{TNLean.Lattice.PartyCircuit.Word.groupByPartyIso_eval_eq_tensorRestrictions}
    \leanok
    \uses{def:peps_party_registers,def:peps_party_partition,
        thm:peps_party_restriction}
    For any source-free composition $W:\mathcal H_\ell\to\mathcal H_m$ and distinct parties $p_1,\ldots,p_s$ with $\mathcal P(W)\subseteq\{p_1,\ldots,p_s\}$ (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}, \texttt{04-compression.tex}, lines 233--251 and 351--381), the grouping unitaries $U_\ell,U_m$ satisfy
    \begin{align}
        U_m W
         & =\left(\bigotimes_{j=1}^s W|_{\{p_j\}}\right)U_\ell.
        \label{eq:peps_actual_singleton_factorization}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_party_binary_factorization,thm:peps_party_factorization,
        thm:peps_party_restriction,thm:peps_singleton_after_excluded_party}
    We induct on $(p_1,\ldots,p_s)$. For $s=0$, $\mathcal P(W)=\emptyset$ gives $W=\operatorname{id}_{\C}$ by Theorem~\ref{thm:peps_party_factorization}. For $(p,p_2,\ldots,p_s)$, applying (\ref{eq:peps_party_binary_factorization}) with $A=\{p\}$ splits $W$ into $W|_{\{p\}}$ and $W|_{P\setminus\{p\}}$, and for each $q\in\{p_2,\ldots,p_s\}$ one has $(W|_{P\setminus\{p\}})|_{\{q\}}=W|_{\{q\}}$ by (\ref{eq:peps_singleton_after_excluded_party}), completing the induction.
\end{proof}
