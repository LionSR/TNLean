\section{Incident factors of the sampled physical density}
\label{sec:peps_incident_sampled_physical_density}

The source-replaced physical operator is expressed in terms of prepared
operators determined by incident gate labels and sampled endpoint operators
indexed by the original incident source slots. The ket and bra labels are
independent, and each source uses one shared sample index at its two endpoints.

\begin{definition}[Comparison of word evaluations]
    \label{def:peps_word_eval_comparison}
    \lean{TNLean.Lattice.PartyCircuit.Word.IsEvalComparison}
    \leanok
    \uses{def:peps_party_layout}
    For chronological words $u,v$ in the proof of Theorem~5.2
    of~\cite{OpenAI2026PolynomialPEPS}, their evaluations agree when
    $\operatorname{eval}(u)$ and $\operatorname{eval}(v)$ are heterogeneously
    equal as operators between their recorded input and output register spaces
    (Definition~\ref{def:peps_party_layout}).
\end{definition}

\begin{lemma}[Evaluation comparison on fixed layouts]
    \label{lem:peps_word_eval_comparison_fixed_layouts}
    \lean{TNLean.Lattice.PartyCircuit.Word.IsEvalComparison.eval_eq}

    heterogeneous equality (Definition~\ref{def:peps_word_eval_comparison})

\end{proof}
