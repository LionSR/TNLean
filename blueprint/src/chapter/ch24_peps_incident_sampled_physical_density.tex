\section{Incident factors of the sampled physical density}
\label{sec:peps_incident_sampled_physical_density}

The source-replaced physical operator is expressed in terms of prepared
operators determined by incident gate labels and sampled endpoint operators
indexed by the original incident source slots. The ket and bra labels are
independent, and each source uses one shared sample index at its two endpoints.

\begin{lemma}[Remaining operations at an owner with a singleton fibre]
    \label{thm:peps_singleton_fibre_local_part}
    \lean{QuantumCircuit.Party.Word.localPartOp_coarsen_of_fiber}
    \leanok
    \uses{thm:peps_source_preparation,def:peps_owner_composition,def:peps_single_party_selector,def:peps_party_registers}
    Let $f:P\to Q$ satisfy $f^{-1}(\{q\})=\{p\}$, and let $U:a\to b$ be a
    chronological word with source-register list $s_U$ and local part
    $L(U):s_Ua\to b$ (Lemma~\ref{thm:peps_source_preparation};
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}). Under the canonical
    identifications of $((s_Ua)_p)^f$ and $(b_p)^f$,
    \begin{align}
        (L(U^f))|_q & =\bigl((L(U))|_p\bigr)^f.
        \label{eq:peps_singleton_fibre_local_part}
    \end{align}
    % Source: 04-compression.tex, lines 383--434 and 565--588.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_source_free_local_part_restriction,thm:peps_append_spectators_restriction,thm:peps_owner_selection_commutation,thm:peps_owner_spectator_extension,thm:peps_owner_evaluation_identification}
    We induct on $U$. Source-free elementary steps satisfy~(\ref{eq:peps_singleton_fibre_local_part})
    by Lemma~\ref{thm:peps_source_free_local_part_restriction},
    Lemma~\ref{thm:peps_owner_selection_commutation}, and
    Lemma~\ref{thm:peps_owner_evaluation_identification}. Since
    $f^{-1}(\{q\})=\{p\}$, the source registers of $U^f$ at $q$ are those of
    $U$ at $p$, relabelled by $f$; hence compositions and framed words follow
    from Lemma~\ref{thm:peps_append_spectators_restriction} and
    Lemma~\ref{thm:peps_owner_spectator_extension}. At a pair source with
    distinct owners $p_1\ne p_2$, either $f(p_1)\ne f(p_2)$ so both local parts
    are $\Id$, or $f(p_1)=f(p_2)\ne q$ (since $f^{-1}(\{q\})=\{p\}$ has size
    $1$) so its merged preparation is erased at $q$.
\end{proof}

For the following statements, fix $p\in P$, let $g_{\rm all}:P\to Q_{\rm all}=P\sqcup\{*\}$
retain every party, and write $p_{\rm all}=g_{\rm all}(p)$.

\begin{lemma}[The prepared operator determined by incident labels]
    \label{thm:peps_prepared_incident_operator}
    \lean{QuantumCircuit.Party.SourceCircuit.partyOp_partialSourceWord_eq_incidentSourceOp}
    \leanok
    \uses{def:peps_actual_private_preparation,thm:peps_incident_gate_labels,def:peps_canonical_source_residual,thm:peps_partial_fixed_layout,def:peps_incident_source_word}
    Let $\mathcal W:a\to b$ be a chronological circuit with private product
    input $\psi$ (Definition~\ref{def:peps_actual_private_preparation}) and
    all-party partial choice $\xi$ with incident gate labels $\eta=\xi_p$
    (Lemma~\ref{thm:peps_incident_gate_labels};
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}). Let $T_\xi$ be the
    prepared all-party word (Definition~\ref{def:peps_canonical_source_residual})
    and $D_{p,\eta}$ the prepared operation from incident labels
    (Definition~\ref{def:peps_incident_source_word}) between
    $(S_{\rm all}\varnothing)_{p_{\rm all}}$ and $(b^{g_{\rm all}})_{p_{\rm all}}$.
    Then
    \begin{align}
        T_\xi|_{p_{\rm all}} & =D_{p,\xi_p}.
        \label{eq:peps_prepared_incident_operator}
    \end{align}
    % Source: 04-compression.tex, lines 137--139, 246--267, 351--434 and 565--588.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_incident_source_word,thm:peps_incident_source_word_well_defined}
    The choice $\xi$ itself has incident labels $\xi_p$, so by
    Definition~\ref{def:peps_incident_source_word} and
    Lemma~\ref{thm:peps_incident_source_word_well_defined}, $D_{p,\xi_p}$ is
    the operation of $T_\xi$ at $p_{\rm all}$, which
    is~(\ref{eq:peps_prepared_incident_operator}).
\end{proof}

\begin{definition}[Ordered endpoint operators at one party]
    \label{def:peps_ordered_endpoint_register_operators}
    \lean{QuantumCircuit.Party.pairFactorMatrix}
    \lean{QuantumCircuit.Party.SourceInventory.slotSiteEquiv}
    \leanok
    \uses{def:peps_source_inventory,def:peps_source_slots,def:peps_source_register_operator_word}
    Let $R=((p_i,q_i))_{i=0}^{n-1}$ be an ordered pair-source list with
    endpoint spaces $U_i,V_i$ of finite orthonormal bases, let $A_i$ and $B_i$
    be operators on $U_i$ and $V_i$, and let the source registers stand in
    front of a system with space $K_p$ at each party $p$. For a party $p$ and
    $i\in\{0,\ldots,n-1\}$, put $H_{i,p}=U_i$ and $F_{i,p}=A_i$ if $p=p_i$,
    $H_{i,p}=V_i$ and $F_{i,p}=B_i$ if $p=q_i$, and $H_{i,p}=\C$ and
    $F_{i,p}=\Id_{\C}$ otherwise. The space of $p$ in the source registers is
    $H_{0,p}\otimes\cdots\otimes H_{n-1,p}\otimes K_p$, in this order, and the
    ordered endpoint operator at $p$ is
    \begin{align}
        \mathcal E_p(A,B)
         & =F_{0,p}\otimes\cdots\otimes F_{n-1,p}\otimes I_{K_p},
        \label{eq:peps_ordered_endpoint_data}
    \end{align}
    recording the endpoint operations at $p$ in
    \cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}.
    % Source: 04-compression.tex, lines 279--309 and 565--588.
\end{definition}

\begin{lemma}[Restriction retains the ordered endpoint operators]
    \label{thm:peps_restricted_ordered_endpoint_operators}
    \lean{QuantumCircuit.Party.SourceInventory.partyOp_endpointWord}
    \leanok
    \uses{def:peps_ordered_endpoint_register_operators,def:peps_source_register_operator_word,thm:peps_party_restriction,thm:peps_source_register_operator_local_congruence}
    For a party $p$, the endpoint word $Z_R=Z_R(A,B)$
    (Definition~\ref{def:peps_source_register_operator_word}) satisfies, in
    the coordinates of Definition~\ref{def:peps_ordered_endpoint_register_operators},
    \begin{align}
        Z_R|_p & =\mathcal E_p(A,B).
        \label{eq:peps_restricted_ordered_endpoint_operators}
    \end{align}
    In particular, $Z_R|_p$ depends only on the operators $A_i$ with $p_i=p$
    and $B_i$ with $q_i=p$.
    % Source: 04-compression.tex, lines 279--309, 409--434 and 565--588.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_ordered_endpoint_register_operators,thm:peps_party_restriction}
    Induct on $R$. For $R=\varnothing$, $Z_R=\Id$. For a head slot
    $(p_0,q_0)$, the word applies $A_0$ at $p_0$ and $B_0$ at $q_0$ after the
    tail word on the remaining slots; at $p$ this is $F_{0,p}$ on the first
    factor $H_{0,p}$ beside the restriction of the tail word, and the induction
    hypothesis gives~(\ref{eq:peps_restricted_ordered_endpoint_operators}).
\end{proof}

\begin{definition}[The sampled endpoint word from incident labels]
    \label{def:peps_incident_sampled_endpoint_word}
    \lean{QuantumCircuit.Party.SourceCircuit.incidentSampledEndpointOp}
    \leanok
    \uses{thm:peps_incident_gate_labels,def:peps_partial_fixed_positions,def:peps_original_sampled_endpoint_factors,def:peps_source_register_operator_word,def:peps_all_party_source_registers}
    For a chronological circuit $\mathcal W:a\to b$ with all-party
    source-register list $S_{\rm all}$, fix incident label assignments
    $\eta,\theta$ at $p$, enumerate the source occurrences incident to $p$ in
    order as $e_0,\ldots,e_{n_p-1}$, and fix a local sample assignment
    $j_p:\{0,\ldots,n_p-1\}\to\{0,\ldots,k-1\}$ and Gaussian arrays
    $\omega_{e,\alpha\beta}$ (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}).
    With $r_e=\min(u_e,v_e)$, endpoint matrix families $L,H$
    (Definition~\ref{def:peps_original_sampled_endpoint_factors}), and gate
    labels $\alpha_i,\beta_i$ reconstructed from $\eta,\theta$ at $e_i$
    (Lemma~\ref{thm:peps_incident_gate_labels}), set
    \begin{align}
        A_i & =L_{e_i,\alpha_i\beta_i,j_p(i)}
        (\omega_{e_i,\alpha_i\beta_i}),       \\
        B_i & =H_{e_i,\alpha_i\beta_i,j_p(i)}
        (\omega_{e_i,\alpha_i\beta_i}).
        \label{eq:peps_incident_sampled_endpoints}
    \end{align}
    Let $Z^{p}_{\eta,\theta,j_p,\omega}$ be the endpoint word
    (Definition~\ref{def:peps_source_register_operator_word}) on the all-party
    source registers $S_{\rm all}$
    (Definition~\ref{def:peps_all_party_source_registers}) whose endpoint
    operators are $A_i,B_i$ at the slot of $e_i$ and identities at every slot
    not incident to $p$, and define the operator on
    $(S_{\rm all}\varnothing)_{p_{\rm all}}$
    \begin{align}
        U_{p,\eta,\theta,j_p,\omega}
         & =Z^{p}_{\eta,\theta,j_p,\omega}|_{p_{\rm all}}.
        \label{eq:peps_incident_endpoint_word}
    \end{align}
    % Source: 04-compression.tex, lines 253--267, 279--355, 409--434 and 565--588.
\end{definition}

\begin{lemma}[The sampled endpoint operator determined by incident labels]
    \label{thm:peps_incident_sampled_endpoint_comparison}
    \lean{QuantumCircuit.Party.SourceCircuit.partyOp_sampledEndpointWord_eq_incidentSampledEndpointOp}
    \leanok
    \uses{def:peps_incident_sampled_endpoint_word,def:peps_actual_sampled_endpoint_word,thm:peps_incident_gate_labels}
    For a chronological circuit $\mathcal W:a\to b$, all-party choices $\xi,\zeta$,
    the endpoint matrix families, sample number $k$, and Gaussian arrays $\omega$
    of Definition~\ref{def:peps_incident_sampled_endpoint_word}, and a global
    sample assignment $j:\mathcal E\to\{0,\ldots,k-1\}$ with restriction
    $j_p(i)=j(e_i)$, the all-party sampled endpoint word
    $U^{\rm all}_{\xi,\zeta,j,\omega}$
    (Definition~\ref{def:peps_actual_sampled_endpoint_word}) satisfies
    \begin{align}
        U^{\rm all}_{\xi,\zeta,j,\omega}|_{p_{\rm all}}
         & =U_{p,\xi_p,\zeta_p,j_p,\omega}
        \label{eq:peps_incident_sampled_endpoint_comparison}
    \end{align}
    on $(S_{\rm all}\varnothing)_{p_{\rm all}}$
    (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}).
    % Source: 04-compression.tex, lines 253--267, 279--355, 409--434 and 565--588.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_restricted_ordered_endpoint_operators,thm:peps_incident_gate_labels}
    By Lemma~\ref{thm:peps_restricted_ordered_endpoint_operators},
    $U^{\rm all}_{\xi,\zeta,j,\omega}|_{p_{\rm all}}$ and
    $U_{p,\xi_p,\zeta_p,j_p,\omega}$ depend only on the endpoint operators at
    the slots incident to $p$. By Lemma~\ref{thm:peps_incident_gate_labels},
    $\xi_p$ and $\zeta_p$ preserve the gate labels at every occurrence $e_i$
    incident to $p$, so there both use the same source position, label pair,
    sample index $j_p(i)=j(e_i)$, and Gaussian array, and their endpoint
    operators coincide with~(\ref{eq:peps_incident_sampled_endpoints}). This
    yields~(\ref{eq:peps_incident_sampled_endpoint_comparison}).
\end{proof}

\begin{definition}[The physical factor constructed from incident labels]
    \label{def:peps_incident_sampled_physical_factor}
    \lean{QuantumCircuit.Party.SourceCircuit.incidentSampledPhysicalFactor}
    \leanok
    \uses{def:peps_incident_source_word,def:peps_incident_sampled_endpoint_word,def:peps_selected_physical_private_basis,def:peps_private_product_input}
    For $d:P\to\N$, party enumeration $p_0,\ldots,p_{m-1}$ with physical
    register list $h_d$ ($\mathcal H_{p_i}=\C^{d_{p_i}}$), private output list $c$,
    and chronological circuit $\mathcal W:a\to h_dc$ with product input $\psi$
    (Definition~\ref{def:peps_private_product_input}), choose finite orthonormal
    bases $B_q=(B_q(i))_{i\in I_q}$ of $(S_{\rm all}\varnothing)_q$ and
    $G_q=(G_q(e))_{e\in E_q}$ of $(c^{g_{\rm all}})_q$, with induced
    physical/private basis $C_q$ on $((h_dc)^{g_{\rm all}})_q$
    (Definition~\ref{def:peps_selected_physical_private_basis}). For $p\in P$,
    incident labels $\eta,\theta$, local samples $j_p$, and $x\in\{0,\ldots,d_p-1\}$,
    $e\in E_{p_{\rm all}}$, $i,l\in I_{p_{\rm all}}$, define
    \begin{align}
        A_{p,\eta}((x,e),i)
         & =\bigl\langle C_{p_{\rm all}}(x,e),
        D_{p,\eta}B_{p_{\rm all}}(i)\bigr\rangle,                   \\
        N_{p,\eta,\theta,j_p,\omega}(i,l)
         & =\bigl\langle B_{p_{\rm all}}(i),
        U_{p,\eta,\theta,j_p,\omega}B_{p_{\rm all}}(l)\bigr\rangle, \\
        M^{\rm inc}_{p,\eta,\theta,j_p,\omega}
         & =\operatorname{tr}_{E_{p_{\rm all}}}
        \bigl(A_{p,\eta}N_{p,\eta,\theta,j_p,\omega}A_{p,\theta}^{\dagger}\bigr),
        \label{eq:peps_incident_sampled_physical_factor}
    \end{align}
    where $D_{p,\eta}$ is the operator of Definition~\ref{def:peps_incident_source_word}
    and $U_{p,\eta,\theta,j_p,\omega}$ is the operator of
    Definition~\ref{def:peps_incident_sampled_endpoint_word}
    (\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS}).
    % Source: 04-compression.tex, lines 253--267, 279--355, 409--434 and 574--579.
\end{definition}

\begin{lemma}[The physical factor equals its incident construction]
    \label{thm:peps_incident_sampled_physical_factor_comparison}

    In the setting of Definition~\ref{def:peps_incident_sampled_physical_factor},
    for all-party choices $\xi,\zeta$ and a global sample assignment $j$ with
    restriction $j_p(i)=j(e_i)$, the local physical factor
    $M^{\xi,\zeta}_{j,p_{\rm all}}$
    (Lemma~\ref{thm:peps_actual_sampled_physical_factors}) satisfies
    \begin{align}
        M^{\xi,\zeta}_{j,p_{\rm all}}
         & =M^{\rm inc}_{p,\xi_p,\zeta_p,j_p,\omega}.
        \label{eq:peps_incident_sampled_physical_factor_comparison}
    \end{align}
    % Source: 04-compression.tex, lines 565--588, especially 574--579.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_prepared_incident_operator,thm:peps_incident_sampled_endpoint_comparison,def:peps_incident_sampled_physical_factor}
    By Lemma~\ref{thm:peps_prepared_incident_operator} and
    Lemma~\ref{thm:peps_incident_sampled_endpoint_comparison}, the matrices of
    $T_\xi|_{p_{\rm all}}$, $T_\zeta|_{p_{\rm all}}$, and
    $U^{\rm all}_{\xi,\zeta,j,\omega}|_{p_{\rm all}}$ in the bases $B_{p_{\rm all}},C_{p_{\rm all}}$

    substituting into~(\ref{eq:peps_incident_sampled_physical_factor}) gives~(\ref{eq:peps_incident_sampled_physical_factor_comparison}).
\end{proof}


\begin{lemma}[Restricting samples to the original incident slots]
    \label{thm:peps_incident_source_sample_compatibility}
    \lean{QuantumCircuit.Party.SourceCircuit.restrict_sourceSampleAssignment_eq}
    \leanok
    \uses{def:peps_original_source_sample_indices,def:peps_local_source_sample_indices,def:peps_partial_fixed_positions}
    Let $\tau:\mathcal E\to\mathcal S$ be the canonical bijection between original
    source positions $\mathcal E$ and distributed sample positions $\mathcal S$
    (Definition~\ref{def:peps_original_source_sample_indices}), and let
    $\tau_p:\{0,\ldots,n_p-1\}\to\mathcal S_p$ be the local bijection on sample
    positions incident to $p$ (Definition~\ref{def:peps_local_source_sample_indices}
    and Definition~\ref{def:peps_partial_fixed_positions}). For any
    $j:\mathcal E\to\{0,\ldots,k-1\}$, $j^{\rm samp}(s)=j(\tau^{-1}(s))$ and
    $j_p(i)=j(e_i)$ satisfy
    \begin{align}
        (j^{\rm samp}|_{\mathcal S_p})\circ\tau_p & =j_p.
        \label{eq:peps_incident_source_sample_compatibility}
    \end{align}
    % Source: 04-compression.tex, lines 342--417 and 568--579.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:peps_original_source_sample_indices,def:peps_local_source_sample_indices}
    By Definition~\ref{def:peps_original_source_sample_indices} and
    Definition~\ref{def:peps_local_source_sample_indices}, $\tau_p(i)=\tau(e_i)$,
    so $j^{\rm samp}(\tau_p(i))=j(\tau^{-1}(\tau(e_i)))=j(e_i)=j_p(i)$,
    giving~(\ref{eq:peps_incident_source_sample_compatibility}).
\end{proof}

\begin{lemma}[The original physical sum in incident factors]
    \label{thm:peps_incident_pure_sampled_physical_sum}

    For finite $P$, in the setting of
    Definition~\ref{def:peps_incident_sampled_physical_factor} and
    Lemma~\ref{thm:peps_actual_pure_sampled_physical_sum} with $N$ source

        \widehat\rho_{\mathrm{phys},\psi}(\widehat X)

        \sum_{j\in\{0,\ldots,k-1\}^N}
        \bigotimes_{p\in P}
        M^{\rm inc}_{p,\xi_p,\zeta_p,j_p,\omega}.
        \label{eq:peps_incident_pure_sampled_physical_sum}
    \end{align}
    % Source: 04-compression.tex, lines 253--267, 351--381 and 565--588.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:peps_actual_pure_sampled_physical_sum,thm:peps_incident_sampled_physical_factor_comparison}

    from~(\ref{eq:peps_incident_sampled_physical_factor_comparison}) into each
    summand of~(\ref{eq:peps_actual_pure_sampled_physical_sum}) of
    Lemma~\ref{thm:peps_actual_pure_sampled_physical_sum}.
\end{proof}
