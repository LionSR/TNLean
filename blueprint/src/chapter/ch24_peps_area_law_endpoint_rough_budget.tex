\section{Endpoint norm loss and selected density}
\label{sec:arealaw-actual-endpoint-budget}

Write $H=nm$ for the physical scan horizon, $a>0$ for its per-band weight
and $W=aK$. Fix a unit truncated physical vector, a target spectral set of
cardinality $d$ and positive mass $z$, and a Schmidt--Bell label sequence
whose replica prevectors have norm at most one and satisfy the polynomial
mass lower bound; all root and filtered norms below use this sequence.

\begin{definition}[The endpoint replica remainder]
    \label{def:arealaw-rough-endpoint-remainder}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughEndpointRemainder}
    \leanok
    Let $R_k^{\rm Bell}(d,c,K)$ be the rough Bell-overlap remainder.  Put
    \begin{align}
        R_k^{\rm end}(d,c,K,C)
         & =R_k^{\rm Bell}(d,c,K)+C\frac{\log(k+2)}k.
        \label{eq:arealaw-rough-endpoint-remainder}
    \end{align}
\end{definition}

\begin{lemma}[Vanishing endpoint remainder]
    \label{thm:arealaw-rough-endpoint-remainder-limit}
    \lean{TNLean.PEPS.AreaLaw.Scan.tendsto_roughEndpointRemainder}
    \leanok
    For fixed $d,c,K,C$, the remainder in
    (\ref{eq:arealaw-rough-endpoint-remainder}) tends to zero as $k\to\infty$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:arealaw-rough-endpoint-remainder}
    Both $R_k^{\rm Bell}(d,c,K)$ and $\log(k+2)/k$ tend to zero as $k\to\infty$.
\end{proof}

\begin{lemma}[Initial-to-final physical endpoint loss]
    \label{thm:arealaw-physical-endpoint-loss}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.eventually_actualScan_suppliedPrevector_endpoint_budget}
    \leanok
    Suppose the ambient target is nonempty, the scanner graph is the
    induced domain graph, its depth is the target's ambient depth,
    $n,m,M>0$, $r_0\leq D$, $D\geq1$,
    $4D\leq m$ and $8Km\leq L$.  Assume every ambient target layer up
    to $L$ has size at most $n$ and every whole-row positive-depth prefix
    of the truncated state has entropy at most $B_{\rm sh}$.
    There are nonnegative constants $c,C_i$, before the replica count,
    such that eventually the final squared filtered norm is positive and
    \begin{align}
        \frac{\log N_k^{\rm initial}-\log N_k^{\rm final}}k
         & \leq B_{\rm rough}+R_k^{\rm end}(d,c,K,C_i),\nonumber \\
        B_{\rm rough}
         & =2\log d-\log z+
        W\left(\frac{B_{\rm sh}}z+1+3nD\log q\right).
        \label{eq:arealaw-physical-endpoint-loss}
    \end{align}
    Here $N_k^{\rm initial}$ and $N_k^{\rm final}$ are the squared filtered
    norms at the fill endpoints $(0,0)$ and $(H,0)$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_prevector_norm,thm:al_scan_actual_initial_log_norm,
        thm:al_scan_actual_rough_upper,
        thm:arealaw-rough-endpoint-remainder-limit}
    Because the coordinate isometry preserves the norm of the replica
    prevector, the initial log-norm floor bounds $\log N_k^{\rm initial}$
    by $C_i\log(k+2)$.
    Since $H>0$, the final fill root at $(H,0)$ coincides with the last
    charge root at $(H-1,1)$, which lies within the horizon of the rough
    upper estimate; combining that upper estimate with the initial
    log-norm bound gives the endpoint loss inequality.
\end{proof}

\begin{lemma}[The original-state entropy input]
    \label{thm:arealaw-source-endpoint-loss}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_source_actualScan_suppliedPrevector_endpoint_budget,TNLean.PEPS.AreaLaw.Scan.CollarScan.source_actualScan_suppliedPrevector_endpoint_budget_of_threshold_le}
    \leanok
    Fix $q\geq1$, a positive normalized gap $g$ and a volume coefficient
    $C_{\rm vol}$. For all sufficiently large $n$ (above a threshold
    depending only on $q$, $g$ and $C_{\rm vol}$), suppose the original and
    truncated vectors are unit, the compact collar has at most
    $C_{\rm vol}n^2$ sites, and the trace distance between the two states is
    at most $\sqrt{2\min\{n^{-1000},g/4\}/g}$. If every original-state
    positive-depth prefix up to $L$ has entropy at most
    $C_e(nL^e+n)$, then the endpoint bound
    (\ref{eq:arealaw-physical-endpoint-loss}) holds with
    \begin{align}
        B_{\rm sh} & =C_e(nL^e+n)+n^{-247}.
        \label{eq:arealaw-source-endpoint-shell}
    \end{align}
    All geometric and same-sequence premises of the preceding theorem
    are retained.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:al_positive_prefix_entropy_transfer,
        thm:arealaw-physical-endpoint-loss}
    Entropy continuity bounds the entropy of the truncated state on every
    positive-depth prefix up to $L$ by $B_{\rm sh}$, and the preceding
    theorem applied to the same truncated state, target spectral set and
    label sequence then yields the endpoint loss bound.
\end{proof}

\begin{lemma}[The accumulated selection numerator]
    \label{thm:arealaw-rough-selection-numerator}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughSelectionBudget_le_of_endpoint_budget}
    \leanok
    Let $k>0$, $A_k=\log N_k^{\rm initial}-\log N_k^{\rm final}$,
    and $\beta_k\leq C_\beta\log(k+2)$.  Assume the endpoint inequality
    (\ref{eq:arealaw-physical-endpoint-loss}).  For any real rate $T$,
    \begin{align}
        \frac{A_k+2H(kT+\beta_k)}k
         & \leq B_{\rm rough}+2HT+
        R_k^{\rm end}(d,c,K,C_i+2HC_\beta).
        \label{eq:arealaw-rough-selection-numerator}
    \end{align}
    In the scanner entropy bound,
    $T=C_{\rm ent}aK\,a^{1/4}\ell^{e_{\rm ent}}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:arealaw-rough-endpoint-remainder}
    Divide the endpoint and logarithmic error bounds by $k$ and expand.
\end{proof}

\begin{lemma}[Density at the same selected point]
    \label{thm:arealaw-selected-density-normalization}
    \lean{TNLean.PEPS.AreaLaw.Scan.selected_density_le_of_normalized_budget}
    \leanok
    Let $k,n,m,D,K$ be positive integers and $a,C_1,b$ positive real
    numbers, set $H=nm$ and
    $\rho=ka/(2(C_1+1)nD)$. If a selected point satisfies
    $Q\leq B/(\rho H b)$ and $B/k\leq R$, then at that point
    \begin{align}
        \frac{Q}{KnD} & \leq\frac{2(C_1+1)}{aKHb}R.
        \label{eq:arealaw-selected-density-normalization}
    \end{align}
    Here $b=p_1-p_0$ for the selected parameter interval.
\end{lemma}
\begin{proof}
    \leanok
    Dividing $Q\leq B/(\rho H b)$ by $KnD>0$ and substituting
    $\rho=ka/(2(C_1+1)nD)$ gives
    $Q/(KnD)\leq\frac{2(C_1+1)}{aKHb}(B/k)$, which combines with
    $B/k\leq R$ (without any sign assumption on $B$) to yield the stated
    density bound.
\end{proof}

Together, these estimates bound the normalized charge density $Q/(KnD)$
in terms of the spectral mass $z$, target dimension $d$, prefix entropy
bound $B_{\rm sh}$ and replica remainder $R_k^{\rm end}$.
