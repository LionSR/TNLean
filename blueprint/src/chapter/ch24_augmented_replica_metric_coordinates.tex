\section{Replica metrics in augmented site coordinates}

Write $I=V\sqcup\{C,R\}$ for the site set obtained by adjoining the two
auxiliary registers $C$ and $R$ to the physical sites $V$, and write
$J=V\sqcup\{\mathrm{false},\mathrm{true}\}$.
Let $a:I\to J$ fix each site of $V$ and send $C$ to
$\mathrm{false}$ and $R$ to $\mathrm{true}$.
Each physical site in $V$ has local dimension $q$ and each auxiliary
register has dimension $d$. Denote the resulting single-copy configuration
spaces by $\mathcal X_I$ and $\mathcal X_J$. Relabelling sites along $a$
induces a bijection $e_0:\mathcal X_I\to\mathcal X_J$ and, on $k$ replicas,
the coordinatewise bijection $e_k:\mathcal X_I^k\to\mathcal X_J^k$.

\begin{theorem}[Subsystem permutations in augmented coordinates]
    \label{thm:al_augmented_subsystem_action}
    \lean{TNLean.PEPS.AreaLaw.Scan.augmentedReplicaConfigurationEquiv_subsystemPerm}
    \leanok
    \uses{def:al_scan_auxiliary_coordinates}
    Let $V$ be a set, let $q,d,k\ge0$ be integers, and let $Q\subseteq I$
    be finite. For $\sigma\in S_k$, write $\rho_Q(\sigma)$ for the
    operator that permutes the $k$ replicas at the sites in $Q$ and fixes
    all other sites. Then, for every $x\in\mathcal X_I^k$,
    \begin{align*}
        e_k\bigl(\rho_Q(\sigma)x\bigr)=\rho_{a(Q)}(\sigma)e_k(x).
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    At site $a(i)$ and replica $j$, both sides take the value of $x$
    at site $i$ and replica $\sigma^{-1}(j)$ when $i\in Q$, and at replica $j$
    otherwise. Since $a$ is bijective, $i\in Q$ if and only if
    $a(i)\in a(Q)$, so the two configurations agree at every coordinate.
\end{proof}

\begin{theorem}[Replica metrics in augmented coordinates]
    \label{thm:al_augmented_replica_metric_coordinates}
    \lean{TNLean.PEPS.AreaLaw.Scan.replicaMetric_submatrix_augmentedReplicaConfigurationEquiv}
    \leanok
    Suppose that $V$ is finite. Put
    $N=|\mathcal X_I|=|\mathcal X_J|$, and form the replica metrics on
    $\mathcal X_I^k$ and $\mathcal X_J^k$ with the weight
    $w_{k,N,t}(\lambda)$ at the full single-copy dimension $N$.
    For every real $t$, every integer $k\ge0$, and every $Q\subseteq I$,
    \begin{align*}
        W^{J}_{a(Q),k}(x,y)=W^{I}_{Q,k}\bigl(e_k^{-1}x,e_k^{-1}y\bigr).
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:al_augmented_subsystem_action}
    The bijection $e_0$ gives $|\mathcal X_I|=|\mathcal X_J|=N$, so the
    weights $w_{k,N,t}(\lambda)$ coincide on both spaces. By the subsystem
    permutation identity, conjugation by $e_k$
    maps each permutation matrix $\rho_Q(\sigma)$ to
    $\rho_{a(Q)}(\sigma)$, and therefore maps each Young-symmetrizer
    projector on $Q$ to the corresponding projector on $a(Q)$. Summing
    these projectors with the common weights $w_{k,N,t}(\lambda)$ gives
    the stated equality, which relates the two coordinate conventions in
    \cite[Sections~5--8]{OpenAI2026AreaLaw}.
\end{proof}
