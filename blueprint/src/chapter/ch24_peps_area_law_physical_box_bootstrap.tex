\section{The physical rough safe-box bootstrap}
\label{sec:arealaw-physical-box-bootstrap}

The following construction uses the fixed bootstrap scales of the
September 24 manuscript.  Its input is a uniform safe-box estimate at the
current exponent, and its conclusion is an improved exponent in the
original state.  The finite-dimensional replica limit is taken before
changing the physical size.  No sharp marginal-moment estimate is used.

\begin{theorem}[Generic physical charge-energy coefficient]
    \label{thm:arealaw-generic-charge-energy}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScannerExponents.exists_actualCharge_energy_coefficients}
    \leanok
    Fix $q\geq1$, a scanner exponent tuple, positive radius and charge constants,
    a target-volume coefficient $\ge0$, an anchor-multiplicity bound, a tail exponent $P$,
    and the two round-energy coefficients. There exist $C\geq1$, $C_l\geq0$, and $N$
    (prior to any finite physical system, auxiliary dimension, replica vector, round, and $p\in(0,1)$)
    such that under the rounded scales and domain, row, anchor, and clearance hypotheses,
    for $W\geq0$ and a nonzero symmetric replica vector, the energy-error term is at most
    \begin{align}
        C W^2 n^\ell D^2(\log n)^{C_l}
        \left((1-p)\left(\frac{Q}{KnD}\right)^{1/8}
        +(1-p)n^{-P}+p\right),
        \qquad 0<p<1.
        \label{eq:arealaw-generic-charge-cap}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:al_scanner_scale_facts,
        thm:al_scan_transport_coefficients_log}
    Theorem~\ref{thm:al_scan_transport_coefficients_log} bounds the two round-energy coefficients by
    $C_d(\log n)^{C_l}$, and Lemma~\ref{lem:al_scanner_scale_facts} gives $nD^2/(mK)\leq C_s n^\ell D^2$.
    Substituting $a=W/K$ into the physical history density-plus-tail bound yields
    (\ref{eq:arealaw-generic-charge-cap}) with $C=C_sC_d$.
\end{proof}


\begin{theorem}[The same selected charge energy]
    \label{thm:arealaw-generic-selected-energy}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.actualCharge_defect_le_of_density_energy_bound_at_exponent}
    \leanok
    For positive physical and auxiliary dimensions, $\Delta>0$, and $W\geq0$,
    the round-energy inequality and the coefficient cap $\mathrm{cap}$ of
    (\ref{eq:arealaw-generic-charge-cap}) at any selected charge point imply
    \begin{align}
        E_{\rm def} & \leq \frac{e+\mathrm{cap}+r_k}{g/2},
        \label{eq:arealaw-generic-selected-energy}
    \end{align}
    with the truncated eigenvalue $e$, physical prevector, selected point, and mean energy unchanged.
\end{theorem}
\begin{proof}
    \leanok
    Since the energy-error integral is nonnegative, replacing the signed round-energy
    coefficient by its absolute value increases the numerator; substituting $\mathrm{cap}$
    and dividing by $g/2>0$ gives (\ref{eq:arealaw-generic-selected-energy}).
\end{proof}

\begin{theorem}[Principal density at the selected point]
    \label{thm:arealaw-rough-selected-density}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScannerExponents.exists_rough_selectedDensity_principal}
    \leanok
    Fix the scanner tuple, $q\geq1$, $C_e,C_t\geq0$, $C_1>0$, and a logarithmic exponent.
    There exist $C_\delta>0$ and a threshold prior to every $e\geq\kappa$ and finite instance
    such that the target, shell, selected-dimension, mass, rough-endpoint, and logarithmic
    round-energy bounds imply, for all large $k$,
    \begin{align}
        \frac{Q}{KnD} & \leq\delta,
                      & \delta      & =C_\delta\epsilon^{-1}
        \left(n^{e-\mu}+a^{1/4}(\log n)^{C_{l,\delta}}\right),
                      & a           & =4/K,                  & \epsilon & =n^{-\nu}.
        \label{eq:arealaw-rough-principal-density}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:arealaw-source-rough-budget-power,
        thm:arealaw-selected-density-principal,
        lem:al_scanner_scale_facts}
    By Theorem~\ref{thm:arealaw-source-rough-budget-power}, Theorem~\ref{thm:arealaw-selected-density-principal},
    and $D\leq2n^e$, $m\geq n^\mu/2$ (Lemma~\ref{lem:al_scanner_scale_facts}), the non-remainder
    density contribution is at most $\delta/2$ for suitable $C_\delta$, while at fixed physical
    instance the replica remainder tends to $0$ and is eventually $\le\delta/2$.
\end{proof}


\begin{theorem}[Normalized selected energy at bootstrap scales]
    \label{thm:arealaw-bootstrap-selected-energy}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_selected_energy_bound}
    \leanok
    Fix $0<e_0<1$, cap and density coefficients $\ge0$, logarithmic exponents, and $\gamma>0$.
    There is a uniform threshold for all $e\leq e_0$ above which, at bootstrap scales,
    $0\leq\rho\leq\delta$, $p\in[\epsilon,2\epsilon]$ with $p\leq1$, and truncated eigenvalue
    $e_{\rm tr}\le n^{-1000}$ imply that the normalized selected energy cap is at most
    $(1+\gamma^{-1})n^{-\nu/2}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_bootstrap_principal_energy}
    Since $0\leq\rho\leq\delta$ and $\epsilon\leq p\leq2\epsilon$, the parenthesized factor in
    (\ref{eq:arealaw-generic-charge-cap}) is at most $2(\delta^{1/8}+\epsilon+n^{-100})$, so
    Theorem~\ref{thm:area_law_bootstrap_principal_energy} with coefficient $2C/\gamma$ bounds the
    cap term by $n^{-\nu/2}$, while $e_{\rm tr}/\gamma\le\gamma^{-1}n^{-1000}\leq\gamma^{-1}n^{-\nu/2}$
    for $n\geq2$.
\end{proof}


\begin{theorem}[Restored additive entropy bound]
    \label{thm:arealaw-bootstrap-restored-budget}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_bootstrap_restored_entropy_budget}
    \leanok
    For $e_*<e_0<1$, $q\geq1$, and $C_e\geq0$, there exist $C>0$ and a threshold uniform in
    $e\in(e_*,e_0]$ such that for $z\ge1-n^{-100}>0$ and $0\leq\tau\leq1$, the sum of the
    typical-mass ($7(w-\log z)-\log z$), whole-prefix, mismatch, binary-entropy, and
    $n^{-247}$ continuity errors is at most $C n^{1+(1-\ell)e}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_bootstrap_comparison_budget}
    For $n\geq2$, $z\ge1-n^{-100}\ge1/2$ gives $z^{-1}\leq2$ and $-\log z\leq1$, so with
    $C_b=8C_e+12\log q+30$ the total error is at most $C_b(nL^e+nD+n^{3/5}+h(\tau)/a+1)$,
    which Theorem~\ref{thm:area_law_bootstrap_comparison_budget} bounds by
    $C_b(5+\frac{1}{4}\log2)n^{1+(1-\ell)e}$.
\end{proof}


\begin{theorem}[Original-state scanner entropy improvement]
    \label{thm:arealaw-source-bootstrap-entropy}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_sourceBootstrap_entropy_improvement,TNLean.PEPS.AreaLaw.Scan.CollarScan.sourceBootstrapEntropyThreshold,TNLean.PEPS.AreaLaw.Scan.CollarScan.exists_sourceBootstrap_entropy_improvement_of_threshold_le}
    \leanok
    Fix $e_*<e_0<1$, $C_e\geq0$, and the constants of Theorem~\ref{thm:arealaw-source-rough-comparison}.
    There exist $C_r,C_{\rm out}>0$ and $N\geq2$ uniform in $e\in(e_*,e_0]$ and the finite system
    such that, at bootstrap exponents and rounded scales under the graph, target, depth, anchor,
    row, and clearance hypotheses, if $S_\Omega(X)\le C_e n^{1+e}$ and every positive-depth
    prefix $P$ satisfies $S_\Omega(P)\le C_e(nL^e+n)$, then
    \begin{align}
        S_\Omega(X) & \leq C_{\rm out}n^{1+(1-\ell(e_0))e}.
        \label{eq:arealaw-source-bootstrap-improvement}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:fixed_weight_four_transport_smallness,
        thm:scanner_rounded_admissibility,
        thm:arealaw-source-rough-comparison,
        thm:arealaw-rough-selected-density,
        thm:arealaw-bootstrap-selected-energy,
        thm:arealaw-bootstrap-restored-budget,
        thm:arealaw-target-entropy-transfer,
        thm:area_law_selected_schmidt_entropy_bounds,
        thm:area_law_rough_typical_entropy_absorption}
    By Theorems~\ref{thm:fixed_weight_four_transport_smallness} and \ref{thm:scanner_rounded_admissibility},
    Theorems~\ref{thm:arealaw-source-rough-comparison}, \ref{thm:arealaw-rough-selected-density},
    and \ref{thm:arealaw-bootstrap-selected-energy} apply at the selected point. Setting
    $\tau=n^{-\nu/4}$ in the rough comparisons gives relative loss $6(\tau+E_{\rm def}/\tau)$ with
    vanishing replica remainders; substituting the Schmidt entropy bound
    (Theorem~\ref{thm:area_law_selected_schmidt_entropy_bounds}), absorbing typical errors
    (Theorem~\ref{thm:area_law_rough_typical_entropy_absorption}), restoring $\Omega$ via
    Theorem~\ref{thm:arealaw-target-entropy-transfer}, and applying
    Theorem~\ref{thm:arealaw-bootstrap-restored-budget} yields (\ref{eq:arealaw-source-bootstrap-improvement}).
\end{proof}

\begin{theorem}[Weak-rectangle source entropy]
    \label{thm:arealaw-weak-rectangle-source-entropy}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_weakRectangle_source_entropy_bounds,TNLean.PEPS.AreaLaw.Scan.weakRectangleSourceEntropyCoefficient,TNLean.PEPS.AreaLaw.Scan.weakRectangle_source_entropy_bounds,TNLean.PEPS.AreaLaw.Scan.weakRectangleSourceEntropyCoefficient_nonneg}
    \leanok
    Fix $b>0$ and $C_{\rm box}\geq0$, and set $C_e=C_{\rm box}(25+64/(2^b-1))$.
    For every integer $D_0\geq0$, finite state, weakly safe rectangle of size $s$,
    row scale $n\geq s$, collar $1\leq L\leq s$ with $(D_0+1)L\leq D_0s$, and $e\geq b$,
    if the state obeys the pointwise safe-box bound with constant $C_{\rm box}$ at exponent $e$,
    then its target and positive-depth prefixes obey the hypotheses of
    Theorem~\ref{thm:arealaw-source-bootstrap-entropy} with constant $C_e$.
\end{theorem}
\begin{proof}
    \leanok
    The safe-box bound gives target entropy $\le C_{\rm box}s^{1+e}\le C_{\rm box}n^{1+e}$,
    while the weak-rectangle dyadic shell cover bounds each positive prefix by
    $C_{\rm box}(24+64/(2^e-1))sL^e\le C_{\rm box}(24+64/(2^b-1))nL^e$ since $e\ge b$ and $s\le n$.
\end{proof}


\begin{theorem}[Physical powers and the box supremum]
    \label{thm:arealaw-box-supremum-powers}
    \lean{TNLean.PEPS.AreaLaw.regionalEntropy_le_of_boxEntropy_power}
    \leanok
    For $q\geq1$, if $F_{\rm box}(r)\le Cr^{1+e}$ for all $r\ge1$ with $1+e>0$, then every
    safe rectangle $Q$ satisfies $S_\Omega(A\cap Q)\le C\operatorname{size}(Q)^{1+e}$.
\end{theorem}
\begin{proof}
    \leanok
    For $s=\operatorname{size}(Q)\ge1$, $S_\Omega(A\cap Q)\le F_{\rm box}(s)\le Cs^{1+e}$ by definition;
    for $s=0$, $A\cap Q=\varnothing$ gives $S_\Omega(A\cap Q)=0=Cs^{1+e}$.
\end{proof}

\begin{theorem}[Bounded rectangle sizes]
    \label{thm:arealaw-box-supremum-small-sizes}
    \lean{TNLean.PEPS.AreaLaw.boxEntropy_power_le_of_large_rectangles}
    \leanok
    Let $q\geq1$, $C,p\geq0$, and $N\in\mathbb N$. If every safe rectangle $Q$ of size
    $s\geq N$ satisfies $S_\Omega(A\cap Q)\le Cs^p$, then
    $F_{\rm box}(r)\le(C+N^2\log q+1)r^p$ for every $r\geq1$.
\end{theorem}
\begin{proof}
    \leanok
    For $N\le s\le r$, $S_\Omega(A\cap Q)\le Cs^p\le Cr^p$; for $s<N$, the volume bound gives
    $S_\Omega(A\cap Q)\le N^2\log q\le(N^2\log q)r^p$ since $r^p\ge1$.
\end{proof}


\begin{theorem}[Reducing the safety parameter]
    \label{thm:arealaw-box-clearance-comparison}
    \lean{TNLean.PEPS.AreaLaw.boxEntropy_power_le_of_succ_clearance}
    \leanok
    \uses{def:area_law_box_entropy}
    Fix $q,R\in\mathbb N$ with $q\geq1$ and $J,\Delta\in\mathbb R$, and write $F_{\mathrm{box},D}$
    for the safe-box entropy (Definition~\ref{def:area_law_box_entropy}) at safety parameter $D\ge3$.
    If $C,p\geq0$ and $F_{\mathrm{box},D+1}(r)\leq Cr^p$ for all $r\geq1$, then for all $r\geq1$,
    \begin{align}
        F_{\mathrm{box},D}(r) & \leq (4C+4\log q+1)r^p.
        \label{eq:arealaw-box-clearance-comparison}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_box_children,thm:arealaw-box-supremum-small-sizes}
    For any $D$-safe rectangle $Q$ of size $s\geq2$, Theorem~\ref{thm:area_law_box_children} ($M=2$)
    partitions $Q$ into at most $4$ child rectangles $Q_i$ of size $\le r=\lceil s/2\rceil$.
    Since $D\ge3$ and $s\ge2$ imply $(D+1)r\le Ds$, each $Q_i\subseteq Q$ has distance $>Ds\ge(D+1)\operatorname{size}(Q_i)$
    from $Z$ and is $(D+1)$-safe. Subadditivity gives
    \begin{align}
        S_\Omega(A\cap Q) & \leq \sum_i S_\Omega(A\cap Q_i) \leq 4Cr^p \leq 4Cs^p,
        \notag
    \end{align}
    and Theorem~\ref{thm:arealaw-box-supremum-small-sizes} with $N=2$ yields (\ref{eq:arealaw-box-clearance-comparison}).
\end{proof}

        \quad\Longrightarrow\quad

\begin{theorem}[Improved entropy bound for safe rectangles]
    \label{thm:arealaw-physical-safe-rectangle-real}
    \lean{TNLean.PEPS.AreaLaw.exists_regionalEntropy_safe_rect_le_boxError}
    \leanok
    \uses{def:area_law_hamiltonian,def:area_law_ground_entropy,
        def:area_law_safe_rectangle}
    Fix integers $q\geq1$, $R\geq0$ and reals $J\geq0$, $\Delta>0$. There exists $C>0$
    (depending only on $q,R,J,\Delta$) such that for every real $D_0>2R+10$, finite domain
    $\Lambda\subseteq\mathbb Z^2$, $(R,J)$-Hamiltonian $H$ with $\Delta$-gapped ground state
    $\Omega$ (Definition~\ref{def:area_law_hamiltonian}), cut $A\subseteq\Lambda$ with crossing-edge
    endpoints $Z$, and discrete rectangle $Q$ satisfying $D_0\operatorname{size}(Q)<\|t-z\|_\infty$
    for all $t\in Q$, $z\in Z$ (Definition~\ref{def:area_law_safe_rectangle}),
    \begin{align}
        S_\Omega(A\cap Q)
         & \leq C\operatorname{size}(Q)^{1+e_*},
         & e_*                                   & =2\cdot10^{-6}.
        \label{eq:arealaw-physical-real-box}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:arealaw-physical-box-fixed-exponent,
        thm:arealaw-box-clearance-comparison,
        thm:arealaw-box-supremum-powers}
    Put $d=2R+10\ge3$. Theorem~\ref{thm:arealaw-physical-box-fixed-exponent} gives $C_b>0$ with
    $F_{\mathrm{box},d+1}(r)\leq C_b r^{1+e_*}$, and Theorem~\ref{thm:arealaw-box-clearance-comparison}
    gives
    \begin{align}
        F_{\mathrm{box},d}(r)
         & \leq C r^{1+e_*},
         & C                 & =4C_b+4\log q+1>0.
        \label{eq:arealaw-physical-real-box-constant}
    \end{align}
    Since $D_0>d$, every $D_0$-safe $Q$ is $d$-safe, so Theorem~\ref{thm:arealaw-box-supremum-powers}
    and (\ref{eq:arealaw-physical-real-box-constant}) yield (\ref{eq:arealaw-physical-real-box}).
\end{proof}

Taking the supremum of (\ref{eq:arealaw-physical-real-box}) over $\operatorname{size}(Q)\le r$
equivalently gives $F_{\mathrm{box},D_0}(r)\leq Cr^{1+e_*}$ uniformly in real $D_0>2R+10$, proving
\cite[Proposition~3.3 and Proposition~9.5]{OpenAI2026AreaLaw} with $e_0=e_*$
(the collar bound is Theorem~\ref{thm:area_law_sublinear_collar}).
