\begin{lemma}[Direct-sum classification of Fibonacci representations]%
    \label{thm:mpo_symmetry_fibonacci_decomposition}
    \lean{FibonacciCompression.exists_fibNim_pair,
        FibonacciCompression.fibNimPartner,
        FibonacciCompression.fibNimPartner_involutive,
        FibonacciCompression.exists_equiv_prod_of_fibNim_sq,
        FibonacciCompression.fibNim_symmetric_of_sq,
        FibonacciCompression.exists_equiv_prod_of_isNIMRep_fibNim}
    \leanok
    \uses{def:asymex_fib_anomaly_nim}
    Let $T$ be a matrix indexed by a finite set $K$, with entries in $\N$,
    satisfying $T^2=I+T$. Set $R=\{x\in K:T_{xx}=0\}$ and
    $F=\begin{pmatrix}0&1\\1&1\end{pmatrix}$. There is a bijection
    $\sigma:K\longrightarrow R\times\{0,1\}$ for which
    \begin{align}
        T_{\sigma^{-1}(r,i),\sigma^{-1}(s,j)}
         & =\delta_{rs}F_{ij}.
        \label{eq:mpo_fibonacci_direct_sum}
    \end{align}
    In particular, $T$ is symmetric. Every unital nonnegative integer
    representation of the Fibonacci fusion ring is therefore a direct sum
    of regular representations, yielding the decomposition behind the
    two-block action of \cite[Section~6]{GarreRubio2023MPOSym} once the
    missing indecomposability hypothesis recorded in
    \cite{gap:glm23_fibonacci_module_rank_scope} is imposed.
    % Source anchor: arXiv:2203.12563, lines 564--565 and 1991--1993.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:asymex_fib_anomaly_nim}
    The diagonal equation gives $T_{xx}\in\{0,1\}$ and
    $\sum_{z\ne x}T_{xz}T_{zx}=1$. Thus $x$ has a unique partner $y\ne x$
    with $T_{xy}=T_{yx}=1$. For $z\notin\{x,y\}$,
    $T_{xz}T_{zx}=0$ and the $(z,x)$ equation gives $T_{zy}\leq T_{zx}$.
    Hence $T_{xz}T_{zy}=0$, and the $(x,y)$ equation reduces to
    $T_{xx}+T_{yy}=1$. Orienting the pair by $T_{xx}=0$ and $T_{yy}=1$,
    the $(x,z)$ and $(y,z)$ equations imply
    $T_{yz}\leq T_{xz}$ and $T_{xz}+T_{yz}\leq T_{yz}$, respectively;
    hence $T_{xz}=T_{yz}=0$. The partner operation is an involution,
    because $x$ is the only possible partner of $y$. Its pairs partition
    $K$, proving (\ref{eq:mpo_fibonacci_direct_sum}). In a unital
    Fibonacci representation the fusion relation at $a=b=\tau$ is
    precisely $T^2=I+T$.
\end{proof}

\begin{lemma}[Finite normal families with Fibonacci symmetry]%
    \label{thm:mpo_symmetry_fibonacci_normal_decomposition}
    \lean{FibonacciCompression.exists_equiv_prod_of_isMPOSymmetricFamily_fibNim}
    \leanok
    \uses{def:asymex_fib_anomaly_nim, def:mpo_symmetric_family, def:normal}
    Let $O_1,O_\tau$ be a Fibonacci matrix product operator algebra and
    $(A_x)_{x\in K}$ a finite family of normal tensors of positive bond
    dimensions. Suppose its periodic vectors are invariant with
    length-independent coefficients $M_{a,x}^y$, and linearly independent
    at a positive length $N_0$. Assume that $O_1^{(N_0)}$ fixes each of
    those vectors. Set $R=\{x\in K:M_{\tau,x}^x=0\}$.
    There is a bijection $\sigma:K\longrightarrow R\times\{0,1\}$ such that
    \begin{align}
        M_{a,\sigma^{-1}(r,i)}^{\sigma^{-1}(s,j)}
         & =\delta_{rs}N_{ai}^j.
        \label{eq:mpo_fibonacci_normal_sum}
    \end{align}
    In particular, for arbitrary finite $K$ (including the empty family)
    and for the Fibonacci operators of Section~\ref{sec:asymex_fibonacci}
    \cite[Sections~3 and~6]{GarreRubio2023MPOSym}, the cardinality
    $|K|=2|R|$ is even.
    % Source anchor: arXiv:2203.12563, lines 458--460, 564--568, 1991--1993.
    % This theorem classifies multiplicities, not the gauge classes of L-symbols.
\end{lemma}

\begin{proof}
    \uses{thm:mpo_symmetric_nimrep, thm:mpo_symmetry_fibonacci_decomposition}
    \leanok
    The symmetry and normality hypotheses give natural-number coefficients
    forming a representation of the fusion ring. Linear independence at
    $N_0$ compares $O_1^{(N_0)}\ket{V^{(N_0)}(A_x)}$ with
    $\ket{V^{(N_0)}(A_x)}$, and forces $M_{1,x}^y=\delta_{xy}$.
    The direct-sum theorem then gives (\ref{eq:mpo_fibonacci_normal_sum}).
    Its representatives have zero diagonal coefficient, so the displayed
    bijection gives the cardinality formula.
\end{proof}

\begin{lemma}[The indecomposable Fibonacci action]%
    \label{thm:mpo_symmetry_fibonacci_indecomposable}
    \lean{FibonacciCompression.exists_equiv_of_isIndecomposable_fibNim}
    \leanok
    \uses{def:asymex_fib_anomaly_nim}
    A nonempty finite indecomposable nonnegative integer representation
    of the Fibonacci ring, with the unit acting as the identity, has
    exactly two blocks. Here indecomposability means that $M_\tau$ cannot
    be permuted into block-triangular form with two nonempty diagonal blocks.
    As in \cite[Section~6]{GarreRubio2023MPOSym}, the blocks can be
    labelled $x_1,x_\tau$ so that $\tau\cdot x_1=x_\tau$ and
    $\tau\cdot x_\tau=x_1+x_\tau$.
    % Source anchor: arXiv:2203.12563, line 1993.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:mpo_symmetry_fibonacci_decomposition,
        thm:mpo_symmetry_fibonacci_rank_two}
    Each isolated pair in (\ref{eq:mpo_fibonacci_direct_sum}) is a nonempty
    invariant subset. Indecomposability makes it all of $K$, and the
    two-block classification identifies the action with the regular one.
\end{proof}

\begin{lemma}[Exact trees of the two-block Fibonacci realization]%
    \label{lem:mpo_symmetry_fibonacci_exact_trees}
    \lean{MPSTensor.isBiorthogonalDecomposition_complexOfRing,
        FibonacciCompression.isBiorthogonalDecomposition_fibFusion,
        FibonacciCompression.isBiorthogonalDecomposition_fibAct}
    \leanok
    \uses{def:glm_boundary_decomposition, def:asymex_fib,
        thm:asymex_fib_anomaly_nim_rep}
    Let $O_1,O_\tau$ be the Fibonacci operator blocks and let
    $A_{x_1},A_{x_\tau}$ be the two normal states of
    Lemma~\ref{thm:asymex_fib_anomaly_nim_rep}. For every pair of labels
    the product $O_aO_b$ is an exact biorthogonal decomposition onto the
    blocks $O_c$ with multiplicity $N_{ab}^c$, and for every label $a$ and
    every state block $x$ the action $O_a\cdot A_x$ is an exact biorthogonal
    decomposition onto the blocks $A_y$ with multiplicity $N_{ax}^y$, as in
    \cite[Section~3]{GarreRubio2023MPOSym}. The analysis and synthesis maps
    have entries in $\mathbb{Z}[\sigma]$, $\sigma=\varphi^{-1/2}$.
    % Source anchor: arXiv:2203.12563, lines 361--391, 459--490 and 1991--1993.
\end{lemma}

\begin{proof}
    \leanok
    Each retraction, orthogonality and letter identity is an identity of
    matrices over $\mathbb{Z}[\sigma]$, checked there and carried to the
    complex matrices by the embedding of $\mathbb{Z}[\sigma]$. Every letter
    of each product and action tensor is the sum of its blocks, so no
    remainder occurs.
\end{proof}

\begin{lemma}[The Fibonacci L-symbol is the F-symbol up to gauge]%
    \label{lem:mpo_symmetry_fibonacci_l_symbol}
    \lean{FibonacciCompression.fib_actionLMatrix_eq,
        FibonacciCompression.fib_actionLMatrix_eq_gauge_fSymbol}
    \leanok
    \uses{lem:mpo_symmetry_fibonacci_exact_trees,
        def:glm_boundary_action_l_matrix, def:asymex_fib_fsymbols}
    For the exact trees of Lemma~\ref{lem:mpo_symmetry_fibonacci_exact_trees},
    the L-symbol $L_{\tau\tau x}^{y}$ equals $1$ on the three
    one-dimensional blocks, and on the block $x=y=x_\tau$ it equals
    \begin{align}
        L_{\tau\tau x_\tau}^{x_\tau} & =
        \begin{pmatrix}
            \varphi^{-1} & \varphi       \\
            \varphi^{-2} & -\varphi^{-1}
        \end{pmatrix},
        \label{eq:mpo_fibonacci_l_symbol}
    \end{align}
    with rows indexed by the intermediate block $z$ and columns by the
    fusion channel $c$. With nonzero multiplicity gauges $X_{ax}^{y}$, the
    hatted gauge equal to the inverse and a trivial fusion gauge,
    \begin{align}
        \bigl(L_{\tau\tau x}^{y}\bigr)_{c}^{z}
         & =X_{\tau z}^{y}X_{\tau x}^{z}\bigl(X_{cx}^{y}\bigr)^{-1}
        \bigl(F^{\tau\tau x}_{y}\bigr)_{c}^{z}
        \label{eq:mpo_fibonacci_l_equals_f}
    \end{align}
    for all labels, so the L-symbol of the regular module is the gauge
    transform of the F-symbol, which is how the equality of L- and F-symbols in
    \cite[Section~3.1]{GarreRubio2023MPOSym} is read.
    % Source anchor: arXiv:2203.12563, lines 517--546 and 593--605;
    % arXiv:1511.08090, lines 1245--1257.
\end{lemma}

\begin{proof}
    \leanok
    The normalized trace defining each entry is the image of a trace of a
    product of matrices over $\mathbb{Z}[\sigma]$, which is computed
    there. The gauges $X_{1x_1}^{x_1}=X_{1x_\tau}^{x_\tau}
        =X_{\tau x_1}^{x_\tau}=\varphi^{3/2}$, with all other gauges equal
    to $1$, then give (\ref{eq:mpo_fibonacci_l_equals_f}) entry by entry.
    Both sides of (\ref{eq:mpo_fibonacci_l_equals_f}) on the block
    $x=y=x_\tau$ have the gauge-invariant cross ratio $-\varphi^{-1}$.
\end{proof}


\begin{theorem}[A standard gauge for every Fibonacci action sector]%
    \label{thm:mpo_symmetry_fibonacci_l_all_sectors}
    \lean{FibonacciCompression.Boundary.gauge_mul_inv,
        FibonacciCompression.Boundary.gauge_ne_zero,
        FibonacciCompression.Boundary.fusion_isBiorthogonal,
        FibonacciCompression.Boundary.action_isBiorthogonal,
        FibonacciCompression.Boundary.states_isNormal,
        FibonacciCompression.Boundary.actionTrees_isBiorthogonal,
        FibonacciCompression.Boundary.actionLMatrix_eq_fSymbol,
        FibonacciCompression.Boundary.inverseActionLMatrix_eq_fSymbol,
        FibonacciCompression.Boundary.actionLMatrix_tau,
        FibonacciCompression.Boundary.actionLMatrix_inverse,
        FibonacciCompression.Boundary.fusionFMatrix_eq_fSymbol,
        FibonacciCompression.Boundary.actionLMatrix_eq_fusionFMatrix}
    \uses{lem:mpo_symmetry_fibonacci_exact_trees,
        lem:mpo_symmetry_fibonacci_l_symbol}
    For the two normal blocks and exact maps of
    Lemma~\ref{lem:mpo_symmetry_fibonacci_exact_trees}, multiply both fusion
    and action analysis in the $(\tau,\tau,1)$ and $(\tau,\tau,\tau)$
    channels by $\sigma$ and $-\sigma^{-1}$, respectively, and multiply
    synthesis by the reciprocal factors. All other channel scales are $1$.
    The maps remain exact biorthogonal decompositions. For every choice of
    labels $a,b,x,y$, the resulting action comparison and the fusion
    comparison computed from the same maps satisfy
    \[
        (L_{abx}^{y})_{c,k\mu}^{z,ij}
        = [F^{abx}_{y}]_{c,\mu k}^{z,ji}.
    \]
    Every nonempty one-dimensional sector is $1$. With intermediate
    labels ordered as $1,\tau$, the only two-dimensional sector is
    \[
        L_{\tau\tau\tau}^{\tau}
        =F^{\tau\tau\tau}_{\tau}
        =\begin{pmatrix}\sigma^2&\sigma\\\sigma&-\sigma^2\end{pmatrix}.
    \]
    The reverse action comparison has the transposed coefficients.
    The forward and reverse matrices are inverse, including on empty
    multiplicity spaces.
\end{theorem}
\begin{proof}
    The reciprocal channel factors preserve retractions and letter
    reconstruction. In common incoming bond coordinates, each sequential
    analysis followed by fusion-then-action synthesis is the displayed
    coefficient times the identity of the final state bond. The normalized
    trace gives $L$. Left fusion analysis followed by right fusion
    synthesis gives the same coefficient, with the opposite left and right
    multiplicity index types. The reverse contractions and multiplication
    of the displayed coefficient matrices give both inverse identities.
\end{proof}
