\section{The memory of several sheets}
\label{sec:peps_sheet_stacks}
In the protocol of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} the memory
holds the main sheet, the active auxiliary sheets, the frozen auxiliary sheets
and the copies of $\Omega$ not yet in use. Every copy of $\Omega$, including the
main copy at the root, is initialized privately at its future owner, and the
reference vector at every stage is the tensor product of the reference vectors
of the encoded frames of the sheets, every factor of norm at most one
(\texttt{06-geometry.tex}, lines 653--681). This section collects the
bookkeeping of this memory: the initial private maps, the reorderings of the
sheets, the changes that leave a frame unchanged, and the norm of the reference
vector. The initialization assumes that the product of the reference vectors of the
frames of each party has norm one; for the raw frames of the initial memory, which have
no holes, this is the normalization of $\Omega$.

\begin{definition}[The memory of several sheets]
    \label{def:peps_sheet_stack}
    \lean{TNLean.PEPS.EncodedFrame.stackRegs,
        TNLean.PEPS.EncodedFrame.SheetStack,
        TNLean.PEPS.EncodedFrame.Frame.regRef,
        TNLean.PEPS.EncodedFrame.stackVec,
        TNLean.PEPS.EncodedFrame.framesVec}
    \leanok
    \uses{def:peps_frame_registers}
    For a list of encoded frames $\mathcal F_1,\dots,\mathcal F_m$ and a list $\ell$
    of further registers, the memory of the sheets is the tensor product of the
    registers of the frames, in order, followed by those of $\ell$. For a vector
    $\xi$ of the registers of $\ell$, the reference vector of the sheets is
    $\Omega_{\mathcal F_1}\otimes\cdots\otimes\Omega_{\mathcal F_m}\otimes\xi$.
\end{definition}

\begin{theorem}[Reordering the sheets]
    \label{thm:peps_sheet_reordering}
    \lean{TNLean.PEPS.EncodedFrame.stackRegs_append,
        TNLean.PEPS.EncodedFrame.stackVec_append,
        TNLean.PEPS.EncodedFrame.exists_reordering_stackVec,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_stackVec_of_perm}
    \leanok
    \uses{def:peps_sheet_stack}
    Let $\mathcal F'_1,\dots,\mathcal F'_m$ be a permutation of the frames
    $\mathcal F_1,\dots,\mathcal F_m$. There is a word of exchanges of tensor
    factors from the memory of the first list, followed by $\ell$, to that of the
    second list, followed by $\ell$, which carries the reference vector of the first
    list to that of the second for every vector $\xi$ of $\ell$. As an original
    circuit it has no nonprivate gate. In particular an auxiliary sheet can be moved
    next to the main sheet before a lens exchange.
\end{theorem}

\begin{proof}\leanok
    Induct on the permutation. Prepending a common frame frames the word by its
    registers, and two adjacent frames are interchanged by exchanging their two
    blocks of registers.
\end{proof}

\begin{theorem}[The initial private maps]
    \label{thm:peps_sheet_initialization}
    \lean{TNLean.PEPS.EncodedFrame.framesSpace,
        TNLean.PEPS.EncodedFrame.splitMap,
        TNLean.PEPS.EncodedFrame.splitMap_tmul_one,
        TNLean.PEPS.EncodedFrame.norm_splitMap_le,
        TNLean.PEPS.EncodedFrame.splitCircuit,
        TNLean.PEPS.EncodedFrame.nonprivateCount_splitCircuit,
        TNLean.PEPS.EncodedFrame.isExpansionBounded_splitCircuit,
        TNLean.PEPS.EncodedFrame.isMonomialBounded_splitCircuit,
        TNLean.PEPS.EncodedFrame.participants_splitCircuit_subset,
        TNLean.PEPS.EncodedFrame.eval_splitCircuit,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_initial}
    \leanok
    \uses{def:peps_sheet_stack}
    Let every frame of a list be held entirely by one party: all its registers,
    raw and tag, belong to that party. Give each party, as its whole initial private
    memory, the registers of its frames in the order of the list, and as its
    initial vector the product of their reference vectors, assumed of norm one.
    Then one private map per party, followed by a reordering of the frames, carries
    the product of the initial vectors of all parties exactly to the reference
    vector of the list. The resulting circuit has no nonprivate gate. This is the
    pure product initialization of \texttt{06-geometry.tex}, lines 658--662.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_reordering}
    Each private map identifies the whole memory of a party with the memory of its
    frames. The concatenation, over the parties, of their lists of frames is a
    permutation of the list: a frame occurs in the list of its party exactly as
    often as in the list. Theorem~\ref{thm:peps_sheet_reordering} reorders the
    frames.
\end{proof}

\begin{theorem}[The initial step from raw copies]
    \label{thm:peps_sheet_raw_initialization}
    \lean{TNLean.PEPS.EncodedFrame.Frame.norm_refVec_of_holes_eq_nil,
        TNLean.PEPS.EncodedFrame.norm_framesVec_of_holes_eq_nil,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_initial_of_holes_eq_nil}
    \leanok
    \uses{thm:peps_sheet_initialization}
    A frame without holes has reference vector of norm $\lVert\Omega\rVert$. Let
    $\lVert\Omega\rVert=1$ and let every frame of the list have no hole and be held
    entirely by one party: the main copy at the root and the auxiliary copies at
    their future owners. Then the product of the reference vectors of the frames of
    every party has norm one, and the initial private maps of
    Theorem~\ref{thm:peps_sheet_initialization} apply (\texttt{06-geometry.tex},
    lines 658--667).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_initialization, thm:peps_frame_norm}
    Without holes the product of the hole projectors is the identity, so
    $\lVert\Omega_{\mathcal F}\rVert=\lVert\Omega\rVert=1$, and the norm of a product
    of reference vectors is the product of their norms.
\end{proof}

\begin{theorem}[Changes that leave a frame unchanged]
    \label{thm:peps_identity_change}
    \lean{TNLean.PEPS.EncodedFrame.exists_originalCircuit_of_frame_eq,
        TNLean.PEPS.EncodedFrame.Frame.changeOwner_empty,
        TNLean.PEPS.Approximation.emptyTarget_step}
    \leanok
    \uses{def:peps_sheet_stack}
    A change between equal frames, such as a birth or death whose raw target is
    empty or the resize of a hole whose radius does not change, is the circuit
    without operations: it has no nonprivate gate and carries the reference vector
    of the frame to itself. Changing the owner of an empty set of raw registers
    leaves a frame unchanged (\texttt{06-geometry.tex}, lines 432--438 and
    529--541).
\end{theorem}

\begin{proof}\leanok
    Immediate.
\end{proof}

\begin{theorem}[The norm of the reference vector of several sheets]
    \label{thm:peps_sheet_stack_norm}
    \lean{TNLean.PEPS.EncodedFrame.one_sub_mul_le_norm_stackVec,
        TNLean.PEPS.EncodedFrame.one_sub_div_pow_le_norm_stackVec}
    \leanok
    \uses{def:peps_sheet_stack,thm:peps_frame_norm}
    Let $\lVert\Omega\rVert=1$, and let every hole of every frame
    $\mathcal F_1,\dots,\mathcal F_m$ satisfy
    $\lVert(\id-P)\Omega\rVert\le\varepsilon$. If the frames have $r$ holes in
    total, then
    \begin{align*}
        (1-r\varepsilon)\lVert\xi\rVert
         & \le\lVert\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi\rVert .
    \end{align*}
    In particular, if $\varepsilon=L^{-60}$, $r\le CL^3$ and
    $\lVert\xi\rVert=1$, the reference vector has norm between $1-CL^{-57}$ and
    $1$. The bound $r\le CL^3$ on the total number of holes is a hypothesis here;
    in the source it is the count of frozen auxiliary holes,
    \texttt{06-geometry.tex}, lines 653--656 and 674--678.
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_frame_norm}, the frame $\mathcal F_i$ with $r_i$ holes
    has $1-r_i\varepsilon\le\lVert\Omega_{\mathcal F_i}\rVert\le1$. For
    $1-a\le x\le1$ and $(1-b)z\le y\le z$ one has
    $xy=y-(1-x)y\ge y-az\ge(1-a-b)z$; induct over the frames.
\end{proof}

        TNLean.PEPS.EncodedFrame.readoutCircuit}
    \leanok
    \uses{def:peps_sheet_stack}
    Let the main frame $\mathcal M$ be the first sheet of the memory. The final regrouping moves the raw registers of
    $\mathcal M$ in front of its tag registers, renames the raw register of each
    site, by a private map at its party, as the physical register of that party,
    and lets every party without a site prepare a physical register of dimension
    one by a private map. The physical registers are listed in the order of the
    sites, followed by the parties without a site; the private registers are the
    tags of $\mathcal M$ followed by the registers of the other sheets and of the
    idle copies (\texttt{06-geometry.tex}, lines 48--51 and 669--681).
\end{definition}

\begin{theorem}[The final reference vector]
    \label{thm:peps_final_reference}
    \lean{TNLean.PEPS.PairEffect.OriginalCircuit.isEmpty_nonprivateLocations_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.participants_subset_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.isExpansionBounded_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.isMonomialBounded_of_nonprivateCount_eq_zero,
        TNLean.PEPS.EncodedFrame.eq_trivialTag,
        TNLean.PEPS.EncodedFrame.rawProd_trivialTag,
        TNLean.PEPS.EncodedFrame.Frame.regRef_of_outer_eq_empty,
        TNLean.PEPS.EncodedFrame.familyPhysicalListBasis_eq_physVec,
        TNLean.PEPS.EncodedFrame.familyLabelledPhysicalBasis_eq_physVec,
        TNLean.PEPS.EncodedFrame.eval_freshCircuit,
        TNLean.PEPS.EncodedFrame.eval_siteCircuit,
        TNLean.PEPS.EncodedFrame.nodup_siteOrder,
        TNLean.PEPS.EncodedFrame.mem_siteOrder,
        TNLean.PEPS.EncodedFrame.nonprivateCount_readoutCircuit,
        TNLean.PEPS.EncodedFrame.eval_readoutCircuit_stackVec,
        TNLean.PEPS.EncodedFrame.readout_eval_readoutCircuit_stackVec,
        TNLean.PEPS.EncodedFrame.exists_readoutCircuit_norm,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv_apply_val,
        TNLean.PEPS.Approximation.DyadicProtocol.readout_eq_euclideanTensorVector}
    \leanok
    \uses{def:peps_final_regrouping, def:peps_dyadic_protocol}
    The final regrouping has no nonprivate gate. Let the raw owner of the main frame $\mathcal M$ be
    injective, and let every hole of the main frame
    $\mathcal M$ have empty physical outer sample, let the physical dimension be
    $q$ at the party of every site and one at every other party, and read the
    physical configurations of the parties as configurations of the sites. Then,
    for every $\Omega$ and every vector $\xi$ of the idle registers, the
    physical-private coordinates of the regrouped reference vector
    $\Omega_{\mathcal M}\otimes\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi$ are those of $\Omega\otimes\Xi$, where $\Xi$
    is the vector of the private registers and
    $\lVert\Xi\rVert=\lVert\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi\rVert$. If moreover
    $\lVert\Omega\rVert=\lVert\xi\rVert=1$, every hole of the other frames
    satisfies $\lVert(\id-P)\Omega\rVert\le L^{-60}$ and the other frames have at
    most $CL^3$ holes in total, then $1-CL^{-57}\le\lVert\Xi\rVert\le1$. The bound
    on the number of holes is a hypothesis here. For a protocol whose final layout
    is the output of the regrouping, the readout of the protocol is then
    $\Omega\otimes\Xi$ (\texttt{06-geometry.tex}, lines 43--51 and 669--681).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_stack_norm}
    A hole with empty outer sample has only the identity tag, with identity raw
    part, so $\Omega_{\mathcal M}=\sum_c\Omega(c)\,\lvert\tau_0\rangle\otimes
        \lvert c\rangle$. On a basis vector $\lvert c\rangle$ of the raw registers, the
    exchanges move $\lvert c\rangle$ to the front, the renaming sends it to the
    physical basis vector of $c$, and the fresh registers add the basis vector of
    the one-point coordinate. By linearity, the regrouped vector is
    $\sum_c\Omega(c)\,\lvert c\rangle\otimes\pi$, with $\pi$ the product of the tags
    and the other factors, and its coordinates are $\Omega(c)$ times those of
    $\pi$. The norm bound is Theorem~\ref{thm:peps_sheet_stack_norm}.
\end{proof}

\begin{theorem}[The initial step of the dyadic protocol]
    \label{thm:peps_dyadic_protocol_initial_step}
    \lean{TNLean.PEPS.Approximation.DyadicProtocol.exists_originalCircuit_initial,
        TNLean.PEPS.Approximation.rootSquare,
        TNLean.PEPS.Approximation.initialFrames,
        TNLean.PEPS.Approximation.length_initialFrames_le,
        TNLean.PEPS.Approximation.rawHolder,
        TNLean.PEPS.Approximation.mem_initialFrames,
        TNLean.PEPS.Approximation.DyadicProtocol.exists_originalCircuit_initialFrames}
    \leanok
    \uses{def:peps_sheet_stack, def:peps_dyadic_protocol,
        def:peps_dyadic_protocol_parties, def:peps_dyadic_protocol_schedule}
    Let $L\ge1$ and $\lVert\Omega\rVert=1$. The initial sheets are the main copy of $\Omega$,
    a frame without holes held entirely by the party of the root square, and one
    copy for each change of the schedule that initializes an auxiliary sheet,
    a frame without holes held entirely by the future owner of that copy; there
    are at most $1+4(\#\text{squares}+\#\text{junctions})$ of them. Suppose that
    the whole initial private memory of each party of a protocol is the memory
    of the initial sheets it holds, and that its initial vector is the product of
    their reference vectors. Then one private map per party followed by a
    reordering of the sheets, with no nonprivate gate, carries the initial
    reference vector of the protocol exactly to the reference vector of the
    initial sheets. The future owners of the auxiliary copies are arbitrary
    (\texttt{06-geometry.tex}, lines 21--23, 576, 649--652 and 658--667).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_raw_initialization, thm:peps_dyadic_protocol_schedule_counts}
    Every initial sheet has no hole and is held by one party, so
    Theorem~\ref{thm:peps_sheet_raw_initialization} applies; the initial layout of
    the protocol is the layout of whole party memories, identified with the
    memory of the frames of each party. The number of auxiliary copies is the
    number of changes of the schedule that initialize an auxiliary sheet, at most
    four for each square and junction.
\end{proof}

    Let $L\le 2^k<2L$ and $\lVert\Omega\rVert=1$. At the end of the last level
    $n=1$ every unit block of the root square carries its own label, so the raw
    owner of a site, at the cell center $(i+\frac12,j+\frac12)$, is the party of
    its own unit square, and distinct sites have distinct owners. Let the holes of
    the main frame sit at grid corners with patches of outer radius below
    $\frac12$. Let the other sheets be frozen auxiliary frames, one for each
    change of the schedule that initializes an auxiliary sheet, each with at most
    as many holes as its final guide has true vertices and with patches supplied
    by the adaptive patch constraint at accuracy $L^{-60}$, followed by idle registers with a

    the party of its unit square, and its final layout and final reference vector
    are those of the final regrouping of these sheets, then the readout of the
    final reference vector is $\Omega\otimes\Xi$ with
    $1-214\,L^{-57}\le\lVert\Xi\rVert\le1$. The frozen auxiliary frames and the

\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_final_reference, thm:peps_guide_frame_norm,

        thm:peps_dyadic_repainting_valid_labels}
    A site at $(i+\frac12,j+\frac12)$ lies in the unit block $(i,j)$, which lies in
    the root square and has been repainted to its own label; that label is held
    by the party of the unit square $(i,j)$. By Theorem~\ref{thm:peps_guide_frame_norm}
    every hole of the main frame has empty physical outer sample. Supplied patches
    have projector error at most $L^{-60}$, and by
    Lemma~\ref{lem:peps_dyadic_frozen_hole_count} the frozen auxiliary frames carry
    at most $214\,L^3$ holes in total. Theorem~\ref{thm:peps_final_reference} gives
    the readout.
\end{proof}

\begin{definition}[Bounded changes of the sheets]
    \label{def:peps_bounded_change}
    \lean{TNLean.PEPS.PairEffect.OriginalCircuit.isExpansionBounded_mono,
        TNLean.PEPS.PairEffect.OriginalCircuit.norm_eval_le_norm,
        TNLean.PEPS.PairEffect.OriginalCircuit.IsBoundedChange,
        TNLean.PEPS.EncodedFrame.HasBoundedChange}
    \leanok
    \uses{def:peps_sheet_stack}
    An original circuit is a bounded change from a vector $x$ to a vector $y$, with
    participants in a set $S$ and bounds $r$ and $K$ and error $\varepsilon$, if it
    has at most one nonprivate gate, every participant of that gate lies in $S$,
    every monomial of its expansion has at most $r$ pair effects, its coefficient
    sum and its number of monomials are at most $K$, and it takes $x$ within
    $\varepsilon$ of $y$. A bounded change of the sheets from the frames
    $\mathcal F_1,\dots,\mathcal F_m$ to the frames
    $\mathcal F'_1,\dots,\mathcal F'_{m'}$, followed by the same further registers
    $\ell$ with vector $\xi$, is a bounded change between their memories from
    $\Omega_{\mathcal F_1}\otimes\cdots\otimes\Omega_{\mathcal F_m}\otimes\xi$ to
    $\Omega_{\mathcal F'_1}\otimes\cdots\otimes\Omega_{\mathcal F'_{m'}}\otimes\xi$.
    These are the per-change requirements of items 2 and 4 of
    \cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} (\texttt{06-geometry.tex},
    lines 25--41 and 572--583).
\end{definition}

\begin{theorem}[Operations that preserve bounded changes]
    \label{thm:peps_bounded_change_closure}
    \lean{TNLean.PEPS.PairEffect.OriginalCircuit.IsBoundedChange.mono,
        TNLean.PEPS.PairEffect.OriginalCircuit.IsBoundedChange.comp_right,
        TNLean.PEPS.PairEffect.OriginalCircuit.IsBoundedChange.comp_left,
        TNLean.PEPS.PairEffect.OriginalCircuit.isBoundedChange_id,
        TNLean.PEPS.PairEffect.Word.isBoundedChange_toOriginalCircuit,
        TNLean.PEPS.PairEffect.PartyChain.isBoundedChange_toOriginalCircuit,
        TNLean.PEPS.EncodedFrame.pairRef_eq_twoLayoutIso_symm,
        TNLean.PEPS.EncodedFrame.HasBoundedChange.mono,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_of_perm,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_refl,
        TNLean.PEPS.EncodedFrame.HasBoundedChange.perm_left,
        TNLean.PEPS.EncodedFrame.HasBoundedChange.perm_right,
        TNLean.PEPS.EncodedFrame.isBoundedChange_liftOne,
        TNLean.PEPS.EncodedFrame.isBoundedChange_liftTwo,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_liftOne,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_liftTwo,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_activate,
        TNLean.PEPS.EncodedFrame.hasBoundedChange_perm_liftOne,
        TNLean.PEPS.EncodedFrame.exists_isBoundedChange_of_frame_eq}
    \leanok
    \uses{def:peps_bounded_change, thm:peps_sheet_reordering, thm:peps_identity_change}
    Let $\lVert\Omega\rVert\le1$ and $\lVert\xi\rVert\le1$.
    \begin{enumerate}
        \item A permutation of the sheets, and the change between equal frames, are
              bounded changes with any participants, any bounds and any error
              $\varepsilon\ge0$. A permutation of the sheets before or after a bounded
              change gives a bounded change with the same data.
        \item A bounded change of one frame $\mathcal F$ into $\mathcal F'$, from
              $\Omega_{\mathcal F}$ to $\Omega_{\mathcal F'}$, is a bounded change of the
              sheets $\mathcal F_1,\dots,\mathcal F,\dots,\mathcal F_m$ into
              $\mathcal F_1,\dots,\mathcal F',\dots,\mathcal F_m$ with the same data; so
              is a bounded change of two adjacent frames, from
              $\Omega_{\mathcal F_1}\otimes\Omega_{\mathcal F_2}$ to
              $\Omega_{\mathcal F'_1}\otimes\Omega_{\mathcal F'_2}$. In particular an idle
              auxiliary copy can be moved next to the main sheet and then changed together
              with it.
        \item An allowed word using only the parties of $S$ with at most one pair source
              is a bounded change with its own error, for every $r$ and every $K\ge1$. An
              allowed chain using only the parties of $S$, without pair sources and with at
              most one pair effect, is a bounded change for every $r\ge1$ and $K\ge1$.
    \end{enumerate}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_reordering}
    A permutation of the sheets is a word of exchanges, without nonprivate gate and
    without error, and an original circuit is a contraction, so composing with it
    does not increase the error. The spectator sheets of a lifted change have
    reference vectors of norm at most one. A word becomes an original circuit with
    one two-party gate for each pair source, and a chain one more gate for each pair
    effect, each with a single monomial of coefficient one.
\end{proof}

\begin{theorem}[Frames under maps agreeing off the placeholder]
    \label{thm:peps_guide_frame_owner_agreement}
    \lean{TNLean.PEPS.Approximation.guideFrame_eq_of_owner_eq}
    \leanok
    \uses{def:peps_guide_frame, def:peps_dyadic_protocol_parties}
    Fix the hierarchy, the sites, their positions and the local configuration size.
    Let $g$ be a guide taking only valid labels, with no placeholder label at any
    site. Let $\rho,\rho'$ be any two maps from guide labels to protocol parties such
    that
    $\rho(\ell)=\rho'(\ell)$ for every valid non-placeholder label $\ell$.
    Fix the same ranking of labels, a finite set $V$ of true vertices of $g$, and
    the same square patches with pairwise disjoint physical outer samples at
    distinct centers in $V$. Then the encoded frames of $g$ read through
    $\rho$ and $\rho'$ are equal.
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_guide_frame, lem:peps_guide_frame_tags, lem:peps_dyadic_tags}
    The raw owners agree at every site because its label is not the placeholder.
    The canonical tag holder at each true vertex is a valid non-placeholder
    label, by Lemmas~\ref{lem:peps_guide_frame_tags}
    and~\ref{lem:peps_dyadic_tags}. Thus the tag owners also agree. The site
    owners and the list of holes are equal, so the frames are equal.
\end{proof}

    different rates. So the readings agree, and with them the true vertices, the
    owners and the tags. The party map may be read through any repainting, since the
    readings carry no placeholder at the sites. The uniform auxiliary guide has no true
    vertex.
\end{proof}

\begin{lemma}[Chains of changes]
    \label{lem:peps_protocol_change_chains}
    \lean{TNLean.PEPS.Approximation.IsStepChain,
        TNLean.PEPS.Approximation.IsStepChain.refl,
        TNLean.PEPS.Approximation.IsStepChain.single,
        TNLean.PEPS.Approximation.IsStepChain.trans,
        TNLean.PEPS.Approximation.IsStepChain.map,

        TNLean.PEPS.Approximation.isStepChain_main,

        TNLean.PEPS.Approximation.isStepChain_aux,
        TNLean.PEPS.Approximation.hasSourceBoundedChange_of_hasPairChange}
    \leanok
    \uses{thm:peps_protocol_level_frame_facts, thm:peps_protocol_lens_exchange_change}
    Chains of $m$ and $n$ steps compose to a chain of $m+n$ steps. If adding one mark
    of a list to any set of marks is a step, there is a chain of as many steps as marks
    from no mark to all of them, and likewise for removing the marks. With a bulk step
    between, these give a chain of $2m+1$ steps. A chain of bounded changes of the main
    sheet, or of one auxiliary sheet, is a chain of bounded changes of the memory, and
    so is a bounded change of the main and one auxiliary sheet
    (\texttt{06-geometry.tex}, lines 328--339 and 572--583).
\end{lemma}

\begin{proof}\leanok
    Concatenate the sequences of states. The memory changes are the lifts of one or
    two sheets of the sheet stack, after a reordering of the sheets.
\end{proof}

\begin{theorem}[The edge construction as changes of the memory]
    \label{thm:peps_protocol_edge_construction_chain}
    \lean{TNLean.PEPS.Approximation.isStepChain_auxBirths,
        TNLean.PEPS.Approximation.isStepChain_mainSteps,
        TNLean.PEPS.Approximation.isStepChain_lensExchange,
        TNLean.PEPS.Approximation.isStepChain_edgeConstruction}
    \leanok
    \uses{def:peps_protocol_treated_frame, thm:peps_protocol_sheet_operation_chains}
    Assume the angular information bound at the scale $t=D\log L$, as provided by
    Proposition~3.4. In the setting of
    Theorem~\ref{thm:peps_protocol_sheet_operation_chains}, let the memory carry the
    main sheet with the central birth and some edges done, treated at no mark, and the
    raw copy held by the party of the label $C_e$ across an edge $e$ not yet done. The
    initialization of the auxiliary sheet of $e$, its three births, the lens exchange
    and the four operations of the main sheet, with their point treatments, are $45$
    bounded changes of the memory with the bounds of the small-patch rewrites. They
    complete $e$ on the main sheet and leave the auxiliary sheet of $e$ with the
    exchanged auxiliary word (\texttt{06-geometry.tex}, lines 183--221, 328--339 and
    389--430).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_protocol_sheet_operation_chains, thm:peps_protocol_lens_exchange_change,
        thm:peps_protocol_treated_frame_ends, lem:peps_protocol_change_chains,
        thm:peps_dyadic_no_changed_vertex, thm:peps_protocol_event_segments}
    The initialization is the identity change, since the fresh auxiliary sheet is the
    raw copy. The three auxiliary births are $15$ changes of the auxiliary sheet. The
    main sheet passes to the starting word without change. The lens exchange, with the
    point treatments of both sheets about the endpoints of $e$, is $9$ changes, and the
    four operations of the main sheet are $20$ changes. Each birth and death introduces
    one of the labels $A$, $B$, $C_e$, which are local labels of the repainting.
\end{proof}
