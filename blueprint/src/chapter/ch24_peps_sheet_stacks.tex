\section{The memory of several sheets}
\label{sec:peps_sheet_stacks}
In the protocol of~\cite[Proposition~7.1]{OpenAI2026PolynomialPEPS} the memory
holds the main sheet, the active auxiliary sheets, the frozen auxiliary sheets
and the copies of $\Omega$ not yet in use. Every copy of $\Omega$, including the
main copy at the root, is initialized privately at its future owner, and the
reference vector at every stage is the tensor product of the reference vectors
of the encoded frames of the sheets, every factor of norm at most one
(\texttt{06-geometry.tex}, lines 653--681). This section collects the
bookkeeping of this memory: the initial private maps, the reorderings of the
sheets, the changes that leave a frame unchanged, and the norm of the reference
vector. The initialization assumes that the product of the reference vectors of the
frames of each party has norm one; for the raw frames of the initial memory, which have
no holes, this is the normalization of $\Omega$.

\begin{definition}[The memory of several sheets]
    \label{def:peps_sheet_stack}
    \lean{TNLean.PEPS.EncodedFrame.stackRegs,
        TNLean.PEPS.EncodedFrame.SheetStack,
        TNLean.PEPS.EncodedFrame.Frame.regRef,
        TNLean.PEPS.EncodedFrame.stackVec,
        TNLean.PEPS.EncodedFrame.framesVec}
    \leanok
    \uses{def:peps_frame_registers}
    For a list of encoded frames $\mathcal F_1,\dots,\mathcal F_m$ and a list $\ell$
    of further registers, the memory of the sheets is the tensor product of the
    registers of the frames, in order, followed by those of $\ell$. For a vector
    $\xi$ of the registers of $\ell$, the reference vector of the sheets is
    $\Omega_{\mathcal F_1}\otimes\cdots\otimes\Omega_{\mathcal F_m}\otimes\xi$.
\end{definition}

\begin{theorem}[Reordering the sheets]
    \label{thm:peps_sheet_reordering}
    \lean{TNLean.PEPS.EncodedFrame.stackRegs_append,
        TNLean.PEPS.EncodedFrame.stackVec_append,
        TNLean.PEPS.EncodedFrame.exists_reordering_stackVec,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_stackVec_of_perm}
    \leanok
    \uses{def:peps_sheet_stack}
    Let $\mathcal F'_1,\dots,\mathcal F'_m$ be a permutation of the frames
    $\mathcal F_1,\dots,\mathcal F_m$. There is a word of exchanges of tensor
    factors from the memory of the first list, followed by $\ell$, to that of the
    second list, followed by $\ell$, which carries the reference vector of the first
    list to that of the second for every vector $\xi$ of $\ell$. As an original
    circuit it has no nonprivate gate. In particular an auxiliary sheet can be moved
    next to the main sheet before a lens exchange.
\end{theorem}

\begin{proof}\leanok
    Induct on the permutation. Prepending a common frame frames the word by its
    registers, and two adjacent frames are interchanged by exchanging their two
    blocks of registers.
\end{proof}

\begin{theorem}[The initial private maps]
    \label{thm:peps_sheet_initialization}
    \lean{TNLean.PEPS.EncodedFrame.framesSpace,
        TNLean.PEPS.EncodedFrame.splitMap,
        TNLean.PEPS.EncodedFrame.splitMap_tmul_one,
        TNLean.PEPS.EncodedFrame.norm_splitMap_le,
        TNLean.PEPS.EncodedFrame.splitCircuit,
        TNLean.PEPS.EncodedFrame.nonprivateCount_splitCircuit,
        TNLean.PEPS.EncodedFrame.isExpansionBounded_splitCircuit,
        TNLean.PEPS.EncodedFrame.isMonomialBounded_splitCircuit,
        TNLean.PEPS.EncodedFrame.participants_splitCircuit_subset,
        TNLean.PEPS.EncodedFrame.eval_splitCircuit,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_initial}
    \leanok
    \uses{def:peps_sheet_stack}
    Let every frame of a list be held entirely by one party: all its registers,
    raw and tag, belong to that party. Give each party, as its whole initial private
    memory, the registers of its frames in the order of the list, and as its
    initial vector the product of their reference vectors, assumed of norm one.
    Then one private map per party, followed by a reordering of the frames, carries
    the product of the initial vectors of all parties exactly to the reference
    vector of the list. The resulting circuit has no nonprivate gate. This is the
    pure product initialization of \texttt{06-geometry.tex}, lines 658--662.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_reordering}
    Each private map identifies the whole memory of a party with the memory of its
    frames. The concatenation, over the parties, of their lists of frames is a
    permutation of the list: a frame occurs in the list of its party exactly as
    often as in the list. Theorem~\ref{thm:peps_sheet_reordering} reorders the
    frames.
\end{proof}

\begin{theorem}[The initial step from raw copies]
    \label{thm:peps_sheet_raw_initialization}
    \lean{TNLean.PEPS.EncodedFrame.Frame.norm_refVec_of_holes_eq_nil,
        TNLean.PEPS.EncodedFrame.norm_framesVec_of_holes_eq_nil,
        TNLean.PEPS.EncodedFrame.exists_originalCircuit_initial_of_holes_eq_nil}
    \leanok
    \uses{thm:peps_sheet_initialization}
    A frame without holes has reference vector of norm $\lVert\Omega\rVert$. Let
    $\lVert\Omega\rVert=1$ and let every frame of the list have no hole and be held
    entirely by one party: the main copy at the root and the auxiliary copies at
    their future owners. Then the product of the reference vectors of the frames of
    every party has norm one, and the initial private maps of
    Theorem~\ref{thm:peps_sheet_initialization} apply (\texttt{06-geometry.tex},
    lines 658--667).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_initialization, thm:peps_frame_norm}
    Without holes the product of the hole projectors is the identity, so
    $\lVert\Omega_{\mathcal F}\rVert=\lVert\Omega\rVert=1$, and the norm of a product
    of reference vectors is the product of their norms.
\end{proof}

\begin{theorem}[Changes that leave a frame unchanged]
    \label{thm:peps_identity_change}
    \lean{TNLean.PEPS.EncodedFrame.exists_originalCircuit_of_frame_eq,
        TNLean.PEPS.EncodedFrame.Frame.changeOwner_empty,
        TNLean.PEPS.Approximation.emptyTarget_step}
    \leanok
    \uses{def:peps_sheet_stack}
    A change between equal frames, such as a birth or death whose raw target is
    empty or the resize of a hole whose radius does not change, is the circuit
    without operations: it has no nonprivate gate and carries the reference vector
    of the frame to itself. Changing the owner of an empty set of raw registers
    leaves a frame unchanged (\texttt{06-geometry.tex}, lines 432--438 and
    529--541).
\end{theorem}

\begin{proof}\leanok
    Immediate.
\end{proof}

\begin{theorem}[The norm of the reference vector of several sheets]
    \label{thm:peps_sheet_stack_norm}
    \lean{TNLean.PEPS.EncodedFrame.one_sub_mul_le_norm_stackVec,
        TNLean.PEPS.EncodedFrame.one_sub_div_pow_le_norm_stackVec}
    \leanok
    \uses{def:peps_sheet_stack,thm:peps_frame_norm}
    Let $\lVert\Omega\rVert=1$, and let every hole of every frame
    $\mathcal F_1,\dots,\mathcal F_m$ satisfy
    $\lVert(\id-P)\Omega\rVert\le\varepsilon$. If the frames have $r$ holes in
    total, then
    \begin{align*}
        (1-r\varepsilon)\lVert\xi\rVert
         & \le\lVert\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi\rVert .
    \end{align*}
    In particular, if $\varepsilon=L^{-60}$, $r\le CL^3$ and
    $\lVert\xi\rVert=1$, the reference vector has norm between $1-CL^{-57}$ and
    $1$. The bound $r\le CL^3$ on the total number of holes is a hypothesis here;
    in the source it is the count of frozen auxiliary holes,
    \texttt{06-geometry.tex}, lines 653--656 and 674--678.
\end{theorem}

\begin{proof}\leanok
    By Theorem~\ref{thm:peps_frame_norm}, the frame $\mathcal F_i$ with $r_i$ holes
    has $1-r_i\varepsilon\le\lVert\Omega_{\mathcal F_i}\rVert\le1$. For
    $1-a\le x\le1$ and $(1-b)z\le y\le z$ one has
    $xy=y-(1-x)y\ge y-az\ge(1-a-b)z$; induct over the frames.
\end{proof}

\begin{definition}[The final regrouping]
    \label{def:peps_final_regrouping}
    \lean{TNLean.PEPS.EncodedFrame.trivialTag,
        TNLean.PEPS.EncodedFrame.physVec,
        TNLean.PEPS.EncodedFrame.freshMap,
        TNLean.PEPS.EncodedFrame.freshMap_apply,

        TNLean.PEPS.EncodedFrame.renameMap,
        TNLean.PEPS.EncodedFrame.renameMap_tmul,
        TNLean.PEPS.EncodedFrame.norm_renameMap_le,
        TNLean.PEPS.EncodedFrame.finOneOf,
        TNLean.PEPS.EncodedFrame.nonprivateCount_freshCircuit,
        TNLean.PEPS.EncodedFrame.nonprivateCount_siteCircuit,
        TNLean.PEPS.EncodedFrame.mem_nonsiteParties,
        TNLean.PEPS.EncodedFrame.nonsiteParties_dim,
        TNLean.PEPS.EncodedFrame.freshCircuit,
        TNLean.PEPS.EncodedFrame.siteCircuit,
        TNLean.PEPS.EncodedFrame.nonsiteParties,
        TNLean.PEPS.EncodedFrame.siteOrder,
        TNLean.PEPS.EncodedFrame.readoutPrivate,
        TNLean.PEPS.EncodedFrame.readoutCircuit}
    \leanok
    \uses{def:peps_sheet_stack}
    Let the main frame $\mathcal M$ be the first sheet of the memory. The final regrouping moves the raw registers of
    $\mathcal M$ in front of its tag registers, renames the raw register of each
    site, by a private map at its party, as the physical register of that party,
    and lets every party without a site prepare a physical register of dimension
    one by a private map. The physical registers are listed in the order of the
    sites, followed by the parties without a site; the private registers are the
    tags of $\mathcal M$ followed by the registers of the other sheets and of the
    idle copies (\texttt{06-geometry.tex}, lines 48--51 and 669--681).
\end{definition}

\begin{theorem}[The final reference vector]
    \label{thm:peps_final_reference}
    \lean{TNLean.PEPS.PairEffect.OriginalCircuit.isEmpty_nonprivateLocations_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.participants_subset_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.isExpansionBounded_of_nonprivateCount_eq_zero,
        TNLean.PEPS.PairEffect.OriginalCircuit.isMonomialBounded_of_nonprivateCount_eq_zero,
        TNLean.PEPS.EncodedFrame.eq_trivialTag,
        TNLean.PEPS.EncodedFrame.rawProd_trivialTag,
        TNLean.PEPS.EncodedFrame.Frame.regRef_of_outer_eq_empty,
        TNLean.PEPS.EncodedFrame.familyPhysicalListBasis_eq_physVec,
        TNLean.PEPS.EncodedFrame.familyLabelledPhysicalBasis_eq_physVec,
        TNLean.PEPS.EncodedFrame.eval_freshCircuit,
        TNLean.PEPS.EncodedFrame.eval_siteCircuit,
        TNLean.PEPS.EncodedFrame.nodup_siteOrder,
        TNLean.PEPS.EncodedFrame.mem_siteOrder,
        TNLean.PEPS.EncodedFrame.nonprivateCount_readoutCircuit,
        TNLean.PEPS.EncodedFrame.eval_readoutCircuit_stackVec,
        TNLean.PEPS.EncodedFrame.readout_eval_readoutCircuit_stackVec,
        TNLean.PEPS.EncodedFrame.exists_readoutCircuit_norm,
        TNLean.PEPS.Approximation.DyadicProtocol.physicalConfigEquiv_apply_val,
        TNLean.PEPS.Approximation.DyadicProtocol.readout_eq_euclideanTensorVector}
    \leanok
    \uses{def:peps_final_regrouping, def:peps_dyadic_protocol}
    The final regrouping has no nonprivate gate. Let the raw owner of the main frame $\mathcal M$ be
    injective, and let every hole of the main frame
    $\mathcal M$ have empty physical outer sample, let the physical dimension be
    $q$ at the party of every site and one at every other party, and read the
    physical configurations of the parties as configurations of the sites. Then,
    for every $\Omega$ and every vector $\xi$ of the idle registers, the
    physical-private coordinates of the regrouped reference vector
    $\Omega_{\mathcal M}\otimes\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi$ are those of $\Omega\otimes\Xi$, where $\Xi$
    is the vector of the private registers and
    $\lVert\Xi\rVert=\lVert\Omega_{\mathcal F_1}\otimes\cdots\otimes
        \Omega_{\mathcal F_m}\otimes\xi\rVert$. If moreover
    $\lVert\Omega\rVert=\lVert\xi\rVert=1$, every hole of the other frames
    satisfies $\lVert(\id-P)\Omega\rVert\le L^{-60}$ and the other frames have at
    most $CL^3$ holes in total, then $1-CL^{-57}\le\lVert\Xi\rVert\le1$. The bound
    on the number of holes is a hypothesis here. For a protocol whose final layout
    is the output of the regrouping, the readout of the protocol is then
    $\Omega\otimes\Xi$ (\texttt{06-geometry.tex}, lines 43--51 and 669--681).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_sheet_stack_norm}
    A hole with empty outer sample has only the identity tag, with identity raw
    part, so $\Omega_{\mathcal M}=\sum_c\Omega(c)\,\lvert\tau_0\rangle\otimes
        \lvert c\rangle$. On a basis vector $\lvert c\rangle$ of the raw registers, the
    exchanges move $\lvert c\rangle$ to the front, the renaming sends it to the
    physical basis vector of $c$, and the fresh registers add the basis vector of
    the one-point coordinate. By linearity, the regrouped vector is
    $\sum_c\Omega(c)\,\lvert c\rangle\otimes\pi$, with $\pi$ the product of the tags
    and the other factors, and its coordinates are $\Omega(c)$ times those of
    $\pi$. The norm bound is Theorem~\ref{thm:peps_sheet_stack_norm}.
\end{proof}
