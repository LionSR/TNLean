\begin{theorem}[Regular closures and local parent conditions]%
    \label{thm:peps_regular_closure_parent_constraint}
    \lean{TNLean.PEPS.IsGInjective.commute_of_cornerParent_annihilates_torusGClosure,
        TNLean.PEPS.IsGInjective.regionParent_annihilates_torusGClosure_of_isSimplyConnected,
        TNLean.PEPS.IsGInjective.regionParentHamiltonian_annihilates_torusGClosure,
        TNLean.PEPS.IsGInjective.span_torusGClosure_commuting_le_parentKernel,
        TNLean.PEPS.IsGInjective.card_commutingPairConjugacyClass_le_finrank_parentKernel}
    \leanok
    \uses{thm:peps_regular_torus_parent_flatness,
        thm:peps_regular_region_parent_flatness,
        def:peps_region_parent_interaction, thm:peps_pair_centralizer_count}
    On a square torus $\Z_{L_x}\times\Z_{L_y}$ with $L_x,L_y\ge 3$ and a
    native regular $G$-injective tensor, let $\Psi_{g,h}$ be the closure
    vector with seam labels $g,h\in G$. If the original parent interaction
    $h_{P_0}^{\mathrm{global}}$ on the seam-intersection plaquette $P_0$
    satisfies $h_{P_0}^{\mathrm{global}}\Psi_{g,h}=0$, then $gh=hg$.
    Conversely, if $gh=hg$, then $h_R^{\mathrm{global}}\Psi_{g,h}=0$
    (Definition~\ref{def:peps_region_parent_interaction}) for every connected
    region $R$ whose closed-cell realization is simply connected, even when
    $R$ meets either seam.

    For any finite family $\mathcal R$ of such regions and any parent
    Hamiltonian $H=\sum_{R\in\mathcal R}h_R^{\mathrm{global}}$ with
    $h_R\ge 0$ and $\ker h_R=\mathcal G_R$, one has
    $\spn\{\Psi_{g,h}:gh=hg\}\subseteq\ker H$ and
    \begin{align}
        \dim\ker H &\ge
        \sum_{C\in\operatorname{Conj}(G)}
        \left|\operatorname{Conj}\bigl(Z_G(c_C)\bigr)\right|,
        \label{eq:peps_closure_parent_dim_bound}
    \end{align}
    where $c_C\in C$ is a class representative. No $G$-isometry is required.
    This proves the inclusion and the resulting lower bound for
    the regular case of \cite[Theorems~5.5 and~5.9]{Schuch2010PEPS}.
    The assertion that every parent-kernel vector lies in the
    closure span remains separate.
    The regular native-torus restriction and the missing reverse global spanning inclusion are recorded in
    \cite{gap:rmp_peps_quantum_double_g_isometry}.
\end{theorem}

\begin{proof}\leanok
    Regular $G$-injectivity implies $\Psi_{g,h}\ne 0$. By
    Theorem~\ref{thm:peps_regular_torus_parent_flatness},
    $h_{P_0}^{\mathrm{global}}\Psi_{g,h}=0$ forces
    $\operatorname{hol}_{g,h}(P_0)=h^{-1}g^{-1}hg=1$, i.e., $gh=hg$.

    Conversely, a simply connected closed-cell region $R$ admits an integer
    coordinate lift; when $gh=hg$, this lift defines a vertex gauge on $R$
    removing all internal seam insertions. By
    Theorem~\ref{thm:peps_regular_region_parent_flatness}, the inserted and
    original regional spaces satisfy $\mathcal G_R^{g,h}=\mathcal G_R$, so
    every slice of $\Psi_{g,h}$ lies in $\mathcal G_R=\ker h_R$ and
    $h_R^{\mathrm{global}}\Psi_{g,h}=0$. Since $h_R\ge 0$,
    $\ker H=\bigcap_{R\in\mathcal R}\ker h_R^{\mathrm{global}}\supseteq\spn\{\Psi_{g,h}:gh=hg\}$.
    Linear independence of the commuting pair-conjugacy classes and
    Theorem~\ref{thm:peps_pair_centralizer_count} then yield~(\ref{eq:peps_closure_parent_dim_bound}).
\end{proof}
