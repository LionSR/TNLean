\section{Weighted dyadic covering of rectangle shells}

\begin{theorem}[Summation of selected dyadic squares by scale]
    \label{thm:area_law_capped_sum_by_scale}
    \lean{TNLean.PEPS.AreaLaw.Geometry.sum_cappedDyadicPartition_by_scale}
    \leanok
    \uses{def:area_law_capped_dyadic_partition}
    Let $S\subseteq\mathbb Z^2$ be finite, let $K\in\mathbb N$, and let
    $f:\mathbb N\to\mathbb R$ be arbitrary. Then
    \begin{align}
        \sum_{(r,z)\in\mathcal P_K(S)} f(r)
         & =\sum_{k=0}^{K}
        \#\{(r,z)\in\mathcal P_K(S):r=k\}\,f(k).
    \end{align}
    This is the finite summation by scale used in the covering argument of
    \cite[Lemma~9.4, lines~656--664]{OpenAI2026AreaLaw}.
    The identity includes $S=\varnothing$ and $K=0$ and imposes no sign
    condition on $f$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_capped_dyadic_partition}
    Membership in $\mathcal P_K(S)$ implies $r\le K$. Group its finitely
    many terms according to $r$. On each group with $r=k$, the summand is
    the constant $f(k)$, so
    \begin{align}
        \sum_{\substack{(r,z)\in\mathcal P_K(S)      \\r=k}}f(k)
         & =\#\{(r,z)\in\mathcal P_K(S):r=k\}\,f(k).
    \end{align}
    Summing over $k=0,\ldots,K$ gives the identity.
\end{proof}

% Finite capped-partition moment estimate.
\begin{theorem}[A dyadic moment bound from scale counts]
    \label{thm:area_law_capped_dyadic_moment}
    \lean{TNLean.PEPS.AreaLaw.Geometry.sum_rpow_cappedDyadicPartition_le}
    \leanok
    \uses{def:area_law_capped_dyadic_partition}
    Let $S\subseteq\mathbb Z^2$ be finite and let $K,L\in\mathbb N$
    satisfy $2^K\le L$. Let $e,A,B\in\mathbb R$ with $e>0$ and $B\ge0$.
    Write $N_k=\#\{(r,z)\in\mathcal P_K(S):r=k\}$.
    Suppose $2^K N_K\le A$ and $2^k N_k\le B$ for every $0\le k<K$.
    Then
    \begin{align}
        \sum_{(k,z)\in\mathcal P_K(S)}(2^k)^{1+e}
         & \le\left(A+\frac{B}{2^e-1}\right)L^e.
    \end{align}
    This is the finite moment estimate used in the covering argument of
    \cite[Lemma~9.4, lines~656--664]{OpenAI2026AreaLaw}.
    It includes $S=\varnothing$ and $K=0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_capped_sum_by_scale,thm:area_law_weighted_dyadic_sum}
    Group the selected squares by scale and apply the finite weighted
    dyadic-sum estimate:
    \begin{align}
        \sum_{(k,z)\in\mathcal P_K(S)}(2^k)^{1+e}
         & =\sum_{k=0}^{K}N_k(2^k)^{1+e} \\
         & \le\left(A+\frac{B}{2^e-1}\right)(2^K)^e
         \le\left(A+\frac{B}{2^e-1}\right)L^e.
    \end{align}
    The cap bound gives $A\ge0$, while $2^e-1>0$. Hence the coefficient
    is nonnegative, and the last inequality follows from $e>0$ and $2^K\le L$.
\end{proof}

\begin{theorem}[Weighted dyadic covering of a rectangle shell]
    \label{thm:area_law_rectangle_shell_weighted_cover}
    \lean{TNLean.PEPS.AreaLaw.IntRect.sum_rpow_cappedDyadicPartition_shell_le}
    \leanok
    \uses{def:area_law_safe_rectangle,def:area_law_capped_dyadic_partition}
    Let $Q$ be a nonempty integer rectangle and put $s=\operatorname{size}(Q)$. Let $j,L,K\in\N$
    satisfy $j\le L\le s$ and $2^K\le L<2^{K+1}$, and let $e>0$.
    Set $S_j=Q^{+j}\setminus Q$, with both rectangles viewed as sets of
    lattice sites. Then the capped maximal-contained dyadic partition satisfies
    \begin{align}
        \sum_{(k,z)\in\mathcal P_K(S_j)}(2^k)^{1+e}
         & \le \left(24+\frac{64}{2^e-1}\right)sL^e.
    \end{align}
    This is the geometric weighted covering estimate used in
    \cite[proof of Proposition~9.5, lines~717--729]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_rectangle_shell_cap_count,
        thm:area_law_rectangle_shell_selected_count,
        thm:area_law_capped_dyadic_moment}
    Let $N_k$ denote the number of selected squares of scale $k$.
    The rectangle-shell counting estimates give the following bounds.
    The second holds for every $k$ with $0\le k<K$.
    \begin{align}
        2^K N_K & \le24s, \\
        2^k N_k & \le64s.
    \end{align}
    For the second estimate, the required bounds are
    $j\le L\le s$ and $2^{k+1}\le2^K\le L\le s$.

    Applying Theorem~\ref{thm:area_law_capped_dyadic_moment} with cap
    budget $24s$ and lower-scale budget $64s$ gives the stated bound.
    Its proof groups terms by scale and estimates the geometric series as follows:
    \begin{align}
        \sum_{(k,z)\in\mathcal P_K(S_j)}(2^k)^{1+e}
         & =\sum_{k=0}^{K}N_k(2^k)^{1+e}                \\
         & \le\left(24s+\frac{64s}{2^e-1}\right)(2^K)^e \\
         & \le\left(24+\frac{64}{2^e-1}\right)sL^e.
    \end{align}
    The last inequality uses $e>0$ and $2^K\le L$; its coefficient is
    nonnegative because $2^e-1>0$.
    If $j=0$, the shell and its partition are empty. If $K=0$, the
    lower-scale sum is empty and the cap inequalities force $L=1$.
    The argument also applies to thin rectangles and negative coordinates.
\end{proof}
