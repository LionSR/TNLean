\subsection{The four-block closure property}
\label{sec:peps_full_four_cut_closure}

The four blocks in \cite[Theorem~5.5]{Schuch2010PEPS} occupy
$\Lambda=(\mathbb Z/2\mathbb Z)^2$ with eight labelled bonds (two between
each adjacent pair of blocks). Each bond $e$ carries a finite virtual
alphabet $X_e$ and semi-regular representation $U_e$ (acting by $U_e$ at
the head and $U_e^{-\mathsf T}$ at the tail), and each block $v$ carries a
finite physical alphabet $Y_v$.
% Exact source: Papers/1001.3807/paper_v3.tex:1424--1513,
% thm:2d:closure and eq:2d:closure-intersection. Link matching is
% paper_v3.tex:1310--1316. No common virtual or physical dimension is imposed.

\begin{definition}[Four cut spaces and commuting closures]%
    \label{def:peps_full_four_cut_space}
    \lean{TNLean.PEPS.DependentTorus.Vertex,
        TNLean.PEPS.DependentTorus.Bond,
        TNLean.PEPS.DependentTorus.LocalConfig,
        TNLean.PEPS.DependentTorus.cutBonds,
        TNLean.PEPS.DependentTorus.fourCutSpace,
        TNLean.PEPS.DependentTorus.mem_fourCutSpace_iff,
        TNLean.PEPS.DependentTorus.cutBonds_cover,
        TNLean.PEPS.DependentTorus.closure,
        TNLean.PEPS.DependentTorus.commutingClosureSpan}
    \leanok
    \uses{def:peps_labelled_bonds, def:peps_labelled_cut,
        def:peps_labelled_closure, def:peps_dependent_cut}
    Let $C_{c,r}$ be the cut of horizontal bonds with head column $c$ and
    vertical bonds with tail row $r$, with eight-incidence virtual domain
    $\mathcal X_{\partial C_{c,r}}$. For $p^{g,h}$ equal to $h$ on the
    horizontal seam, $g$ on the vertical seam, and $1$ elsewhere, define
    \begin{align}
        \mathcal I_A & =\bigcap_{c,r\in\mathbb Z/2\mathbb Z}
        \operatorname{im}\mathcal C_{A,C_{c,r}},
        \label{eq:peps_full_four_cut_space}                  \\
        \zeta_A(g,h) & =\mathcal N_A((U_e(p^{g,h}_e))_e),
        \label{eq:peps_full_four_cut_closure}                \\
        \mathcal Z_A & =\spn_{\C}\{\zeta_A(g,h):gh=hg\}.
        \label{eq:peps_full_four_cut_closure_span}
    \end{align}
    Every bond is uncut in the cut $C_{c,r}$ of its base vertex.
\end{definition}

\begin{theorem}[Four-block closure for independently sized $G$-injective tensors]%
    \label{thm:peps_four_cut_closure}
    \lean{TNLean.PEPS.DependentTorus.fourCutSpace_eq_commutingClosureSpan}
    \leanok
    \uses{def:peps_full_four_cut_space, def:peps_g_injective,
        def:peps_dependent_incident_action, def:peps_semiregular}
    For finite $G$, semi-regular bond representations $U_e$, and
    $G$-injective site tensors $A_v$,
    \begin{align}
        \bigcap_{c,r\in\mathbb Z/2\mathbb Z}
        \{\mathcal C_{A,C_{c,r}}(M):M\in\C^{\mathcal X_{\partial C_{c,r}}}\}
        =\left\{\sum_{gh=hg}\lambda_{g,h}\zeta_A(g,h):
        \lambda_{g,h}\in\C\right\}.
        \label{eq:peps_full_four_cut_equality}
    \end{align}
    In this form of \cite[Theorem~5.5]{Schuch2010PEPS}, the bond and physical
    dimensions may differ independently, each boundary $M$ is an arbitrary
    tensor on all eight open incidences, and algebraic semi-regularity suffices.
    % Source ownership: thm:2d:closure, eq:2d:closure-intersection.
    % Proof source labels: eq:2d:closure-inv, eq:2d:close-in-in.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_full_four_cut_transport,
        thm:peps_full_four_cut_canonical_equality}
    For the canonical averaging tensors $A^0$ and the product
    $\mathcal T=\bigotimes_vT_v$ of the local coefficient maps,
    Lemma~\ref{thm:peps_full_four_cut_transport} gives
    $\mathcal I_A=\mathcal T(\mathcal I_{A^0})$ and
    $\mathcal Z_A=\mathcal T(\mathcal Z_{A^0})$, while
    Lemma~\ref{thm:peps_full_four_cut_canonical_equality} gives
    $\mathcal I_{A^0}=\mathcal Z_{A^0}$.
\end{proof}

\begin{lemma}[Common physical image of cuts and closures]%
    \label{thm:peps_full_four_cut_transport}
    \lean{TNLean.PEPS.DependentTorus.physicalMap_closure,
        TNLean.PEPS.DependentTorus.map_commutingClosureSpan,
        TNLean.PEPS.DependentTorus.fourCutSpace_eq_map_averagingSite,
        TNLean.PEPS.DependentTorus.map_commutingClosureSpan_recover}
    \leanok
    \uses{def:peps_full_four_cut_space, def:peps_dependent_physical_site}
    For local physical maps $F_v$ and $\mathcal F=\bigotimes_vF_v$,
    $\mathcal F\zeta_A(g,h)=\zeta_{FA}(g,h)$ and
    $\mathcal F(\mathcal Z_A)=\mathcal Z_{FA}$.
    For the canonical averaging tensors $A^0$ and $\mathcal T=\bigotimes_vT_v$,
    invariance of $A$ gives $\mathcal T(\mathcal Z_{A^0})=\mathcal Z_A$,
    and $G$-injectivity of $A$ gives
    $\mathcal I_A=\mathcal T(\mathcal I_{A^0})$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dependent_physical_contraction,
        thm:peps_dependent_common_inverse}
    Commuting $\mathcal F$ through (\ref{eq:peps_full_four_cut_closure}) and
    taking spans gives the first two identities; $T_vP_v=T_v$ then gives
    $\mathcal T(\mathcal Z_{A^0})=\mathcal Z_A$. Applying
    (\ref{eq:peps_dependent_cut_transport}) across the four cuts gives
    $\mathcal I_A=\mathcal T(\mathcal I_{A^0})$.
\end{proof}

\begin{definition}[Canonical four-block tensors and projection]%
    \label{def:peps_full_four_cut_canonical}
    \lean{TNLean.PEPS.DependentTorus.canonicalSites,
        TNLean.PEPS.DependentTorus.canonicalProjector}
    \leanok
    \uses{def:peps_dependent_averaging_site, def:peps_dependent_average_projection}
    On $\Lambda=(\mathbb Z/2\mathbb Z)^2$, the canonical tensors are
    $A_v^0(\eta,s)=(P_v)_{s,\eta}$ with physical alphabet $\mathcal X_v$,
    and their product projection is $\mathcal P=\bigotimes_{v\in\Lambda}P_v$.
\end{definition}

\begin{lemma}[Moving a commuting closure to every cut]%
    \label{thm:peps_full_four_cut_forward}
    \lean{TNLean.PEPS.DependentTorus.canonicalClosure_mem_cutSpace,
        TNLean.PEPS.DependentTorus.commutingClosureSpan_le_fourCutSpace_canonical}
    \leanok
    \uses{def:peps_full_four_cut_space, def:peps_full_four_cut_canonical}
    Without assuming semi-regularity, $gh=hg$ implies
    $\zeta_{A^0}(g,h)\in\mathcal S_{A^0,C_{c,r}}$ for every $(c,r)$, and hence
    $\mathcal Z_{A^0}\subseteq\mathcal I_{A^0}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_labelled_closure_gauge, thm:peps_labelled_closure_support,
        thm:peps_dependent_vertex_gauge, thm:peps_dependent_cut_basis}
    By Lemmas~\ref{thm:peps_labelled_closure_gauge}
    and~\ref{thm:peps_labelled_closure_support}, a vertex gauge shifts the
    seams to labels $p^{g,h;c,r}$ equal to $1$ off $C_{c,r}$. By
    Lemmas~\ref{thm:peps_dependent_vertex_gauge}
    and~\ref{thm:peps_dependent_cut_basis}, $\mathcal C_{A^0,C_{c,r}}(M)=\zeta_{A^0}(g,h)$
    for $M(\eta)=\prod_{e\in C_{c,r}}U_e(p^{g,h;c,r}_e)_{\eta_{e,+},\eta_{e,-}}$;
    spanning and intersecting over $(c,r)$ gives
    $\mathcal Z_{A^0}\subseteq\mathcal I_{A^0}$.
\end{proof}

\begin{lemma}[Projected flat products are commuting closures]%
    \label{thm:peps_full_four_cut_flat_projection}
    \lean{TNLean.PEPS.DependentTorus.canonicalProjector_bondProduct,
        TNLean.PEPS.DependentTorus.exists_canonicalProjector_bondProduct_eq_closure,
        TNLean.PEPS.DependentTorus.canonicalProjector_bondProduct_mem_closureSpan}
    \leanok
    \uses{def:peps_full_four_cut_canonical, def:peps_dependent_bond_product,
        def:peps_bond_flat}
    For every bond assignment $p$,
    $\mathcal P b_p=\mathcal N_{A^0}((U_e(p_e))_e)$.
    If $p$ is flat, then $\mathcal P b_p=\zeta_{A^0}(g,h)\in\mathcal Z_{A^0}$
    for some commuting pair $(g,h)$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dependent_average_projection,
        thm:peps_labelled_flat_classification, thm:peps_dependent_vertex_gauge}
    Lemma~\ref{thm:peps_dependent_average_projection} gives
    $\mathcal P b_p=\mathcal N_{A^0}((U_e(p_e))_e)$. For flat $p$,
    Lemma~\ref{thm:peps_labelled_flat_classification} provides a vertex
    gauge to commuting seams $(g,h)$, and
    Lemma~\ref{thm:peps_dependent_vertex_gauge} gives
    $\mathcal P b_p=\zeta_{A^0}(g,h)$.
\end{proof}

\begin{lemma}[Nonflat coefficients vanish in the four-cut intersection]%
    \label{thm:peps_full_four_cut_nonflat_zero}
    \lean{TNLean.PEPS.DependentTorus.coefficient_eq_zero_of_mem_fourCutSpace_not_flat}
    \leanok
    \uses{def:peps_full_four_cut_space, def:peps_dependent_bond_coefficient,
        def:peps_bond_flat}
    If every $U_e$ is semi-regular, $\psi\in\mathcal I_{A^0}$, and $p$ is
    not flat, then $c_p(\psi)=0$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dependent_uncut_coefficients, thm:peps_labelled_uncut_flat}
    If $c_p(\psi)\ne0$, then on the cut of a plaquette where $p$ is not flat,
    (\ref{eq:peps_dependent_uncut_coefficients}) and
    Lemma~\ref{thm:peps_labelled_uncut_flat} force $p$ to satisfy that
    plaquette equation, a contradiction.
\end{proof}

\begin{lemma}[Canonical four-cut equality]%
    \label{thm:peps_full_four_cut_canonical_equality}
    \lean{TNLean.PEPS.DependentTorus.fourCutSpace_canonical_le_commutingClosureSpan,
        TNLean.PEPS.DependentTorus.fourCutSpace_canonical_eq_commutingClosureSpan}
    \leanok
    \uses{def:peps_full_four_cut_space, def:peps_full_four_cut_canonical}
    If all eight bond representations are semi-regular, then
    $\mathcal I_{A^0}\subseteq\mathcal Z_{A^0}$ and
    $\mathcal I_{A^0}=\mathcal Z_{A^0}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_dependent_average_projection, thm:peps_dependent_cut_expansion,
        thm:peps_full_four_cut_nonflat_zero, thm:peps_full_four_cut_flat_projection,
        thm:peps_full_four_cut_forward}
    For $\psi\in\mathcal I_{A^0}$, membership in $\mathcal S_{A^0,C_{0,0}}$
    gives $\mathcal P\psi=\psi$ by
    Lemma~\ref{thm:peps_dependent_average_projection}. Since the four cuts
    leave every bond uncut at least once, (\ref{eq:peps_dependent_extracted_expansion}),
    Lemma~\ref{thm:peps_full_four_cut_nonflat_zero}, and
    Lemma~\ref{thm:peps_full_four_cut_flat_projection} give
    \begin{align}
        \psi=\mathcal P\psi=\sum_p c_p(\psi)\mathcal P b_p
        =\sum_{p\ \mathrm{flat}}c_p(\psi)\mathcal P b_p
        \in\mathcal Z_{A^0},
        \label{eq:peps_full_four_cut_reconstruction}
    \end{align}
    and Lemma~\ref{thm:peps_full_four_cut_forward} gives the reverse inclusion.
\end{proof}
