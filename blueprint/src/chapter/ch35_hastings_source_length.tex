\section{An admissible length and the printed entropy bound}
\label{sec:hastings_source_length}

The preceding factorization and entropy estimates use a length
$\zeta=20(1+c_RJ/\Delta)$, where
$c_R=2(R+1)2^{2R+1}e^{2R}$. At that fixed length, the corrections in
\cite{gap:hastings07_gaussian_average_exponent,gap:hastings07_doubling_constants}
remain necessary. The Lieb--Robinson parameters in
\cite[lines 96--112, 139--142]{Hastings2007AreaLaw} are chosen existentially.
A sufficiently large admissible choice absorbs those losses. Throughout this
section $J\ge0$, $\Delta>0$, and the length convention is
$e^{-d/\xi_C}$ in the cone $v|t|\le d$, as explained in
\cite{gap:hastings07_lemma1_decay_length}.

\begin{definition}[A physical velocity and its length]
    \label{def:hastings_source_length}
    \lean{TNLean.MPS.AreaLaw.hastingsLRVelocity,
        TNLean.MPS.AreaLaw.hastingsLRLength}
    \leanok
    For a range $R$ and interaction bound $J$, put
    \begin{align*}
        V=40c_RJ,\qquad \Xi=\max\{2V/\Delta,80\}.
    \end{align*}
    The velocity is independent of the gap, chain length, and local dimension.
\end{definition}

\begin{lemma}[Numerical Lieb--Robinson estimate]
    \label{lem:hastings_source_lieb_robinson}
    \lean{TNLean.MPS.AreaLaw.ChainHamiltonian.norm_heisenberg_commutator_le_numeric,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_heisenberg_commutator_le_hastings,
        TNLean.MPS.AreaLaw.ChainHamiltonian.norm_heisenberg_commutator_le_hastings_cone,
        TNLean.MPS.AreaLaw.chainLRVelocity_sub_one_pos}
    \leanok
    \uses{def:chain_model,def:hastings_source_length}
    Let $A$ and $B$ be supported on arbitrary finite sets $X$ and $Y$ of the
    chain, and let $d$ be a nonnegative integer lower bound for their distance.
    Then
    \begin{align*}
        \|[A(t),B]\|\le4|X|\|A\|\|B\|
            e^{2c_RJ|t|-d}.
    \end{align*}
    In particular, if $V|t|\le d$, then
    $\|[A(t),B]\|\le4|X|\|A\|\|B\|e^{-d/80}$.
    More generally, any $v>0$ with $v\ge8c_RJ$ and any $\xi_C\ge8$ give
    $\|[A(t),B]\|\le4|X|\|A\|\|B\|e^{-d/\xi_C}$ when $|t|\le d/v$.
    For $J>0$, the pair $(V,80)$ is therefore admissible in the sense of
    \cite[eq. (lrb)]{Hastings2007AreaLaw}. For $J=0$, the multiplication
    form of the cone remains meaningful.
\end{lemma}
\begin{proof}\leanok
    Apply the graph Lieb--Robinson estimate to the path graph. Every
    interaction has diameter at most $R$ and at most $R+1$ sites, and the
    sum of norms of terms containing any site is at most $2^{2R+1}J$.
    This gives the first estimate without a factor depending on $|Y|$.
    In the chosen cone, $2c_RJ|t|\le d/20$, so the exponent is at most
    $-19d/20\le-d/80$. In the general cone it is at most $-3d/4$,
    which is at most $-d/\xi_C$ for $\xi_C\ge8$. Empty supports give
    a vanishing commutator.
\end{proof}

\begin{theorem}[Positive factorization at the admissible length]
    \label{thm:hastings_source_length_factorization}
    \lean{TNLean.MPS.AreaLaw.two_mul_lemmaOneEndpointLength_le_sourceLength,
        TNLean.MPS.AreaLaw.exists_projectorFactorization_posSemidef_sourceLength}
    \leanok
    \uses{def:hastings_source_length,thm:hastings_lemma1_endpoint,
        def:hastings_factorization_at}
    There is a constant $K\ge0$, depending only on $R$, such that every
    unit ground vector $\Omega$ of a range-$R$ chain Hamiltonian with term
    norms at most $J$ and ground-line gap $\Delta$ has the following property.
    At every cut and every buffer width $l$, there are the supported
    contractions $O_B,O_L,O_R$ of Definition~\ref{def:hastings_factorization_at},
    with $O_B$ positive semidefinite, and
    \begin{align*}
        \|O_BO_LO_R-|\Omega\rangle\langle\Omega|\|
            \le K\Xi^3e^{-l/(6\Xi)}.
    \end{align*}
    This uses positivity in the semidefinite sense of the note added to
    \cite{Hastings2007AreaLaw}; it does not assert strict positivity of
    the projections.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint}
    The scalar inequality $2\zeta\le\Xi$ follows from
    $\Xi\ge80c_RJ/\Delta$ and $\Xi\ge80$.
    The positive endpoint factorization has error
    $K\zeta^3e^{-l/(12\zeta)}$. Enlarge its polynomial factor and its
    decay length using $2\zeta\le\Xi$.
\end{proof}

\begin{theorem}[The printed entropy-bound shape at the constructed length]
    \label{thm:hastings_source_length_entropy}
    \lean{TNLean.MPS.AreaLaw.Hastings.exists_regionEntropy_prefix_le_printed,
        TNLean.MPS.AreaLaw.Hastings.exists_regionEntropy_prefix_le_sourceLength}
    \leanok
    \uses{def:hastings_source_length,thm:hastings_area_law}
    There is $c_0>0$, depending only on $R$, such that every prefix of
    every ground vector in the preceding theorem satisfies
    \begin{align*}
        S(\Omega_{[0,i)})\le c_0\xi'\log\xi'\log q\,2^{\xi'\log q},
        \qquad \xi'=6\Xi.
    \end{align*}
    For nearest-neighbour interactions $c_0$ is numerical. More generally,
    the same form holds with $\Xi$ replaced by any $s\ge2\zeta$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_area_law,thm:hastings_area_law_bound_le}
    At $\xi_0'=6\zeta$, the factorization constant is
    $C=\max\{K\zeta^3,1\}$ and
    $\log C\le(\log(K+1)+3)\log\xi_0'$.
    The corrected doubling estimate and the bootstrap remainder therefore
    give $c_0\xi_0'\log\xi_0'\log q\,2^{2\xi_0'\log q}$, with $c_0$
    depending only on $R$. Since $2\zeta\le\Xi$, both the prefactor and
    exponent are bounded by the displayed expression. For $q=1$ every
    entropy is zero. The fixed-length doubling correction is unchanged.
\end{proof}
