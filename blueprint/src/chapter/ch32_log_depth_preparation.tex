\chapter{Preparation of Matrix Product States by Log-Depth Circuits}
\label{ch:log_depth_preparation}

A translation-invariant matrix product state on $N$ sites generated by a
normal tensor is approximated by blocking $q$ sites, replacing the positive
part of the polar decomposition of the blocked tensor by its renormalization
fixed point, and keeping the isometric part. The fixed point is a product of
nearest-neighbour entangled pairs, and the isometric part factorizes into
isometries with bounded bond dimension, either sequentially or along a binary
tree. With $q$ proportional to $\log(N/\varepsilon)$ this gives a circuit of
depth $O(\log(N/\varepsilon))$ with error $\varepsilon$, and no circuit of
depth $o(\log N)$ achieves a small error. The presentation follows
\cite{Malz2024Preparation}.

Throughout, $A$ is a tensor with physical dimension $d$ and bond dimension
$D$, $\E_A$ is its transfer map (Definition~\ref{def:transfer_map}), and
\begin{align}
    \ket{\phi_N(A)}
     & =\sum_{i_1,\dots,i_N=0}^{d-1}\Tr\left(A^{i_1}\cdots A^{i_N}\right)
    \ket{i_1\cdots i_N}
    \label{eq:ldp_ti_mps}
\end{align}
is the translation-invariant vector it generates, that is,
$\ket{\phi_N(A)}=\ket{V^{(N)}(A)}$ in the notation of
Definition~\ref{def:mpv_state}; the chapter keeps the source's notation
$\phi_N$. For two unit vectors the error is
$\varepsilon(\phi,\psi)=1-\lvert\braket{\phi}{\psi}\rvert$.

\section{The approximating state}

Write $\mathcal B_q$ for the physical matrix of the $q$-blocked tensor of $A$
(Definitions~\ref{def:blocked_tensor} and~\ref{def:ldpolar_physical_matrix})
and $\mathcal B_q=VP$ for its polar decomposition
(Theorem~\ref{thm:ldpolar_tensor_decomposition}).
In this section $A$ is normal (Definition~\ref{def:normal}) and in the gauge
\begin{align}
    \sum_{i=0}^{d-1}(A^i)^\dagger A^i
     & =\Id,
     &
    \E_A(\rho)
     & =\rho,
     &
    \rho
     & >0,
     &
    \Tr\rho
     & =1,
    \label{eq:ldp_normal_gauge}
\end{align}
so that $\E_A^{\,n}(X)\to\Tr(X)\,\rho$ as $n\to\infty$
\cite[eq.~(5)]{Malz2024Preparation}. The limit uses normality as follows.
Normality means that the words of some fixed length $L_0$ span $\MN{D}$, so
by Theorem~\ref{thm:wielandt_proposition_three} the map $\E_A$ is strongly
irreducible: its peripheral spectrum is $\{1\}$ and its fixed point $\rho$
is unique up to scale. Since $\E_A$ is trace preserving, $1$ is a
semisimple eigenvalue, and all other eigenvalues have modulus less than
$1$. A periodic tensor is not normal, because no blocking of it is
injective.

\begin{lemma}[Convergence of the positive part]
    \label{lem:ldp_positive_part_converges}
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor}
    Let $P_q=(\mathcal B_q^\dagger\mathcal B_q)^{1/2}$. Then $P_q\to P_\infty$ as
    $q\to\infty$, where $P_\infty$ is read as the map
    $\ket{\alpha,\beta}\mapsto\sum_\gamma(\sqrt\rho)_{\alpha\gamma}\ket{\gamma,\beta}$.
    This is the limit in \cite[eq.~(8)]{Malz2024Preparation}.
\end{lemma}
\begin{proof}
    The matrix of $\mathcal B_q^\dagger\mathcal B_q$ is a rearrangement of
    $\E_A^{\,q}$, which converges to $X\mapsto\Tr(X)\rho$; the rearrangement of
    the limit is $\rho^{\mathsf T}\otimes\Id$, and the square root is continuous
    on positive semidefinite matrices.
\end{proof}

\begin{definition}[Approximating state]
    \label{def:ldp_approximating_state}
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} and $N=qM$, the
    \emph{approximating state} is
    $\ket{\widetilde\phi_N}=\ket{\phi_M(VP_\infty)}/\lVert\phi_M(VP_\infty)\rVert$.
    By Lemma~\ref{lem:mpv_rotate_physical} and
    Theorem~\ref{thm:prep_mpv_fixed_point_tensor},
    $\ket{\phi_M(VP_\infty)}=V^{\otimes M}\bigotimes_k\ket{\omega}_{R_kL_{k+1}}$.
    This is \cite[eqs.~(9) and~(10)]{Malz2024Preparation}.
\end{definition}

\begin{lemma}[A unitary implementing the isometry]
    \label{lem:ldp_unitary_implementing_isometry}
    \uses{thm:ldpolar_tensor_decomposition, thm:seqanc_unitary_extension}
    Let $d\ge2$, put $r=\lceil\log_dD\rceil$, and suppose $q\ge2r$ and
    $\mathcal B_q$ is injective. Write
    $(\C^d)^{\otimes q}=L\otimes C\otimes R$, where $L$ and $R$ are the first
    and the last $r$ sites of the block and $C$ is the remaining $q-2r$ sites,
    embed $\C^D$ into $L$ and into $R$ as the span of the first $D$
    computational basis vectors, and let $\ket{0}_C=\ket{0}^{\otimes(q-2r)}$.
    Then there is a unitary $U$ on $(\C^d)^{\otimes q}$ with
    $U\bigl(\ket{\gamma}_L\ket{0}_C\ket{\delta}_R\bigr)=V\ket{\gamma,\delta}$
    for all $\gamma,\delta$, and
    \begin{align}
        \ket{\widetilde\phi_N}
         & =\Bigl(\bigotimes_{k=1}^M U_k\Bigr)
        \bigotimes_{k=1}^M\bigl(\ket{\omega}_{R_kL_{k+1}}\ket{0}_{C_k}\bigr).
        \label{eq:ldp_unitary_preparation}
    \end{align}
    This is \cite[eqs.~(10) and~(11)]{Malz2024Preparation}.
\end{lemma}

\section{Exact factorizations of the isometry}

\begin{theorem}[Successive decompositions with a matrix remainder]
    \label{thm:ldp_sweep_matrix_remainder}
    \lean{MPSPreparation.exists_isometric_chain_mul,
        MPSPreparation.sum_star_mulVec_dotProduct_of_isIsometryOn,
        MPSPreparation.sum_star_eval_mulVec_dotProduct}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    Let $a\le D$, let $R_{-1}\in\MN{D}$ be row-supported below $a$, and let
    $P_0,\dots,P_{n-1}$ be families of $D\times D$ matrices. Then there are
    $b_0=a$ and $b_1,\dots,b_n\le D$, families $Q_0,\dots,Q_{n-1}$ such that
    $Q_p^i$ is row-supported below $b_p$, vanishes on the columns
    $\beta\ge b_{p+1}$, and $Q_p$ is isometric on its first $b_{p+1}$ columns,
    and a matrix $R$ row-supported below $b_n$, such that for every
    configuration $\sigma$
    \begin{align}
        R_{-1}\,P^\sigma
         & =Q^\sigma\,R.
        \label{eq:ldp_sweep_matrix_remainder}
    \end{align}
    Moreover $b_{p+1}\le c$ whenever every $P_p^i$ vanishes on the columns
    $\gamma\ge c$. If $Q$ is isometric on its first $b$ columns and $v,w$ are
    supported below $b$, then
    $\sum_i\langle Q^iv,Q^iw\rangle=\langle v,w\rangle$. Consequently, if
    $Q_p^i$ is row-supported below $b_p$, $Q_p$ is isometric on its first
    $b_{p+1}$ columns, and $v,w$ are supported below $b_n$, then
    \begin{align}
        \sum_\sigma\langle Q^\sigma v,Q^\sigma w\rangle
         & =\langle v,w\rangle.
        \label{eq:ldp_sweep_inner_product}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_one_step, thm:seqanc_support_propagation}
    By induction on $n$. For $n=0$ take $R=R_{-1}$. Otherwise apply
    Theorem~\ref{thm:seqanc_one_step} to $Y^i=R_{-1}P_0^i$, which gives
    $Y^i=Q_0^iR_0$ with $R_0$ row-supported below $b_1$, and apply the
    induction hypothesis to $R_0$ and $P_1,\dots,P_{n-1}$:
    \begin{align}
        R_{-1}P_0^{\sigma_0}P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
         & =Q_0^{\sigma_0}R_0P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
        =Q_0^{\sigma_0}Q_1^{\sigma_1}\cdots Q_{n-1}^{\sigma_{n-1}}R.
        \notag
    \end{align}
    For one site, expand
    \begin{align}
        \sum_i\langle Q^iv,Q^iw\rangle
         & =\sum_{\beta,\beta'<b}\overline{v_\beta}w_{\beta'}
        \sum_{i,\alpha}\overline{Q^i_{\alpha\beta}}Q^i_{\alpha\beta'}
        =\sum_{\beta<b}\overline{v_\beta}w_\beta
        \notag
    \end{align}
    by the isometry condition. For the chain, write
    $Q^{\sigma}=Q_0^{\sigma_0}Q'^{\sigma'}$, where $Q'$ is the chain
    $Q_1,\dots,Q_{n-1}$ and $\sigma'=(\sigma_1,\dots,\sigma_{n-1})$. The
    vectors $Q'^{\sigma'}v$ and $Q'^{\sigma'}w$ are supported below $b_1$ by
    Theorem~\ref{thm:seqanc_support_propagation}, so the one-site identity for
    $Q_0$ and the induction hypothesis for $Q'$ give
    \begin{align}
        \sum_{\sigma_0,\sigma'}
        \langle Q_0^{\sigma_0}Q'^{\sigma'}v,Q_0^{\sigma_0}Q'^{\sigma'}w\rangle
         & =\sum_{\sigma'}\langle Q'^{\sigma'}v,Q'^{\sigma'}w\rangle
        =\langle v,w\rangle.
        \notag
    \end{align}
\end{proof}

\begin{theorem}[Sequential factorization of a matrix product isometry]
    \label{thm:ldp_sequential_factorization_general}
    \lean{MPSPreparation.exists_isometric_chain_of_eq_mul}
    \leanok
    Let $A$ have bond dimension $D$, let $G$ be a $D^2\times D^2$ matrix, and
    let $V\colon\C^D\otimes\C^D\to(\C^d)^{\otimes q}$, $q\ge1$, be an isometry
    with matrix elements
    $\bra{i_1\cdots i_q}V\ket{x}
        =\sum_{\alpha,\beta}(A^{i_1}\cdots A^{i_q})_{\alpha\beta}\,
        G_{(\alpha,\beta),x}$.
    Then there are integers $b_0=1$, $b_q=D^2$ and $b_p\le D^2$, and families
    of $b_{p-1}\times b_p$ matrices $X_p^i$, $p=1,\dots,q$, with
    $\sum_i(X_p^i)^\dagger X_p^i=\Id_{b_p}$ for every $p$, such that
    $\bra{i_1\cdots i_q}V\ket{x}=(X_1^{i_1}X_2^{i_2}\cdots X_q^{i_q})_{1x}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_sweep_matrix_remainder, thm:seqanc_support_propagation}
    The matrix elements of $V$ are
    $\bigl(j^{\mathsf T}(\Id_D\otimes A^{i_1})\cdots(\Id_D\otimes A^{i_q})G\bigr)_x$
    with $j=\sum_\gamma\ket{\gamma}\otimes\ket{\gamma}$. Apply
    Theorem~\ref{thm:ldp_sweep_matrix_remainder} to the row matrix of $j$ and the
    sites $\Id_D\otimes A^i$; this gives isometric sites $Q_1,\dots,Q_q$ and a
    remainder $C=RG$ with $V=Q\,C$ in the sense of
    (\ref{eq:ldp_sweep_matrix_remainder}). By
    (\ref{eq:ldp_sweep_inner_product}), $V^\dagger V=C^\dagger C$, so $C$ is an
    isometry because $V$ is. Absorb $C$ into the last site, $X_q^i=Q_q^iC$ and
    $X_p=Q_p$ for $p<q$; the last site is again isometric,
    $\sum_i C^\dagger(Q_q^i)^\dagger Q_q^iC=C^\dagger C=\Id$.
\end{proof}

\begin{theorem}[Sequential factorization]
    \label{thm:ldp_sequential_factorization}
    \lean{MPSPreparation.exists_isometric_chain_polarIsoMatrix}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition}
    Suppose $\mathcal B_q$ is injective, $q\ge1$, and let $V$ be as in
    Theorem~\ref{thm:ldpolar_tensor_decomposition}. There are integers
    $D'_1=D^2$, $D'_{q+1}=1$, and $D'_k\le D^2$, and matrices
    $W_k^i\in\C^{D'_{k+1}\times D'_k}$ for $k=1,\dots,q$ and $i=1,\dots,d$, with
    \begin{align}
        \sum_{i=1}^d\bigl(W_k^i\bigr)^\dagger W_k^i
         & =\Id_{D'_k}
        \quad\text{and}\quad
        \bra{i_1\cdots i_q}V\ket{x}
        =\bigl(W_q^{i_1}W_{q-1}^{i_2}\cdots W_1^{i_q}\bigr)_{1x}
        \label{eq:ldp_sequential_factorization}
    \end{align}
    for every $k$, every configuration $i_1\cdots i_q$ and every $x$, where $\C^{D'_1}$ is
    identified with $\C^D\otimes\C^D$. This is the matrix product form of
    \cite[eqs.~(13)--(15)]{Malz2024Preparation}, where the isometry carrying the
    input is written $\widetilde C=V_1$. The block length $q\ge1$ is implicit
    in the source, which draws at least one site; the empty block $q=0$ is
    excluded, see~\cite{gap:mswc24_sequential_factorization_positive_block_length}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldpolar_tensor_injective,
        thm:ldp_sequential_factorization_general}
    Since $\mathcal B_q$ is injective, $P$ is positive definite and
    $V=\mathcal B_qP^{-1}$, so
    $\bra{i_1\cdots i_q}V\ket{x}
        =\sum_{\alpha,\beta}(A^{i_1}\cdots A^{i_q})_{\alpha\beta}(P^{-1})_{(\alpha,\beta),x}$.
    This is an open-boundary matrix product map with bond dimension at most
    $D^2$: the site matrices are $\Id_D\otimes A^i$ (the source writes
    $A^i\otimes\Id_D$; the order of the two bond factors only permutes the
    bond index), the two bond factors are
    joined at the left end, and $P^{-1}$ is absorbed into the right end
    \cite[eq.~(13)]{Malz2024Preparation}. Theorem~\ref{thm:ldp_sequential_factorization_general}
    with $G=P^{-1}$ makes every site isometric by successive decompositions
    from the left \cite[eq.~(14)]{Malz2024Preparation}; the remainder at the
    input end is an isometry because $V$ is
    \cite[eq.~(15)]{Malz2024Preparation}. The sites of that theorem are
    numbered from the output end and those here from the input end, so its
    $p$-th site is the site $q+1-p$ here, $X_p^i=W_{q+1-p}^i$ for
    $p=1,\dots,q$, which turns $(X_1^{i_1}\cdots X_q^{i_q})_{1x}$ into
    $(W_q^{i_1}\cdots W_1^{i_q})_{1x}$ and $b_p$ into $D'_{q+1-p}$.
\end{proof}

\begin{remark}[Sequential factorization as a composition of isometries]
    \label{rem:ldp_sequential_factorization_composition}
    \uses{thm:ldp_sequential_factorization}
    The source states Theorem~\ref{thm:ldp_sequential_factorization} as an
    ordered composition. With the matrices of
    (\ref{eq:ldp_sequential_factorization}), let
    $W_k\colon\C^{D'_k}\to\C^{D'_{k+1}}\otimes\C^d$ be the map with
    $\bra{\alpha,i}W_k\ket{\beta}=(W_k^i)_{\alpha\beta}$. The first identity in
    (\ref{eq:ldp_sequential_factorization}) says that $W_k$ is an isometry,
    $W_k^\dagger W_k=\sum_i(W_k^i)^\dagger W_k^i=\Id_{D'_k}$. Expanding the
    composition in the product basis, with the tensor factor $\C^d$ of $W_k$
    placed at the site $q+1-k$ of the block and $\C^{D'_{q+1}}=\C$, the second
    identity becomes
    \begin{align}
        V
         & =\bigl(W_q\otimes\Id_{d}^{\otimes(q-1)}\bigr)\cdots
        \bigl(W_2\otimes\Id_d\bigr)W_1,
        \label{eq:ldp_sequential_factorization_composition}
    \end{align}
    which is \cite[eqs.~(13)--(15)]{Malz2024Preparation} as printed. This
    operator form is not formalized; the formal statement is the matrix product
    form (\ref{eq:ldp_sequential_factorization}).
\end{remark}

\begin{theorem}[Tree factorization]
    \label{thm:ldp_tree_factorization}
    \uses{thm:ldpolar_tensor_decomposition, lem:mpv_rotate_physical}
    Put $T_0=A$ and, for $j\ge1$, let $\mathcal B^{(j)}=V^{(j)}P_j$ be the polar
    decomposition of the two-site blocked map of $T_{j-1}$, and let $T_j$ be the
    tensor of $P_j$ (Definition~\ref{def:ldpolar_tensor}). Then for
    $q=2^k$,
    \begin{align}
        \mathcal B_q
         & =\Bigl(V^{(1)}\Bigr)^{\otimes2^{k-1}}
        \Bigl(V^{(2)}\Bigr)^{\otimes2^{k-2}}\cdots V^{(k)}\,P_k.
        \label{eq:ldp_tree_factorization}
    \end{align}
    Here $V^{(1)}$ maps $\C^{D^2}$ to $(\C^d)^{\otimes2}$ and each $V^{(j)}$ with
    $j\ge2$ maps $\C^{D^2}$ to $\C^{D^2}\otimes\C^{D^2}$. The product of the
    layers is a partial isometry with initial space the range of $P_k$, so by
    uniqueness of the polar decomposition it equals $V$ and $P_k=P_q$. This is
    \cite[eq.~(16)]{Malz2024Preparation}.
\end{theorem}
\begin{proof}
    Induction on $k$ using Lemma~\ref{lem:mpv_rotate_physical} for
    open products. The range of the two-site blocked map of $T_{j-1}$ lies in
    $\operatorname{ran}P_{j-1}\otimes\operatorname{ran}P_{j-1}$, so each layer
    is applied inside the initial space of the next.
\end{proof}

\section{Approximation error}

For a normal tensor $A$, let $\lambda_2$ be an eigenvalue of $\E_A$ of
largest modulus other than the Perron eigenvalue $1$. By the remark after
(\ref{eq:ldp_normal_gauge}), $\lvert\lambda_2\rvert<1$. The
\emph{correlation length} of $A$ is the quantity
$\xi=-1/\log\lvert\lambda_2\rvert$ of Definition~\ref{def:correlation_length}
at this $\lambda_2$. It is finite, it is positive exactly when
$\lambda_2\neq0$, and $\lvert\lambda_2\rvert=e^{-1/\xi}$ in that case.

\begin{lemma}[Overlap of the positive part with the fixed point]
    \label{lem:ldp_positive_part_overlap}
    \uses{def:prep_fixed_point_tensor, thm:ldpolar_blocked_state,
        def:correlation_length}
    Let $A$ be normal with correlation length $\xi$ and let $0<\gamma<1/2$.
    Then, for $N=qM$,
    \begin{align}
        \Bigl\lvert1-\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\bigr\rvert\Bigr\rvert
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi}\right).
        \label{eq:ldp_positive_part_overlap}
    \end{align}
    This is \cite[Supplemental Material, eq.~(S29)]{Piroli2021LOCC}, quoted as
    \cite[eq.~(S9)]{Malz2024Preparation}.
\end{lemma}

\begin{theorem}[Approximation error, normal case]
    \label{thm:ldp_approximation_error_normal}
    \uses{lem:ldp_positive_part_overlap, def:ldp_approximating_state,
        def:correlation_length}
    Let $A$ be normal with correlation length $\xi$, let $\ket{\phi_N}$ be the
    normalized vector (\ref{eq:ldp_ti_mps}), and let $0<\gamma<1/2$. For
    $N=qM$ with $q\to\infty$ and $q=o(N)$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi}\right).
        \label{eq:ldp_approximation_error_normal}
    \end{align}
    In particular $q=\lceil\xi(1+\eta)\ln N/\gamma\rceil$ gives
    $e^{-\gamma q/\xi}\le N^{-(1+\eta)}$ and hence
    $\varepsilon=O(N^{-\eta})$ for every $\eta>0$. This is
    \cite[Lemma~1 and Lemma~1$'$(i)]{Malz2024Preparation}.
\end{theorem}

\section{Depth of the preparation}

\begin{definition}[Local circuit of depth $T$]
    \label{def:ldp_local_circuit}
    \lean{MPSPreparation.IsLocalCircuitOfDepth, MPSPreparation.IsPreparedInDepth,
        MPSPreparation.Layer, MPSPreparation.circuitOp,
        MPSPreparation.supportedOperators}
    \leanok
    A \emph{local circuit of depth $T$} on a chain of $N$ sites is a product of
    $T$ layers, each a tensor product of unitaries acting on disjoint pairs of
    neighbouring sites. A state is \emph{prepared in depth $T$} if it equals a
    local circuit of depth $T$ applied to a product state.

    The chain is a ring: the neighbouring pairs are $\{k,k+1\}$ with indices
    modulo $N$, and a circuit whose gates avoid the pair $\{N,1\}$ is a circuit
    on the open chain. An operator \emph{acts on} a set of sites $X$ if it
    lies in $\MN{d}^{\otimes X}\otimes\mathbb 1$, the span of the products
    $\bigotimes_i m_i$ with $m_i=\mathbb 1$ for $i\notin X$. Two sets of sites
    are at distance larger than $s$ if no site of one is within ring distance
    $s$ of a site of the other.
\end{definition}

\begin{lemma}[Light cone of a local circuit]
    \label{lem:ldp_light_cone}
    \uses{def:ldp_local_circuit}
    \lean{MPSPreparation.conj_circuitOp_mem_supportedOperators}
    \leanok
    Let $U$ be a local circuit of depth $T$ and let $Z$ act on a set of sites
    $X$. Then $U^\dagger ZU$ acts on the sites within ring distance $T$ of $X$.
    This is the finite light cone of \cite[Theorem~1]{Malz2024Preparation}.
\end{lemma}
\begin{proof}
    \leanok
    By induction on the layers it suffices to treat one layer $W$. Split its
    gates into those whose pair meets $X$, with product $G$, and the others,
    with product $H$. The gates commute, so $W=HG$. The operator $H$ is unitary
    and acts on the complement of $X$, so it commutes with $Z$ and
    $W^\dagger ZW=G^\dagger ZG$. The pairs of $G$ lie within distance $1$ of
    $X$, and products and adjoints of operators acting on a set act on it.
\end{proof}

\begin{lemma}[No connected correlations beyond the light cone]
    \label{lem:ldp_light_cone_correlations}
    \uses{lem:ldp_light_cone}
    \lean{MPSPreparation.expect_mul_eq_of_isPreparedInDepth,
        MPSPreparation.expect_mul_mul_expect_one_of_isPreparedInDepth}
    \leanok
    Let $\ket{\psi}$ be a unit vector prepared in depth $T$, and let $Z$, $Z'$
    act on sets of sites at distance larger than $2T$. Then
    $\bra{\psi}ZZ'\ket{\psi}=\bra{\psi}Z\ket{\psi}\bra{\psi}Z'\ket{\psi}$.
    This is the statement in the proof of
    \cite[Theorem~1]{Malz2024Preparation} that every connected correlation of
    such operators vanishes.
\end{lemma}
\begin{proof}
    \leanok
    Write $\ket{\psi}=U\ket{\pi}$ with $\ket{\pi}=\bigotimes_i\ket{v_i}$. Since
    $UU^\dagger=\mathbb 1$,
    $\bra{\psi}ZZ'\ket{\psi}=\bra{\pi}(U^\dagger ZU)(U^\dagger Z'U)\ket{\pi}$,
    and by Lemma~\ref{lem:ldp_light_cone} the two factors act on disjoint sets
    of sites. For products $\bigotimes_i m_i$ and $\bigotimes_i m'_i$ with
    disjoint supports the expectation in $\ket{\pi}$ is
    $\prod_i\bra{v_i}m_im'_i\ket{v_i}$, and at each site one of $m_i$, $m'_i$
    is the identity; bilinearity then gives
    $\bra{\pi}AB\ket{\pi}\braket{\pi}{\pi}
        =\bra{\pi}A\ket{\pi}\bra{\pi}B\ket{\pi}$ for $A$, $B$ acting on
    disjoint sets.
\end{proof}

\begin{theorem}[Preparation in depth $O(\log(N/\varepsilon))$]
    \label{thm:ldp_depth_upper_bound}
    \uses{def:ldp_local_circuit, lem:ldp_unitary_implementing_isometry,
        thm:ldp_sequential_factorization, thm:ldp_tree_factorization,
        thm:ldp_approximation_error_normal}
    Let $A$ be normal. There is a constant $c$ depending only on $A$ such that,
    for every $N\ge2$ and $0<\varepsilon\le1$, a state $\ket{\psi_N}$ with
    $\varepsilon(\psi_N,\phi_N)\le\varepsilon$ is prepared in depth at most
    $c\log(N/\varepsilon)$. This is \cite[eq.~(1)]{Malz2024Preparation}.
\end{theorem}

The correlation estimate below rests on four ingredients, which are proved
first. For an operator $X$ on $(\C^d)^{\otimes L}$ write $\E_X$ for the map
$Z\mapsto\sum_{w,w'}\bra{w'}X\ket{w}A^wZ(A^{w'})^\dagger$ on $\MN{D}$, where
$w,w'$ run over words of length $L$ and $A^w$ is the product of the matrices
of $A$ along $w$.

\begin{lemma}[Every map is an inserted transfer map]
    \label{lem:ldp_observable_transfer_surjective}
    \lean{MPSTensor.exists_physicalObservableTransfer_eq}
    \leanok
    \uses{def:transfer_map}
    Suppose the products of $L$ matrices of $A$ span $\MN{D}$. Then every
    linear map $\Phi$ on $\MN{D}$ equals $\E_X$ for some operator $X$ on
    $(\C^d)^{\otimes L}$. For $L=1$ and injective $A$ this is the step
    ``we can always choose $\mathcal O$ (and $\mathcal O'$), such that the
    corresponding transfer matrix $E_{\mathcal O}=\ket{A}\bra{B}$ for
    arbitrary $A,B$ (up to a normalization constant)'' in the proof of
    \cite[Lemma~2]{Malz2024Preparation}.
\end{lemma}
\begin{proof}
    \leanok
    Choose coefficients $c^{ab}_w$ with $\sum_wc^{ab}_wA^w=e_{ab}$. The operator
    with matrix elements $\bra{w'}X\ket{w}=c^{ab}_w\overline{c^{ce}_{w'}}$ gives
    $\E_X(Z)=e_{ab}Ze_{ec}=Z_{be}\,e_{ac}$. These maps span all linear maps on
    $\MN{D}$, and $X\mapsto\E_X$ is linear.
\end{proof}

\begin{lemma}[The limit correlator at an eigenvector]
    \label{lem:ldp_limit_correlator_eigenvalue}
    \lean{MPSTensor.limitCorrelator, MPSTensor.exists_limitCorrelator_eq_pow}
    \leanok
    \uses{def:transfer_map}
    Suppose the products of $L$ matrices of $A$ span $\MN{D}$ and
    $\sum_i(A^i)^\dagger A^i=\Id$, let $\Tr\rho=1$, put $P(Z)=\Tr(Z)\,\rho$,
    and let $R\neq0$ satisfy $\E_A(R)=\lambda R$ with $\lambda\neq1$. Then there
    are operators $X,Y$ on $(\C^d)^{\otimes L}$ such that for every $t\ge0$
    \begin{align}
        \Tr\Bigl(\E_X\bigl((\E_A-P)^t\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr)
         & =\lambda^t.
        \notag
    \end{align}
    When $\rho$ is the fixed point of the gauge (\ref{eq:ldp_normal_gauge}),
    the left side is the limit correlator (\ref{eq:ldp_limit_correlator}) at
    $t=s-L-1$. This is the choice of observables in the proof of
    \cite[Lemma~2]{Malz2024Preparation}, without the source's requirement that
    the one-point functions vanish.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:ldp_observable_transfer_surjective}
    Trace preservation gives $\lambda\Tr R=\Tr R$, so $\Tr R=0$, $P(R)=0$, and
    $(\E_A-P)^tR=\lambda^tR$. By
    Lemma~\ref{lem:ldp_observable_transfer_surjective} take
    $\E_Y(Z)=\Tr(Z)\,R$ and $\E_X(Z)=(Z_{ij}/R_{ij})\,\rho$ for an entry
    $R_{ij}\neq0$.
\end{proof}

\begin{lemma}[Exponential sums on windows]
    \label{lem:ldp_vandermonde_window}
    \lean{Complex.exists_window_le_norm_sum_mul_pow}
    \leanok
    Let $\mu_1,\dots,\mu_K$ be distinct complex numbers of modulus one and
    $a\in\C^K$ nonzero. There is $c_0>0$ such that for every integer $t\ge0$
    some $u\in\{0,\dots,K-1\}$ satisfies
    $\bigl\lvert\sum_{j=1}^Ka_j\mu_j^{\,t+u}\bigr\rvert\ge c_0$. This gives
    the lower bound of (\ref{eq:ldp_vandermonde_window}) for some positive
    constant $c_0$, not necessarily the explicit constant displayed there.
\end{lemma}
\begin{proof}
    \leanok
    The Vandermonde map $v\mapsto(\sum_jv_j\mu_j^{\,u})_{0\le u<K}$ is
    injective, so it is bounded below on $\C^K$. The vector
    $(a_j\mu_j^t)_j$ has the maximum norm of $a$.
\end{proof}

\begin{lemma}[Expectations in overlapping states]
    \label{lem:ldp_expectation_overlap}
    \lean{LinearMap.IsSymmetric.norm_inner_mul_norm_sub_le}
    \leanok
    For a Hermitian operator $Z$ and unit vectors $\phi,\psi$, with
    $\sigma_\chi(Z)=\lVert(Z-\langle Z\rangle_\chi)\chi\rVert$, the inequality
    (\ref{eq:ldp_expectation_overlap}) holds:
    $\lvert\braket{\phi}{\psi}\rvert\,
        \lvert\langle Z\rangle_\phi-\langle Z\rangle_\psi\rvert
        \le\sigma_\phi(Z)+\sigma_\psi(Z)$.
\end{lemma}
\begin{proof}
    \leanok
    Since $\langle Z\rangle_\phi$ is real,
    $(\langle Z\rangle_\phi-\langle Z\rangle_\psi)\braket{\phi}{\psi}
        =\bra{\phi}(Z-\langle Z\rangle_\psi)\ket{\psi}
        -\bra{\phi}(Z-\langle Z\rangle_\phi)\ket{\psi}$; apply the
    Cauchy--Schwarz inequality to both terms.
\end{proof}

\begin{lemma}[Exponentially decaying correlations]
    \label{lem:ldp_decaying_correlations}
    \uses{def:normal, def:correlation_length}
    Let $A$ be normal with correlation length $\xi>0$, let $L\ge1$ be a length
    such that the products of $L$ matrices of $A$ span $\MN{D}$ (one may take
    $L=1$ when $A$ is injective), and let $\ket{\phi_N}$ be the normalized
    vectors (\ref{eq:ldp_ti_mps}). For operators $X,Y$ on $(\C^d)^{\otimes L}$
    let $X_1$ act on the sites $1,\dots,L$, let $Y_s$ act on the sites
    $s,\dots,s+L-1$, and put
    \begin{align}
        G_N(X,Y;s)
         & =\bra{\phi_N}X_1Y_s\ket{\phi_N}
        -\bra{\phi_N}X_1\ket{\phi_N}\bra{\phi_N}Y_s\ket{\phi_N}.
        \label{eq:ldp_connected_correlator}
    \end{align}
    There are Hermitian operators $\mathcal O,\mathcal O'$ on
    $(\C^d)^{\otimes L}$ of norm one, a constant $c>0$, an integer $K\ge1$ and
    an integer $s_0\ge L+2$ such that for every $s\ge s_0$ some
    $s'\in\{s,s+1,\dots,s+K-1\}$ satisfies, for every $N\ge3s'$,
    \begin{align}
        \bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
         & \ge c\,e^{-(s'-1)/\xi}.
        \label{eq:ldp_decaying_correlations}
    \end{align}
    One may take $K$ to be the number of distinct eigenvalues of $\E_A$ of
    modulus $\lvert\lambda_2\rvert$. For injective $A$ and $L=1$ this is the
    correlation estimate of \cite[Lemma~2]{Malz2024Preparation}, with the
    separation $s$ ranging over windows of $K$ consecutive values.
\end{lemma}
\begin{proof}
    \uses{def:transfer_map, lem:ldp_observable_transfer_surjective,
        lem:ldp_limit_correlator_eigenvalue, lem:ldp_vandermonde_window}
    With the inserted transfer maps $\E_X$ defined before
    Lemma~\ref{lem:ldp_observable_transfer_surjective},
    $\E_{\Id}=\E_A^{\,L}$. Expanding the traces in
    (\ref{eq:ldp_ti_mps}) gives, for $L<s$ and $s+L-1\le N$,
    \begin{align}
        \bra{\phi_N(A)}X_1Y_s\ket{\phi_N(A)}
         & =\operatorname{tr}\bigl(\E_X\,\E_A^{\,s-L-1}\,\E_Y\,
        \E_A^{\,N-s-L+1}\bigr),
        \notag
    \end{align}
    the trace of a linear map on $\MN{D}$. Let $P(Z)=\Tr(Z)\,\rho$. In the
    gauge (\ref{eq:ldp_normal_gauge}), $\E_A^{\,n}=P+(\E_A-P)^n$ for $n\ge1$,
    and $\E_A-P$ has spectral radius $\lvert\lambda_2\rvert$. Fix $r$ with
    $\lvert\lambda_2\rvert<r<1$ and $r^2<\lvert\lambda_2\rvert$; then
    $\lVert(\E_A-P)^n\rVert\le C_rr^n$. Replacing the last power
    $\E_A^{\,N-s-L+1}$ by $P$, and likewise in the normalization
    $\operatorname{tr}\E_A^{\,N}$ and in the one-point functions, changes
    $G_N(X,Y;s)$ by at most $C_{X,Y}\,r^{N-s-L+1}$, which is at most
    $C'_{X,Y}\,r^{2s}$ when $N\ge3s$. The remaining term is
    \begin{align}
        G_\infty(X,Y;s)
         & =\Tr\Bigl(\E_X\bigl((\E_A-P)^{s-L-1}\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr),
        \label{eq:ldp_limit_correlator}
    \end{align}
    which is bilinear in $(X,Y)$.

    The matrices $A^w\otimes\overline{A^{w'}}$ with $w,w'$ of length $L$ span
    $\MN{D}\otimes\MN{D}$, so every linear map on $\MN{D}$ is of the form
    $\E_X$. Since $\xi>0$, $\lambda_2\neq0$. Let $R$ be an eigenvector,
    $\E_A(R)=\lambda_2R$; then $\Tr R=0$ because $\E_A$ is trace preserving,
    so $P(R)=0$. Choose $Y$ with $\E_Y(\rho)=R$ and $X$ with
    $\Tr\E_X(Z)=\Tr(M^\dagger Z)$ for all $Z$, where $\Tr(M^\dagger R)=1$.
    Then $G_\infty(X,Y;s)=\lambda_2^{\,s-L-1}$.

    Let $\lambda_2=\lambda^{(1)},\dots,\lambda^{(K)}$ be the distinct
    eigenvalues of $\E_A$ of modulus $\lvert\lambda_2\rvert$, put
    $\mu_j=\lambda^{(j)}/\lvert\lambda_2\rvert$, and fix $\delta>0$ with all
    other eigenvalues of $\E_A-P$ of modulus less than
    $\lvert\lambda_2\rvert-\delta$. The Jordan decomposition of $\E_A-P$ gives,
    for Hermitian $X',Y'$ and $t=s-L-1$,
    \begin{align}
        G_\infty(X',Y';s)
         & =\lvert\lambda_2\rvert^t\sum_{j=1}^Kp_j(t)\,\mu_j^t
        +O\bigl((\lvert\lambda_2\rvert-\delta)^t\bigr),
        \label{eq:ldp_peripheral_expansion}
    \end{align}
    with polynomials $p_j$ depending bilinearly on $(X',Y')$; the polynomials
    are constants when the eigenvalues of modulus $\lvert\lambda_2\rvert$ are
    semisimple. Write $X=X^{(1)}+iX^{(2)}$ and $Y=Y^{(1)}+iY^{(2)}$ with
    Hermitian parts. If all four pairs $(X^{(a)},Y^{(b)})$ had vanishing
    polynomials $p_j$, bilinearity would give
    $G_\infty(X,Y;s)=O((\lvert\lambda_2\rvert-\delta)^t)$, which is false. Take
    $\mathcal O,\mathcal O'$ to be a pair $X^{(a)},Y^{(b)}$ with some
    $p_j\neq0$, scaled to norm one. Let $m$ be the largest degree of the
    $p_j$ and $a_j$ the coefficient of $t^m$ in $p_j$, so
    $a=(a_1,\dots,a_K)\neq0$.

    The Vandermonde matrix $W=(\mu_j^{\,u})_{0\le u<K,\,1\le j\le K}$ is
    invertible because the $\mu_j$ are distinct. For every $t$ the vector
    with entries $\sum_ja_j\mu_j^{\,t+u}$, $0\le u<K$, is $W$ applied to
    $(a_j\mu_j^t)_j$, whose maximum norm equals that of $a$ because
    $\lvert\mu_j\rvert=1$. Hence some $u\in\{0,\dots,K-1\}$ satisfies
    \begin{align}
        \Bigl\lvert\sum_{j=1}^Ka_j\mu_j^{\,t+u}\Bigr\rvert
         & \ge c_0,
         &
        c_0
         & =\frac{\max_j\lvert a_j\rvert}{\lVert W^{-1}\rVert_\infty}>0,
        \label{eq:ldp_vandermonde_window}
    \end{align}
    so a nonzero exponential sum of this kind cannot be small at $K$
    consecutive integers. At $t'=t+u$, (\ref{eq:ldp_peripheral_expansion})
    gives
    $\lvert G_\infty(\mathcal O,\mathcal O';s+u)\rvert\ge
        \lvert\lambda_2\rvert^{t'}\bigl(t'^mc_0-O(t'^{m-1})\bigr)
        -O\bigl((\lvert\lambda_2\rvert-\delta)^{t'}\bigr)
        \ge\tfrac{c_0}{2}\lvert\lambda_2\rvert^{t'}$
    once $s$ is large. With $s'=s+u$, $N\ge3s'$ and $r^2<\lvert\lambda_2\rvert$,
    the finite-size correction $O(r^{2s'})$ is at most
    $\tfrac{c_0}{4}\lvert\lambda_2\rvert^{t'}$ after enlarging $s_0$. Finally
    $\lvert\lambda_2\rvert^{t'}=e^{-(s'-L-1)/\xi}\ge e^{-(s'-1)/\xi}$, which
    gives (\ref{eq:ldp_decaying_correlations}) with $c=c_0/4$.
\end{proof}

\begin{theorem}[No preparation in depth $o(\log N)$]
    \label{thm:ldp_depth_lower_bound}
    \uses{def:ldp_local_circuit, def:correlation_length}
    Let $A$ be normal with correlation length $\xi>0$, let $\ket{\phi_N}$ be the
    normalized vectors (\ref{eq:ldp_ti_mps}), and let $\ket{\psi_N}$ be prepared
    in depth $T_N$ with $T_N=o(\log N)$. Then there is $N_0$ such that
    $\varepsilon(\phi_N,\psi_N)>1/2$ for all $N>N_0$. This is
    \cite[Theorem~1]{Malz2024Preparation}.
\end{theorem}
\begin{proof}
    \uses{lem:ldp_decaying_correlations, lem:ldp_expectation_overlap}
    Take $L,\mathcal O,\mathcal O',c,K,s_0$ from
    Lemma~\ref{lem:ldp_decaying_correlations}, and suppose
    $\varepsilon(\phi_N,\psi_N)\le1/2$, that is
    $\lvert\braket{\phi_N}{\psi_N}\rvert\ge1/2$. Write $T=T_N$ and
    $\ket{\psi_N}=U\ket{\pi}$ with $U$ a local circuit of depth $T$ and
    $\ket{\pi}$ a product state. If $Z$ acts on an interval of sites, then
    $U^\dagger ZU$ acts on the interval enlarged by $T$ sites on each side, so
    expectations in $\ket{\psi_N}$ of products of operators whose enlarged
    intervals are pairwise disjoint factorize.

    Let $s_N=\max(s_0,2T+L+1)$ and let $s'\in\{s_N,\dots,s_N+K-1\}$ be given
    by the lemma; the window changes $s'$ by at most $K-1$, so still
    $s'=o(\log N)$ and $N\ge3s'$ for large $N$. Put $\Delta=s'+L+2T\le2s'$,
    $n=\lfloor N/\Delta\rfloor\ge N/(4s')$, and $b_k=1+(k-1)\Delta$ for
    $k=1,\dots,n$. With $e=\bra{\phi_N}\mathcal O_1\ket{\phi_N}$ and
    $e'=\bra{\phi_N}\mathcal O'_1\ket{\phi_N}$ let $\mathcal O^{(k)}$ be
    $\mathcal O-e$ on the sites $b_k,\dots,b_k+L-1$ and $\mathcal O'^{(k)}$ be
    $\mathcal O'-e'$ on the sites $b_k+s',\dots,b_k+s'+L-1$. These have norm at
    most $2$, zero expectation in $\ket{\phi_N}$ by translation invariance,
    and $\bra{\phi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\phi_N}
        =G_N(\mathcal O,\mathcal O';s')$. Since $s'\ge2T+L+1$ and
    $\Delta\ge s'+L+2T$, the $2n$ enlarged intervals are pairwise disjoint,
    also around the ring.

    Two estimates are used. For Hermitian $Z$ and unit vectors $\phi,\psi$,
    with $\sigma_\chi(Z)=\lVert(Z-\langle Z\rangle_\chi)\chi\rVert$,
    \begin{align}
        \lvert\braket{\phi}{\psi}\rvert\,
        \bigl\lvert\langle Z\rangle_\phi-\langle Z\rangle_\psi\bigr\rvert
         & \le\sigma_\phi(Z)+\sigma_\psi(Z),
        \label{eq:ldp_expectation_overlap}
    \end{align}
    because
    $(\langle Z\rangle_\phi-\langle Z\rangle_\psi)\braket{\phi}{\psi}
        =\bra{\phi}(Z-\langle Z\rangle_\psi)\ket{\psi}
        -\bra{\phi}(Z-\langle Z\rangle_\phi)\ket{\psi}$. Secondly, the
    decomposition $\E_A^{\,n}=P+(\E_A-P)^n$ from the proof of
    Lemma~\ref{lem:ldp_decaying_correlations}, together with the fact that in
    the gauge (\ref{eq:ldp_normal_gauge}) the maps $\E_Z$ have norm bounded by
    a multiple of $\lVert Z\rVert$ independent of the size of the support,
    shows that the connected correlation in $\ket{\phi_N}$ of two operators of
    norm at most $4$ on intervals separated by $g$ and $g'$ sites around the
    ring is at most $C(r^g+r^{g'})$, with $C$ and $r<1$ depending only on $A$.
    Consecutive intervals $[b_k,b_k+s'+L-1]$ are separated by at least $2T$
    sites, and intervals $j$ apart by at least $(j-1)\Delta$, so every
    average of $n$ of these operators, or of their pair products, has
    variance at most $C_1/n$ in $\ket{\phi_N}$.

    For signs $\epsilon_k\in\{\pm1\}$ apply (\ref{eq:ldp_expectation_overlap})
    to $Z=\frac1n\sum_k\epsilon_k\mathcal O^{(k)}$. Its expectation in
    $\ket{\phi_N}$ is $0$; in $\ket{\psi_N}$ it is
    $\frac1n\sum_k\epsilon_kx_k$ with
    $x_k=\bra{\psi_N}\mathcal O^{(k)}\ket{\psi_N}\in\R$, and by factorization
    its variance there is at most $4/n$. Hence
    $\lvert\frac1n\sum_k\epsilon_kx_k\rvert\le C_2n^{-1/2}$, and the choice
    $\epsilon_k=\operatorname{sign}x_k$ gives
    $\frac1n\sum_k\lvert x_k\rvert\le C_2n^{-1/2}$. Next apply
    (\ref{eq:ldp_expectation_overlap}) to
    $Z=\frac1n\sum_k\mathcal O^{(k)}\mathcal O'^{(k)}$, which is Hermitian
    because the two factors act on disjoint sites. Its expectation in
    $\ket{\phi_N}$ is $G_N(\mathcal O,\mathcal O';s')$. In $\ket{\psi_N}$ the
    light cone gives
    $\bra{\psi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\psi_N}=x_ky_k$ with
    $\lvert y_k\rvert\le2$, so the expectation is at most $2C_2n^{-1/2}$ in
    modulus and the variance is at most $16/n$. With
    (\ref{eq:ldp_decaying_correlations}) and $n\ge N/(4s')$,
    \begin{align}
        c\,e^{-(s'-1)/\xi}
         & \le\bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
        \le C_3\Bigl(\frac{s'}{N}\Bigr)^{1/2}.
        \label{eq:ldp_depth_contradiction}
    \end{align}
    Since $s'\le s_N+K-1=o(\log N)$, the left side is $N^{-o(1)}$ and the
    right side is $N^{-1/2+o(1)}$, so (\ref{eq:ldp_depth_contradiction}) fails
    for all $N$ larger than some $N_0$ determined by $A$ and the sequence
    $(T_N)$. For those $N$, $\varepsilon(\phi_N,\psi_N)>1/2$.
\end{proof}

\section{Tensors that are not normal}

\begin{definition}[Fixed point of a basis of normal tensors]
    \label{def:ldp_non_normal_fixed_point}
    \uses{def:bnt, def:prep_fixed_point_tensor, thm:ldpolar_tensor_decomposition,
        def:correlation_length}
    Let $A^i=\bigoplus_{j=1}^b\operatorname{diag}(\mu_{j,1},\dots,\mu_{j,m_j})
        \otimes A_j^i$ with normal $A_j$, each in the gauge
    (\ref{eq:ldp_normal_gauge}), forming a basis of normal tensors, and put
    $\beta_j=\sum_{k=1}^{m_j}\mu_{j,k}^N$. Consider only lengths $N=qM$ with
    $(\beta_1,\dots,\beta_b)\neq0$; for the other lengths
    $\ket{\phi_N(A)}=\sum_j\beta_j\ket{\phi_N(A_j)}=0$ and there is nothing to
    prepare. Write $\xi_j$ for the correlation length of the normal tensor
    $A_j$, formed from the subleading eigenvalue of its own transfer map
    $\E_{A_j}$. With $\ket{\omega_j}$ the pair state of $A_j$ and
    $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$, the
    \emph{fixed-point state} is
    \begin{align}
        \ket{\Omega'}
         & =\sum_{j=1}^b\alpha^{(N)}_j\bigotimes_{k=1}^{M}\ket{\omega_j}_{R_kL_{k+1}}.
        \label{eq:ldp_non_normal_fixed_point}
    \end{align}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} for the blocked map of
    $A$, the \emph{approximating state} is
    $\ket{\widetilde\phi_N}=V^{\otimes M}\ket{\Omega'}/
        \lVert V^{\otimes M}\ket{\Omega'}\rVert$, whenever the denominator is
    nonzero.
    This is \cite[eqs.~(19) and~(S7)]{Malz2024Preparation}, with the
    canonical form of \cite{Cirac2017MPDO_AnnPhys}.
\end{definition}

\begin{definition}[Fixed-point state of a family of pairs]
    \label{def:ldp_pair_family_fixed_point}
    \lean{MPSTensor.bntWeight, MPSTensor.ghzAmplitude,
        MPSTensor.nonNormalFixedPointState, MPSTensor.nonNormalApproxState,
        MPSTensor.pairIsometry, MPSTensor.ghzState, Matrix.tensorPower}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, thm:ldpolar_tensor_decomposition}
    Let $\omega_1,\dots,\omega_b\in\mathbb C^D\otimes\mathbb C^D$ be any vectors,
    $\alpha\in\mathbb C^b$ any coefficients, and $M\ge0$. The state
    $\ket{\Omega'}=\sum_j\alpha_j\bigotimes_{k=1}^M\ket{\omega_j}_{R_kL_{k+1}}$ is
    formed as in (\ref{eq:ldp_non_normal_fixed_point}), and for a tensor $A$ and a
    blocking length $q$, with $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition}
    for the $q$-site blocked map of $A$, the approximating state is
    $V^{\otimes M}\ket{\Omega'}/\lVert V^{\otimes M}\ket{\Omega'}\rVert$, and zero when
    the denominator vanishes. For multiplicity coefficients $\mu_{j,k}$ put
    $\beta_j=\sum_k\mu_{j,k}^N$ and $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$.
    Let $W\colon\mathbb C^b\to\mathbb C^D\otimes\mathbb C^D$,
    $\ket{j}\mapsto\ket{\omega_j}$, be the map whose $j$-th column is $\omega_j$,
    let $X^{\otimes M}$ denote the $M$-fold tensor power of a matrix $X$, with entries
    $(X^{\otimes M})_{p,s}=\prod_{k=1}^M X_{p_k,s_k}$, and put
    $\ket{\chi_M}=\sum_j\alpha_j\ket{j}^{\otimes M}\in(\mathbb C^b)^{\otimes M}$.
    Definition~\ref{def:ldp_non_normal_fixed_point} is the instance in which
    $\omega_j$ is the pair state of the normal block $A_j$ embedded in the bond space
    of $A$, $\alpha=\alpha^{(N)}$, and $N=qM$; none of these three identifications is
    imposed here.
\end{definition}

\begin{theorem}[GHZ form of the fixed-point state]
    \label{thm:ldp_ghz_form}
    \lean{MPSTensor.isIsometry_pairIsometry, Matrix.IsIsometry.tensorPower,
        MPSTensor.nonNormalFixedPointState_eq_tensorPower_mulVec_ghzState,
        MPSTensor.nonNormalFixedPointState_norm_sq}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, def:ldp_pair_family_fixed_point}
    With $W$, $X^{\otimes M}$ and $\chi_M$ as in
    Definition~\ref{def:ldp_pair_family_fixed_point}, formed with
    $\alpha=\alpha^{(N)}$, if $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$, then
    $W$ and $W^{\otimes M}$ are isometries and
    $\ket{\Omega'}=W^{\otimes M}\ket{\chi_M}$, up to the
    regrouping of the legs $R_kL_{k+1}$. If moreover $M\geq1$ and
    $(\beta_1,\dots,\beta_b)\neq0$, then $\braket{\Omega'}{\Omega'}=1$.
    This is the preparation of \cite[eq.~(19)]{Malz2024Preparation}. The same
    holds for the state of Definition~\ref{def:ldp_pair_family_fixed_point} with
    $\alpha=\alpha^{(N)}$ for any orthonormal pairs $\omega_j$. The source asserts
    the orthogonality of its pairs after \cite[eq.~(S7)]{Malz2024Preparation},
    citing the canonical form; here it is a hypothesis
    \cite{gap:mswc24_ghz_form_local_orthogonality}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:prep_pair_product_norm}
    The $(j,j')$ entry of $W^\dagger W$ is $\braket{\omega_j}{\omega_{j'}}$, and the
    entries of $(W^{\otimes M})^\dagger W^{\otimes M}$ factor over the sites,
    \begin{align*}
        \bigl((W^{\otimes M})^\dagger W^{\otimes M}\bigr)_{s,t}
        =\prod_{k=1}^M(W^\dagger W)_{s_k,t_k}=\delta_{st}.
    \end{align*}
    Expanding
    $\ket{j}^{\otimes M}$ in the product basis gives
    $W^{\otimes M}\ket{j}^{\otimes M}=\bigotimes_k\ket{\omega_j}$, and summing with the
    weights $\alpha^{(N)}_j$ gives $\ket{\Omega'}$. Overlaps of the product states factor
    over the bonds,
    $\braket{\Omega_j}{\Omega_{j'}}=\braket{\omega_j}{\omega_{j'}}^M=\delta_{jj'}$, so
    $\braket{\Omega'}{\Omega'}=\sum_j\lvert\alpha^{(N)}_j\rvert^2=1$.
\end{proof}

\begin{theorem}[Pairs with disjoint supports are orthogonal]
    \label{thm:ldp_pair_disjoint_orthogonal}
    \lean{MPSTensor.inner_pair_eq_zero_of_disjoint}
    \leanok
    \uses{def:ldp_non_normal_fixed_point}
    Let $S,S'\subseteq\{0,\dots,D-1\}$ be disjoint, and let
    $\omega,\omega'\in\C^D\otimes\C^D$ satisfy $\omega_{(a,c)}=0$ for $a\notin S$
    and $\omega'_{(a,c)}=0$ for $a\notin S'$. Then
    $\braket{\omega}{\omega'}=0$. The embedded pairs of distinct blocks in
    Definition~\ref{def:ldp_non_normal_fixed_point} are supported in this way, so
    this gives the off-diagonal half of the orthogonality
    $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$; the unit norms and the
    identification of the supports are not supplied here
    \cite{gap:mswc24_ghz_form_local_orthogonality}.
\end{theorem}

\begin{proof}
    \leanok
    Every term of the sum vanishes: if $a\in S$ then $a\notin S'$ and
    $\omega'_{(a,c)}=0$, and otherwise $\omega_{(a,c)}=0$, so
    \begin{align*}
        \braket{\omega}{\omega'}
        =\sum_{(a,c)}\overline{\omega_{(a,c)}}\,\omega'_{(a,c)}=0.
    \end{align*}
\end{proof}

\begin{theorem}[Approximation error, general case]
    \label{thm:ldp_approximation_error_general}
    \uses{def:ldp_non_normal_fixed_point, def:correlation_length}
    With the notation of Definition~\ref{def:ldp_non_normal_fixed_point},
    including the approximating state $\widetilde\phi_N$ defined there,
    $\xi_{\mathrm{diag}}=\max_j\xi_j$ the largest correlation length of the
    blocks $A_j$, $0<\gamma<1/2$, and $q=o(N)$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi_{\mathrm{diag}}}\right).
        \label{eq:ldp_approximation_error_general}
    \end{align}
    This is \cite[Lemma~1$'$(ii)]{Malz2024Preparation}.
\end{theorem}
