\chapter{Preparation of Matrix Product States by Log-Depth Circuits}
\label{ch:log_depth_preparation}

A translation-invariant matrix product state on $N$ sites generated by a
normal tensor is approximated by blocking $q$ sites, replacing the positive
part of the polar decomposition of the blocked tensor by its renormalization
fixed point, and retaining the isometric part \cite{Malz2024Preparation}. The fixed point is a product of
nearest-neighbor entangled pairs, and the isometric part factorizes into
isometries with bounded bond dimension sequentially or along a binary
tree. Choosing $q=O(\log(N/\varepsilon))$ yields a quantum circuit of
depth $O(\log(N/\varepsilon))$ with error $\varepsilon$, whereas no circuit of
depth $o(\log N)$ achieves small error.

Throughout, $A$ is a tensor with physical dimension $d$ and bond dimension
$D$, $\E_A$ is its transfer map (Definition~\ref{def:transfer_map}), and
\begin{align}
    \ket{\phi_N(A)}
     & =\sum_{i_1,\dots,i_N=0}^{d-1}\Tr\left(A^{i_1}\cdots A^{i_N}\right)
    \ket{i_1\cdots i_N}
    \label{eq:ldp_ti_mps}
\end{align}
is the translation-invariant vector it generates, with
$\ket{\phi_N(A)}=\ket{V^{(N)}(A)}$ in the notation of
Definition~\ref{def:mpv_state}. For unit vectors $\phi,\psi$, the
error is $\varepsilon(\phi,\psi)=1-\lvert\braket{\phi}{\psi}\rvert$.

\section{The approximating state}

Write $\mathcal B_q$ for the physical matrix of the $q$-blocked tensor of $A$
(Definitions~\ref{def:blocked_tensor} and~\ref{def:ldpolar_physical_matrix})
and $\mathcal B_q=VP$ for its polar decomposition
(Theorem~\ref{thm:ldpolar_tensor_decomposition}).
In this section $A$ is normal (Definition~\ref{def:normal}) and in the gauge
\begin{align}
    \sum_{i=0}^{d-1}(A^i)^\dagger A^i
     & =\Id,
     &
    \E_A(\rho)
     & =\rho,
     &
    \rho
     & >0,
     &
    \Tr\rho
     & =1,
    \label{eq:ldp_normal_gauge}
\end{align}
so that $\E_A^{\,n}(X)\to\Tr(X)\,\rho$ as $n\to\infty$
\cite[eq.~(5)]{Malz2024Preparation}. Under normality, words of length $L_0$
span $\MN{D}$, so $\E_A$ is strongly irreducible by
Theorem~\ref{thm:wielandt_proposition_three}: the peripheral spectrum is
$\{1\}$, the positive fixed point $\rho$ is unique up to scale, and all
remaining eigenvalues have modulus strictly less than $1$.

\begin{lemma}[Convergence of the positive part]
    \label{lem:ldp_positive_part_converges}
    \lean{MPSTensor.tendsto_polarPosTensor_blockTensor_of_isNormal,
        MPSTensor.tendsto_polarPosTensor_of_tendsto_transferMap,
        MPSTensor.tendsto_polarPos_physicalMatrix_of_tendsto_transferMap}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor,
        thm:prep_fixed_point_tensor_psd, thm:prep_transfer_fixed_point_tensor,
        thm:prep_blocked_transfer_limit_normal}
    Let $P_q=(\mathcal B_q^\dagger\mathcal B_q)^{1/2}$. Then $P_q\to P_\infty$ as
    $q\to\infty$, where $P_\infty=(\sqrt\rho)^{\mathsf T}\otimes\Id$
    \cite[eq.~(8)]{Malz2024Preparation}.
    More generally, for any tensors $B_q$ whose transfer maps satisfy
    $\E_{B_q}(X)\to\Tr(X)\,\rho$ for every $X$, where $\rho\ge0$, the positive
    parts of the physical matrices of $B_q$ converge to
    $(\sqrt\rho)^{\mathsf T}\otimes\Id$, and the corresponding tensors
    converge to $P_\infty$.
\end{lemma}
\begin{proof}
    \leanok
    The matrix of $\mathcal B_q^\dagger\mathcal B_q$ is a rearrangement of
    $\E_A^{\,q}$:
    $(\mathcal B_q^\dagger\mathcal B_q)_{(\alpha,\beta),(\alpha',\beta')}
        =\E_A^{\,q}(\ketbra{\beta'}{\beta})_{\alpha'\alpha}$.
    By normality and the gauge~(\ref{eq:ldp_normal_gauge}),
    $\E_A^{\,q}(X)\to\Tr(X)\rho$, so the rearranged matrices converge to
    $\rho^{\mathsf T}\otimes\Id$. The square root of
    $\rho^{\mathsf T}\otimes\Id$ is $(\sqrt\rho)^{\mathsf T}\otimes\Id$,
    since the latter is positive semidefinite and squares to the former, and
    the square root is continuous on positive semidefinite matrices.
\end{proof}

\begin{theorem}[Scalar lower bound for the square root]
    \label{thm:ldp_sqrt_scalar_lower_bound}
    \lean{CFC.algebraMap_sqrt_le_sqrt}
    \leanok
    Let $b$ be an element of a C$^*$-algebra and $c\ge0$ a real number with
    $b\ge c\,\Id$. Then $\sqrt b\ge\sqrt c\,\Id$.
\end{theorem}
\begin{proof}
    \leanok
    The element $\sqrt c\,\Id$ is positive and its square is $c\,\Id\le b$, so
    operator monotonicity of the square root gives
    $\sqrt c\,\Id=\sqrt{c\,\Id}\le\sqrt b$.
\end{proof}

\begin{lemma}[One-sided bounds for the square root above a scalar floor]
    \label{lem:ldp_sqrt_floor_bounds}
    \lean{CFC.sq_sqrt_add_algebraMap, CFC.sqrt_sub_sqrt_le_algebraMap_of_le,
        CFC.sqrt_sub_sqrt_le_algebraMap_of_le'}
    \leanok
    Let $a,b$ be positive elements of a C$^*$-algebra, and let $c,s\ge0$ be
    real numbers with $b\ge c\,\Id$. Write $\varepsilon=\lVert a-b\rVert$.
    \begin{enumerate}
        \item If $\varepsilon\le2s\sqrt c+s^2$, then $\sqrt a-\sqrt b\le s$.
        \item If $s\le\sqrt c$ and $\varepsilon+s^2\le2s\sqrt c$, then
              $\sqrt b-\sqrt a\le s$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:ldp_sqrt_scalar_lower_bound}
    For every real $t$,
    $(\sqrt b+t)^2=b+2t\sqrt b+t^2$. By
    Theorem~\ref{thm:ldp_sqrt_scalar_lower_bound}, $\sqrt b\ge\sqrt c$, and
    $-\varepsilon\le a-b\le\varepsilon$. For the first bound,
    $(\sqrt b+s)^2\ge b+2s\sqrt c+s^2\ge b+\varepsilon\ge a$. For the second,
    $\sqrt b-s\ge\sqrt c-s\ge0$ and
    $(\sqrt b-s)^2\le b-2s\sqrt c+s^2\le a+\varepsilon+s^2-2s\sqrt c\le a$.
    Operator monotonicity of the square root gives $\sqrt a\le\sqrt b+s$ and
    $\sqrt b-s\le\sqrt a$.
\end{proof}

\begin{theorem}[H\"older bound for the square root]
    \label{thm:ldp_sqrt_holder}
    \lean{CFC.norm_sqrt_sub_sqrt_le, CFC.sqrt_sub_sqrt_le_algebraMap}
    \leanok
    Let $a,b$ be positive elements of a C$^*$-algebra. Then
    $\lVert\sqrt a-\sqrt b\rVert\le\lVert a-b\rVert^{1/2}$.
    This is the operator inequality quoted from Bhatia in
    \cite[Supplemental Material, eq.~(26)]{Piroli2021LOCC}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:ldp_sqrt_floor_bounds}
    Write $\varepsilon=\lVert a-b\rVert$ and $s=\sqrt\varepsilon$. The first
    bound of Lemma~\ref{lem:ldp_sqrt_floor_bounds} with $c=0$ gives
    $\sqrt a-\sqrt b\le s$, since $\varepsilon=s^2$. Exchanging $a$ and $b$
    gives $\sqrt b-\sqrt a\le s$. Since $\sqrt a-\sqrt b$ is self-adjoint and
    lies between $-s$ and $s$, its norm is at most $s$.
\end{proof}

\begin{theorem}[Square root near a positive definite element]
    \label{thm:ldp_sqrt_lipschitz}
    \lean{CFC.norm_sqrt_sub_sqrt_le_div}
    \leanok
    Let $a,b$ be positive elements of a C$^*$-algebra with $b\ge c\,\Id$ for
    a real number $c>0$. Then
    $\lVert\sqrt a-\sqrt b\rVert\le\lVert a-b\rVert/\sqrt c$.
    For matrices and the Frobenius norm, the corresponding bound with the
    smallest eigenvalue of $\sqrt b$ in place of $\sqrt c$ is
    \cite[App.~B3, eqs.~(S27)--(S31)]{Murota2026ExactLogDepth}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_sqrt_holder, lem:ldp_sqrt_floor_bounds}
    Write $\varepsilon=\lVert a-b\rVert$. If $\varepsilon>c$, the bound
    $\lVert\sqrt a-\sqrt b\rVert\le\sqrt\varepsilon$ of
    Theorem~\ref{thm:ldp_sqrt_holder} suffices, since
    $\sqrt\varepsilon\le\varepsilon/\sqrt c$. Otherwise let
    $s=\varepsilon/\sqrt c\le\sqrt c$, so that $2s\sqrt c=2\varepsilon$ and
    $s^2\le\varepsilon$. Both hypotheses of
    Lemma~\ref{lem:ldp_sqrt_floor_bounds} hold, so
    $\sqrt b-s\le\sqrt a\le\sqrt b+s$, that is,
    $\lVert\sqrt a-\sqrt b\rVert\le s$.
\end{proof}

\begin{theorem}[Lipschitz constant of the square root at an invertible positive element]
    \label{thm:ldp_sqrt_lipschitz_constant}
    \lean{IsStrictlyPositive.exists_norm_sqrt_sub_sqrt_le}
    \leanok
    Let $b$ be a positive invertible element of a C$^*$-algebra. There is a
    real number $K\ge0$ such that
    $\lVert\sqrt a-\sqrt b\rVert\le K\lVert a-b\rVert$ for every positive
    element $a$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_sqrt_lipschitz}
    If the algebra is zero, take $K=0$. Otherwise the spectrum of $b$ is a
    compact subset of $(0,\infty)$, so $b\ge c\,\Id$ for some real $c>0$, and
    Theorem~\ref{thm:ldp_sqrt_lipschitz} gives the bound with
    $K=1/\sqrt c$.
\end{proof}

\begin{theorem}[Rate of convergence of the positive part]
    \label{thm:ldp_positive_part_rate}
    \lean{MPSTensor.lt_exp_neg_div_correlationLength,
        MPSTensor.exists_norm_transferMap_pow_sub_le,
        MPSTensor.exists_norm_gram_blockTensor_sub_le,
        MPSTensor.exists_norm_polarPos_blockTensor_sub_le}
    \leanok
    \uses{def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}), let
    $\lambda_2\in\C$ satisfy $\lvert\mu\rvert\le\lvert\lambda_2\rvert$ for every
    eigenvalue $\mu\ne1$ of $\E_A$, let $\xi$ be the correlation length at
    $\lambda_2$ of Definition~\ref{def:correlation_length}, and let
    $0<\gamma<1$. Write $r=e^{-\gamma/\xi}$, where $-\gamma/\xi$ is read as $0$
    when $\xi=0$; thus $r=\lvert\lambda_2\rvert^{\gamma}$ when $\lambda_2\ne0$,
    and $r=1$ when $\lambda_2=0$. Then $a<r$ for every real number $a<1$ with
    $a\le\lvert\lambda_2\rvert$. There are constants $C,K,K'$ depending only
    on $A$, $\rho$, $\lambda_2$, and $\gamma$ such that
    $\lVert\E_A^{\,n}(X)-\Tr(X)\rho\rVert\le Cr^n\lVert X\rVert$
    for all $n$ and $X$, and
    \begin{align*}
        \lVert\mathcal B_q^\dagger\mathcal B_q-\rho^{\mathsf T}\otimes\Id\rVert
         & \le Kr^q, \\
        \lVert P_q-P_\infty\rVert
         & \le K'r^q
    \end{align*}
    for all $q$.
    In \cite[Supplemental Material, eq.~(26)]{Piroli2021LOCC} the bound on
    $P_q-P_\infty$ comes from Theorem~\ref{thm:ldp_sqrt_holder}, which halves
    the exponent. This gives the rate of
    \cite[eq.~(34)]{Piroli2021LOCC} and the range $0<\gamma<1/2$ of
    \cite[Lemma~1 and eq.~(S9)]{Malz2024Preparation}. The Lipschitz bound at
    the positive definite limit $P_\infty^2$ gives the range $0<\gamma<1$.
    This step is \cite[App.~B3, eqs.~(S27)--(S32)]{Murota2026ExactLogDepth}
    for diagonalizable transfer matrices, and eqs.~(S33)--(S38) there extend
    it to every normal tensor at each correlation length larger than $\xi$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_sqrt_lipschitz_constant}
    Let $a<1$ with $a\le\lvert\lambda_2\rvert$. If
    $0<\lvert\lambda_2\rvert<1$, then
    $r=\lvert\lambda_2\rvert^{\gamma}>\lvert\lambda_2\rvert\ge a$,
    because $\gamma<1$. Otherwise $\lambda_2=0$ and $r=1$, or
    $\lvert\lambda_2\rvert\ge1$ and $r=\lvert\lambda_2\rvert^{\gamma}\ge1$;
    in both cases $r\ge1>a$.

    Write $\E_\infty(X)=\Tr(X)\rho$. The eigenvalues $z$ of
    $\E_A-\E_\infty$ other than $0$ are eigenvalues of $\E_A$ different from
    $1$, so $\lvert z\rvert\le\lvert\lambda_2\rvert$, and
    $\lvert z\rvert<1$ by normality. Taking $a=\lvert z\rvert$ gives
    $\lvert z\rvert<r$. Hence the spectral radius of
    $\E_A-\E_\infty$ is less than $r$, and
    $\E_A^{\,n}-\E_\infty=(\E_A-\E_\infty)^n$ gives the first bound for
    $n\ge1$. For $n=0$ it reads
    $\lVert X-\E_\infty(X)\rVert\le C\lVert X\rVert$, which holds once
    $C\ge1+\lVert\E_\infty\rVert$; enlarge $C$ accordingly.

    The transfer map of the $q$-site blocked tensor is $\E_A^{\,q}$, so the
    entries of $\mathcal B_q^\dagger\mathcal B_q$ are entries of $\E_A^{\,q}$
    applied to matrix units:
    $(\mathcal B_q^\dagger\mathcal B_q)_{(\alpha,\beta),(\alpha',\beta')}
        =\E_A^{\,q}(\ketbra{\beta'}{\beta})_{\alpha'\alpha}$. In the same way
    $(\rho^{\mathsf T}\otimes\Id)_{(\alpha,\beta),(\alpha',\beta')}
        =\Tr(\ketbra{\beta'}{\beta})\,\rho_{\alpha'\alpha}$. Hence every entry
    of $\mathcal B_q^\dagger\mathcal B_q-\rho^{\mathsf T}\otimes\Id$ is an
    entry of $\E_A^{\,q}(X)-\Tr(X)\rho$ at a matrix unit $X$, and is at most
    a fixed multiple of $Cr^q$ by the first bound. Bounding the operator norm
    by the sum of the moduli of the finitely many entries gives
    $\lVert P_q^2-P_\infty^2\rVert
        =\lVert\mathcal B_q^\dagger\mathcal B_q-\rho^{\mathsf T}\otimes\Id\rVert
        \le Kr^q$. Since
    $P_\infty^2=\rho^{\mathsf T}\otimes\Id$ is positive definite,
    Theorem~\ref{thm:ldp_sqrt_lipschitz_constant} gives $L\ge0$ with
    $\lVert P_q-P_\infty\rVert\le LKr^q$, so $K'=LK$.
\end{proof}

\begin{definition}[Approximating state]
    \label{def:ldp_approximating_state}
    \lean{MPSTensor.approximatingTensor, MPSTensor.mpv_approximatingTensor,
        MPSTensor.mpv_approximatingTensor_norm_sq}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor,
        thm:prep_mpv_fixed_point_tensor, lem:mpv_rotate_physical,
        thm:ldpolar_tensor_injective, thm:prep_pair_product_norm}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} and $N=qM$, the
    \emph{approximating state} is
    $\ket{\widetilde\phi_N}=\ket{\phi_M(VP_\infty)}/\lVert\phi_M(VP_\infty)\rVert$.
    By Lemma~\ref{lem:mpv_rotate_physical} and
    Theorem~\ref{thm:prep_mpv_fixed_point_tensor}, for $M\ge1$,
    $\ket{\phi_M(VP_\infty)}=V^{\otimes M}\bigotimes_k\ket{\omega}_{R_kL_{k+1}}$.
    When $\mathcal B_q$ is injective, $V$ is an isometry and
    $\lVert\phi_M(VP_\infty)\rVert=1$, so no normalization is needed.
    This is \cite[eqs.~(9) and~(10)]{Malz2024Preparation}.
\end{definition}

\begin{lemma}[A unitary implementing the isometry]
    \label{lem:ldp_unitary_implementing_isometry}
    \lean{Matrix.exists_mem_unitaryGroup_apply_embedding_eq,
        MPSTensor.embeddedPairState,
        MPSTensor.mpv_approximatingTensor_eq_sum_embeddedPairState,
        MPSTensor.exists_unitary_mpv_approximatingTensor}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, thm:ldpolar_tensor_injective,
        def:ldp_approximating_state, def:prep_pair_product_state}
    Suppose $\mathcal B_q$ is injective, and let $\iota$ be an injective map
    from the pairs $(\gamma,\delta)$ with $0\le\gamma,\delta<D$ to the
    computational basis of $(\C^d)^{\otimes q}$. Then there is a unitary $U$
    on $(\C^d)^{\otimes q}$ with $U\ket{\iota(\gamma,\delta)}=V\ket{\gamma,\delta}$
    for all $\gamma,\delta$, and for every $M\ge1$
    \begin{align}
        \ket{\widetilde\phi_N}
         & =\Bigl(\bigotimes_{k=1}^M U_k\Bigr)
        \sum_{c}\Omega(c)\bigotimes_{k=1}^M\ket{\iota(c_k)},
        \label{eq:ldp_unitary_preparation}
    \end{align}
    where $c$ runs over the configurations $c_k=(l_k,r_k)$ and $\Omega$ is the
    pair product state~(\ref{eq:prep_pair_product}). The source takes $D=d$,
    so that $L$ and $R$ below are single sites and the remaining central sites
    carry $\ket{0}$; the padding of the legs for general $D$ is this chapter's
    extension. For $d\ge2$,
    $r=\lceil\log_dD\rceil$, $q\ge2r$, and $(\C^d)^{\otimes q}=L\otimes C\otimes R$,
    where $L$ and $R$ are the first and last $r$ sites of the block and
    $C$ denotes the remaining $q-2r$ sites, embed $\C^D$ into $L$ and $R$
    as the span of the first $D$ computational basis vectors, and set
    $\iota(\gamma,\delta)=\ket{\gamma}_L\ket{0}_C\ket{\delta}_R$ with
    $\ket{0}_C=\ket{0}^{\otimes(q-2r)}$. The right-hand side
    of~(\ref{eq:ldp_unitary_preparation}) then evaluates to
    $\bigl(\bigotimes_kU_k\bigr)\bigotimes_k\bigl(\ket{\omega}_{R_kL_{k+1}}\ket{0}_{C_k}\bigr)$
    \cite[eqs.~(10) and~(11)]{Malz2024Preparation}.
\end{lemma}

\section{Exact factorizations of the isometry}

\begin{theorem}[Sequential factorization of a matrix product isometry]
    \label{thm:ldp_sequential_factorization_general}
    \lean{MPSPreparation.exists_isometric_chain_of_eq_mul, MPSPreparation.eval_pairEmbed}
    \leanok
    Let $A_1,\dots,A_q$, $q\ge1$, be tensors of bond dimension $D$, one on each
    site, let $G$ be a $D^2\times D^2$ matrix, and let
    $V\colon\C^D\otimes\C^D\to(\C^d)^{\otimes q}$ be an isometry with matrix
    elements
    $\bra{i_1\cdots i_q}V\ket{x}
        =\sum_{\alpha,\beta}(A_1^{i_1}\cdots A_q^{i_q})_{\alpha\beta}\,
        G_{(\alpha,\beta),x}$.
    Then there are integers $b_0=1$, $b_q=D^2$ and $b_p\le D^2$, and families
    of $b_{p-1}\times b_p$ matrices $X_p^i$, $p=1,\dots,q$, with
    $\sum_i(X_p^i)^\dagger X_p^i=\Id_{b_p}$ for every $p$, such that
    $\bra{i_1\cdots i_q}V\ket{x}=(X_1^{i_1}X_2^{i_2}\cdots X_q^{i_q})_{1x}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_sweep_matrix_remainder, thm:seqanc_support_propagation}

    $\bigl(j^{\mathsf T}(\Id_D\otimes A_1^{i_1})\cdots(\Id_D\otimes A_q^{i_q})G\bigr)_x$
    with $j=\sum_\gamma\ket{\gamma}\otimes\ket{\gamma}$. Apply
    Theorem~\ref{thm:seqanc_sweep_matrix_remainder} to the row matrix of $j$ and the

    (\ref{eq:seqanc_sweep_inner_product}), $V^\dagger V=C^\dagger C$, so $C$ is an

\end{proof}

\begin{theorem}[Sequential factorization]
    \label{thm:ldp_sequential_factorization}
    \lean{MPSPreparation.exists_isometric_chain_polarIsoMatrix}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition}
    Let $A_1,\dots,A_q$, $q\ge1$, be tensors of bond dimension $D$, one on each
    site, suppose their blocked tensor $B$ with
    $B^{i_1\cdots i_q}=A_1^{i_1}\cdots A_q^{i_q}$ is injective, and let $B=VP$
    be its polar decomposition as in
    Theorem~\ref{thm:ldpolar_tensor_decomposition}. For equal tensors $A_p=A$,
    $B$ is the blocked tensor $\mathcal B_q$. There are integers
    $D'_1=D^2$, $D'_{q+1}=1$, and $D'_k\le D^2$, and matrices
    $W_k^i\in\C^{D'_{k+1}\times D'_k}$ for $k=1,\dots,q$ and $i=1,\dots,d$, with
    \begin{align}
        \sum_{i=1}^d\bigl(W_k^i\bigr)^\dagger W_k^i
         & =\Id_{D'_k},
        \label{eq:ldp_sequential_factorization_isometry}             \\
        \bra{i_1\cdots i_q}V\ket{x}
         & =\bigl(W_q^{i_1}W_{q-1}^{i_2}\cdots W_1^{i_q}\bigr)_{1x},

    \end{align}

    identified with $\C^D\otimes\C^D$. This is the matrix product form of
    \cite[eqs.~(13)--(15)]{Malz2024Preparation}, where the isometry carrying the
    input is written $\widetilde C=V_1$. The block length $q\ge1$ is implicit
    in the source, which draws at least one site; the empty block $q=0$ is
    excluded, see~\cite{gap:mswc24_sequential_factorization_positive_block_length}.
\end{theorem}

\begin{theorem}[Uniqueness of the polar decomposition]
    \label{thm:ldp_polar_uniqueness}
    \lean{Matrix.polarPos_eq_of_eq_mul, Matrix.polarSupport_eq_of_eq_mul,
        Matrix.polarIso_eq_of_eq_mul,
        Matrix.mul_eq_self_of_conjTranspose_mul_self_eq,
        Matrix.polarPos_eq_of_mul_self_eq, Matrix.polarSupport_eq_of_range_eq}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldpolar_support}
    Let $M=WQ$ with $Q\ge0$, and suppose $W^\dagger W=E$ for the orthogonal
    projector $E$ onto the range of $Q$. Then $Q=(M^\dagger M)^{1/2}$, $E$ is the
    support projector of $M$, and $W$ is the partial isometry of the polar
    decomposition of $M$. In particular, every positive semidefinite $Q$ with
    $Q^2=M^\dagger M$ is the positive part of $M$, and every orthogonal projector
    with the range of that positive part is the support projector.
\end{theorem}
\begin{proof}
    \leanok
    Since $EQ=Q$ and $WE=W$, $M^\dagger M=QE\,Q=Q^2$, so $Q$ is the positive
    square root of $M^\dagger M$. Two orthogonal projectors with the same range
    coincide, which identifies $E$. Finally $(V-W)Q=M-M=0$ for the partial
    isometry $V$ of $M$, and both $V$ and $W$ are fixed by $E$, whose range lies
    in the range of $Q$; hence $V=W$.
\end{proof}

\begin{theorem}[Tree factorization]
    \label{thm:ldp_tree_factorization}
    \lean{MPSTensor.treeTensor, MPSTensor.treePosTensor,
        MPSTensor.treeIsoMatrix, MPSTensor.blockKron,
        MPSTensor.blockTensor_pairRegroupEquiv,
        MPSTensor.blockKron_polarSupportMatrix_mul_physicalMatrix_blockTensor,
        MPSTensor.blockTensor_eq_rotatePhysical_treeIsoMatrix,
        MPSTensor.physicalMatrix_blockTensor_eq_treeIsoMatrix_mul,
        MPSTensor.conjTranspose_blockKron_polarIsoMatrix_mul_self,
        MPSTensor.conjTranspose_treeIsoMatrix_mul_self,
        MPSTensor.polarIsoMatrix_blockTensor_eq_treeIsoMatrix,
        MPSTensor.polarPosTensor_blockTensor_eq_treePosTensor,
        MPSTensor.polarSupportMatrix_blockTensor_eq_treeSupportMatrix}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition}
    Put $T_0=A$ and, for $j\ge1$, let $\mathcal B^{(j)}=V^{(j)}P_j$ be the polar
    decomposition of the two-site blocked map of $T_{j-1}$, with
    $V^{(j)\dagger}V^{(j)}=\Pi_j$ the support projector of $P_j$, and let $T_j$
    be the tensor of $P_j$ (Definition~\ref{def:ldpolar_tensor}). Then for
    $q=2^k$ with $k\ge1$,
    \begin{align}
        \mathcal B_q
         & =\Bigl(V^{(1)}\Bigr)^{\otimes2^{k-1}}
        \Bigl(V^{(2)}\Bigr)^{\otimes2^{k-2}}\cdots V^{(k)}\,P_k.
        \label{eq:ldp_tree_factorization}
    \end{align}
    Here $V^{(1)}$ maps $\C^{D^2}$ to $(\C^d)^{\otimes2}$ and each $V^{(j)}$ with
    $j\ge2$ maps $\C^{D^2}$ to $\C^{D^2}\otimes\C^{D^2}$; the Kronecker powers
    act on the regrouping of the $q$ sites into $2^{k-1}$ neighbouring pairs,
    then pairs of pairs, and so on. Each layer
    $\bigl(V^{(j)}\bigr)^{\otimes m}$ is a partial isometry with initial
    projector $\Pi_j^{\otimes m}$. The product of the layers equals $V$, the
    positive part $P_k$ equals $P_q$, and the support projector of $P_k$ is the
    support projector of $\mathcal B_q$, so the product of the layers is a
    partial isometry with initial space the range of $P_k$. The
    factorization (\ref{eq:ldp_tree_factorization}) is
    \cite[eq.~(16)]{Malz2024Preparation}, where the tree is said to act ``to the
    same effect'' as blocking directly; the identification with the polar
    decomposition of the blocked map is left implicit there and is supplied
    here.
\end{theorem}

\section{Approximation error}

For a normal tensor $A$, let $\lambda_2$ be an eigenvalue of $\E_A$ of
largest modulus other than the Perron eigenvalue $1$. By the remark after
(\ref{eq:ldp_normal_gauge}), $\lvert\lambda_2\rvert<1$. The
\emph{correlation length} of $A$ is the quantity
$\xi=-1/\log\lvert\lambda_2\rvert$ of Definition~\ref{def:correlation_length}
at this $\lambda_2$. It is finite, it is positive exactly when
$\lambda_2\neq0$, and $\lvert\lambda_2\rvert=e^{-1/\xi}$ in that case.
As in the previous section, $A$ is in the gauge~(\ref{eq:ldp_normal_gauge})
throughout this section. The estimates below use $\lambda_2$ only through the
bound $\lvert\mu\rvert\le\lvert\lambda_2\rvert$ for the eigenvalues $\mu\ne1$
of $\E_A$. When $\lambda_2=0$, so that $\xi=0$, they are applied with any
$\lambda_2'$ satisfying $0<\lvert\lambda_2'\rvert<1$ in place of $\lambda_2$,
and $\xi$ is read as the correlation length at $\lambda_2'$.

\begin{lemma}[Overlap of the positive part with the fixed point]
    \label{lem:ldp_positive_part_overlap}
    \lean{MPSTensor.exists_abs_one_sub_norm_mpvOverlap_polarPosTensor_le_mul,
        MPSTensor.exists_abs_one_sub_norm_mpvOverlap_polarPosTensor_le,
        MPSTensor.exists_norm_mpvOverlap_polarPosTensor_sub_one_le,
        MPSTensor.exists_norm_trace_prod_range_transferMatrix_sub_one_le,
        MPSTensor.exists_norm_mpvState_sq_le}
    \leanok
    \uses{def:prep_fixed_point_tensor, thm:ldpolar_blocked_transfer,
        def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, and let $0<\gamma<1$.
    Then, for $N=qM$,
    \begin{align}
        \Bigl\lvert1-\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\bigr\rvert\Bigr\rvert
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi}\right).
        \label{eq:ldp_positive_part_overlap}
    \end{align}
    for $q\ge1$ \cite[Supplemental Material, eq.~(S29)]{Piroli2021LOCC}, quoted as
    \cite[eq.~(S9)]{Malz2024Preparation}. The sources state this for
    $0<\gamma<1/2$; the range $0<\gamma<1$ follows from the Lipschitz bound of
    Theorem~\ref{thm:ldp_positive_part_rate}. Explicitly, there is a constant $C>0$
    depending only on $A$, $\rho$, and $\gamma$ such that
    $\bigl\lvert1-\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\rvert\bigr\rvert
        \le Cy\,e^{Cy}$ for all $q\ge0$ and $M\ge1$, where $y=M\,e^{-\gamma q/\xi}$.
    The same bound holds for the overlap itself:
    $\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle-1\bigr\rvert\le Cy\,e^{Cy}$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:ldp_positive_part_rate}
    By Theorem~\ref{thm:ldp_positive_part_rate},
    $\lVert P_q-P_\infty\rVert\le K_2e^{-\gamma q/\xi}$.

    Let $\tau_q$ and $\tau_\infty$ be the transfer matrices of the mixed
    transfer maps of $P_q$ and of $P_\infty$ against $P_\infty$. Then
    $\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle=\Tr\tau_q^M$,
    $1=\Tr\tau_\infty^M$, $\tau_\infty^2=\tau_\infty$, and
    $\delta=\lVert\tau_q-\tau_\infty\rVert\le K_3e^{-\gamma q/\xi}$, since
    $\tau_q-\tau_\infty$ depends linearly on $P_q-P_\infty$. Writing
    $\tau_q^M-\tau_\infty^M=\sum_{k<M}\tau_q^k(\tau_q-\tau_\infty)\tau_\infty^{M-1-k}$
    and bounding $\lVert\tau_q^k\rVert\le c+\lVert\tau_q^k-\tau_\infty^k\rVert$
    with $c\ge\max(1,\lVert\tau_\infty\rVert)$, induction on $M$ gives
    $\lVert\tau_q^M-\tau_\infty^M\rVert\le c\bigl((1+c\delta)^M-1\bigr)
        \le c\,Mc\delta\,e^{Mc\delta}$.
    Since $\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle-1=\Tr\tau_q^M-\Tr\tau_\infty^M$,
    this bounds the overlap itself, and hence also its modulus, which proves the
    explicit bounds. When $y\le1$ it is at most $Ce^{C}y$.
    When $y>1$, the overlap is at most
    $\lVert\phi_M(P_q)\rVert=\lVert\phi_N(A)\rVert$, and these norms are
    bounded because $\lVert\phi_N(A)\rVert^2=\Tr\E_A^{\,N}$ converges to $1$.
\end{proof}

\begin{theorem}[Block lengths below the injectivity lengths]
    \label{thm:ldp_injectivity_cutoff}
    \lean{Real.exp_neg_mul_div_eq_pow,

    bound follows from $\varepsilon\leq1\leq Cy\leq Cy\,e^{Cy}$.
\end{proof}

\begin{theorem}[Powers of a perturbed idempotent without first-order term]
    \label{thm:ldp_idempotent_second_order}
    \lean{IsIdempotentElem.norm_trace_add_pow_sub_le_of_le,
        IsIdempotentElem.norm_trace_add_pow_sub_le}
    \leanok
    Let $e$ be an idempotent of a normed ring, $f=1-e$, and let $Z$ satisfy
    $eZe=0$ and $\lVert fZe\rVert,\lVert eZf\rVert,\lVert fZf\rVert\le z\le1/2$.
    Let $\Tr$ be additive with $\Tr(xy)=\Tr(yx)$ and
    $\lvert\Tr x\rvert\le K\lVert x\rVert$. Then for $M\ge2$,
    \begin{align}
        \bigl\lvert\Tr(e+Z)^M-\Tr e\bigr\rvert
         & \le K\bigl(\lVert e\rVert+3\lVert f\rVert\bigr)\,2Mz^2\,e^{2Mz^2}.
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Put $T=e+Z$, $W=fZf$, $\beta=\lVert fZe\rVert$, $\beta'=\lVert eZf\rVert$,
    $\zeta=\lVert W\rVert$, and $\rho=1+\beta\beta'/(1-\zeta)$. For $P\in\{e,f\}$
    the blocks $x_M=PT^Me$ and $y_M=PT^Mf$ satisfy
    $x_{M+1}=x_M+y_M\,fZe$ and $y_{M+1}=x_M\,eZf+y_MW$, because $eTe=e$.
    Induction on $M$ gives $\lVert eT^Me-e\rVert\le\lVert e\rVert(\rho^M-1)$ and
    $\lVert eT^Mf\rVert\le\lVert e\rVert\beta'\rho^M/(1-\zeta)$, and also
    $\lVert fT^Me\rVert\le\lVert f\rVert\beta(\rho^M-\zeta^M)/(1-\zeta)$ and
    $\lVert fT^Mf-fW^M\rVert\le\lVert f\rVert\beta\beta'(\rho^M-\zeta^M)/(1-\zeta)^2$.
    Since $\Tr x=\Tr(exe)+\Tr(fxf)$, it follows that $\Tr T^M-\Tr e$ is the sum of
    the traces of $eT^Me-e$, of $fT^Mf-fW^M$, and of $fW^M$, the last of norm at
    most $\lVert f\rVert\zeta^M\le\lVert f\rVert z^2$. With $z\le1/2$ one has
    $\rho\le1+2z^2$ and $\beta\beta'/(1-\zeta)^2\le4z^2$, and
    $(1+2z^2)^M-1\le2Mz^2e^{2Mz^2}$.
\end{proof}

\begin{theorem}[Overlap of the positive part with the fixed point, to second order]
    \label{thm:ldp_positive_part_overlap_second_order}
    \lean{MPSTensor.exists_one_sub_norm_mpvOverlap_polarPosTensor_le_sq,
        MPSTensor.exists_norm_mpvState_sq_sub_norm_mpvOverlap_sq_le,
        MPSTensor.transferMatrix_fixedPointTensor_mul_mul,
        MPSTensor.inner_sqrtWeightLM, MPSTensor.exists_norm_map_polarPosTensor_sub_le}
    \leanok
    \uses{thm:ldp_idempotent_second_order, thm:ldp_positive_part_rate,
        def:prep_fixed_point_tensor, thm:ldpolar_blocked_transfer}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, and let $0<\gamma<1$. There is $C>0$ such that for all $q$ and
    all $M\ge2$ with $Cy<1$, where $y=M\,e^{-2\gamma q/\xi}$,
    \begin{align}
        1-\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\bigr\rvert
         & \le Cy\,e^{Cy}.
        \notag
    \end{align}
    For $M=1$ there is $K$ with
    $\lVert\phi_1(P_q)\rVert^2
        -\lvert\bra{\phi_1(P_\infty)}\phi_1(P_q)\rangle\rvert^2\le Ke^{-2\gamma q/\xi}$
    for all $q$. These bounds are of second order in $e^{-\gamma q/\xi}$.
    Lemma~\ref{lem:ldp_positive_part_overlap}, the estimate of
    \cite{Piroli2021LOCC} quoted as \cite[eq.~(S9)]{Malz2024Preparation}, is
    of first order, and so is the
    bound of \cite[App.~B4, eq.~(S52)]{Murota2026ExactLogDepth}, which is
    linear in the residual. The second-order bounds are new here.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_unit_vector_deficit}
    By Theorem~\ref{thm:ldp_positive_part_rate}, every linear function of the
    tensor $P_q$ is within $K'e^{-\gamma q/\xi}$ of its value at $P_\infty$. Let
    $\tau_q$ and $\tau_\infty$ be as in the proof of
    Lemma~\ref{lem:ldp_positive_part_overlap}, so that the overlap is
    $\Tr\tau_q^M$. The idempotent $\tau_\infty$ is the transfer matrix of the
    rank-one map $X\mapsto\Tr(X)\rho$; hence $\tau_\infty\tau_q\tau_\infty
        =\alpha\tau_\infty$ with
    $\alpha=\Tr\sum_iP_q^i\rho P_\infty^{i\dagger}=\braket{v_\infty}{v_q}$, where
    $v_q$ and $v_\infty$ are the vectors $(P^i\sqrt\rho)_i$. Both are unit
    vectors, because $\E_{P_q}=\E_A^q$ fixes $\rho$ and $\Tr\rho=1$. Hence
    $1-\lvert\alpha\rvert\le1-\operatorname{Re}\alpha
        =\tfrac12\lVert v_q-v_\infty\rVert^2=O(e^{-2\gamma q/\xi})$, and
    $\lvert1-\alpha\rvert\le\lVert v_q-v_\infty\rVert$. Write
    $\tau_q=\alpha(\tau_\infty+Z)$. Then $\tau_\infty Z\tau_\infty=0$ and
    $\lVert Z\rVert=O(e^{-\gamma q/\xi})$, so
    Theorem~\ref{thm:ldp_idempotent_second_order} bounds
    $\lvert\Tr(\tau_\infty+Z)^M-1\rvert$ by a multiple of $y\,e^{c y}$, and
    $\lvert\alpha\rvert^M\ge1-M(1-\lvert\alpha\rvert)$ is at least $1$ minus a
    multiple of $y$.
    For $M=1$, the vector $\phi_1(P_\infty)$ is a unit vector, and
    $\lVert u\rVert^2-\lvert\braket wu\rvert^2\le\lVert u-w\rVert^2$ for every
    unit vector $w$; here
    $\lVert\phi_1(P_q)-\phi_1(P_\infty)\rVert=O(e^{-\gamma q/\xi})$, since
    $\phi_1$ is linear in the tensor.
\end{proof}

\begin{theorem}[Approximation error, normal case]
    \label{thm:ldp_approximation_error_normal}
    \lean{MPSTensor.exists_approximationError_le_mul,
        MPSTensor.exists_approximationError_le, MPSTensor.normalizedMPVState,
        MPSTensor.approximatingMPVState,
        MPSTensor.exists_abs_norm_mpvState_sq_sub_one_le,
        MPSTensor.exists_norm_transferMatrix_pow_sub_le,
        MPSTensor.one_sub_norm_inner_smul_inv_norm_le_two_mul}
    \leanok
    \uses{thm:ldp_positive_part_overlap_second_order, def:ldp_approximating_state,
        thm:ldpolar_blocked_state, thm:ldpolar_tensor_injective,
        thm:ldp_injectivity_cutoff, def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, let $\ket{\phi_N}$ be the
    normalized vector (\ref{eq:ldp_ti_mps}), and let $0<\gamma<1$. For
    $N=qM$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)
         & =O\!\left(\frac Nq\,e^{-2\gamma q/\xi}\right)
        \label{eq:ldp_approximation_error_normal}
    \end{align}
    for $q\ge1$.
    Explicitly, there is a constant $C>0$ depending only on $A$, $\rho$, and $\gamma$
    such that $\varepsilon(\widetilde\phi_N,\phi_N)\le Cy\,e^{Cy}$ for all $q\ge0$
    and $M\ge1$, where $y=M\,e^{-2\gamma q/\xi}$.
    The rate $e^{-\gamma q/\xi}$ for $0<\gamma<1/2$ is
    \cite[Lemma~1, eq.~(17), and Lemma~1$'$(i)]{Malz2024Preparation}, and
    \cite[App.~B4, eqs.~(S52) and~(S53)]{Murota2026ExactLogDepth} gives the
    rate $e^{-q/\xi}$. The rate $e^{-2\gamma q/\xi}$ for every $0<\gamma<1$
    and every normal tensor is new here. Numerically, the error decays as
    $e^{-\gamma q/\xi}$ with $\gamma\approx2$
    \cite[Supplemental Material, discussion of Fig.~S1]{Malz2024Preparation},
    \cite[eq.~(4)]{Wei2026ParallelSequential}.
\end{theorem}
\begin{proof}
    \leanok
    Write $c_N=\lVert\phi_N(A)\rVert$. For $q$ at least the injectivity length
    of $A$, the blocked tensor is injective, $V$ is an isometry, and
    $\ket{\widetilde\phi_N}=V^{\otimes M}\ket{\phi_M(P_\infty)}$ is
    normalized. Since $\ket{\phi_N(A)}=V^{\otimes M}\ket{\phi_M(P_q)}$ by
    Theorem~\ref{thm:ldpolar_blocked_state}, this gives
    $c_N\braket{\widetilde\phi_N}{\phi_N}
        =\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle=:a$, and
    $c_N=\lVert\phi_M(P_q)\rVert$. With $t=\lvert a\rvert/c_N\le1$,
    \begin{align}
        \varepsilon=1-t\le1-t^2=\frac{c_N^2-\lvert a\rvert^2}{c_N^2},
        \notag
    \end{align}
    which is at most $2(c_N^2-\lvert a\rvert^2)$ once $c_N^2\ge1/2$. The
    normalization satisfies $\lvert c_N^2-1\rvert=\lvert\Tr\E_A^{\,N}-1\rvert
        \le Ke^{-\gamma' N/\xi}$ for every $0<\gamma'<1$; we use it at $\gamma'=\gamma$.
    For $M=1$, Theorem~\ref{thm:ldp_positive_part_overlap_second_order} bounds
    $c_N^2-\lvert a\rvert^2$ directly. For $M\ge2$,
    $c_N^2-\lvert a\rvert^2\le\lvert c_N^2-1\rvert+2\max(0,1-\lvert a\rvert)$,
    with $\lvert c_N^2-1\rvert\le Ke^{-2\gamma q/\xi}$ because $N\ge2q$, and
    $1-\lvert a\rvert$ is bounded by
    Theorem~\ref{thm:ldp_positive_part_overlap_second_order}. The finitely many
    $q$ below the injectivity length, and the case $Cy\ge1$, are covered by
    $\varepsilon\le1$.
\end{proof}

\begin{definition}[Approximating state for blocks of prescribed lengths]
    \label{def:ldp_block_isometry_state}
    \lean{MPSTensor.blockIsometryState, MPSTensor.mpvFamily,
        MPSPreparation.blockOffset, MPSPreparation.blockSite,
        MPSPreparation.blockCfgEquiv, MPSTensor.blockIndexEquiv}
    \leanok
    \uses{def:ldp_approximating_state, thm:ldpolar_tensor_decomposition,
        def:prep_pair_product_state}
    Cut the ring of $N$ sites into $M\ge1$ consecutive blocks of lengths
    $\ell_1,\dots,\ell_M$ with $\sum_k\ell_k=N$, block $k$ consisting of the
    sites $o_k+1,\dots,o_k+\ell_k$ with $o_k=\sum_{i<k}\ell_i$. Let
    $\mathcal B_\ell=V_\ell P_\ell$ be the polar decomposition of the
    $\ell$-site blocked tensor, and let $\omega$ be a vector on
    $\C^D\otimes\C^D$. The state
    \begin{align}
        \ket{\psi_{\ell,\omega}}
         & =\Bigl(\bigotimes_{k=1}^MV_{\ell_k}\Bigr)
        \bigotimes_{k=1}^M\ket{\omega}_{R_kL_{k+1}}
        \label{eq:ldp_block_isometry_state}
    \end{align}
    has the coefficient
    $\sum_c\prod_k\bra{s_k}V_{\ell_k}\ket{c_k}\,\Omega(c)$ at the configuration
    $s=(s_1,\dots,s_M)$ read block by block, where $c_k=(l_k,r_k)$ and $\Omega$ is
    the pair product state~(\ref{eq:prep_pair_product}) of $\omega$. If the
    $q$-site blocked tensor is injective, $\ell_k=q$ for every $k$, and $\omega$
    is the pair of a fixed point $\sigma\ge0$ with $\Tr\sigma=1$, it is the
    approximating state of Definition~\ref{def:ldp_approximating_state}.
\end{definition}

\begin{lemma}[Norm and overlap for blocks of prescribed lengths]
    \label{lem:ldp_block_isometry_state_overlap}
    \lean{MPSTensor.sum_star_blockIsometryState,
        MPSTensor.norm_blockIsometryState,
        MPSTensor.inner_blockIsometryState_mpvState,
        MPSTensor.mpv_eq_mpvFamily_blockTensor,
        MPSTensor.sum_mpvFamily_mul_star_mpv,
        MPSTensor.mpvFamily_rotatePhysical,
        MPSPreparation.prod_ofFn_blockSite,
        Matrix.IsIsometry.sum_star_mul_prod,
        Matrix.IsIsometry.sum_star_mul_tensorPower}
    \leanok
    \uses{def:ldp_block_isometry_state, thm:ldpolar_tensor_injective,
        thm:ldpolar_tensor_decomposition, thm:prep_pair_product_norm,
        thm:prep_mpv_fixed_point_tensor, thm:mpu_transfer_matrix_overlap}
    Suppose every blocked tensor $\mathcal B_{\ell_k}$ is injective. Then
    $\braket{\psi_{\ell,\omega}}{\psi_{\ell,\omega}}=\braket\omega\omega^M$, and
    for the pair $\omega$ of the fixed point
    \begin{align}
        \braket{\psi_{\ell,\omega}}{\phi_N(A)}
         & =\Tr\bigl(\tau_{\ell_1}\tau_{\ell_2}\cdots\tau_{\ell_M}\bigr),
        \label{eq:ldp_block_isometry_state_overlap}
    \end{align}
    where $\tau_\ell$ is the transfer matrix of the mixed transfer map
    $X\mapsto\sum_iP_\ell^iX(P_\infty^i)^\dagger$.
\end{lemma}
\begin{proof}
    \leanok
    An ordered product over the sites of the ring is the ordered product over
    the blocks of the products over their sites, so
    $\phi_N(A)(s)=\Tr(\mathcal B_{\ell_1}^{s_1}\cdots\mathcal B_{\ell_M}^{s_M})$,
    and $\mathcal B_{\ell_k}=V_{\ell_k}P_{\ell_k}$ gives
    $\ket{\phi_N(A)}=(\bigotimes_kV_{\ell_k})\ket{\phi_M(P_{\ell_1},\dots,P_{\ell_M})}$,
    the periodic state with the tensor $P_{\ell_k}$ at site $k$. Isometries
    $W_k$, one on each site, preserve inner products:
    $\sum_s\overline{((\bigotimes_kW_k)\psi)(s)}((\bigotimes_kW_k)\varphi)(s)
        =\braket\psi\varphi$, since
    $\sum_{s_k}\overline{(W_k)_{s_ka}}(W_k)_{s_kb}=\delta_{ab}$ factor by factor.
    This gives the norm by Theorem~\ref{thm:prep_pair_product_norm}, and reduces
    the overlap to
    $\sum_u\Tr(P_{\ell_1}^{u_1}\cdots P_{\ell_M}^{u_M})
        \overline{\Tr(P_\infty^{u_1}\cdots P_\infty^{u_M})}$ by
    Theorem~\ref{thm:prep_mpv_fixed_point_tensor}. Expanding the trace of
    $\tau_{\ell_1}\cdots\tau_{\ell_M}$ as a sum over cyclic paths of pairs of
    bond indices, with
    $(\tau_\ell)_{(j,i),(l,m)}=\sum_a(P_\ell^a)_{im}\overline{(P_\infty^a)_{jl}}$,
    and summing the two paths separately gives the same sum, as in
    Theorem~\ref{thm:mpu_transfer_matrix_overlap}.
\end{proof}

\begin{theorem}[Approximation error for blocks of unequal lengths]
    \label{thm:ldp_block_approximation_error}
    \lean{MPSTensor.exists_blockApproximationError_le_mul}
    \leanok
    \uses{thm:ldp_unequal_absolute_trace_second_order,
        thm:ldp_approximation_error_normal, def:ldp_block_isometry_state,
        lem:ldp_block_isometry_state_overlap, def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, and let $0<\gamma<1$. There is $C>0$, depending only on
    the tensor, its normalized fixed state and the chosen spectral bound
    and rate, such that every partition of $N$ sites into $M\ge1$ blocks
    of lengths $\ell_j\ge q$ with injective blocked tensors satisfies
    $\varepsilon(\psi_{\ell,\omega},\phi_N)\le CM e^{-2\gamma q/\xi}$.
    Here $\psi_{\ell,\omega}$ is the state of
    Definition~\ref{def:ldp_block_isometry_state} constructed from the
    fixed-point pair, and $\phi_N$ is the normalized target.
    This refines the first-order rate of
    \cite[Lemma~1$'$(i), eq.~(S11)]{Malz2024Preparation}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:ldp_unequal_absolute_trace_second_order,
        thm:ldp_positive_part_overlap_second_order,
        thm:ldp_approximation_error_normal,
        lem:ldp_block_isometry_state_overlap}
    Put $v=\phi_N(A)$, $c_N=\lVert v\rVert$ and
    $a=\langle\psi_{\ell,\omega},v\rangle$.
    The block isometries give $\lVert\psi_{\ell,\omega}\rVert=1$ and
    $a=\Tr(\tau_{\ell_0}\cdots\tau_{\ell_{M-1}})$.
    As in Theorem~\ref{thm:ldp_approximation_error_normal}, once
    $c_N^2\ge1/2$, normalization gives
    $\varepsilon\le2(c_N^2-|a|^2)$.
    Write $u=Me^{-2\gamma q/\xi}$.
    If $M\ge2$, Theorem~\ref{thm:ldp_unequal_absolute_trace_second_order}
    bounds $1-|a|$ by $Cu e^{Cu}$ in the small-error regime. The transfer
    gap gives $|c_N^2-1|\le K e^{-\gamma N/\xi}$; since
    $N\ge Mq\ge2q$, this is at most $K e^{-2\gamma q/\xi}$.
    If $M=1$, Theorem~\ref{thm:ldp_positive_part_overlap_second_order}
    bounds $c_N^2-|a|^2$ directly by $K e^{-2\gamma N/\xi}$, hence by
    $K e^{-2\gamma q/\xi}$.
    The first-order normalization estimate also yields $c_N^2\ge1/2$
    when a sufficiently large fixed multiple of $u$ is below one.
    The remaining regime is covered by $\varepsilon\le1$.
    Finally $Cu e^{Cu}\le Ce^C u$ for $u\le1$, giving the assertion
    after increasing the constant.
\end{proof}

\begin{theorem}[The approximating state in depth $O(q)$]

    There is $C$, depending only on $d$ and $D$, such that the following
    holds. Let $A$ be a tensor, let $\omega$ be a unit vector on
    $\C^D\otimes\C^D$, and cut the ring of $N\ge1$ sites into $M\ge1$ blocks
    of lengths $3D\le\ell_k\le L$ for which every blocked tensor is injective.
    Then the state $\ket{\psi_{\ell,\omega}}$ of
    Definition~\ref{def:ldp_block_isometry_state} is prepared in depth at
    most $CL$. In particular, for $\sigma\ge0$ with $\Tr\sigma=1$, a block
    length $q\ge3D$ for which the $q$-site blocked tensor is injective, and
    $M\ge1$, the approximating state of
    Definition~\ref{def:ldp_approximating_state} on $N=Mq$ sites is prepared

\begin{proof}
    \leanok
    \uses{lem:ldp_unitary_implementing_isometry,
        thm:ldp_sequential_factorization, lem:seqanc_staircase,
        thm:qc_bounded_unitary_two_site_gates, lem:qc_parallel_circuits,
        lem:qc_placed_operators}
    The case $D=0$ is trivial. If $d\ge2$ then $D<d^D$. If $d\le1$ and
    $D\ge1$, injectivity of the blocked tensor gives $D^2\le d^q$, which
    forces $d=D=1$. In either case $D\le d^D$, so each bond index can be
    encoded in $D$ sites. Encode the two bond indices
    of a block in its first and last $D$ sites, and write $\ket{l,0,r}$ for the block with $l$ on the first $D$ sites, $r$
    on the last $D$ sites, and $\ket0$ elsewhere. This defines an injective map $E$ from
    bond-pair configurations to physical configurations. Its restriction to
    block $k$ is the prescribed $\ket{l_k,0,r_k}$; its restriction to the
    pair window is $(r_k,l_{k+1})$. These pair-window restrictions, together
    with the zero values off the windows, characterize $E$.
    Within each layer the windows are disjoint, so their embedded operators
    commute and the layer's matrix coefficients factor over the windows.
    The all-zero product input has coefficient $\prod_i\delta_{s_i,0}$.
    For equal block length $q$, the site $(k,j)$ is $kq+j$, and reading a
    configuration block by block agrees with the uniform blocking equivalence.
    A first layer of unitaries,
    one on the last $D$ sites of each block and the first $D$ sites of the
    next, sends $\ket{0\cdots0}$ to
    $\bigotimes_k\ket\omega_{R_kL_{k+1}}\ket0_{C_k}$
    \cite[eq.~(12)]{Malz2024Preparation}; by
    Theorem~\ref{thm:qc_bounded_unitary_two_site_gates} it has constant depth.
    On each block $k$, in parallel, a unitary $U_k$ with
    $U_k\ket{l,0,r}=V_{\ell_k}\ket{l,r}$ is applied, as in
    Lemma~\ref{lem:ldp_unitary_implementing_isometry}. SWAP
    gates move the first $D$ sites next to the last ones in $O(q)$ gates.
    By Theorem~\ref{thm:ldp_sequential_factorization}, $V_\ell$ is the ordered
    product of $\ell$ isometries, each from the bond space into the bond space
    and one site; each is extended to a unitary on the bond register of $2D$
    sites and one fresh site, and the register moves one site to the left at
    each step, so the staircase consists of $q-2D$ unitaries on $2D+1$ sites
    followed by one unitary on the first $2D$ sites
    \cite[paragraph ``The sequential-RG circuit'']{Malz2024Preparation}. Each
    of them has bounded depth by
    Theorem~\ref{thm:qc_bounded_unitary_two_site_gates}, so $U_k$ has depth
    $O(\ell_k)\le O(L)$, and circuits on disjoint blocks run in the same layers.
    A configuration contributes to the pair layer applied to $\ket{0\cdots0}$
    only if it carries a pair on every window and $\ket0$ elsewhere, which
    identifies the output with~(\ref{eq:ldp_block_isometry_state}).
\end{proof}

\begin{theorem}[Preparation in depth $O(\log(N/\varepsilon))$]
    \label{thm:ldp_depth_upper_bound}

    \leanok
    Let $A$ be normal. There is a constant $c$ depending only on $A$ such that,
    for every $N\ge2$ and $0<\varepsilon\le1$ for which the periodic state
    $\ket{\phi_N(A)}$ is nonzero, a state $\ket{\psi_N}$ with
    $\varepsilon(\psi_N,\phi_N)\le\varepsilon$ is prepared in depth at most
    $c\log(N/\varepsilon)$. If $D\ge1$, there is $N_0$ such that
    $\ket{\phi_N(A)}\neq0$ for every $N\ge N_0$, so for $N\ge N_0$ the condition can be dropped. This is
    \cite[eq.~(1)]{Malz2024Preparation}; the source normalizes
    $\ket{\phi_N}=\ket{\phi_N(A)}/c_N$ with $c_N>0$, and for some normal tensors
    $\ket{\phi_2(A)}=0$ \cite{gap:mswc24_depth_upper_bound_nonzero_state}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:normalized_primitive_gauge, lem:ldp_normalized_state_rescaling,
        thm:ldp_block_approximation_error, thm:ldp_approximating_state_depth,
        thm:ldp_short_chain_preparation, lem:ldp_periodic_state_nonzero,
        lem:ldp_block_isometry_state_overlap, thm:wordspan_blk_inj_succ,
        lem:isNBlkInjective_iff_blockTensor_isInjective,
        thm:gauge_invariance_normal, thm:normality_rescaling_invariance,
        thm:is_normal_tensor_of_is_normal_left_canonical, thm:eigenvalue_bound,
        thm:primitive_fixed_point_trace_normalization, thm:transfer_channel}
    Bring $A$ into the gauge~(\ref{eq:ldp_normal_gauge}) as in the proof of
    Theorem~\ref{thm:ldp_depth_upper_bound_divisible}, with fixed point $\sigma$,
    a bound $t<1$ on the moduli of the eigenvalues other than $1$, and
    $r=-\log t$; the error is unchanged by
    Lemma~\ref{lem:ldp_normalized_state_rescaling}. Let $K$ be the constant of
    Theorem~\ref{thm:ldp_block_approximation_error} at $\gamma=1/2$, so the error
    is at most $KMe^{-rq}$, and let $L_0$ be a length at which the blocked
    tensor is injective. Put $a=1/r$, $b=\max(\log K,0)/r+L_0+3D+1$,
    $Q=a\log(N/\varepsilon)+b$ and $q=\lceil Q\rceil$, so $Q\le q<2Q$ and
    $Q\le(a+b/\log2)\log(N/\varepsilon)$.

    If $q\le N$, write $N=(M-1)q+q'$ with $M=\lfloor N/q\rfloor$ and
    $q'=q+(N\bmod q)$, so $q\le q'<2q$, and cut the ring into $M-1$ blocks of
    length $q$ followed by one of length $q'$, as in the Supplemental Material
    of \cite{Malz2024Preparation}, also used in its lower-bound proof of
    Theorem~1. All blocked tensors are
    injective since $q\ge L_0$, and
    Theorem~\ref{thm:ldp_block_approximation_error} gives error at most
    $KMe^{-rq}\le KNe^{-rQ}=Ke^{-rb}\varepsilon\le\varepsilon$. By
    Theorem~\ref{thm:ldp_approximating_state_depth} with $L=2q$ the depth is at
    most $2Cq\le4CQ$.

    If $N<q$, then $N<Q$. If $N\ge L_0+3D$, the state $\ket{\phi_N}$ is prepared
    exactly in depth at most $C'N\le C'Q$ by
    Theorem~\ref{thm:ldp_short_chain_preparation}; otherwise $N$ is bounded by a
    constant depending only on $A$, and the same theorem prepares $\ket{\phi_N}$
    exactly in depth at most $K'\le(K'/\log2)\log(N/\varepsilon)$. In every case
    the depth is at most
    $c\log(N/\varepsilon)$ with $c=(4C+C')(a+b/\log2)+K'/\log2$. The second
    statement follows from Lemma~\ref{lem:ldp_periodic_state_nonzero}.
\end{proof}

\begin{lemma}[Triangle step for the error]
    \label{lem:ldp_error_triangle}
    \lean{MPSPreparation.one_sub_norm_inner_le_norm_sub_smul_sq,
        MPSPreparation.exists_norm_sub_smul_sq_eq,
        MPSPreparation.one_sub_norm_inner_le_two_mul_add,
        Matrix.star_mulVec_dotProduct_mulVec_of_mem_unitary,
        Matrix.inner_eq_of_mulVec_eq, Matrix.norm_eq_of_mulVec_eq}
    \leanok
    For unit vectors $a$, $b$ in a complex inner product space and every
    scalar $c$ with $\lvert c\rvert=1$,
    $\varepsilon(a,b)\le\lVert a-cb\rVert^2/2$, with equality for some such
    $c$. For unit vectors $x$, $y$, $z$,
    $\varepsilon(x,z)\le2\bigl(\varepsilon(x,y)+\varepsilon(y,z)\bigr)$.
    A unitary matrix $U$ satisfies
    $\overline{(Ux)}\cdot(Uy)=\overline{x}\cdot y$ for coordinate vectors
    $x$, $y$, and so preserves inner products and norms.
\end{lemma}
\begin{proof}
    \leanok
    For unit vectors $a$, $b$ and $\lvert c\rvert=1$,
    $\lVert a-cb\rVert^2=2-2\operatorname{Re}(c\braket{a}{b})
        \ge2\varepsilon(a,b)$, with equality for
    $c=\overline{\braket{a}{b}}/\lvert\braket{a}{b}\rvert$ (any $c$ if
    $\braket{a}{b}=0$). Choose $c_1$, $c_2$ attaining equality for $(x,y)$ and
    $(y,z)$. Then
    $\lVert x-c_1c_2z\rVert\le\lVert x-c_1y\rVert+\lVert y-c_2z\rVert$, and
    $(s+t)^2\le2(s^2+t^2)$ gives
    $2\varepsilon(x,z)\le\lVert x-c_1c_2z\rVert^2
        \le4\bigl(\varepsilon(x,y)+\varepsilon(y,z)\bigr)$.
\end{proof}

\begin{corollary}[Normal states are transformed into each other in depth $O(\log(N/\varepsilon))$]
    \label{cor:ldp_normal_states_circuit}
    \uses{def:qc_local_circuit}
    \lean{MPSPreparation.exists_isLocalCircuitOfDepth_le_log_of_mpvState_ne_zero,
        MPSPreparation.exists_isLocalCircuitOfDepth_le_log}
    \leanok
    Let $A$ and $B$ be normal tensors with the same physical dimension $d$.
    There is a constant $c$ depending only on $A$ and $B$ such that, for every
    $N\ge2$ and $0<\varepsilon\le1$ for which $\ket{\phi_N(A)}$ and
    $\ket{\phi_N(B)}$ are nonzero, some local circuit $U$ of depth at most
    $c\log(N/\varepsilon)$ satisfies
    $\varepsilon(U\hat\phi_N(A),\hat\phi_N(B))\le\varepsilon$, where
    $\hat\phi_N(A)=\ket{\phi_N(A)}/\lVert\phi_N(A)\rVert$ and
    $\hat\phi_N(B)=\ket{\phi_N(B)}/\lVert\phi_N(B)\rVert$ are the normalized
    states. If both bond dimensions are at least
    $1$, there are $c$ and $N_0$ such that the same holds for every
    $N\ge N_0$ and $0<\varepsilon\le1$ without the nonvanishing condition.

    The paragraph ``Discussion and outlook'' of \cite{Malz2024Preparation}
    states that its results imply that states in the same phase are
    transformed into each other by a circuit of logarithmic depth. Normal
    states lie in the trivial phase of the classification without symmetry;
    the corollary covers only this case. Non-normal states and phases defined
    with a symmetry are not treated
    \cite{gap:mswc24_same_phase_circuit_normal_case}.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:ldp_depth_upper_bound, lem:ldp_periodic_state_nonzero,
        lem:qc_circuit_inverse_composition, lem:qc_preparation_from_zero,
        lem:ldp_error_triangle}
    By Theorem~\ref{thm:ldp_depth_upper_bound} with error $\varepsilon/4$ and
    Lemma~\ref{lem:qc_preparation_from_zero}, there are local circuits $U_A$,
    $U_B$ and unit vectors $\psi_A=aU_A\ket{0\cdots0}$,
    $\psi_B=bU_B\ket{0\cdots0}$ with $\varepsilon(\psi_A,\hat\phi_N(A))$ and
    $\varepsilon(\psi_B,\hat\phi_N(B))$ at most $\varepsilon/4$, of depths
    $T_A+2$ and $T_B+2$ with $T_A,T_B$ at most a constant times
    $\log(4N/\varepsilon)\le3\log(N/\varepsilon)$. Put $U=U_BU_A^\dagger$, a
    local circuit of depth $T_A+T_B+4$ by
    Lemma~\ref{lem:qc_circuit_inverse_composition}; the term $4$ is at most
    $(4/\log2)\log(N/\varepsilon)$. Then $U\psi_A=(a/b)\psi_B$ is a unit
    vector, and since $U$ is unitary,
    $\varepsilon(U\hat\phi_N(A),(a/b)\psi_B)=\varepsilon(\hat\phi_N(A),\psi_A)$.
    Lemma~\ref{lem:ldp_error_triangle} gives
    $\varepsilon(U\hat\phi_N(A),\hat\phi_N(B))\le2(\varepsilon/4+\varepsilon/4)
        =\varepsilon$. The second statement follows from
    Lemma~\ref{lem:ldp_periodic_state_nonzero}.
\end{proof}

\begin{lemma}[Rescaling changes the normalized state by a phase]
    \label{lem:ldp_normalized_state_rescaling}
    \lean{MPSPreparation.norm_inner_normalizedMPVState_of_mpv_eq}
    \leanok
    Let $A$ and $B$ be tensors with $\ket{\phi_N(B)}=\zeta^N\ket{\phi_N(A)}$ for

    $\ket{\phi_N(A)}$ (zero when $\ket{\phi_N(A)}=0$). Then every vector
    $\ket\psi$ on $N$ sites satisfies
    $\lvert\braket{\psi}{\hat\phi_N(B)}\rvert=\lvert\braket{\psi}{\hat\phi_N(A)}\rvert$,

\end{lemma}

    \leanok
    $\ket{\hat\phi_N(B)}=(\zeta^N/\lvert\zeta^N\rvert)\ket{\hat\phi_N(A)}$,
    and if $\ket{\phi_N(A)}=0$ both normalized states vanish.
\end{proof}

\begin{theorem}[Block length $a\log(N/\varepsilon)+b$ for every slope $a>\xi/2$]
    \label{thm:ldp_depth_upper_bound_slope}
    \uses{def:qc_local_circuit}
    \lean{MPSPreparation.exists_isPreparedInDepth_approximationError_le_of_slope}
    \leanok
    There is $C$, depending only on $d$ and $D$, such that the following holds.
    Let $B$ be a normal tensor in the gauge~(\ref{eq:ldp_normal_gauge}), with a
    positive definite fixed point $\sigma$ of trace one, and let $A$ be a tensor
    with $\ket{\phi_N(B)}=\zeta^N\ket{\phi_N(A)}$ for all $N$ and some
    $\zeta\neq0$. Let $0<t<1$ bound the moduli of the eigenvalues of $\E_B$ other
    than $1$, and let $\xi=-1/\log t$. For every $a>\xi/2$ there is $b\ge1$ such
    that for $0<\varepsilon\le1$, every $N\ge1$ and every $q$ dividing $N$ with
    $q\ge a\log(N/\varepsilon)+b$, a unit vector $\ket\psi$ with
    $\varepsilon(\psi,\phi_N(A))\le\varepsilon$ is prepared in depth at most $Cq$.
    The block length $q=\lceil2\xi(1+\eta)\ln N\rceil$ of
    \cite[paragraph after Lemma~1]{Malz2024Preparation} has slope $2\xi$, and
    \cite[App.~B4, eq.~(S54)]{Murota2026ExactLogDepth} takes
    $q=O(\xi\log(N/\varepsilon))$. The slopes $a>\xi/2$ are new here.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_approximating_state_depth,
        thm:ldp_approximation_error_normal,
        thm:wordspan_blk_inj_succ,
        lem:ldp_normalized_state_rescaling,
        lem:isNBlkInjective_iff_blockTensor_isInjective,
        lem:ldp_approximating_state_unit_norm,
        lem:ldp_block_length_threshold_error,
        lem:ldp_approximation_error_slope_threshold}
    Since $a>\xi/2$, the number $\gamma=\xi/(2a)$ satisfies $0<\gamma<1$, and
    Theorem~\ref{thm:ldp_approximation_error_normal} at this $\gamma$ gives $K$
    with $\varepsilon(\widetilde\phi_N,\phi_N(B))\le K\,(N/q)\,e^{-q/a}$, since
    $2\gamma/\xi=1/a$. Take $b=a\max(\log K,0)+L_0+3D+1$, with $L_0$ a length at
    which the blocked tensor is injective. Then $q\ge b$ makes the blocked
    tensor $\mathcal B_q$ injective by
    Theorem~\ref{thm:wordspan_blk_inj_succ} and
    Lemma~\ref{lem:isNBlkInjective_iff_blockTensor_isInjective}, and $q\ge3D$.
    Moreover $q/a\ge\log(N/\varepsilon)+\log K\ge\log(N/(q\varepsilon))+\log K$,
    so $K(N/q)e^{-q/a}\le\varepsilon$. The approximating state is a unit vector
    by Lemma~\ref{lem:ldp_approximating_state_unit_norm}, is prepared in depth at
    most $Cq$ by Theorem~\ref{thm:ldp_approximating_state_depth}, and has the
    same error against $\phi_N(A)$ as against $\phi_N(B)$ by
    Lemma~\ref{lem:ldp_normalized_state_rescaling}.
\end{proof}

\begin{theorem}[Preparation in depth $O(\log(N/\varepsilon))$, block length dividing $N$]

    dividing $N$ with
    $q\ge a\log(N/\varepsilon)+b$, a unit vector $\ket\psi$ with
    $\varepsilon(\psi,\phi_N)\le\varepsilon$ is prepared in depth at most $Cq$.

\begin{proof}
    \leanok

        lem:ldp_depth_log_window, lem:ldp_normal_gap,
        thm:normality_rescaling_invariance,
        thm:gauge_invariance_normal,

        thm:eigenvalue_bound,
        thm:primitive_fixed_point_trace_normalization,
        thm:transfer_channel}

    gauge~(\ref{eq:ldp_normal_gauge}) by
    Theorem~\ref{thm:normalized_primitive_gauge}, and the positive definite
    fixed point $\rho$ it provides is replaced by
    $\sigma=\rho/\Tr\rho$, still a fixed point of trace one, by
    Theorem~\ref{thm:primitive_fixed_point_trace_normalization}; they multiply
    $\ket{\phi_N(A)}$ by a nonzero scalar. The new tensor is normal by
    Theorems~\ref{thm:normality_rescaling_invariance}
    and~\ref{thm:gauge_invariance_normal}. By
    Theorem~\ref{thm:is_normal_tensor_of_is_normal_left_canonical} its transfer

\end{proof}

\begin{lemma}[Inserted transfer maps of products and adjoints]

    \begin{align}
        (X\otimes Y)(X'\otimes Y')
         & =XX'\otimes YY',
         &
        \E_{X\otimes Y}
         & =\E_X\,\E_Y,
        \notag                       \\
        \E_{\Id}
         & =\E_A^{\,L},
         &
        \E_{X^\dagger}(Z)
         & =\E_X(Z^\dagger)^\dagger.
        \notag
    \end{align}
    For an operator $O$ on $(\C^d)^{\otimes N}$ and the unnormalized vector
    $\ket{\phi_N(A)}$ of (\ref{eq:ldp_ti_mps}),

\end{lemma}

    \leanok

    $L_1$ and $L_2$, with $A^{ww'}=A^wA^{w'}$ and
    $\bra{ww'}X\otimes Y\ket{vv'}=\bra{w}X\ket{v}\bra{w'}Y\ket{v'}$; summing

    induction on $L$. The adjoint formula is the conjugate of the defining sum.
    Writing $\braket{w}{\phi_N(A)}=\Tr A^w$ and expanding
    $\sum_{w,w'}\overline{\Tr A^{w'}}\bra{w'}O\ket{w}\Tr A^w$ in matrix units

\end{proof}

    $(\C^d)^{\otimes L}$. For $L=1$ and injective $A$, this corresponds to the
    choice of transfer matrices in the proof of
    \cite[Lemma~2]{Malz2024Preparation}. If moreover
    $\Phi(Z^\dagger)=\Phi(Z)^\dagger$ for every $Z$, then $X$ can be chosen
    Hermitian.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:physical_observable_word_span_mul, lem:ldp_observable_transfer_products}
    Choose coefficients $c^{ab}_w$ with $\sum_wc^{ab}_wA^w=e_{ab}$. The operator
    with matrix elements $\bra{w'}X\ket{w}=c^{ab}_w\overline{c^{ce}_{w'}}$ gives
    $\E_X(Z)=e_{ab}Ze_{ec}=Z_{be}\,e_{ac}$. These maps span all linear maps on
    $\MN{D}$, and $X\mapsto\E_X$ is linear. If $\Phi$ preserves adjoints and
    $\E_X=\Phi$, then by
    Lemma~\ref{lem:ldp_observable_transfer_products} also
    $\E_{X^\dagger}=\Phi$, so the Hermitian part $(X+X^\dagger)/2$ has the

\begin{lemma}[Two windows on the ring]
    \label{lem:ldp_two_window_trace}
    \lean{MPSTensor.chainWindowOperator_apply,

        MPSTensor.chainWindowOperator_self, MPSTensor.chainWindowOperator_one,

    $m,n\ge0$ and $N=2L+m+n$, and let $X_1$ act on the sites $1,\dots,L$ and
    $Y_{L+m+1}$ on the sites $L+m+1,\dots,2L+m$ of a ring of $N$ sites. Then
    \begin{align}
        \bra{\phi_N(A)}X_1Y_{L+m+1}\ket{\phi_N(A)}
         & =\operatorname{tr}\bigl(\E_X\,\E_A^{\,m}\,\E_Y\,\E_A^{\,n}\bigr).
        \notag
    \end{align}
    The expectation in the normalized vector $\ket{\phi_N}$ is
    $\bra{\phi_N(A)}O\ket{\phi_N(A)}/\braket{\phi_N(A)}{\phi_N(A)}$.
\end{lemma}
\begin{proof}
    \leanok

    product of the identity on the sites before it, the operator on the window,
    and the identity on the sites after it. Hence $X_1Y_{L+m+1}$ is
    $X\otimes\Id\otimes Y\otimes\Id$ over the blocks of lengths $L,m,L,n$, and
    Lemma~\ref{lem:ldp_observable_transfer_products} gives the trace formula.
\end{proof}

\begin{lemma}[The limit correlator at an eigenvector]
    \label{lem:ldp_limit_correlator_eigenvalue}
    \lean{MPSTensor.limitCorrelator, MPSTensor.exists_limitCorrelator_eq_pow,
        MPSTensor.limitCorrelator_smul_smul, MPSTensor.limitCorrelator_zero_left,

    \leanok

    $\sum_i(A^i)^\dagger A^i=\Id$, let $\Tr\rho=1$, put $P(Z)=\Tr(Z)\,\rho$,

    are operators $X,Y$ on $(\C^d)^{\otimes L}$ such that for every $t\ge0$
    \begin{align}
        \Tr\Bigl(\E_X\bigl((\E_A-P)^t\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr)
         & =\lambda^t.
        \notag
    \end{align}
    When $\rho$ is the fixed point of the gauge (\ref{eq:ldp_normal_gauge}),
    the left side is the limit correlator (\ref{eq:ldp_limit_correlator}) at
    $t=s-L-1$; it is bilinear in $(X,Y)$. The inner insertion has zero
    trace, so centering it leaves the formula unchanged even at $t=0$. This is the choice of observables in
    the proof of \cite[Lemma~2]{Malz2024Preparation}, without the source's
    requirement that the one-point functions vanish.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:ldp_observable_transfer_surjective}
    Trace preservation gives $\lambda\Tr R=\Tr R$, so $\Tr R=0$, $P(R)=0$, and
    $(\E_A-P)^tR=\lambda^tR$. By
    Lemma~\ref{lem:ldp_observable_transfer_surjective} take
    $\E_Y(Z)=\Tr(Z)\,R$ and $\E_X(Z)=(Z_{ij}/R_{ij})\,\rho$ for an entry
    $R_{ij}\neq0$.
\end{proof}

\begin{lemma}[Hermitian observables for the second eigenvalue]
    \label{lem:ldp_hermitian_observables}
    \lean{MPSTensor.exists_hermitian_vector_functional_compl_pow,

    \leanok
    \uses{lem:ldp_limit_correlator_eigenvalue,
        lem:ldp_observable_transfer_surjective}

    $\sum_i(A^i)^\dagger A^i=\Id$, let $\rho$ be Hermitian with $\Tr\rho=1$,

    $\E_A$. Then there are Hermitian operators $\mathcal O,\mathcal O'$ on
    $(\C^d)^{\otimes L}$ of norm one, an integer $K\in\{1,2\}$ with $K=1$ when
    $\lambda$ is real, distinct complex numbers $\mu_1,\dots,\mu_K$ of modulus
    one and a nonzero $a\in\C^K$ such that for every $t\ge0$
    \begin{align}
        \Tr\Bigl(\E_{\mathcal O}\bigl((\E_A-P)^t

         & =\lvert\lambda\rvert^t\sum_{j=1}^Ka_j\mu_j^{\,t}.
        \notag
    \end{align}
    The inner insertion is traceless, so the displayed expression equals the

    of observables in the proof of \cite[Lemma~2]{Malz2024Preparation}
    by one with Hermitian observables: for
    non-real $\lambda$ the eigenvector is not Hermitian, and a Hermitian

\end{lemma}
\begin{proof}
    \leanok

    Theorem~\ref{thm:ldp_decaying_correlations}: a traceless Hermitian vector $v$ and a
    functional $\ell$ with $\ell(Z^\dagger)=\overline{\ell(Z)}$ are built from
    an eigenvector $R$ and its adjoint, the maps $Z\mapsto\Tr(Z)\,v$ and
    $Z\mapsto\ell(Z)\,\rho$ are realized by Hermitian observables through
    Lemma~\ref{lem:ldp_observable_transfer_surjective}, and scaling to norm one
    multiplies $a$ by a positive constant.
\end{proof}

\begin{lemma}[Finite-size corrections]
    \label{lem:ldp_finite_size_correction}
    \lean{MPSTensor.exists_compl_pow_bound,
        MPSTensor.mpvConnectedCorrelator_eq_of_compl,
        Complex.norm_div_sub_div_mul_div_sub_le,
        Matrix.norm_linearMap_trace_le_mul_norm,
        MPSTensor.exp_neg_div_correlationLength_le_norm_pow}
    \leanok
    \uses{lem:ldp_two_window_trace, thm:wielandt_proposition_three,
        def:correlation_length}
    Let $A$ be in the gauge (\ref{eq:ldp_normal_gauge}), with the products of
    $L\ge1$ matrices spanning $\MN{D}$, let $P(Z)=\Tr(Z)\,\rho$, and let
    $\lambda_2$ be an eigenvalue of $\E_A$ of largest modulus among those
    different from $1$.
    \begin{enumerate}
        \item For every $r>\lvert\lambda_2\rvert$ there is $C_r>0$ with
              $\lVert(\E_A-P)^n\rVert\le C_rr^n$ for all $n\ge0$.
        \item For $t,n\ge1$ and $N=2L+t+n$, the connected correlator
              $G_N(X,Y;L+t+1)$ of (\ref{eq:ldp_connected_correlator}) equals
              $(\alpha\beta+G_\infty+\delta_{XY})/(1+\delta)
                  -(\alpha+\delta_X)(\beta+\delta_Y)/(1+\delta)^2$, where
              $\alpha=\Tr\E_X(\rho)$, $\beta=\Tr\E_Y(\rho)$, $G_\infty$ is the
              limit correlator (\ref{eq:ldp_limit_correlator}) at
              separation $t$, and $\delta,\delta_X,\delta_Y,\delta_{XY}$ are
              traces of linear maps containing $(\E_A-P)^n$, $(\E_A-P)^N$ or a
              power of $\E_A-P$ of exponent at least $n$.
        \item If $\lvert z-1\rvert$, $\lvert n_{XY}-(ab+g)\rvert$,
              $\lvert n_X-a\rvert$ and $\lvert n_Y-b\rvert$ are at most
              $\varepsilon\le\tfrac12$ and $\lvert a\rvert,\lvert b\rvert,
                  \lvert g\rvert\le M$, then
              $\lvert n_{XY}/z-(n_X/z)(n_Y/z)-g\rvert\le4(M+2)^2\varepsilon$.
        \item The trace of a linear map on $\MN{D}$ is bounded by a constant
              depending only on $D$ times its operator norm, and
              $e^{-(s-1)/\xi}\le\lvert\lambda_2\rvert^t$ whenever $t+1\le s$ and
              $0<\lvert\lambda_2\rvert<1$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    A nonzero eigenvalue of $\E_A-P$ is an eigenvalue of $\E_A$ with a traceless
    eigenvector, hence different from $1$ by strong irreducibility
    (Theorem~\ref{thm:wielandt_proposition_three}), so the spectral radius of
    $\E_A-P$ is at most $\lvert\lambda_2\rvert$ and Gelfand's formula gives
    (1). Part (2) substitutes $\E_A^{\,k}=P+(\E_A-P)^k$ for $k\ge1$ into the
    trace formula of Lemma~\ref{lem:ldp_two_window_trace} and into the
    one-point functions and the normalization. Part (3) is elementary
    algebra with $\lvert1/z\rvert\le2$, and part (4) expands the trace in
    matrix units and uses $\xi=-1/\log\lvert\lambda_2\rvert$.
\end{proof}

\begin{theorem}[Exponentially decaying correlations]
    \label{thm:ldp_decaying_correlations}
    \lean{MPSTensor.exists_decayingCorrelations,
        MPSTensor.mpvConnectedCorrelator, MPSTensor.mpvExpectation,
        MPSTensor.chainWindowOperator}
    \leanok
    \uses{def:normal, def:correlation_length}
    Let $A$ be normal and in the gauge (\ref{eq:ldp_normal_gauge}), with
    correlation length $\xi>0$, let $L\ge1$ be a length such that the products
    of $L$ matrices of $A$ span $\MN{D}$ (one may take $L=1$ when $A$ is
    injective), and let $\ket{\phi_N}$ be the normalized vectors
    (\ref{eq:ldp_ti_mps}). For operators $X,Y$ on $(\C^d)^{\otimes L}$ let
    $X_1$ act on the sites $1,\dots,L$, let $Y_s$ act on the sites
    $s,\dots,s+L-1$, and put
    \begin{align}
        G_N(X,Y;s)
         & =\bra{\phi_N}X_1Y_s\ket{\phi_N}
        -\bra{\phi_N}X_1\ket{\phi_N}\bra{\phi_N}Y_s\ket{\phi_N}.
        \label{eq:ldp_connected_correlator}
    \end{align}
    There are Hermitian operators $\mathcal O,\mathcal O'$ on
    $(\C^d)^{\otimes L}$ of norm one, a constant $c>0$, an integer
    $K\in\{1,2\}$ and an integer $s_0\ge L+2$ such that for every $s\ge s_0$
    some $s'\in\{s,s+1,\dots,s+K-1\}$ satisfies, for every $N\ge3s'$,
    \begin{align}
        \bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
         & \ge c\,e^{-(s'-1)/\xi}.
        \label{eq:ldp_decaying_correlations}
    \end{align}
    One may take $K=1$ when $\lambda_2$ is real and $K=2$ otherwise. This is a
    Hermitian variant of the correlation estimate of
    \cite[Lemma~2]{Malz2024Preparation}, with the separation $s$ ranging over
    windows of $K$ consecutive values and the connected correlator. The
    source's own observables cannot be Hermitian when $\lambda_2$ is not real;
    for them the bound holds in modulus at every separation, while the printed
    real inequality fails when $\lambda_2$ is not real or is
    negative~\cite{gap:mswc24_decaying_correlations_windowed_connected}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:transfer_map, lem:ldp_observable_transfer_products,
        lem:ldp_observable_transfer_surjective, lem:ldp_two_window_trace,
        lem:ldp_hermitian_observables, lem:ldp_finite_size_correction,
        lem:ldp_vandermonde_window, thm:wielandt_proposition_three}
    With the inserted transfer maps $\E_X$ defined before
    Lemma~\ref{lem:ldp_observable_transfer_surjective}, and for operators
    $X,Y$ on consecutive blocks,
    \begin{align}
        \E_{\Id} & =\E_A^{\,L},
                 &
        \E_{X\otimes Y}
                 & =\E_X\,\E_Y.
        \notag
    \end{align}
    Expanding the traces in (\ref{eq:ldp_ti_mps})
    gives, for $L<s$ and $s+L-1\le N$,
    \begin{align}
        \bra{\phi_N(A)}X_1Y_s\ket{\phi_N(A)}
         & =\operatorname{tr}\bigl(\E_X\,\E_A^{\,s-L-1}\,\E_Y\,
        \E_A^{\,N-s-L+1}\bigr),
        \notag
    \end{align}
    the trace of a linear map on $\MN{D}$. Let $P(Z)=\Tr(Z)\,\rho$. In the
    gauge (\ref{eq:ldp_normal_gauge}), $\E_A^{\,n}=P+(\E_A-P)^n$ for $n\ge1$.
    An eigenvalue $\nu\neq0$ of $\E_A-P$ is an eigenvalue of $\E_A$ with a
    traceless eigenvector, so $\nu\neq1$ because $\E_A$ is strongly
    irreducible (Theorem~\ref{thm:wielandt_proposition_three}), and hence
    $\lvert\nu\rvert\le\lvert\lambda_2\rvert$. Fix $r$ with
    $\lvert\lambda_2\rvert<r<1$ and $r^2<\lvert\lambda_2\rvert$; Gelfand's
    formula gives $\lVert(\E_A-P)^n\rVert\le C_rr^n$. Replacing the last power
    $\E_A^{\,N-s-L+1}$ by $P$, and likewise in the normalization
    $\operatorname{tr}\E_A^{\,N}$ and in the one-point functions, changes
    $G_N(X,Y;s)$ by at most $C_{X,Y}\,r^{N-s-L+1}$, which is at most
    $C_{X,Y}\,r^{2(s-L-1)}$ when $N\ge3s$. The remaining term is
    \begin{align}
        G_\infty(X,Y;s)
         & =\Tr\Bigl(\E_X\bigl((\E_A-P)^{s-L-1}\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr).
        \label{eq:ldp_limit_correlator}
    \end{align}

    Let $R\neq0$ satisfy $\E_A(R)=\lambda_2R$. Since $\xi>0$,
    $\lambda_2\neq0$, and trace preservation gives $\Tr R=0$, so
    $(\E_A-P)^tR=\lambda_2^{\,t}R$; since $\E_A(Z^\dagger)=\E_A(Z)^\dagger$,
    also $(\E_A-P)^tR^\dagger=\bar\lambda_2^{\,t}R^\dagger$. If $\lambda_2$ is
    real, one of $R+R^\dagger$ and $i(R-R^\dagger)$ is a nonzero Hermitian
    eigenvector $H$; put $v=H$ and $\ell(Z)=\Tr(HZ)/\Tr(H^2)$, so that
    $\ell((\E_A-P)^tv)=\lambda_2^{\,t}$. If $\lambda_2$ is not real, $R$ and
    $R^\dagger$ are eigenvectors for distinct eigenvalues; take a linear
    functional $f$ with $f(R)=1$ and $f(R^\dagger)=0$, put $v=R+R^\dagger$ and
    $\ell(Z)=f(Z)+\overline{f(Z^\dagger)}$, so that
    $\ell((\E_A-P)^tv)=\lambda_2^{\,t}+\bar\lambda_2^{\,t}$. In both cases
    $v$ is Hermitian, $\ell(Z^\dagger)=\overline{\ell(Z)}$, and
    $\ell((\E_A-P)^tv)=\lvert\lambda_2\rvert^t\sum_{j=1}^Ka_j\mu_j^t$ with
    $K\le2$, distinct $\mu_j$ of modulus one and $a\neq0$.

    The maps $Z\mapsto\Tr(Z)\,v$ and $Z\mapsto\ell(Z)\,\rho$ preserve
    Hermitian matrices. By Lemma~\ref{lem:ldp_observable_transfer_surjective}
    they are $\E_Y$ and $\E_X$ for some $X,Y$, and since
    $\E_{X^\dagger}(Z)=\E_X(Z^\dagger)^\dagger$ the Hermitian parts of $X$ and
    $Y$ have the same maps. Then
    $G_\infty(X,Y;s)=\lvert\lambda_2\rvert^t\sum_ja_j\mu_j^t$ with $t=s-L-1$,
    and $X,Y\neq0$ because this sum does not vanish identically. Take
    $\mathcal O,\mathcal O'$ to be $X,Y$ scaled to norm one; this multiplies
    $a$ by a positive constant.

    The Vandermonde matrix $W=(\mu_j^{\,u})_{0\le u<K,\,1\le j\le K}$ is
    invertible because the $\mu_j$ are distinct. For every $t$ the vector with
    entries $\sum_ja_j\mu_j^{\,t+u}$, $0\le u<K$, is $W$ applied to
    $(a_j\mu_j^t)_j$, whose maximum norm equals that of $a$ because
    $\lvert\mu_j\rvert=1$. Hence some $u\in\{0,\dots,K-1\}$ satisfies
    \begin{align}
        \Bigl\lvert\sum_{j=1}^Ka_j\mu_j^{\,t+u}\Bigr\rvert
         & \ge c_0,
         &
        c_0
         & =\frac{\max_j\lvert a_j\rvert}{\lVert W^{-1}\rVert_\infty}>0,
        \label{eq:ldp_vandermonde_window}
    \end{align}
    so $\lvert G_\infty(\mathcal O,\mathcal O';s+u)\rvert\ge
        c_0\lvert\lambda_2\rvert^{t'}$ with $t'=t+u$. With $s'=s+u$ and
    $N\ge3s'$ the finite-size correction is at most
    $C\,r^{2t'}=C\,(r^2/\lvert\lambda_2\rvert)^{t'}\lvert\lambda_2\rvert^{t'}$,
    which is at most $\tfrac{c_0}{2}\lvert\lambda_2\rvert^{t'}$ after
    enlarging $s_0$, since $r^2<\lvert\lambda_2\rvert$. Finally
    $\lvert\lambda_2\rvert^{t'}=e^{-(s'-L-1)/\xi}\ge e^{-(s'-1)/\xi}$, which
    gives (\ref{eq:ldp_decaying_correlations}) with $c=c_0/2$.
\end{proof}

    \leanok
    \uses{thm:ldp_decaying_correlations, lem:ldp_expectation_overlap,
        thm:wielandt_proposition_three, lem:ldp_window_averages,
        thm:ldp_depth_lower_bound_gauge, lem:ldp_normal_gauge_reduction}
    We may assume that $A$ is in the gauge (\ref{eq:ldp_normal_gauge}). Since
    $A$ is normal, $\E_A$ is strongly irreducible
    (Theorem~\ref{thm:wielandt_proposition_three}), so its spectral radius
    $\lambda_1$ is a positive eigenvalue whose right eigenvector $\rho$ and
    whose eigenvector $\sigma$ for the dual map are positive definite. Replacing
    $A^i$ by $\lambda_1^{-1/2}\sigma^{1/2}A^i\sigma^{-1/2}$ gives
    (\ref{eq:ldp_normal_gauge}), with fixed point $\sigma^{1/2}\rho\sigma^{1/2}$
    scaled to unit trace. The scaling multiplies
    $\ket{\phi_N}$ by a positive constant and the conjugation leaves every trace
    in (\ref{eq:ldp_ti_mps}) unchanged, so $\phi_N/\lVert\phi_N\rVert$ is
    unchanged; the eigenvalues of $\E_A$ are divided by $\lambda_1$, which is
    the normalization $\lambda_1=1$ in which $\xi$ is computed.

    Take $L,\mathcal O,\mathcal O',c,K,s_0$ from
    Theorem~\ref{thm:ldp_decaying_correlations}, and suppose
    $\varepsilon(\phi_N,\psi_N)\le1/2$, that is
    $\lvert\braket{\phi_N}{\psi_N}\rvert\ge1/2$. Write $T=T_N$ and
    $\ket{\psi_N}=U\ket{\pi}$ with $U$ a local circuit of depth $T$ and
    $\ket{\pi}$ a product state. If $Z$ acts on an interval of sites, then
    $U^\dagger ZU$ acts on the interval enlarged by $T$ sites on each side, so
    expectations in $\ket{\psi_N}$ of products of operators whose enlarged
    intervals are pairwise disjoint factorize.

    Let $s_N=\max(s_0,2T+L+1)$ and let $s'\in\{s_N,\dots,s_N+K-1\}$ be given
    by the lemma; the window changes $s'$ by at most $K-1$, so still
    $s'=o(\log N)$ and $N\ge3s'$ for large $N$. Put $\Delta=s'+L+2T\le2s'$,
    $n=\lfloor N/\Delta\rfloor\ge N/(4s')$, and $b_k=1+(k-1)\Delta$ for
    $k=1,\dots,n$. With $e=\bra{\phi_N}\mathcal O_1\ket{\phi_N}$ and
    $e'=\bra{\phi_N}\mathcal O'_1\ket{\phi_N}$ let $\mathcal O^{(k)}$ be
    $\mathcal O-e$ on the sites $b_k,\dots,b_k+L-1$ and $\mathcal O'^{(k)}$ be
    $\mathcal O'-e'$ on the sites $b_k+s'-1,\dots,b_k+s'+L-2$. These have norm at
    most $2$, zero expectation in $\ket{\phi_N}$ by translation invariance,
    and $\bra{\phi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\phi_N}
        =G_N(\mathcal O,\mathcal O';s')$. Since $s'\ge2T+L+1$ and
    $\Delta\ge s'+L+2T$, the $2n$ enlarged intervals are pairwise disjoint,
    also around the ring.

    Two estimates are used. For Hermitian $Z$ and unit vectors $\phi,\psi$,
    with $\sigma_\chi(Z)=\lVert(Z-\langle Z\rangle_\chi)\chi\rVert$,
    \begin{align}
        \lvert\braket{\phi}{\psi}\rvert\,
        \bigl\lvert\langle Z\rangle_\phi-\langle Z\rangle_\psi\bigr\rvert
         & \le\sigma_\phi(Z)+\sigma_\psi(Z),
        \label{eq:ldp_expectation_overlap}
    \end{align}
    because
    $(\langle Z\rangle_\phi-\langle Z\rangle_\psi)\braket{\phi}{\psi}
        =\bra{\phi}(Z-\langle Z\rangle_\psi)\ket{\psi}
        -\bra{\phi}(Z-\langle Z\rangle_\phi)\ket{\psi}$. Secondly, the
    decomposition $\E_A^{\,n}=P+(\E_A-P)^n$ from the proof of
    Theorem~\ref{thm:ldp_decaying_correlations}, together with the fact that in
    the gauge (\ref{eq:ldp_normal_gauge}) the maps $\E_Z$ have norm bounded by
    a multiple of $\lVert Z\rVert$ independent of the size of the support,
    shows that the connected correlation in $\ket{\phi_N}$ of two operators of
    norm at most $4$ on intervals separated by $g$ and $g'$ sites around the
    ring is at most $C(r^g+r^{g'})$, with $C$ and $r<1$ depending only on $A$.
    Consecutive intervals $[b_k,b_k+s'+L-2]$ are separated by at least $2T$
    sites, and intervals $j$ apart by at least $(j-1)\Delta$, so every
    average of $n$ of these operators, or of their pair products, has
    variance at most $C_1/n$ in $\ket{\phi_N}$.

    For signs $\epsilon_k\in\{\pm1\}$ apply (\ref{eq:ldp_expectation_overlap})
    to $Z=\frac1n\sum_k\epsilon_k\mathcal O^{(k)}$. Its expectation in
    $\ket{\phi_N}$ is $0$; in $\ket{\psi_N}$ it is
    $\frac1n\sum_k\epsilon_kx_k$ with
    $x_k=\bra{\psi_N}\mathcal O^{(k)}\ket{\psi_N}\in\R$, and by factorization
    its variance there is at most $4/n$. Hence
    $\lvert\frac1n\sum_k\epsilon_kx_k\rvert\le C_2n^{-1/2}$, and the choice
    $\epsilon_k=\operatorname{sign}x_k$ gives
    $\frac1n\sum_k\lvert x_k\rvert\le C_2n^{-1/2}$. Next apply
    (\ref{eq:ldp_expectation_overlap}) to
    $Z=\frac1n\sum_k\mathcal O^{(k)}\mathcal O'^{(k)}$, which is Hermitian
    because the two factors act on disjoint sites. Its expectation in
    $\ket{\phi_N}$ is $G_N(\mathcal O,\mathcal O';s')$. In $\ket{\psi_N}$ the
    light cone gives
    $\bra{\psi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\psi_N}=x_ky_k$ with
    $\lvert y_k\rvert\le2$, so the expectation is at most $2C_2n^{-1/2}$ in
    modulus and the variance is at most $16/n$. With
    (\ref{eq:ldp_decaying_correlations}) and $n\ge N/(4s')$,
    \begin{align}
        c\,e^{-(s'-1)/\xi}
         & \le\bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
        \le C_3\Bigl(\frac{s'}{N}\Bigr)^{1/2}.
        \label{eq:ldp_depth_contradiction}
    \end{align}
    Since $s'\le s_N+K-1=o(\log N)$, the left side is $N^{-o(1)}$ and the
    right side is $N^{-1/2+o(1)}$, so (\ref{eq:ldp_depth_contradiction}) fails
    for all $N$ larger than some $N_0$ determined by $A$ and the sequence
    $(T_N)$. For those $N$, $\varepsilon(\phi_N,\psi_N)>1/2$.

    The argument is quantitative. With the overlap bound $1/2$ replaced by
    $\eta\in(0,1]$, the constants $C_2$ and $C_3$ grow at most like
    $\eta^{-1}$. Since $s'\le2T+C$, and the bound below is trivial when
    $T\ge\xi\ln N/4$, (\ref{eq:ldp_depth_contradiction}) shows that
    $\lvert\braket{\phi_N}{\psi_N}\rvert\ge\eta$ requires
    \begin{align}
        T
         & \ge\frac\xi4\bigl(\ln N-\ln\ln N\bigr)-\xi\ln\frac1\eta-C_A,
        \label{eq:ldp_depth_constant}
    \end{align}
    with $C_A$ depending only on $A$; for $\eta=1/2$ it follows from the form
    (\ref{eq:ldp_depth_lower_bound_constant}) of
    Theorem~\ref{thm:ldp_depth_lower_bound_constant}. The same constant
    $\xi/4$ in front of
    $\ln N$ is implicit in the final inequality of the proof in
    \cite{Malz2024Preparation}, which compares $N/\ln N$ with a multiple of
    $e^{4T/\xi}$.
\end{proof}

\begin{lemma}[Averages over separated windows]
    \label{lem:ldp_window_averages}
    \lean{MPSTensor.exists_norm_mpvExpectation_mul_sub_mul_le,
        MPSTensor.exists_norm_sub_inner_smul_sq_le_mpv,
        MPSTensor.mpvExpectation_chainWindowOperator_eq,
        MPSTensor.mpvExpectation_chainWindowOperator_mul_eq,
        MPSTensor.norm_toContinuousLinearMap_physicalObservableTransfer_le,
        MPSPreparation.norm_sub_inner_smul_sq_le,
        MPSPreparation.norm_sub_inner_smul_sq_le_of_isPreparedInDepth,
        MPSTensor.exists_norm_sub_inner_smul_add_le_div_sqrt,
        MPSPreparation.isSeparatedBy_window}
    \leanok
    \uses{lem:ldp_two_window_trace, lem:ldp_finite_size_correction,
        lem:qc_light_cone_correlations}
    Let $A$ be in the gauge (\ref{eq:ldp_normal_gauge}), with the products of
    $L$ matrices spanning $\MN{D}$, and let $\lvert\lambda_2\rvert<r<1$ and
    $M\ge0$. For an operator $X$ on $(\C^d)^{\otimes w}$ let $X^{(k)}$ act on
    the sites $k\Delta+1,\dots,k\Delta+w$ of the ring of $N$ sites, for
    $0\le k<n$ with $n\Delta\le N$, and let $Z=\sum_kc_kX^{(k)}$ with real
    $\lvert c_k\rvert\le\gamma$.
    \begin{enumerate}
        \item Expectations in $\ket{\phi_N}$ of operators on one window, or on
              two windows of the same length at a fixed distance, do not depend
              on the position of the windows, as long as the windows lie in the
              sites $1,\dots,N$ without wrapping around the ring.
        \item In the gauge (\ref{eq:ldp_normal_gauge}),
              $\lVert\E_X\rVert\le D^3\lVert X\rVert$ whatever the number of
              sites $X$ acts on.
        \item There is $C$ such that for $\lVert X\rVert,\lVert Y\rVert\le M$
              on windows of $w$ sites separated by $g\ge1$ and $g'\ge1$ sites
              around the ring, the connected correlation in $\ket{\phi_N}$ is at
              most $C(r^g+r^{g'})$.
        \item There is $C'$ such that for Hermitian $X$ with
              $\lVert X\rVert\le M$ and $w<\Delta$, and for every $N$
              with $\ket{\phi_N(A)}\neq0$,
              $\lVert(Z-\langle Z\rangle_{\phi_N})\phi_N\rVert^2\le
                  3n\gamma^2C'$.
        \item If $\ket{\psi}$ is a unit vector prepared in depth $T$ and
              $w+2T\le\Delta$, then
              $\lVert(Z-\langle Z\rangle_\psi)\psi\rVert^2\le
                  6n\gamma^2\lVert X\rVert^2$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    Part (1) is the trace formula of Lemma~\ref{lem:ldp_two_window_trace} at
    an arbitrary offset: the exponent of $\E_A$ closing the ring changes by
    the offset and the trace absorbs it. For (2), the entry of $\E_X$ applied
    to a matrix unit is $\bra{p}X\ket{q}$ for vectors $p,q$ of entries of
    products of matrices of $A$, and $\sum_i(A^i)^\dagger A^i=\Id$ makes their
    norms at most one. Part (3) substitutes $\E_A^{\,k}=P+(\E_A-P)^k$ into
    the trace formulas, as in Lemma~\ref{lem:ldp_finite_size_correction}. For
    (4) and (5), $\lVert(Z-\langle Z\rangle)\chi\rVert^2$ is the double sum
    of $c_kc_l$ times the connected correlations of $X^{(k)}$ and $X^{(l)}$;
    in $\ket{\phi_N}$ these are bounded by (3) with $g\ge l-k$ and
    $g'\ge n-l+k$, and in $\ket{\psi}$ they vanish for $k\ne l$ by
    Lemma~\ref{lem:qc_light_cone_correlations}, since the windows are at
    least $2T$ sites apart on both sides.
\end{proof}

\begin{theorem}[No preparation in depth $o(\log N)$, gauge form]
    \label{thm:ldp_depth_lower_bound_gauge}
    \lean{MPSTensor.eventually_one_half_lt_infidelity_of_isLittleO_log,
        MPSTensor.exists_depth_lower_bound,
        MPSTensor.exists_log_le_of_infidelity_le_one_half,
        MPSTensor.norm_mpvConnectedCorrelator_le_of_overlap,
        MPSTensor.mpvExpectation_centred_pair,
        MPSPreparation.sum_norm_inner_div_le_of_overlap}
    \leanok
    \uses{def:qc_local_circuit, def:correlation_length}
    Let $A$ be normal and in the gauge (\ref{eq:ldp_normal_gauge}), let $L\ge1$
    be a length such that the products of $L$ matrices of $A$ span $\MN{D}$,
    and let $\lambda_2\neq1$ be an eigenvalue of $\E_A$ of largest modulus
    among those different from $1$, with $\xi=-1/\log\lvert\lambda_2\rvert>0$.
    \begin{enumerate}
        \item There are $B\ge0$ and $C>0$ such that for every $T\ge0$ and
              $N\ge B(T+1)$, if a unit vector $\ket{\psi}$ prepared in depth $T$
              satisfies $\lvert\braket{\phi_N}{\psi}\rvert\ge1/2$, then
              $N\le C(T+1)\,e^{4T/\xi}$.
        \item If $\ket{\psi_N}$ are unit vectors prepared in depth $T_N$ with
              $T_N=o(\log N)$, then $\varepsilon(\phi_N,\psi_N)>1/2$ for all
              sufficiently large $N$.
        \item There is $C'$ such that for all $N\ge1$ and $T\ge0$, if a unit
              vector $\ket{\psi}$ prepared in depth $T$ satisfies
              $\varepsilon(\phi_N,\psi)\le1/2$, then
              $\frac\xi4\ln N\le T+\frac\xi4\ln(T+1)+C'$.
    \end{enumerate}
    Part (2) is Theorem~\ref{thm:ldp_depth_lower_bound} for a tensor in the
    gauge (\ref{eq:ldp_normal_gauge}); Lemma~\ref{lem:ldp_normal_gauge_reduction}
    removes the restriction~\cite{gap:mswc24_depth_lower_bound_gauge}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_decaying_correlations, lem:ldp_expectation_overlap,
        lem:ldp_window_averages, lem:qc_light_cone_correlations}
    Part (1) is the argument of the proof of
    Theorem~\ref{thm:ldp_depth_lower_bound} from its second paragraph on, with $s'\le s_0+2T+L+2$,
    $\Delta\le(s_0+2L+4)(T+1)$ and $n\ge N/(2\Delta)$. Squaring
    (\ref{eq:ldp_depth_contradiction}) gives
    $N\le C_1\Delta\,e^{2(s'-1)/\xi}$, and
    $e^{2(s'-1)/\xi}\le e^{2(s_0+L+1)/\xi}e^{4T/\xi}$, which is (1). For (2),
    put $b=4/\xi$; $T_N\le\log N/(2(b+1))$ for large $N$ gives
    $C(T_N+1)e^{b(T_N+1)}\le Ce^{b+1}\sqrt N<N$ and $B(T_N+1)\le N$, so
    $\lvert\braket{\phi_N}{\psi_N}\rvert<1/2$ by (1). For (3), if
    $N\ge B(T+1)$ take logarithms in (1); otherwise
    $\ln N\le\ln(B+1)+\ln(T+1)$. In both cases
    $\ln N\le\lvert\ln C\rvert+\ln(B+1)+\ln(T+1)+4T/\xi$; multiply by
    $\xi/4$.
\end{proof}

\begin{lemma}[Reduction to the gauge]
    \label{lem:ldp_normal_gauge_reduction}
    \lean{MPSTensor.IsNormalTensor.exists_normalGauge,
        MPSTensor.exists_normalGauge_of_isNormal,
        MPSTensor.hasEigenvalue_transferMap_iff_of_gaugeEquiv,
        MPSTensor.transferMap_smul_eq_norm_sq_smul,
        MPSTensor.normalizedMPVState_eq_of_gaugeEquiv,
        MPSTensor.norm_inner_normalizedMPVState_smul}
    \leanok
    \uses{def:normal, thm:cpsv_normal_tp_gauge,
        thm:normalized_normal_strongly_irreducible,
        thm:cpsv_normal_eventually_injective, thm:gauge_invariance_normal}
    Let the products of some fixed number of matrices of $A$ span $\MN{D}$, and
    let $\lambda_1$ be an eigenvalue of $\E_A$ of largest modulus. Then
    $\lambda_1\ne0$, and there are $\zeta\ne0$, an invertible $X$ and $L\ge1$
    such that $B^i=X\zeta A^iX^{-1}$ satisfies the
    gauge~(\ref{eq:ldp_normal_gauge}) for some $\rho$, the products of $L$
    matrices of $B$ span $\MN{D}$, the eigenvalues of $\E_B$ are those of
    $\E_A$ divided by $\lambda_1$, and $\ket{\phi_N(B)}=\zeta^N\ket{\phi_N(A)}$
    for every $N$. In particular, whenever $\ket{\phi_N(A)}\neq0$,
    $\lvert\braket{\phi_N(B)}{\psi}\rvert/\lVert\phi_N(B)\rVert
        =\lvert\braket{\phi_N(A)}{\psi}\rvert/\lVert\phi_N(A)\rVert$ for every
    vector $\ket{\psi}$; when $\ket{\phi_N(A)}=0$, also $\ket{\phi_N(B)}=0$.
    If instead $A$ is irreducible and $\E_A$ has the unique eigenvalue $1$ of
    largest modulus, the same holds with $\zeta=1$.
\end{lemma}
\begin{proof}
    \leanok
    The eigenvalues of $\E_{\zeta A}=\lvert\zeta\rvert^2\E_A$ are those of
    $\E_A$ times $\lvert\zeta\rvert^2$, and a gauge $B^i=XA^iX^{-1}$ maps an
    eigenvector $Y$ of $\E_A$ to the eigenvector $XYX^\dagger$ of $\E_B$ with
    the same eigenvalue; it leaves every trace in (\ref{eq:ldp_ti_mps})
    unchanged. Since $A$ is normal, $\E_A$ is irreducible, so it has a
    positive eigenvector for its spectral radius $r>0$ and a positive
    eigenvector $\sigma$ of its dual; with $\zeta=r^{-1/2}$ and
    $X=\sigma^{1/2}$, the tensor $B$ satisfies $\sum_i(B^i)^\dagger B^i=\Id$
    (Theorem~\ref{thm:cpsv_normal_tp_gauge}). By
    Theorem~\ref{thm:normalized_normal_strongly_irreducible}, $\E_B$ has a fixed point
    $\rho>0$, which we scale to unit trace, and $1$ is its only eigenvalue of
    modulus one. All eigenvalues of the trace-preserving map $\E_B$ have
    modulus at most one, and $\lvert\zeta\rvert^2\lambda_1$ is one of largest
    modulus, so $\lvert\zeta\rvert^2\lambda_1=1$. When $A$ is irreducible and
    $1$ is the unique eigenvalue of $\E_A$ of largest modulus, $r=1$, so
    $\zeta=1$; primitivity of $\E_A$ passes to $\E_B$, and some blocking of $B$
    is injective by Theorems~\ref{thm:cpsv_normal_eventually_injective}
    and~\ref{thm:gauge_invariance_normal}.
\end{proof}

\begin{theorem}[The depth lower bound with the constant $\xi/4$]
    \label{thm:ldp_depth_lower_bound_constant}
    \lean{MPSTensor.exists_log_le_of_infidelity_le_one_half_of_isNormal,
        MPSTensor.exists_mul_log_le_of_infidelity_le_one_half_of_isNormal,
        Real.log_le_mul_sub_one_sub_log}
    \leanok
    \uses{def:qc_local_circuit, def:correlation_length,
        thm:ldp_depth_lower_bound}
    Let $A$ and $\xi=-1/\log\lvert\lambda_2/\lambda_1\rvert>0$ be as in
    Theorem~\ref{thm:ldp_depth_lower_bound}.
    \begin{enumerate}
        \item There is $C$ such that for all $N\ge1$ and $T\ge0$, if a unit
              vector $\ket{\psi}$ prepared in depth $T$ satisfies
              $\varepsilon(\phi_N,\psi)\le1/2$, then
              \begin{align}
                  \frac\xi4\ln N
                   & \le T+\frac\xi4\ln(T+1)+C.
                  \label{eq:ldp_depth_lower_bound_constant}
              \end{align}
        \item For every $0<a<\xi/4$ there is $C_a$ such that under the same
              conditions $a\ln N\le T+C_a$.
    \end{enumerate}
    This is the explicit form of \cite[Theorem~1]{Malz2024Preparation} that
    its proof gives: its final inequality compares $N/\ln N$ with a multiple
    of $e^{4T/\xi}$. Two constants of that inequality need
    repair~\cite{gap:mswc24_depth_lower_bound_constant}; the rate $4/\xi$
    does not change.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_depth_lower_bound_gauge, lem:ldp_normal_gauge_reduction}
    By Lemma~\ref{lem:ldp_normal_gauge_reduction}, $A$ may be replaced by a
    tensor in the gauge (\ref{eq:ldp_normal_gauge}); this changes
    $\ket{\phi_N}$ by a phase and divides the eigenvalues of $\E_A$ by
    $\lambda_1$, so $\xi$ and $\lvert\braket{\phi_N}{\psi}\rvert$ are
    unchanged. Part (1) is then part (3) of
    Theorem~\ref{thm:ldp_depth_lower_bound_gauge}. For (2) put
    $\theta=4a/\xi<1$ and $l=(1-\theta)/a$. Multiplying
    (\ref{eq:ldp_depth_lower_bound_constant}) by $\theta$ gives
    $a\ln N\le\theta T+a\ln(T+1)+\theta C$, and
    $\ln x=\ln(lx)-\ln l\le lx-1-\ln l$ gives
    $a\ln(T+1)\le(1-\theta)(T+1)-a(1+\ln l)$. Adding,
    $a\ln N\le T+C_a$ with $C_a=1-\theta-a(1+\ln l)+\theta C$.
\end{proof}

\begin{corollary}[Depth $\Theta(\log N)$ at fixed error]
    \label{cor:ldp_depth_theta_log}
    \lean{MPSTensor.exists_depth_isTheta_log}
    \leanok
    \uses{def:qc_local_circuit, def:correlation_length,
        thm:ldp_depth_lower_bound}
    Let $A$ and $\xi$ be as in Theorem~\ref{thm:ldp_depth_lower_bound}, and
    fix $0<\varepsilon\le1/2$. There are $a>0$, $c$, $C$ and $N_0$ such that
    for every $N\ge N_0$:
    \begin{enumerate}
        \item some state $\ket{\psi}$ with
              $\varepsilon(\psi,\phi_N)\le\varepsilon$ is prepared in depth
              at most $c\ln N$;
        \item every state $\ket{\psi}$ prepared in depth $T$ with
              $\varepsilon(\phi_N,\psi)\le\varepsilon$ has $T\ge a\ln N-C$.
    \end{enumerate}
    Any $a<\xi/4$ is admissible. Thus the optimal depth is $\Theta(\log N)$,
    as stated in the introduction of~\cite{Malz2024Preparation}.
\end{corollary}
\begin{proof}
    \leanok
    \uses{thm:ldp_depth_upper_bound, thm:ldp_depth_lower_bound_constant}
    For (1), Theorem~\ref{thm:ldp_depth_upper_bound} gives depth at most
    $c_0\ln(N/\varepsilon)$ for $N\ge N_0$, and for $N\ge2$,
    $\ln(N/\varepsilon)=\ln N+\ln(1/\varepsilon)
        \le\bigl(1+\ln(1/\varepsilon)/\ln2\bigr)\ln N$. Part (2) is part
    (2) of Theorem~\ref{thm:ldp_depth_lower_bound_constant}, since
    $\varepsilon\le1/2$.
\end{proof}

\section{Tensors that are not normal}

\begin{definition}[Fixed point of a basis of normal tensors]
    \label{def:ldp_non_normal_fixed_point}
    \uses{def:bnt, def:sector_weight_data, def:prep_fixed_point_tensor,
        thm:ldpolar_tensor_decomposition, def:correlation_length}
    Let $A^i=\bigoplus_{j=1}^b\operatorname{diag}(\mu_{j,1},\dots,\mu_{j,m_j})
        \otimes A_j^i$ with normal $A_j$, each in the gauge
    (\ref{eq:ldp_normal_gauge}), forming a basis of normal tensors, and put
    $\beta_j=\sum_{k=1}^{m_j}\mu_{j,k}^N$. Consider lengths $N=qM$ with
    $(\beta_1,\dots,\beta_b)\neq0$. Write $\xi_j$ for the correlation length of
    the normal tensor $A_j$, formed from the subleading eigenvalue of its own
    transfer map $\E_{A_j}$. With $\ket{\omega_j}$ the pair state of $A_j$ and
    $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$, the
    \emph{fixed-point state} is
    \begin{align}
        \ket{\Omega'}
         & =\sum_{j=1}^b\alpha^{(N)}_j\bigotimes_{k=1}^{M}\ket{\omega_j}_{R_kL_{k+1}}.
        \label{eq:ldp_non_normal_fixed_point}
    \end{align}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} for the blocked map of
    $A$, the \emph{approximating state} is
    $\ket{\widetilde\phi_N}=V^{\otimes M}\ket{\Omega'}/
        \lVert V^{\otimes M}\ket{\Omega'}\rVert$, whenever the denominator is
    nonzero.
    This is \cite[eqs.~(19) and~(S7)]{Malz2024Preparation}, with the
    canonical form of \cite{Cirac2017MPDO_AnnPhys}.

    When some block has multiplicity $m_j\geq2$, the source's construction
    fails: no product of pairs carrying the weights $\beta_j$ gives, after
    $V^{\otimes M}$, an approximation of $\ket{\phi_N}$. For
    $A^0=\operatorname{diag}(1,1,0)$ and $A^1=\operatorname{diag}(0,0,1)$ the
    overlap with $\ket{\phi_N}\propto2\ket{0\cdots0}+\ket{1\cdots1}$ tends to
    $1/\sqrt5$ as $M\to\infty$, for every $q$
    \cite{gap:mswc24_multiplicity_fixed_point}. The failure lies in the
    coefficients $\beta_j$. With the pairs of each block on one of its copies
    $k_j$ and the coefficients $\beta'_j=\beta_j(c_j/\mu_{j,k_j}^q)^M$ of
    Definition~\ref{def:ldp_one_copy_weight}, with the copy norms $c_j$ of
    Definition~\ref{def:ldp_repeated_block_sum}, in place of $\beta_j$, the
    construction converges for all nonzero complex weights, also when the
    $q$-site states of distinct blocks overlap, at a rate that includes the
    correlation lengths of the mixed transfer maps
    (Lemma~\ref{lem:ldp_one_copy_state} and
    Theorem~\ref{thm:ldp_approximation_error_one_copy})
    \cite{gap:mswc24_repeated_block_corrected_state,gap:mswc24_multiplicity_fixed_point}.

    When a weight is not a nonnegative real number, the coefficients
    $\beta_j=\mu_j^N$ count its phase twice, since $V$ already carries it. For
    $A^0=\operatorname{diag}(1,0)$ and $A^1=\operatorname{diag}(0,-1)$ and odd $N$
    the approximating state is orthogonal to $\ket{\phi_N}$; for multiplicity one the
    coefficients $\beta_j=\lvert\mu_j\rvert^N$ repair the construction
    (Section~\ref{sec:ldp_complex_weights})
    \cite{gap:mswc24_block_form_mixed_overlap}.
\end{definition}

\begin{definition}[Fixed-point state of a sector decomposition]
    \label{def:ldp_sector_fixed_point_state}
    \lean{MPSTensor.SectorDecomposition.copyCoord,
        MPSTensor.SectorDecomposition.embeddedFixedPointPair,
        MPSTensor.SectorDecomposition.canonicalFixedPointState,
        MPSTensor.embedPair}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, def:sector_weight_data,
        def:prep_pair_product_state}
    Let $A$ be given by a sector decomposition with blocks $A_1,\dots,A_b$ of
    bond dimensions $D_j$, multiplicities $m_j$ and coefficients $\mu_{j,k}$,
    whose flattened direct sum $\bigoplus_{(j,k)}\mu_{j,k}A_j$ is the tensor of
    Definition~\ref{def:ldp_non_normal_fixed_point} up to the order of the
    bond coordinates, so the bond space of $A$ is $\C^D$ with
    $D=\sum_jm_jD_j$. The blocks carry no normality, basis-of-normal-tensors
    or gauge condition. For each $j$ let $\sigma_j$ be any matrix on
    $\C^{D_j}$ and choose a copy $k_j\in\{1,\dots,m_j\}$. The pair of block
    $j$ is $\sum_{r,l}(\sqrt{\sigma_j})_{rl}\ket{r}\ket{l}$ placed, on both
    legs, on the bond coordinates of copy $k_j$, which gives
    $\ket{\omega_j}\in\C^D\otimes\C^D$. With $\beta_j$ and $\alpha^{(N)}_j$ as
    in Definition~\ref{def:ldp_non_normal_fixed_point}, the fixed-point state
    $\ket{\Omega'}$ is formed from these pairs by
    (\ref{eq:ldp_non_normal_fixed_point}). Here each $\sigma_j$ is an arbitrary
    matrix on $\C^{D_j}$, and the copy $k_j$ is specified explicitly.
\end{definition}

\begin{definition}[Fixed-point state of a family of pairs]
    \label{def:ldp_pair_family_fixed_point}
    \lean{MPSTensor.bntWeight, MPSTensor.ghzAmplitude,
        MPSTensor.sum_norm_sq_pos_of_ne_zero, MPSTensor.ghzAmplitude_norm_sq,
        MPSTensor.nonNormalFixedPointState, MPSTensor.nonNormalApproxState,
        MPSTensor.pairIsometry, Matrix.tensorPower}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, thm:ldpolar_tensor_decomposition,
        def:qc_ghz_state}
    Let $\omega_1,\dots,\omega_b\in\mathbb C^D\otimes\mathbb C^D$ be any vectors,
    $\alpha\in\mathbb C^b$ any coefficients, and $M\ge0$. The state
    $\ket{\Omega'}=\sum_j\alpha_j\bigotimes_{k=1}^M\ket{\omega_j}_{R_kL_{k+1}}$ is
    formed as in (\ref{eq:ldp_non_normal_fixed_point}), and for a tensor $A$ and a
    blocking length $q$, with $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition}
    for the $q$-site blocked map of $A$, the approximating state is
    $V^{\otimes M}\ket{\Omega'}/\lVert V^{\otimes M}\ket{\Omega'}\rVert$, and zero when
    the denominator vanishes. For multiplicity coefficients $\mu_{j,k}$ put
    $\beta_j=\sum_k\mu_{j,k}^N$ and $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$.
    For any $\beta\in\mathbb C^b$ with $\beta\neq0$, $\sum_l\lvert\beta_l\rvert^2>0$ and
    $\sum_j\lvert\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}\rvert^2=1$.
    Let $W\colon\mathbb C^b\to\mathbb C^D\otimes\mathbb C^D$,
    $\ket{j}\mapsto\ket{\omega_j}$, be the map whose $j$-th column is $\omega_j$,
    let $X^{\otimes M}$ denote the $M$-fold tensor power of a matrix $X$, with entries
    $(X^{\otimes M})_{p,s}=\prod_{k=1}^M X_{p_k,s_k}$, and put
    $\ket{\chi_M}=\sum_j\alpha_j\ket{j}^{\otimes M}\in(\mathbb C^b)^{\otimes M}$.
    For $\omega_j$ given by the embedded pair state of each normal block $A_j$,
    $\alpha=\alpha^{(N)}$, and $N=qM$, this specializes to
    Definition~\ref{def:ldp_non_normal_fixed_point}.
\end{definition}

\begin{theorem}[Local orthogonality of the pairs]
    \label{thm:ldp_local_orthogonality}
    \lean{MPSTensor.SectorDecomposition.inner_embeddedFixedPointPair,
        MPSTensor.inner_embedPair_self, MPSTensor.inner_embedPair_eq_zero_of_disjoint,
        MPSTensor.SectorDecomposition.copyCoord_injective,
        MPSTensor.SectorDecomposition.copyCoord_ne_of_ne,
        MPSTensor.SectorDecomposition.toTensor_copyCoord,
        MPSTensor.pairProductState_embedPair,
        MPSTensor.mpv_toTensorFromBlocks_fixedPointTensor}
    \leanok
    \uses{def:ldp_sector_fixed_point_state}
    In Definition~\ref{def:ldp_sector_fixed_point_state}, let each $\sigma_j$ be
    positive semidefinite with $\operatorname{Tr}\sigma_j=1$. For every choice
    of the copies $k_j$ the embedded pairs satisfy
    $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$. The chosen coordinates of
    block $j$ carry the block $\mu_{j,k_j}A_j$ of $A$, and embedding the pair
    embeds its product state. Moreover, for $M\geq1$, in the bond space
    $\C^{D_j}\otimes\C^{D_j}$ of block $j$ alone, the tensor
    $\bigoplus_k\mu_{j,k}^qP_{j,\infty}$ generates on $M$ sites the state
    $\bigl(\sum_k\mu_{j,k}^{qM}\bigr)\ket{\Omega_j}=\beta_j\ket{\Omega_j}$ with
    $N=qM$ and $\ket{\Omega_j}$ formed from the unembedded pair. The
    orthogonality is asserted after \cite[eq.~(S7)]{Malz2024Preparation},
    where the reason given is that the blocks form a basis of normal tensors;
    here it follows from the placement of the pairs alone. The identification
    of these pairs with the fixed points of the positive parts of the blocked
    tensor is not proved here and can fail when a block has multiplicity
    greater than one \cite{gap:mswc24_ghz_form_local_orthogonality}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:ldp_pair_disjoint_orthogonal, thm:prep_pair_norm,
        thm:prep_mpv_fixed_point_tensor, thm:mpv_decomposition}
    Copies of distinct blocks occupy distinct bond coordinates, so the right
    legs of $\omega_j$ and $\omega_{j'}$ with $j\neq j'$ are supported on
    disjoint sets and the pairs are orthogonal by
    Theorem~\ref{thm:ldp_pair_disjoint_orthogonal}. The coordinates of one
    copy are distinct, so embedding preserves overlaps and
    $\braket{\omega_j}{\omega_j}=\operatorname{Tr}\sigma_j=1$. The direct sum
    generates the sum of the states of its blocks with the $M$-th powers of the
    weights, and $P_{j,\infty}$ generates $\ket{\Omega_j}$.
\end{proof}

\begin{theorem}[GHZ form of the fixed-point state]
    \label{thm:ldp_ghz_form}
    \lean{MPSTensor.isIsometry_pairIsometry, Matrix.IsIsometry.tensorPower,
        MPSTensor.nonNormalFixedPointState_eq_tensorPower_mulVec_ghzState,
        MPSTensor.nonNormalFixedPointState_norm_sq,
        MPSTensor.SectorDecomposition.isIsometry_pairIsometry_embeddedFixedPointPair,
        MPSTensor.SectorDecomposition.canonicalFixedPointState_eq_tensorPower_mulVec_ghzState,
        MPSTensor.SectorDecomposition.canonicalFixedPointState_norm_sq}
    \leanok
    \uses{def:ldp_sector_fixed_point_state, def:ldp_pair_family_fixed_point,
        thm:ldp_local_orthogonality}
    With $W$, $X^{\otimes M}$ and $\chi_M$ as in
    Definition~\ref{def:ldp_pair_family_fixed_point}, formed with
    $\alpha=\alpha^{(N)}$ and the pairs of
    Definition~\ref{def:ldp_sector_fixed_point_state}, where each $\sigma_j$ is
    positive semidefinite of unit trace, $W$ and $W^{\otimes M}$ are isometries
    and $\ket{\Omega'}=W^{\otimes M}\ket{\chi_M}$, up to the regrouping of the
    legs $R_kL_{k+1}$. If moreover $M\geq1$ and $(\beta_1,\dots,\beta_b)\neq0$,
    then $\braket{\Omega'}{\Omega'}=1$. This is the preparation of
    \cite[eq.~(19)]{Malz2024Preparation}. The same holds for the state of
    Definition~\ref{def:ldp_pair_family_fixed_point} with $\alpha=\alpha^{(N)}$
    for any orthonormal pairs $\omega_j$; the identity
    $\ket{\Omega'}=W^{\otimes M}\ket{\chi_M}$ holds for any pairs.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:prep_pair_product_norm, thm:ldp_local_orthogonality}
    The pairs of Definition~\ref{def:ldp_sector_fixed_point_state} are
    orthonormal by Theorem~\ref{thm:ldp_local_orthogonality}. The $(j,j')$ entry of $W^\dagger W$ is $\braket{\omega_j}{\omega_{j'}}$, and the
    entries of $(W^{\otimes M})^\dagger W^{\otimes M}$ factor over the sites,
    \begin{align*}
        \bigl((W^{\otimes M})^\dagger W^{\otimes M}\bigr)_{s,t}
        =\prod_{k=1}^M(W^\dagger W)_{s_k,t_k}=\delta_{st}.
    \end{align*}
    Expanding
    $\ket{j}^{\otimes M}$ in the product basis gives
    $W^{\otimes M}\ket{j}^{\otimes M}=\bigotimes_k\ket{\omega_j}$, and summing with the
    weights $\alpha^{(N)}_j$ gives $\ket{\Omega'}$. Overlaps of the product states factor
    over the bonds,
    $\braket{\Omega_j}{\Omega_{j'}}=\braket{\omega_j}{\omega_{j'}}^M=\delta_{jj'}$, so
    $\braket{\Omega'}{\Omega'}=\sum_j\lvert\alpha^{(N)}_j\rvert^2=1$.
\end{proof}

\begin{theorem}[Pairs with disjoint supports are orthogonal]
    \label{thm:ldp_pair_disjoint_orthogonal}
    \lean{MPSTensor.inner_pair_eq_zero_of_disjoint}
    \leanok
    \uses{def:ldp_sector_fixed_point_state}
    Let $S,S'\subseteq\{0,\dots,D-1\}$ be disjoint, and let
    $\omega,\omega'\in\C^D\otimes\C^D$ satisfy $\omega_{(a,c)}=0$ for $a\notin S$
    and $\omega'_{(a,c)}=0$ for $a\notin S'$. Then
    $\braket{\omega}{\omega'}=0$. The embedded pairs of distinct blocks in
    Definition~\ref{def:ldp_sector_fixed_point_state} are supported in this way, so
    this gives the off-diagonal half of the orthogonality
    $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$, which
    Theorem~\ref{thm:ldp_local_orthogonality} completes.
\end{theorem}

\begin{proof}
    \leanok
    Every term of the sum vanishes: if $a\in S$ then $a\notin S'$ and
    $\omega'_{(a,c)}=0$, and otherwise $\omega_{(a,c)}=0$, so
    \begin{align*}
        \braket{\omega}{\omega'}
        =\sum_{(a,c)}\overline{\omega_{(a,c)}}\,\omega'_{(a,c)}=0.
    \end{align*}
\end{proof}

    blocks $A_j$, $0<\gamma<1/2$, and $q=o(N)$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi_{\mathrm{diag}}}\right).
        \label{eq:ldp_approximation_error_general}
    \end{align}

\end{theorem}

\begin{theorem}[The approximating state for a repeated block]
    \label{thm:ldp_approximation_error_repeated_block}
    \lean{MPSTensor.isBNTCanonicalForm_and_not_approximationError_le_repeatedBlock,
        MPSTensor.isBNTCanonicalForm_repeatedBlockSector,
        MPSTensor.repeatedBlockSector, MPSTensor.coeff_repeatedBlockSector,
        MPSTensor.reindex_toTensor_repeatedBlockSector,
        MPSTensor.nonNormalApproxOverlap_repeatedBlockTensor,
        MPSTensor.tendsto_norm_nonNormalApproxOverlap_repeatedBlockTensor,
        MPSTensor.norm_le_of_hasEigenvalue_transferMap_repeatedBlockBasis,
        MPSTensor.not_approximationError_le_repeatedBlockTensor}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, def:ldp_sector_fixed_point_state,
        def:correlation_length, def:sector_bnt_cf, def:nt_cpsv,
        def:cpsv_cfii}
    Let $A^0=\operatorname{diag}(1,1,0)$ and $A^1=\operatorname{diag}(0,0,1)$,
    the canonical form with the normal block $A_1=(1,0)$ of multiplicity two and
    weights $(1,1)$ and the normal block $A_2=(0,1)$ of multiplicity one, so that
    $\beta=(2,1)$ and $\ket{\phi_N}\propto2\ket{0\cdots0}+\ket{1\cdots1}$.
    The blocks are normal and in canonical form~II, they form a basis of normal
    tensors whose states become orthogonal as $N\to\infty$, and the weights satisfy $\lvert\mu_{j,k}\rvert\leq1$ with one of
    modulus one, so $A$ satisfies every hypothesis of
    \cite[Lemma~1$'$(ii)]{Malz2024Preparation}. Place
    the pair of the first block on its first copy. For every $q\geq1$ and
    $M\geq1$ the approximating state of
    Definition~\ref{def:ldp_non_normal_fixed_point} satisfies
    \begin{align}
        \braket{\widetilde\phi_N}{\phi_N}
         & =\frac{4\cdot2^{-M/2}+1}{\sqrt5\,(4\cdot2^{-M}+1)^{1/2}},
        \label{eq:ldp_repeated_block_overlap}
    \end{align}
    which does not depend on $q$ and tends to $1/\sqrt5$ as $M\to\infty$. The
    transfer maps of both blocks have no eigenvalue other than one, so every
    $\lambda_2$ with $0<\lvert\lambda_2\rvert<1$ bounds their subleading
    eigenvalues. For every such $\lambda_2$, every $\gamma>0$ and every constant
    $C$ there is $M\geq1$ such that, with $q=M$, $N=M^2$ and
    $y=(N/q)e^{-\gamma q/\xi_{\mathrm{diag}}}$,
    $\varepsilon(\widetilde\phi_N,\phi_N)>Cy\,e^{Cy}$. The sequence $q=M$
    satisfies $q=o(N)$, so the bound (\ref{eq:ldp_approximation_error_general})
    fails for this tensor \cite{gap:mswc24_multiplicity_fixed_point}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldp_polar_uniqueness}
    Each block is one-dimensional with $\sum_i\lvert a_i\rvert^2=1$, so its
    transfer map is the identity. It is therefore irreducible and normal, and
    left-canonical with the diagonal positive-definite fixed point $1$, which is
    the canonical-form~II condition; the pairs below are built from this same
    fixed point. Its self-overlap is one. The
    states $\ket{0\cdots0}$ and $\ket{1\cdots1}$ of the blocks are linearly
    independent for $N\geq1$, and the blocks are not related by a gauge
    transformation and a phase, since $A_1^1=0\neq A_2^1$. With the bond
    coordinates ordered as the two copies of $A_1$ followed by the copy of
    $A_2$, and the weights $(1,1)$ and $1$, the direct sum of the copies is $A$:
    for $i=0,1$,
    \begin{align}
        A^i & =A_1^i\oplus A_1^i\oplus A_2^i
        =\operatorname{diag}\bigl(A_1^i,A_1^i,A_2^i\bigr).
        \label{eq:ldp_repeated_block_assembly}
    \end{align}

    For a tensor of diagonal matrices the blocked map factors as $B=CJ^\dagger$,
    where $J\colon\ket e\mapsto\ket{e\,e}$ and the column $e$ of $C$ is the state
    of the diagonal coordinate $e$ on $q$ sites; here these are $\ket{0\cdots0}$
    for $e=0,1$ and $\ket{1\cdots1}$ for $e=2$. Hence $B^\dagger B=JGJ^\dagger$ with
    $G$ the Gram matrix of the columns. The positive semidefinite square root of $G$
    is $Q=2^{-1/2}\bigl(\begin{smallmatrix}1&1\\1&1\end{smallmatrix}\bigr)\oplus1$,
    and $E=\tfrac12\bigl(\begin{smallmatrix}1&1\\1&1\end{smallmatrix}\bigr)\oplus1$
    is the projector onto its range, so uniqueness of positive square roots gives
    $P=JQJ^\dagger$ and $\Pi=JEJ^\dagger$. The pairs are $e_{00}$ and $e_{22}$, so
    $V^{\otimes M}\sum_j\alpha_j\ket{\Omega_j}$ has components
    $\sum_j\alpha_j\prod_kV_{\tau_k,(c_j,c_j)}$. From $V^\dagger B=P$ and
    $V^\dagger V=\Pi$, the overlap with the unnormalized target is
    $\sum_j\sum_e\overline{\alpha_j}Q_{c_je}^M$, the squared norm of the
    approximating vector is $\sum_{j,j'}\overline{\alpha_j}\alpha_{j'}E_{c_jc_{j'}}^M$,
    and $\lVert\phi_N(A)\rVert^2=\sum_{e,e'}G_{ee'}^M=5$. With
    $\alpha=(2,1)/\sqrt5$ this gives (\ref{eq:ldp_repeated_block_overlap}). Since
    $1/\sqrt5<1/2$, the error exceeds $1/2$ for large $M$, while $Cy\,e^{Cy}$ with
    $y=Mr^M$ and $r=e^{-\gamma/\xi_{\mathrm{diag}}}<1$ tends to zero.
\end{proof}

    Put $s=(3/5)^q$,
    the overlap of the states of the two blocks on $q$ sites,
    $u=\bigl(\sqrt{1+s}+\sqrt{1-s}\bigr)/2$ and
    $v=\bigl(\sqrt{1+s}-\sqrt{1-s}\bigr)/2$. For every $q\geq1$ and $M\geq1$ the
    approximating state of Definition~\ref{def:ldp_non_normal_fixed_point}
    satisfies
    \begin{align}
        \braket{\widetilde\phi_N}{\phi_N}
         & =\frac{u^M+v^M}{(1+s^M)^{1/2}},
        \label{eq:ldp_overlapping_blocks_overlap}
    \end{align}
    and for $M\geq3$ the error is at least $(9/25)^q/16$, while for every $M\geq1$
    it is at most $M(9/25)^q$, so the construction converges whenever
    $(N/q)(9/25)^q\to0$. The transfer maps of both
    blocks have no eigenvalue other than one, so every $\lambda_2$ bounds their
    subleading eigenvalues. If $r=e^{-\gamma/\xi_{\mathrm{diag}}}<9/25$, then for every
    constant $C$ there is $q\geq3$ such that, with $M=q$, $N=q^2$ and
    $y=(N/q)e^{-\gamma q/\xi_{\mathrm{diag}}}$,
    $\varepsilon(\widetilde\phi_N,\phi_N)>Cy\,e^{Cy}$. The sequence $q=M$ satisfies
    $q=o(N)$, so the bound (\ref{eq:ldp_approximation_error_general}) fails even
    when every multiplicity is one \cite{gap:mswc24_block_form_mixed_overlap}. What
    fails is the rate: for $M\geq3$ the error lies between $(9/25)^q/16$ and
    $M(9/25)^q$, where $9/25=\tau^2$ is the square of the eigenvalue $\tau=3/5$ of
    the mixed transfer map of the two blocks.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldp_polar_uniqueness}
    The hypotheses are checked as in the proof of
    Theorem~\ref{thm:ldp_approximation_error_repeated_block}: both blocks are
    one-dimensional with $\sum_i\lvert a_i\rvert^2=1$, their states
    $\ket{0\cdots0}$ and $(\tfrac35\ket0+\tfrac45\ket1)^{\otimes N}$ are
    linearly independent for $N\geq1$, and $A_1^1=0\neq A_2^1$.

    As in the proof of Theorem~\ref{thm:ldp_approximation_error_repeated_block},
    the Gram matrix of the blocked map on the diagonal coordinates is
    $G=\bigl(\begin{smallmatrix}1&s\\s&1\end{smallmatrix}\bigr)$, whose positive
    square root is $Q=\bigl(\begin{smallmatrix}u&v\\v&u\end{smallmatrix}\bigr)$,
    since $u^2+v^2=1$ and $2uv=s$. The determinant of $Q$ is $\sqrt{1-s^2}>0$, so
    the support projector is $JJ^\dagger$. With $\alpha=(1,1)/\sqrt2$ the overlap
    with the unnormalized target is $\sqrt2\,(u^M+v^M)$, the approximating vector
    has norm one, and $\lVert\phi_N(A)\rVert^2=2+2s^M$, which gives
    (\ref{eq:ldp_overlapping_blocks_overlap}). For $M\geq3$,
    \begin{align*}
        u^M+v^M\leq u^3+v^2=1-u^2(1-u)\leq1-\frac{s^2}{16},
    \end{align*}
    because $u^2=\bigl(1+\sqrt{1-s^2}\bigr)/2\geq1/2$ and
    $1-u\geq(1-u^2)/2\geq s^2/8$. For the upper bound, the overlap is one when
    $M=1$, since $u+v=\sqrt{1+s}$. For $M\geq2$,
    $u\geq u^2=\bigl(1+\sqrt{1-s^2}\bigr)/2\geq1-s^2/2$, so $u^M\geq1-Ms^2/2$, and
    $(1+s^M)^{1/2}\leq1+s^2/2$; when $Ms^2<1$ this gives
    \begin{align*}
        \frac{u^M+v^M}{(1+s^M)^{1/2}}\geq\frac{1-Ms^2/2}{1+s^2/2}\geq1-Ms^2,
    \end{align*}
    and when $Ms^2\geq1$ the bound $\varepsilon\leq1$ suffices. Finally $Cy\,e^{Cy}$
    with $y=qr^q$ is eventually below $(9/25)^q/16$, because $q(25r/9)^q\to0$.
\end{proof}

\begin{theorem}[The polar matrix has a first-order perturbation]
    \label{thm:ldp_overlapping_polar_norm_lower_bound}
    \lean{MPSTensor.posDef_diagGram_overlappingBlockDiag,
        MPSTensor.overlappingBlockOverlap_div_two_le_norm_sqrt_diagGram_sub_one,
        MPSTensor.overlappingBlockOverlap_div_two_le_norm_polarPos_sub_limit,
        MPSTensor.exists_norm_polarPos_overlappingBlockTensor_sub_limit_gt_sq}
    \leanok
    \uses{thm:ldp_approximation_error_overlapping_blocks}
    For the tensor of Theorem~\ref{thm:ldp_approximation_error_overlapping_blocks},
    put $s=(3/5)^q$, let $G_q$ be the Gram matrix on its diagonal bond pairs,
    and let $J\ket j=\ket{j,j}$. For every $q\ge1$, $G_q$ is positive definite
    and the positive polar matrix is $P_q=J\sqrt{G_q}J^\dagger$. In operator norm,
    \[
        \lVert\sqrt{G_q}-I\rVert\ge s/2,
        \qquad \lVert P_q-JJ^\dagger\rVert\ge s/2.
    \]
    For every $C\ge0$ and every threshold $q_0$, there is $q\ge\max(1,q_0)$
    such that $\lVert P_q-JJ^\dagger\rVert>Cs^2$. Thus this matrix difference
    is not $O((3/5)^{2q})$. The quadratic normalized-state error of
    Theorem~\ref{thm:ldp_approximation_error_overlapping_second_order}
    concerns a different quantity.
\end{theorem}
\begin{proof}\leanok
    Write $\sqrt{G_q}=\bigl(\begin{smallmatrix}u&v\\v&u\end{smallmatrix}\bigr)$
    as in the preceding theorem. Since $u,v\ge0$, $u^2+v^2=1$ and $2uv=s$,
    one has $u\le1$ and $v\ge s/2$. Each operator norm is at least the
    absolute value of this off-diagonal entry. The Gram matrix is positive
    definite since its positive square root is invertible for $q\ge1$.
    Finally, $(3/5)^q\to0$, so beyond any threshold one may choose $q$ with
    $C(3/5)^q<1/2$, giving $Cs^2<s/2$.
\end{proof}

\begin{lemma}[Polar decomposition of an orthogonal sum]
    \label{lem:ldp_orthogonal_sum_polar}
    \lean{Matrix.polarIso_sum_of_orthogonal, Matrix.exists_polarIso_eq_mul,
        Matrix.conjTranspose_polarIso_mul_polarIso_eq_zero,
        Matrix.conjTranspose_polarIso_mul_eq_zero}
    \leanok
    \uses{thm:ldp_polar_uniqueness}
    Let $B=\sum_jc_jB_jK_j^\dagger$ with $c_j>0$, isometries $K_j$ with
    orthogonal ranges, so that $K_j^\dagger K_{j'}=\delta_{jj'}$, and
    $B_j^\dagger B_{j'}=0$ for $j\neq j'$. With $B_j=V_jP_j$ the polar
    decompositions of the summands, the polar decomposition of $B$ has
    \begin{align*}
        V=\sum_jV_jK_j^\dagger,\qquad P=\sum_jc_jK_jP_jK_j^\dagger.
    \end{align*}
    This is the block form of the positive part asserted in
    \cite[eq.~(S5)]{Malz2024Preparation} (``Since $V$ is an isometry, $P$
    inherits the block structure of $A$''), for blocks of multiplicity one
    under the orthogonality of their states, which the source does not state
    and without which the block form fails
    \cite{gap:mswc24_block_form_mixed_overlap}.
\end{lemma}
\begin{proof}
    \leanok
    Every partial isometry is $V_j=B_jR_j$ for some $R_j$: the support
    projector $\Pi_j$ satisfies $\Pi_j=P_jR_j$ for some $R_j$, because both
    have the same range, and $V_j=V_j\Pi_j=B_jR_j$. Hence
    $V_j^\dagger V_{j'}=R_j^\dagger B_j^\dagger B_{j'}R_{j'}=0$ for $j\neq j'$. Put
    $W=\sum_jV_jK_j^\dagger$, $Q=\sum_jc_jK_jP_jK_j^\dagger\geq0$ and
    $E=\sum_jK_j\Pi_jK_j^\dagger$. The orthogonality of the $K_j$ and of the
    $V_j$ removes all cross terms, so $WQ=B$, $W^\dagger W=E$, $E^2=E=E^\dagger$,
    $EQ=Q$ and $Q\sum_jc_j^{-1}K_jR_jK_j^\dagger=E$. Thus $E$ is the projector
    onto the range of $Q$, and Theorem~\ref{thm:ldp_polar_uniqueness} gives
    $V=W$ and $P=Q$.
\end{proof}

\begin{theorem}[Approximation error for orthogonal blocks of multiplicity one]
    \label{thm:ldp_approximation_error_orthogonal_blocks}
    \lean{MPSTensor.exists_approximationError_le_mul_blockSum,
        MPSTensor.exists_approximationError_le_blockSum,
        MPSTensor.nonNormalApproxVector_blockSum,
        MPSTensor.mixedMapLM_eq_zero_of_conjTranspose_physicalMatrix_mul_eq_zero,
        MPSTensor.mpvOverlap_eq_zero,
        MPSTensor.eq_zero_of_hasEigenvalue_mixedMapLM_of_conjTranspose_mul_eq_zero}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, def:ldp_pair_family_fixed_point,
        lem:ldp_orthogonal_sum_polar, lem:ldp_mixed_transfer_decay,
        thm:ldp_approximation_error_overlapping_weights, def:correlation_length}
    Let $A^i=\bigoplus_{j=1}^b\mu_jA_j^i$ with $b\geq1$, weights $\mu_j>0$, and
    normal blocks $A_j$ in the gauge (\ref{eq:ldp_normal_gauge}) with fixed
    points $\rho_j$, each placed on its own bond coordinates. Let $\lambda_2$
    bound the moduli of the eigenvalues other than one of every transfer map
    $\E_{A_j}$, so that its correlation length $\xi$ bounds every $\xi_j$, and
    let $0<\gamma<1$. Form the approximating state $\widetilde\phi_N$ of
    Definition~\ref{def:ldp_non_normal_fixed_point} with the pairs of the
    $\rho_j$ on the bond coordinates of their blocks and $\beta_j=\mu_j^N$.
    There is $C>0$ such that for all weights $\mu_j>0$, every block length $q$
    at which the $q$-site states of distinct blocks are orthogonal,
    $\sum_{i_1,\dots,i_q}\overline{(A_j^{i_1}\cdots A_j^{i_q})_{ab}}\,
        (A_{j'}^{i_1}\cdots A_{j'}^{i_q})_{a'b'}=0$ for $j\neq j'$ and all
    $a,b,a',b'$, and every $M\geq1$, with $N=qM$ and
    $y=(N/q)\,e^{-\gamma q/\xi}$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)\leq Cy
        \qquad\text{and}\qquad
        \varepsilon(\widetilde\phi_N,\phi_N)\leq Cy\,e^{Cy}.
        \label{eq:ldp_approximation_error_orthogonal_blocks}
    \end{align}
    This is \cite[Lemma~1$'$(ii)]{Malz2024Preparation} restricted to
    multiplicity one and to blocks whose $q$-site states are orthogonal, for
    $0<\gamma<1$ in place of $0<\gamma<1/2$. For
    multiplicity one the phase of $\mu_j$ can be absorbed into $A_j$, which
    stays normal with the same transfer map, so the condition $\mu_j>0$ only
    fixes this presentation. The orthogonality is an added hypothesis: without
    it the statement fails
    (Theorem~\ref{thm:ldp_approximation_error_overlapping_blocks}). Under it no
    condition $q=o(N)$ is needed
    \cite{gap:mswc24_block_form_mixed_overlap}.
\end{theorem}
\begin{proof}
    \leanok
    If the $q$-site states of distinct blocks are orthogonal for no $q$, there is
    nothing to prove. Otherwise let them be orthogonal at $q_0$, and let $B_j$ be
    the $q_0$-site blocked map of $A_j$. The entries of the mixed transfer map
    $Z\mapsto\sum_iB_j^iZ(B_{j'}^i)^\dagger$ are sums
    $\sum_iB_j^{i}{}_{ab}\overline{B_{j'}^{i}{}_{a'b'}}$, the complex conjugates of
    the entries of $B_{j'}^\dagger B_j=0$, so this map vanishes. It is
    $\E_{jj'}^{\,q_0}$, so $\E_{jj'}$ is nilpotent and all its eigenvalues are
    zero. Hence $\lambda_2$ also bounds the moduli of the eigenvalues of the mixed
    transfer maps, and Theorem~\ref{thm:ldp_approximation_error_overlapping_weights}
    with $w_j=\mu_j$ gives both bounds, with a constant that does not depend on the
    weights.
\end{proof}
