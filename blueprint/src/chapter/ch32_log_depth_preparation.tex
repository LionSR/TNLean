\chapter{Preparation of Matrix Product States by Log-Depth Circuits}
\label{ch:log_depth_preparation}

A translation-invariant matrix product state on $N$ sites generated by a
normal tensor is approximated by blocking $q$ sites, replacing the positive
part of the polar decomposition of the blocked tensor by its renormalization
fixed point, and keeping the isometric part. The fixed point is a product of
nearest-neighbour entangled pairs, and the isometric part factorizes into
isometries with bounded bond dimension, either sequentially or along a binary
tree. With $q$ proportional to $\log(N/\varepsilon)$ this gives a circuit of
depth $O(\log(N/\varepsilon))$ with error $\varepsilon$, and no circuit of
depth $o(\log N)$ achieves a small error. The presentation follows
\cite{Malz2024Preparation}.

Throughout, $A$ is a tensor with physical dimension $d$ and bond dimension
$D$, $\E_A$ is its transfer map (Definition~\ref{def:transfer_map}), and
\begin{align}
    \ket{\phi_N(A)}
     & =\sum_{i_1,\dots,i_N=0}^{d-1}\Tr\left(A^{i_1}\cdots A^{i_N}\right)
    \ket{i_1\cdots i_N}
    \label{eq:ldp_ti_mps}
\end{align}
is the translation-invariant vector it generates, that is,
$\ket{\phi_N(A)}=\ket{V^{(N)}(A)}$ in the notation of
Definition~\ref{def:mpv_state}; the chapter keeps the source's notation
$\phi_N$. For two unit vectors the error is
$\varepsilon(\phi,\psi)=1-\lvert\braket{\phi}{\psi}\rvert$.

\section{The approximating state}

Write $\mathcal B_q$ for the physical matrix of the $q$-blocked tensor of $A$
(Definitions~\ref{def:blocked_tensor} and~\ref{def:ldpolar_physical_matrix})
and $\mathcal B_q=VP$ for its polar decomposition
(Theorem~\ref{thm:ldpolar_tensor_decomposition}).
In this section $A$ is normal (Definition~\ref{def:normal}) and in the gauge
\begin{align}
    \sum_{i=0}^{d-1}(A^i)^\dagger A^i
     & =\Id,
     &
    \E_A(\rho)
     & =\rho,
     &
    \rho
     & >0,
     &
    \Tr\rho
     & =1,
    \label{eq:ldp_normal_gauge}
\end{align}
so that $\E_A^{\,n}(X)\to\Tr(X)\,\rho$ as $n\to\infty$
\cite[eq.~(5)]{Malz2024Preparation}. The limit uses normality as follows.
Normality means that the words of some fixed length $L_0$ span $\MN{D}$, so
by Theorem~\ref{thm:wielandt_proposition_three} the map $\E_A$ is strongly
irreducible: its peripheral spectrum is $\{1\}$ and its fixed point $\rho$
is unique up to scale. Since $\E_A$ is trace preserving, $1$ is a
semisimple eigenvalue, and all other eigenvalues have modulus less than
$1$. A periodic tensor is not normal, because no blocking of it is
injective.

\begin{lemma}[Convergence of the positive part]
    \label{lem:ldp_positive_part_converges}
    \lean{MPSTensor.tendsto_polarPosTensor_blockTensor_of_isNormal,
        MPSTensor.tendsto_polarPosTensor_of_tendsto_transferMap,
        MPSTensor.tendsto_polarPos_physicalMatrix_of_tendsto_transferMap}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor,
        thm:prep_fixed_point_tensor_psd, thm:prep_transfer_fixed_point_tensor,
        thm:prep_blocked_transfer_limit_normal}
    Let $P_q=(\mathcal B_q^\dagger\mathcal B_q)^{1/2}$. Then $P_q\to P_\infty$ as
    $q\to\infty$, where $P_\infty$ is read as the map
    $\ket{\alpha,\beta}\mapsto\sum_\gamma(\sqrt\rho)_{\alpha\gamma}\ket{\gamma,\beta}$,
    that is, $P_\infty=(\sqrt\rho)^{\mathsf T}\otimes\Id$.
    This is the limit in \cite[eq.~(8)]{Malz2024Preparation}.
    More generally, for any tensors $B_q$ whose transfer maps satisfy
    $\E_{B_q}(X)\to\Tr(X)\,\rho$ for every $X$, with $\rho\ge0$, the positive
    parts of the physical matrices of $B_q$ converge to
    $(\sqrt\rho)^{\mathsf T}\otimes\Id$, and the corresponding tensors
    converge to $P_\infty$.
\end{lemma}
\begin{proof}
    \leanok
    The matrix of $\mathcal B_q^\dagger\mathcal B_q$ is a rearrangement of
    $\E_A^{\,q}$:
    $(\mathcal B_q^\dagger\mathcal B_q)_{(\alpha,\beta),(\alpha',\beta')}
        =\E_A^{\,q}(\ketbra{\beta'}{\beta})_{\alpha'\alpha}$.
    By normality and the gauge~(\ref{eq:ldp_normal_gauge}),
    $\E_A^{\,q}(X)\to\Tr(X)\rho$, so the rearranged matrices converge to
    $\rho^{\mathsf T}\otimes\Id$. The square root of
    $\rho^{\mathsf T}\otimes\Id$ is $(\sqrt\rho)^{\mathsf T}\otimes\Id$,
    since the latter is positive semidefinite and squares to the former, and
    the square root is continuous on positive semidefinite matrices.
\end{proof}

\begin{definition}[Approximating state]
    \label{def:ldp_approximating_state}
    \lean{MPSTensor.approximatingTensor, MPSTensor.mpv_approximatingTensor,
        MPSTensor.mpv_approximatingTensor_norm_sq}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, def:prep_fixed_point_tensor,
        thm:prep_mpv_fixed_point_tensor, lem:mpv_rotate_physical,
        thm:ldpolar_tensor_injective, thm:prep_pair_product_norm}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} and $N=qM$, the
    \emph{approximating state} is
    $\ket{\widetilde\phi_N}=\ket{\phi_M(VP_\infty)}/\lVert\phi_M(VP_\infty)\rVert$.
    By Lemma~\ref{lem:mpv_rotate_physical} and
    Theorem~\ref{thm:prep_mpv_fixed_point_tensor}, for $M\ge1$,
    $\ket{\phi_M(VP_\infty)}=V^{\otimes M}\bigotimes_k\ket{\omega}_{R_kL_{k+1}}$.
    When $\mathcal B_q$ is injective, $V$ is an isometry and
    $\lVert\phi_M(VP_\infty)\rVert=1$, so no normalization is needed.
    This is \cite[eqs.~(9) and~(10)]{Malz2024Preparation}.
\end{definition}

\begin{lemma}[A unitary implementing the isometry]
    \label{lem:ldp_unitary_implementing_isometry}
    \lean{Matrix.exists_mem_unitaryGroup_apply_embedding_eq,
        MPSTensor.embeddedPairState,
        MPSTensor.mpv_approximatingTensor_eq_sum_embeddedPairState,
        MPSTensor.exists_unitary_mpv_approximatingTensor}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, thm:ldpolar_tensor_injective,
        def:ldp_approximating_state, def:prep_pair_product_state}
    Suppose $\mathcal B_q$ is injective, and let $\iota$ be an injective map
    from the pairs $(\gamma,\delta)$ with $0\le\gamma,\delta<D$ to the
    computational basis of $(\C^d)^{\otimes q}$. Then there is a unitary $U$
    on $(\C^d)^{\otimes q}$ with $U\ket{\iota(\gamma,\delta)}=V\ket{\gamma,\delta}$
    for all $\gamma,\delta$, and for every $M\ge1$
    \begin{align}
        \ket{\widetilde\phi_N}
         & =\Bigl(\bigotimes_{k=1}^M U_k\Bigr)
        \sum_{c}\Omega(c)\bigotimes_{k=1}^M\ket{\iota(c_k)},
        \label{eq:ldp_unitary_preparation}
    \end{align}
    where $c$ runs over the configurations $c_k=(l_k,r_k)$ and $\Omega$ is the
    pair product state~(\ref{eq:prep_pair_product}). The source takes $D=d$,
    so that $L$ and $R$ below are single sites and the remaining $\ell$ central
    sites carry $\ket{0}$; the padding of the legs for general $D$ is this
    chapter's extension. With $d\ge2$, $r=\lceil\log_dD\rceil$, $q\ge2r$, and $(\C^d)^{\otimes q}=L\otimes C\otimes R$,
    where $L$ and $R$ are the first and the last $r$ sites of the block and
    $C$ is the remaining $q-2r$ sites; $\C^D$ is embedded into $L$ and into
    $R$ as the span of the first $D$ computational basis vectors, and
    $\iota(\gamma,\delta)=\ket{\gamma}_L\ket{0}_C\ket{\delta}_R$ with
    $\ket{0}_C=\ket{0}^{\otimes(q-2r)}$. The right-hand side
    of~(\ref{eq:ldp_unitary_preparation}) is then
    $\bigl(\bigotimes_kU_k\bigr)\bigotimes_k\bigl(\ket{\omega}_{R_kL_{k+1}}\ket{0}_{C_k}\bigr)$.
    This is \cite[eqs.~(10) and~(11)]{Malz2024Preparation}.
\end{lemma}
\begin{proof}
    \leanok
    Since $V$ is an isometry, its columns $V\ket{\gamma,\delta}$ are
    orthonormal. Place them at the positions $\iota(\gamma,\delta)$ and extend
    them to an orthonormal basis of $(\C^d)^{\otimes q}$; the matrix with these
    columns is the unitary $U$. Expanding
    $\bigl(\bigotimes_kU_k\bigr)\sum_c\Omega(c)\bigotimes_k\ket{\iota(c_k)}$
    gives $\sum_c\Omega(c)\bigotimes_kV\ket{c_k}
        =V^{\otimes M}\bigotimes_k\ket{\omega}_{R_kL_{k+1}}$, which is
    $\ket{\widetilde\phi_N}$ by Definition~\ref{def:ldp_approximating_state}.
\end{proof}

\section{Exact factorizations of the isometry}

\begin{theorem}[Successive decompositions with a matrix remainder]
    \label{thm:ldp_sweep_matrix_remainder}
    \lean{MPSPreparation.exists_isometric_chain_mul,
        MPSPreparation.sum_star_mulVec_dotProduct_of_isIsometryOn,
        MPSPreparation.sum_star_eval_mulVec_dotProduct}
    \leanok
    \uses{def:seqanc_isometry_on, def:seqanc_support}
    Let $a\le D$, let $R_{-1}\in\MN{D}$ be row-supported below $a$, and let
    $P_0,\dots,P_{n-1}$ be families of $D\times D$ matrices. Then there are
    $b_0=a$ and $b_1,\dots,b_n\le D$, families $Q_0,\dots,Q_{n-1}$ such that
    $Q_p^i$ is row-supported below $b_p$, vanishes on the columns
    $\beta\ge b_{p+1}$, and $Q_p$ is isometric on its first $b_{p+1}$ columns,
    and a matrix $R$ row-supported below $b_n$, such that for every
    configuration $\sigma$
    \begin{align}
        R_{-1}\,P^\sigma
         & =Q^\sigma\,R.
        \label{eq:ldp_sweep_matrix_remainder}
    \end{align}
    Moreover $b_{p+1}\le c$ whenever every $P_p^i$ vanishes on the columns
    $\gamma\ge c$. If $Q$ is isometric on its first $b$ columns and $v,w$ are
    supported below $b$, then
    $\sum_i\langle Q^iv,Q^iw\rangle=\langle v,w\rangle$. Consequently, if
    $Q_p^i$ is row-supported below $b_p$, $Q_p$ is isometric on its first
    $b_{p+1}$ columns, and $v,w$ are supported below $b_n$, then
    \begin{align}
        \sum_\sigma\langle Q^\sigma v,Q^\sigma w\rangle
         & =\langle v,w\rangle.
        \label{eq:ldp_sweep_inner_product}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:seqanc_one_step, thm:seqanc_support_propagation}
    By induction on $n$. For $n=0$ take $R=R_{-1}$. Otherwise apply
    Theorem~\ref{thm:seqanc_one_step} to $Y^i=R_{-1}P_0^i$, which gives
    $Y^i=Q_0^iR_0$ with $R_0$ row-supported below $b_1$, and apply the
    induction hypothesis to $R_0$ and $P_1,\dots,P_{n-1}$:
    \begin{align}
        R_{-1}P_0^{\sigma_0}P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
         & =Q_0^{\sigma_0}R_0P_1^{\sigma_1}\cdots P_{n-1}^{\sigma_{n-1}}
        =Q_0^{\sigma_0}Q_1^{\sigma_1}\cdots Q_{n-1}^{\sigma_{n-1}}R.
        \notag
    \end{align}
    For one site, expand
    \begin{align}
        \sum_i\langle Q^iv,Q^iw\rangle
         & =\sum_{\beta,\beta'<b}\overline{v_\beta}w_{\beta'}
        \sum_{i,\alpha}\overline{Q^i_{\alpha\beta}}Q^i_{\alpha\beta'}
        =\sum_{\beta<b}\overline{v_\beta}w_\beta
        \notag
    \end{align}
    by the isometry condition. For the chain, write
    $Q^{\sigma}=Q_0^{\sigma_0}Q'^{\sigma'}$, where $Q'$ is the chain
    $Q_1,\dots,Q_{n-1}$ and $\sigma'=(\sigma_1,\dots,\sigma_{n-1})$. The
    vectors $Q'^{\sigma'}v$ and $Q'^{\sigma'}w$ are supported below $b_1$ by
    Theorem~\ref{thm:seqanc_support_propagation}, so the one-site identity for
    $Q_0$ and the induction hypothesis for $Q'$ give
    \begin{align}
        \sum_{\sigma_0,\sigma'}
        \langle Q_0^{\sigma_0}Q'^{\sigma'}v,Q_0^{\sigma_0}Q'^{\sigma'}w\rangle
         & =\sum_{\sigma'}\langle Q'^{\sigma'}v,Q'^{\sigma'}w\rangle
        =\langle v,w\rangle.
        \notag
    \end{align}
\end{proof}

\begin{theorem}[Sequential factorization of a matrix product isometry]
    \label{thm:ldp_sequential_factorization_general}
    \lean{MPSPreparation.exists_isometric_chain_of_eq_mul}
    \leanok
    Let $A$ have bond dimension $D$, let $G$ be a $D^2\times D^2$ matrix, and
    let $V\colon\C^D\otimes\C^D\to(\C^d)^{\otimes q}$, $q\ge1$, be an isometry
    with matrix elements
    $\bra{i_1\cdots i_q}V\ket{x}
        =\sum_{\alpha,\beta}(A^{i_1}\cdots A^{i_q})_{\alpha\beta}\,
        G_{(\alpha,\beta),x}$.
    Then there are integers $b_0=1$, $b_q=D^2$ and $b_p\le D^2$, and families
    of $b_{p-1}\times b_p$ matrices $X_p^i$, $p=1,\dots,q$, with
    $\sum_i(X_p^i)^\dagger X_p^i=\Id_{b_p}$ for every $p$, such that
    $\bra{i_1\cdots i_q}V\ket{x}=(X_1^{i_1}X_2^{i_2}\cdots X_q^{i_q})_{1x}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldp_sweep_matrix_remainder, thm:seqanc_support_propagation}
    The matrix elements of $V$ are
    $\bigl(j^{\mathsf T}(\Id_D\otimes A^{i_1})\cdots(\Id_D\otimes A^{i_q})G\bigr)_x$
    with $j=\sum_\gamma\ket{\gamma}\otimes\ket{\gamma}$. Apply
    Theorem~\ref{thm:ldp_sweep_matrix_remainder} to the row matrix of $j$ and the
    sites $\Id_D\otimes A^i$; this gives isometric sites $Q_1,\dots,Q_q$ and a
    remainder $C=RG$ with $V=Q\,C$ in the sense of
    (\ref{eq:ldp_sweep_matrix_remainder}). By
    (\ref{eq:ldp_sweep_inner_product}), $V^\dagger V=C^\dagger C$, so $C$ is an
    isometry because $V$ is. Absorb $C$ into the last site, $X_q^i=Q_q^iC$ and
    $X_p=Q_p$ for $p<q$; the last site is again isometric,
    $\sum_i C^\dagger(Q_q^i)^\dagger Q_q^iC=C^\dagger C=\Id$.
\end{proof}

\begin{theorem}[Sequential factorization]
    \label{thm:ldp_sequential_factorization}
    \lean{MPSPreparation.exists_isometric_chain_polarIsoMatrix}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition}
    Suppose $\mathcal B_q$ is injective, $q\ge1$, and let $V$ be as in
    Theorem~\ref{thm:ldpolar_tensor_decomposition}. There are integers
    $D'_1=D^2$, $D'_{q+1}=1$, and $D'_k\le D^2$, and matrices
    $W_k^i\in\C^{D'_{k+1}\times D'_k}$ for $k=1,\dots,q$ and $i=1,\dots,d$, with
    \begin{align}
        \sum_{i=1}^d\bigl(W_k^i\bigr)^\dagger W_k^i
         & =\Id_{D'_k}
        \quad\text{and}\quad
        \bra{i_1\cdots i_q}V\ket{x}
        =\bigl(W_q^{i_1}W_{q-1}^{i_2}\cdots W_1^{i_q}\bigr)_{1x}
        \label{eq:ldp_sequential_factorization}
    \end{align}
    for every $k$, every configuration $i_1\cdots i_q$ and every $x$, where $\C^{D'_1}$ is
    identified with $\C^D\otimes\C^D$. This is the matrix product form of
    \cite[eqs.~(13)--(15)]{Malz2024Preparation}, where the isometry carrying the
    input is written $\widetilde C=V_1$. The block length $q\ge1$ is implicit
    in the source, which draws at least one site; the empty block $q=0$ is
    excluded, see~\cite{gap:mswc24_sequential_factorization_positive_block_length}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldpolar_tensor_injective,
        thm:ldp_sequential_factorization_general}
    Since $\mathcal B_q$ is injective, $P$ is positive definite and
    $V=\mathcal B_qP^{-1}$, so
    $\bra{i_1\cdots i_q}V\ket{x}
        =\sum_{\alpha,\beta}(A^{i_1}\cdots A^{i_q})_{\alpha\beta}(P^{-1})_{(\alpha,\beta),x}$.
    This is an open-boundary matrix product map with bond dimension at most
    $D^2$: the site matrices are $\Id_D\otimes A^i$ (the source writes
    $A^i\otimes\Id_D$; the order of the two bond factors only permutes the
    bond index), the two bond factors are
    joined at the left end, and $P^{-1}$ is absorbed into the right end
    \cite[eq.~(13)]{Malz2024Preparation}. Theorem~\ref{thm:ldp_sequential_factorization_general}
    with $G=P^{-1}$ makes every site isometric by successive decompositions
    from the left \cite[eq.~(14)]{Malz2024Preparation}; the remainder at the
    input end is an isometry because $V$ is
    \cite[eq.~(15)]{Malz2024Preparation}. The sites of that theorem are
    numbered from the output end and those here from the input end, so its
    $p$-th site is the site $q+1-p$ here, $X_p^i=W_{q+1-p}^i$ for
    $p=1,\dots,q$, which turns $(X_1^{i_1}\cdots X_q^{i_q})_{1x}$ into
    $(W_q^{i_1}\cdots W_1^{i_q})_{1x}$ and $b_p$ into $D'_{q+1-p}$.
\end{proof}

\begin{remark}[Sequential factorization as a composition of isometries]
    \label{rem:ldp_sequential_factorization_composition}
    \uses{thm:ldp_sequential_factorization}
    The source states Theorem~\ref{thm:ldp_sequential_factorization} as an
    ordered composition. With the matrices of
    (\ref{eq:ldp_sequential_factorization}), let
    $W_k\colon\C^{D'_k}\to\C^{D'_{k+1}}\otimes\C^d$ be the map with
    $\bra{\alpha,i}W_k\ket{\beta}=(W_k^i)_{\alpha\beta}$. The first identity in
    (\ref{eq:ldp_sequential_factorization}) says that $W_k$ is an isometry,
    $W_k^\dagger W_k=\sum_i(W_k^i)^\dagger W_k^i=\Id_{D'_k}$. Expanding the
    composition in the product basis, with the tensor factor $\C^d$ of $W_k$
    placed at the site $q+1-k$ of the block and $\C^{D'_{q+1}}=\C$, the second
    identity becomes
    \begin{align}
        V
         & =\bigl(W_q\otimes\Id_{d}^{\otimes(q-1)}\bigr)\cdots
        \bigl(W_2\otimes\Id_d\bigr)W_1,
        \label{eq:ldp_sequential_factorization_composition}
    \end{align}
    which is \cite[eqs.~(13)--(15)]{Malz2024Preparation} as printed. This
    operator form is not formalized; the formal statement is the matrix product
    form (\ref{eq:ldp_sequential_factorization}).
\end{remark}

\begin{theorem}[Uniqueness of the polar decomposition]
    \label{thm:ldp_polar_uniqueness}
    \lean{Matrix.polarPos_eq_of_eq_mul, Matrix.polarSupport_eq_of_eq_mul,
        Matrix.polarIso_eq_of_eq_mul,
        Matrix.mul_eq_self_of_conjTranspose_mul_self_eq,
        Matrix.polarPos_eq_of_mul_self_eq, Matrix.polarSupport_eq_of_range_eq}
    \leanok
    \uses{thm:ldpolar_decomposition, thm:ldpolar_support}
    Let $M=WQ$ with $Q\ge0$, and suppose $W^\dagger W=E$ for the orthogonal
    projector $E$ onto the range of $Q$. Then $Q=(M^\dagger M)^{1/2}$, $E$ is the
    support projector of $M$, and $W$ is the partial isometry of the polar
    decomposition of $M$. In particular, every positive semidefinite $Q$ with
    $Q^2=M^\dagger M$ is the positive part of $M$, and every orthogonal projector
    with the range of that positive part is the support projector. This is the
    step needed to identify the layers of
    \cite[eq.~(16)]{Malz2024Preparation} with the isometry of the blocked
    tensor, which the source asserts (``to the same effect'') without proof.
\end{theorem}
\begin{proof}
    \leanok
    Since $EQ=Q$ and $WE=W$, $M^\dagger M=QE\,Q=Q^2$, so $Q$ is the positive
    square root of $M^\dagger M$. Two orthogonal projectors with the same range
    coincide, which identifies $E$. Finally $(V-W)Q=M-M=0$ for the partial
    isometry $V$ of $M$, and both $V$ and $W$ are fixed by $E$, whose range lies
    in the range of $Q$; hence $V=W$.
\end{proof}

\begin{theorem}[Tree factorization]
    \label{thm:ldp_tree_factorization}
    \lean{MPSTensor.treeTensor, MPSTensor.treePosTensor,
        MPSTensor.treeIsoMatrix, MPSTensor.blockKron,
        MPSTensor.blockTensor_pairRegroupEquiv,
        MPSTensor.blockKron_polarSupportMatrix_mul_physicalMatrix_blockTensor,
        MPSTensor.blockTensor_eq_rotatePhysical_treeIsoMatrix,
        MPSTensor.physicalMatrix_blockTensor_eq_treeIsoMatrix_mul,
        MPSTensor.conjTranspose_blockKron_polarIsoMatrix_mul_self,
        MPSTensor.conjTranspose_treeIsoMatrix_mul_self,
        MPSTensor.polarIsoMatrix_blockTensor_eq_treeIsoMatrix,
        MPSTensor.polarPosTensor_blockTensor_eq_treePosTensor}
    \leanok
    \uses{thm:ldpolar_tensor_decomposition, thm:ldp_polar_uniqueness}
    Put $T_0=A$ and, for $j\ge1$, let $\mathcal B^{(j)}=V^{(j)}P_j$ be the polar
    decomposition of the two-site blocked map of $T_{j-1}$, with
    $V^{(j)\dagger}V^{(j)}=\Pi_j$ the support projector of $P_j$, and let $T_j$
    be the tensor of $P_j$ (Definition~\ref{def:ldpolar_tensor}). Then for
    $q=2^k$ with $k\ge1$,
    \begin{align}
        \mathcal B_q
         & =\Bigl(V^{(1)}\Bigr)^{\otimes2^{k-1}}
        \Bigl(V^{(2)}\Bigr)^{\otimes2^{k-2}}\cdots V^{(k)}\,P_k.
        \label{eq:ldp_tree_factorization}
    \end{align}
    Here $V^{(1)}$ maps $\C^{D^2}$ to $(\C^d)^{\otimes2}$ and each $V^{(j)}$ with
    $j\ge2$ maps $\C^{D^2}$ to $\C^{D^2}\otimes\C^{D^2}$; the Kronecker powers
    act on the regrouping of the $q$ sites into $2^{k-1}$ neighbouring pairs,
    then pairs of pairs, and so on. Each layer
    $\bigl(V^{(j)}\bigr)^{\otimes m}$ is a partial isometry with initial
    projector $\Pi_j^{\otimes m}$. The product of the
    layers is a partial isometry with initial space the range of $P_k$, so by
    uniqueness of the polar decomposition it equals $V$ and $P_k=P_q$. The
    factorization (\ref{eq:ldp_tree_factorization}) is
    \cite[eq.~(16)]{Malz2024Preparation}, where the tree is said to act ``to the
    same effect'' as blocking directly; the identification with the polar
    decomposition of the blocked map is left implicit there and is supplied
    here.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:physical_rotation_mul, lem:ldpolar_physical_matrix_rotate,
        lem:physical_rotation_state, thm:block_tensor_rotate_physical}
    Blocking commutes with maps on the physical leg: blocking $C_W$ over $m$
    sites gives the tensor $C$ blocked over $m$ sites, rotated by
    $W^{\otimes m}$ (Theorem~\ref{thm:block_tensor_rotate_physical}). Blocking
    $2^{k}$ sites is blocking $2^{k-1}$ pairs of the two-site blocked tensor,
    and the outer layer composes with the maps of the inner tree by
    $(A_{W'})_W=A_{WW'}$
    (Lemma~\ref{lem:physical_rotation_mul}), so induction on $k$ gives
    (\ref{eq:ldp_tree_factorization}).
    The range of the two-site blocked map of $T_{j}$ lies in
    $\operatorname{ran}P_{j}\otimes\operatorname{ran}P_{j}$, because
    $\Pi_j$ fixes the tensor of $P_j$; so $\Pi_j^{\otimes2^{k-j}}$ fixes the
    product of the later layers, and the product of all layers is a partial
    isometry with initial projector the support projector of $P_k$.
    Theorem~\ref{thm:ldp_polar_uniqueness} then identifies it with $V$ and $P_k$
    with $P_q$.
\end{proof}

\section{Approximation error}

For a normal tensor $A$, let $\lambda_2$ be an eigenvalue of $\E_A$ of
largest modulus other than the Perron eigenvalue $1$. By the remark after
(\ref{eq:ldp_normal_gauge}), $\lvert\lambda_2\rvert<1$. The
\emph{correlation length} of $A$ is the quantity
$\xi=-1/\log\lvert\lambda_2\rvert$ of Definition~\ref{def:correlation_length}
at this $\lambda_2$. It is finite, it is positive exactly when
$\lambda_2\neq0$, and $\lvert\lambda_2\rvert=e^{-1/\xi}$ in that case.
As in the previous section, $A$ is in the gauge~(\ref{eq:ldp_normal_gauge})
throughout this section. The estimates below use $\lambda_2$ only through the
bound $\lvert\mu\rvert\le\lvert\lambda_2\rvert$ for the eigenvalues $\mu\ne1$
of $\E_A$. When $\lambda_2=0$, so that $\xi=0$, they are applied with any
$\lambda_2'$ satisfying $0<\lvert\lambda_2'\rvert<1$ in place of $\lambda_2$,
and $\xi$ is read as the correlation length at $\lambda_2'$.

\begin{lemma}[Overlap of the positive part with the fixed point]
    \label{lem:ldp_positive_part_overlap}
    \lean{MPSTensor.exists_abs_one_sub_norm_mpvOverlap_polarPosTensor_le_mul,
        MPSTensor.exists_abs_one_sub_norm_mpvOverlap_polarPosTensor_le,
        MPSTensor.exists_norm_mpvOverlap_polarPosTensor_sub_one_le}
    \leanok
    \uses{def:prep_fixed_point_tensor, thm:ldpolar_blocked_transfer,
        lem:ldp_positive_part_converges, def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, and let $0<\gamma<1/2$.
    Then, for $N=qM$,
    \begin{align}
        \Bigl\lvert1-\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\bigr\rvert\Bigr\rvert
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi}\right).
        \label{eq:ldp_positive_part_overlap}
    \end{align}
    This is \cite[Supplemental Material, eq.~(S29)]{Piroli2021LOCC}, quoted as
    \cite[eq.~(S9)]{Malz2024Preparation}. The constant depends only on $A$,
    $\rho$ and $\gamma$, and the bound reads $\le Cy$ for all $q\ge0$ and
    $M\ge1$, where $y=M\,e^{-\gamma q/\xi}$ equals $(N/q)\,e^{-\gamma q/\xi}$
    when $q\ge1$. Explicitly, there is $C>0$ with
    $\bigl\lvert1-\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle\rvert\bigr\rvert
        \le Cy\,e^{Cy}$ for all $q\ge0$ and $M\ge1$. The same bound holds for the
    overlap itself:
    $\bigl\lvert\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle-1\bigr\rvert\le Cy\,e^{Cy}$.
\end{lemma}
\begin{proof}
    \leanok
    Write $\E_\infty(X)=\Tr(X)\rho$ and $r=e^{-2\gamma/\xi}$, so that
    $r=\lvert\lambda_2\rvert^{2\gamma}>\lvert\lambda_2\rvert$ when
    $0<\lvert\lambda_2\rvert<1$, because $2\gamma<1$. The eigenvalues of
    $\E_A-\E_\infty$ other than $0$ are eigenvalues of $\E_A$ different from
    $1$, so the spectral radius of $\E_A-\E_\infty$ is less than $r$, and
    $\E_A^{\,n}-\E_\infty=(\E_A-\E_\infty)^n$ for $n\ge1$ gives
    $\lVert\E_A^{\,n}(X)-\Tr(X)\rho\rVert\le Kr^n\lVert X\rVert$. By the
    rearrangement in the proof of
    Lemma~\ref{lem:ldp_positive_part_converges},
    $\lVert P_q^2-P_\infty^2\rVert\le K_1r^q$, and since
    $\lVert\sqrt X-\sqrt Y\rVert\le\lVert X-Y\rVert^{1/2}$ for $X,Y\ge0$,
    $\lVert P_q-P_\infty\rVert\le K_2e^{-\gamma q/\xi}$.

    Let $\tau_q$ and $\tau_\infty$ be the transfer matrices of the mixed
    transfer maps of $P_q$ and of $P_\infty$ against $P_\infty$. Then
    $\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle=\Tr\tau_q^M$,
    $1=\Tr\tau_\infty^M$, $\tau_\infty^2=\tau_\infty$, and
    $\delta=\lVert\tau_q-\tau_\infty\rVert\le K_3e^{-\gamma q/\xi}$, since
    $\tau_q-\tau_\infty$ depends linearly on $P_q-P_\infty$. Writing
    $\tau_q^M-\tau_\infty^M=\sum_{k<M}\tau_q^k(\tau_q-\tau_\infty)\tau_\infty^{M-1-k}$
    and bounding $\lVert\tau_q^k\rVert\le c+\lVert\tau_q^k-\tau_\infty^k\rVert$
    with $c\ge\max(1,\lVert\tau_\infty\rVert)$, induction on $M$ gives
    $\lVert\tau_q^M-\tau_\infty^M\rVert\le c\bigl((1+c\delta)^M-1\bigr)
        \le c\,Mc\delta\,e^{Mc\delta}$.
    Since $\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle-1=\Tr\tau_q^M-\Tr\tau_\infty^M$,
    this bounds the overlap itself, and hence also its modulus, which proves the
    explicit bounds. When $y\le1$ it is at most $Ce^{C}y$.
    When $y>1$, the overlap is at most
    $\lVert\phi_M(P_q)\rVert=\lVert\phi_N(A)\rVert$, and these norms are
    bounded because $\lVert\phi_N(A)\rVert^2=\Tr\E_A^{\,N}$ converges to $1$.
\end{proof}

\begin{theorem}[Approximation error, normal case]
    \label{thm:ldp_approximation_error_normal}
    \lean{MPSTensor.exists_approximationError_le_mul,
        MPSTensor.exists_approximationError_le, MPSTensor.normalizedMPVState,
        MPSTensor.approximatingMPVState}
    \leanok
    \uses{lem:ldp_positive_part_overlap, def:ldp_approximating_state,
        thm:ldpolar_blocked_state, thm:ldpolar_tensor_injective,
        def:correlation_length}
    Let $A$ be normal in the gauge~(\ref{eq:ldp_normal_gauge}) with correlation
    length $\xi$, let $\ket{\phi_N}$ be the
    normalized vector (\ref{eq:ldp_ti_mps}), and let $0<\gamma<1/2$. For
    $N=qM$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)
         & =O\!\left(\frac Nq\,e^{-\gamma q/\xi}\right).
        \label{eq:ldp_approximation_error_normal}
    \end{align}
    This is \cite[Lemma~1 and Lemma~1$'$(i)]{Malz2024Preparation}. The constant
    depends only on $A$, $\rho$ and $\gamma$, and the bound reads $\le Cy$ for
    all $q\ge0$ and $M\ge1$, where $y=M\,e^{-\gamma q/\xi}$ equals
    $(N/q)\,e^{-\gamma q/\xi}$ when $q\ge1$. Explicitly, there is $C>0$ with
    $\varepsilon(\widetilde\phi_N,\phi_N)\le Cy\,e^{Cy}$ for all $q\ge0$ and
    $M\ge1$.
\end{theorem}
\begin{proof}
    \leanok
    Write $c_N=\lVert\phi_N(A)\rVert$. For $q$ at least the injectivity length
    of $A$, the blocked tensor is injective, $V$ is an isometry, and
    $\ket{\widetilde\phi_N}=V^{\otimes M}\ket{\phi_M(P_\infty)}$ is
    normalized. Since $\ket{\phi_N(A)}=V^{\otimes M}\ket{\phi_M(P_q)}$ by
    Theorem~\ref{thm:ldpolar_blocked_state}, this gives
    $c_N\braket{\widetilde\phi_N}{\phi_N}
        =\bra{\phi_M(P_\infty)}\phi_M(P_q)\rangle$. The triangle inequality
    \begin{align}
        \varepsilon
         & \le\bigl\lvert1-c_N\lvert\braket{\widetilde\phi_N}{\phi_N}\rvert\bigr\rvert
        +\lvert c_N-1\rvert\,\lvert\braket{\widetilde\phi_N}{\phi_N}\rvert
        \notag
    \end{align}
    bounds the first term by Lemma~\ref{lem:ldp_positive_part_overlap}. For the
    second, $\lvert c_N-1\rvert\le\lvert c_N^2-1\rvert
        =\lvert\Tr\E_A^{\,N}-1\rvert\le Ke^{-2\gamma N/\xi}$, and
    $e^{-2\gamma N/\xi}\le e^{-\gamma q/\xi}$ because $N\ge q$. The finitely
    many smaller $q$, and the case $y>1$, are covered by
    $\varepsilon\le1$.
\end{proof}
%% The source's proof of Lemma 1'(i) drops the normalization term
%% |c_N - 1| = O(exp(-N/xi)) using q = o(N). Since N >= q, that term is at most
%% a constant times exp(-gamma q/xi) <= (N/q) exp(-gamma q/xi), so the
%% statement above carries no condition relating q and N, as in the source's
%% statement.

In particular, when $\xi>0$, the choice $q=\lceil\xi(1+\eta)\ln N/\gamma\rceil$
gives $e^{-\gamma q/\xi}\le N^{-(1+\eta)}$ and hence $\varepsilon=O(N^{-\eta})$
for every $\eta>0$. The source takes
$q=\lceil2\xi(1+\eta)\ln N\rceil$, which corresponds to the endpoint
$\gamma=1/2$ that the estimate above does not cover; hence the factor
$1/\gamma$ in the block length here
(compare \cite[after Lemma~1]{Malz2024Preparation}).

\section{Depth of the preparation}

\begin{definition}[Local circuit of depth $T$]
    \label{def:ldp_local_circuit}
    \lean{MPSPreparation.IsLocalCircuitOfDepth, MPSPreparation.IsPreparedInDepth,
        MPSPreparation.Layer, MPSPreparation.circuitOp,
        MPSPreparation.supportedOperators}
    \leanok
    A \emph{local circuit of depth $T$} on a chain of $N$ sites is a product of
    $T$ layers, each a tensor product of unitaries acting on disjoint pairs of
    neighbouring sites. A state is \emph{prepared in depth $T$} if it equals a
    local circuit of depth $T$ applied to a product state.

    The chain is a ring: the neighbouring pairs are $\{k,k+1\}$ with indices
    modulo $N$, and a circuit whose gates avoid the pair $\{N,1\}$ is a circuit
    on the open chain. An operator \emph{acts on} a set of sites $X$ if it
    lies in $\MN{d}^{\otimes X}\otimes\mathbb 1$, the span of the products
    $\bigotimes_i m_i$ with $m_i=\mathbb 1$ for $i\notin X$. Two sets of sites
    are at distance larger than $s$ if no site of one is within ring distance
    $s$ of a site of the other.
\end{definition}

\begin{lemma}[Light cone of a local circuit]
    \label{lem:ldp_light_cone}
    \uses{def:ldp_local_circuit}
    \lean{MPSPreparation.conj_circuitOp_mem_supportedOperators}
    \leanok
    Let $U$ be a local circuit of depth $T$ and let $Z$ act on a set of sites
    $X$. Then $U^\dagger ZU$ acts on the sites within ring distance $T$ of $X$.
    This is the finite light cone of \cite[Theorem~1]{Malz2024Preparation}.
\end{lemma}
\begin{proof}
    \leanok
    By induction on the layers it suffices to treat one layer $W$. Split its
    gates into those whose pair meets $X$, with product $G$, and the others,
    with product $H$. The gates commute, so $W=HG$. The operator $H$ is unitary
    and acts on the complement of $X$, so it commutes with $Z$ and
    $W^\dagger ZW=G^\dagger ZG$. The pairs of $G$ lie within distance $1$ of
    $X$, and products and adjoints of operators acting on a set act on it.
\end{proof}

\begin{lemma}[No connected correlations beyond the light cone]
    \label{lem:ldp_light_cone_correlations}
    \uses{lem:ldp_light_cone}
    \lean{MPSPreparation.expect_mul_eq_of_isPreparedInDepth,
        MPSPreparation.expect_mul_mul_expect_one_of_isPreparedInDepth}
    \leanok
    Let $\ket{\psi}$ be a unit vector prepared in depth $T$, and let $Z$, $Z'$
    act on sets of sites at distance larger than $2T$. Then
    $\bra{\psi}ZZ'\ket{\psi}=\bra{\psi}Z\ket{\psi}\bra{\psi}Z'\ket{\psi}$.
    This is the statement in the proof of
    \cite[Theorem~1]{Malz2024Preparation} that every connected correlation of
    such operators vanishes.
\end{lemma}
\begin{proof}
    \leanok
    Write $\ket{\psi}=U\ket{\pi}$ with $\ket{\pi}=\bigotimes_i\ket{v_i}$. Since
    $UU^\dagger=\mathbb 1$,
    $\bra{\psi}ZZ'\ket{\psi}=\bra{\pi}(U^\dagger ZU)(U^\dagger Z'U)\ket{\pi}$,
    and by Lemma~\ref{lem:ldp_light_cone} the two factors act on disjoint sets
    of sites. For products $\bigotimes_i m_i$ and $\bigotimes_i m'_i$ with
    disjoint supports the expectation in $\ket{\pi}$ is
    $\prod_i\bra{v_i}m_im'_i\ket{v_i}$, and at each site one of $m_i$, $m'_i$
    is the identity; bilinearity then gives
    $\bra{\pi}AB\ket{\pi}\braket{\pi}{\pi}
        =\bra{\pi}A\ket{\pi}\bra{\pi}B\ket{\pi}$ for $A$, $B$ acting on
    disjoint sets.
\end{proof}

\begin{theorem}[Preparation in depth $O(\log(N/\varepsilon))$]
    \label{thm:ldp_depth_upper_bound}
    \uses{def:ldp_local_circuit, lem:ldp_unitary_implementing_isometry,
        thm:ldp_sequential_factorization, thm:ldp_tree_factorization,
        thm:ldp_approximation_error_normal}
    Let $A$ be normal. There is a constant $c$ depending only on $A$ such that,
    for every $N\ge2$ and $0<\varepsilon\le1$, a state $\ket{\psi_N}$ with
    $\varepsilon(\psi_N,\phi_N)\le\varepsilon$ is prepared in depth at most
    $c\log(N/\varepsilon)$. This is \cite[eq.~(1)]{Malz2024Preparation}.
\end{theorem}

%% R_k are realized on groups of ceil(log_d D) sites as in
%% Lemma lem:ldp_unitary_implementing_isometry; the depth estimate is

The correlation estimate below rests on the following lemmas, which are
proved first. For an operator $X$ on $(\C^d)^{\otimes L}$ write $\E_X$ for the map
$Z\mapsto\sum_{w,w'}\bra{w'}X\ket{w}A^wZ(A^{w'})^\dagger$ on $\MN{D}$, where
$w,w'$ run over words of length $L$ and $A^w$ is the product of the matrices
of $A$ along $w$.

\begin{lemma}[Inserted transfer maps of products and adjoints]
    \label{lem:ldp_observable_transfer_products}
    \lean{MPSTensor.physicalObservableTransfer_apply, MPSTensor.appendObservable,
        MPSTensor.appendObservable_apply, MPSTensor.appendObservable_append,
        MPSTensor.appendObservable_mul, MPSTensor.appendObservable_one_one,
        MPSTensor.physicalObservableTransfer_appendObservable,
        MPSTensor.physicalObservableTransfer_one,
        MPSTensor.physicalObservableTransfer_conjTranspose,
        MPSTensor.inner_mpvState_toEuclideanLin}
    \leanok
    \uses{def:transfer_map}
    For operators $X,X'$ on $(\C^d)^{\otimes L_1}$ and $Y,Y'$ on
    $(\C^d)^{\otimes L_2}$, with $X\otimes Y$ acting on $L_1+L_2$ consecutive
    sites,
    \begin{align}
        (X\otimes Y)(X'\otimes Y')
         & =XX'\otimes YY',
         &
        \E_{X\otimes Y}
         & =\E_X\,\E_Y,
        \notag                       \\
        \E_{\Id}
         & =\E_A^{\,L},
         &
        \E_{X^\dagger}(Z)
         & =\E_X(Z^\dagger)^\dagger.
        \notag
    \end{align}
    For an operator $O$ on $(\C^d)^{\otimes N}$ and the unnormalized vector
    $\ket{\phi_N(A)}$ of (\ref{eq:ldp_ti_mps}),
    $\bra{\phi_N(A)}O\ket{\phi_N(A)}=\operatorname{tr}\E_O$, the trace of a
    linear map on $\MN{D}$.
\end{lemma}
\begin{proof}
    \leanok
    A word of length $L_1+L_2$ is the concatenation $ww'$ of words of lengths
    $L_1$ and $L_2$, with $A^{ww'}=A^wA^{w'}$ and
    $\bra{ww'}X\otimes Y\ket{vv'}=\bra{w}X\ket{v}\bra{w'}Y\ket{v'}$; summing
    gives $\E_{X\otimes Y}=\E_X\E_Y$, and $\E_{\Id}=\E_A^{\,L}$ follows by
    induction on $L$. The adjoint formula is the conjugate of the defining sum.
    Writing $\braket{w}{\phi_N(A)}=\Tr A^w$ and expanding
    $\sum_{w,w'}\overline{\Tr A^{w'}}\bra{w'}O\ket{w}\Tr A^w$ in matrix units
    gives the trace of $\E_O$.
\end{proof}

\begin{lemma}[Every map is an inserted transfer map]
    \label{lem:ldp_observable_transfer_surjective}
    \lean{MPSTensor.exists_physicalObservableTransfer_eq,
        MPSTensor.exists_isHermitian_physicalObservableTransfer_eq}
    \leanok
    \uses{def:transfer_map}
    Suppose the products of $L$ matrices of $A$ span $\MN{D}$. Then every
    linear map $\Phi$ on $\MN{D}$ equals $\E_X$ for some operator $X$ on
    $(\C^d)^{\otimes L}$. For $L=1$ and injective $A$ this is the step
    ``we can always choose $\mathcal O$ (and $\mathcal O'$), such that the
    corresponding transfer matrix $E_{\mathcal O}=\ket{A}\bra{B}$ for
    arbitrary $A,B$ (up to a normalization constant)'' in the proof of
    \cite[Lemma~2]{Malz2024Preparation}. If moreover
    $\Phi(Z^\dagger)=\Phi(Z)^\dagger$ for every $Z$, then $X$ can be chosen
    Hermitian.
\end{lemma}
\begin{proof}
    \leanok
    Choose coefficients $c^{ab}_w$ with $\sum_wc^{ab}_wA^w=e_{ab}$. The operator
    with matrix elements $\bra{w'}X\ket{w}=c^{ab}_w\overline{c^{ce}_{w'}}$ gives
    $\E_X(Z)=e_{ab}Ze_{ec}=Z_{be}\,e_{ac}$. These maps span all linear maps on
    $\MN{D}$, and $X\mapsto\E_X$ is linear. If $\Phi$ preserves adjoints and
    $\E_X=\Phi$, then by
    Lemma~\ref{lem:ldp_observable_transfer_products} also
    $\E_{X^\dagger}=\Phi$, so the Hermitian part $(X+X^\dagger)/2$ has the
    same map.
\end{proof}

\begin{lemma}[Two windows on the ring]
    \label{lem:ldp_two_window_trace}
    \lean{MPSTensor.chainWindowOperator_apply,
        MPSTensor.chainWindowOperator_add_left,
        MPSTensor.chainWindowOperator_add_right,
        MPSTensor.chainWindowOperator_self, MPSTensor.chainWindowOperator_one,
        MPSTensor.chainWindowOperator_zero_eq_appendObservable,
        MPSTensor.chainWindowOperator_add_eq_appendObservable,
        MPSTensor.inner_mpvState_chainWindowOperator_mul,
        MPSTensor.inner_mpvState_chainWindowOperator_mul_eq_physicalTwoPointExpectation,
        MPSTensor.mpvExpectation_eq_div}
    \leanok
    \uses{lem:ldp_observable_transfer_products}
    Let $X,Y$ be operators on $(\C^d)^{\otimes L}$ with $L\ge1$, let
    $m,n\ge0$ and $N=2L+m+n$, and let $X_1$ act on the sites $1,\dots,L$ and
    $Y_{L+m+1}$ on the sites $L+m+1,\dots,2L+m$ of a ring of $N$ sites. Then
    \begin{align}
        \bra{\phi_N(A)}X_1Y_{L+m+1}\ket{\phi_N(A)}
         & =\operatorname{tr}\bigl(\E_X\,\E_A^{\,m}\,\E_Y\,\E_A^{\,n}\bigr).
        \notag
    \end{align}
    The expectation in the normalized vector $\ket{\phi_N}$ is
    $\bra{\phi_N(A)}O\ket{\phi_N(A)}/\braket{\phi_N(A)}{\phi_N(A)}$.
\end{lemma}
\begin{proof}
    \leanok
    An operator on a window that does not wrap around the ring is the tensor
    product of the identity on the sites before it, the operator on the window,
    and the identity on the sites after it. Hence $X_1Y_{L+m+1}$ is
    $X\otimes\Id\otimes Y\otimes\Id$ over the blocks of lengths $L,m,L,n$, and
    Lemma~\ref{lem:ldp_observable_transfer_products} gives the trace formula.
\end{proof}

\begin{lemma}[The limit correlator at an eigenvector]
    \label{lem:ldp_limit_correlator_eigenvalue}
    \lean{MPSTensor.limitCorrelator, MPSTensor.exists_limitCorrelator_eq_pow,
        MPSTensor.limitCorrelator_smul_smul, MPSTensor.limitCorrelator_zero_left,
        MPSTensor.limitCorrelator_zero_right,

    and let $R\neq0$ satisfy $\E_A(R)=\lambda R$ with $\lambda\neq1$. Then there
    are operators $X,Y$ on $(\C^d)^{\otimes L}$ such that for every $t\ge0$
    \begin{align}
        \Tr\Bigl(\E_X\bigl((\E_A-P)^t\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr)
         & =\lambda^t.
        \notag
    \end{align}
    When $\rho$ is the fixed point of the gauge (\ref{eq:ldp_normal_gauge}),
    the left side is the limit correlator (\ref{eq:ldp_limit_correlator}) at
    $t=s-L-1$; it is bilinear in $(X,Y)$. This is the choice of observables in the proof of
    \cite[Lemma~2]{Malz2024Preparation}, without the source's requirement that
    the one-point functions vanish.
\end{lemma}
\begin{proof}
    \leanok
    \uses{lem:ldp_observable_transfer_surjective}
    Trace preservation gives $\lambda\Tr R=\Tr R$, so $\Tr R=0$, $P(R)=0$, and
    $(\E_A-P)^tR=\lambda^tR$. By
    Lemma~\ref{lem:ldp_observable_transfer_surjective} take
    $\E_Y(Z)=\Tr(Z)\,R$ and $\E_X(Z)=(Z_{ij}/R_{ij})\,\rho$ for an entry
    $R_{ij}\neq0$.
\end{proof}

\begin{lemma}[Exponential sums on windows]

\begin{lemma}[Expectations in overlapping states]
    \label{lem:ldp_expectation_overlap}
    \lean{LinearMap.IsSymmetric.norm_inner_mul_norm_sub_le}
    \leanok
    For a Hermitian operator $Z$ and unit vectors $\phi,\psi$, with
    $\sigma_\chi(Z)=\lVert(Z-\langle Z\rangle_\chi)\chi\rVert$, the inequality
    (\ref{eq:ldp_expectation_overlap}) holds:
    $\lvert\braket{\phi}{\psi}\rvert\,
        \lvert\langle Z\rangle_\phi-\langle Z\rangle_\psi\rvert
        \le\sigma_\phi(Z)+\sigma_\psi(Z)$.
\end{lemma}
\begin{proof}
    \leanok

    $(\langle Z\rangle_\phi-\langle Z\rangle_\psi)\braket{\phi}{\psi}

        -\bra{\phi}(Z-\langle Z\rangle_\phi)\ket{\psi}$; apply the
    Cauchy--Schwarz inequality to both terms.
\end{proof}

\begin{lemma}[Hermitian observables for the second eigenvalue]
    \label{lem:ldp_hermitian_observables}
    \lean{MPSTensor.exists_hermitian_vector_functional_compl_pow,
        MPSTensor.exists_normalized_observables_limitCorrelator}
    \leanok
    \uses{lem:ldp_limit_correlator_eigenvalue,
        lem:ldp_observable_transfer_surjective}
    Suppose the products of $L$ matrices of $A$ span $\MN{D}$ and
    $\sum_i(A^i)^\dagger A^i=\Id$, let $\rho$ be Hermitian with $\Tr\rho=1$,
    put $P(Z)=\Tr(Z)\,\rho$, and let $\lambda\neq0,1$ be an eigenvalue of
    $\E_A$. Then there are Hermitian operators $\mathcal O,\mathcal O'$ on
    $(\C^d)^{\otimes L}$ of norm one, an integer $K\in\{1,2\}$ with $K=1$ when
    $\lambda$ is real, distinct complex numbers $\mu_1,\dots,\mu_K$ of modulus
    one and a nonzero $a\in\C^K$ such that for every $t\ge0$
    \begin{align}
        \Tr\Bigl(\E_{\mathcal O}\bigl((\E_A-P)^t
        \bigl(\E_{\mathcal O'}(\rho)\bigr)\bigr)\Bigr)
         & =\lvert\lambda\rvert^t\sum_{j=1}^Ka_j\mu_j^{\,t}.
        \notag
    \end{align}
    This replaces the choice of observables in the proof of
    \cite[Lemma~2]{Malz2024Preparation} by one with Hermitian observables: for
    non-real $\lambda$ the eigenvector is not Hermitian, and a Hermitian
    observable pairs it with its adjoint, an eigenvector for $\bar\lambda$.
\end{lemma}
\begin{proof}
    \leanok
    This is the second and third paragraph of the proof of
    Theorem~\ref{thm:ldp_decaying_correlations}: a Hermitian vector $v$ and a
    functional $\ell$ with $\ell(Z^\dagger)=\overline{\ell(Z)}$ are built from
    an eigenvector $R$ and its adjoint, the maps $Z\mapsto\Tr(Z)\,v$ and
    $Z\mapsto\ell(Z)\,\rho$ are realized by Hermitian observables through
    Lemma~\ref{lem:ldp_observable_transfer_surjective}, and scaling to norm one
    multiplies $a$ by a positive constant.
\end{proof}

\begin{lemma}[Finite-size corrections]
    \label{lem:ldp_finite_size_correction}
    \lean{MPSTensor.exists_compl_pow_bound,
        MPSTensor.mpvConnectedCorrelator_eq_of_compl,
        Complex.norm_div_sub_div_mul_div_sub_le,
        Matrix.norm_linearMap_trace_le_mul_norm,
        MPSTensor.exp_neg_div_correlationLength_le_norm_pow}
    \leanok
    \uses{lem:ldp_two_window_trace, thm:wielandt_proposition_three,
        def:correlation_length}
    Let $A$ be in the gauge (\ref{eq:ldp_normal_gauge}), with the products of
    $L\ge1$ matrices spanning $\MN{D}$, let $P(Z)=\Tr(Z)\,\rho$, and let
    $\lambda_2$ be an eigenvalue of $\E_A$ of largest modulus among those
    different from $1$.
    \begin{enumerate}
        \item For every $r>\lvert\lambda_2\rvert$ there is $C_r>0$ with
              $\lVert(\E_A-P)^n\rVert\le C_rr^n$ for all $n\ge0$.
        \item For $t,n\ge1$ and $N=2L+t+n$, the connected correlator
              $G_N(X,Y;L+t+1)$ of (\ref{eq:ldp_connected_correlator}) equals
              $(\alpha\beta+G_\infty+\delta_{XY})/(1+\delta)
                  -(\alpha+\delta_X)(\beta+\delta_Y)/(1+\delta)^2$, where
              $\alpha=\Tr\E_X(\rho)$, $\beta=\Tr\E_Y(\rho)$, $G_\infty$ is the
              limit correlator (\ref{eq:ldp_limit_correlator}) at
              separation $t$, and $\delta,\delta_X,\delta_Y,\delta_{XY}$ are
              traces of linear maps containing $(\E_A-P)^n$, $(\E_A-P)^N$ or a
              power of $\E_A-P$ of exponent at least $n$.
        \item If $\lvert z-1\rvert$, $\lvert n_{XY}-(ab+g)\rvert$,
              $\lvert n_X-a\rvert$ and $\lvert n_Y-b\rvert$ are at most
              $\varepsilon\le\tfrac12$ and $\lvert a\rvert,\lvert b\rvert,
                  \lvert g\rvert\le M$, then
              $\lvert n_{XY}/z-(n_X/z)(n_Y/z)-g\rvert\le4(M+2)^2\varepsilon$.
        \item The trace of a linear map on $\MN{D}$ is bounded by a constant
              depending only on $D$ times its operator norm, and
              $e^{-(s-1)/\xi}\le\lvert\lambda_2\rvert^t$ whenever $t+1\le s$ and
              $0<\lvert\lambda_2\rvert<1$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \leanok
    A nonzero eigenvalue of $\E_A-P$ is an eigenvalue of $\E_A$ with a traceless
    eigenvector, hence different from $1$ by strong irreducibility
    (Theorem~\ref{thm:wielandt_proposition_three}), so the spectral radius of
    $\E_A-P$ is at most $\lvert\lambda_2\rvert$ and Gelfand's formula gives
    (1). Part (2) substitutes $\E_A^{\,k}=P+(\E_A-P)^k$ for $k\ge1$ into the
    trace formula of Lemma~\ref{lem:ldp_two_window_trace} and into the
    one-point functions and the normalization. Part (3) is elementary
    algebra with $\lvert1/z\rvert\le2$, and part (4) expands the trace in
    matrix units and uses $\xi=-1/\log\lvert\lambda_2\rvert$.
\end{proof}

\begin{theorem}[Exponentially decaying correlations]
    \label{thm:ldp_decaying_correlations}
    \lean{MPSTensor.exists_decayingCorrelations,
        MPSTensor.mpvConnectedCorrelator, MPSTensor.mpvExpectation,
        MPSTensor.chainWindowOperator}
    \leanok
    \uses{def:normal, def:correlation_length}
    Let $A$ be normal and in the gauge (\ref{eq:ldp_normal_gauge}), with
    correlation length $\xi>0$, let $L\ge1$ be a length such that the products
    of $L$ matrices of $A$ span $\MN{D}$ (one may take $L=1$ when $A$ is
    injective), and let $\ket{\phi_N}$ be the normalized vectors
    (\ref{eq:ldp_ti_mps}). For operators $X,Y$ on $(\C^d)^{\otimes L}$ let
    $X_1$ act on the sites $1,\dots,L$, let $Y_s$ act on the sites
    $s,\dots,s+L-1$, and put
    \begin{align}
        G_N(X,Y;s)
         & =\bra{\phi_N}X_1Y_s\ket{\phi_N}
        -\bra{\phi_N}X_1\ket{\phi_N}\bra{\phi_N}Y_s\ket{\phi_N}.
        \label{eq:ldp_connected_correlator}
    \end{align}
    There are Hermitian operators $\mathcal O,\mathcal O'$ on
    $(\C^d)^{\otimes L}$ of norm one, a constant $c>0$, an integer
    $K\in\{1,2\}$ and an integer $s_0\ge L+2$ such that for every $s\ge s_0$
    some $s'\in\{s,s+1,\dots,s+K-1\}$ satisfies, for every $N\ge3s'$,
    \begin{align}
        \bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
         & \ge c\,e^{-(s'-1)/\xi}.
        \label{eq:ldp_decaying_correlations}
    \end{align}
    One may take $K=1$ when $\lambda_2$ is real and $K=2$ otherwise. For
    injective $A$ and $L=1$ this is the correlation estimate of
    \cite[Lemma~2]{Malz2024Preparation}, with the separation $s$ ranging over
    windows of $K$ consecutive values; the correction of the source's
    statement is recorded in~\cite{gap:mswc24_decaying_correlations_windowed_connected}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:transfer_map, lem:ldp_observable_transfer_products,
        lem:ldp_observable_transfer_surjective, lem:ldp_two_window_trace,
        lem:ldp_hermitian_observables, lem:ldp_finite_size_correction,
        lem:ldp_vandermonde_window, thm:wielandt_proposition_three}
    With the inserted transfer maps $\E_X$ defined before
    Lemma~\ref{lem:ldp_observable_transfer_surjective}, and for operators
    $X,Y$ on consecutive blocks,
    \begin{align}
        \E_{\Id} & =\E_A^{\,L},
                 &
        \E_{X\otimes Y}
                 & =\E_X\,\E_Y.
        \notag
    \end{align}
    Expanding the traces in (\ref{eq:ldp_ti_mps})
    gives, for $L<s$ and $s+L-1\le N$,
    \begin{align}
        \bra{\phi_N(A)}X_1Y_s\ket{\phi_N(A)}
         & =\operatorname{tr}\bigl(\E_X\,\E_A^{\,s-L-1}\,\E_Y\,
        \E_A^{\,N-s-L+1}\bigr),
        \notag
    \end{align}
    the trace of a linear map on $\MN{D}$. Let $P(Z)=\Tr(Z)\,\rho$. In the
    gauge (\ref{eq:ldp_normal_gauge}), $\E_A^{\,n}=P+(\E_A-P)^n$ for $n\ge1$.
    An eigenvalue $\nu\neq0$ of $\E_A-P$ is an eigenvalue of $\E_A$ with a
    traceless eigenvector, so $\nu\neq1$ because $\E_A$ is strongly
    irreducible (Theorem~\ref{thm:wielandt_proposition_three}), and hence
    $\lvert\nu\rvert\le\lvert\lambda_2\rvert$. Fix $r$ with
    $\lvert\lambda_2\rvert<r<1$ and $r^2<\lvert\lambda_2\rvert$; Gelfand's
    formula gives $\lVert(\E_A-P)^n\rVert\le C_rr^n$. Replacing the last power
    $\E_A^{\,N-s-L+1}$ by $P$, and likewise in the normalization
    $\operatorname{tr}\E_A^{\,N}$ and in the one-point functions, changes
    $G_N(X,Y;s)$ by at most $C_{X,Y}\,r^{N-s-L+1}$, which is at most
    $C_{X,Y}\,r^{2(s-L-1)}$ when $N\ge3s$. The remaining term is
    \begin{align}
        G_\infty(X,Y;s)
         & =\Tr\Bigl(\E_X\bigl((\E_A-P)^{s-L-1}\bigl(\E_Y(\rho)\bigr)\bigr)\Bigr).
        \label{eq:ldp_limit_correlator}
    \end{align}

    Let $R\neq0$ satisfy $\E_A(R)=\lambda_2R$. Since $\xi>0$,
    $\lambda_2\neq0$, and trace preservation gives $\Tr R=0$, so
    $(\E_A-P)^tR=\lambda_2^{\,t}R$; since $\E_A(Z^\dagger)=\E_A(Z)^\dagger$,
    also $(\E_A-P)^tR^\dagger=\bar\lambda_2^{\,t}R^\dagger$. If $\lambda_2$ is
    real, one of $R+R^\dagger$ and $i(R-R^\dagger)$ is a nonzero Hermitian
    eigenvector $H$; put $v=H$ and $\ell(Z)=\Tr(HZ)/\Tr(H^2)$, so that
    $\ell((\E_A-P)^tv)=\lambda_2^{\,t}$. If $\lambda_2$ is not real, $R$ and
    $R^\dagger$ are eigenvectors for distinct eigenvalues; take a linear
    functional $f$ with $f(R)=1$ and $f(R^\dagger)=0$, put $v=R+R^\dagger$ and
    $\ell(Z)=f(Z)+\overline{f(Z^\dagger)}$, so that
    $\ell((\E_A-P)^tv)=\lambda_2^{\,t}+\bar\lambda_2^{\,t}$. In both cases
    $v$ is Hermitian, $\ell(Z^\dagger)=\overline{\ell(Z)}$, and
    $\ell((\E_A-P)^tv)=\lvert\lambda_2\rvert^t\sum_{j=1}^Ka_j\mu_j^t$ with
    $K\le2$, distinct $\mu_j$ of modulus one and $a\neq0$.

    The maps $Z\mapsto\Tr(Z)\,v$ and $Z\mapsto\ell(Z)\,\rho$ preserve
    Hermitian matrices. By Lemma~\ref{lem:ldp_observable_transfer_surjective}
    they are $\E_Y$ and $\E_X$ for some $X,Y$, and since
    $\E_{X^\dagger}(Z)=\E_X(Z^\dagger)^\dagger$ the Hermitian parts of $X$ and
    $Y$ have the same maps. Then
    $G_\infty(X,Y;s)=\lvert\lambda_2\rvert^t\sum_ja_j\mu_j^t$ with $t=s-L-1$,
    and $X,Y\neq0$ because this sum does not vanish identically. Take
    $\mathcal O,\mathcal O'$ to be $X,Y$ scaled to norm one; this multiplies
    $a$ by a positive constant.

    The Vandermonde matrix $W=(\mu_j^{\,u})_{0\le u<K,\,1\le j\le K}$ is
    invertible because the $\mu_j$ are distinct. For every $t$ the vector with
    entries $\sum_ja_j\mu_j^{\,t+u}$, $0\le u<K$, is $W$ applied to
    $(a_j\mu_j^t)_j$, whose maximum norm equals that of $a$ because
    $\lvert\mu_j\rvert=1$. Hence some $u\in\{0,\dots,K-1\}$ satisfies
    \begin{align}
        \Bigl\lvert\sum_{j=1}^Ka_j\mu_j^{\,t+u}\Bigr\rvert
         & \ge c_0,
         &
        c_0
         & =\frac{\max_j\lvert a_j\rvert}{\lVert W^{-1}\rVert_\infty}>0,
        \label{eq:ldp_vandermonde_window}
    \end{align}
    so $\lvert G_\infty(\mathcal O,\mathcal O';s+u)\rvert\ge
        c_0\lvert\lambda_2\rvert^{t'}$ with $t'=t+u$. With $s'=s+u$ and
    $N\ge3s'$ the finite-size correction is at most
    $C\,r^{2t'}=C\,(r^2/\lvert\lambda_2\rvert)^{t'}\lvert\lambda_2\rvert^{t'}$,
    which is at most $\tfrac{c_0}{2}\lvert\lambda_2\rvert^{t'}$ after
    enlarging $s_0$, since $r^2<\lvert\lambda_2\rvert$. Finally
    $\lvert\lambda_2\rvert^{t'}=e^{-(s'-L-1)/\xi}\ge e^{-(s'-1)/\xi}$, which
    gives (\ref{eq:ldp_decaying_correlations}) with $c=c_0/2$.
\end{proof}

\begin{theorem}[No preparation in depth $o(\log N)$]
    \label{thm:ldp_depth_lower_bound}
    \uses{def:ldp_local_circuit, def:correlation_length}
    Let $A$ be normal with correlation length $\xi>0$, let $\ket{\phi_N}$ be the
    normalized vectors (\ref{eq:ldp_ti_mps}), and let $\ket{\psi_N}$ be prepared
    in depth $T_N$ with $T_N=o(\log N)$. Then there is $N_0$ such that
    $\varepsilon(\phi_N,\psi_N)>1/2$ for all $N>N_0$. This is
    \cite[Theorem~1]{Malz2024Preparation}.
\end{theorem}
\begin{proof}
    \uses{thm:ldp_decaying_correlations, lem:ldp_expectation_overlap,
        thm:wielandt_proposition_three}
    We may assume that $A$ is in the gauge (\ref{eq:ldp_normal_gauge}). Since
    $A$ is normal, $\E_A$ is strongly irreducible
    (Theorem~\ref{thm:wielandt_proposition_three}), so its spectral radius
    $\lambda_1$ is a positive eigenvalue whose right eigenvector $\rho$ and
    whose eigenvector $\sigma$ for the dual map are positive definite. Replacing
    $A^i$ by $\lambda_1^{-1/2}\sigma^{1/2}A^i\sigma^{-1/2}$ gives
    (\ref{eq:ldp_normal_gauge}), with fixed point $\sigma^{1/2}\rho\sigma^{1/2}$
    scaled to unit trace. The scaling multiplies
    $\ket{\phi_N}$ by a positive constant and the conjugation leaves every trace
    in (\ref{eq:ldp_ti_mps}) unchanged, so $\phi_N/\lVert\phi_N\rVert$ is
    unchanged; the eigenvalues of $\E_A$ are divided by $\lambda_1$, which is
    the normalization $\lambda_1=1$ in which $\xi$ is computed.

    Take $L,\mathcal O,\mathcal O',c,K,s_0$ from
    Theorem~\ref{thm:ldp_decaying_correlations}, and suppose
    $\varepsilon(\phi_N,\psi_N)\le1/2$, that is
    $\lvert\braket{\phi_N}{\psi_N}\rvert\ge1/2$. Write $T=T_N$ and
    $\ket{\psi_N}=U\ket{\pi}$ with $U$ a local circuit of depth $T$ and
    $\ket{\pi}$ a product state. If $Z$ acts on an interval of sites, then
    $U^\dagger ZU$ acts on the interval enlarged by $T$ sites on each side, so
    expectations in $\ket{\psi_N}$ of products of operators whose enlarged
    intervals are pairwise disjoint factorize.

    Let $s_N=\max(s_0,2T+L+1)$ and let $s'\in\{s_N,\dots,s_N+K-1\}$ be given
    by the lemma; the window changes $s'$ by at most $K-1$, so still
    $s'=o(\log N)$ and $N\ge3s'$ for large $N$. Put $\Delta=s'+L+2T\le2s'$,
    $n=\lfloor N/\Delta\rfloor\ge N/(4s')$, and $b_k=1+(k-1)\Delta$ for
    $k=1,\dots,n$. With $e=\bra{\phi_N}\mathcal O_1\ket{\phi_N}$ and
    $e'=\bra{\phi_N}\mathcal O'_1\ket{\phi_N}$ let $\mathcal O^{(k)}$ be
    $\mathcal O-e$ on the sites $b_k,\dots,b_k+L-1$ and $\mathcal O'^{(k)}$ be
    $\mathcal O'-e'$ on the sites $b_k+s'-1,\dots,b_k+s'+L-2$. These have norm at
    most $2$, zero expectation in $\ket{\phi_N}$ by translation invariance,
    and $\bra{\phi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\phi_N}
        =G_N(\mathcal O,\mathcal O';s')$. Since $s'\ge2T+L+1$ and
    $\Delta\ge s'+L+2T$, the $2n$ enlarged intervals are pairwise disjoint,
    also around the ring.

    Two estimates are used. For Hermitian $Z$ and unit vectors $\phi,\psi$,
    with $\sigma_\chi(Z)=\lVert(Z-\langle Z\rangle_\chi)\chi\rVert$,
    \begin{align}
        \lvert\braket{\phi}{\psi}\rvert\,
        \bigl\lvert\langle Z\rangle_\phi-\langle Z\rangle_\psi\bigr\rvert
         & \le\sigma_\phi(Z)+\sigma_\psi(Z),
        \label{eq:ldp_expectation_overlap}
    \end{align}
    because
    $(\langle Z\rangle_\phi-\langle Z\rangle_\psi)\braket{\phi}{\psi}
        =\bra{\phi}(Z-\langle Z\rangle_\psi)\ket{\psi}
        -\bra{\phi}(Z-\langle Z\rangle_\phi)\ket{\psi}$. Secondly, the
    decomposition $\E_A^{\,n}=P+(\E_A-P)^n$ from the proof of
    Theorem~\ref{thm:ldp_decaying_correlations}, together with the fact that in
    the gauge (\ref{eq:ldp_normal_gauge}) the maps $\E_Z$ have norm bounded by
    a multiple of $\lVert Z\rVert$ independent of the size of the support,
    shows that the connected correlation in $\ket{\phi_N}$ of two operators of
    norm at most $4$ on intervals separated by $g$ and $g'$ sites around the
    ring is at most $C(r^g+r^{g'})$, with $C$ and $r<1$ depending only on $A$.
    Consecutive intervals $[b_k,b_k+s'+L-2]$ are separated by at least $2T$
    sites, and intervals $j$ apart by at least $(j-1)\Delta$, so every
    average of $n$ of these operators, or of their pair products, has
    variance at most $C_1/n$ in $\ket{\phi_N}$.

    For signs $\epsilon_k\in\{\pm1\}$ apply (\ref{eq:ldp_expectation_overlap})
    to $Z=\frac1n\sum_k\epsilon_k\mathcal O^{(k)}$. Its expectation in
    $\ket{\phi_N}$ is $0$; in $\ket{\psi_N}$ it is
    $\frac1n\sum_k\epsilon_kx_k$ with
    $x_k=\bra{\psi_N}\mathcal O^{(k)}\ket{\psi_N}\in\R$, and by factorization
    its variance there is at most $4/n$. Hence
    $\lvert\frac1n\sum_k\epsilon_kx_k\rvert\le C_2n^{-1/2}$, and the choice
    $\epsilon_k=\operatorname{sign}x_k$ gives
    $\frac1n\sum_k\lvert x_k\rvert\le C_2n^{-1/2}$. Next apply
    (\ref{eq:ldp_expectation_overlap}) to
    $Z=\frac1n\sum_k\mathcal O^{(k)}\mathcal O'^{(k)}$, which is Hermitian
    because the two factors act on disjoint sites. Its expectation in
    $\ket{\phi_N}$ is $G_N(\mathcal O,\mathcal O';s')$. In $\ket{\psi_N}$ the
    light cone gives
    $\bra{\psi_N}\mathcal O^{(k)}\mathcal O'^{(k)}\ket{\psi_N}=x_ky_k$ with
    $\lvert y_k\rvert\le2$, so the expectation is at most $2C_2n^{-1/2}$ in
    modulus and the variance is at most $16/n$. With
    (\ref{eq:ldp_decaying_correlations}) and $n\ge N/(4s')$,
    \begin{align}
        c\,e^{-(s'-1)/\xi}
         & \le\bigl\lvert G_N(\mathcal O,\mathcal O';s')\bigr\rvert
        \le C_3\Bigl(\frac{s'}{N}\Bigr)^{1/2}.
        \label{eq:ldp_depth_contradiction}
    \end{align}
    Since $s'\le s_N+K-1=o(\log N)$, the left side is $N^{-o(1)}$ and the
    right side is $N^{-1/2+o(1)}$, so (\ref{eq:ldp_depth_contradiction}) fails
    for all $N$ larger than some $N_0$ determined by $A$ and the sequence
    $(T_N)$. For those $N$, $\varepsilon(\phi_N,\psi_N)>1/2$.
\end{proof}

\section{Tensors that are not normal}

\begin{definition}[Fixed point of a basis of normal tensors]
    \label{def:ldp_non_normal_fixed_point}
    \uses{def:bnt, def:sector_weight_data, def:prep_fixed_point_tensor,
        thm:ldpolar_tensor_decomposition, def:correlation_length}
    Let $A^i=\bigoplus_{j=1}^b\operatorname{diag}(\mu_{j,1},\dots,\mu_{j,m_j})
        \otimes A_j^i$ with normal $A_j$, each in the gauge
    (\ref{eq:ldp_normal_gauge}), forming a basis of normal tensors, and put
    $\beta_j=\sum_{k=1}^{m_j}\mu_{j,k}^N$. Consider only lengths $N=qM$ with
    $(\beta_1,\dots,\beta_b)\neq0$; for the other lengths
    $\ket{\phi_N(A)}=\sum_j\beta_j\ket{\phi_N(A_j)}=0$ and there is nothing to
    prepare. Write $\xi_j$ for the correlation length of the normal tensor
    $A_j$, formed from the subleading eigenvalue of its own transfer map
    $\E_{A_j}$. With $\ket{\omega_j}$ the pair state of $A_j$ and
    $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$, the
    \emph{fixed-point state} is
    \begin{align}
        \ket{\Omega'}
         & =\sum_{j=1}^b\alpha^{(N)}_j\bigotimes_{k=1}^{M}\ket{\omega_j}_{R_kL_{k+1}}.
        \label{eq:ldp_non_normal_fixed_point}
    \end{align}
    With $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition} for the blocked map of
    $A$, the \emph{approximating state} is
    $\ket{\widetilde\phi_N}=V^{\otimes M}\ket{\Omega'}/
        \lVert V^{\otimes M}\ket{\Omega'}\rVert$, whenever the denominator is
    nonzero.
    This is \cite[eqs.~(19) and~(S7)]{Malz2024Preparation}, with the
    canonical form of \cite{Cirac2017MPDO_AnnPhys}.

    When some block has multiplicity $m_j\geq2$, the source's construction
    fails: no product of pairs carrying the weights $\beta_j$ gives, after
    $V^{\otimes M}$, an approximation of $\ket{\phi_N}$. For
    $A^0=\operatorname{diag}(1,1,0)$ and $A^1=\operatorname{diag}(0,0,1)$ the
    overlap with $\ket{\phi_N}\propto2\ket{0\cdots0}+\ket{1\cdots1}$ tends to
    $1/\sqrt5$ as $M\to\infty$, for every $q$
    \cite{gap:mswc24_multiplicity_fixed_point}.
\end{definition}

\begin{definition}[Fixed-point state of a sector decomposition]
    \label{def:ldp_sector_fixed_point_state}
    \lean{MPSTensor.SectorDecomposition.copyCoord,
        MPSTensor.SectorDecomposition.embeddedFixedPointPair,
        MPSTensor.SectorDecomposition.canonicalFixedPointState,
        MPSTensor.embedPair}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, def:sector_weight_data,
        def:prep_pair_product_state}
    Let $A$ be given by a sector decomposition with blocks $A_1,\dots,A_b$ of
    bond dimensions $D_j$, multiplicities $m_j$ and coefficients $\mu_{j,k}$,
    whose flattened direct sum $\bigoplus_{(j,k)}\mu_{j,k}A_j$ is the tensor of
    Definition~\ref{def:ldp_non_normal_fixed_point} up to the order of the
    bond coordinates, so the bond space of $A$ is $\C^D$ with
    $D=\sum_jm_jD_j$. The blocks carry no normality, basis-of-normal-tensors
    or gauge condition. For each $j$ let $\sigma_j$ be any matrix on
    $\C^{D_j}$ and choose a copy $k_j\in\{1,\dots,m_j\}$. The pair of block
    $j$ is $\sum_{r,l}(\sqrt{\sigma_j})_{rl}\ket{r}\ket{l}$ placed, on both
    legs, on the bond coordinates of copy $k_j$, which gives
    $\ket{\omega_j}\in\C^D\otimes\C^D$. With $\beta_j$ and $\alpha^{(N)}_j$ as
    in Definition~\ref{def:ldp_non_normal_fixed_point}, the fixed-point state
    $\ket{\Omega'}$ is formed from these pairs by
    (\ref{eq:ldp_non_normal_fixed_point}).

    Apart from the missing conditions on the blocks, this construction differs
    from Definition~\ref{def:ldp_non_normal_fixed_point} in three ways. The source
    takes $\sigma_j$ to be the unit-trace right fixed point of $\E_{A_j}$ in
    the gauge (\ref{eq:ldp_normal_gauge}), whereas here $\sigma_j$ is a free
    parameter. The source calls the pairs vectors of
    $\C^{D^2}=\C^D\otimes\C^D$ without saying where the pair of block $j$ sits
    when $m_j>1$, whereas here the copy $k_j$ is chosen explicitly. The
    approximating state is not defined here; it is defined only for a family
    of arbitrary pairs (Definition~\ref{def:ldp_pair_family_fixed_point}).
\end{definition}

\begin{definition}[Fixed-point state of a family of pairs]
    \label{def:ldp_pair_family_fixed_point}
    \lean{MPSTensor.bntWeight, MPSTensor.ghzAmplitude,
        MPSTensor.nonNormalFixedPointState, MPSTensor.nonNormalApproxState,
        MPSTensor.pairIsometry, MPSTensor.ghzState, Matrix.tensorPower}
    \leanok
    \uses{def:ldp_non_normal_fixed_point, thm:ldpolar_tensor_decomposition}
    Let $\omega_1,\dots,\omega_b\in\mathbb C^D\otimes\mathbb C^D$ be any vectors,
    $\alpha\in\mathbb C^b$ any coefficients, and $M\ge0$. The state
    $\ket{\Omega'}=\sum_j\alpha_j\bigotimes_{k=1}^M\ket{\omega_j}_{R_kL_{k+1}}$ is
    formed as in (\ref{eq:ldp_non_normal_fixed_point}), and for a tensor $A$ and a
    blocking length $q$, with $V$ from Theorem~\ref{thm:ldpolar_tensor_decomposition}
    for the $q$-site blocked map of $A$, the approximating state is
    $V^{\otimes M}\ket{\Omega'}/\lVert V^{\otimes M}\ket{\Omega'}\rVert$, and zero when
    the denominator vanishes. For multiplicity coefficients $\mu_{j,k}$ put
    $\beta_j=\sum_k\mu_{j,k}^N$ and $\alpha^{(N)}_j=\beta_j/(\sum_l\lvert\beta_l\rvert^2)^{1/2}$.
    Let $W\colon\mathbb C^b\to\mathbb C^D\otimes\mathbb C^D$,
    $\ket{j}\mapsto\ket{\omega_j}$, be the map whose $j$-th column is $\omega_j$,
    let $X^{\otimes M}$ denote the $M$-fold tensor power of a matrix $X$, with entries
    $(X^{\otimes M})_{p,s}=\prod_{k=1}^M X_{p_k,s_k}$, and put
    $\ket{\chi_M}=\sum_j\alpha_j\ket{j}^{\otimes M}\in(\mathbb C^b)^{\otimes M}$.
    Definition~\ref{def:ldp_non_normal_fixed_point} is the instance in which
    $\omega_j$ is the pair state of the normal block $A_j$ embedded in the bond space
    of $A$, $\alpha=\alpha^{(N)}$, and $N=qM$; none of these three identifications is
    imposed here.
\end{definition}

\begin{theorem}[Local orthogonality of the pairs]
    \label{thm:ldp_local_orthogonality}
    \lean{MPSTensor.SectorDecomposition.inner_embeddedFixedPointPair,
        MPSTensor.inner_embedPair_self, MPSTensor.inner_embedPair_eq_zero_of_disjoint,
        MPSTensor.SectorDecomposition.copyCoord_injective,
        MPSTensor.SectorDecomposition.copyCoord_ne_of_ne,
        MPSTensor.SectorDecomposition.toTensor_copyCoord,
        MPSTensor.pairProductState_embedPair,
        MPSTensor.mpv_toTensorFromBlocks_fixedPointTensor}
    \leanok
    \uses{def:ldp_sector_fixed_point_state}
    In Definition~\ref{def:ldp_sector_fixed_point_state}, let each $\sigma_j$ be
    positive semidefinite with $\operatorname{Tr}\sigma_j=1$. For every choice
    of the copies $k_j$ the embedded pairs satisfy
    $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$. The chosen coordinates of
    block $j$ carry the block $\mu_{j,k_j}A_j$ of $A$, and embedding the pair
    embeds its product state. Moreover, for $M\geq1$, in the bond space
    $\C^{D_j}\otimes\C^{D_j}$ of block $j$ alone, the tensor
    $\bigoplus_k\mu_{j,k}^qP_{j,\infty}$ generates on $M$ sites the state
    $\bigl(\sum_k\mu_{j,k}^{qM}\bigr)\ket{\Omega_j}=\beta_j\ket{\Omega_j}$ with
    $N=qM$ and $\ket{\Omega_j}$ formed from the unembedded pair. The
    orthogonality is asserted after \cite[eq.~(S7)]{Malz2024Preparation},
    where the reason given is that the blocks form a basis of normal tensors;
    here it follows from the placement of the pairs alone, and the
    identification of these pairs with the fixed points of the positive parts
    of the blocked tensor is not formalized
    \cite{gap:mswc24_ghz_form_local_orthogonality}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:ldp_pair_disjoint_orthogonal, thm:prep_pair_norm,
        thm:prep_mpv_fixed_point_tensor, thm:mpv_decomposition}
    Copies of distinct blocks occupy distinct bond coordinates, so the right
    legs of $\omega_j$ and $\omega_{j'}$ with $j\neq j'$ are supported on
    disjoint sets and the pairs are orthogonal by
    Theorem~\ref{thm:ldp_pair_disjoint_orthogonal}. The coordinates of one
    copy are distinct, so embedding preserves overlaps and
    $\braket{\omega_j}{\omega_j}=\operatorname{Tr}\sigma_j=1$. The direct sum
    generates the sum of the states of its blocks with the $M$-th powers of the
    weights, and $P_{j,\infty}$ generates $\ket{\Omega_j}$.
\end{proof}

\begin{theorem}[GHZ form of the fixed-point state]
    \label{thm:ldp_ghz_form}
    \lean{MPSTensor.isIsometry_pairIsometry, Matrix.IsIsometry.tensorPower,
        MPSTensor.nonNormalFixedPointState_eq_tensorPower_mulVec_ghzState,
        MPSTensor.nonNormalFixedPointState_norm_sq,
        MPSTensor.SectorDecomposition.isIsometry_pairIsometry_embeddedFixedPointPair,
        MPSTensor.SectorDecomposition.canonicalFixedPointState_eq_tensorPower_mulVec_ghzState,
        MPSTensor.SectorDecomposition.canonicalFixedPointState_norm_sq}
    \leanok
    \uses{def:ldp_sector_fixed_point_state, def:ldp_pair_family_fixed_point,
        thm:ldp_local_orthogonality}
    With $W$, $X^{\otimes M}$ and $\chi_M$ as in
    Definition~\ref{def:ldp_pair_family_fixed_point}, formed with
    $\alpha=\alpha^{(N)}$ and the pairs of
    Definition~\ref{def:ldp_sector_fixed_point_state}, where each $\sigma_j$ is
    positive semidefinite of unit trace, $W$ and $W^{\otimes M}$ are isometries
    and $\ket{\Omega'}=W^{\otimes M}\ket{\chi_M}$, up to the regrouping of the
    legs $R_kL_{k+1}$. If moreover $M\geq1$ and $(\beta_1,\dots,\beta_b)\neq0$,
    then $\braket{\Omega'}{\Omega'}=1$. This is the preparation of
    \cite[eq.~(19)]{Malz2024Preparation}. The same holds for the state of
    Definition~\ref{def:ldp_pair_family_fixed_point} with $\alpha=\alpha^{(N)}$
    for any orthonormal pairs $\omega_j$; the identity
    $\ket{\Omega'}=W^{\otimes M}\ket{\chi_M}$ holds for any pairs.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:prep_pair_product_norm, thm:ldp_local_orthogonality}
    The pairs of Definition~\ref{def:ldp_sector_fixed_point_state} are
    orthonormal by Theorem~\ref{thm:ldp_local_orthogonality}. The $(j,j')$ entry of $W^\dagger W$ is $\braket{\omega_j}{\omega_{j'}}$, and the
    entries of $(W^{\otimes M})^\dagger W^{\otimes M}$ factor over the sites,
    \begin{align*}
        \bigl((W^{\otimes M})^\dagger W^{\otimes M}\bigr)_{s,t}
        =\prod_{k=1}^M(W^\dagger W)_{s_k,t_k}=\delta_{st}.
    \end{align*}
    Expanding
    $\ket{j}^{\otimes M}$ in the product basis gives
    $W^{\otimes M}\ket{j}^{\otimes M}=\bigotimes_k\ket{\omega_j}$, and summing with the
    weights $\alpha^{(N)}_j$ gives $\ket{\Omega'}$. Overlaps of the product states factor
    over the bonds,
    $\braket{\Omega_j}{\Omega_{j'}}=\braket{\omega_j}{\omega_{j'}}^M=\delta_{jj'}$, so
    $\braket{\Omega'}{\Omega'}=\sum_j\lvert\alpha^{(N)}_j\rvert^2=1$.
\end{proof}

\begin{theorem}[Pairs with disjoint supports are orthogonal]
    \label{thm:ldp_pair_disjoint_orthogonal}
    \lean{MPSTensor.inner_pair_eq_zero_of_disjoint}
    \leanok
    \uses{def:ldp_sector_fixed_point_state}
    Let $S,S'\subseteq\{0,\dots,D-1\}$ be disjoint, and let
    $\omega,\omega'\in\C^D\otimes\C^D$ satisfy $\omega_{(a,c)}=0$ for $a\notin S$
    and $\omega'_{(a,c)}=0$ for $a\notin S'$. Then
    $\braket{\omega}{\omega'}=0$. The embedded pairs of distinct blocks in
    Definition~\ref{def:ldp_sector_fixed_point_state} are supported in this way, so
    this gives the off-diagonal half of the orthogonality
    $\braket{\omega_j}{\omega_{j'}}=\delta_{jj'}$, which
    Theorem~\ref{thm:ldp_local_orthogonality} completes.
\end{theorem}

\begin{proof}
    \leanok
    Every term of the sum vanishes: if $a\in S$ then $a\notin S'$ and
    $\omega'_{(a,c)}=0$, and otherwise $\omega_{(a,c)}=0$, so
    \begin{align*}
        \braket{\omega}{\omega'}
        =\sum_{(a,c)}\overline{\omega_{(a,c)}}\,\omega'_{(a,c)}=0.
    \end{align*}
\end{proof}

\begin{theorem}[The approximating state for a repeated block]
    \label{thm:ldp_approximation_error_repeated_block}
    \lean{MPSTensor.nonNormalApproxOverlap_repeatedBlockTensor,
        MPSTensor.tendsto_norm_nonNormalApproxOverlap_repeatedBlockTensor,
        MPSTensor.norm_le_of_hasEigenvalue_transferMap_repeatedBlockBasis,
        MPSTensor.not_approximationError_le_repeatedBlockTensor}
    \uses{def:ldp_non_normal_fixed_point, def:ldp_sector_fixed_point_state,
        def:correlation_length}
    Let $A^0=\operatorname{diag}(1,1,0)$ and $A^1=\operatorname{diag}(0,0,1)$,
    the canonical form with the normal block $A_1=(1,0)$ of multiplicity two and
    weights $(1,1)$ and the normal block $A_2=(0,1)$ of multiplicity one, so that
    $\beta=(2,1)$ and $\ket{\phi_N}\propto2\ket{0\cdots0}+\ket{1\cdots1}$. Place
    the pair of the first block on its first copy. For every $q\geq1$ and
    $M\geq1$ the approximating state of
    Definition~\ref{def:ldp_non_normal_fixed_point} satisfies
    \begin{align}
        \braket{\widetilde\phi_N}{\phi_N}
         & =\frac{4\cdot2^{-M/2}+1}{\sqrt5\,(4\cdot2^{-M}+1)^{1/2}},
        \label{eq:ldp_repeated_block_overlap}
    \end{align}
    which does not depend on $q$ and tends to $1/\sqrt5$ as $M\to\infty$. The
    transfer maps of both blocks have no eigenvalue other than one, so every
    $\lambda_2$ with $0<\lvert\lambda_2\rvert<1$ bounds their subleading
    eigenvalues. For every such $\lambda_2$, every $\gamma>0$ and every constant
    $C$ there is $M\geq1$ such that, with $q=M$, $N=M^2$ and
    $y=(N/q)e^{-\gamma q/\xi_{\mathrm{diag}}}$,
    $\varepsilon(\widetilde\phi_N,\phi_N)>Cy\,e^{Cy}$. The sequence $q=M$
    satisfies $q=o(N)$, so the bound (\ref{eq:ldp_approximation_error_general})
    fails for this tensor \cite{gap:mswc24_multiplicity_fixed_point}.
\end{theorem}

\begin{proof}
    \uses{thm:ldpolar_decomposition, thm:ldp_polar_uniqueness}
    For a tensor of diagonal matrices the blocked map factors as $B=CJ^\dagger$,
    where $J\colon\ket e\mapsto\ket{e\,e}$ and the column $e$ of $C$ is the state
    of the diagonal coordinate $e$ on $q$ sites; here these are $\ket{0\cdots0}$
    for $e=0,1$ and $\ket{1\cdots1}$ for $e=2$. Hence $B^\dagger B=JGJ^\dagger$ with
    $G$ the Gram matrix of the columns. The positive semidefinite square root of $G$
    is $Q=2^{-1/2}\bigl(\begin{smallmatrix}1&1\\1&1\end{smallmatrix}\bigr)\oplus1$,
    and $E=\tfrac12\bigl(\begin{smallmatrix}1&1\\1&1\end{smallmatrix}\bigr)\oplus1$
    is the projector onto its range, so uniqueness of positive square roots gives
    $P=JQJ^\dagger$ and $\Pi=JEJ^\dagger$. The pairs are $e_{00}$ and $e_{22}$, so
    $V^{\otimes M}\sum_j\alpha_j\ket{\Omega_j}$ has components
    $\sum_j\alpha_j\prod_kV_{\tau_k,(c_j,c_j)}$. From $V^\dagger B=P$ and
    $V^\dagger V=\Pi$, the overlap with the unnormalized target is
    $\sum_j\sum_e\overline{\alpha_j}Q_{c_je}^M$, the squared norm of the
    approximating vector is $\sum_{j,j'}\overline{\alpha_j}\alpha_{j'}E_{c_jc_{j'}}^M$,
    and $\lVert\phi_N(A)\rVert^2=\sum_{e,e'}G_{ee'}^M=5$. With
    $\alpha=(2,1)/\sqrt5$ this gives (\ref{eq:ldp_repeated_block_overlap}). Since
    $1/\sqrt5<1/2$, the error exceeds $1/2$ for large $M$, while $Cy\,e^{Cy}$ with
    $y=Mr^M$ and $r=e^{-\gamma/\xi_{\mathrm{diag}}}<1$ tends to zero.
\end{proof}

\begin{theorem}[The approximating state for overlapping blocks]
    \label{thm:ldp_approximation_error_overlapping_blocks}
    \lean{MPSTensor.nonNormalApproxOverlap_overlappingBlockTensor,
        MPSTensor.le_one_sub_norm_nonNormalApproxOverlap_overlappingBlockTensor,
        MPSTensor.norm_le_of_hasEigenvalue_transferMap_overlappingBlockBasis,
        MPSTensor.not_approximationError_le_overlappingBlockTensor}
    \uses{def:ldp_non_normal_fixed_point, def:ldp_sector_fixed_point_state,
        def:correlation_length}
    Let $A^0=\operatorname{diag}(1,3/5)$ and $A^1=\operatorname{diag}(0,4/5)$,
    the canonical form with the normal blocks $A_1=(1,0)$ and $A_2=(3/5,4/5)$, each
    of multiplicity one and weight one, so that $\beta=(1,1)$. Put $s=(3/5)^q$,
    the overlap of the states of the two blocks on $q$ sites,
    $u=\bigl(\sqrt{1+s}+\sqrt{1-s}\bigr)/2$ and
    $v=\bigl(\sqrt{1+s}-\sqrt{1-s}\bigr)/2$. For every $q\geq1$ and $M\geq1$ the
    approximating state of Definition~\ref{def:ldp_non_normal_fixed_point}
    satisfies
    \begin{align}
        \braket{\widetilde\phi_N}{\phi_N}
         & =\frac{u^M+v^M}{(1+s^M)^{1/2}},
        \label{eq:ldp_overlapping_blocks_overlap}
    \end{align}
    and for $M\geq3$ the error is at least $(9/25)^q/16$. The transfer maps of both
    blocks have no eigenvalue other than one, so every $\lambda_2$ bounds their
    subleading eigenvalues. If $r=e^{-\gamma/\xi_{\mathrm{diag}}}<9/25$, then for every
    constant $C$ there is $q\geq3$ such that, with $M=q$, $N=q^2$ and
    $y=(N/q)e^{-\gamma q/\xi_{\mathrm{diag}}}$,
    $\varepsilon(\widetilde\phi_N,\phi_N)>Cy\,e^{Cy}$. The sequence $q=M$ satisfies
    $q=o(N)$, so the bound (\ref{eq:ldp_approximation_error_general}) fails even
    when every multiplicity is one \cite{gap:mswc24_block_form_mixed_overlap}.
\end{theorem}

\begin{proof}
    \uses{thm:ldpolar_decomposition, thm:ldp_polar_uniqueness}
    As in the proof of Theorem~\ref{thm:ldp_approximation_error_repeated_block},
    the Gram matrix of the blocked map on the diagonal coordinates is
    $G=\bigl(\begin{smallmatrix}1&s\\s&1\end{smallmatrix}\bigr)$, whose positive
    square root is $Q=\bigl(\begin{smallmatrix}u&v\\v&u\end{smallmatrix}\bigr)$,
    since $u^2+v^2=1$ and $2uv=s$. The determinant of $Q$ is $\sqrt{1-s^2}>0$, so
    the support projector is $JJ^\dagger$. With $\alpha=(1,1)/\sqrt2$ the overlap
    with the unnormalized target is $\sqrt2\,(u^M+v^M)$, the approximating vector
    has norm one, and $\lVert\phi_N(A)\rVert^2=2+2s^M$, which gives
    (\ref{eq:ldp_overlapping_blocks_overlap}). For $M\geq3$,
    \begin{align*}
        u^M+v^M\leq u^3+v^2=1-u^2(1-u)\leq1-\frac{s^2}{16},
    \end{align*}
    because $u^2=\bigl(1+\sqrt{1-s^2}\bigr)/2\geq1/2$ and
    $1-u\geq(1-u^2)/2\geq s^2/8$. Finally $Cy\,e^{Cy}$ with $y=qr^q$ is eventually
    below $(9/25)^q/16$, because $q(25r/9)^q\to0$.
\end{proof}

\begin{lemma}[Polar decomposition of an orthogonal sum]
    \label{lem:ldp_orthogonal_sum_polar}
    \lean{Matrix.polarIso_sum_of_orthogonal, Matrix.exists_polarIso_eq_mul,
        Matrix.conjTranspose_polarIso_mul_polarIso_eq_zero,
        Matrix.conjTranspose_polarIso_mul_eq_zero}
    \leanok
    \uses{thm:ldp_polar_uniqueness}
    Let $B=\sum_jc_jB_jK_j^\dagger$ with $c_j>0$, isometries $K_j$ with
    orthogonal ranges, so that $K_j^\dagger K_{j'}=\delta_{jj'}$, and
    $B_j^\dagger B_{j'}=0$ for $j\neq j'$. With $B_j=V_jP_j$ the polar
    decompositions of the summands, the polar decomposition of $B$ has
    \begin{align*}
        V=\sum_jV_jK_j^\dagger,\qquad P=\sum_jc_jK_jP_jK_j^\dagger.
    \end{align*}
    This is the block form of the positive part asserted in
    \cite[eq.~(S5)]{Malz2024Preparation} (``Since $V$ is an isometry, $P$
    inherits the block structure of $A$''), for blocks of multiplicity one
    under the orthogonality of their states, which the source does not state
    and without which the block form fails
    \cite{gap:mswc24_block_form_mixed_overlap}.
\end{lemma}
\begin{proof}
    \leanok
    Every partial isometry is $V_j=B_jR_j$ for some $R_j$: the support
    projector $\Pi_j$ satisfies $\Pi_j=P_jR_j$ for some $R_j$, because both
    have the same range, and $V_j=V_j\Pi_j=B_jR_j$. Hence
    $V_j^\dagger V_{j'}=R_j^\dagger B_j^\dagger B_{j'}R_{j'}=0$ for $j\neq j'$. Put
    $W=\sum_jV_jK_j^\dagger$, $Q=\sum_jc_jK_jP_jK_j^\dagger\geq0$ and
    $E=\sum_jK_j\Pi_jK_j^\dagger$. The orthogonality of the $K_j$ and of the
    $V_j$ removes all cross terms, so $WQ=B$, $W^\dagger W=E$, $E^2=E=E^\dagger$,
    $EQ=Q$ and $Q\sum_jc_j^{-1}K_jR_jK_j^\dagger=E$. Thus $E$ is the projector
    onto the range of $Q$, and Theorem~\ref{thm:ldp_polar_uniqueness} gives
    $V=W$ and $P=Q$.
\end{proof}

    bound the moduli of the eigenvalues other than one of every transfer map
    $\E_{A_j}$, so that its correlation length $\xi$ bounds every $\xi_j$, and
    let $0<\gamma<1/2$. Form the approximating state $\widetilde\phi_N$ of
    Definition~\ref{def:ldp_non_normal_fixed_point} with the pairs of the
    $\rho_j$ on the bond coordinates of their blocks and $\beta_j=\mu_j^N$.
    There is $C>0$ such that for every block length $q$ at which the $q$-site
    states of distinct blocks are orthogonal,
    $\sum_{i_1,\dots,i_q}\overline{(A_j^{i_1}\cdots A_j^{i_q})_{ab}}\,
        (A_{j'}^{i_1}\cdots A_{j'}^{i_q})_{a'b'}=0$ for $j\neq j'$ and all
    $a,b,a',b'$, and every $M\geq1$, with $N=qM$ and
    $y=(N/q)\,e^{-\gamma q/\xi}$,
    \begin{align}
        \varepsilon(\widetilde\phi_N,\phi_N)\leq Cy
        \qquad\text{and}\qquad
        \varepsilon(\widetilde\phi_N,\phi_N)\leq Cy\,e^{Cy}.
        \label{eq:ldp_approximation_error_orthogonal_blocks}
    \end{align}
    This is \cite[Lemma~1$'$(ii)]{Malz2024Preparation} restricted to
    multiplicity one and to blocks whose $q$-site states are orthogonal. For
    multiplicity one the phase of $\mu_j$ can be absorbed into $A_j$, which
    stays normal with the same transfer map, so the condition $\mu_j>0$ only
    fixes this presentation. The orthogonality is an added hypothesis: without
    it the statement fails
    (Theorem~\ref{thm:ldp_approximation_error_overlapping_blocks}). Under it no
    condition $q=o(N)$ is needed
    \cite{gap:mswc24_block_form_mixed_overlap}.
\end{theorem}
\begin{proof}
    \leanok
    Let $B_j$ be the $q$-site blocked map of $A_j$, with polar decomposition
    $B_j=V_jP_j$, and $K_j$ the isometry placing the bond pairs of block $j$.
    For $q\geq1$ every word of $A$ of length $q$ is the direct sum of the
    words of the blocks, so the blocked map of $A$ is
    $B=\sum_j\mu_j^qB_jK_j^\dagger$, and
    Lemma~\ref{lem:ldp_orthogonal_sum_polar} gives $V=\sum_jV_jK_j^\dagger$.
    The pair of block $j$ is $K_j^{\otimes M}$ applied to the product of the
    pairs of $\rho_j$, and $VK_j=V_j$, so
    $V^{\otimes M}\ket{\Omega'}=\sum_j\alpha_j\ket{s_j}$ with
    $\ket{s_j}=\ket{\phi_M(V_jP_{j,\infty})}$, while
    $\ket{\phi_N(A)}=\sum_j\mu_j^N\ket{t_j}$ with
    $\ket{t_j}=\ket{\phi_N(A_j)}$. The tensors generating $s_j$, $t_j$, $s_{j'}$
    and $t_{j'}$ have mixed transfer maps equal to zero for $j\neq j'$, because
    $B_j^\dagger B_{j'}=0$ and hence $V_j^\dagger B_{j'}=V_j^\dagger V_{j'}=0$;
    so all cross overlaps vanish.

    Let $q$ be at least the injectivity lengths of all blocks; the smaller $q$
    are covered by $\varepsilon\leq1$ once $C$ is large. Then
    $\braket{s_j}{s_j}=1$, $\braket{s_j}{t_j}=z_j$ with
    $z_j=\bra{\phi_M(P_{j,\infty})}\phi_M(P_{j,q})\rangle$, and
    $\braket{t_j}{t_j}=c_j=\Tr\E_{A_j}^{\,N}$. With $p_j=\mu_j^{2N}$ this gives
    \begin{align*}
        \lvert\braket{\widetilde\phi_N}{\phi_N}\rvert
        =\frac{\bigl\lvert\sum_jp_jz_j\bigr\rvert}
        {\bigl(\sum_jp_j\bigr)^{1/2}\bigl(\sum_jp_jc_j\bigr)^{1/2}}.
    \end{align*}
    For weights $w_j=p_j/\sum_lp_l$ put $a=\sum_jw_jz_j$ and
    $s^2=\sum_jw_jc_j$. Then $\lvert a-1\rvert\leq\sum_j\lvert z_j-1\rvert$ and
    $\lvert s^2-1\rvert\leq\sum_j\lvert c_j-1\rvert=:\eta$. If $\eta>1/2$ then
    $\varepsilon\leq1<2\eta$. Otherwise $s\geq1/2$ and
    $\varepsilon=(s-\lvert a\rvert)/s\leq2\bigl(\lvert s-1\rvert+\lvert a-1\rvert\bigr)$,
    so in both cases
    $\varepsilon\leq2\sum_j\bigl(\lvert z_j-1\rvert+\lvert c_j-1\rvert\bigr)$.
    The proof of Lemma~\ref{lem:ldp_positive_part_overlap} bounds
    $\lvert z_j-1\rvert$ itself, not only $\bigl\lvert1-\lvert z_j\rvert\bigr\rvert$,
    by $C_jy\,e^{C_jy}$, and the proof of
    Theorem~\ref{thm:ldp_approximation_error_normal} gives
    $\lvert c_j-1\rvert\leq K_je^{-2\gamma N/\xi}\leq K_jy$. Summing over the
    blocks gives the second bound, and the first follows as in
    Theorem~\ref{thm:ldp_approximation_error_normal}, because $\varepsilon\leq1$.
\end{proof}
