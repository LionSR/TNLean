\section{Composition and exact expansions of original circuits}
\label{sec:peps_original_chronology_bounds}

Distributed compression of a dyadic protocol composes its stage changes in
chronological order while tracking each nonprivate occurrence and its local
monomial expansion \cite[Proposition~7.1]{OpenAI2026PolynomialPEPS}.

\begin{definition}[Chronological composite]
    \label{def:peps_original_chronology_composite}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.sequence}
    For original distributed circuits
    $W_t:\mathcal H_{A_t}\to\mathcal H_{A_{t+1}}$, define
    $W^{[0]}=1$ and $W^{[t+1]}=W_tW^{[t]}$.
\end{definition}

\begin{theorem}[Chronological composition]
    \label{thm:peps_original_chronology_composition}
    \leanok
    \uses{def:peps_original_chronology_composite}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.eval_sequence_zero}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.eval_sequence_succ}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isExpansionBounded_sequence}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.nonprivateCount_sequence}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.forall_participants_sequence}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isMonomialBounded_sequence}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.participationCount_comp}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.participationCount_sequence}
    For original distributed circuits $W_t:\mathcal H_{A_t}\to\mathcal H_{A_{t+1}}$
    ($0\le t<T$) and $v\in\mathcal H_{A_0}$, $W^{[0]}v=v$ and
    $W^{[t+1]}v=W_t(W^{[t]}v)$. Writing $N(W)$ for the number of nonprivate
    occurrences in $W$ and $n_p(W)$ for those involving party $p$ (retaining
    separate occurrences with equal local operators),
    \begin{align}
        N(W^{[T]})   & =\sum_{t<T}N(W_t),   &
        n_p(W^{[T]}) & =\sum_{t<T}n_p(W_t),
        \label{eq:peps_original_chronology_counts}
    \end{align}
    and any participating-set property or monomial bound $(K,r,S)$ (at most $K$
    terms, $r$ pair effects per term, and coefficient sum $S$) holding for all
    $W_t$ also holds for $W^{[T]}$.
\end{theorem}
\begin{proof}\leanok
    By induction on $T$, $W^{[t+1]}=W_tW^{[t]}$ (Definition~\ref{def:peps_original_chronology_composite})
    has occurrences equal to the disjoint union of those of $W^{[t]}$ and $W_t$
    with unchanged participating sets and monomial lists, giving~(\ref{eq:peps_original_chronology_counts})
    and preserving all occurrence-wise bounds.
\end{proof}

\begin{definition}[Recorded local operators]
    \label{def:peps_original_recorded_gate_maps}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.canonicalGateMaps}
    For an original distributed circuit $W$, let $\widehat W$ be the evaluation
    of the chronology of $W$ when each nonprivate occurrence with monomial list
    $L$ is assigned its recorded sum $\sum_{m\in L}m$.
\end{definition}

\begin{theorem}[Exact local operator sums]
    \label{thm:peps_original_exact_gate_maps}
    \leanok
    \uses{def:peps_original_recorded_gate_maps}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.evalWithGateMaps_canonicalGateMaps}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isGateApproximation_zero_canonicalGateMaps}
    For every original distributed circuit $W$, $\widehat W=W$ (including
    private operations), with zero local approximation error and unchanged
    contraction and pair-effect conditions.
\end{theorem}
\begin{proof}\leanok
    At a nonprivate occurrence with monomial list $L$, the prescribed gate is
    $\sum_{m\in L}m$ (error $\bigl\|\sum_{m\in L}m-\sum_{m\in L}m\bigr\|=0$),
    while $\widehat{VU}=\widehat V\,\widehat U$,
    $\widehat{1\otimes U}=1\otimes\widehat U$, and private/exchange steps are
    unchanged; induction on $W$ gives $\widehat W=W$.
\end{proof}

\begin{theorem}[Original monomial and expansion bounds]
    \label{thm:peps_original_produced_resource_bounds}
    \leanok
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.monomialCount_produce_le_of_isMonomialBounded}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.isExpansionBounded_mono}
    If every original monomial list has $|L|\le K$, then every produced
    occurrence has $|L|\le K$; if $W$ has pair-effect bound $r$ and
    coefficient-sum bound $S$, then any $r'\ge r$ and $S'\ge S$ also bound $W$.
\end{theorem}
\begin{proof}\leanok
    Each produced occurrence retains its original list $L$ (so $|L|\le K$), and
    transitivity of $\le$ extends $(r,S)$ to $(r',S')$ across all monomials.
\end{proof}
