\section{The constants produced by the proof of the polynomial PEPS approximation}
\label{sec:polynomial_peps_explicit_constants}

Theorem~1.1 of \cite{OpenAI2026PolynomialPEPS} asserts that its constants
exist. Its proof in Section~8 fixes them in a definite order
(\texttt{07-assembly.tex}, lines 179--214). This section writes the constants
of the large sizes as functions of three inputs: the constant of the square
area law, and the slope and the threshold of Proposition~3.4. Every other
choice is a fixed number.

\begin{lemma}[The distribution protocol at a given slope and given patch constants]
    \label{lem:peps_dyadic_protocol_at_given_scale}
    \lean{TNLean.PEPS.Approximation.largeSizeDyadicProtocol_of_angularScale}
    \leanok
    \uses{def:peps_dyadic_protocol,def:peps_angular_scale,
        lem:peps_adaptive_patch_supply,
        lem:peps_dyadic_protocol_mesh_ratio}
    Fix $q\ge2$ and reals $J,\Delta$. Let $\epsilon_0=1/30000$,
    $\nu=1/600000000$, $a_0=1/5000$ and
    $b=\max\{18,\lceil((4+8\epsilon_0+2\nu)/\nu+1)^2\rceil\}$. Let $D\ge1$ and
    $L_{\rm ang}$ be a slope and a threshold at accuracy $L^{-60}$ for the
    clearance $a_0/2$, the diameter ratio $1$ and at most $4$ marked points. Let
    $C>0$ and $c$ be constants with which Proposition~4.1 supplies adaptive
    patches on the cell-center sites at the radius constant $B_0=20D$ and power
    $60$, for every gapped ground vector of side $L\ge2$. Put $C'=\lceil C\rceil$,
    $c'=\lceil c\rceil$,
    \begin{align}
        G   & =(C'+1)^{3b}\bigl(16b^2(C'+1)^{6b}+1\bigr)^{2b}, &
        k_K & =3bc'+4b(3bc'+30).
    \end{align}
    Then for every count constant $C_N\ge5042$, expansion constant $C_K\ge G$
    and error constant $C_E\ge\max\{10080,2b+1\}$, every $L\ge2$ with
    $L\ge\lceil\max\{L_{\rm ang},1\}\rceil$ and every
    $(H,E_0,\Omega)\in\mathcal G_{q,J,\Delta}(L)$, there is a protocol with
    the Section~8 bounds of Definition~\ref{def:peps_dyadic_protocol} at the
    constants $C_N$, $C_K$, $C_E$, the bound $64\cdot189$, the exponent $k_K$
    and $2b$ pair effects.
\end{lemma}

\begin{proof}\leanok
    \uses{def:peps_dyadic_protocol_parties,
        def:peps_protocol_frozen_frames,
        thm:peps_angular_scale_monotonicity,
        thm:peps_dyadic_direct_repainting_constants,
        thm:peps_dyadic_padding_bounds,
        thm:peps_dyadic_protocol_schedule_counts,
        thm:peps_protocol_constants,
        thm:peps_protocol_from_large_repaintings,
        thm:peps_protocol_frozen_frame_bounds,
        thm:peps_protocol_large_repainting_segment,
        thm:peps_protocol_level_events,
        thm:peps_protocol_point_treatment_change}
    This is the argument of
    Theorem~\ref{thm:peps_protocol_from_angular_and_patch_bounds}, with the
    slope and the patch constants given instead of chosen. The mesh ratio $\nu$
    serves the resizes, the point treatments and the direct repaintings at once,
    so the constants $K_0=20$, $\epsilon_0$, $\nu$, $a=a_0/2$, $D$ and
    $B_0=K_0D$ satisfy the conditions of the protocol constants. For
    $L\ge\lceil\max\{L_{\rm ang},1\}\rceil$ the slope applies at side $L$.
\end{proof}

\begin{definition}[The constants of the polynomial PEPS approximation for large sizes]
    \label{def:polynomial_peps_explicit_constants}
    \lean{TNLean.PEPS.Approximation.pepsLinkExponentOf,
        TNLean.PEPS.Approximation.pepsLinkPrefactorOf,
        TNLean.PEPS.Approximation.pepsCongestion,
        TNLean.PEPS.Approximation.pepsCongestion_eq,
        TNLean.PEPS.Approximation.pepsNetworkErrorConstant,
        TNLean.PEPS.Approximation.pepsSizeThresholdOf,
        TNLean.PEPS.Approximation.pepsBondExponentOf,
        TNLean.PEPS.Approximation.pepsLargeSizePrefactorOf,
        TNLean.PEPS.Approximation.pepsLargeSizePrefactorOf_pos,
        TNLean.PEPS.Approximation.pepsBondExponentOf_pos}
    \leanok
    \uses{def:peps_adaptive_patch_constants,def:peps_compression_constants,
        def:peps_sampled_network_constants,
        thm:peps_dyadic_protocol_explicit_constants}
    Fix an integer $q$ and reals $J,\Delta,C_A,D,L_{\rm ang}$. Let
    $\epsilon_0=1/30000$, $\nu=1/600000000$, $b_{\rm prot}=64\cdot189$ and
    \begin{align}
        b=\max\{18,\lceil((4+8\epsilon_0+2\nu)/\nu+1)^2\rceil\}
        =5760768040000960009.
    \end{align}
    Let $C=\max\{q^{81},1+10D\}$ and $c$ be the constants of
    Definition~\ref{def:peps_adaptive_patch_constants} at $B_0=20D$, $p=60$
    and $C_A$, and let $G$ and $k_K$ be as in
    Theorem~\ref{thm:peps_dyadic_protocol_explicit_constants}, with
    $C'=\lceil C\rceil$ and $c'=\lceil c\rceil$. Let
    $C_E=\max\{10080,2b+1\}$. The link exponent and the link prefactor are
    \begin{align}
        k   & =\max\bigl\{2(k_K+4b(8+k_K)),\,2(8+4b_{\rm prot}k_K)\bigr\},   &
        C_1 & =C_\Sigma(2b,b_{\rm prot},5042\,b_{\rm prot},q,\max\{G,0\},G),
    \end{align}
    the constants of Definition~\ref{def:peps_sampled_network_constants} and
    Definition~\ref{def:peps_compression_constants} at $2b$ pair effects,
    arity $b_{\rm prot}$, gate-count exponent $3$, physical-dimension exponent
    $0$, monomial and coefficient-sum exponent $k_K$ and accuracy $L^{-5}$. Put
    \begin{align}
        \chi        & =8(2b_{\rm prot}+1)\,b_{\rm prot}\cdot2b_{\rm prot}(b_{\rm prot}-1)
        =685013510424821760,                                                              \\
        C_{\rm net} & =1+2\cdot5042\,C_E+2C_E,                                            \\
        L_0         & =\max\bigl\{\lceil\max\{L_{\rm ang},1\}\rceil,
        \lceil2C_{\rm net}\rceil\bigr\}.
    \end{align}
    The bond exponent and the bond prefactor for large sizes are
    \begin{align}
        c_{\rm PEPS}  & =k\chi+1,              &
        C_{\rm large} & =\max\{C_1,1\}^{\chi}.
    \end{align}
    Both are positive. Only $c_{\rm PEPS}$ depends on $C_A$; $C_{\rm large}$
    depends on $q$ and $D$, and $L_0$ on $L_{\rm ang}$ alone.
\end{definition}

\begin{theorem}[The polynomial PEPS approximation at given area-law and angular constants]
    \label{thm:polynomial_peps_of_ratios}
    \lean{TNLean.PEPS.Approximation.polynomialPEPSApproximation_of_ratios}
    \leanok
    \uses{def:polynomial_peps_explicit_constants,def:polynomial_peps_model,
        def:square_entropy_area_law_at,def:peps_angular_scale}
    Fix an integer $q\ge2$ and reals $J,\Delta>0$. Assume the square-grid
    entropy area law at $q$, $J$, $\Delta$ with constant $C_A$, and let $D\ge1$
    and $L_{\rm ang}$ be a slope and a threshold at accuracy $L^{-60}$ for the
    clearance $a_0/2$, $a_0=1/5000$, the diameter ratio $1$ and at most $4$
    marked points. Let $c_{\rm PEPS}$, $C_{\rm large}$ and $L_0$ be the
    constants of Definition~\ref{def:polynomial_peps_explicit_constants}. Then
    for every $L\ge2$ with $L\ge L_0$, every Hamiltonian of the model of
    Theorem~\ref{thm:openai_polynomial_peps} and every unit ground vector
    $\Omega$ with gap $\Delta$, there is a nonzero PEPS $\Phi$ on the
    $L\times L$ grid with $D_e\le C_{\rm large}L^{c_{\rm PEPS}}$ for every edge
    $e$ and
    \begin{align}
        \min_{\theta\in\mathbb R}
        \left\|\frac{\Phi}{\|\Phi\|}-e^{i\theta}\Omega\right\| & \le L^{-1}.
    \end{align}
    The area law and the slope are inputs of the proof of
    Theorem~\ref{thm:openai_polynomial_peps}, not hypotheses of that theorem,
    and the sizes $L<L_0$ are not covered here.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_adaptive_patch_supply_of_area_law_at,
        lem:peps_dyadic_protocol_at_given_scale,
        thm:peps_large_size_network_from_protocol,
        thm:peps_ground_state_approximation_conditional}
    By Theorem~\ref{thm:peps_adaptive_patch_supply_of_area_law_at} the area law
    supplies adaptive patches at $B_0=20D$ and power $60$ with the constants
    $C$ and $c$; moving them to the cell-center sites keeps the constants.
    Lemma~\ref{lem:peps_dyadic_protocol_at_given_scale} at $C_N=5042$, $C_K=G$
    and $C_E=\max\{10080,2b+1\}$ gives the protocol for
    $L\ge\lceil\max\{L_{\rm ang},1\}\rceil$. Compression at accuracy $L^{-5}$,
    as in Theorem~\ref{thm:peps_large_size_network_from_protocol}, gives an
    operator network with error $C_{\rm net}L^{-5}$, link dimensions at most
    $C_1L^k$, at most $b_{\rm prot}$ parties per block, at most
    $2b_{\rm prot}(b_{\rm prot}-1)$ links per party and separations at most
    $b_{\rm prot}$. For $L\ge\lceil2C_{\rm net}\rceil$ the routing argument of
    Theorem~\ref{thm:peps_ground_state_approximation_conditional} gives bonds at
    most $\max\{C_1,1\}^\chi L^{k\chi+1}$ and error at most $L^{-1}$.
\end{proof}
