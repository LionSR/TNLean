\begin{theorem}[Target entropy transfer]
    \label{thm:arealaw-target-entropy-transfer}
    \lean{TNLean.PEPS.AreaLaw.Scan.eventually_abs_regionalEntropy_sub_le_of_truncation}
    \leanok
    For $q\ge1$, $g>0$, $C_{\rm vol}$, $n\gg1$, any unit vectors $\Omega,\widetilde\Omega$, and any region $X$ with $|X|\le C_{\rm vol}n^2$,
    \begin{align}
        \operatorname{dist}_{\rm tr}
        (|\widetilde\Omega\rangle\langle\widetilde\Omega|,
        |\Omega\rangle\langle\Omega|)
        \le\sqrt{2\min\{n^{-1000},g/4\}/g}
        \implies
        |S(\widetilde\Omega_X)-S(\Omega_X)| \le n^{-247}.
        \label{eq:arealaw-target-entropy-transfer}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_regional_entropy_continuity}
    At trace distance $t\le\sqrt{2/g}\,n^{-500}$, Theorem~\ref{thm:area_law_regional_entropy_continuity} gives $|S(\widetilde\Omega_X)-S(\Omega_X)|\le4\sqrt t(1+|X|\log q)\le4(2/g)^{1/4}n^{-250}(1+C_{\rm vol}n^2\log q)\le n^{-247}$ for $n\gg1$.
\end{proof}

\begin{theorem}[The rough bound has the source entropy power]
    \label{thm:arealaw-source-rough-budget-power}
    \lean{TNLean.PEPS.AreaLaw.Scan.roughBudget_le_source_power}
    \leanok
    For $n\ge2$, $q\ge1$, $L\le n$, $1\le D\le2n^e$, $e,C_e\ge0$, and $W\ge1$, if
    \begin{align}
        1-n^{-100}   & \le z,                                                &
        S_t          & \le C_en^{1+e}+n^{-247},\notag                          \\
        |\log d-S_t| & \le n^{3/5}-\log z,                                   &
        B_{\rm sh}   & \le C_e(nL^e+n)+n^{-247}, \label{eq:rough_budget_hyp}
    \end{align}
    then
    \begin{align}
        2\log d-\log z+
        W\left(\frac{B_{\rm sh}}z+1+3nD\log q\right)
         & \le(6C_e+13+6\log q)Wn^{1+e}.
        \label{eq:arealaw-source-rough-budget-power}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Since $n\ge2$, $z\ge1-n^{-100}\ge1/2$ and $-\log z\le1$, so with $N=n^{1+e}$, (\ref{eq:rough_budget_hyp}) gives $2\log d-\log z\le(2C_e+7)N\le(2C_e+7)WN$, $B_{\rm sh}/z\le4(C_e+1)N$, and $1+3nD\log q\le2(1+3\log q)N$; summing yields (\ref{eq:arealaw-source-rough-budget-power}).
\end{proof}

\begin{theorem}[Principal terms of the selected-density bound]
    \label{thm:arealaw-selected-density-principal}
    \lean{TNLean.PEPS.AreaLaw.Scan.selectedDensity_budget_le_principal}
    \leanok
    For integers $n,m\ge1$, $W,\varepsilon,C_1>0$, $a,C_r,C_t\ge0$, and $e,C_l,B,G,R\in\mathbb R$, if $B\le C_rWn^{1+e}$ and $G\le C_t(\log n)^{C_l}$, then with $C=2(C_1+1)(C_r+2C_t)$,
    \begin{align}
        \frac{2(C_1+1)}{Wnm\varepsilon}
        \bigl(B+2nm\,GW a^{1/4}+R\bigr)
         & \le C\frac{n^e/m+a^{1/4}(\log n)^{C_l}}{\varepsilon}
        +\frac{2(C_1+1)}{Wnm\varepsilon}R.
        \label{eq:arealaw-selected-density-principal}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Substituting $B\le C_rWn^{1+e}$ and $G\le C_t(\log n)^{C_l}$, cancelling $Wn$, and bounding each nonnegative coefficient by $C=2(C_1+1)(C_r+2C_t)$ gives (\ref{eq:arealaw-selected-density-principal}).
\end{proof}

\begin{theorem}[Selected density at arbitrary scanner exponents]
    \label{thm:arealaw-selected-density-delta}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScannerExponents.selectedDensity_budget_le_delta}
    \leanok
    Under the exponents and rounded scales of Lemma~\ref{lem:al_scanner_scale_facts} with constant $C_s\ge1$, let $W,C_1>0$, $a=W/K$, $C_r,C_t\ge0$, $B\le C_rWn^{1+e}$, and $G\le C_t(\log n)^{C_l}$.
    Setting $c=\frac{2(C_1+1)}{Wnm(\epsilon/2)}$ and $C_\delta=4(C_1+1)(C_r+2C_t)C_s$, for all $R\in\mathbb R$,
    \begin{align}
        c\bigl(B+2nm\,GW a^{1/4}+R\bigr)
         & \le C_\delta\epsilon^{-1}
        \bigl(n^{e-\mu}+a^{1/4}(\log n)^{C_l}\bigr)+cR. \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:al_scanner_scale_facts,thm:arealaw-selected-density-principal}
    Applying (\ref{eq:arealaw-selected-density-principal}) of Theorem~\ref{thm:arealaw-selected-density-principal} at $\varepsilon=\epsilon/2$ and using $n^e/m\le C_sn^{e-\mu}$ ($C_s\ge1$) from Lemma~\ref{lem:al_scanner_scale_facts} yields the bound.
\end{proof}

\begin{theorem}[Energy from the selected density]
    \label{thm:arealaw-selected-energy-density}
    \lean{TNLean.PEPS.AreaLaw.Scan.selectedEnergy_le_of_density_le_add}
    \leanok
    For $d,\delta,\rho,A,t\ge0$, $g>0$, $d\le\delta+\rho$, $0\le p\le\epsilon$, and $e,r,E\in\mathbb R$,
    \begin{align}
        E \le\frac{e+A((1-p)d^{1/8}+(1-p)t+p)+r}{g}
        \implies
        E \le\frac A g(\delta^{1/8}+\epsilon+t)
        +\frac e g+\frac{A\rho^{1/8}+r}{g}. \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Combine $d^{1/8}\le(\delta+\rho)^{1/8}\le\delta^{1/8}+\rho^{1/8}$, $1-p\le1$, and $p\le\epsilon$.
\end{proof}

\begin{theorem}[Vanishing selected-energy remainder]
    \label{thm:arealaw-selected-energy-vanishing-remainder}
    \lean{TNLean.PEPS.AreaLaw.Scan.tendsto_selectedEnergy_remainder}
    \leanok
    For fixed $A,g\in\mathbb R$ (with division by $0$ equal to $0$), if $\rho_k\to0$ and $r_k\to0$ along a filter, then
    \begin{align}
        \frac{A|\rho_k|^{1/8}+r_k}{g} \longrightarrow0. \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Continuity of $x\mapsto|x|^{1/8}$ at $0$ and linearity give $A|\rho_k|^{1/8}+r_k\to0$, which is preserved upon dividing by $g$.
\end{proof}

\begin{theorem}[Ordered-limit closure at the selected point]
    \label{thm:arealaw-selected-density-power}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_final_selectedDensity_le}
    \leanok
    For $q\ge1$, $C_e,C_t\ge0$, $C_1>0$, and $C_l\in\mathbb R$, there exists $N\ge2$ such that for all $n\ge N$ with rounded scales
    \begin{align}
        L & =\lfloor n^{1-\ell}\rfloor,   &
        m & =\lfloor n^\mu\rfloor,        &
        K & =L\mathbin{\mathrm{div}}(8m), &
        D & =\lceil n^\kappa\rceil,\notag                                                            \\
        a & =W/K,                         & \varepsilon & =n^{-\nu}, & 1 & \le W\le n^\omega, \notag
    \end{align}
    if (\ref{eq:rough_budget_hyp}) holds at the source box-error exponent $e$, $G\le C_t(\log n)^{C_l}$, $R_k\to0$, and for $k\gg1$ there exists $(x,p)$ with $p\in[p_0,p_1]$ satisfying $\mathcal P_k(x,p)$ and
    \begin{align}
        d_k(x,p) & \le\frac{2(C_1+1)}{Wnm\varepsilon}
        \bigl(B_{\rm rough}+2nm\,GW a^{1/4}+R_k\bigr), \notag
    \end{align}
    then eventually $d_k(x,p)\le\varepsilon$ and $\mathcal P_k(x,p)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:arealaw-source-rough-budget-power,
        thm:arealaw-selected-density-principal,
        thm:arealaw-source-density-final}
    Combining (\ref{eq:arealaw-source-rough-budget-power}) and (\ref{eq:arealaw-selected-density-principal}) with Theorem~\ref{thm:arealaw-source-density-final} at coefficient $2C$ bounds the principal part by $\varepsilon/2$ for $n\ge N$, while $R_k\to0$ ensures $\frac{2(C_1+1)}{Wnm\varepsilon}R_k\le\varepsilon/2$ for $k\gg1$.
\end{proof}

In the physical application, fixing the target window and label sequence yields (\ref{eq:rough_budget_hyp}), and Theorem~\ref{thm:al_scanner_source} discharges the selection premise while preserving the energy and sharp-comparison bounds at $(x,p)$ (prior to final energy absorption, the entropy telescope, and the area law).
