\section{Marginal surprisal tails at a lattice cut}
\label{sec:area_law_marginal_tails}

The marginal moment and tail bounds of~\cite[Lemma~3.1]{OpenAI2026AreaLaw}
hold for any finite tensor product with designated supports. Here they are
instantiated for the finite-range Hamiltonians of
Definition~\ref{def:area_law_hamiltonian}.

\begin{definition}[Cut logarithmic budget]
    \label{def:area_law_cut_log_budget}
    \lean{TNLean.PEPS.AreaLaw.cutLogBudget}
    \leanok
    \uses{def:area_law_hamiltonian}
    For a cut $A\subseteq\Lambda$ put
    $\mathcal B_A=1+\sum_X\log^2(\mathrm e\,q^{|X|})$, the sum running over
    the admissible supports $X$ that meet both $A$ and its complement.
\end{definition}

\begin{theorem}[Supported operators act as the identity outside their support]
    \label{thm:area_law_supported_identity}
    \lean{QuantumCircuit.isSupportedOn_of_mem_supportedOperators}
    \leanok
    \uses{def:qc_local_circuit}
    An operator in the algebra generated by products acting as the identity
    outside $S$ has vanishing matrix elements between configurations that
    differ outside $S$, and its other matrix elements do not depend on the
    common values outside $S$.
\end{theorem}
\begin{proof}\leanok
    For a product $X=\bigotimes_im_i$ of one-site operators with $m_i=I$ for
    $i\notin S$,
    \begin{align}
        X_{\sigma\tau} & =\prod_i(m_i)_{\sigma_i\tau_i}
        =\prod_{i\in S}(m_i)_{\sigma_i\tau_i}\prod_{i\notin S}\delta_{\sigma_i\tau_i},
    \end{align}
    which vanishes when $\sigma$ and $\tau$ differ outside $S$ and otherwise
    depends only on $\sigma|_S,\tau|_S$. Both properties are preserved by sums
    and scalar multiples.
\end{proof}

\begin{theorem}[Marginal moment and tail bounds at a lattice cut]
    \label{thm:area_law_marginal_tails}
    \lean{TNLean.PEPS.AreaLaw.LocalHamiltonian.log_surprisalMoment_reducedState_le,
        TNLean.PEPS.AreaLaw.LocalHamiltonian.surprisalTail_reducedState_le}
    \leanok
    \uses{def:area_law_ground_entropy, def:area_law_cut_log_budget}
    Let $\Omega$ be a unit vector satisfying (\ref{eq:area_law_ground_vector})
    and (\ref{eq:area_law_projector_gap}) with $\Delta>0$ for a Hamiltonian whose terms have norm
    at most $J\ge0$, and let $A\subseteq\Lambda$. Let $K$ take the value
    $-\log p_j$ with probability $p_j$ for the eigenvalues $p_j$ of
    $\rho_A$, and put $\vartheta=J/\Delta$. Then
    $\log\mathbb E\,\mathrm e^{uK}\le uS_\Omega(A)+512\mathrm e\vartheta
        \mathcal B_Au^2$ for $|u|\le1/(32\sqrt{(1+\vartheta)\mathcal B_A})$, and
    for every real $w$
    \begin{align}
        \Pr\{|K-S_\Omega(A)|>w\} & \le\min\Bigl\{1,\,2\mathrm e^{\mathrm e/2}
        \exp\Bigl(-\frac{w}{32\sqrt{(1+\vartheta)\mathcal B_A}}\Bigr)\Bigr\}.
        \label{eq:area_law_marginal_tail}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_supported_identity, thm:area_law_cut_marginal}
    By Theorem~\ref{thm:area_law_cut_marginal} the regional state is the
    marginal of $\Omega$ across the cut $A\sqcup A^c$. Every term is supported
    on its admissible support, of dimension $q^{|X|}$, by
    Theorem~\ref{thm:area_law_supported_identity}, so the general lemma
    applies.
\end{proof}

\begin{theorem}[Regional state as a cut marginal]
    \label{thm:area_law_cut_marginal}
    \lean{TNLean.PEPS.AreaLaw.partialTraceRight_cutVector}
    \leanok
    \uses{def:area_law_ground_entropy}
    Writing $\Omega$ in the coordinates of $\mathcal H_A\otimes\mathcal H_{A^c}$,
    \begin{align}
        \rho_A & =\operatorname{Tr}_{A^c}|\Omega\rangle\langle\Omega|.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Both sides have entries
    $\sum_{z}\Omega(x\cup z)\overline{\Omega(y\cup z)}$ over configurations
    $z$ of $A^c$.
\end{proof}

\begin{theorem}[Typical marginal spectrum]
    \label{thm:area_law_typical_spectrum}
    \lean{TNLean.PEPS.AreaLaw.concentrationWidth,
        TNLean.PEPS.AreaLaw.LocalHamiltonian.one_sub_rpow_le_typicalMass_reducedState}
    \leanok
    \uses{def:area_law_ground_entropy, def:area_law_cut_log_budget,
        thm:area_law_marginal_tails}
    Under the hypotheses and notation of
    Theorem~\ref{thm:area_law_marginal_tails}, let $n\ge0$ be an
    integer and $M$ real, and put
    $w=32\sqrt{(1+\vartheta)\mathcal B_A}\,(M\log(n+2)+\mathrm e/2+\log2)$.
    The positive eigenvalues $p_j$ of $\rho_A$ with
    $|-\log p_j-S_\Omega(A)|\le w$ carry total mass at least $1-(n+2)^{-M}$.
    This is the concentration statement following Lemma~3.1 of
    \cite{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_marginal_tails}
    At this width the second argument of the minimum in
    (\ref{eq:area_law_marginal_tail}) equals $(n+2)^{-M}$, so the tail
    probability is at most $(n+2)^{-M}$; for $M<0$ this bound exceeds one and
    the conclusion is trivial. The indices of positive weight outside the
    typical set are exactly those of the tail event.
\end{proof}

\begin{theorem}[Supports crossing a general cut]
    \label{thm:area_law_crossing_supports}
    \lean{TNLean.PEPS.AreaLaw.card_admissibleSupport_containing_le,
        TNLean.PEPS.AreaLaw.card_crossingTerms_le_edgeBoundary}
    \leanok
    \uses{def:area_law_hamiltonian,
        thm:area_diamond_count, thm:area_diamond_budget}
    Put $v_R=1+2R(R+1)$ and $\mu_R=2^{v_R-1}$. At most $\mu_R$ admissible
    supports contain a given site, and for every $A\subseteq\Lambda$ at most
    $|\partial_\Lambda A|\,v_R\mu_R$ admissible supports meet both $A$ and its
    complement.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_diamond_count, thm:area_diamond_budget}
    Let $X$ meet $A$ in $v$ and its complement in $w$. A walk of length at
    most $R$ from $v$ to $w$ in the induced graph crosses an edge
    $\{x,y\}\in\partial_\Lambda A$ with $x\in A$, and its initial segment
    joins $v$ to $x$ in at most $R$ steps. Thus $X$ contains a site of the
    ball of radius $R$ about the endpoint in $A$ of a boundary edge. This
    ball has at most $v_R$ sites by Theorem~\ref{thm:area_diamond_count}, and
    each of them lies in at most $\mu_R$ admissible supports by
    Theorem~\ref{thm:area_diamond_budget}.
\end{proof}

\begin{theorem}[Tail parameter of a general cut]
    \label{thm:area_law_tail_parameter}
    \lean{TNLean.PEPS.AreaLaw.tailParameterConstant,
        TNLean.PEPS.AreaLaw.one_le_tailParameterConstant,
        TNLean.PEPS.AreaLaw.cutLogBudget_le_edgeBoundary}
    \leanok
    \uses{def:area_law_cut_log_budget, thm:area_law_crossing_supports}
    Let $q\ge1$ and put $C=v_R\mu_R\log^2(\mathrm e\,q^{v_R})$. Then $C\ge1$,
    and for every $A\subseteq\Lambda$,
    \begin{align}
        \mathcal B_A & \le C\,(1+|\partial_\Lambda A|).
        \label{eq:area_law_tail_parameter}
    \end{align}
    This is the bound on the tail parameter stated after Lemma~3.1 of
    \cite{OpenAI2026AreaLaw}; the constant depends only on $q$ and $R$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_crossing_supports, thm:area_diamond_budget}
    An admissible support has at most $v_R$ sites, so each summand of
    $\mathcal B_A$ is at most $\log^2(\mathrm e\,q^{v_R})$, and by
    Theorem~\ref{thm:area_law_crossing_supports} there are at most
    $|\partial_\Lambda A|\,v_R\mu_R$ summands. Hence
    $\mathcal B_A\le1+C|\partial_\Lambda A|\le C(1+|\partial_\Lambda A|)$,
    the last step because $C\ge1$.
\end{proof}

\begin{theorem}[Typical marginal spectrum at the boundary width]
    \label{thm:area_law_typical_spectrum_boundary}
    \lean{TNLean.PEPS.AreaLaw.typicalWidthConstant,
        TNLean.PEPS.AreaLaw.concentrationWidth_le,
        TNLean.PEPS.AreaLaw.typicalMass_le_typicalMass_of_le,
        TNLean.PEPS.AreaLaw.LocalHamiltonian.one_sub_rpow_le_typicalMass_edgeBoundary}
    \leanok
    \uses{thm:area_law_typical_spectrum, thm:area_law_tail_parameter}
    Under the hypotheses and notation of
    Theorem~\ref{thm:area_law_marginal_tails}, let $q\ge1$, let $n\ge0$ be
    an integer and $M\ge0$, and put
    \begin{align}
        C_M & =32\sqrt{(1+\vartheta)C}\,
        \Bigl(M+\frac{\mathrm e/2+\log2}{\log2}\Bigr)
    \end{align}
    with $C$ as in Theorem~\ref{thm:area_law_tail_parameter}. The positive
    eigenvalues $p_j$ of $\rho_A$ with
    \begin{align}
        |-\log p_j-S_\Omega(A)| & \le C_M\sqrt{1+|\partial_\Lambda A|}\,\log(n+2)
    \end{align}
    carry total mass at least $1-(n+2)^{-M}$. This is the concentration
    statement following Lemma~3.1 of \cite{OpenAI2026AreaLaw}, which is
    stated there for $n\ge1$ and $M>0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:area_law_typical_spectrum, thm:area_law_tail_parameter}
    Since $\log(n+2)\ge\log2$,
    \begin{align}
        M\log(n+2)+\mathrm e/2+\log2
         & \le\Bigl(M+\frac{\mathrm e/2+\log2}{\log2}\Bigr)\log(n+2),
    \end{align}
    and $\sqrt{(1+\vartheta)\mathcal B_A}\le\sqrt{(1+\vartheta)C}\,
    \sqrt{1+|\partial_\Lambda A|}$ by (\ref{eq:area_law_tail_parameter}).
    Both factors being nonnegative, the width of
    Theorem~\ref{thm:area_law_typical_spectrum} is at most
    $C_M\sqrt{1+|\partial_\Lambda A|}\log(n+2)$, and the mass of the typical
    set does not decrease when the width grows.
\end{proof}
