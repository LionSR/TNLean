\section{Chronological finite channel words}
\label{sec:al_channel_words}

Let $I,J$ be finite sets, $q\in\N$, $\mathcal H=(\C^q)^{\otimes I}$, and
$\mathcal K=\C^J$. For an event label set $F$, write $\mathcal E_i$ for the
full spectator root channel of $k_i$
(Definition~\ref{def:al_local_root_channel}). All operators on
$\mathcal H\otimes\mathcal K$ carry the Euclidean operator norm. Write $wi$
for the word $w$ with $i\in F$ appended on the right.

\begin{definition}[Observable after a chronological word]
    \label{def:al_channel_word}
    \uses{def:al_local_root_channel}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord}
    \leanok
    For $w=(i_0,\ldots,i_{m-1})\in F^m$ ($m\ge1$), define
    \begin{align}
        \label{eq:al_word_composition}
        \mathcal E_w(B)
         & =(\mathcal E_{i_{m-1}}\circ\cdots\circ\mathcal E_{i_0})(B),
    \end{align}
    and set $\mathcal E_{\varnothing}=\mathrm{id}$. For an initial operator
    $B_0$, write $B_w=\mathcal E_w(B_0)$.
\end{definition}

\begin{theorem}[Empty words and appending an event]
    \label{thm:al_channel_word_append}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_nil,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_append_singleton}
    \leanok
    For every operator $B$, finite word $w$, and label $i\in F$,
    \begin{align}
        \label{eq:al_word_empty}
        \mathcal E_{\varnothing}(B) & =B,                             \\
        \label{eq:al_word_append}
        \mathcal E_{wi}(B)          & =\mathcal E_i(\mathcal E_w(B)),
    \end{align}
    so $B_{\varnothing}=B_0$ and $B_{wi}=\mathcal E_i(B_w)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word}
    Immediate from Definition~\ref{def:al_channel_word} and
    (\ref{eq:al_word_composition}).
\end{proof}

\begin{theorem}[Relabelling chronological words]
    \label{thm:al_channel_word_relabel}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_map}
    \leanok
    For a map $f:E\to F$, set $\widetilde k_j=k_{f(j)}$ and
    $\widetilde{\mathcal E}_j=\mathcal E_{f(j)}$ ($j\in E$). For a word
    $w=(j_0,\ldots,j_{m-1})$ over $E$ and $f(w)=(f(j_0),\ldots,f(j_{m-1}))$,
    every operator $B$ satisfies
    \begin{align}
        \label{eq:al_word_relabel}
        \mathcal E_{f(w)}(B) & =\widetilde{\mathcal E}_w(B).
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word}
    Both sides of~(\ref{eq:al_word_relabel}) equal
    $(\mathcal E_{f(j_{m-1})}\circ\cdots\circ\mathcal E_{f(j_0)})(B)$ by
    (\ref{eq:al_word_composition}).
\end{proof}

\begin{theorem}[Channel words on a common ground vector]
    \label{thm:al_channel_word_ground}
    \uses{def:al_channel_word,def:al_local_root_channel}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannel_mulVec_ground,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_mulVec_ground}
    \leanok
    Following \cite[Section~10]{OpenAI2026AreaLaw}, suppose
    $0\le k_i\le\mathbb 1$ and $k_i\Omega=0$ for all $i\in F$, where
    $\Omega\in\mathcal H$. Set $G_i=(\mathbb 1-k_i)^{1/2}$,
    $R_w=G_{i_{m-1}}\cdots G_{i_0}$ for $w=(i_0,\ldots,i_{m-1})$, and
    $R_{\varnothing}=\mathbb 1$. For every $\xi\in\mathcal K$ and operator $B$
    on $\mathcal H\otimes\mathcal K$,
    \begin{align}
        \label{eq:al_word_ground_vector}
        \mathcal E_w(B)(\Omega\otimes\xi)
         & =(R_w\otimes\mathbb 1)B(\Omega\otimes\xi).
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_local_root_channel,thm:al_channel_word_append}
    Functional calculus gives $G_i\Omega=\Omega$ and $k_i^{1/2}\Omega=0$, so
    \begin{align}
        \label{eq:al_single_ground_vector}
        \mathcal E_i(B)(\Omega\otimes\xi)
         & =(G_i\otimes\mathbb 1)B(\Omega\otimes\xi).
    \end{align}
    Inducting on $w$ via~(\ref{eq:al_word_empty}), (\ref{eq:al_word_append}),
    and~(\ref{eq:al_single_ground_vector}) yields
    (\ref{eq:al_word_ground_vector}) since $G_iR_w=R_{wi}$.
\end{proof}

\begin{theorem}[Vector difference bounded by channel difference]
    \label{thm:al_channel_word_vector_difference}
    \uses{thm:al_channel_word_ground}
    \lean{TNLean.PEPS.AreaLaw.norm_rootProduct_sub_le_channelWord_sub}
    \leanok
    As in \cite[\texttt{09-amplification.tex}, lines 218--224]{OpenAI2026AreaLaw},
    under the hypotheses of Theorem~\ref{thm:al_channel_word_ground}, any two
    finite words $u,w$ satisfy
    \begin{align}
        \label{eq:al_word_vector_difference}
        \bigl\|(R_u\otimes\mathbb 1)B(\Omega\otimes\xi)
        -(R_w\otimes\mathbb 1)B(\Omega\otimes\xi)\bigr\|
         & \le\|\mathcal E_u(B)-\mathcal E_w(B)\|
        \|\Omega\otimes\xi\|.
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_ground}
    By (\ref{eq:al_word_ground_vector}), the left-hand side of
    (\ref{eq:al_word_vector_difference}) equals
    $\|(\mathcal E_u(B)-\mathcal E_w(B))(\Omega\otimes\xi)\|\le
        \|\mathcal E_u(B)-\mathcal E_w(B)\|\|\Omega\otimes\xi\|$.
\end{proof}

\begin{theorem}[Norm contraction along a finite word]
    \label{thm:al_channel_word_norm}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.norm_spectatorRootChannelWord_le}
    \leanok
    If $0\le k_i\le\mathbb 1$ for all $i\in F$, then
    $\|\mathcal E_w(B)\|\le\|B\|$ for every finite word $w$ and operator $B$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word,thm:al_root_channel_contraction}
    By induction on $w$: $\mathcal E_{\varnothing}(B)=B$, and for $w=iu$,
    Theorem~\ref{thm:al_root_channel_contraction} and the induction hypothesis
    give
    \begin{align}
        \label{eq:al_word_norm_step}
        \|\mathcal E_{iu}(B)\|
         & =\|\mathcal E_u(\mathcal E_i(B))\|
        \le\|\mathcal E_i(B)\|\le\|B\|.
    \end{align}
\end{proof}

\begin{theorem}[Oscillation bounded by the initial norm]
    \label{thm:al_channel_word_oscillation}
    \uses{def:al_channel_word,def:qc_site_oscillation}

\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_norm,thm:qc_site_oscillation_bounds}
    Theorems~\ref{thm:qc_site_oscillation_bounds}
    and~\ref{thm:al_channel_word_norm} give
    $0\le d_y(B_w)\le2\|B_w\|\le2\|B_0\|$.
\end{proof}

If $q>0$ and $(B_0)_{ab}\in\mathcal A_S$ for all $a,b\in J$,
then~(\ref{eq:al_word_empty}) and

\begin{theorem}[Summed append increments of the current observable]
    \label{thm:al_channel_word_kernel_increment}
    \uses{def:al_channel_word,def:qc_site_oscillation,
        def:al_graph_event_kernel,def:qc_local_circuit}
    \lean{TNLean.PEPS.AreaLaw.sum_siteOscillation_spectatorRootChannelWord_append_sub_le_graphChannelEventKernel}
    \leanok
    Suppose $q>0$, $F$ is finite, $G$ is a simple graph on $I$, and
    $0\le k_i\le\mathbb 1$ is supported on $C(a_i)$ ($a_i\in I$, $i\in F$).
    Let $C\ge0$, $c>0$, $0<\alpha\le1$, and $N\ge\max_{i\in F}R_{a_i}$. If
    every $i\in F$ and $0\le l\le N$ satisfy
    \begin{align}
        \label{eq:al_word_localization_error}
        \|k_i-E_{N_l(a_i)}(k_i)\| & \le C\exp(-cl^\alpha),
    \end{align}
    then for the kernel $\mathsf A^{(N)}$ of
    (\ref{eq:al_propagation_event_kernel}) with
    $D=2(2+8\sqrt C\exp(c/2))$ and $b=c/2$, every finite word $w$, $y\in I$,
    and $B_0$ satisfy
    \begin{align}
        \label{eq:al_word_kernel_increment}
        \sum_{i\in F}\bigl(d_y(B_{wi})-d_y(B_w)\bigr)
         & \le\sum_{z\in I}\mathsf A^{(N)}_{yz}d_z(B_w).
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_actual_channel_kernel_domination,thm:al_channel_word_append}
    Apply Theorem~\ref{thm:al_actual_channel_kernel_domination} to $B=B_w$ and
    use $\mathcal E_i(B_w)=B_{wi}$ from~(\ref{eq:al_word_append}).
\end{proof}

For $F_P=\{i\in F:P(i)\}$ with inclusion $j:F_P\hookrightarrow F$, let
$\mathcal E_w^P$ be the word composition for $k\circ j$. By
Theorem~\ref{thm:al_channel_word_relabel},
$\mathcal E_w^P(B_0)=B_{j(w)}$, so
(\ref{eq:al_word_kernel_increment}) on $F_P$ gives
\begin{align}
    \label{eq:al_word_retained_increment}
    \sum_{i\in F_P}\bigl(d_y(B_{j(wi)})-d_y(B_{j(w)})\bigr)
     & \le\sum_{z\in I}\mathsf A^{(N,P)}_{yz}d_z(B_{j(w)}).
\end{align}
For the lattice constraints of
Theorem~\ref{thm:al_lattice_channel_uniform_bounds}, applying
(\ref{eq:al_propagation_lattice_increment}) to $\mathcal E_w^P(B_0)$ via
(\ref{eq:al_word_relabel}) and~(\ref{eq:al_word_append}) yields
(\ref{eq:al_word_retained_increment}) with the same $C,c,v$ chosen before $P$.

\section{Expected spatial oscillation of channel words}
\label{sec:al_channel_poisson_growth}

For finite $F$ and $t\ge0$, assign each word $w=(i_0,\ldots,i_{m-1})\in F^m$
the Poisson mass
\begin{align}
    \label{eq:al_poisson_word_mass}
    \nu_t(\{w\}) & =e^{-|F|t}\frac{t^m}{m!}.
\end{align}
With $\rho_G$ from Definition~\ref{def:al_graph_extended_metric},
$d_G(y,S)=\inf_{z\in S}\rho_G(y,z)$, $\beta\ge0$, and $0<\alpha\le1$, define
\begin{align}
    \label{eq:al_poisson_distance_profile}
    H_S(y) & =
    \left\{\begin{array}{ll}
               \exp(-\beta d_G(y,S)^\alpha), & d_G(y,S)<\infty, \\
               0,                            & d_G(y,S)=\infty,
           \end{array}\right.
\end{align}
so $H_{\varnothing}=0$.

\begin{theorem}[A physical channel kernel gives expected spatial decay]
    \label{thm:al_channel_poisson_kernel_growth}
    \uses{def:al_channel_word,def:qc_site_oscillation,
        def:al_graph_extended_metric}
    \lean{TNLean.PEPS.AreaLaw.integrable_and_integral_siteOscillation_le_of_graph_kernel,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_ofFn_append_singleton}
    \leanok
    Suppose $q>0$, $G$ is a simple graph on $I$, and $A_{yz}\ge0$ satisfies
    $A_{yz}=0$ when $\rho_G(y,z)=\infty$. Assume that for all operators $W$
    and $y\in I$,
    \begin{align}
        \label{eq:al_poisson_physical_increment}
        \sum_{i\in F}\bigl(d_y(\mathcal E_iW)-d_y(W)\bigr)
         & \le\sum_{z\in I}A_{yz}d_z(W), \\
        \label{eq:al_poisson_weighted_rows}
        \sum_{z\in I}A_{yz}
        \exp\bigl(\beta (\rho_G(y,z)_{\R})^\alpha\bigr)
         & \le v,
    \end{align}
    where $\beta,v\ge0$ and $0<\alpha\le1$. If $B_{ab}\in\mathcal A_S$ for all
    $a,b\in J$ ($S\subseteq I$), then $w\mapsto d_y(B_w)$ is
    $\nu_t$-integrable for all $t\ge0$ and $y\in I$, with
    \begin{align}
        \label{eq:al_poisson_physical_decay}
        \mathbb E_{\nu_t}d_y(B_w)
         & \le e^{vt}\,2\|B\|\,H_S(y),
    \end{align}
    and $d_y(B_w)=0$ for all $w$ whenever $d_G(y,S)=\infty$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_append,thm:al_spectator_support_indicator,
        thm:al_graph_extended_components,thm:al_graph_extended_real_value}
    Following \cite[\texttt{09-amplification.tex}, lines 161--188]{OpenAI2026AreaLaw},
    set $f(w,y)=d_y(B_w)\ge0$. Theorem~\ref{thm:al_spectator_support_indicator}
    gives $f(\varnothing,y)\le2\|B\|\mathbf1_S(y)$, and
    $B_{wi}=\mathcal E_i(B_w)$ turns~(\ref{eq:al_poisson_physical_increment})
    into
    \begin{align}
        \label{eq:al_poisson_append_increment}
        \sum_{i\in F}\bigl(f(wi,y)-f(w,y)\bigr)
         & \le\sum_{z\in I}A_{yz}f(w,z).
    \end{align}
    Subadditivity of $r\mapsto r^\alpha$ and the triangle inequality give
    $(AH_S)(y)\le vH_S(y)$ from~(\ref{eq:al_poisson_weighted_rows}), so the
    Poisson-word growth bound yields integrability,
    (\ref{eq:al_poisson_physical_decay}), and $f(w,y)=0$ when
    $d_G(y,S)=\infty$.
\end{proof}

\begin{theorem}[Chronological restriction of a full word]
    \label{thm:al_channel_word_filter}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_filter}
    \leanok
    For $F_P=\{i\in F:P(i)\}$ and a word $w$ over $F$, let $\pi_P(w)$ be the
    subword over $F_P$ of letters satisfying $P$ and $w|_P=j(\pi_P(w))$ over
    $F$. Then
    \begin{align}
        \label{eq:al_channel_word_filter}
        \mathcal E_{w|_P}(B_0)
         & =\mathcal E^P_{\pi_P(w)}(B_0).
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_relabel}
    Apply Theorem~\ref{thm:al_channel_word_relabel} to the inclusion
    $j:F_P\hookrightarrow F$, since $j(\pi_P(w))=w|_P$.
\end{proof}

\begin{theorem}[Expectations of retained physical words]
    \label{thm:al_channel_word_filter_expectation}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.integrable_spectatorRootChannelWord_filter_iff,
        TNLean.PEPS.AreaLaw.integral_spectatorRootChannelWord_filter}
    \leanok
    Let $F$ be finite, $t\ge0$, and $\nu_{F,t}$ the measure
    (\ref{eq:al_poisson_word_mass}). For any predicate $P$ on $F$ and map
    $\Phi$ from operators on $\mathcal H\otimes\mathcal K$ to a normed real
    vector space, $w\mapsto\Phi(\mathcal E_{w|_P}(B_0))$ is
    $\nu_{F,t}$-integrable if and only if $u\mapsto\Phi(\mathcal E^P_u(B_0))$
    is $\nu_{F_P,t}$-integrable, and
    \begin{align}
        \label{eq:al_channel_word_filter_expectation}
        \int\Phi(\mathcal E_{w|_P}(B_0))\,d\nu_{F,t}(w)
         & =\int\Phi(\mathcal E^P_u(B_0))\,d\nu_{F_P,t}(u).
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_filter}
    Since $(\pi_P)_*\nu_{F,t}=\nu_{F_P,t}$ on discrete countable word spaces,
    substituting~(\ref{eq:al_channel_word_filter}) and changing variables along
    $\pi_P$ yields the equivalence and
    (\ref{eq:al_channel_word_filter_expectation}).
\end{proof}
