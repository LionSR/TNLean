\section{Chronological finite channel words}
\label{sec:al_channel_words}

Let $I,J$ be finite sets, let $q\in\N$, and put
$\mathcal H=(\C^q)^{\otimes I}$ and $\mathcal K=\C^J$.
The spectator dimension may be zero. Let $F$ be a set of event labels,
and write $\mathcal E_i$ for the full spectator root map associated
with the physical matrix $k_i$, as in
Definition~\ref{def:al_local_root_channel}.
All input operators are arbitrary complex matrices on
$\mathcal H\otimes\mathcal K$, with the Euclidean operator norm.
Write $wi$ for the word obtained by appending a single label $i$ to
the right of a finite word $w$.

\begin{definition}[Observable after a chronological word]
    \label{def:al_channel_word}
    \uses{def:al_local_root_channel}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord}
    \leanok
    For a nonempty word $w=(i_0,\ldots,i_{m-1})$, define
    \begin{align}
        \label{eq:al_word_composition}
        \mathcal E_w(B)
         & =(\mathcal E_{i_{m-1}}\circ\cdots\circ\mathcal E_{i_0})(B).
    \end{align}
    For the empty word $\varnothing$, take
    $\mathcal E_{\varnothing}=\mathrm{id}$.
    Thus the first event acts first. Given an initial operator $B_0$,
    write $B_w=\mathcal E_w(B_0)$.
\end{definition}

\begin{theorem}[Empty words and appending an event]
    \label{thm:al_channel_word_append}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_nil,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_append_singleton}
    \leanok
    For every operator $B$, every finite word $w$, and every label $i$,
    \begin{align}
        \label{eq:al_word_empty}
        \mathcal E_{\varnothing}(B) & =B,
    \end{align}
    \begin{align}
        \label{eq:al_word_append}
        \mathcal E_{wi}(B) & =\mathcal E_i(\mathcal E_w(B)).
    \end{align}
    In particular, $B_{\varnothing}=B_0$ and $B_{wi}=\mathcal E_i(B_w)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word}
    The empty composition is the identity. In the composition for
    $wi$, the last map is $\mathcal E_i$, after every map indexed by
    $w$, which gives~(\ref{eq:al_word_append}).
\end{proof}

\begin{theorem}[Relabelling chronological words]
    \label{thm:al_channel_word_relabel}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannelWord_map}
    \leanok
    Let $E$ be any label set and let $f:E\to F$ be any map.
    Define $\widetilde k_j=k_{f(j)}$ and
    $\widetilde{\mathcal E}_j=\mathcal E_{f(j)}$ for $j\in E$.
    For a word $w$ over $E$, let $f(w)$ be the word obtained by
    applying $f$ to each label. Then, for every operator $B$,
    \begin{align}
        \label{eq:al_word_relabel}
        \mathcal E_{f(w)}(B) & =\widetilde{\mathcal E}_w(B).
    \end{align}
    The map $f$ may identify distinct labels.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word}
    At every position in the chronological composition, both sides
    of~(\ref{eq:al_word_relabel}) apply the same map
    $\mathcal E_{f(j)}$. They start from the same input $B$ and
    apply these maps in the same order.
\end{proof}

\begin{theorem}[Channel words on a common ground vector]
    \label{thm:al_channel_word_ground}
    \uses{def:al_channel_word,def:al_local_root_channel}
    \lean{TNLean.PEPS.AreaLaw.spectatorRootChannel_mulVec_ground,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_mulVec_ground}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ and $k_i\Omega=0$ for every
    $i\in F$, where $\Omega\in\mathcal H$. Put
    $G_i=(\mathbb 1-k_i)^{1/2}$, and, for a chronological word
    $w=(i_0,\ldots,i_{m-1})$, put
    $R_w=G_{i_{m-1}}\cdots G_{i_0}$, with
    $R_{\varnothing}=\mathbb 1$.
    For every $\xi\in\mathcal K$ and every operator $B$ on
    $\mathcal H\otimes\mathcal K$, one has
    \begin{align}
        \label{eq:al_word_ground_vector}
        \mathcal E_w(B)(\Omega\otimes\xi)
         & =(R_w\otimes\mathbb 1)B(\Omega\otimes\xi).
    \end{align}
    This is the finite composition identity in
    \cite[Section~10]{OpenAI2026AreaLaw}.
    Neither vector need be normalized, and $B$ need not be Hermitian.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_local_root_channel,thm:al_channel_word_append}
    Functional calculus gives $G_i\Omega=\Omega$ and
    $k_i^{1/2}\Omega=0$. Consequently
    \begin{align}
        \label{eq:al_single_ground_vector}
        \mathcal E_i(B)(\Omega\otimes\xi)
         & =(G_i\otimes\mathbb 1)B(\Omega\otimes\xi).
    \end{align}
    Induct by appending the final event to a word. The empty word
    gives the identity. For $wi$, apply
    (\ref{eq:al_single_ground_vector}) to $\mathcal E_w(B)$,
    then the induction hypothesis; the resulting left factor is
    $G_iR_w=R_{wi}$.
\end{proof}

\begin{theorem}[Vector difference bounded by channel difference]
    \label{thm:al_channel_word_vector_difference}
    \uses{thm:al_channel_word_ground}
    \lean{TNLean.PEPS.AreaLaw.norm_rootProduct_sub_le_channelWord_sub}
    \leanok
    Under the hypotheses and notation of
    Theorem~\ref{thm:al_channel_word_ground}, let $u,w$ be any
    two finite words. Then
    \begin{align}
        \label{eq:al_word_vector_difference}
        \bigl\|(R_u\otimes\mathbb 1)B(\Omega\otimes\xi)
        -(R_w\otimes\mathbb 1)B(\Omega\otimes\xi)\bigr\|
         & \le\|\mathcal E_u(B)-\mathcal E_w(B)\|
        \|\Omega\otimes\xi\|.
    \end{align}
    Vector norms are Euclidean and the matrix norm is the induced
    operator norm. This is the pointwise conversion used in
    \cite[\texttt{09-amplification.tex}, lines 218--224]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_ground}
    By (\ref{eq:al_word_ground_vector}), the vector inside the norm
    is $(\mathcal E_u(B)-\mathcal E_w(B))(\Omega\otimes\xi)$.
    Apply the defining operator-norm inequality.
\end{proof}

\begin{theorem}[Norm contraction along a finite word]
    \label{thm:al_channel_word_norm}
    \uses{def:al_channel_word}
    \lean{TNLean.PEPS.AreaLaw.norm_spectatorRootChannelWord_le}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ for every $i\in F$.
    Then $\|\mathcal E_w(B)\|\le\|B\|$ for every finite word $w$
    and every operator $B$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:al_channel_word,thm:al_root_channel_contraction}
    Induct on the word, allowing an arbitrary input operator.
    The empty word gives equality. If the first label is $i$ and
    the remaining word is $u$, then
    \begin{align}
        \label{eq:al_word_norm_step}
        \|\mathcal E_{iu}(B)\|
         & =\|\mathcal E_u(\mathcal E_i(B))\|
        \le\|\mathcal E_i(B)\|\le\|B\|.
    \end{align}
    Here the first inequality is the induction hypothesis at input
    $\mathcal E_i(B)$ and the second is
    Theorem~\ref{thm:al_root_channel_contraction}.
\end{proof}

\begin{theorem}[Oscillation bounded by the initial norm]
    \label{thm:al_channel_word_oscillation}
    \uses{def:al_channel_word,def:qc_site_oscillation}
    \lean{TNLean.PEPS.AreaLaw.siteOscillation_spectatorRootChannelWord_bounds}
    \leanok
    Suppose $0\le k_i\le\mathbb 1$ for every $i\in F$.
    For every finite word $w$, site $y\in I$, and initial operator
    $B_0$, one has $0\le d_y(B_w)\le2\|B_0\|$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_norm,thm:qc_site_oscillation_bounds}
    The oscillation is nonnegative, and
    $d_y(B_w)\le2\|B_w\|\le2\|B_0\|$ by
    Theorems~\ref{thm:qc_site_oscillation_bounds} and
    \ref{thm:al_channel_word_norm}.
\end{proof}

At the empty word, the initial support bound is
Theorem~\ref{thm:al_spectator_support_indicator}.
If $q>0$ and every physical matrix block of $B_0$ is supported on
$S\subseteq I$, then~(\ref{eq:al_word_empty}) gives
\begin{align}
    \label{eq:al_word_initial_support}
    d_y(B_{\varnothing}) & =d_y(B_0)\le2\|B_0\|\mathbf1_S(y).
\end{align}

\begin{theorem}[Summed append increments of the current observable]
    \label{thm:al_channel_word_kernel_increment}
    \uses{def:al_channel_word,def:qc_site_oscillation,
        def:al_graph_event_kernel,def:qc_local_circuit}
    \lean{TNLean.PEPS.AreaLaw.sum_siteOscillation_spectatorRootChannelWord_append_sub_le_graphChannelEventKernel}
    \leanok
    Suppose $q>0$ and $F$ is finite. Let $G$ be a simple graph on
    $I$, and choose anchors $a_i\in I$ for $i\in F$.
    Assume $0\le k_i\le\mathbb 1$ and that $k_i$ is supported on
    the connected component of $a_i$ for every $i$.
    Let $C\ge0$, $c>0$, and $0<\alpha\le1$.
    Fix a cutoff $N\in\N$ with $N\ge R_{a_i}$ for every $i$,
    where $R_{a_i}$ is the maximum distance from $a_i$ within its
    component. Suppose that, for every $i\in F$ and every integer
    $l$ with $0\le l\le N$,
    \begin{align}
        \label{eq:al_word_localization_error}
        \|k_i-E_{N_l(a_i)}(k_i)\| & \le C\exp(-cl^\alpha).
    \end{align}
    Use the kernel~(\ref{eq:al_propagation_event_kernel}) with
    $D=2(2+8\sqrt C\exp(c/2))$ and $b=c/2$.
    Then, for every finite word $w$, every $y\in I$, and every
    initial operator $B_0$,
    \begin{align}
        \label{eq:al_word_kernel_increment}
        \sum_{i\in F}\bigl(d_y(B_{wi})-d_y(B_w)\bigr)
         & \le\sum_{z\in I}\mathsf A^{(N)}_{yz}d_z(B_w).
    \end{align}
    Every oscillation on the right is evaluated at the same current
    observable $B_w$. Labels with the same anchor remain distinct
    summands. The estimate holds for arbitrary finite spectators
    and arbitrary complex initial operators $B_0$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_actual_channel_kernel_domination,thm:al_channel_word_append}
    Apply Theorem~\ref{thm:al_actual_channel_kernel_domination} to
    the operator $B_w$. For each event $i$, replace
    $\mathcal E_i(B_w)$ by $B_{wi}$ using~(\ref{eq:al_word_append}).
    The resulting inequality is~(\ref{eq:al_word_kernel_increment}).
\end{proof}

For a finite retained family $F_P=\{i\in F:P(i)\}$, let
$j:F_P\hookrightarrow F$ be the inclusion and let $\mathcal E_w^P$
denote composition using the restricted effects $k\circ j$.
Theorem~\ref{thm:al_channel_word_relabel} identifies
the observable $\mathcal E_w^P(B_0)$ with
$B_{j(w)}=\mathcal E_{j(w)}(B_0)$ in the original family.
Under the hypotheses of Theorem~\ref{thm:al_channel_word_kernel_increment}
for this restricted family, (\ref{eq:al_word_kernel_increment}) gives
\begin{align}
    \label{eq:al_word_retained_increment}
    \sum_{i\in F_P}\bigl(d_y(B_{j(wi)})-d_y(B_{j(w)})\bigr)
     & \le\sum_{z\in I}\mathsf A^{(N,P)}_{yz}d_z(B_{j(w)}),
\end{align}
where $\mathsf A^{(N,P)}$ counts the retained labels and the cutoff
need only cover their anchors.
For the lattice constraints of
Theorem~\ref{thm:al_lattice_channel_uniform_bounds}, one may instead
apply~(\ref{eq:al_propagation_lattice_increment}) directly to
$\mathcal E_w^{\,P}(B_0)$ and use~(\ref{eq:al_word_relabel}) and
(\ref{eq:al_word_append}). Thus the same $C,c,v$, chosen before $P$,
give both the lattice weighted row bound and
(\ref{eq:al_word_retained_increment}) for every retained word.
Every effect in this composition is still constructed from the
original full Hamiltonian and its gapped ground vector.

\section{Expected spatial oscillation of channel words}
\label{sec:al_channel_poisson_growth}

Suppose $F$ is finite. For $t\ge0$, give each chronological word
$w=(i_0,\ldots,i_{m-1})$ the probability
\begin{align}
    \label{eq:al_poisson_word_mass}
    \nu_t(\{w\}) & =e^{-|F|t}\frac{t^m}{m!}.
\end{align}
These masses sum to one. The empty alphabet has only the empty word,
whose mass is one. In this section, $B_w$ always denotes the actual
composition~(\ref{eq:al_word_composition}), and expectations are taken
with respect to $\nu_t$.

Use the extended graph distance $\rho_G$ from
Definition~\ref{def:al_graph_extended_metric}, and write
$d_G(y,S)=\inf_{z\in S}\rho_G(y,z)$. For $\beta\ge0$ and
$0<\alpha\le1$, define the distance profile by
\begin{align}
    \label{eq:al_poisson_distance_profile}
    H_S(y) & =
    \begin{cases}
        \exp(-\beta d_G(y,S)^\alpha),&d_G(y,S)<\infty,\\
        0,&d_G(y,S)=\infty.
    \end{cases}
\end{align}
In particular, $H_{\varnothing}=0$.

\begin{theorem}[A physical channel kernel gives expected spatial decay]
    \label{thm:al_channel_poisson_kernel_growth}
    \uses{def:al_channel_word,def:qc_site_oscillation,
        def:al_graph_extended_metric}
    \lean{TNLean.PEPS.AreaLaw.integrable_and_integral_siteOscillation_le_of_graph_kernel,
        TNLean.PEPS.AreaLaw.spectatorRootChannelWord_ofFn_append_singleton}
    \leanok
    Suppose $q>0$, let $G$ be a simple graph on the finite physical
    site set $I$, and let $A_{yz}\ge0$ be a real kernel.
    Assume that, for every operator $W$ and every $y\in I$,
    \begin{align}
        \label{eq:al_poisson_physical_increment}
        \sum_{i\in F}\bigl(d_y(\mathcal E_iW)-d_y(W)\bigr)
         & \le\sum_{z\in I}A_{yz}d_z(W).
    \end{align}
    Suppose $A_{yz}=0$ when $\rho_G(y,z)=\infty$, and suppose
    $\beta\ge0$, $0<\alpha\le1$, and $v\ge0$ satisfy
    \begin{align}
        \label{eq:al_poisson_weighted_rows}
        \sum_{z\in I}A_{yz}
        \exp\bigl(\beta (\rho_G(y,z)_{\R})^\alpha\bigr)
         & \le v
    \end{align}
    for every $y\in I$. Let $S\subseteq I$, and suppose every
    physical matrix block of $B$ is supported on $S$.
    Then $w\mapsto d_y(B_w)$ is integrable for every $t\ge0$ and
    every $y\in I$, with
    \begin{align}
        \label{eq:al_poisson_physical_decay}
        \mathbb E_{\nu_t}d_y(B_w)
         & \le e^{vt}\,2\|B\|\,H_S(y).
    \end{align}
    If $d_G(y,S)=\infty$, then $d_y(B_w)=0$ for every word $w$.
    The input $B$ may be non-Hermitian and the spectator space may
    have dimension zero.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_channel_word_append,thm:al_spectator_support_indicator,
        thm:al_graph_extended_components,thm:al_graph_extended_real_value}
    Put $f(w,y)=d_y(B_w)\ge0$. The empty word gives
    $f(\varnothing,y)\le2\|B\|\mathbf1_S(y)$ by
    Theorem~\ref{thm:al_spectator_support_indicator}.
    The chronological identity $B_{wi}=\mathcal E_i(B_w)$ turns
    (\ref{eq:al_poisson_physical_increment}) into
    \begin{align}
        \label{eq:al_poisson_append_increment}
        \sum_{i\in F}\bigl(f(wi,y)-f(w,y)\bigr)
         & \le\sum_{z\in I}A_{yz}f(w,z).
    \end{align}
    The extended triangle inequality and subadditivity of
    $r\mapsto r^\alpha$ give $AH_S\le vH_S$ from
    (\ref{eq:al_poisson_weighted_rows}). Applying the spatial
    growth theorem for the atomic Poisson-word law to $f$ gives
    integrability and~(\ref{eq:al_poisson_physical_decay}).
    Its zero-profile conclusion gives $f(w,y)=0$ at infinite
    distance for every word. This is the finite-word form of the
    argument in \cite[\texttt{09-amplification.tex}, lines 161--188]{OpenAI2026AreaLaw}.
\end{proof}

\begin{theorem}[Localization tails give expected spatial decay]
    \label{thm:al_channel_poisson_graph_growth}
    \uses{thm:al_channel_word_kernel_increment,
        thm:al_channel_poisson_kernel_growth}
    \lean{TNLean.PEPS.AreaLaw.integrable_and_integral_siteOscillation_le_graphChannelEventKernel}
    \leanok
    Under the effect, component support, tail and cutoff hypotheses
    of Theorem~\ref{thm:al_channel_word_kernel_increment}, put
    $\beta=c/(4\cdot2^\alpha)$ and use the actual kernel
    $\mathsf A^{(N)}$ with $D=2(2+8\sqrt C e^{c/2})$ and $b=c/2$.
    Suppose its weighted rows satisfy
    (\ref{eq:al_poisson_weighted_rows}) for some $v\ge0$.
    For every initial operator $B$ whose physical blocks are
    supported on $S$, all conclusions of
    Theorem~\ref{thm:al_channel_poisson_kernel_growth} hold.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_actual_channel_kernel_domination,
        thm:al_graph_event_kernel_nonneg,thm:al_graph_kernel_disconnected,
        thm:al_channel_poisson_kernel_growth}
    Theorem~\ref{thm:al_actual_channel_kernel_domination} supplies
    (\ref{eq:al_poisson_physical_increment}) for the actual kernel.
    The kernel is nonnegative and has no entries between distinct
    components. Apply
    Theorem~\ref{thm:al_channel_poisson_kernel_growth}.
\end{proof}

\begin{theorem}[Uniform expected spatial decay on lattice domains]
    \label{thm:al_channel_poisson_lattice_growth}
    \uses{thm:al_lattice_channel_uniform_bounds,
        thm:al_channel_poisson_kernel_growth}
    \lean{TNLean.PEPS.AreaLaw.exists_latticeChannelWord_spatial_growth}
    \leanok
    Fix $p\ge1$, interaction radius $R\in\N$, interaction bound
    $J\ge0$, and gap $\Delta>0$. There are $c>0$ and $v\ge0$
    depending only on $p,R,J,\Delta$ with the following property.
    Take any local dimension $q>0$, finite lattice domain $\Lambda$,
    Hamiltonian, anchors and gapped ground vector satisfying the
    hypotheses of Theorem~\ref{thm:al_lattice_channel_uniform_bounds}.
    Retain any subfamily $F_P$ of its event labels, and use its actual
    positive constraints $k_i$, formed from the original Hamiltonian.
    Put $\alpha=p/(p+1)$ and $\beta=c/(4\cdot2^\alpha)$.
    For every finite spectator, support $S\subseteq\Lambda$,
    operator $B$ whose physical blocks are supported on $S$, and
    time $t\ge0$, the retained chronological word satisfies
    \begin{align}
        \label{eq:al_poisson_lattice_decay}
        \mathbb E_{\nu_t^{F_P}}d_y(B_w)
         & \le e^{vt}\,2\|B\|\,H_S(y).
    \end{align}
    The oscillation is integrable. It vanishes at every word when
    the component of $y$ does not meet $S$. The same $c,v$ work for
    all domains, local dimensions, retained families, spectators
    and initial operators, including empty retained families and
    empty supports.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_lattice_channel_uniform_bounds,
        thm:al_graph_event_kernel_nonneg,thm:al_graph_kernel_disconnected,
        thm:al_channel_poisson_kernel_growth}
    Choose the constants from
    Theorem~\ref{thm:al_lattice_channel_uniform_bounds} before all
    lattice and operator data. For a fixed retained family, take
    $N$ to be the maximum component radius of its anchors, with
    maximum zero for the empty family. That theorem gives both
    (\ref{eq:al_poisson_physical_increment}) and
    (\ref{eq:al_poisson_weighted_rows}) for the actual finite
    event kernel. The kernel is nonnegative and zero between
    components, so Theorem~\ref{thm:al_channel_poisson_kernel_growth}
    gives~(\ref{eq:al_poisson_lattice_decay}) and the remaining
    conclusions.
\end{proof}
