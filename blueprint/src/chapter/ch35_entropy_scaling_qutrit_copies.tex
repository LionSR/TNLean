\section{The repeated qutrit family}

\begin{definition}[A qutrit family on cubic-length chains]
    \label{def:qutrit_entropy_scaling_copies}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.qutritCopies,TNLean.MPS.AreaLaw.EntropyScaling.qutritCopies_length}
    \leanok
    \uses{def:qutrit_entropy_scaling_state,def:chain_copy_product}
    For $N\ge1$ and $\psi_{2N}=\psi_{2N}(1/N)$
    (Definition~\ref{def:qutrit_entropy_scaling_state}), define on consecutive
    $2N$-site blocks~\cite[lines 421--425]{Schuch2008EntropyScaling}:
    \begin{align}
        \label{eq:qutrit_entropy_scaling_copies}
        \chi_M=\psi_{2N}^{\otimes N^2},\qquad M=N^2(2N)=2N^3.
    \end{align}
\end{definition}

\begin{lemma}[Normalization of the repeated family]
    \label{thm:qutrit_entropy_scaling_copies_norm}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.norm_qutritCopies}
    \leanok
    \uses{def:qutrit_entropy_scaling_copies}
    For every $N\ge1$, $\|\chi_M\|=1$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qutrit_entropy_scaling_normalization,thm:chain_copy_product_inner}
    By Lemmas~\ref{thm:qutrit_entropy_scaling_normalization}
    and~\ref{thm:chain_copy_product_inner},
    $\|\chi_M\|=\|\psi_{2N}\|^{N^2}=1$.
\end{proof}

\begin{lemma}[A uniform four-bit entropy bound]
    \label{thm:qutrit_entropy_scaling_copies_entropy}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.regionEntropy_interval_qutritCopies_le}
    \leanok
    \uses{def:qutrit_entropy_scaling_copies}
    For every $N\ge1$ and every interval $I\subseteq[0,M)$,
    $S_{\chi_M}(I)\le4\log2$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qutrit_entropy_scaling_family_entropy,thm:chain_copy_interval_entropy,thm:qutrit_entropy_scaling_normalization}
    By Lemma~\ref{thm:qutrit_entropy_scaling_family_entropy} each single-copy
    region has entropy at most $2\log2$, and $I$ partially intersects at most
    two copies, so Lemma~\ref{thm:chain_copy_interval_entropy} gives
    $S_{\chi_M}(I)\le2(2\log2)=4\log2$.
\end{proof}

\begin{lemma}[The single-copy overlap obstruction]
    \label{thm:qutrit_entropy_scaling_copy_overlap}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.normSq_overlap_qutritFamily_le_kyFan}
    \leanok
    \uses{def:qutrit_entropy_scaling_state,def:qutrit_entropy_scaling_middle,def:chain_open_mps}
    For $N\ge1$, any unit MPS $\phi$ on $2N$ qutrits of bond dimension at most
    $D$, and the middle-cut marginal $\rho_N$ of $\psi_{2N}$ with Ky Fan
    $D$-norm $\|\rho_N\|_{(D)}$,
    \begin{align}
        |\langle\psi_{2N},\phi\rangle|^2\le\|\rho_N\|_{(D)}.
        \label{eq:qutrit_entropy_scaling_kyfan}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_cut_rank_of_mps}
    By Theorem~\ref{thm:chain_cut_rank_of_mps} the middle-cut coefficient
    matrix of $\phi$ has rank at most $D$, so Eckart--Young gives
    (\ref{eq:qutrit_entropy_scaling_kyfan}) for $D<3^N$, while for $D\ge3^N$
    Cauchy--Schwarz gives $|\langle\psi_{2N},\phi\rangle|^2\le1=\|\rho_N\|_{(D)}$.
\end{proof}

\begin{lemma}[The single-copy scalar error obstruction]
    \label{thm:qutrit_entropy_scaling_copy_scalar_error}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.qutritFamily_scalar_error_lower_bound}
    \leanok
    \uses{def:qutrit_entropy_scaling_state,def:chain_open_mps}
    For $N\ge2$ and any unit MPS $\phi$ of bond dimension at most $D$, set
    $f=|\langle\psi_{2N},\phi\rangle|^2$ and $T(f)=2\sqrt{f(1-f)}$ for
    $f\ge1/2$ ($T(f)=1$ for $f<1/2$). Then
    \begin{align}
        \label{eq:qutrit_entropy_scaling_scalar_tail}
        \frac{2^N-(D-1)}{N2^N}\le T(f).
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qutrit_entropy_scaling_copy_overlap,thm:qutrit_middle_discarded_mass,thm:qutrit_entropy_scaling_normalization}
    By Lemma~\ref{thm:qutrit_middle_discarded_mass} and
    (\ref{eq:qutrit_entropy_scaling_kyfan}),
    $(2^N-(D-1))/(N2^N)\le1-\|\rho_N\|_{(D)}\le1-f\le T(f)$.
\end{proof}

\begin{lemma}[Exponential bond dimension from scalar error]
    \label{thm:qutrit_entropy_scaling_scalar_dimension}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.bondDimension_qutritCopies_ge_of_scalar_error_lt_one}
    \leanok
    \uses{def:qutrit_entropy_scaling_copies,def:chain_open_mps}
    For $N\ge2$ and any unit MPS $\Phi$ on $M=2N^3$ sites of bond dimension at
    most $D$, if $t=T(|\langle\chi_M,\Phi\rangle|^2)<1$, then
    $D\ge2^N(1-2t)+1$ (with constant $2$ in place of the printed $8$ of
    \cite[lines 475--479]{Schuch2008EntropyScaling};
    see~\cite{gap:swvc08_qutrit_copy_constant}).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:chain_mps_copy_error_amplification,thm:qutrit_entropy_scaling_copy_scalar_error,thm:qutrit_entropy_scaling_normalization}
    Lemma~\ref{thm:chain_mps_copy_error_amplification} yields a unit
    single-copy MPS $\phi$ of bond dimension at most $D$ with
    $(N/2)T(|\langle\psi_{2N},\phi\rangle|^2)\le t$, so
    (\ref{eq:qutrit_entropy_scaling_scalar_tail}) gives
    $(2^N-(D-1))/(2\cdot2^N)\le t$.
\end{proof}

\begin{lemma}[Exponential bond dimension from full trace error]
    \label{thm:qutrit_entropy_scaling_trace_dimension}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.bondDimension_qutritCopies_ge}
    \leanok
    \uses{def:qutrit_entropy_scaling_copies,def:chain_open_mps}
    For $N\ge2$, any MPS $\Phi$ on $M=2N^3$ sites of bond dimension at most
    $D$, and $\delta=\|\lvert\chi_M\rangle\langle\chi_M\rvert-\lvert\Phi\rangle\langle\Phi\rvert\|_1$,
    \begin{align}
        D\ge2^N(1-2\delta)+1
        \label{eq:qutrit_entropy_scaling_trace_dim}
    \end{align}
    (with constant $2$ and unnormalized error $\delta$ in place of $8$ and the
    optimized error of \cite[lines 475--479]{Schuch2008EntropyScaling};
    see~\cite{gap:swvc08_qutrit_copy_constant}).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qutrit_entropy_scaling_scalar_dimension,thm:qutrit_entropy_scaling_copies_norm}
    For $\delta\ge1$, $2^N(1-2\delta)+1\le0\le D$. For $\delta<1$, $\Phi\ne0$
    and $\widehat\Phi=\Phi/\|\Phi\|$ is a unit MPS of bond dimension at most $D$
    with
    \begin{align}
        \label{eq:qutrit_entropy_scaling_actual_error}
        T\bigl(|\langle\chi_M,\widehat\Phi\rangle|^2\bigr)
        \le\bigl\|\lvert\chi_M\rangle\langle\chi_M\rvert-
        \lvert\Phi\rangle\langle\Phi\rvert\bigr\|_1=\delta<1,
    \end{align}
    so Lemma~\ref{thm:qutrit_entropy_scaling_scalar_dimension} and
    (\ref{eq:qutrit_entropy_scaling_actual_error}) give
    (\ref{eq:qutrit_entropy_scaling_trace_dim}).
\end{proof}

\begin{theorem}[No polynomial bond bound at fixed accuracy]
    \label{thm:qutrit_entropy_scaling_no_polynomial}
    \lean{TNLean.MPS.AreaLaw.EntropyScaling.eventually_no_polynomial_mps_approximation_qutritCopies}
    \leanok
    \uses{def:qutrit_entropy_scaling_copies,def:chain_open_mps}
    For any $\delta<1/8$, $C\in\R$, and $k\in\N$, and all sufficiently large
    $N$, no MPS $\Phi$ on $M=2N^3$ sites satisfies both $D\le CM^k$ and
    $\|\lvert\chi_M\rangle\langle\chi_M\rvert-\lvert\Phi\rangle\langle\Phi\rvert\|_1\le\delta$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:qutrit_entropy_scaling_trace_dimension}
    Since $1-2\delta>0$, for all sufficiently large $N$
    Lemma~\ref{thm:qutrit_entropy_scaling_trace_dimension} gives
    \begin{align}
        \label{eq:qutrit_entropy_scaling_polynomial_contradiction}
        C(2N^3)^k<2^N(1-2\delta)+1\le D,
    \end{align}
    contradicting $D\le CM^k$.
\end{proof}
