\section{Templates of belt tiles and replacements}
\label{sec:area_law_belt_replacement_templates}

The belt tiles and replacements of \cite[Section~11]{OpenAI2026AreaLaw} admit
templates (Definition~\ref{def:area_law_template}). A belt tile at layer $k$ has
side scale $u=t_k$ and is a fan run of its fine cell. A replacement with removal
half-side $u=d(S)$ is the image under $x\mapsto v+ux$ of a run of the normalized
eight-triangle fan on $[-1,1]^2$ (joining the center to the corners and side
midpoints, possibly with angle $>\pi$ or equal to the whole square). Throughout,
a point in $\mathbb Z^2$ supplies nonemptiness (templates are used only for
regions meeting the cut $A\subseteq\mathbb Z^2$), and replacements assume
$S_*\ge1$; both restrictions are recorded
in~\cite{gap:arealaw2d_belt_replacement_template_scope}.

\begin{lemma}[A transported normalized fan run]
    \label{thm:area_law_centered_fan_run_image}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanRunGraph_centered,
        TNLean.PEPS.AreaLaw.Geometry.cellFanRunRegion_centered_image}
    \leanok
    \uses{def:area_law_fan_runs,lem:area_law_isolated_star_mono}
    For $c\in\mathbb R^2$, $\ell\in\mathbb N$, and the closed dyadic square
    $c+[-2^{\ell-1},2^{\ell-1}]^2$ of side $2^\ell$, the fan run graph for any
    midpoint subdivisions and colors equals that of the normalized fan on
    $[-1,1]^2$, and the image of any normalized run under
    $\phi(x)=c+2^{\ell-1}x$ equals the union of the corresponding run on
    $c+[-2^{\ell-1},2^{\ell-1}]^2$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:area_law_isolated_star_mono}
    Since $\phi(x)=c+2^{\ell-1}x$ is a bijection mapping the endpoints of each
    normalized elementary side segment to those of the scaled square, adjacency
    in the run graphs (Definition~\ref{def:area_law_fan_runs}) and the union of
    indexed triangles are preserved under $\phi$.
\end{proof}

\begin{lemma}[Templates of initial belt tiles]
    \label{thm:area_law_belt_templates}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialBelt_exists_template}
    \leanok
    \uses{thm:area_law_fan_run_template,def:area_law_initial_birth_regions,
        def:area_law_fine_belt_scales}
    Let $C_{\mathrm{tpl}}\ge1$ and $C_B\ge16C_{\mathrm{tpl}}$. If a belt tile
    $B$ at layer $k$ satisfies $B\cap\mathbb Z^2\ne\varnothing$, then with
    $u=t_k$ there is a template $T=B\cap\mathbb Z^2$ with parameters
    $(s_0,n)=(u,\lceil C_Bu\rceil)$ and $N_0\le8$ closed fan triangles.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fan_run_template}
    By Definition~\ref{def:area_law_initial_birth_regions}, $B$ is a fan run of
    a fine cell of side $t_k$, so Lemma~\ref{thm:area_law_fan_run_template}
    applies.
\end{proof}

\begin{lemma}[Templates of replacements]
    \label{thm:area_law_replacement_templates}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_nat_eq_repairHalfSide,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementBirthRegion_exists_template,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacement_exists_template}
    \leanok
    \uses{thm:area_law_fan_run_template,thm:area_law_centered_fan_run_image,
        def:area_law_repair_birth,def:area_law_repair_square,
        def:area_law_repair_stage,def:area_law_repair_ancestors}
    Let $S_*\ge1$, $C_{\mathrm{tpl}}\ge1$, and $C_B\ge32C_{\mathrm{tpl}}$. If a
    replacement with parent scale $S$, removal half-side $u=d(S)$, and birth
    region $B$ satisfies $B\cap\mathbb Z^2\ne\varnothing$, then there is a
    template $T=B\cap\mathbb Z^2$ with parameters
    $(s_0,n)=(2u,\lceil C_Bu\rceil)$ and $N_0\le8$ fan triangles of the removal
    square (including wide sectors and homogeneous squares).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fan_run_template,thm:area_law_centered_fan_run_image}
    Since $S$ is nonterminal, $u=2^m\ge S_*\ge1$ ($m\in\mathbb N$). By
    Lemma~\ref{thm:area_law_centered_fan_run_image} with $\ell=m+1$, $B$ is a
    fan run of the square of side $2u$ centered at the parent mark; applying
    Lemma~\ref{thm:area_law_fan_run_template} with
    $C_B/2\ge16C_{\mathrm{tpl}}$ gives $(s_0,n)=(2u,\lceil C_Bu\rceil)$.
\end{proof}

\begin{lemma}[One scale constant for belt tiles and replacements]
    \label{thm:area_law_belt_replacement_templates}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_belt_replacement_templates}
    \leanok
    \uses{thm:area_law_belt_templates,thm:area_law_replacement_templates}
    For every $C_{\mathrm{tpl}}\ge1$, setting $C_B=32C_{\mathrm{tpl}}$ (before
    $S_*$ and $k_0$) ensures that every belt tile and replacement meeting
    $\mathbb Z^2$ has a template $T$ as in Lemmas~\ref{thm:area_law_belt_templates}
    and~\ref{thm:area_law_replacement_templates} with $N_0\le8$ and
    \begin{align}
        n                            & \ge C_{\mathrm{tpl}}N_0(s_0+1), &
        \operatorname{diam}_\infty T & \le n.
        \label{eq:area_law_belt_replacement_template_bounds}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_belt_templates,thm:area_law_replacement_templates}
    Take $C_B=32C_{\mathrm{tpl}}$ in Lemmas~\ref{thm:area_law_belt_templates}
    and~\ref{thm:area_law_replacement_templates}.
\end{proof}

\begin{lemma}[The dilation count is small]
    \label{thm:area_law_belt_replacement_scale_limit}
    \lean{TNLean.PEPS.AreaLaw.Geometry.eventually_beltReplacementScaleLimit}
    \leanok
    For any $C_B\in\mathbb R$ and $\varepsilon>0$, all sufficiently large $u$
    satisfy, with $\lceil x\rceil_+=\max\{0,\lceil x\rceil\}$,
    \begin{align}
        n=\lceil C_Bu\rceil_+,\qquad
        L=\bigl\lfloor n^{1-\ell}\bigr\rfloor & \le\varepsilon u,
        \label{eq:area_law_belt_replacement_scale_limit}
    \end{align}
    so $L=o(u)$.
\end{lemma}
\begin{proof}\leanok
    With $B=\max\{C_B,0\}+1$, $u\ge1$ gives $n\le Bu$ and
    $L\le B^{1-\ell}u^{-\ell}u\le\varepsilon u$ once
    $u^{-\ell}\le\varepsilon/B^{1-\ell}$ ($\ell>0$).
\end{proof}

\begin{lemma}[Retained regions have lattice points in their births]
    \label{thm:area_law_retained_birth_lattice_point}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_birth_oldInclusion,
        TNLean.PEPS.AreaLaw.Geometry.repairStage_exists_integerPoint_mem_birth,
        TNLean.PEPS.AreaLaw.Geometry.repairStage_exists_integerPoint_mem_birth_of_oldInclusion,
        TNLean.PEPS.AreaLaw.Geometry.repairStage_retained_exists_integerPoint_mem_birth}
    \leanok
    \uses{thm:area_law_repair_current_birth_containment,
        def:area_law_repair_inclusions,def:area_law_repair_region_sites}
    Retaining an identifier from generation $j$ to $j'\ge j$ preserves its birth
    region $B$; if its current region $R_{j'}$ at generation $j'$ meets
    $A\subseteq\mathbb Z^2$, then $B\cap\mathbb Z^2\ne\varnothing$ (in
    particular for every belt tile and replacement whose final region meets $A$).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_repair_current_birth_containment}
    By Definition~\ref{def:area_law_repair_inclusions} and
    Lemma~\ref{thm:area_law_repair_current_birth_containment},
    $\varnothing\ne R_{j'}\cap A\subseteq B\cap\mathbb Z^2$.
\end{proof}
