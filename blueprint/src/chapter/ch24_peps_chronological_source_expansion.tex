\section{Chronological partial source expansions}

We follow the chronological expansion in
\cite[proof of Theorem~5.2]{OpenAI2026PolynomialPEPS},
\texttt{04-compression.tex}, lines 342--434. Each gate has its own
participating parties and the common source data constructed above.
The following statements establish the exact partial expansion and
its scalar cost. The estimate for corrected source operators requires
the subsequent separation and second-moment arguments.

\begin{definition}[A chronological composition]
    \label{def:peps_source_circuit}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.IsAllowed}
    \leanok
    \uses{def:peps_prepared_source_gate,def:peps_party_registers}
    For a party set $P$, a chronological composition from register list $a$ to
    $b$ is built from identities, consecutive composition, private operations,
    adjacent register swaps, idle-register extensions, and nonprivate gates
    (each carrying a finite participant set $C\hookrightarrow P$ and common
    finite source data on $C$, acting on its input registers and as the identity
    elsewhere). The composition is \emph{allowed} if every private operation is
    a contraction; its complete operator $W:\mathcal H_a\to\mathcal H_b$
    evaluates the aggregate operator at each nonprivate gate.
\end{definition}

\begin{theorem}[Norm of the complete composition]
    \label{thm:peps_source_circuit_norm}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.norm_eval_le_one}
    \leanok
    \uses{def:peps_source_circuit}
    Every allowed chronological composition satisfies $\|W\|\leq1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_prepared_gate_grouped}
    By Theorem~\ref{thm:peps_prepared_gate_grouped} each gate is a contraction,
    and composition, identity extension, and register swaps preserve
    $\|\cdot\|\le1$.
\end{proof}

\begin{definition}[Gate occurrences and source positions]
    \label{def:peps_source_circuit_positions}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.gateLocations}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.gateLocationsFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.slotCount}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceLocations}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceLocationsFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.participants}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.endpoints}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.branchLabels}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.branchLabelsFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.branchCoefficient}
    \leanok
    \uses{def:peps_source_circuit}
    Let $\mathcal G$ be the finite set of nonprivate gate occurrences $g$, each
    with participants $C_g\subseteq P$, ordered source slots $E_g$, branch set
    $\Xi_g$, and coefficients $c_{g,\xi}$. The source positions are
    $\mathcal E=\{(g,e):g\in\mathcal G,\ e\in E_g\}$, with endpoints mapped
    from $e$ along $C_g\hookrightarrow P$.
\end{definition}

\begin{theorem}[Incidence of the original source positions]
    \label{thm:peps_source_circuit_incidence}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.endpoints_mem_participants}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.endpoints_ne}
    \leanok
    \uses{def:peps_source_circuit_positions}
    Each $(g,e)\in\mathcal E$ has two distinct endpoints in $C_g$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_prepared_source_gate}
    By Definition~\ref{def:peps_prepared_source_gate}, each slot $e\in E_g$ has
    distinct endpoints, which are preserved by the injection $C_g\hookrightarrow P$.
\end{proof}

\begin{definition}[The partial branch choices]
    \label{def:peps_partial_source_choices}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.IsTouched}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isTouched_gate_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.Choices}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.choicesFintype}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.partialWord}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.coefficient}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedEval}
    \leanok
    \uses{def:peps_source_circuit_positions,def:peps_affected_owner}
    For affected parties $\mathcal A\subseteq P$, let
    $\mathcal G_{\mathcal A}=\{g\in\mathcal G:C_g\cap\mathcal A\neq\varnothing\}$
    be the touched gate occurrences and
    $\Xi_{\mathcal A}=\prod_{g\in\mathcal G_{\mathcal A}}\Xi_g$ the partial
    branch choices $\xi=(\xi_g)_{g\in\mathcal G_{\mathcal A}}$. Each $\xi$
    defines a composition $W_\xi$ (replacing $g\in\mathcal G_{\mathcal A}$ by
    branch $\xi_g$, keeping complete operators at $g\notin\mathcal G_{\mathcal A}$,
    and grouping exterior owners via Definition~\ref{def:peps_affected_owner})
    with coefficient $c_\xi=\prod_{g\in\mathcal G_{\mathcal A}}c_{g,\xi_g}$ and
    weighted sum $\sum_{\xi\in\Xi_{\mathcal A}}c_\xi W_\xi$.
\end{definition}

\begin{theorem}[Allowed partial branches and their retained sources]
    \label{thm:peps_partial_source_inventory}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.isAllowed_partialWord}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedSources}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sources_partialWord}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.mem_sources_partialWord}
    \leanok
    \uses{def:peps_partial_source_choices}
    If the original composition is allowed, every $W_\xi$ ($\xi\in\Xi_{\mathcal A}$)
    is allowed, and its source list retains, in order and with multiplicity,
    precisely the sources of $g\in\mathcal G_{\mathcal A}$ having an endpoint in
    $\mathcal A$ (with exterior owners grouped and original endpoint spaces and
    vectors preserved).
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_affected_source_occurrences,thm:peps_prepared_gate_properties,thm:peps_grouped_block_map,thm:peps_prepared_gate_grouped}
    Induct on the composition using Theorems~\ref{thm:peps_prepared_gate_properties},
    \ref{thm:peps_prepared_gate_grouped}, \ref{thm:peps_grouped_block_map}, and
    \ref{thm:peps_affected_source_occurrences}: untouched gates are contractions
    on exterior memory, and exterior-endpoint pairs become local normalized
    preparations under owner grouping.
\end{proof}

\begin{theorem}[Exact chronological operator identity]
    \label{thm:peps_partial_source_evaluation}
    \lean{TNLean.Lattice.PartyCircuit.Word.eval_sum_appendTail}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.eval_partial_gate}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedEval_comp}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedEval_frame}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedEval_mapOwner}
    \leanok
    \uses{def:peps_partial_source_choices}
    With $I_a,I_b$ the isometries identifying the input and output memories with
    their grouped-owner memories,
    \begin{align}
        \left(\sum_{\xi\in\Xi_{\mathcal A}}c_\xi W_\xi\right)I_a
         & = I_bW.
        \label{eq:peps_partial_source_evaluation}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_grouped_gate_sum,thm:peps_prepared_gate_grouped,thm:peps_grouped_block_map,def:peps_source_circuit}
    By Theorem~\ref{thm:peps_grouped_gate_sum}, summing $\xi_g\in\Xi_g$ at
    $g\in\mathcal G_{\mathcal A}$ recovers the gate operator, while
    $g\notin\mathcal G_{\mathcal A}$ already carries its aggregate operator;
    inducting over consecutive compositions and idle-register extensions gives
    (\ref{eq:peps_partial_source_evaluation}).
\end{proof}

\begin{theorem}[Local contributions to the scalar cost]
    \label{thm:peps_partial_source_local_cost}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_gate_of_touches}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_gate_of_exterior}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_comp}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_frame}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_id}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_localMap}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_swap}
    \leanok
    \uses{def:peps_partial_source_choices}
    In $\sum_{\xi\in\Xi_{\mathcal A}}|c_\xi|$, a touched gate $g\in\mathcal G_{\mathcal A}$
    contributes $\sum_{\xi\in\Xi_g}|c_{g,\xi}|$, untouched gates, identities,
    private operations, and swaps contribute $1$, consecutive compositions
    multiply these sums, and idle-register extensions preserve them.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:peps_partial_source_choices}
    Direct from Definition~\ref{def:peps_partial_source_choices} and
    multiplicativity of $|\cdot|$ over Cartesian products of choice sets.
\end{proof}

\begin{theorem}[The exact product of the gate costs]
    \label{thm:peps_partial_source_product_cost}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_eq_prod}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_le_pow_card}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_mul_conj_eq_prod_sq}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sum_norm_coefficient_mul_conj_le_pow_card}
    \leanok
    \uses{def:peps_partial_source_choices}
    With $t=|\mathcal G_{\mathcal A}|$,
    \begin{align}
        \sum_{\xi\in\Xi_{\mathcal A}}|c_\xi|
                                         & =\prod_{g\in\mathcal G_{\mathcal A}}
        \sum_{\eta\in\Xi_g}|c_{g,\eta}|, &
        \sum_{\xi,\zeta\in\Xi_{\mathcal A}}|c_\xi\overline{c_\zeta}|
                                         & =\left(\sum_{\xi\in\Xi_{\mathcal A}}|c_\xi|\right)^2,
        \label{eq:peps_partial_source_product_cost}
    \end{align}
    which are bounded by $B^t$ and $B^{2t}$ when
    $\sum_{\eta\in\Xi_g}|c_{g,\eta}|\le B$ for all $g\in\mathcal G_{\mathcal A}$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_partial_source_local_cost,def:peps_partial_source_choices}
    Induction via Theorem~\ref{thm:peps_partial_source_local_cost} yields
    (\ref{eq:peps_partial_source_product_cost}), and bounding each of the $t$
    factors by $B$ gives $B^t$ and $B^{2t}$.
\end{proof}

\begin{theorem}[The density expansion in orthonormal coordinates]
    \label{thm:peps_partial_source_density}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.expandedEval_eq_mapOwner}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.toMatrix_eval_eq_sum_partialWord}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.density_eval_eq_sum_partialWord}
    \leanok
    \uses{thm:peps_partial_source_evaluation}
    In finite orthonormal bases of the grouped input and output memories, let
    $M$ and $K_\xi$ be the matrices of $I_bWI_a^{-1}$ and $W_\xi$. For every
    input matrix $\rho$,
    \begin{align}
        M & =\sum_{\xi\in\Xi_{\mathcal A}}c_\xi K_\xi,
        \label{eq:peps_partial_source_matrix}          \\
        M\rho M^\dagger
          & =\sum_{\xi,\zeta\in\Xi_{\mathcal A}}
        c_\xi\overline{c_\zeta}\,
        K_\xi\rho K_\zeta^\dagger.
        \label{eq:peps_partial_source_density}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_partial_source_evaluation}
    Right-multiplying (\ref{eq:peps_partial_source_evaluation}) by $I_a^{-1}$
    gives (\ref{eq:peps_partial_source_matrix}), and expanding $M\rho M^\dagger$
    gives (\ref{eq:peps_partial_source_density}).
\end{proof}

\begin{theorem}[Fixed positions for every partial source inventory]
    \label{thm:peps_partial_source_order}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceOrder}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.mem_sourceOrder}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.nodup_sourceOrder}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceOrder_gate}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceDims}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceAt}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceAt_isSome_iff}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sourceAt_eq_some_spec}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.filterMap_sourceAt_comp}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.filterMap_sourceAt_gate}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.sources_partialWord_eq_filterMap}
    \lean{TNLean.Lattice.PartyCircuit.SourceCircuit.layout_sources_partialWord}
    \leanok
    \uses{def:peps_source_circuit_positions,def:peps_partial_source_choices}
    The set $\mathcal E$ admits a fixed duplicate-free ordering (listing later
    compositions first and preserving intra-gate slot order), with fixed
    halfspaces $\C^{u_e},\C^{v_e}$ at each $e\in\mathcal E$. For every
    $\xi\in\Xi_{\mathcal A}$, the source inventory of $W_\xi$ is the sublist of
    positions $e\in\mathcal E$ meeting $\mathcal A$, each carrying its grouped
    endpoints, fixed halfspaces $\C^{u_e},\C^{v_e}$, and normalized branch
    vector; hence the surviving source-register layout is independent of $\xi$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:peps_partial_source_inventory,thm:peps_affected_source_occurrences}
    Induct on the composition using Theorems~\ref{thm:peps_partial_source_inventory}
    and \ref{thm:peps_affected_source_occurrences}: disjoint occurrence sets
    preserve duplicate-freeness under concatenation, and owner grouping drops a
    slot $e\in\mathcal E$ precisely when both endpoints lie outside $\mathcal A$.
\end{proof}
