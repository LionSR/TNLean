\paragraph{One virtual operation on every staircase window.}

The construction below gives the forward virtual/physical-operation
correspondence in the proof sketch of the two-dimensional corollary
\cite[corollary after Lemma~5, lines~2297--2444]{Molnar2018NormalPEPS}.

\begin{definition}[Partial state across a region cut]%
    \label{def:peps_region_partial_state}
    \lean{TNLean.PEPS.regionPartialState}
    \leanok
    For $R\subseteq V$ and a physical configuration $\tau$ on $V\setminus R$,
    define $\psi^\tau_{A,R}$ on the physical space of $R$ by
    \begin{align}
        \psi^\tau_{A,R}(\sigma)
         & =c_A(\omega_{R,\sigma,\tau}).
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Partial states of equal closed states]%
    \label{thm:peps_region_partial_state_same_state}
    \lean{TNLean.PEPS.regionPartialState_sameState}
    \leanok
    \uses{def:peps_region_partial_state}
    If $c_A=c_B$, then $\psi^\tau_{A,R}=\psi^\tau_{B,R}$ for every $R\subseteq V$
    and $\tau$.
\end{theorem}
\begin{proof}
    \leanok
    By Definition~\ref{def:peps_region_partial_state},
    $\psi^\tau_{A,R}(\sigma)=c_A(\omega_{R,\sigma,\tau})=c_B(\omega_{R,\sigma,\tau})=\psi^\tau_{B,R}(\sigma)$.
\end{proof}

\begin{theorem}[Blocked-tensor expansion of a partial state]%
    \label{thm:peps_region_partial_state_blocked_expansion}
    \lean{TNLean.PEPS.regionInteriorBondProd_smul_regionPartialState}
    \leanok
    \uses{def:peps_region_partial_state,
        def:peps_regionComplementBoundaryConfig}
    With $P_A(R)=\prod_{e\notin\partial R}D_A(e)$, every complement
    configuration $\tau$ satisfies
    \begin{align}
        P_A(R)\psi^\tau_{A,R}
         & =\sum_{\mu\in\partial R}
        T_{A,V\setminus R}(\bar\mu)(\tau)T_{A,R}(\mu).
        \label{eq:peps_partial_state_blocked_expansion}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Since $\bar\mu=\bar\nu\iff\mu=\nu$ on $\partial R$, the double boundary sum
    collapses to the diagonal $\mu=\nu$, and the non-boundary bonds contribute
    $\prod_{e\notin\partial R}D_A(e)=P_A(R)$, giving
    (\ref{eq:peps_partial_state_blocked_expansion}).
\end{proof}

\begin{definition}[Physical operator of a region insert]%
    \label{def:peps_physical_operator_of_region_insert}
    \lean{TNLean.PEPS.physicalOpOfRegionInsert}
    \leanok
    \uses{def:peps_region_out_endpoint_block_inverse}
    For an injective region $R$, a physical family
    $C=\{C(\mu)\}_{\mu\in\partial R}$, and a left inverse $L_{A,R}$ of the
    blocked tensor map, define
    \begin{align}
        O_C
         & =\left(c\longmapsto\sum_{\mu}c_\mu C(\mu)\right)L_{A,R}.
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Open-boundary realization of a region insert]%
    \label{thm:peps_physical_operator_of_region_insert_realizes}
    \lean{TNLean.PEPS.physicalOpOfRegionInsert_realizes}
    \leanok
    \uses{def:peps_physical_operator_of_region_insert}
    For every $\mu\in\partial R$, $O_C(T_{A,R}(\mu))=C(\mu)$.
\end{theorem}
\begin{proof}
    \leanok
    Since $L_{A,R}(T_{A,R}(\mu))=e_\mu$ by
    Definition~\ref{def:peps_region_out_endpoint_block_inverse},
    Definition~\ref{def:peps_physical_operator_of_region_insert} gives
    $O_C(T_{A,R}(\mu))=C(\mu)$.
\end{proof}

\begin{definition}[Region insert of a physical operator]%
    \label{def:peps_region_insert_of_physical_operator}
    \lean{TNLean.PEPS.regionInsertOfPhysicalOp}
    \leanok
    For a physical operator $O$ on $R$, define the insert
    $C^O_{A,R}(\mu)=O(T_{A,R}(\mu))$.
\end{definition}

\begin{theorem}[Closed state of a physical operator]%
    \label{thm:peps_closed_state_of_physical_operator}
    \lean{TNLean.PEPS.deformedRegionState_regionInsertOfPhysicalOp}
    \leanok
    \uses{def:peps_region_insert_of_physical_operator,
        thm:peps_region_partial_state_blocked_expansion}
    For every complement physical configuration $\tau$,
    \begin{align}
        \Psi_{A,R,C^O_{A,R}}(\sigma,\tau)
         & =\bigl[O\bigl(P_A(R)\psi^\tau_{A,R}\bigr)\bigr](\sigma).
        \label{eq:peps_physical_op_partial_state}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Apply $O$ to (\ref{eq:peps_partial_state_blocked_expansion}) by linearity
    and evaluate at $\sigma$.
\end{proof}

\begin{theorem}[Physical operator transferred between equal closed states]%
    \label{thm:peps_physical_operator_same_state_transfer}
    \lean{TNLean.PEPS.deformedRegionState_regionInsertOfPhysicalOp_sameState}
    \leanok
    \uses{thm:peps_closed_state_of_physical_operator,
        thm:peps_region_partial_state_same_state}
    If $c_A=c_B$, then
    \begin{align}
        P_B(R)\Psi_{A,R,C^O_{A,R}}(\sigma,\tau)
         & =P_A(R)\Psi_{B,R,C^O_{B,R}}(\sigma,\tau).
        \label{eq:peps_physical_op_same_state}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    By Theorem~\ref{thm:peps_region_partial_state_same_state},
    $\psi^\tau_{A,R}=\psi^\tau_{B,R}$, so (\ref{eq:peps_physical_op_partial_state})
    gives (\ref{eq:peps_physical_op_same_state}).
\end{proof}

\begin{definition}[Interior-bond region insert]%
    \label{def:peps_interior_bond_inserted_region_insert}
    \lean{TNLean.PEPS.interiorBondInsertedRegionInsert}
    \leanok
    \uses{def:peps_bond_inserted_region_insert, def:peps_corner_extension}
    For an ordered edge $e=(u,w)$ with $u,w\in R$, $u<w$, and
    $X\in\MN{D_A(e)}$, put $U=\{u\}$ and define the interior-bond insert on $R$ by
    \begin{align}
        I^{\mathrm{int}}_{A,R,e,X}
         & =
        \frac{P_A(R)}{P_A(U)}
        \Ext_{U\subseteq R}(I_{A,U,e,X}).
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Closed state of an interior-bond insert]%
    \label{thm:peps_interior_bond_inserted_state}
    \lean{TNLean.PEPS.deformedRegionStateAssembled_interiorBondInserted_eq_smul_edgeInsertedCoeff}
    \leanok
    \uses{def:peps_interior_bond_inserted_region_insert, thm:peps_corner_extension_state_preserves, thm:peps_deformedRegionStateAssembled_bondInserted_eq_smul_edgeInsertedCoeff}
    If $D_A(g)>0$ for all $g$, then for every global physical configuration
    $\omega$,
    \begin{align}
        \Psi^{\mathrm{as}}_{A,R,I^{\mathrm{int}}_{A,R,e,X}}(\omega)
         & =P_A(R)E_A(e,\omega,X).
        \label{eq:peps_interior_insert_state}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Since $u\in U=\{u\}$, $\mathcal O_{U,e}(X)=X$, so state preservation under
    $\Ext_{U\subseteq R}$ gives
    $\Psi^{\mathrm{as}}_{A,R,\Ext_{U\subseteq R}(I_{A,U,e,X})}(\omega)=P_A(U)E_A(e,\omega,X)$;
    multiplying by $P_A(R)/P_A(U)$ yields (\ref{eq:peps_interior_insert_state}).
\end{proof}

\begin{definition}[Staircase family for one virtual operation]%
    \label{def:peps_staircase_virtual_operation_insert}
    \lean{TNLean.PEPS.staircaseVirtualOperationInsert}
    \leanok
    \uses{def:peps_bond_inserted_region_insert, def:peps_interior_bond_inserted_region_insert, lem:peps_reference_edge_boundary_windows, lem:peps_reference_edge_interior_windows}
    For the horizontal reference edge $e$ and $X\in\MN{D_B(e)}$, define the
    insert $C_j(X)$ on the staircase window $W_j$ ($0\le j<L+K$) by
    \begin{align}
        C_j(X)
         & =
        \begin{cases}
            I_{B,W_0,e,X^{\mathsf T}},    & j=0,        \\
            I^{\mathrm{int}}_{B,W_j,e,X}, & 1\le j<L,   \\
            I_{B,W_j,e,X},                & L\le j<L+K.
        \end{cases}
        \notag
    \end{align}
    Here $W_0$ contains only the right endpoint of $e$, $W_1,\ldots,W_{L-1}$
    contain both endpoints, and $W_L,\ldots,W_{L+K-1}$ contain only the left
    endpoint.
\end{definition}

\begin{theorem}[Physical operators on the staircase windows]%
    \label{thm:peps_staircase_virtual_operation_physical_realization}
    \lean{TNLean.PEPS.staircaseVirtualOperationPhysicalOp}
    \lean{TNLean.PEPS.staircaseVirtualOperationPhysicalOp_realizes}
    \leanok
    \uses{def:peps_staircase_virtual_operation_insert, def:peps_physical_operator_of_region_insert, thm:peps_physical_operator_of_region_insert_realizes}
    If every $L\times K$ window is injective, then for each $0\leq j<L+K$ there
    is a physical operator $O_j(X)=O_{C_j(X)}$ on $W_j$ satisfying
    \begin{align}
        O_j(X)\bigl(T_{B,W_j}(\mu)\bigr) & =C_j(X)(\mu)
        \qquad (\mu\in\partial W_j).
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Each $W_j$ is an injective $L\times K$ translate, so
    Theorem~\ref{thm:peps_physical_operator_of_region_insert_realizes} yields
    $O_j(X)=O_{C_j(X)}$.
\end{proof}

\begin{lemma}[The two end inserts of the staircase family]%
    \label{lem:peps_staircase_virtual_operation_end_inserts}
    \lean{TNLean.PEPS.staircaseVirtualOperationInsert_zero}
    \lean{TNLean.PEPS.staircaseVirtualOperationInsert_last}
    \leanok
    \uses{def:peps_staircase_virtual_operation_insert}
    The end inserts of Definition~\ref{def:peps_staircase_virtual_operation_insert}
    are
    \begin{align}
        C_0(X)       & =I_{B,W_0,e,X^{\mathsf T}},
        \label{eq:peps_staircase_first}            \\
        C_{L+K-1}(X) & =I_{B,W_{L+K-1},e,X}.
        \label{eq:peps_staircase_last}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Evaluate Definition~\ref{def:peps_staircase_virtual_operation_insert} at
    $j=0$ and at $j=L+K-1\ge L$ (for $K>0$).
\end{proof}

\begin{theorem}[Common state of the staircase virtual-operation family]%
    \label{thm:peps_staircase_virtual_operation_common_state}
    \lean{TNLean.PEPS.deformedRegionStateAssembled_staircaseVirtualOperationInsert}
    \leanok
    \uses{def:peps_staircase_virtual_operation_insert, lem:peps_staircase_virtual_operation_end_inserts, lem:peps_reference_edge_left_endpoint_staircase_window, thm:peps_interior_bond_inserted_state, thm:peps_deformedRegionStateAssembled_bondInserted_eq_smul_edgeInsertedCoeff, thm:peps_regionInteriorBondProd_staircaseWindow_eq}
    If $B$ is translation invariant with $D_B(g)>0$ for all $g$, then for
    $0\le j<L+K$ and every global physical configuration $\omega$,
    \begin{align}
        \Psi^{\mathrm{as}}_{B,W_j,C_j(X)}(\omega)
         & =P_B(W_0)E_B(e,\omega,X).
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Since $\mathcal O_{W_0,e}(X^{\mathsf T})=X$ and
    $\mathcal O_{W_j,e}(X)=X$ for $L\le j<L+K$, while
    (\ref{eq:peps_interior_insert_state}) covers $1\le j<L$, each case gives
    $\Psi^{\mathrm{as}}_{B,W_j,C_j(X)}(\omega)=P_B(W_j)E_B(e,\omega,X)$, and
    translation invariance gives $P_B(W_j)=P_B(W_0)$.
\end{proof}

\begin{theorem}[Common staircase state transferred between tensors]%
    \label{thm:peps_staircase_physical_operation_same_state}
    \lean{TNLean.PEPS.deformedRegionStateAssembled_staircasePhysicalOp_sameState}
    \leanok
    \uses{thm:peps_physical_operator_same_state_transfer,
        thm:peps_staircase_virtual_operation_physical_realization,
        thm:peps_staircase_virtual_operation_common_state,
        thm:peps_regionInteriorBondProd_staircaseWindow_eq}
    Let $L,K>0$, $a+2L\leq\mathrm{width}$, and $2K\leq\mathrm{height}$. Let
    $A,B$ be translation-invariant tensors with $c_A=c_B$, every cyclic
    $L\times K$ window of $A$ injective, and $D_A(g)>0$ for all $g$. For
    $X\in\MN{D_A(e)}$, put
    $C_j^{A\to B}(X)(\mu)=O_j^A(X)(T_{B,W_j}(\mu))$. Then for all
    $j,j'\in\mathbb{N}$ (with $W_j=W_{L+K-1}$ for $j\geq L+K-1$),
    \begin{align}
        \Psi^{\mathrm{as}}_{B,W_j,C_j^{A\to B}(X)}
         & =\Psi^{\mathrm{as}}_{B,W_{j'},C_{j'}^{A\to B}(X)}.
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    By (\ref{eq:peps_physical_op_same_state}) and
    Theorem~\ref{thm:peps_staircase_virtual_operation_physical_realization},
    \begin{align}
        P_B(W_j)\Psi^{\mathrm{as}}_{A,W_j,C_j^A(X)}
         & =P_A(W_j)\Psi^{\mathrm{as}}_{B,W_j,C_j^{A\to B}(X)}.
        \notag
    \end{align}
    By Theorem~\ref{thm:peps_staircase_virtual_operation_common_state} and
    translation invariance ($P_A(W_j)=P_A(W_0)>0$, $P_B(W_j)=P_B(W_0)$), the
    left-hand side is independent of $j$; cancelling $P_A(W_0)$ gives the claim.
\end{proof}

\begin{theorem}[Cross-tensor virtual operation on the staircase edge]%
    \label{thm:peps_cross_tensor_virtual_operation_of_staircase}
    \lean{TNLean.PEPS.existsUnique_crossTensorVirtualOperation_of_staircasePhysicalOp_sameState}
    \leanok
    \uses{def:peps_staircase_O1_physical_operation,
        def:peps_staircase_O3_physical_operation}
    Let $L,K>0$ and choose staircase coordinates $(a,b)$ satisfying
    \begin{align}
        1    & \leq a,
             & a+2L                 & \leq \mathrm{width},
             & b+2K-1               & \leq \mathrm{height},
        \notag                                              \\
        2L+1 & \leq \mathrm{width},
             & 2K+1                 & \leq \mathrm{height}.
        \notag
    \end{align}
    Suppose $A,B$ are translation invariant with $c_A=c_B$, every cyclic
    $L\times K$ window of $A$ and of $B$ is injective, and $D_A(g),D_B(g)>0$
    for all $g$. For each $X\in\MN{D_A(e)}$, there is a unique
    $Y\in\MN{D_B(e)}$ such that for all $\eta_{\mathrm L},\eta_{\mathrm R},k$,
    \begin{align}
        O_1^A(X)\bigl(T_{B,W_{\mathrm L}}(k,\eta_{\mathrm L})\bigr)
         & =\sum_jY_{jk}T_{B,W_{\mathrm L}}(j,\eta_{\mathrm L}),
        \label{eq:peps_staircase_o1_cross_tensor_relation}       \\
        O_3^A(X)\bigl(T_{B,W_{\mathrm R}}(k,\eta_{\mathrm R})\bigr)
         & =\sum_jY_{kj}T_{B,W_{\mathrm R}}(j,\eta_{\mathrm R}).
        \label{eq:peps_staircase_o3_cross_tensor_relation}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_staircase_physical_operation_same_state,
        thm:peps_staircase_pair_insert_eq_open,
        thm:peps_existsUnique_virtualOperation_of_horizontal_staircase_end_pair,
        thm:peps_regionInjectivityUnionClosure_of_overlap}
    By Theorem~\ref{thm:peps_staircase_physical_operation_same_state}, the
    inserts $C_j^{A\to B}(X)$ share one closed $B$-state. Overlapping-union
    injectivity and corner completion identify the open-boundary end inserts
    $C_0^{A\to B}(X)=O_3^A(X)\circ T_{B,W_{\mathrm R}}$ and
    $C_{L+K-1}^{A\to B}(X)=O_1^A(X)\circ T_{B,W_{\mathrm L}}$, so the two-end
    comparison theorem yields the unique $Y\in\MN{D_B(e)}$ satisfying
    (\ref{eq:peps_staircase_o1_cross_tensor_relation})--(\ref{eq:peps_staircase_o3_cross_tensor_relation}).
\end{proof}

\begin{theorem}[Composition of the canonical staircase end operations]%
    \label{thm:peps_staircase_end_physical_operation_multiplication}
    \lean{TNLean.PEPS.staircaseO1PhysicalOp_mul}
    \lean{TNLean.PEPS.staircaseO3PhysicalOp_mul}
    \leanok
    \uses{def:peps_staircase_O1_physical_operation,
        def:peps_staircase_O3_physical_operation}
    Let $L,K>0$, $a+2L\leq\mathrm{width}$, and $2K\leq\mathrm{height}$, and
    suppose every cyclic $L\times K$ window of $A$ is injective. For all
    $X,Y\in\MN{D_A(e)}$,
    \begin{align}
        O_1^A(XY) & =O_1^A(X)O_1^A(Y),
        \notag                         \\
        O_3^A(XY) & =O_3^A(Y)O_3^A(X).
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_staircase_virtual_operation_end_inserts,
        thm:peps_lemma5_end_operation_multiplication}
    Under the boundary split $E$, the reference-edge insert $C_M$ on an end
    window $W$ satisfies
    $C_M(E^{-1}(j,\eta))=\sum_i M_{ij}T_{A,W}(E^{-1}(i,\eta))$, so
    $L_{A,W}\circ T_{A,W}=\Id$ gives $O_W(M)O_W(N)=O_W(MN)$. By
    Lemma~\ref{lem:peps_staircase_virtual_operation_end_inserts}, $O_1^A$ uses
    $M=X$ while $O_3^A$ uses $M=X^{\mathsf T}$, and
    $(XY)^{\mathsf T}=Y^{\mathsf T}X^{\mathsf T}$.
\end{proof}

\begin{theorem}[Multiplicativity of the cross-tensor assignment]%
    \label{thm:peps_staircase_cross_tensor_virtual_operation_multiplication}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_mul}
    \leanok
    \uses{def:peps_staircase_cross_tensor_virtual_operation}
    Under the hypotheses of
    Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase}, for all
    $X_1,X_2\in\MN{D_A(e)}$,
    \begin{align}
        Y_{A\to B}(X_1X_2)
         & =Y_{A\to B}(X_1)Y_{A\to B}(X_2).
        \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_staircase_end_physical_operation_multiplication,
        thm:peps_lemma5_end_operation_multiplication}
    Set $Y_r=Y_{A\to B}(X_r)$ for $r=1,2$. Iterating
    (\ref{eq:peps_staircase_o1_cross_tensor_relation}) and
    (\ref{eq:peps_staircase_o3_cross_tensor_relation}) gives
    \begin{align}
        O_1^A(X_1)\Bigl(O_1^A(X_2)
        \bigl(T_{B,W_{\mathrm L}}(k,\eta_{\mathrm L})\bigr)\Bigr)
         & =\sum_i (Y_1Y_2)_{ik}\,
        T_{B,W_{\mathrm L}}(i,\eta_{\mathrm L}),
        \notag                     \\
        O_3^A(X_2)\Bigl(O_3^A(X_1)
        \bigl(T_{B,W_{\mathrm R}}(k,\eta_{\mathrm R})\bigr)\Bigr)
         & =\sum_j (Y_1Y_2)_{kj}\,
        T_{B,W_{\mathrm R}}(j,\eta_{\mathrm R}).
        \notag
    \end{align}
    By Theorem~\ref{thm:peps_staircase_end_physical_operation_multiplication},
    the left-hand sides equal $O_1^A(X_1X_2)(T_{B,W_{\mathrm L}}(k,\eta_{\mathrm L}))$
    and $O_3^A(X_1X_2)(T_{B,W_{\mathrm R}}(k,\eta_{\mathrm R}))$, so uniqueness
    in Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase} gives
    $Y_{A\to B}(X_1X_2)=Y_1Y_2$.
\end{proof}

\paragraph{The algebra isomorphism on the displayed staircase edge.}

\begin{definition}[Separating one boundary coordinate]%
    \label{def:peps_boundary_split}
    \lean{TNLean.PEPS.regionBoundaryConfigSplitAt}
    \leanok
    For $f\in\partial R$, let $E_f:\mu\mapsto(\mu_f,\mu|_{\partial R\setminus\{f\}})$
    be the boundary coordinate split.
\end{definition}

\begin{theorem}[A bond insertion in separated coordinates]%
    \label{thm:peps_bond_insert_split}
    \lean{TNLean.PEPS.bondInsertedRegionInsert_splitAt}
    \leanok
    \uses{def:peps_boundary_split, def:peps_bond_inserted_region_insert}
    For every exterior configuration $\eta$ and bond index $j$,
    \begin{align}
        I_{A,R,f,M}\bigl(E_f^{-1}(j,\eta)\bigr)
         & =\sum_i M_{ij}T_{A,R}\bigl(E_f^{-1}(i,\eta)\bigr).
        \label{eq:peps_bond_insert_split}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Under $E_f$, $\mu|_{\partial R\setminus\{f\}}=E_f^{-1}(j,\eta)|_{\partial R\setminus\{f\}}$
    fixes the exterior coordinate to $\eta$, reducing the sum in
    Definition~\ref{def:peps_bond_inserted_region_insert} to
    (\ref{eq:peps_bond_insert_split}).
\end{proof}

\begin{definition}[Identification of physical operations and inserts]%
    \label{def:peps_region_identification}
    \lean{TNLean.PEPS.regionPhysicalOperationCongr}
    \lean{TNLean.PEPS.regionInsertCongr}
    \leanok
    An equality $R=S$ transports a physical operation $O$ and a region insert
    $C$ on $R$ to $O|_S$ and $C|_S$ on $S$.
\end{definition}

\begin{theorem}[Physical realization under region identification]%
    \label{thm:peps_realization_identification}
    \lean{TNLean.PEPS.regionInsertOfPhysicalOp_congr}

        def:peps_region_insert_of_physical_operator}

\end{theorem}
\begin{proof}
    \leanok
    Substitute $R=S$ in Definition~\ref{def:peps_region_insert_of_physical_operator}.
\end{proof}

\begin{theorem}[Blocked images of equal states]%
    \label{thm:peps_common_blocked_image}
    \lean{TNLean.PEPS.range_regionBlockedTensorMap_eq_span}
    \lean{TNLean.PEPS.range_regionBlockedTensorMap_eq_of_sameState}
    \leanok
    The blocked tensor map $\mathcal T_{A,R}$ has image
    $\operatorname{im}\mathcal T_{A,R}=\operatorname{span}\{T_{A,R}(\mu):\mu\in\partial R\}$.
    If $c_A=c_B$, $A$ and $B$ are injective on $V\setminus R$, and
    $D_A(g),D_B(g)>0$ for all $g$, then
    $\operatorname{im}\mathcal T_{A,R}=\operatorname{im}\mathcal T_{B,R}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_region_partial_state_same_state}
    Applying the left inverse of $\mathcal T_{A,V\setminus R}$ to
    (\ref{eq:peps_partial_state_blocked_expansion}) identifies
    $\operatorname{im}\mathcal T_{A,R}$ with $\operatorname{span}_\tau\{\psi^\tau_{A,R}\}$,
    and Theorem~\ref{thm:peps_region_partial_state_same_state} gives
    $\psi^\tau_{A,R}=\psi^\tau_{B,R}$.
\end{proof}

\begin{theorem}[Agreement of physical operations on a common blocked image]%
    \label{thm:peps_physical_operations_common_image}
    \lean{TNLean.PEPS.physicalOps_eq_on_blocks_of_sameState}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:peps_common_blocked_image}, if
    $OT_{A,R}(\mu)=PT_{A,R}(\mu)$ for all $\mu\in\partial R$, then
    $OT_{B,R}(\nu)=PT_{B,R}(\nu)$ for all $\nu\in\partial R$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_common_blocked_image}
    By Theorem~\ref{thm:peps_common_blocked_image},
    $T_{B,R}(\nu)\in\operatorname{im}\mathcal T_{B,R}=\operatorname{im}\mathcal T_{A,R}=\operatorname{span}\{T_{A,R}(\mu)\}$,
    where $O$ and $P$ agree by linearity.
\end{proof}

\begin{theorem}[Linearity and identity for boundary insertions]%
    \label{thm:peps_bond_insert_algebra}
    \lean{TNLean.PEPS.bondInsertedRegionInsert_add}
    \lean{TNLean.PEPS.bondInsertedRegionInsert_smul}
    \lean{TNLean.PEPS.bondInsertedRegionInsert_one}
    \leanok
    For every $c\in\mathbb C$ and $\mu\in\partial R$,
    $I_{A,R,f,M+N}=I_{A,R,f,M}+I_{A,R,f,N}$,
    $I_{A,R,f,cM}=cI_{A,R,f,M}$, and
    $I_{A,R,f,1}(\mu)=T_{A,R}(\mu)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_bond_insert_split}
    Linearity in $M$ and the specialization $M_{ij}=\delta_{ij}$ in
    (\ref{eq:peps_bond_insert_split}) give all three identities.
\end{proof}

\begin{theorem}[Boundary insertions determine the bond matrix]%
    \label{thm:peps_bond_insert_injective}
    \lean{TNLean.PEPS.bondInsertedRegionInsert_injective}
    \leanok
    If $A$ is injective on $R$ and $V\setminus R$ with $D_A(g)>0$ for all $g$,
    then $I_{A,R,f,M}=I_{A,R,f,N}$ implies $M=N$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_regionInsertedCoeff_injective,
        thm:peps_deformedRegionState_bondInsertedRegionInsert}
    By Theorem~\ref{thm:peps_deformedRegionState_bondInsertedRegionInsert},
    $I_{A,R,f,M}=I_{A,R,f,N}$ implies
    $C_A(R,f,M;\sigma,\tau)=C_A(R,f,N;\sigma,\tau)$, so
    Theorem~\ref{thm:peps_regionInsertedCoeff_injective} gives $M=N$.
\end{proof}

\begin{theorem}[Left staircase realization and linearity]%
    \label{thm:peps_staircase_left_realization}
    \lean{TNLean.PEPS.staircaseO1PhysicalOp_realizes_insert}
    \lean{TNLean.PEPS.staircaseO1PhysicalOp_add}
    \lean{TNLean.PEPS.staircaseO1PhysicalOp_smul}
    \leanok
    Let $L,K>0$, $1\leq a$, $a+2L\leq\mathrm{width}$,
    $b+2K-1\leq\mathrm{height}$, and $2K\leq\mathrm{height}$, and suppose every
    $L\times K$ window of $A$ is injective. Then for all $\mu\in\partial W_{\mathrm L}$
    and $c\in\mathbb C$,
    $O_1^A(X)T_{A,W_{\mathrm L}}(\mu)=I_{A,W_{\mathrm L},e,X}(\mu)$,
    $O_1^A(X+Y)=O_1^A(X)+O_1^A(Y)$, and $O_1^A(cX)=cO_1^A(X)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_staircase_virtual_operation_physical_realization,
        lem:peps_staircase_virtual_operation_end_inserts,
        thm:peps_realization_identification, thm:peps_bond_insert_algebra,
        def:peps_physical_operator_of_region_insert}
    Combine Theorem~\ref{thm:peps_staircase_virtual_operation_physical_realization},
    (\ref{eq:peps_staircase_last}), and Theorem~\ref{thm:peps_bond_insert_algebra}
    with the linearity of $C\mapsto O_C$ in
    Definition~\ref{def:peps_physical_operator_of_region_insert}.
\end{proof}

\begin{theorem}[A global left end relation is a boundary insertion]%
    \label{thm:peps_global_left_relation_insert}
    \lean{TNLean.PEPS.leftEnd_regionInsertOfPhysicalOp_eq_bondInserted}
    \leanok
    Let $L,K>0$, $1\leq a$, $a+2L\leq\mathrm{width}$, and
    $b+2K-1\leq\mathrm{height}$. If
    $OT_{A,W_{\mathrm L}}(j,\eta)=\sum_i M_{ij}T_{A,W_{\mathrm L}}(i,\eta)$
    for all $j,\eta$, then $C^O_{A,W_{\mathrm L}}=I_{A,W_{\mathrm L},e,M}$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_bond_insert_split}
    Compare the hypothesis with (\ref{eq:peps_bond_insert_split}) under the
    bijection identifying $(j,\eta)$ with $E_e(\mu)$.
\end{proof}

\begin{theorem}[Cross-tensor realization and normalized coefficients]%
    \label{thm:peps_cross_tensor_realization_coefficients}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_realizes}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_normalized_coeff}
    \leanok
    Under the hypotheses of
    Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase},
    write $F_{AB}(X)=Y_{A\to B}(X)$. Then for all $\nu\in\partial W_{\mathrm L}$
    and $\sigma,\tau$,
    $O_1^A(X)T_{B,W_{\mathrm L}}(\nu)=I_{B,W_{\mathrm L},e,F_{AB}(X)}(\nu)$ and
    \begin{align}
        P_B(W_{\mathrm L})C_A(W_{\mathrm L},e,X;\sigma,\tau)
         & =P_A(W_{\mathrm L})C_B(W_{\mathrm L},e,F_{AB}(X);\sigma,\tau).
        \label{eq:peps_cross_tensor_normalized_coefficients}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_cross_tensor_virtual_operation_of_staircase,
        thm:peps_global_left_relation_insert, thm:peps_staircase_left_realization,
        thm:peps_physical_operator_same_state_transfer,
        thm:peps_deformedRegionState_bondInsertedRegionInsert}
    Apply Theorem~\ref{thm:peps_global_left_relation_insert} to
    (\ref{eq:peps_staircase_o1_cross_tensor_relation}) to get
    $C^{O_1^A(X)}_{B,W_{\mathrm L}}=I_{B,W_{\mathrm L},e,F_{AB}(X)}$, and combine
    with $C^{O_1^A(X)}_{A,W_{\mathrm L}}=I_{A,W_{\mathrm L},e,X}$ from
    Theorem~\ref{thm:peps_staircase_left_realization} in
    (\ref{eq:peps_physical_op_same_state}) and
    Theorem~\ref{thm:peps_deformedRegionState_bondInsertedRegionInsert}.
\end{proof}

\begin{theorem}[Injectivity of a single-window complement]%
    \label{thm:peps_torus_window_complement_injectivity}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.windowComplement_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.regionBlockedTensorInjective_windowComplement}
    \leanok
    If $L,K>0$, $2L+1\leq\mathrm{width}$, $2K+1\leq\mathrm{height}$, every
    cyclic $L\times K$ window is injective, and injectivity is closed under
    binary unions, then $V\setminus W$ is injective for every cyclic $L\times K$
    window $W$.
\end{theorem}
\begin{proof}
    \leanok
    Decompose $V\setminus W$ as the union of a full-height band of width
    $\mathrm{width}-L\geq L+1$ and an $L$-column band of height
    $\mathrm{height}-K\geq K+1$, each covered by translates of $L\times K$
    windows, and apply union closure.
\end{proof}

\begin{theorem}[The cross-tensor algebraic laws and inverse]%
    \label{thm:peps_cross_tensor_algebra_laws}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_add}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_smul}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_one}
    \lean{TNLean.PEPS.staircaseCrossTensorVirtualOperation_reverse}
    \leanok
    Under the hypotheses of
    Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase}, for all
    $X,X'\in\MN{D_A(e)}$ and $c\in\mathbb C$,
    $F_{AB}(X+X')=F_{AB}(X)+F_{AB}(X')$, $F_{AB}(cX)=cF_{AB}(X)$,
    $F_{AB}(1)=1$, and $F_{BA}(F_{AB}(X))=X$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_cross_tensor_realization_coefficients,
        thm:peps_staircase_left_realization, thm:peps_bond_insert_algebra,
        thm:peps_bond_insert_injective, thm:peps_physical_operations_common_image,
        thm:peps_torus_window_complement_injectivity,
        thm:peps_regionInjectivityUnionClosure_of_overlap,
        lem:peps_horizontal_staircase_window_injectivity}
    Linearity follows from Theorems~\ref{thm:peps_staircase_left_realization},
    \ref{thm:peps_cross_tensor_realization_coefficients},
    \ref{thm:peps_bond_insert_algebra}, and~\ref{thm:peps_bond_insert_injective}
    (using Theorem~\ref{thm:peps_torus_window_complement_injectivity} for
    injectivity on $V\setminus W_{\mathrm L}$). Since $O_1^A(1)$ fixes
    $T_{A,W_{\mathrm L}}(\mu)$, Theorem~\ref{thm:peps_physical_operations_common_image}
    shows it fixes $T_{B,W_{\mathrm L}}(\nu)$, giving $F_{AB}(1)=1$. Similarly,
    $O_1^B(F_{AB}(X))$ and $O_1^A(X)$ agree on $T_{B,W_{\mathrm L}}(\nu)$ and
    hence on $T_{A,W_{\mathrm L}}(\mu)$, giving $F_{BA}(F_{AB}(X))=X$.
\end{proof}

\begin{theorem}[Equality of the displayed staircase bond dimensions]%
    \label{thm:peps_staircase_cross_tensor_bond_dimension}
    \lean{TNLean.PEPS.staircaseCrossTensor_bondDim_eq}
    \leanok
    Under the hypotheses of
    Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase},
    $D_A(e)=D_B(e)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_staircase_cross_tensor_algebra_isomorphism,
        thm:matrixAlgEquiv_fin_eq}
    By Definition~\ref{def:peps_staircase_cross_tensor_algebra_isomorphism},
    $D_A(e)^2=\dim_{\mathbb C}\MN{D_A(e)}=\dim_{\mathbb C}\MN{D_B(e)}=D_B(e)^2$,
    so $D_A(e)=D_B(e)$.
\end{proof}

\begin{theorem}[The conjugating gauge on the displayed staircase bond]%
    \label{thm:peps_staircase_cross_tensor_normalized_gauge}
    \lean{TNLean.PEPS.exists_staircaseWindowConjCoeffIdentity_normalized}
    \leanok
    Under the hypotheses of
    Theorem~\ref{thm:peps_cross_tensor_virtual_operation_of_staircase},
    there exist $h:D_A(e)=D_B(e)$ and $Z\in\mathrm{GL}(D_B(e),\mathbb C)$ such
    that for all $X\in\MN{D_A(e)}$ and $\sigma,\tau$,
    \begin{align}
        P_B(W_{\mathrm L})C_A(W_{\mathrm L},e,X;\sigma,\tau)
         & =P_A(W_{\mathrm L})
        C_B(W_{\mathrm L},e,Z\iota_h(X)Z^{-1};\sigma,\tau),
        \notag
    \end{align}
    where $\iota_h:\MN{D_A(e)}\to\MN{D_B(e)}$ is the reindexing induced by $h$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{def:peps_staircase_cross_tensor_algebra_isomorphism,
        thm:matrixAlgEquiv_inner_of_fin,
        thm:peps_cross_tensor_realization_coefficients}
    By the Skolem--Noether theorem
    (Theorem~\ref{thm:matrixAlgEquiv_inner_of_fin}),
    $F_{AB}(X)=Z\iota_h(X)Z^{-1}$ for some $Z\in\mathrm{GL}(D_B(e),\mathbb C)$;
    substituting into (\ref{eq:peps_cross_tensor_normalized_coefficients}) gives
    the claim.
\end{proof}
