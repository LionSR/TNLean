\subsection{Four cut spaces with common coordinate alphabets}
\label{sec:peps_four_cut_common_boundaries}

The following constructions give the common-coordinate specialization of the
four boundary spaces in \cite[Theorem~5.5]{Schuch2010PEPS}, whose full
statement is Theorem~\ref{thm:peps_four_cut_closure}. Let
$X=\Z/w\Z\times\Z/\ell\Z$ with $w,\ell>0$, $e=(1,0)$, and $n=(0,1)$;
horizontal bonds run from $v$ to $v+e$, vertical bonds from $v+n$ to $v$, and
all eight bonds remain distinct when $w=\ell=2$. Let $V$ and $P$ be finite
virtual and physical alphabets, and write $\mathcal N_A(O^{\mathrm h},O^{\mathrm v})$
for the physical vector of Definition~\ref{def:peps_torus_bond_network} with
independent site tensors $A_v:V^4\times P\to\C$.

\begin{definition}[Correlated cut boundaries and their ranges]%
    \label{def:peps_four_cut_common_boundary}
    \lean{TNLean.PEPS.TorusCutBoundaryConfig,
        TNLean.PEPS.torusCutBoundaryRestriction,
        TNLean.PEPS.torusCutInteriorWeight,
        TNLean.PEPS.torusCutCoeff,
        TNLean.PEPS.torusCutMap,
        TNLean.PEPS.torusCutMap_apply,
        TNLean.PEPS.torusCutSpace,
        TNLean.PEPS.mem_torusCutSpace_iff,
        TNLean.PEPS.fourTorusCutSpace,
        TNLean.PEPS.mem_fourTorusCutSpace_iff,
        TNLean.PEPS.card_torusCutBoundaryConfig_two}
    \leanok
    \uses{def:peps_torus_bond_network}
    The boundary configurations for the cut $(c,r)$ form
    $\mathcal E=(V\times V)^{\Z/\ell\Z}\times(V\times V)^{\Z/w\Z}$, with each
    pair ordered $(\text{tail},\text{head})$. For $\beta:X\to V^4$, put
    \begin{align}
        R_{c,r}(\beta)
                   & =\bigl(y\mapsto(\beta^1_{(c-1,y)},\beta^2_{(c-1,y)}),
        x\mapsto(\beta^3_{(x,r-1)},\beta^4_{(x,r-1)})\bigr),
        \label{eq:peps_common_boundary_restriction}                        \\
        s_v(\beta) & =(\beta_v^4,\beta_v^1,\beta_{v-n}^3,\beta_{v-e}^2),
        \notag                                                             \\
        W_{c,r}(\beta)
                   & =\prod_{v\in X}
        \begin{cases}1,&v_x+1=c,\\\delta_{\beta_v^2,\beta_v^1},&v_x+1\ne c\end{cases}
        \begin{cases}1,&v_y+1=r,\\\delta_{\beta_v^4,\beta_v^3},&v_y+1\ne r.\end{cases}
        \notag
    \end{align}
    For $M:\mathcal E\to\C$, define $\mathcal C^A_{c,r}$ by
    \begin{align}
        (\mathcal C^A_{c,r}M)(\sigma)
         & =\sum_{\beta:X\to V^4}M(R_{c,r}(\beta))W_{c,r}(\beta)
        \prod_{v\in X}A_v(s_v(\beta);\sigma_v),
        \label{eq:peps_common_cut_contraction}
    \end{align}
    and put $\mathcal S^A_{c,r}=\operatorname{ran}\mathcal C^A_{c,r}$, so
    $\psi\in\mathcal S^A_{c,r}$ iff $\psi=\mathcal C^A_{c,r}M$ for some $M$.
    On $X=(\Z/2\Z)^2$, put $\mathcal S^A=\bigcap_{c,r}\mathcal S^A_{c,r}$, with
    $|\mathcal E|=|V|^8$.
\end{definition}
\begin{proof}\leanok
    Linearity of (\ref{eq:peps_common_cut_contraction}) in $M$ gives the range
    and intersection characterizations, and $|\mathcal E|=|V|^{2\cdot2+2\cdot2}=|V|^8$
    on $(\Z/2\Z)^2$.
\end{proof}

\begin{definition}[Product boundaries and fixed endpoints]%
    \label{def:peps_four_cut_common_product_boundary}
    \lean{TNLean.PEPS.torusCutBondBoundary,
        TNLean.PEPS.torusCutBondBoundary_restriction,
        TNLean.PEPS.torusCutHorizontalUnit,
        TNLean.PEPS.torusCutVerticalUnit,
        TNLean.PEPS.torusCutBondBoundary_units}
    \leanok
    \uses{def:peps_four_cut_common_boundary}
    For bond matrices $O^{\mathrm h}_v,O^{\mathrm v}_v$, define
    \begin{align}
        M^{O}_{c,r}(\eta)
         & =\prod_y O^{\mathrm h}_{(c-1,y)}
        (\eta^{\mathrm h,2}_y,\eta^{\mathrm h,1}_y)
        \prod_x O^{\mathrm v}_{(x,r-1)}
        (\eta^{\mathrm v,2}_x,\eta^{\mathrm v,1}_x).
        \label{eq:peps_common_product_boundary}
    \end{align}
    Then $M^O_{c,r}(R_{c,r}(\beta))$ is the product of
    $O^{\mathrm h}_v(\beta_v^2,\beta_v^1)$ on $v_x+1=c$ and
    $O^{\mathrm v}_v(\beta_v^4,\beta_v^3)$ on $v_y+1=r$. For matrix units
    $E^{\mathrm h}_{\eta,v}=E_{\eta^{\mathrm h,2}_{v_y},\eta^{\mathrm h,1}_{v_y}}$
    and
    $E^{\mathrm v}_{\eta,v}=E_{\eta^{\mathrm v,2}_{v_x},\eta^{\mathrm v,1}_{v_x}}$,
    one has $M^{E_\eta}_{c,r}(\theta)=\delta_{\eta,\theta}$.
\end{definition}
\begin{proof}\leanok
    Substituting (\ref{eq:peps_common_boundary_restriction}) into
    (\ref{eq:peps_common_product_boundary}) selects the seam bonds, and the
    product of the matrix units $E^{\mathrm h}_{\eta,v},E^{\mathrm v}_{\eta,v}$
    equals $\delta_{\eta,\theta}$.
\end{proof}

\begin{lemma}[Fixed-boundary expansion of a cut range]%
    \label{thm:peps_four_cut_common_boundary_span}
    \lean{TNLean.PEPS.torusCutCoeff_bondBoundary,
        TNLean.PEPS.torusCutMap_eq_sum_single,
        TNLean.PEPS.torusCutSpace_eq_span_single,
        TNLean.PEPS.torusCutMap_single_eq_bondNetwork}
    \leanok
    \uses{def:peps_four_cut_common_product_boundary}
    Write $\widehat O^{\mathrm h,c}_v=O^{\mathrm h}_v$ on $v_x+1=c$ and $\Id$
    elsewhere, and define $\widehat O^{\mathrm v,r}$ analogously. Then
    \begin{align}
        \mathcal C^A_{c,r}M^O_{c,r}
         & =\mathcal N_A(\widehat O^{\mathrm h,c},\widehat O^{\mathrm v,r}),
        \label{eq:peps_common_boundary_network}                              \\
        \mathcal C^A_{c,r}M
         & =\sum_{\eta\in\mathcal E}M(\eta)\mathcal C^A_{c,r}\delta_\eta,
        \notag                                                               \\
        \mathcal S^A_{c,r}
         & =\spn_{\C}\{\mathcal C^A_{c,r}\delta_\eta:\eta\in\mathcal E\}.
        \notag
    \end{align}
    In particular, $\mathcal C^A_{c,r}\delta_\eta=\mathcal N_A(\widehat E^{\mathrm h,c}_\eta,\widehat E^{\mathrm v,r}_\eta)$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_four_cut_common_product_boundary}
    Multiplying $M^O_{c,r}(R_{c,r}(\beta))$ of
    Definition~\ref{def:peps_four_cut_common_product_boundary} by $W_{c,r}(\beta)$
    gives (\ref{eq:peps_common_boundary_network}). Expanding
    $M=\sum_\eta M(\eta)\delta_\eta$ by linearity of $\mathcal C^A_{c,r}$ gives
    the span description of $\mathcal S^A_{c,r}$, and $M^{E_\eta}_{c,r}=\delta_\eta$
    gives the final specialization.
\end{proof}

\begin{definition}[Matching bond actions and their averaging tensors]%
    \label{def:peps_four_cut_common_matched_action}
    \lean{TNLean.PEPS.torusMatchedLegMatrix,
        TNLean.PEPS.torusMatchedLegRep,
        TNLean.PEPS.torusMatchedLegRep_apply,
        TNLean.PEPS.torusMatchedLegRep_const,
        TNLean.PEPS.IsUnitaryTorusBondFamily,
        TNLean.PEPS.IsSemiRegularTorusBondFamily,
        TNLean.PEPS.representationAveragingSite,
        TNLean.PEPS.siteMap_representationAveragingSite,
        TNLean.PEPS.representationAveragingSite_torusLegRep,
        TNLean.PEPS.torusMatchedAveragingSites}
    \leanok
    \uses{def:peps_torus_bond_network,def:peps_g_injective,def:peps_semiregular}
    Following \cite[Definition~5.1]{Schuch2010PEPS}, let $G$ be a group and
    assign representations $U^{\mathrm h}_v,U^{\mathrm v}_v$ on $\C^V$ to the
    individual bonds, acting on the $(\text{top},\text{right},\text{down},\text{left})$
    coordinates of site $v$ by
    \begin{align}
        \rho_v(k)
         & =U^{\mathrm v}_v(k)\otimes
        U^{\mathrm h}_v(k^{-1})^{\mathsf T}\otimes
        U^{\mathrm v}_{v-n}(k^{-1})^{\mathsf T}\otimes
        U^{\mathrm h}_{v-e}(k).
        \label{eq:peps_common_matched_action}
    \end{align}
    The bond family is unitary when every $U^d_v(k)$ is unitary, and for finite
    $G$ is semi-regular when every $U^d_v$ is semi-regular (Definition~\ref{def:peps_semiregular}).
    For any representation $\rho$ of finite $G$ on $\C^{V^4}$, put
    $\Pi_\rho=|G|^{-1}\sum_k\rho(k)$ and
    $A^0_\rho(t,r,b,l;s)=(\Pi_\rho)_{s,(t,r,b,l)}$, with site map $\Pi_\rho$;
    write $A^0_v=A^0_{\rho_v}$ for the matched family, which recovers the
    uniform four-leg action when $U^d_v=U$ is constant.
\end{definition}
\begin{proof}\leanok
    Since $k\mapsto U^d_v(k^{-1})^{\mathsf T}$ is a representation, so is
    (\ref{eq:peps_common_matched_action}); the site map of $A^0_\rho$ and the
    constant specialization $U^d_v=U$ follow directly from the definitions.
\end{proof}

\begin{definition}[Site-dependent matched closures]%
    \label{def:peps_four_cut_common_closure}
    \lean{TNLean.PEPS.sitewiseTorusGClosure,
        TNLean.PEPS.sitewiseTorusGClosure_const,
        TNLean.PEPS.sitewiseCommutingClosureSpan,
        TNLean.PEPS.torusMatchedHorizontalClosureAt,
        TNLean.PEPS.torusMatchedVerticalClosureAt,
        TNLean.PEPS.matchedTorusGClosure,
        TNLean.PEPS.matchedTorusGClosure_const,
        TNLean.PEPS.matchedCommutingClosureSpan}
    \leanok
    \uses{def:peps_four_cut_common_matched_action,def:peps_torus_g_closure}
    For $g,h\in G$, define
    \begin{align}
        H^c_v(h)          & =\begin{cases}U^{\mathrm h}_v(h),&v_x+1=c,\\\Id,&v_x+1\ne c,\end{cases}
        \notag                                                                                      \\
        V^r_v(g)          & =\begin{cases}U^{\mathrm v}_v(g),&v_y+1=r,\\\Id,&v_y+1\ne r,\end{cases}
        \notag                                                                                      \\
        \mathcal M_A(g,h) & =\mathcal N_A(H^0(h),V^0(g)),
        \label{eq:peps_common_matched_closure}                                                      \\
        \mathcal Z_A      & =\spn_{\C}\{\mathcal M_A(g,h):gh=hg\}.
        \notag
    \end{align}
    When $U^d_v=U$ is constant, these give the site-dependent uniform closures,
    and if also $A_v=A$ is constant, $\mathcal M_A(g,h)=\mathcal M(A\mid g,h)$
    of Definition~\ref{def:peps_torus_g_closure}.
\end{definition}

\begin{lemma}[Uniform closures belong to every cut range]%
    \label{thm:peps_four_cut_common_uniform_membership}
    \lean{TNLean.PEPS.torusBondNetwork_closureAt_eq_sitewiseTorusGClosure,
        TNLean.PEPS.torusCutMap_closureBoundary,
        TNLean.PEPS.sitewiseTorusGClosure_mem_torusCutSpace,
        TNLean.PEPS.sitewiseCommutingClosureSpan_le_fourTorusCutSpace}
    \leanok
    \uses{def:peps_four_cut_common_closure,def:peps_four_cut_common_product_boundary}
    Suppose $U^d_v=U$ on every bond and $\mathcal P(A_v)\rho(k)=\mathcal P(A_v)$
    for all $v\in X$ and $k\in G$. If $gh=hg$, then for every cut $(c,r)$,
    \begin{align}
        \mathcal N_A(H^c(h),V^r(g))
         & =\mathcal C^A_{c,r}M^{(U(h),U(g))}_{c,r}
        =\mathcal M_A(g,h)\in\mathcal S^A_{c,r},
        \label{eq:peps_common_uniform_membership}
    \end{align}
    and on $X=(\Z/2\Z)^2$, $\mathcal Z_A\subseteq\mathcal S^A$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_torus_commuting_closure_seams,thm:peps_four_cut_common_boundary_span}
    Setting $B(t,r,b,l;(v,s))=A_v(t,r,b,l;s)$ preserves $\rho(k)$-invariance at
    each $(v,s)$. Applying Lemma~\ref{thm:peps_torus_commuting_closure_seams}
    to $B$ at $v\mapsto(v,\sigma_v)$ together with
    (\ref{eq:peps_common_boundary_network}) of
    Lemma~\ref{thm:peps_four_cut_common_boundary_span} gives
    (\ref{eq:peps_common_uniform_membership}), and taking spans yields
    $\mathcal Z_A\subseteq\mathcal S^A$.
\end{proof}

\begin{definition}[Cyclic seam changes of basis]%
    \label{def:peps_four_cut_common_seam_gauge}
    \lean{TNLean.PEPS.torusSeamGauge}
    \leanok
    For $x,c\in\Z/m\Z$ with $m>0$ and representatives
    $\bar x,\bar c\in\{0,\ldots,m-1\}$, put $s_{k,c}(x)=k^{-1}$ if
    $\bar x<\bar c$ and $s_{k,c}(x)=1$ otherwise.
\end{definition}

\begin{lemma}[Cyclic seam identities]%
    \label{thm:peps_four_cut_common_seam_gauge}
    \lean{TNLean.PEPS.torusSeamGauge_horizontal,
        TNLean.PEPS.torusSeamGauge_vertical,
        TNLean.PEPS.torusSeamGauge_commute}
    \leanok
    \uses{def:peps_four_cut_common_seam_gauge}
    Write $K^c_k(x)=k$ if $x+1=c$ and $K^c_k(x)=1$ otherwise. For every $k\in G$,
    \begin{align}
        K^c_k(x) & =s_{k,c}(x+1)K^0_k(x)s_{k,c}(x)^{-1},
        \label{eq:peps_common_seam_horizontal}                     \\
        K^c_k(x) & =s_{k^{-1},c}(x)K^0_k(x)s_{k^{-1},c}(x+1)^{-1}.
        \label{eq:peps_common_seam_vertical}
    \end{align}
    If $kl=lk$, then $s_{k,c}(x)l=ls_{k,c}(x)$.
\end{lemma}
\begin{proof}\leanok
    Checking $x+1=0$, $x+1=c$, and the intervals between them gives
    (\ref{eq:peps_common_seam_horizontal}); inverting with $k^{-1}$ in place of
    $k$ gives (\ref{eq:peps_common_seam_vertical}), and $s_{k,c}(x)\in\{1,k^{-1}\}$
    commutes with $l$ when $kl=lk$.
\end{proof}

\begin{lemma}[Matched commuting closures belong to every cut range]%
    \label{thm:peps_four_cut_common_matched_membership}
    \lean{TNLean.PEPS.torusBondNetwork_matchedClosureAt_eq_matchedTorusGClosure,
        TNLean.PEPS.torusCutMap_matchedClosureBoundary,
        TNLean.PEPS.matchedTorusGClosure_mem_torusCutSpace,
        TNLean.PEPS.matchedCommutingClosureSpan_le_fourTorusCutSpace}
    \leanok
    \uses{def:peps_four_cut_common_closure,def:peps_four_cut_common_product_boundary}
    Suppose $\mathcal P(A_v)\rho_v(k)=\mathcal P(A_v)$ for all $v\in X$ and
    $k\in G$. In the first inclusion in the proof of
    \cite[Theorem~5.5]{Schuch2010PEPS} with a common coordinate alphabet, for
    $gh=hg$ and every $(c,r)$,
    \begin{align}
        \mathcal N_A(H^c(h),V^r(g))
         & =\mathcal C^A_{c,r}M^{(U^{\mathrm h}(h),U^{\mathrm v}(g))}_{c,r}
        =\mathcal M_A(g,h)\in\mathcal S^A_{c,r},
        \label{eq:peps_common_matched_membership}
    \end{align}
    and on $X=(\Z/2\Z)^2$, $\mathcal Z_A\subseteq\mathcal S^A$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_four_cut_common_seam_gauge,
        thm:peps_four_cut_common_vertex_gauge,thm:peps_four_cut_common_boundary_span}
    Applying Lemma~\ref{thm:peps_four_cut_common_vertex_gauge} with
    $q_v=s_{h,c}(v_x)$ and then $q_v=s_{g^{-1},r}(v_y)$ uses
    (\ref{eq:peps_common_seam_horizontal})--(\ref{eq:peps_common_seam_vertical})
    of Lemma~\ref{thm:peps_four_cut_common_seam_gauge} to shift both seams to
    $(0,0)$ while commuting with the transverse label, and
    (\ref{eq:peps_common_boundary_network}) of
    Lemma~\ref{thm:peps_four_cut_common_boundary_span} gives
    (\ref{eq:peps_common_matched_membership}) and $\mathcal Z_A\subseteq\mathcal S^A$.
\end{proof}
