\section{Fixed physical ports and bounded local registers}

The intersite gates of~\cite[Supplement, pp.~7--8]{Piroli2021LOCC}
act on fixed-dimensional physical qudits. Local ancillas are separate:
free onsite operations may mix the physical qudit with its local ancillas,
but a counted intersite gate does not act on an arbitrarily enlarged pair
of sites. This section derives bounded-register channel simulations and
unitary realizations with fresh local environments in that convention.
It does not identify the unrestricted local-channel model with the source's phase relation; see
\cite{gap:psc21_physical_port_simulation_scope}.

The register encodings, the routing of registers through the ports, and the
composition of local environments behind the statements below are carried
out in the companion quantum-information
volume~\cite{QICLean2026Blueprint}; only the resulting statements are
recorded here.

\begin{definition}[Physical-port channel protocols]
    \label{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.Geometry.portRing,
        QuantumCircuit.IsProtocol,
        QuantumCircuit.IsProtocol.comp,
        QuantumCircuit.IsProtocol.mono,
        QuantumCircuit.IsProtocol.isKrausCPTP}
    \leanok
    Each site $i\in\Z_N$ carries $W\ge1$ wires $s^{-1}(i)=\{(i,0),\dots,(i,W-1)\}$, each carrying $\C^d$, and the wire $p_i=(i,0)$ is its physical port. Free (depth-$0$) operations are normalized Kraus channels $\Phi(X)=\sum_j K_j X K_j^\dagger$ with $K_j$ supported inside $s^{-1}(i)$ and $\sum_j K_j^\dagger K_j=I$; these include the unitaries supported inside $s^{-1}(i)$ and the wire permutations preserving every site. A counted (depth-$1$) operation is a normalized Kraus channel supported on a pair $\{p_i,p_{i+1}\}$ of neighbouring ports, and operations on disjoint wire sets form one layer. Composition adds physical depths $\operatorname{depth}(\Phi_2\circ\Phi_1)\le\operatorname{depth}(\Phi_2)+\operatorname{depth}(\Phi_1)$, depth bounds are upwards closed, and every protocol is CPTP.
\end{definition}

\begin{lemma}[Uniform bounded-register matching simulation]
    \label{thm:qc_bounded_physical_port_simulation}
    \uses{def:qc_physical_port_protocol, def:qc_dimension_bounded_protocol}
    \lean{QuantumCircuit.Layer.exists_portRing,
        QuantumCircuit.IsBoundedProtocol.exists_portRing}
    \leanok
    Fix $d\ge1$ and $k\ge0$, and consider a ring of $N\ge2$ sites with $2k+1$ qudits of dimension
    $d$ at every site: the physical port, $k$ data qudits and $k$ scratch qudits. Let
    $\mathcal E_{\rm data}$ place a register of $k$ digits at every site into its data qudits.
    Every layer $L$ of CPTP pair channels on a matching of the ring of registers of $k$ digits has
    a physical-port protocol $\Gamma_L$ of depth at most $2k$ with
    $\Gamma_L\circ\mathcal E_{\rm data}=\mathcal E_{\rm data}\circ L$. Consequently, if $\Psi$ is
    a native protocol of depth $T$ whose local dimensions along the protocol are at most
    $B\le d^k$, and its input and output bases are encoded into registers by $\mathcal E_{\rm in}$
    and $\mathcal E_{\rm out}$, there is a physical-port protocol $\Gamma$ of depth at most $2kT$
    with
    $\Gamma\circ\mathcal E_{\rm data}\circ\mathcal E_{\rm in}
        =\mathcal E_{\rm data}\circ\mathcal E_{\rm out}\circ\Psi$;
    for $d\ge2$ one may take $k=\lceil\log_d B\rceil$.
\end{lemma}
\begin{proof}
    \uses{thm:qc_fixed_register_word}
    \leanok
    Move the register of $i$, one digit per port layer, into the scratch qudits of $i+1$ for
    all $i\in M$ simultaneously, at depth $k$; the pair channels then act inside single sites at
    depth $0$, and moving the registers back costs another $k$, giving depth at most $2k$ for one
    layer. For the protocol, encode the bases of every intermediate step into registers by
    Lemma~\ref{thm:qc_fixed_register_word}, so that each encoded layer acts on the code as
    the original layer; the $T$ layers then compose to depth at most $2kT$.
\end{proof}

\subsection{Compiling bounded native protocols}

The intermediate local dimensions of a native channel protocol need a common
bound before a family-level physical-depth comparison is available. The next
construction records that bound explicitly, compiles into one fixed register
alphabet, and then realizes those operations on the fixed physical ports.

\begin{definition}[Dimension-bounded native protocols]
    \label{def:qc_dimension_bounded_protocol}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsBoundedProtocol,
        QuantumCircuit.IsBoundedProtocol.isProtocol,
        QuantumCircuit.IsBoundedProtocol.dimension_bounds,
        QuantumCircuit.IsBoundedProtocol.mono_bound,
        QuantumCircuit.IsBoundedProtocol.comp,
        KrausFamily.IsTP.nonempty_target,
        QuantumCircuit.IsProtocol.exists_dimension_bound}
    \leanok
    A native local-channel protocol $\Psi$ is \emph{$B$-dimension-bounded} of depth $T$ if it is generated by onsite channels (depth $0$) and nearest-neighbor pair-channel layers (depth $1$) whose input, output, and intermediate local dimensions all satisfy $1\le d_t\le B$. The predicate is monotonic in $B$, additive in $T$ under composition, and every finite native protocol with $d_{\rm in}\ge1$ admits some finite bound $B$ (not necessarily uniform in $N$).
\end{definition}


\begin{lemma}[Encoding a native channel layer]
    \label{thm:qc_encode_native_channel_layer}
    \uses{def:qc_local_channel_circuit}
    \lean{QuantumCircuit.encodeCircuit,
        QuantumCircuit.map_encodeCircuit,
        QuantumCircuit.Layer.exists_encoded}
    \leanok
    For injections $e_s$ and $f_s$ of local bases at every site $s$ into larger local bases, write
    $E_e=\bigotimes_s E_{e_s}$ with $E_{e_s}=\sum_a\ket{e_s(a)}\bra{a}$, and
    $\mathcal E_e(X)=E_eXE_e^\dagger$; it is a depth-$0$ onsite channel. Let $\Lambda$ be a layer
    of trace-preserving gates and onsite channels whose input local bases are nonempty, and let
    $e$ and $f$ inject its input and output local bases. Then there is a layer $\widehat\Lambda$
    of trace-preserving gates and onsite channels on the same cells as $\Lambda$ such that
    $\widehat\Lambda\circ\mathcal E_e=\mathcal E_f\circ\Lambda$.
\end{lemma}
\begin{proof}
    \leanok
    Replace each gate $\Phi$ of $\Lambda$, on a cell $C$, by
    $\widehat\Phi=\mathcal E_{f_C}\circ\Phi\circ D_C$,
    where the completed decoder $D_C$ undoes $\mathcal E_{e_C}$ on its range and sends the
    complement to a fixed basis vector of the output. The input bases are nonempty, so by trace
    preservation the output bases are nonempty too and supply these vectors. Each $\widehat\Phi$
    is trace preserving and satisfies $\widehat\Phi\circ\mathcal E_{e_C}=\mathcal E_{f_C}\circ\Phi$,
    and the tensor product over the cells and sites gives
    $\widehat\Lambda\circ\mathcal E_e=\mathcal E_f\circ\Lambda$.
\end{proof}

\begin{lemma}[A fixed-register protocol for a bounded native protocol]
    \label{thm:qc_fixed_register_word}
    \uses{def:qc_dimension_bounded_protocol, thm:qc_encode_native_channel_layer}
    \lean{QuantumCircuit.Circuit.exists_encoded_of_isDimBounded,
        QuantumCircuit.IsBoundedProtocol.exists_encoded}
    \leanok
    Let $B\le q$, and let $\Psi$ be realized by a $B$-dimension-bounded protocol of depth at most
    $T$. For injections $e_s$ and $f_s$ of its input and output local bases at every site $s$ into
    $\{0,\dots,q-1\}$, there is a protocol $\widehat\Psi$ of depth at most $T$ with local
    dimension $q$ at every site such that
    $\widehat\Psi\circ\mathcal E_e=\mathcal E_f\circ\Psi$. The realizing circuit has the same
    depth as that of $\Psi$, consists of trace-preserving operations, and is unbranched when that
    of $\Psi$ is.
\end{lemma}
\begin{proof}
    \leanok
    By induction along the circuit. Every intermediate local basis has at most $B\le q$
    elements, so it injects into $\{0,\dots,q-1\}$, and each layer is encoded between consecutive
    injections by Lemma~\ref{thm:qc_encode_native_channel_layer}. At the end the output is decoded
    from the last injection and encoded along $f$, at depth $0$.
\end{proof}

\subsection{Unitary circuits with one final environment discard}

The source treats onsite local unitaries and ancillas as free
resources~\cite[main text, p.~1]{Piroli2021LOCC}. The following bounds count
only intersite layers through the fixed physical ports. Every extra
environment is a finite local system attached to a wire, is prepared in a
basis state at the start, and is discarded only after the circuit has
finished.


\begin{lemma}[Same-depth unitary realization with one final discard]
    \label{thm:qc_physical_port_unitary_dilation}
    \uses{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.IsUnitaryDilation,
        QuantumCircuit.Layer.isUnitaryDilation,
        QuantumCircuit.IsUnitaryDilation.comp,
        QuantumCircuit.Circuit.isUnitaryDilation,
        QuantumCircuit.IsProtocol.isUnitaryDilation}
    \leanok
    Fix $d\ge1$. Every physical-port protocol $\Phi$ of depth at most $T$ admits finite local environments $A$, one attached to each wire, an initializer $\mathcal I_A$ preparing them in a product basis vector, the partial trace $\mathcal D_A$ over the output environments, and a unitary circuit $U$ on the wires enlarged by their environments such that
    \begin{align}
        \mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A & =\Phi,
                                                              & \operatorname{depth}_{\mathrm{intersite}}(U) & \leq T.
        \label{eq:qc_physical_port_dilation}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    By induction on the layers of $\Phi$: a unitary operation keeps a one-element environment, every other normalized Kraus channel is replaced by a unitary dilation~\cite[Theorem~2.5]{Wolf2012Quantum} whose environment is attached to one wire of its support, and two dilations compose by preparing the environments of the later one at the start and carrying each environment unchanged through the other circuit.
\end{proof}


\begin{lemma}[Bounded unitary simulation through the physical ports]
    \label{thm:qc_bounded_unitary_port_io}
    \uses{def:qc_dimension_bounded_protocol, def:qc_physical_port_protocol,
        thm:qc_physical_port_unitary_dilation}
    \lean{QuantumCircuit.portDataEmbedding,
        QuantumCircuit.Layer.exists_portRing,
        QuantumCircuit.IsBoundedProtocol.exists_portRing}
    \leanok
    Let $d\ge1$, $N\ge2$, $k\ge0$ and $B\le d^k$, and let $\Psi$ be realized by a
    $B$-dimension-bounded protocol of depth at most $T$ on the ring of $N$ sites. Use $2k+1$ qudits
    of dimension $d$ per site: the port, $k$ data qudits and $k$ scratch qudits. Let
    $J^{\rm in}$ and $J^{\rm out}$ store the input and output local bases of each site, through
    injections into registers of $k$ digits, in the $k$ data qudits of that site, with the port
    and scratch qudits in $\ket{0}$. Let $D$ be a set of wires containing a wire of every site.
    Then there are finite local environments $A$, one-element at every wire outside $D$, prepared
    in a basis vector by $\mathcal I_A$ and discarded by the partial trace $\mathcal D_A$, and a
    unitary circuit $U$ such that
    \begin{align}
        \mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A\circ J^{\rm in}
         & =J^{\rm out}\circ\Psi,
         & \operatorname{depth}_{\mathrm{intersite}}(U) & \leq2kT,
        \label{eq:qc_bounded_unitary_port_io}
    \end{align}
    where every wire outside $D$ has dimension at most $d$ throughout $U$. For $k\ge1$ and $D$ the
    non-port wires, every gate of $U$ between neighbouring sites is therefore a unitary on the two
    port wires alone, each of dimension at most $d$. For $d\ge2$ one may take
    $k=\lceil\log_d B\rceil$. Uniform family overhead requires $B$ and $d$ to be fixed
    independently of $N$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_fixed_register_word, thm:qc_physical_port_unitary_dilation}
    Encode $\Psi$ into registers of $k$ digits by Lemma~\ref{thm:qc_fixed_register_word}. Each layer of the encoded protocol is executed on the ports by moving the register of the left site of every gate, one digit per port layer, into the scratch wires of its right neighbour, acting inside that site, and moving the register back, at cost $2k$ port layers. The routing layers are unitary and every other operation acts on all the wires of one site, which meet $D$, so each part has a unitary dilation as in Lemma~\ref{thm:qc_physical_port_unitary_dilation} with environments at wires of $D$, and the dilations compose. The source's QCcc block/control and pure-output system--ancilla conditions remain separate (see~\cite{gap:psc21_physical_port_simulation_scope}).
\end{proof}

\begin{lemma}[Pure output and environment decoupling]
    \label{thm:qc_physical_port_pure_output}
    \uses{thm:qc_physical_port_unitary_dilation}
    \lean{QuantumCircuit.IsProtocol.isUnitaryDilation,
        QuantumCircuit.Circuit.exists_unit_product_approx}
    \leanok
    Let $\Phi$ be a protocol of depth at most $T$ on nonempty local bases, for
    any assignment of costs to the gates, such as the physical-port one. Then
    $\Phi=\mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A$ for a
    unitary circuit $U$ of depth at most $T$, where $\mathcal I_A$ prepares
    local environments in basis states $\ket{0^A}$ and $\mathcal D_A$ discards
    the output environments. If
    $\|\Phi(\ketbra{\psi}{\psi})-\ketbra{\phi}{\phi}\|_1\le\varepsilon$
    for unit vectors $\psi,\phi$, then there is a unit environment vector
    $\eta$ with
    \[
        \|U(\psi\otimes\ket{0^A})-\phi\otimes\eta\|_2\le\sqrt{2\varepsilon}.
    \]
    In particular, if $\Phi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$,
    then $U(\psi\otimes\ket{0^A})=\phi\otimes\eta$ with
    $\langle\eta,\eta\rangle=1$. The environment vector $\eta$ need not
    factor over spatial sites or equal its initial zero state.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_physical_port_unitary_dilation}
    The dilation is built as in Lemma~\ref{thm:qc_physical_port_unitary_dilation}:
    each layer is dilated with fresh local environments at no extra depth, and
    compositions carry the environments of each factor through the other.
    Let $C_{s,e}=\bigl(U(\psi\otimes\ket{0^A})\bigr)_{s,e}$ be the
    coefficient matrix of the joint output, a unit vector since $U$ is
    unitary. Its first marginal is
    \begin{align}
        CC^\dagger & =\tr_A\bigl(U(\ketbra{\psi}{\psi}\otimes Z_A)U^\dagger\bigr)
        =\Phi(\ketbra{\psi}{\psi}).\notag
    \end{align}
    Project $C$ onto $\phi$: the row $\phi^\dagger C$ has squared norm
    $\langle\phi,CC^\dagger\phi\rangle\ge1-\varepsilon$, and normalizing
    it gives $\eta$ with $\|C-\phi\eta^{\mathsf T}\|_2^2\le2\varepsilon$.
    If the row is zero, then $\varepsilon\ge1$ and any unit vector $\eta$
    suffices. For $\varepsilon=0$ the bound is the equality
    $C=\phi\eta^{\mathsf T}$. No factorization assumption is used.
\end{proof}
