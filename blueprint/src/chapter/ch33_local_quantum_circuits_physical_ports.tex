\section{Fixed physical ports and bounded local registers}

The intersite gates of~\cite[Supplement, pp.~7--8]{Piroli2021LOCC}
act on fixed-dimensional physical qudits. Local ancillas are separate:
free onsite operations may mix the physical qudit with its local ancillas,
but a counted intersite gate does not act on an arbitrarily enlarged pair
of sites. This section derives bounded-register channel simulations and
unitary realizations with fresh local environments in that convention.
It does not identify the unrestricted local-channel model with the source's phase relation; see
\cite{gap:psc21_physical_port_simulation_scope}.

\begin{definition}[Physical-port channel protocols]
    \label{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.PhysicalPortLayout,
        QuantumCircuit.IsPhysicalPortUnitary,
        QuantumCircuit.IsPhysicalPortUnitary.mem_unitary,
        QuantumCircuit.IsPhysicalPortUnitary.conjTranspose,
        QuantumCircuit.IsPhysicalPortUnitary.one,
        QuantumCircuit.IsPhysicalPortUnitary.onsite_swap,
        QuantumCircuit.IsPhysicalPortProtocol,
        QuantumCircuit.IsPhysicalPortProtocol.isKrausCPTP,
        QuantumCircuit.IsPhysicalPortProtocol.one}
    \leanok
    Each wire in $W$ carries $\C^d$. A layout assigns a spatial owner $s(w)\in\Z_N$ to each wire $w\in W$ and selects a physical port $p_i\in s^{-1}(i)$ at each site $i\in\Z_N$. Free (depth-$0$) operations are unitaries supported inside $s^{-1}(i)$, wire permutations $\pi$ with $s\circ\pi=s$, and normalized Kraus channels $\Phi(X)=\sum_j K_j X K_j^\dagger$ with $K_j$ supported inside $s^{-1}(i)$ and $\sum_j K_j^\dagger K_j=I$. A counted (depth-$1$) layer places a nearest-neighbor unitary layer $L$ on $\{p_i\}_{i\in\Z_N}$ as $L\otimes I_{W\setminus\{p_i\}}$. Composition adds physical depths $\operatorname{depth}(\Phi_2\circ\Phi_1)\le\operatorname{depth}(\Phi_2)+\operatorname{depth}(\Phi_1)$, depth bounds are upwards closed, $\operatorname{depth}(U^\dagger)=\operatorname{depth}(U)$, and every protocol is CPTP.
\end{definition}

\begin{lemma}[Register codes and completed decoding]
    \label{lem:qc_register_code_completion}
    \lean{QuantumCircuit.registerEncoding,
        QuantumCircuit.registerEncoding_isIsometry,
        QuantumCircuit.registerEncoding_isKrausCPTP,
        QuantumCircuit.registerDecoding,
        QuantumCircuit.registerDecoding_isKrausCPTP,
        QuantumCircuit.registerDecoding_encoding,
        QuantumCircuit.registerDecoding_comp_encoding,
        QuantumCircuit.registerDecoding_encoding_reference,
        QuantumCircuit.registerEncodedMap,
        QuantumCircuit.registerEncodedMap_isKrausCPTP,
        QuantumCircuit.registerEncodedMap_encoding,
        QuantumCircuit.le_registerDimension,
        QuantumCircuit.registerBasisEmbedding,
        QuantumCircuit.pairRegisterBasisEmbedding,
        QuantumCircuit.pairRegisterBasisEmbedding_apply,
        QuantumCircuit.registerEncoding_apply,
        QuantumCircuit.registerEncoding_pair}
    \leanok
    An injection $e\colon\{0,\dots,m-1\}\hookrightarrow\{0,\dots,d^k-1\}$ defines an isometry $E=\sum_a\ket{e(a)}\bra{a}$ ($E^\dagger E=I_m$) and encoding $\mathcal E(X)=EXE^\dagger$. For a density matrix $\rho$, the completed decoder
    \begin{align}
        D(Y) & =E^\dagger YE+\operatorname{tr}[(I-EE^\dagger)Y(I-EE^\dagger)]\rho\notag
    \end{align}
    is CPTP and satisfies $D\circ\mathcal E=\operatorname{id}$ and $(D\otimes\operatorname{id}_R)\circ(\mathcal E\otimes\operatorname{id}_R)=\operatorname{id}$. On a pair of registers, $E_{e_1\times e_2}=E_{e_1}\otimes E_{e_2}$. If $m\le B$ and $d\ge2$, $k=\lceil\log_d B\rceil$ satisfies $m\le d^k$. A CPTP map $\Phi$ extends to $\widehat\Phi=\mathcal E_{\rm out}\circ\Phi\circ D_{\rm in}$, which is CPTP and satisfies $\widehat\Phi\circ\mathcal E_{\rm in}=\mathcal E_{\rm out}\circ\Phi$.
\end{lemma}
\begin{proof}
    \leanok
    Since $E^\dagger E=I_m$, $(I-EE^\dagger)EXE^\dagger=0$, so $D(EXE^\dagger)=E^\dagger EXE^\dagger E=X$, which extends under $-\otimes\operatorname{id}_R$. The pair basis map $(a,b)\mapsto(e_1(a),e_2(b))$ factors as $E_{e_1}\otimes E_{e_2}$, and $m\le B\le d^{\lceil\log_d B\rceil}$.
\end{proof}

        QuantumCircuit.PortMatching.layer_op,
        QuantumCircuit.PortMatching.layer_op_one_site}
    \leanok
    A matching $M$ of ring bonds defines an involution $\sigma_M\in\mathfrak S_N$ exchanging $i\leftrightarrow i+1$ for each $\{i,i+1\}\in M$ and fixing unmatched sites. The layer $L_M$ of SWAP gates on $M$ has operator $P(\sigma_M)$ where $P(\sigma)\ket{x}=\ket{x\circ\sigma^{-1}}$ satisfies $P(1)=I$, $P(\sigma\tau)=P(\sigma)P(\tau)$, and $P(\sigma)^\dagger=P(\sigma^{-1})$. \end{lemma}
\begin{proof}
    \leanok
    Disjoint bond transpositions commute and multiply to $\sigma_M$, and the homomorphism $P$ identifies the product of placed two-site SWAP matrices with $P(\sigma_M)$.
\end{proof}

\begin{lemma}[Selected-memory routing at unchanged physical depth]
    \label{thm:qc_port_register_routing}
    \uses{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.PhysicalPortLayout.selectionPermutation,
        QuantumCircuit.PhysicalPortLayout.selectionPermutation_site,
        QuantumCircuit.PhysicalPortLayout.selectionPermutation_port,
        QuantumCircuit.PhysicalPortLayout.selectionPermutation_symm,
        QuantumCircuit.IsPhysicalPortUnitary.selected_layer,
        QuantumCircuit.embedOp_permOp_viaEmbedding,
        QuantumCircuit.IsPhysicalPortUnitary.selected_permutation_layer,

        QuantumCircuit.registerPermutation_apply,
        QuantumCircuit.registerPermutation_apply_of_notMem,
        QuantumCircuit.registerPermutation_involutive,
        QuantumCircuit.IsPhysicalPortUnitary.register_matching}
    \leanok
    Given $w_i\in s^{-1}(i)$ for each $i\in\Z_N$, the onsite swap $\pi_w=\prod_i(p_i\,w_i)$ is free, and $P(\pi_w)(L\otimes I)P(\pi_w)^\dagger$ places a physical layer $L$ on $\{w_i\}_{i\in\Z_N}$ at physical depth $1$. Repeating this for $k$ sitewise-disjoint wire selections $w^{(1)},\dots,w^{(k)}$ and a matching $M$ implements $\prod_{r=1}^k P(\sigma_{M,r})$ at physical depth at most $k$, acting as $\sigma_M$ on each digit $r$ and as the identity on all unselected wires.
\end{lemma}
\begin{proof}
    \uses{lem:qc_port_matching_swaps, lem:qc_placed_operators}
    \leanok
    Since $s\circ\pi_w=s$, $P(\pi_w)$ has depth $0$, and conjugation by $P(\pi_w)$ reindexes port placement to $\{w_i\}$ by Lemma~\ref{lem:qc_placed_operators}. By Lemma~\ref{lem:qc_port_matching_swaps}, each digit permutation $P(\sigma_{M,r})$ has depth $1$, and the $k$ factors have disjoint supports and multiply to depth at most $k$.
\end{proof}

\begin{lemma}[One shared routing for a channel family]
    \label{thm:qc_port_shared_channel_routing}
    \uses{def:qc_physical_port_protocol, lem:qc_placed_operators}
    \lean{QuantumCircuit.sum_conjTranspose_embedOp_mul,
        QuantumCircuit.routingMatrixEquiv,
        QuantumCircuit.routingMatrixEquiv_apply,
        QuantumCircuit.routingMatrixEquiv_symm_apply,
        QuantumCircuit.routingChannelConj,
        QuantumCircuit.routingChannelConj_apply,
        QuantumCircuit.routingChannelConj_embeddedKraus,
        QuantumCircuit.IsPhysicalPortProtocol.of_routed_map,
        QuantumCircuit.IsPhysicalPortProtocol.of_register_routing,
        QuantumCircuit.IsPhysicalPortProtocol.of_routed_family}
    \leanok
    If a wire permutation $\pi$ with $\operatorname{depth}(P(\pi))\le T$ maps the support of each channel $\Phi_b$ ($b\in M$) in a commuting family into a single spatial site $s^{-1}(i_b)$, then
    \begin{align}
        \prod_{b\in M}\Phi_b & =\operatorname{Ad}_{P(\pi)^\dagger}\circ\Bigl(\prod_{b\in M}\operatorname{Ad}_{P(\pi)}\circ\Phi_b\circ\operatorname{Ad}_{P(\pi)^\dagger}\Bigr)\circ\operatorname{Ad}_{P(\pi)}\notag
    \end{align}
    has physical depth at most $2T$, independently of $|M|$.
\end{lemma}
\begin{proof}
    \leanok
    Conjugation by $P(\pi)$ is an algebra automorphism, so it distributes over $\prod_{b\in M}\Phi_b$ and reindexes each placed Kraus family $\{K_{b,j}\}$ to $\{P(\pi)K_{b,j}P(\pi)^\dagger\}$ supported inside $s^{-1}(i_b)$ with $\sum_j(P(\pi)K_{b,j}P(\pi)^\dagger)^\dagger(P(\pi)K_{b,j}P(\pi)^\dagger)=I$. Each conjugated factor has depth $0$, leaving only $P(\pi)$ and $P(\pi)^\dagger$ of total depth $2T$.
\end{proof}

\begin{lemma}[Uniform bounded-register matching simulation]
    \label{thm:qc_bounded_physical_port_simulation}
    \uses{def:qc_physical_port_protocol, lem:qc_register_code_completion, lem:qc_port_data_scratch}
    \lean{QuantumCircuit.PortRegisters.pairChannel,
        QuantumCircuit.PortRegisters.pairChannel_commute,

        QuantumCircuit.PortRegisters.matchingChannel_isPhysicalPortProtocol,
        QuantumCircuit.boundedPairRegisterCode,
        QuantumCircuit.exists_bounded_matching_simulation,
        QuantumCircuit.IsPhysicalPortProtocol.list_prod_uniform}
    \leanok
    Fix $d\ge2$, $B\ge1$, $m\le B$, and $k=\lceil\log_d B\rceil$. For every ring of $N\ge2$ sites and matching $M$ of CPTP pair channels $\Phi_i$ ($i\in M$), the encoded pair channels $\widehat\Phi_i=\mathcal E_{\rm out}\circ\Phi_i\circ D_{\rm in}$ on $(d_i,d_{i+1})$ satisfy $\widehat\Phi_i\circ\mathcal E_{\rm in}=\mathcal E_{\rm out}\circ\Phi_i$, commute for distinct $i\in M$, and their placed product $\prod_{i\in M}\operatorname{Lift}_{d_i,d_{i+1}}(\widehat\Phi_i)$ has a physical-port protocol of depth at most $2k$. A composition of $L$ such layers has depth at most $2kL$ ($0$ when $B=1$).
\end{lemma}
\begin{proof}
    \uses{lem:qc_register_code_completion, lem:qc_port_data_scratch, thm:qc_port_shared_channel_routing}
    \leanok
    By Lemma~\ref{lem:qc_register_code_completion}, each $\widehat\Phi_i$ is CPTP with a normalized Kraus representation and intertwines with $\Phi_i$. By Lemma~\ref{lem:qc_port_data_scratch}, the depth-$k$ routing $R_M$ colocates $(d_i,d_{i+1})$ inside $s^{-1}(i+1)$ for all $i\in M$ simultaneously, so Lemma~\ref{thm:qc_port_shared_channel_routing} gives depth at most $2k$, and $L$-fold composition gives $2kL$.
\end{proof}

    \leanok
    For an injective wire map $e\colon S\hookrightarrow W$, the configuration split $x\mapsto(x\circ e,x|_{W\setminus e(S)})$ induces a $*$-algebra isomorphism $\Theta_e$ with $\Theta_e(\operatorname{embed}_e(A))=A\otimes I_{W\setminus e(S)}$, invariant under reindexing $W\setminus e(S)$ (including across disjoint sums $W=W_1\sqcup W_2$). Consequently, the lifted channel $\operatorname{Lift}_e(\Phi)=\Theta_e^{-1}\circ(\Phi\otimes\operatorname{id}_{W\setminus e(S)})\circ\Theta_e$ satisfies $\operatorname{Lift}_e(\operatorname{id})=\operatorname{id}$, $\operatorname{Lift}_e(\Psi\circ\Phi)=\operatorname{Lift}_e(\Psi)\circ\operatorname{Lift}_e(\Phi)$, and
    \begin{align}
        \operatorname{Lift}_e\Bigl(\sum_j K_j\cdot K_j'\Bigr)(X) & =\sum_j\operatorname{embed}_e(K_j)\,X\,\operatorname{embed}_e(K_j'),\notag
    \end{align}
    which is independent of the Kraus representation of $\Phi$.
\end{lemma}
\begin{proof}
    \uses{lem:qc_placed_operators}
    \leanok
    By Lemma~\ref{lem:qc_placed_operators}, $\operatorname{embed}_e(A)_{x,y}=A_{x\circ e,y\circ e}\delta_{x|_{e(S)^c},y|_{e(S)^c}}=(A\otimes I)_{(x\circ e,x|_{e(S)^c}),(y\circ e,y|_{e(S)^c})}$. Linearity and multiplicativity of $\Theta_e$ extend this to $\sum_j\operatorname{embed}_e(K_j)X\operatorname{embed}_e(K_j')$.
\end{proof}

\begin{lemma}[Intertwining of placed encoded channels]
    \label{thm:qc_encoded_channel_placement}
    \uses{lem:qc_product_register_encoding, lem:qc_register_channel_lift}

    \end{align}
\end{lemma}

\end{proof}

\subsection{Compiling bounded native protocols}

The intermediate local dimensions of a native channel protocol need a common
bound before a family-level physical-depth comparison is available. The next
construction records that bound explicitly, compiles into one fixed register
alphabet, and then realizes those operations on the fixed physical ports.

\begin{definition}[Dimension-bounded native protocols]
    \label{def:qc_dimension_bounded_protocol}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsDimensionBoundedLocalChannelProtocol,
        QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.toIsLocalChannelProtocol,
        QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.dimension_bounds,
        QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.mono_bound,
        QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.comp,
        QuantumCircuit.OnsiteChannel.output_dimension_pos,
        QuantumCircuit.IsLocalChannelProtocol.exists_dimension_bound}
    \leanok
    A native local-channel protocol $\Psi$ is \emph{$B$-dimension-bounded} of depth $T$ if it is generated by onsite channels (depth $0$) and nearest-neighbor pair-channel layers (depth $1$) whose input, output, and intermediate local dimensions all satisfy $1\le d_t\le B$. The predicate is monotonic in $B$, additive in $T$ under composition, and every finite native protocol with $d_{\rm in}\ge1$ admits some finite bound $B$ (not necessarily uniform in $N$).
\end{definition}

    \begin{align}

\end{proof}

\subsection{Unitary circuits with one final environment discard}

The finite reduced-channel protocols have unitary realizations with the
same physical-port intersite depth. Every extra environment wire has a
specified spatial owner, is initialized in the zero basis state at the
start, and is discarded only after the circuit has finished.

\begin{definition}[Port-preserving wire embeddings]
    \label{def:qc_physical_port_embedding}
    \uses{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.PhysicalPortEmbedding,
        QuantumCircuit.PhysicalPortLayout.append,
        QuantumCircuit.PhysicalPortLayout.appendEmbedding}
    \leanok
    A port-preserving embedding between layouts on $W$ and $W'$ is an injection $\iota\colon W\hookrightarrow W'$ satisfying $s'\circ\iota=s$ and $\iota(p_i)=p'_i$ for all $i\in\Z_N$. Appending $A$ environment wires with assigned spatial owners $s_A\colon A\to\Z_N$ defines the layout $W\sqcup A$ and canonical embedding $\iota\colon W\hookrightarrow W\sqcup A$.
\end{definition}

\begin{lemma}[Preservation of physical depth under wire embeddings]
    \label{thm:qc_physical_port_embedding}
    \uses{def:qc_physical_port_embedding}
    \lean{QuantumCircuit.PhysicalPortEmbedding.unitary}
    \leanok

\end{lemma}
\begin{proof}\leanok
    \uses{def:qc_physical_port_protocol, lem:qc_placed_operators}

\end{proof}

    Restricting $\widetilde V$ to input environment $0^A$ and output environment $a$ gives $\operatorname{embed}_e(K_a)$, so $\mathcal D_A(\widetilde V(X\otimes Z_A)\widetilde V^\dagger)=\sum_a\operatorname{embed}_e(K_a)X\operatorname{embed}_e(K_a)^\dagger=\operatorname{Lift}_e(\sum_a K_a\cdot K_a^\dagger)(X)$ by Lemma~\ref{lem:qc_register_channel_lift}.
\end{proof}

\begin{lemma}[Fresh onsite unitary dilations]
    \label{thm:qc_fresh_onsite_unitary_dilation}
    \uses{def:qc_appended_environment, def:qc_selected_environment}
    \lean{QuantumCircuit.exists_qudit_freshEnvironment_unitary,
        QuantumCircuit.exists_selected_onsite_dilation,
        QuantumCircuit.exists_onsite_appended_dilation}
    \leanok
    Every CPTP map $\Phi$ on $m$ $d$-level qudits admits a unitary $V$ on $m+2m$ qudits with $\Phi=\mathcal D_{2m}\circ\operatorname{Ad}_V\circ\mathcal I_{2m}$. Placing $V$ on $e_i\colon s^{-1}(i)\hookrightarrow W$ and $2|s^{-1}(i)|$ fresh wires owned by site $i$ realizes any normalized Kraus channel supported inside $s^{-1}(i)$ as $\mathcal D_A\circ\operatorname{Ad}_{\widetilde V}\circ\mathcal I_A$ with $\operatorname{depth}(\widetilde V)=0$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_selected_environment_reduction, lem:qc_placed_operators}
    Since $\operatorname{rank}(C_\Phi)\le(d^m)^2=d^{2m}$, Stinespring dilation and unitary extension~\cite[Theorems~2.2 and~2.5]{Wolf2012Quantum} yield $V$ on $m+2m$ wires. For a Kraus channel supported in $s^{-1}(i)$, injectivity of $\operatorname{embed}_{e_i}$ gives local normalization on $s^{-1}(i)$, and Lemma~\ref{thm:qc_selected_environment_reduction} places $V$ inside $s^{-1}(i)\sqcup A$ at depth $0$.
\end{proof}

\begin{lemma}[Same-depth unitary realization with one final discard]
    \label{thm:qc_physical_port_unitary_dilation}
    \uses{def:qc_physical_port_protocol, def:qc_appended_environment}
    \lean{QuantumCircuit.IsPhysicalPortProtocol.exists_appended_unitary_dilation,
        QuantumCircuit.IsPhysicalPortProtocol.exists_appended_unitary_dilation_reference}
    \leanok
    Fix $d\ge1$ and a layout on $N\ge1$ sites. Every physical-port protocol $\Phi$ of depth at most $T$ admits a finite fresh environment $A$ and a unitary $U$ on $W\sqcup A$ such that
    \begin{align}
        \mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A & =\Phi,
                                                              & \operatorname{depth}_{\mathrm{intersite}}(U) & \leq T,
        \label{eq:qc_physical_port_dilation}
    \end{align}
    and $((\mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A)\otimes\operatorname{id}_R)=\Phi\otimes\operatorname{id}_R$ for any reference $R$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_appended_environment_operators, thm:qc_fresh_onsite_unitary_dilation,
        thm:qc_appended_environment_composition}
    By induction on Definition~\ref{def:qc_physical_port_protocol}: unitary primitives take $A=\varnothing$ (Lemma~\ref{thm:qc_appended_environment_operators}), onsite Kraus channels take $2|s^{-1}(i)|$ fresh wires at site $i$ of depth $0$ (Lemma~\ref{thm:qc_fresh_onsite_unitary_dilation}), and compositions combine disjoint environments via~(\ref{eq:qc_appended_environment_composition}).
\end{proof}

\begin{lemma}[Bounded unitary simulation on the original ports]
    \label{thm:qc_bounded_unitary_port_io}
    \uses{def:qc_dimension_bounded_protocol, def:qc_appended_environment,
        thm:qc_physical_port_codes}
    \lean{QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.exists_physicalPortUnitaryIO}
    \leanok
    Let $d\ge2$, $N\ge2$, and let $\Psi$ be a $B$-dimension-bounded native protocol of depth $T$ with $d_{\rm in}=d_{\rm out}=d$. With $k=\lceil\log_d B\rceil$, there exists a unitary $U$ on the physical ports, designated memory, and fresh local environments $A$ such that
    \begin{align}
        \mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A\circ J_{\mathrm{port}}
         & =J_{\mathrm{port}}\circ\Psi,
         & \operatorname{depth}_{\mathrm{intersite}}(U) & \leq2kT.
        \label{eq:qc_bounded_unitary_port_io}
    \end{align}
    Uniform family overhead requires $B$ and $d$ to be fixed independently of $N$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_bounded_physical_port_io, thm:qc_physical_port_unitary_dilation}
    Apply Lemma~\ref{thm:qc_physical_port_unitary_dilation} to the protocol $\Gamma_{\mathrm{port}}$ of Lemma~\ref{thm:qc_bounded_physical_port_io} and compose~(\ref{eq:qc_physical_port_dilation}) with $J_{\mathrm{port}}$; the source's QCcc block/control and pure-output system--ancilla conditions remain separate (see~\cite{gap:psc21_physical_port_simulation_scope}).
\end{proof}


\begin{lemma}[Pure output and environment decoupling]
    \label{thm:qc_physical_port_pure_output}
    \uses{thm:qc_physical_port_unitary_dilation}
    \lean{QuantumCircuit.IsPhysicalPortProtocol.exists_appended_unitary_dilation_pure_output}
    \leanok
    Suppose a physical-port channel $\Phi$ of intersite depth at most $T$
    satisfies $\Phi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$, where
    $\langle\phi,\phi\rangle=1$. Its same-depth unitary realization can
    be chosen with the existing product-zero environment initializer so that
    \[
        U(\psi\otimes\ket{0_A})=\phi\otimes\eta,
        \qquad \langle\eta,\eta\rangle=1.
    \]
    The unitary retains the same physical ports and the full channel
    identity of Lemma~\ref{thm:qc_physical_port_unitary_dilation}.
    The environment vector $\eta$ need not factor over spatial sites or
    equal its initial zero state.
\end{lemma}
\begin{proof}\leanok
    In system/environment coordinates write the actual joint output as
    the coefficient matrix $C$. Its first marginal is
    $CC^\dagger=P=\ketbra{\phi}{\phi}$, with $P^2=P$. Thus
    $(I-P)CC^\dagger=0$. Positivity of a matrix Gram product gives
    $(I-P)C=0$, hence $C=PC$. Set
    $\eta=\phi^\dagger C$. Then each coefficient is $\phi_s\eta_e$.
    Taking the trace of the actual marginal and using
    $\langle\phi,\phi\rangle=1$ gives
    $\langle\eta,\eta\rangle=1$. No factorization assumption is used.
\end{proof}


\begin{lemma}[Environment measurement branches of a pure output]
    \label{cor:qc_physical_port_pure_output_branches}
    \uses{thm:qc_physical_port_pure_output}
    \lean{QuantumCircuit.IsPhysicalPortProtocol.exists_environmentKraus_pure_output}
    \leanok
    Under the hypotheses of Lemma~\ref{thm:qc_physical_port_pure_output}, measuring the final environment in its standard product basis gives branch operators $K_e$ satisfying
    \[
        \Phi(X)=\sum_e K_e X K_e^*,\qquad K_e\psi=\eta_e\phi,\qquad \sum_e|\eta_e|^2=1.
    \]
    Thus every outcome of nonzero probability leaves the same pure system state. The circuit has the same physical ports and intersite depth as in that theorem. This does not impose a fixed number of source QCcc blocks or their internal control restrictions.
\end{lemma}

\begin{proof}\leanok
    Take the matrix blocks of the compiled unitary in the initialized environment column. Their channel is the partial trace of that unitary after pure environment initialization. Each block applied to the input vector is the corresponding environment coordinate of the product output from Lemma~\ref{thm:qc_physical_port_pure_output}.
\end{proof}


\begin{lemma}[Bounded pure-state conversion on the original ports]
    \label{lem:qc_bounded_port_pure_output}
    \lean{QuantumCircuit.IsDimensionBoundedLocalChannelProtocol.exists_physicalPort_pure_output}
    \notready
    \discussion{8640}
    \uses{def:qc_dimension_bounded_protocol, thm:qc_physical_port_codes,
        thm:qc_bounded_physical_port_io, thm:qc_physical_port_pure_output}
    Fix $d\ge2$ and an intermediate dimension bound $B$. Let $N\ge2$,
    let $\Psi$ be a $B$-dimension-bounded native protocol of depth $T$,
    and suppose
    $\Psi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$ with
    $\langle\phi,\phi\rangle=1$ and physical input and output dimension $d$.
    Set $k=\lceil\log_d B\rceil$. Let $V_{\rm port}$ store each original
    physical qudit on its port and set its data and scratch digits to zero,
    so $J_{\rm port}(X)=V_{\rm port}XV_{\rm port}^\dagger$.
    There are fresh local environment wires, a physical-port unitary $U$,
    and a unit environment vector $\eta$ such that
    \begin{align}
        U(V_{\rm port}\psi\otimes\ket{0_A})
         & =V_{\rm port}\phi\otimes\eta,
         & \operatorname{depth}_{\rm intersite}(U) & \le2kT,
        \label{eq:qc_bounded_port_pure_vector}               \\
        \mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A\circ J_{\rm port}
         & =J_{\rm port}\circ\Psi.
        \label{eq:qc_bounded_port_pure_channel}
    \end{align}
    Thus the designated data and scratch memory is zero in the actual
    final vector. The environment $\eta$ need not reset or factor over
    sites. For a uniform family bound, $B$ and $d$ are fixed independently
    of $N$. No adaptive protocol-block or control condition is asserted.
\end{lemma}
\begin{proof}
    \notready
    \uses{thm:qc_bounded_physical_port_io, thm:qc_physical_port_pure_output}
    The explicit port code is an isometry, so $V_{\rm port}\phi$ is
    normalized. The channel identity of
    Lemma~\ref{thm:qc_bounded_physical_port_io} gives
    $\Gamma J_{\rm port}(\ketbra{\psi}{\psi})
        =J_{\rm port}(\ketbra{\phi}{\phi})$.
    Apply Lemma~\ref{thm:qc_physical_port_pure_output} to this actual
    compiled channel and these two encoded vectors. It derives $\eta$
    without changing the unitary's physical depth. Compose its full channel
    identity with $J_{\rm port}$ to obtain
    (\ref{eq:qc_bounded_port_pure_channel}).
\end{proof}
