\section{Fixed physical ports and bounded local registers}

The intersite gates of~\cite[Supplement, pp.~7--8]{Piroli2021LOCC}
act on fixed-dimensional physical qudits. Local ancillas are separate:
free onsite operations may mix the physical qudit with its local ancillas,
but a counted intersite gate does not act on an arbitrarily enlarged pair
of sites. This section derives bounded-register channel simulations and
unitary realizations with fresh local environments in that convention.
It does not identify the unrestricted local-channel model with the source's phase relation; see
\cite{gap:psc21_physical_port_simulation_scope}.

\begin{definition}[Physical-port channel protocols]
    \label{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.Geometry.portRing,
        QuantumCircuit.portDataEmbedding,
        QuantumCircuit.IsUnitaryDilation}
    \leanok
    Fix $d,W\ge1$. The ring with physical ports carries $W$ wires of dimension $d$ at every
    site $i\in\Z_N$, the wire $p_i=(i,0)$ being the physical port of $i$. Its cells are the set
    of all wires of a site, of cost $0$, and the pair of ports $\{p_i,p_{i+1}\}$, of cost $1$.
    For $W=2k+1$, the data embedding places a register $x\in\{0,\dots,d-1\}^k$ of the site $i$
    on the data wires $d_{i,t}=(i,1+t)$, with $0$ on the port and on the scratch wires
    $q_{i,t}=(i,1+k+t)$. Let $D$ be a set of wires containing a wire of every site. A map
    $\Phi$ is a \emph{physical-port protocol of depth at most $T$ with environments in $D$} if
    there are finite environment bases at the wires, with at most one element at every wire
    outside $D$, basis vectors $e$ of the input environments, and an unbranched circuit of
    unitary gates of depth at most $T$ on the ring with ports, with Kraus operator $U$ and every
    wire outside $D$ of dimension at most $d$ throughout, such that
    \begin{align}
        \Phi(X) & =\tr_F\bigl(U(X\otimes\ketbra{e}{e})U^\dagger\bigr),\notag
    \end{align}
    where $F$ are the output environments.
\end{definition}

\begin{lemma}[Register codes and completed decoding]
    \label{lem:qc_register_code_completion}
    \lean{KrausFamily.embed,
        KrausFamily.decode,
        KrausFamily.isTP_embed,
        KrausFamily.isTP_decode,
        KrausFamily.map_decode_apply,
        KrausFamily.map_decode_map_embed,
        KrausFamily.encode,
        KrausFamily.IsTP.encode,
        KrausFamily.map_encode_map_embed}
    \leanok
    \uses{def:mpu_initialized_basis_inclusion,thm:mpu_initialized_isometry_synthesis}
    An injection $e\colon\alpha\hookrightarrow\beta$ of finite sets defines an isometry $E=\sum_a\ket{e(a)}\bra{a}$ ($E^\dagger E=I$) and the encoding $\mathcal E(X)=EXE^\dagger$. For $x_0\in\alpha$, the completed decoder
    \begin{align}
        D(Y) & =E^\dagger YE+\sum_{b\notin e(\alpha)}\bra{b}Y\ket{b}\,\ketbra{x_0}{x_0}\notag
    \end{align}
    is CPTP and satisfies $D\circ\mathcal E=\operatorname{id}$. For a CPTP map $\Phi$ with Kraus operators $K_j$ from $\C^\alpha$ to $\C^\gamma$, an injection $f$ of $\gamma$ with isometry $F$, and a point $y_0$ of the target of $f$, the map $\widehat\Phi$ with the Kraus operators $FK_jE^\dagger$ and $\ket{y_0}\bra{b}(I-EE^\dagger)$ for every $b$ is CPTP and satisfies $\widehat\Phi\circ\mathcal E=\mathcal E_f\circ\Phi$, where $\mathcal E_f(X)=FXF^\dagger$.
\end{lemma}
\begin{proof}
    \leanok
    Since $E^\dagger E=I$, the operators $\ket{x_0}\bra{b}$ with $b\notin e(\alpha)$ vanish on the range of $E$, so $D(EXE^\dagger)=E^\dagger EXE^\dagger E=X$. The effect of $\widehat\Phi$ is $E(\sum_jK_j^\dagger K_j)E^\dagger+(I-EE^\dagger)=I$, and on $EXE^\dagger$ only the operators $FK_jE^\dagger$ contribute, giving $F\Phi(X)F^\dagger$.
\end{proof}

\begin{lemma}[Parallel physical SWAP layers]
    \label{lem:qc_port_matching_swaps}
    \uses{def:qc_local_circuit}
    \lean{QuantumCircuit.permGate,
        QuantumCircuit.isUnitary_permGate,
        QuantumCircuit.kraus_ofCells_permGate}
    \leanok
    Let $\pi$ be a permutation of the wires that maps each cell of a family of pairwise disjoint cells onto itself and fixes every other wire, such as the involution exchanging the two ends of every bond of a matching. On each cell $C$ let $g_C$ be the operator with the entries $(g_C)_{u'u}=[u=u'\circ\pi]$ on the configurations of $C$. Every $g_C$ is unitary, and the layer of the gates $g_C$ has the single Kraus operator $P(\pi)$, where $P(\pi)\ket{x}=\ket{x\circ\pi^{-1}}$.
\end{lemma}
\begin{proof}
    \leanok
    Since $\pi$ maps $C$ onto itself, $g_C$ is the permutation matrix of the bijection $u\mapsto u\circ\pi^{-1}$ of the configurations of $C$. The entry of the layer at $(y,x)$ is the product of the entries $[x|_C=y|_C\circ\pi]$ over the cells and of $[x_w=y_w]$ over the other wires, which is the entry of $P(\pi)$.
\end{proof}

\begin{lemma}[Product onsite coding on arbitrary chain inputs]
    \label{lem:qc_product_register_encoding}
    \uses{def:qc_onsite_channel,lem:qc_register_code_completion}
    \lean{QuantumCircuit.encodeCircuit,
        QuantumCircuit.decodeCircuit,
        QuantumCircuit.map_encodeCircuit,
        QuantumCircuit.map_encodeCircuit_congr,
        QuantumCircuit.map_encodeCircuit_trans,
        QuantumCircuit.map_decodeCircuit_comp_map_encodeCircuit}
    \leanok
    For injections $e_i$ of the local bases at every site $i$, with isometries $E_i$, the
    sitewise encoding $\mathcal E_e=\bigotimes_i\mathcal E_{e_i}$ and the sitewise decoding
    $\mathcal D_e=\bigotimes_iD_{e_i}$ of Lemma~\ref{lem:qc_register_code_completion} are onsite
    channels, so they cost no depth. The encoding is the conjugation by $E=\bigotimes_iE_i$ and
    does not depend on the bonds; for further injections $g_i$,
    $\mathcal E_{g\circ e}=\mathcal E_g\circ\mathcal E_e$; and
    $\mathcal D_e\circ\mathcal E_e=\operatorname{id}$ on every chain operator.
\end{lemma}
\begin{proof}
    \uses{lem:qc_register_code_completion}
    \leanok
    Both maps are layers without bonds whose Kraus families at the sites are those of
    Lemma~\ref{lem:qc_register_code_completion}. A Kraus operator of the sitewise decoding times
    $E$ is the product of the sitewise operators times the $E_i$, which is $I$ for the operators
    $E_i^\dagger$ at every site and $0$ otherwise. Finally $E_{g\circ e}=E_gE_e$.
\end{proof}

\begin{lemma}[Encoding a native channel layer]
    \label{thm:qc_encode_native_channel_layer}
    \uses{def:qc_local_channel_circuit,lem:qc_product_register_encoding}
    \lean{QuantumCircuit.Layer.exists_encoded}
    \leanok
    Let $\Lambda$ be a layer of CPTP channels on pairwise disjoint bonds and onsite channels at
    the remaining sites, on nonempty input bases, and let $e_i$ and $f_i$ be injections of its
    input and output bases at every site. Then there is a layer $\widehat\Lambda$ of CPTP
    channels on the same bonds and onsite channels at the remaining sites with
    $\widehat\Lambda\circ\mathcal E_e=\mathcal E_f\circ\Lambda$.
\end{lemma}
\begin{proof}
    \uses{lem:qc_register_code_completion,lem:qc_product_register_encoding}
    \leanok
    The output bases of a CPTP map are nonempty, so the target of every $f_i$ has a point in
    the image of $f_i$. Carry every channel on a bond and every onsite channel along the
    injections of its sites as in Lemma~\ref{lem:qc_register_code_completion}, with such a
    fallback point. A Kraus operator $\widehat L_p$ of the new layer satisfies
    $\widehat L_pE=FL_p$ when all its factors are of the form $FK_jE^\dagger$, with $L_p$ the
    corresponding Kraus operator of $\Lambda$, and $\widehat L_pE=0$ otherwise. The indices of
    the first kind are in bijection with the Kraus indices of $\Lambda$, so
    $\sum_p\widehat L_pEXE^\dagger\widehat L_p^\dagger=F\Lambda(X)F^\dagger$.
\end{proof}

\subsection{Compiling bounded native protocols}

The intermediate local dimensions of a native channel protocol need a common
bound before a family-level physical-depth comparison is available. The next
construction records that bound explicitly, compiles into one fixed register
alphabet, and then realizes those operations on the fixed physical ports.

\begin{definition}[Dimension-bounded native protocols]
    \label{def:qc_dimension_bounded_protocol}
    \uses{def:qc_local_channel_conversion}
    \lean{QuantumCircuit.IsBoundedProtocol,
        QuantumCircuit.IsBoundedProtocol.isProtocol,
        QuantumCircuit.IsBoundedProtocol.dimension_bounds,
        QuantumCircuit.IsBoundedProtocol.mono_bound,
        QuantumCircuit.IsBoundedProtocol.comp,
        KrausFamily.IsTP.nonempty_target,
        QuantumCircuit.IsProtocol.exists_dimension_bound}
    \leanok
    A native local-channel protocol $\Psi$ is \emph{$B$-dimension-bounded} of depth $T$ if it is generated by onsite channels (depth $0$) and nearest-neighbor pair-channel layers (depth $1$) whose input, output, and intermediate local dimensions all satisfy $1\le d_t\le B$. The predicate is monotonic in $B$, additive in $T$ under composition, and every finite native protocol with $d_{\rm in}\ge1$ admits some finite bound $B$ (not necessarily uniform in $N$).
\end{definition}

\begin{lemma}[A fixed-register word for a bounded native protocol]
    \label{thm:qc_fixed_register_word}
    \uses{def:qc_dimension_bounded_protocol,lem:qc_product_register_encoding}
    \lean{QuantumCircuit.Circuit.exists_seq_encoded,
        QuantumCircuit.IsBoundedProtocol.exists_seq_encoded}
    \leanok
    Let $q\ge B$, let $\Psi$ be a $B$-dimension-bounded native protocol of depth $T$, and let
    $e_i$ and $f_i$ be injections of its input and output bases into $\{0,\dots,q-1\}$. Then
    there are layers $O_1,\dots,O_r$ of CPTP channels on pairwise disjoint bonds and onsite
    channels, every intermediate local basis being $\{0,\dots,q-1\}$, of total depth at most
    $T$, whose composed map $\Phi_w=O_r\circ\cdots\circ O_1$ satisfies
    $\Phi_w\circ\mathcal E_e=\mathcal E_f\circ\Psi$.
\end{lemma}
\begin{proof}
    \uses{thm:qc_encode_native_channel_layer,lem:qc_product_register_encoding}
    \leanok
    By induction on the layers of $\Psi$. The protocol without layers is replaced by the encoding
    of the identity layer along $e$ and $f$ (Lemma~\ref{thm:qc_encode_native_channel_layer}), a
    single layer without bonds. For a first layer $\Lambda$, embed the bases after $\Lambda$ into
    $\{0,\dots,q-1\}$ by injections $g_i$, which exist because they have at most $B\le q$
    elements, encode $\Lambda$ along $e$ and $g$ on its own bonds
    (Lemma~\ref{thm:qc_encode_native_channel_layer}), and put it in front of the sequence given
    by induction for the rest of $\Psi$ along $g$ and $f$. The encoded layer has the bonds, hence
    the depth, of $\Lambda$.
\end{proof}

\begin{lemma}[Channels on complete spatial-site wire blocks]
    \label{lem:qc_whole_site_channels}
    \uses{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.Geometry.sites,
        QuantumCircuit.Layer.exists_portRing_sites}
    \leanok
    Read every site of the ring with ports, with its $2k+1$ wires, as one site whose basis is
    the set of configurations of its wires, and let $D$ be a set of wires containing a wire of
    every site. Every product $\bigotimes_{i\in\Z_N}\Phi_i$ of CPTP channels on these sites is
    realized on the wires by a physical-port protocol of depth $0$ with environments in $D$.
\end{lemma}
\begin{proof}
    \uses{thm:qc_fresh_onsite_unitary_dilation}
    \leanok
    Read each $\Phi_i$ as a gate on the cell of all the wires of the site $i$, of cost $0$.
    The resulting layer has the Kraus operators of $\bigotimes_i\Phi_i$ read on the wires, and
    each of its gates acts on a cell meeting $D$; every basis at a wire is
    $\{0,\dots,d-1\}$. Lemma~\ref{thm:qc_fresh_onsite_unitary_dilation} dilates it in depth $0$
    with environments in $D$.
\end{proof}

\begin{lemma}[Physical costs of native register operations]
    \label{thm:qc_native_port_layer}
    \uses{def:qc_physical_port_protocol}
    \lean{QuantumCircuit.Layer.exists_portRing}
    \leanok
    Let $N\ge2$ and $d\ge1$, let $\Lambda$ be a layer of CPTP channels on pairwise disjoint bonds
    and onsite channels on the ring of $N$ sites, with the register $\{0,\dots,d-1\}^k$ at every
    site, let $\iota$ be the data embedding of the registers into the ring with ports with
    $2k+1$ wires per site, and let $D$ be a set of wires containing a wire of every site. Then
    some physical-port protocol $\Gamma_\Lambda$ with environments in $D$, of depth at most $2k$,
    and of depth $0$ when $\Lambda$ has no bonds, satisfies
    $\Gamma_\Lambda\circ\iota=\iota\circ\Lambda$.
\end{lemma}
\begin{proof}
    \uses{lem:qc_port_matching_swaps,thm:qc_encode_native_channel_layer,lem:qc_whole_site_channels,thm:qc_physical_port_unitary_dilation,thm:qc_appended_environment_composition}
    \leanok
    Let $C$ be the set of the left ends $j$ of the bonds $\{j,j+1\}$ of $\Lambda$. Since $N\ge2$,
    the disjointness of the bonds gives $j+1\notin C$ for $j\in C$. For each $t<k$ run three
    layers of exchanges of wires: the port with $d_{j,t}$ at the sites $j\in C$, the ports of $j$
    and $j+1$ for $j\in C$, and the port with $q_{j+1,t}$ at the sites $j+1$. Each layer has the
    single Kraus operator of a wire permutation (Lemma~\ref{lem:qc_port_matching_swaps}) and
    only the middle one has cost, so the routing $R$ formed by the $3k$ layers has depth at most
    $k$. It sends $d_{j,t}$ to $q_{j+1,t}$ for $j\in C$ and fixes the data wires of the other
    sites, so the registers of $j$ and $j+1$ both lie at the site $j+1$. Carry $\Lambda$ to the
    sites $j+1$ along these moved codes (Lemma~\ref{thm:qc_encode_native_channel_layer}) and read
    its channels as gates on all the wires of a site, of cost $0$
    (Lemma~\ref{lem:qc_whole_site_channels}); then undo $R$ by its adjoint. The three parts
    realize $\iota\circ\Lambda$ on the image of $\iota$, have depths $k$, $0$ and $k$, and have
    unitary realizations with environments in $D$
    (Lemmas~\ref{thm:qc_physical_port_unitary_dilation}
    and~\ref{thm:qc_appended_environment_composition}).
\end{proof}

\begin{lemma}[Uniform bounded-register matching simulation]
    \label{thm:qc_bounded_physical_port_simulation}
    \uses{def:qc_physical_port_protocol,lem:qc_register_code_completion,thm:qc_native_port_layer}
    \lean{QuantumCircuit.IsBoundedProtocol.exists_portRing}
    \leanok
    Fix $d\ge1$ and $k\ge0$, and consider a ring of $N\ge2$ sites with $2k+1$ qudits of dimension
    $d$ at every site: the physical port, $k$ data qudits and $k$ scratch qudits. Let $\iota$
    place a register of $k$ digits at every site into its data qudits, with the port and the
    scratch qudits in $0$, and let $D$ be a set of wires containing a wire of every site. If
    $\Psi$ is a native protocol of depth $T$ whose local dimensions along the protocol are
    between $1$ and $B\le d^k$, and $e$ and $f$ are sitewise injections of its input and output
    bases into the registers, there is a physical-port protocol $\Gamma$ of depth at most $2kT$
    with environments in $D$ such that
    \begin{align}
        \Gamma\circ\iota\circ\mathcal E_e & =\iota\circ\mathcal E_f\circ\Psi.\notag
    \end{align}
    For $d\ge2$ one may take $k=\lceil\log_d B\rceil$. Every wire outside $D$ has dimension at
    most $d$ throughout; for $k\ge1$ and $D$ the wires other than the ports, every counted gate
    acts on the two port wires alone and the environments stay at their sites.
\end{lemma}
\begin{proof}
    \uses{lem:qc_register_code_completion,thm:qc_encode_native_channel_layer,thm:qc_native_port_layer,thm:qc_appended_environment_composition}
    \leanok
    By induction on the layers of $\Psi$. Every basis along $\Psi$ has between $1$ and
    $B\le d^k$ elements, so it has an injection into the registers. Encode each layer along the
    injections of its input and output bases
    (Lemma~\ref{thm:qc_encode_native_channel_layer}), route the encoded layer through the ports
    (Lemma~\ref{thm:qc_native_port_layer}), and compose the resulting protocols
    (Lemma~\ref{thm:qc_appended_environment_composition}); the depths add to at most $2kT$.
\end{proof}

\subsection{Original physical-port input and output}

The source treats onsite local unitaries and ancillas as free
resources~\cite[main text, p.~1]{Piroli2021LOCC}. The following bound counts
only intersite layers through the fixed physical ports; it is not a count
that also charges every onsite operation.

\begin{lemma}[Product-zero port and data codes]
    \label{thm:qc_physical_port_codes}
    \uses{lem:qc_product_register_encoding,def:qc_physical_port_protocol}
    \lean{QuantumCircuit.portWireEmbedding,
        QuantumCircuit.exists_portRing_recode}
    \leanok
    A code of a site of the ring with ports is an injection of a finite set into the
    configurations of the wires of the site. Let $J_{\rm port}$ be the sitewise encoding along
    the port code, which places the logical qudit of each site on $p_i$ with $0$ on $(d_i,q_i)$,
    and let $J_{\rm data}$ be the sitewise encoding along a data code, which places the register
    encoding on $d_i$ with $0$ on $(p_i,q_i)$. For any two codes $a$, $a'$ of nonempty sets at
    the sites, with sitewise encodings $J_a$, $J_{a'}$, and any set $D$ of wires containing a
    wire of every site, some physical-port protocol $T_{a\to a'}$ of depth $0$ with environments
    in $D$ satisfies $T_{a\to a'}\circ J_a=J_{a'}$. In particular there are depth-$0$
    physical-port protocols $T_{{\rm port}\to{\rm data}}$ and $T_{{\rm data}\to{\rm port}}$ with
    \begin{align}
        T_{{\rm port}\to{\rm data}}\circ J_{\rm port} & =J_{\rm data},       &
        T_{{\rm data}\to{\rm port}}\circ J_{\rm data} & =J_{\rm port}.\notag
    \end{align}
\end{lemma}
\begin{proof}
    \uses{thm:qc_encode_native_channel_layer,lem:qc_whole_site_channels,lem:qc_product_register_encoding}
    \leanok
    Encode the identity layer on the single sites along $a$ and $a'$
    (Lemma~\ref{thm:qc_encode_native_channel_layer}) and realize the encoded layer of channels on
    whole sites by Lemma~\ref{lem:qc_whole_site_channels}; the sitewise encodings do not depend on
    the bonds (Lemma~\ref{lem:qc_product_register_encoding}).
\end{proof}

\begin{lemma}[Bounded protocols between codes on the wires]
    \label{lem:qc_physical_register_encoding}
    \uses{thm:qc_bounded_physical_port_simulation,thm:qc_physical_port_codes,def:qc_dimension_bounded_protocol}
    \lean{QuantumCircuit.IsBoundedProtocol.exists_portRing_of_code}
    \leanok
    Let $N\ge2$, $d\ge1$ and $B\le d^k$, and let $\Psi$ be a native protocol of depth $T$ whose
    local dimensions along the protocol are between $1$ and $B$. Let $a$ and $b$ be codes of its
    input and output bases in the wires of the ring with ports with $2k+1$ wires per site, with
    sitewise encodings $J_a$ and $J_b$, and let $D$ be a set of wires containing a wire of every
    site. Then some physical-port protocol $\Gamma$ of depth at most $2kT$ with environments in
    $D$ satisfies
    \begin{align}
        \Gamma\circ J_a & =J_b\circ\Psi.\notag
    \end{align}
    The identity concerns inputs in the code $a$, which fixes every wire of a site: the memory
    wires are part of the code, and no identity is asserted for an arbitrary state of the memory
    wires carried along as a complementary reference. In particular, let $d\ge2$,
    $k=\lceil\log_d B\rceil$, and let $\Psi$ have physical input and output dimension $d$. With
    $a=b$ the port code, whose encoding $J_{\mathrm{port}}$ places the input on the physical ports
    $\{p_i\}$ and initializes all designated memory $(d_i,q_i)$ in $\ketbra{0}{0}$, some
    physical-port protocol $\Gamma_{\mathrm{port}}$ with environments in $D$ satisfies
    \begin{align}
        \Gamma_{\mathrm{port}} J_{\mathrm{port}}
         & = J_{\mathrm{port}}\Psi,
         & \operatorname{depth}_{\mathrm{intersite}}(\Gamma_{\mathrm{port}})
         & \leq 2kT.\notag
    \end{align}
    For $D$ the wires other than the ports, the counted gates act on the port wires alone and
    every environment stays at its site.
\end{lemma}
\begin{proof}
    \uses{thm:qc_physical_port_codes,thm:qc_bounded_physical_port_simulation,thm:qc_appended_environment_composition}
    \leanok
    The bases of $\Psi$ have between $1$ and $B\le d^k$ elements, so they have sitewise
    injections $e$ and $f$ into the registers. Change the code $a$ into the data code of $e$
    (Lemma~\ref{thm:qc_physical_port_codes}), apply
    Lemma~\ref{thm:qc_bounded_physical_port_simulation}, and change the data code of $f$ into $b$;
    the protocols compose (Lemma~\ref{thm:qc_appended_environment_composition}) with depth
    $0+2kT+0$. The data code of $e$ is the data embedding after $\mathcal E_e$.
    For $d\ge2$ the integer $k=\lceil\log_d B\rceil$ satisfies $B\le d^k$, and the port code is
    the one of Lemma~\ref{thm:qc_physical_port_codes}.
\end{proof}

\subsection{Unitary circuits with one final environment discard}

A physical-port protocol is by definition a unitary circuit, with the same
physical ports and intersite depth, between a preparation of basis vectors on
local environments and one final trace over them. The following results show
that every protocol of channels has such a realization at no extra depth, with
its environments at prescribed wires.

\begin{definition}[Environment input and discard]
    \label{def:qc_appended_environment}
    \lean{QuantumCircuit.prepareCircuit,
        QuantumCircuit.traceCircuit,
        QuantumCircuit.relabelCircuit}
    \leanok
    For finite environment bases $E_s$ at the wires and basis vectors $e_s\in E_s$, the
    environment initialization $\mathcal I_E$ is the onsite encoding along $a\mapsto(a,e_s)$, and
    for finite bases $F_s$ the environment discard $\mathcal D_F$ is the onsite partial trace over
    $\C^{F_s}$. The relabelling along bijections of the local bases is the onsite encoding along
    them.
\end{definition}

\begin{lemma}[Operators on original wires with an appended environment]
    \label{thm:qc_appended_environment_operators}
    \uses{def:qc_appended_environment,lem:qc_product_register_encoding}
    \lean{QuantumCircuit.map_prepareCircuit_apply,
        QuantumCircuit.map_traceCircuit_apply,
        QuantumCircuit.map_relabelCircuit_apply,
        QuantumCircuit.depth_relabelCircuit,
        QuantumCircuit.map_traceCircuit_kronecker,
        QuantumCircuit.map_traceCircuit_map_tensorRight}
    \leanok
    The initialization acts as $\mathcal I_E(X)=X\otimes\ketbra{e}{e}$ and the discard as
    $\mathcal D_F(Y)=\tr_FY$; a relabelling is an unbranched unitary circuit of depth $0$. The
    discard satisfies $\tr_F(A\otimes\omega)=(\tr\omega)A$, and for every circuit $\Phi$ and
    every $Y$,
    \begin{align}
        \tr_F\bigl((\Phi\otimes\operatorname{id}_F)(Y)\bigr) & =\Phi(\tr_FY).\notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{lem:qc_product_register_encoding}
    The three circuits are single layers without bonds, so their maps are tensor products of
    the local maps (Lemma~\ref{lem:qc_product_register_encoding}). Both sides of the last
    identity are linear in $Y$, and for $Y=A\otimes\omega$ the left side is
    $\tr_F(\Phi(A)\otimes\omega)=(\tr\omega)\Phi(A)$.
\end{proof}

\begin{lemma}[Composition with disjoint fresh environments]
    \label{thm:qc_appended_environment_composition}
    \uses{def:qc_physical_port_protocol,thm:qc_appended_environment_operators}
    \lean{QuantumCircuit.isUnitaryDilation_id,
        QuantumCircuit.IsUnitaryDilation.mono,
        QuantumCircuit.IsUnitaryDilation.comp}
    \leanok
    The identity is a physical-port protocol of depth $0$, and a physical-port protocol of
    depth at most $T$ is one of every larger depth. If $\Phi_1$ and $\Phi_2$ are physical-port
    protocols of depths at most $T$ and $S$ with environments in $D$, realized by $U_1$ with
    environments $E_1,F_1$ and $U_2$ with environments $E_2,F_2$, then $\Phi_2\circ\Phi_1$ is
    realized with environments $E_1\times E_2$ and $F_2\times F_1$ by a unitary circuit $C$ of
    depth at most $T+S$, and
    \begin{align}
        \mathcal D_{F_2\times F_1}\circ\operatorname{Ad}_C\circ\mathcal I_{E_1\times E_2}
         & =(\mathcal D_{F_2}\circ\operatorname{Ad}_{U_2}\circ\mathcal I_{E_2})
        \circ(\mathcal D_{F_1}\circ\operatorname{Ad}_{U_1}\circ\mathcal I_{E_1}).
        \label{eq:qc_appended_environment_composition}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_appended_environment_operators}
    Prepare both environments, apply $U_1\otimes I_{E_2}$, relabel, apply $U_2\otimes I_{F_1}$
    and relabel; the relabellings have depth $0$. The trace over $F_2\times F_1$ is the trace
    over $F_2$ after the one over $F_1$, which commutes with $U_2\otimes I_{F_1}$ by
    Lemma~\ref{thm:qc_appended_environment_operators}, and the trace over $F_1$ of the first
    output gives $\Phi_1$. Outside $D$ every environment has at most one element, so the
    product environments do too.
\end{proof}

\begin{lemma}[Fresh onsite unitary dilations]
    \label{thm:qc_fresh_onsite_unitary_dilation}
    \uses{def:qc_appended_environment,def:qc_physical_port_protocol}
    \lean{KrausFamily.IsTP.exists_unitary_dilation,
        QuantumCircuit.Layer.IsUnitaryOff,
        QuantumCircuit.Layer.isUnitaryDilation}
    \leanok
    Every CPTP map $\Phi$ from $\C^\alpha$ to $\C^\beta$ with Kraus operators
    $K_1,\dots,K_r$, on a nonempty basis $\alpha$, satisfies
    $\Phi(X)=\tr_{\alpha\times[r]}\bigl(V(X\otimes\ketbra{e_0}{e_0})V^\dagger\bigr)$ for a
    unitary $V$ from $\C^{\alpha\times\beta\times[r]}$ to $\C^{\beta\times\alpha\times[r]}$ and a
    basis vector $e_0$ of $\C^{\beta\times[r]}$. Consequently, let $D$ be a set of wires and let
    a layer of CPTP gates and onsite channels on nonempty input bases have unitary gates on the
    cells disjoint from $D$ and unitary onsite channels outside $D$. Then it is a physical-port
    protocol of its own depth with environments in $D$, realized by one layer of unitary gates
    on the same cells.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:qc_placed_operators}
    The first statement is Stinespring dilation followed by a unitary extension of the isometry
    $X\mapsto\sum_jK_jX\otimes\ket{j}$~\cite[Theorems~2.2 and~2.5]{Wolf2012Quantum}. For the
    layer, keep every unitary gate and onsite channel and replace every other one by its
    dilation, placing the environment of a gate at one site of its cell, a site of $D$ if the
    cell meets $D$. The entries of the dilated layer factor over the cells and the remaining
    sites (Lemma~\ref{lem:qc_placed_operators}), and on every factor the trace over the
    environment gives back the original gate or channel.
\end{proof}

\begin{lemma}[Same-depth unitary realization with one final discard]
    \label{thm:qc_physical_port_unitary_dilation}
    \uses{def:qc_physical_port_protocol,def:qc_appended_environment}
    \lean{QuantumCircuit.Circuit.IsUnitaryOff,
        QuantumCircuit.Circuit.isUnitaryOff_of_forall_isUnitary,
        QuantumCircuit.Circuit.isUnitaryDilation}
    \leanok
    Let $D$ be a set of wires, and let $\Phi$ be realized by a sequence of layers of CPTP
    gates and onsite channels on nonempty input bases, of depth at most $T$, each with unitary
    gates on the cells disjoint from $D$ and unitary onsite channels outside $D$, and with every
    wire outside $D$ of dimension at most $d$. Then there are environments in $D$ and a unitary
    circuit $U$ on the ring with ports with
    \begin{align}
        \mathcal D_F\circ\operatorname{Ad}_U\circ\mathcal I_E & =\Phi,
                                                              & \operatorname{depth}_{\mathrm{intersite}}(U) & \leq T,
        \label{eq:qc_physical_port_dilation}
    \end{align}
    and hence $((\mathcal D_F\circ\operatorname{Ad}_U\circ\mathcal I_E)\otimes\operatorname{id}_R)=\Phi\otimes\operatorname{id}_R$ for any reference $R$. A circuit of unitary gates satisfies the hypothesis for every $D$, and every protocol of depth at most $T$ on nonempty input bases satisfies~(\ref{eq:qc_physical_port_dilation}) with $D$ the set of all wires.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_appended_environment_operators,thm:qc_fresh_onsite_unitary_dilation,thm:qc_appended_environment_composition}
    By induction on the layers: the sequence without layers is the identity, of depth $0$
    (Lemma~\ref{thm:qc_appended_environment_composition}), every layer is dilated at its own
    depth by Lemma~\ref{thm:qc_fresh_onsite_unitary_dilation}, and the dilations compose
    via~(\ref{eq:qc_appended_environment_composition}).
\end{proof}

\begin{lemma}[Pure output and environment decoupling]
    \label{thm:qc_physical_port_pure_output}
    \uses{thm:qc_physical_port_unitary_dilation}
    \lean{QuantumCircuit.IsProtocol.isUnitaryDilation,
        QuantumCircuit.Circuit.exists_unit_product_approx}
    \leanok
    Let $\Phi$ be a protocol of depth at most $T$ on nonempty local bases, for
    any assignment of costs to the gates, such as the physical-port one. Then
    $\Phi=\mathcal D_A\circ\operatorname{Ad}_U\circ\mathcal I_A$ for a
    unitary circuit $U$ of depth at most $T$, where $\mathcal I_A$ prepares
    local environments in basis states $\ket{0^A}$ and $\mathcal D_A$ discards
    the output environments. If
    $\|\Phi(\ketbra{\psi}{\psi})-\ketbra{\phi}{\phi}\|_1\le\varepsilon$
    for unit vectors $\psi,\phi$, then there is a unit environment vector
    $\eta$ with
    \[
        \|U(\psi\otimes\ket{0^A})-\phi\otimes\eta\|_2\le\sqrt{2\varepsilon}.
    \]
    In particular, if $\Phi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$,
    then $U(\psi\otimes\ket{0^A})=\phi\otimes\eta$ with
    $\langle\eta,\eta\rangle=1$. The environment vector $\eta$ need not
    factor over spatial sites or equal its initial zero state.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:qc_physical_port_unitary_dilation}
    The dilation is built as in Lemma~\ref{thm:qc_physical_port_unitary_dilation}:
    each layer is dilated with fresh local environments at no extra depth, and
    compositions carry the environments of each factor through the other.
    Let $C_{s,e}=\bigl(U(\psi\otimes\ket{0^A})\bigr)_{s,e}$ be the
    coefficient matrix of the joint output, a unit vector since $U$ is
    unitary. Its first marginal is
    \begin{align}
        CC^\dagger & =\tr_A\bigl(U(\ketbra{\psi}{\psi}\otimes Z_A)U^\dagger\bigr)
        =\Phi(\ketbra{\psi}{\psi}).\notag
    \end{align}
    Project $C$ onto $\phi$: the row $\phi^\dagger C$ has squared norm
    $\langle\phi,CC^\dagger\phi\rangle\ge1-\varepsilon$, and normalizing
    it gives $\eta$ with $\|C-\phi\eta^{\mathsf T}\|_2^2\le2\varepsilon$.
    If the row is zero, then $\varepsilon\ge1$ and any unit vector $\eta$
    suffices. For $\varepsilon=0$ the bound is the equality
    $C=\phi\eta^{\mathsf T}$. No factorization assumption is used.
\end{proof}


\begin{lemma}[Environment measurement branches of a pure output]
    \label{cor:qc_physical_port_pure_output_branches}
    \uses{thm:qc_physical_port_pure_output}
    Under the hypotheses of Lemma~\ref{thm:qc_physical_port_pure_output} with an exact pure output,
    $\Phi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$, measuring the final environment in its standard product basis gives branch operators $K_e$ satisfying
    \[
        \Phi(X)=\sum_e K_e X K_e^\dagger,\qquad K_e\psi=\eta_e\phi,\qquad \sum_e|\eta_e|^2=1.
    \]
    Thus every outcome of nonzero probability leaves the same pure system state. The circuit has the same physical ports and intersite depth as in that theorem. This does not impose a fixed number of source QCcc blocks or their internal control restrictions.
\end{lemma}

\begin{proof}
    \uses{thm:qc_physical_port_pure_output}
    Let $K_e=(I_W\otimes\bra{e})U(I_W\otimes\ket{0^A})$ be the block of the
    compiled unitary in the initialized environment column. Then
    $\sum_e K_eXK_e^\dagger=\tr_A\bigl(U(X\otimes Z_A)U^\dagger\bigr)=\Phi(X)$,
    and by Lemma~\ref{thm:qc_physical_port_pure_output}
    \[
        K_e\psi=(I_W\otimes\bra{e})\,U(\psi\otimes\ket{0^A})
        =(I_W\otimes\bra{e})(\phi\otimes\eta)=\eta_e\phi.
    \]
\end{proof}




\begin{lemma}[Bounded pure-state conversion on the original ports]
    \label{lem:qc_bounded_port_pure_output}
    \discussion{8640}
    \uses{def:qc_dimension_bounded_protocol,thm:qc_physical_port_codes,lem:qc_physical_register_encoding,thm:qc_physical_port_pure_output}
    Fix $d\ge2$ and an intermediate dimension bound $B$. Let $N\ge2$,
    let $\Psi$ be a $B$-dimension-bounded native protocol of depth $T$,
    and suppose
    $\Psi(\ketbra{\psi}{\psi})=\ketbra{\phi}{\phi}$ with
    $\langle\psi,\psi\rangle=\langle\phi,\phi\rangle=1$ and physical input and output
    dimension $d$. Set $k=\lceil\log_d B\rceil$. Let $V_{\rm port}$ store each original
    physical qudit on its port and set its data and scratch digits to zero,
    so $J_{\rm port}(X)=V_{\rm port}XV_{\rm port}^\dagger$, and let $D$ be a set of wires
    containing a wire of every site.
    There are local environments at the wires of $D$, basis vectors $e$ of them, a
    physical-port unitary $U$, and a unit environment vector $\eta$ such that
    \begin{align}
        U(V_{\rm port}\psi\otimes e)
         & =V_{\rm port}\phi\otimes\eta,
         & \operatorname{depth}_{\rm intersite}(U) & \le2kT,
        \label{eq:qc_bounded_port_pure_vector}               \\
        \mathcal D_F\circ\operatorname{Ad}_U\circ\mathcal I_E\circ J_{\rm port}
         & =J_{\rm port}\circ\Psi.
        \label{eq:qc_bounded_port_pure_channel}
    \end{align}
    Thus the designated data and scratch memory is zero in the
    final vector. The environment $\eta$ need not reset or factor over
    sites. For a uniform family bound, $B$ and $d$ are fixed independently
    of $N$. No adaptive protocol-block or control condition is asserted.
\end{lemma}
\begin{proof}
    \uses{lem:qc_register_code_completion,lem:qc_physical_register_encoding,thm:qc_physical_port_pure_output}
    The explicit port code is an isometry, so $V_{\rm port}\psi$ and $V_{\rm port}\phi$ are
    normalized. The channel identity of
    Lemma~\ref{lem:qc_physical_register_encoding} gives
    $\Gamma J_{\rm port}(\ketbra{\psi}{\psi})
        =J_{\rm port}(\ketbra{\phi}{\phi})$.
    Apply Lemma~\ref{thm:qc_physical_port_pure_output} with $\varepsilon=0$ to the unitary
    realization of this compiled channel and these two encoded vectors. It derives $\eta$ with
    $U(V_{\rm port}\psi\otimes e)=V_{\rm port}\phi\otimes\eta$
    in~(\ref{eq:qc_bounded_port_pure_vector}), without changing the
    unitary's physical depth, and its channel identity composed with $J_{\rm port}$ is
    (\ref{eq:qc_bounded_port_pure_channel}).
\end{proof}


\begin{lemma}[Approximate pure-state conversion on the original ports]
    \label{lem:qc_bounded_port_approximate_output}
    \discussion{8640}
    \uses{def:qc_dimension_bounded_protocol,thm:qc_physical_port_codes,def:qc_approx_local_channel_conversion}
    \lean{QuantumCircuit.IsBoundedProtocol.exists_portRing_unit_product_approx}
    \leanok
    Fix $d\ge2$ and an intermediate dimension bound $B$. Let $N\ge2$ and
    let $\Psi$ be a $B$-dimension-bounded native channel of depth $T$ with
    physical input and output dimension $d$. For unit vectors $\psi,\phi$,
    suppose
    $\|\Psi(\ketbra{\psi}{\psi})-\ketbra{\phi}{\phi}\|_1\le\varepsilon$.
    Set $k=\lceil\log_d B\rceil$, let $V_{\rm port}$ be the explicit
    original-port isometry with zero designated memory, and let $D$ be a set of wires
    containing a wire of every site. There are local environments at the wires of $D$, basis
    vectors $e$ of them, a unitary $U$ and a unit environment vector $\eta$ such that
    \begin{align}
        \|U(V_{\rm port}\psi\otimes e)
        -V_{\rm port}\phi\otimes\eta\|_2        & \le\sqrt{2\varepsilon},
        \label{eq:qc_bounded_port_approximate_vector}                     \\
        \operatorname{depth}_{\rm intersite}(U) & \le2kT.
    \end{align}
    The same unitary satisfies the full logical channel identity
    (\ref{eq:qc_bounded_port_pure_channel}). The estimate uses the Euclidean
    vector norm and the full trace norm. It imposes no strict upper bound on
    $\varepsilon$ and does not reset the additional environment or provide
    an adaptive source protocol-block decomposition. A uniform family fixes
    $B$ before the chain length.
\end{lemma}
\begin{proof}
    \uses{lem:qc_physical_register_encoding,thm:qc_physical_port_unitary_dilation}
    \leanok
    Construct the actual port channel and its unitary dilation. The full
    dilation identity identifies the marginal of its actual output, while
    trace preservation and normalization of $\psi$ give a unit joint vector.
    Conjugation by $V_{\rm port}$ preserves the trace-norm error. Project
    the actual joint coefficient matrix onto the unit vector
    $V_{\rm port}\phi$. Its environment row has squared norm at least
    $1-\varepsilon$; normalizing that row gives product-vector distance at
    most $\sqrt{2\varepsilon}$. If the row is zero, then
    $\varepsilon\ge1$ and a unit coordinate environment vector suffices.
    This changes neither the unitary nor its physical depth.
\end{proof}
