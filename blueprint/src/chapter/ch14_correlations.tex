\chapter{Exponential Decay of Correlations}\label{ch:correlations}

In the thermodynamic limit, spatial correlation functions of matrix product
states are governed by the subleading spectrum of the transfer map $\E_A$.
Subtracting the leading fixed-point contribution leaves a connected correlator
decaying exponentially (with polynomial factors $n^k$ for non-diagonal Jordan
blocks) at a rate defining the spatial correlation length. For tensors
generating parent Hamiltonians (Chapter~\ref{ch:parent_hamiltonian}), this
spectral gap controls ground-state correlations.

While Chapter~\ref{ch:spectral} analyzes the peripheral spectrum and overlap
decay, this chapter develops the one-point and two-point functionals, their
spectral decomposition, and quantitative decay bounds.

\section{Connected correlators from the transfer map}

\begin{definition}[Auxiliary-space one-point functional]\label{def:one_point_expectation}
    \lean{MPSTensor.onePointExpectation}
    \leanok
    \uses{def:transfer_map, thm:qpf}
    Fix an MPS tensor and a right fixed point of its transfer map.
    Write the tensor as $A$ and the fixed point as $\rho^R$.
    Under the hypotheses of Theorem~\ref{thm:qpf}, that fixed point exists
    and is unique up to scaling; we normalize it so that $\tr(\rho^R)=1$.
    For an auxiliary-space insertion $X\in\MN{D}$, define
    \begin{align}
        \langle X_0 \rangle & = \tr(X\rho^R). \notag
    \end{align}
\end{definition}

\begin{definition}[Auxiliary-space two-point functional at distance $n$]\label{def:two_point_expectation}
    \lean{MPSTensor.twoPointExpectation}
    \leanok
    \uses{def:transfer_map, def:one_point_expectation}
    For auxiliary-space insertions $X,Y\in\MN{D}$ and $n\ge0$, define
    \begin{align}
        \langle X_0 Y_n \rangle
         & = \tr\!\left(Y\,\E_A^n(X\rho^R)\right). \notag
    \end{align}
    Equivalently, insert $X$ at site $0$, propagate by $\E_A^n$, then insert
    $Y$ and take the trace.
    \begin{center}
        \tenkzkernel
        % Formula: \langle X_0Y_n\rangle
        %   = \tr\!\left(Y\,\E_A^n(X\rho^R)\right).
        % Source verdict: Cirac2021Matrix, Section II.B.3, supports the
        % correlator discussion but supplies no dedicated picture of this
        % contraction; the displayed definition is authoritative here.
        % For composite indices a=(k,l) and b=(i,j), the doubled map carries

        % \sum_{i,j,k,l}Y_{ji}(\E_A^n)_{ij,kl}(X\rho^R)_{kl}.
        % The east cup is the applied-channel input; reading from the west
        % instead feeds the adjoint and gives the same trace pairing with Y.
        % Boundary signature: (virtual-west=0 | virtual-east=0 |

        \begin{tenkz}[rows={wire,wire}, cols=3,

            \tn[at=(1,2), skin=ring, wires=2]{\E_A^n}
        \end{tenkz}
    \end{center}
\end{definition}

\begin{definition}[Auxiliary-space connected correlator]\label{def:connected_correlator}
    \lean{MPSTensor.connectedCorrelator}
    \leanok
    \uses{def:two_point_expectation, def:one_point_expectation}
    The connected two-point function is
    \begin{align}
        C(X,Y;n)
         & = \langle X_0Y_n\rangle
        - \langle X_0\rangle\langle Y_0\rangle. \notag
    \end{align}
\end{definition}

\begin{definition}[Fixed-point-centered auxiliary insertion]%
    \label{def:correlator_traceless_part}
    \lean{MPSTensor.tracelessPart}
    \leanok
    For a matrix $\rho\in\MN{D}$ and an auxiliary insertion
    $X\in\MN{D}$, define
    \begin{align}
        Z_\rho(X)
         & = X\rho-\tr(X\rho)\rho.
        \label{eq:correlator_traceless_part}
    \end{align}
    When $\tr(\rho)=1$, this removes the trace component of $X\rho$.
\end{definition}

\begin{lemma}[Trace of the centered insertion]%
    \label{lem:trace_correlator_traceless_part}
    \lean{MPSTensor.trace_tracelessPart}
    \leanok
    \uses{def:correlator_traceless_part}
    If $\tr(\rho)=1$, then
    \begin{align}
        \tr\!\bigl(Z_\rho(X)\bigr) & =0.
    \end{align}
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:correlator_traceless_part}
    By linearity of the trace,
    \begin{align}
        \tr\!\bigl(Z_\rho(X)\bigr)
         & =\tr(X\rho)-\tr(X\rho)\tr(\rho)
        =0.
    \end{align}
\end{proof}

\begin{theorem}[Fixed-point centering identity for connected correlators]%
    \label{thm:connected_correlator_traceless_reduction}
    \lean{MPSTensor.connectedCorrelator_eq_trace_transfer_tracelessPart}
    \leanok
    \uses{def:connected_correlator, def:correlator_traceless_part, def:transfer_map}
    Suppose that $\rho$ is a right fixed point of the transfer map,
    $\E_A(\rho)=\rho$.  Then, for every $n\geq0$,
    \begin{align}
        C(X,Y;n)
         & =\tr\!\left(Y\,\E_A^n\bigl(Z_\rho(X)\bigr)\right).
        \label{eq:connected_correlator_traceless_reduction}
    \end{align}
    When moreover $\tr(\rho)=1$, the preceding lemma shows that
    $Z_\rho(X)$ is traceless.  The identity isolates the fixed-point
    contribution before the remaining transfer iterates are analysed.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:connected_correlator, def:correlator_traceless_part}
    Since $\E_A(\rho)=\rho$, one has $\E_A^n(\rho)=\rho$ for every $n$.
    Linearity therefore gives
    \begin{align}
        \tr\!\left(Y\,\E_A^n\bigl(Z_\rho(X)\bigr)\right)
         & =
        \tr\!\left(Y\,\E_A^n(X\rho)\right)
        -\tr(X\rho)\tr(Y\rho),
    \end{align}
    which is the definition of $C(X,Y;n)$.
\end{proof}

\section{Spectral expansion and exponential decay}

\begin{theorem}[Sum of exponentials under diagonalizability]%
    \label{thm:correlator_sum_exponentials}
    \uses{def:connected_correlator}
    Let $\E_A$ be trace-preserving, with fixed point $\rho$
    satisfying $\tr\rho=1$. Suppose its restriction to the traceless matrices
    is diagonalizable, with eigenvalues $\lambda_1,\ldots,\lambda_{D^2-1}$,
    counted with multiplicity. For every $X,Y$, there are coefficients
    $c_j\in\C$ such that, for all $n\ge0$,
    \begin{align}
        C(X,Y;n)
         & = \sum_{j=1}^{D^2-1} c_j\,\lambda_j^n.
        \label{eq:correlations_spectral_expansion}
    \end{align}
    This is the diagonalizable case of the expansion in
    \cite[Section~II.B.3]{Cirac2021Matrix}. Without diagonalizability,
    Jordan blocks can contribute polynomial factors multiplying the
    exponential terms, so the source's unqualified pure-exponential form
    needs this hypothesis \cite{gap:cpgsv21_correlator_diagonalizable_expansion}.
\end{theorem}

\begin{proof}
    \uses{thm:connected_correlator_traceless_reduction, lem:trace_correlator_traceless_part, def:correlator_traceless_part}
    The matrix $Z_\rho(X)=X\rho-\tr(X\rho)\rho$ is traceless. Expanding it
    in an eigenbasis $(v_j)$ of the traceless subspace,
    \begin{align}
        Z_\rho(X)   & =\sum_{j=1}^{D^2-1}\gamma_j(X)\,v_j,
        \\
        \E_A^n(v_j) & =\lambda_j^n v_j .
    \end{align}
    By Theorem~\ref{thm:connected_correlator_traceless_reduction},
    \begin{align}
        C(X,Y;n)
         & =\tr\!\left(Y\,\E_A^n\bigl(Z_\rho(X)\bigr)\right)
        =\sum_{j=1}^{D^2-1}\gamma_j(X)\,\tr(Y v_j)\,\lambda_j^n,
    \end{align}
    so $c_j=\gamma_j(X)\tr(Yv_j)$.
    % The eigenbasis construction is tracked in Issue~\#1447.
\end{proof}

\begin{definition}[Physical connected fixed-point correlator]
    \label{def:physical_connected_correlator}
    \lean{MPSTensor.physicalConnectedCorrelator}
    \leanok
    \uses{def:physical_observable_transfer, def:transfer_map}
    Let $\rho\in\MN{D}$ have nonzero trace, and set
    $P(Z)=\tr(Z)\rho/\tr\rho$. For physical operators $X,Y$ on independently
    chosen blocks of lengths $L_1,L_2$, and $n\geq0$ unobserved intermediate
    sites, define
    \[
        C_{\mathrm{phys}}(X,Y;n)=
        \tr\!\left(\E_X(\E_A-P)^n
        \bigl(\E_Y(\rho)-P(\E_Y(\rho))\bigr)\right).
    \]
    The centering of the inner insertion includes adjacent blocks, $n=0$.
\end{definition}

\begin{lemma}[Physical connected contraction]
    \label{lem:physical_connected_correlator_identity}
    \lean{MPSTensor.physicalConnectedCorrelator_eq_twoPoint_sub,
        MPSTensor.physicalConnectedCorrelator_eq_compl_pow_of_pos,
        MPSTensor.limitCorrelator_eq_compl_pow_of_pos}
    \leanok
    \uses{def:physical_connected_correlator}
    Suppose $\E_A$ is trace preserving, $\E_A(\rho)=\rho$, and $\tr\rho=1$.
    Then, for all $n\geq0$,
    \[
        C_{\mathrm{phys}}(X,Y;n)=
        \tr\!\left(\E_X\E_A^n(\E_Y(\rho))\right)
        -\tr(\E_X(\rho))\tr(\E_Y(\rho)).
    \]
    For $n\geq1$ this also equals
    $\tr(\E_X(\E_A-P)^n(\E_Y(\rho)))$, including the equal-support
    specialization used in preparation estimates.
\end{lemma}
\begin{proof}
    \leanok
    Trace preservation and stationarity give $P\E_A=\E_AP=P$ and $P^2=P$.
    Induction yields
    $(\E_A-P)^n(Z-PZ)=\E_A^nZ-PZ$ at every $n\geq0$.
    Taking $Z=\E_Y(\rho)$ proves the first identity. For $n\geq1$,
    $\E_A^n=P+(\E_A-P)^n$, which proves the second.
\end{proof}

\begin{lemma}[Complementary gap in trace-preserving gauge]
    \label{lem:normal_tensor_complement_gap}
    \lean{MPSTensor.normalTensor_complement_spectralRadius_lt_one}
    \leanok
    \uses{def:nt_cpsv, def:transfer_map}
    Let $A$ be normal in trace-preserving gauge, and let $\rho\succeq0$ satisfy
    $\E_A(\rho)=\rho$ and $\tr\rho=1$. Then
    $\rho_{\spec}(\E_A-P)<1$, where $P(Z)=\tr(Z)\rho$.
\end{lemma}
\begin{proof}
    \leanok
    Normality supplies irreducibility and peripheral spectrum $\{1\}$.
    The complementary transfer-map gap for irreducible primitive channels
    applies to the nonzero positive fixed state
    \cite[``Complementary transfer-map gap for primitive maps'']{QICLean2026Blueprint}.
\end{proof}

\begin{theorem}[Physical correlation decay in trace-preserving gauge]%
    \label{thm:physical_correlator_tp_bound}
    \lean{MPSTensor.exists_physicalConnectedCorrelator_le_geometric,
        MPSTensor.exists_rate_physicalConnectedCorrelator_le_geometric}
    \leanok
    \uses{def:physical_connected_correlator, lem:normal_tensor_complement_gap}
    For any tensor $A$ and matrix $\rho$ of nonzero trace, every rate
    $r>\rho_{\spec}(\E_A-P)$ and every pair of physical operators on independent
    finite supports admit a constant $C_{XY}>0$ such that, for all $n\geq0$,
    \begin{align}
        |C_{\mathrm{phys}}(X,Y;n)|\leq C_{XY}r^n.
        \label{eq:correlations_decay_bound}
    \end{align}
    If $A$ is normal in trace-preserving gauge and $\rho\succeq0$ is a
    normalized fixed state, one can choose a single rate
    $\rho_{\spec}(\E_A-P)<r<1$ for every finite support and every pair of
    observables. Thus the fixed-point infinite-chain connected correlations
    decay exponentially, without a diagonalizability assumption.
    The source's pure-exponential expansion needs the Jordan-block
    correction recorded in
    \cite{gap:cpgsv21_correlator_diagonalizable_expansion}.
\end{theorem}
\begin{proof}
    \leanok
    Put $T=\E_A-P$ and $Z=\E_Y(\rho)-P(\E_Y(\rho))$.
    The Gelfand formula gives a constant $C>0$ such that
    $\|T^nZ\|\leq Cr^n\|Z\|$ at every $n\geq0$.
    The linear functional $W\mapsto\tr(\E_X(W))$ on the finite-dimensional
    matrix space has a bound $|\tr(\E_X(W))|\leq M_X\|W\|$ with $M_X>0$.
    Hence $C_{XY}=M_XC\|Z\|+1$ proves the estimate.
    Lemma~\ref{lem:normal_tensor_complement_gap} permits the choice $r<1$.
\end{proof}

\begin{remark}[Transfer-map spectral hypothesis]%
    \label{rem:spectral_hypothesis}
    \uses{thm:correlator_exponential_bound, thm:transfer_operator_gap_irr_tp, thm:transfer_map_gap_wielandt}
    The rate $r<1$ in Theorem~\ref{thm:correlator_exponential_bound}
    gives exponential decay. It
    is a transfer-map gap condition: all non-leading eigenvalues of $\E_A$ lie
    strictly inside the unit disk. It is distinct from the Hamiltonian spectral gap of
    Chapter~\ref{ch:parent_hamiltonian_commuting_and_gap}.
    The complementary transfer-map gap $\rho_{\spec}(\E_A-P)<1$ is established for
    primitive channels
    (the corresponding theorem in the companion quantum-channel volume~\cite[``Complementary transfer-map gap for primitive maps'']{QICLean2026Blueprint}).
    The mixed-transfer gap needed for separation of non-equivalent canonical
    blocks is supplied for irreducible trace-preserving tensors
    (Theorem~\ref{thm:transfer_operator_gap_irr_tp}), and quantitative
    single-tensor transfer-map bounds for injective tensors are recorded in
    Theorem~\ref{thm:transfer_map_gap_wielandt}.

    These transfer-map gap results give exponential convergence of the
    complementary transfer map at every rate strictly exceeding its
    spectral radius.
\end{remark}

\begin{definition}[Correlation length]\label{def:correlation_length}
    \lean{MPSTensor.correlationLength}
    \leanok
    \uses{def:transfer_map}
    For every $\lambda_2\in\C$, define the associated correlation-length
    quantity by
    \begin{align}
        \xi
         & = -\frac{1}{\log|\lambda_2|}. \notag
    \end{align}
    For a nonzero subleading eigenvalue in the physical range
    $0<|\lambda_2|<1$, this quantity is positive. If all subleading spectral
    values vanish and correlations therefore vanish after finitely many
    transfer steps, the total definition gives the limiting value $\xi=0$.
\end{definition}

\begin{lemma}[Correlation length is positive]\label{lem:correlation_length_pos}
    \lean{MPSTensor.correlationLength_pos}
    \leanok
    \uses{def:correlation_length}
    If $|\lambda_2| < 1$ and $\lambda_2 \ne 0$, then $\xi > 0$.
\end{lemma}

\begin{proof}
    \leanok
    Since $0<|\lambda_2|<1$, we have $\log|\lambda_2|<0$, so
    $\xi=-1/\log|\lambda_2|>0$.
\end{proof}

\section{Quantitative transfer-map gap bounds}

The qualitative bound $\rho_{\spec}(\E_A-P)<1$ for primitive channels
\cite[``Complementary transfer-map gap for primitive maps'']{QICLean2026Blueprint}
admits quantitative refinements in terms of geometric constants and decay rates
for the transfer map $\E_A$.

\begin{theorem}[Exponential convergence of injective primitive channels]%
    \label{thm:exponential_convergence_primitive}
    \lean{MPSTensor.exponential_convergence_of_primitive}
    \leanok
    \uses{thm:primitive_overlap}
    Fix an injective, trace-preserving MPS tensor $A$ with positive
    definite fixed point $\rho$.
    Let
    \begin{align}
        P(X)
         & = \frac{\tr(X)}{\tr(\rho)}\rho
        \label{eq:correlations_fixed_point_projection}
    \end{align}
    be the corresponding fixed-point projection.
    Then there exist $C > 0$ and $0 < \delta \le 1$ such that, for all
    $n\ge 0$ and $X\in\MN{D}$,
    \begin{align}
        \|\E_A^n(X)-P(X)\|
         & \le C(1-\delta)^n\|X\|.
        \label{eq:correlations_channel_convergence}
    \end{align}
    The transfer-map gap $\delta$ exists by primitivity
    (the corresponding theorem in the companion quantum-channel volume~\cite[``Complementary transfer-map gap for primitive maps'']{QICLean2026Blueprint}, the complementary transfer-map
    gap).
\end{theorem}

\begin{proof}
    \leanok
    Injectivity implies primitivity of the transfer map.
    The complementary spectral-radius bound $\rho_{\spec}(\E_A-P)<1$ then follows
    from the primitive transfer-map gap theorem. Choose
    $r$ with $\rho_{\spec}(\E_A-P)<r<1$. The Gelfand formula yields
    $\|(\E_A-P)^n\|\le Cr^n$ for a suitable constant $C>0$.
    Since $P$ is the fixed-point projection, $\E_A P=P\E_A=P$, and hence
    $\E_A^n-P=(\E_A-P)^n$ for $n\ge 1$. The case $n=0$ is absorbed by
    increasing $C$ if necessary. Setting $\delta=1-r$ gives
    (\ref{eq:correlations_channel_convergence}).
\end{proof}

\begin{theorem}[Correlation length bound]\label{thm:correlation_length_bound}
    \lean{MPSTensor.correlation_length_bound}
    \leanok
    \uses{def:correlation_length, thm:exponential_convergence_primitive, def:injective}
    For an injective trace-preserving MPS tensor, there exist
    $C>0$ and $\xi>0$ such that, for all $n\ge 0$ and all
    traceless $X\in\MN{D}$, i.e.\ $\tr(X)=0$,
    \begin{align}
        \|\E_A^n(X)\|
         & \le C e^{-n/\xi}\|X\|.
        \label{eq:correlations_traceless_decay}
    \end{align}
    Traceless matrices lie in $\ker P$, where $P$ is the fixed-point
    projection (\ref{eq:correlations_fixed_point_projection}).
    Since $\E_A-P$ has spectral radius strictly less than~$1$, choose any
    $r$ with $\rho_{\spec}(\E_A-P)<r<1$. The Gelfand formula then gives exponential
    decay with $\xi=-1/\log r$. Choosing a rate strictly above the spectral
    radius is necessary in general because a Jordan block at the spectral
    radius can prevent a uniform bound proportional to
    $\rho_{\spec}(\E_A-P)^n$.
\end{theorem}

\begin{proof}
    \leanok
    For traceless $X$, we have $P(X)=0$, so
    $\E_A^n(X)=(\E_A-P)^n(X)$.
    Applying the Gelfand-formula estimate with
    $\rho_{\spec}(\E_A-P)<r<1$ gives $\|\E_A^n(X)\|\le Cr^n\|X\|$.
    With $\xi=-1/\log r>0$, the identity $r^n=e^{-n/\xi}$ yields
    (\ref{eq:correlations_traceless_decay}).
\end{proof}

\begin{theorem}[Transfer-map gap from injectivity]%
    \label{thm:transfer_map_gap_wielandt}
    \lean{MPSTensor.transfer_map_gap_of_injective}
    \leanok
    \uses{thm:transfer_operator_gap}
    For an injective trace-preserving MPS tensor, there exists
    $\delta>0$ such that every eigenvalue $\mu\ne 1$ of the
    transfer map satisfies $|\mu|\le 1-\delta$.
\end{theorem}

\begin{proof}
    \leanok
    Injectivity implies eventual full Kraus rank. This yields primitivity,
    hence a complementary transfer-map gap. Since the transfer map has
    only finitely many eigenvalues, one may choose a uniform
    $\delta>0$ separating $1$ from the remaining spectrum.
\end{proof}

\section{Relation to parent Hamiltonians}

\begin{remark}\label{rem:correlations_parent_hamiltonian}
    \uses{def:parent_hamiltonian, lem:parent_hamiltonian_annihilates, def:correlation_length, thm:correlator_exponential_bound}
    Suppose that the MPS tensor $A$ generates a parent Hamiltonian $H_N(A,L)$
    (Definition~\ref{def:parent_hamiltonian}) whose ground state is the matrix
    product vector (Lemma~\ref{lem:parent_hamiltonian_annihilates}), and
    consider its fixed-point infinite-chain state. If the hypotheses of
    Theorem~\ref{thm:correlator_exponential_bound} hold, then for every rate
    $r$ with $\rho_{\spec}(\E_A-P)<r<1$ the corresponding ground-state
    correlations decay as $r^n=e^{-n/\xi}$ with finite correlation length
    $\xi=-1/\log r$, the quantity of
    Definition~\ref{def:correlation_length} evaluated at $r$. When $\E_A-P$ is
    diagonalizable, one may take $r=|\lambda_2|$, the largest modulus of a
    subleading eigenvalue, provided $\lambda_2\neq0$.
\end{remark}


\section{Physical correlations in an arbitrary bond basis}

\begin{definition}[Physical connected left-right contraction]
    \label{def:physical_left_right_correlator}
    \lean{MPSTensor.physicalLeftRightConnectedCorrelator}
    \leanok
    \uses{def:physical_observable_transfer, def:transfer_map}
    For matrices $\rho^L,\rho^R\in\MN{D}$ and physical operators $X,Y$
    on independently chosen blocks of lengths $L_1,L_2$, define
    \begin{align}
        C_{\rho^L,\rho^R}^{A}(X,Y;n)
         & =\tr\!\left(\rho^L\E_X\E_A^n(\E_Y(\rho^R))\right)
        -\tr(\rho^L\E_X(\rho^R))\tr(\rho^L\E_Y(\rho^R)).
        \label{eq:correlations_left_right}
    \end{align}
    When $\rho^L,\rho^R$ are positive fixed matrices of the dual transfer
    map and the transfer map, respectively, normalized by
    $\tr(\rho^L\rho^R)=1$, this is the connected infinite-chain
    physical expectation with $n\geq0$ unobserved intermediate sites.
    This is the fixed-point contraction of
    \cite[Section~II.B.3]{Cirac2021Matrix}.
\end{definition}

\begin{theorem}[Gauge covariance of physical insertions]
    \label{thm:physical_insertion_gauge}
    \lean{MPSTensor.physicalObservableTransfer_congruence_of_gauge}
    \leanok
    \uses{def:physical_observable_transfer}
    Suppose $B^i=UA^iU^{-1}$ with $U$ invertible. For every physical
    operator $O$ on $L$ sites and every $Z\in\MN{D}$,
    \begin{align}
        \E_O^B(UZU^\dagger) & =U\E_O^A(Z)U^\dagger.
        \label{eq:correlations_insertion_gauge}
    \end{align}
    This includes $L=0$; no unitarity or normalization is assumed.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:eval_word_gauge, def:physical_observable_transfer}
    Every word satisfies $B^w=UA^wU^{-1}$. Substitute this into the
    finite word sum defining the insertion and cancel $U^{-1}U$ and
    $U^\dagger U^{-\dagger}$.
\end{proof}

\begin{theorem}[Gauge invariance of physical connected correlations]
    \label{thm:physical_correlator_gauge}
    \lean{MPSTensor.physicalLeftRightConnectedCorrelator_eq_of_gauge}
    \leanok
    \uses{def:physical_left_right_correlator}
    Suppose $B^i=UA^iU^{-1}$ with $U$ invertible, and put
    $\widetilde\rho^L=U^{-\dagger}\rho^LU^{-1}$ and
    $\widetilde\rho^R=U\rho^RU^\dagger$. Then, for every pair of
    independently supported physical operators and every $n\geq0$,
    \begin{align}
        C_{\widetilde\rho^L,\widetilde\rho^R}^{B}(X,Y;n)
         & =C_{\rho^L,\rho^R}^{A}(X,Y;n).
        \label{eq:correlations_gauge_invariance}
    \end{align}
    The identity does not require stationarity or positivity of the
    boundary matrices.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:physical_insertion_gauge, lem:ldp_observable_transfer_products,
        def:physical_left_right_correlator}
    Insertion of the identity on $n$ sites is $\E_A^n$, so
    (\ref{eq:correlations_insertion_gauge}) also gives covariance of every
    transfer power. Apply this and
    (\ref{eq:correlations_insertion_gauge}) to the two-point and one-point
    terms. Trace cyclicity gives
    $\tr(U^{-\dagger}\rho^LZU^\dagger)=\tr(\rho^LZ)$ in each term.
\end{proof}

\begin{theorem}[Left-right contraction in trace-preserving gauge]
    \label{thm:physical_left_right_tp}
    \lean{MPSTensor.physicalLeftRightConnectedCorrelator_one}
    \leanok
    \uses{def:physical_left_right_correlator, def:physical_connected_correlator}
    Suppose $\E_A$ is trace preserving, $\E_A(\rho)=\rho$, and
    $\tr\rho=1$. For every pair of independently supported physical
    operators and every $n\geq0$,
    $C_{\Id,\rho}^{A}(X,Y;n)=C_{\mathrm{phys}}(X,Y;n)$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:physical_connected_correlator_identity}
    Multiplication by $\Id$ leaves each contraction unchanged.
    Lemma~\ref{lem:physical_connected_correlator_identity} gives the
    claimed equality at every separation.
\end{proof}

\begin{theorem}[Physical correlation decay for a normal tensor]
    \label{thm:correlator_exponential_bound}
    \lean{MPSTensor.IsNormalTensor.exists_physicalLeftRightCorrelation_decay}
    \leanok
    \uses{def:nt_cpsv, def:physical_left_right_correlator}
    Let $A$ be a normal tensor, with leading transfer eigenvalue normalized
    to one. There are positive definite matrices $\rho^L,\rho^R$ satisfying
    \begin{align}
        \E_A^*(\rho^L)    & =\rho^L,
        \\
        \E_A(\rho^R)      & =\rho^R,
        \\
        \tr(\rho^L\rho^R) & =1,
    \end{align}
    and a rate $0<r<1$ such that every pair $X,Y$ of physical operators
    on independently chosen finite supports admits a constant $C_{XY}>0$
    with
    $|C_{\rho^L,\rho^R}^{A}(X,Y;n)|\leq C_{XY}r^n$ for every $n\geq0$.
    No trace-preserving gauge or diagonalizability hypothesis is imposed.
    This proves the decay conclusion of
    \cite[Section~II.B.3]{Cirac2021Matrix} for normal tensors. The more
    general class whose unique eigenvalue of largest modulus is algebraically
    simple, with possibly singular fixed matrices, is outside this statement.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:ldp_normal_gauge_reduction, thm:physical_correlator_tp_bound,
        thm:physical_correlator_gauge, thm:physical_left_right_tp}
    Choose $B^i=UA^iU^{-1}$ in trace-preserving gauge, with a positive
    definite fixed state $\rho$ of trace one. Set
    $\rho^L=U^\dagger U$ and $\rho^R=U^{-1}\rho U^{-\dagger}$.
    Trace preservation of $\E_B$ implies $\E_A^*(\rho^L)=\rho^L$;
    transfer covariance gives $\E_A(\rho^R)=\rho^R$.
    Congruence preserves positive definiteness, and trace cyclicity gives
    $\tr(\rho^L\rho^R)=\tr\rho=1$.
    Theorem~\ref{thm:physical_correlator_tp_bound} supplies a common
    rate $r<1$ in the gauge of $B$.
    Since its left-right contraction with left matrix $\Id$ equals the
    centered fixed-point correlator, (\ref{eq:correlations_gauge_invariance})
    transports the bound to $A$ at every separation, including $n=0$.
\end{proof}
