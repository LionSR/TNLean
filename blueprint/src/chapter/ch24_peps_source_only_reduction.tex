    A source-only operation is a composition of local contractions,
    register permutations, and normalized pair-source preparations.
    A local vector $x$ is prepared by $y\mapsto x\otimes y$.
    Preparing an ordered list of pair vectors tensored with the input
    produces a vector of norm one whenever every pair vector is a unit vector.

    For an operation $T$ and an additional register memory $R$, retaining
    $R$ means applying $I_R\otimes T$. Exchanging two register blocks
    sends $r\otimes s\otimes y$ to $s\otimes r\otimes y$, and equal register
    lists are identified by the natural memory isometry. These operations
    introduce no additional nonprivate gates, allowing each allowed monomial
    expansion to be replaced by its prepared-source expansion.
\end{definition}

    An original circuit is an ordered composition of contractions on
    owned registers. A private operation acts on a single party. Every
    nonprivate operation has a specified participating set of parties and an
    expansion $G_j=\sum_\xi c_{j,\xi}M_{j,\xi}$ into allowed monomials, with
    registers outside the participating set acting as spectators.
    Let $N$ denote the number of nonprivate gates, and fix $r$ and $S$
    so that each monomial has at most $r$ pair effects and
    $\sum_\xi|c_{j,\xi}|\le S$ at every nonprivate gate.
\end{definition}

\begin{theorem}[Common auxiliary vector and operator error]
    \label{thm:peps_circuit_effect_error}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.referenceEval_apply}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.norm_auxiliaryVector}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.norm_replacement_sub_reference_le}
    \lean{TNLean.Lattice.PartyCircuit.EffectCircuit.isAllowed_replacement}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.eval_produce}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.expandedGateCount_produce}
    \leanok
    \uses{def:peps_original_circuit,cor:peps_pair_effects_rescaling}
    Let $T$ be the operator of an original circuit and let $\delta>0$.
    With $m=\lceil(rS/\delta)^2\rceil+1$, replacing each nonprivate operation
    by its rescaled source-only expansion yields a contraction $\widetilde T$
    and a unit auxiliary vector $\Gamma$ such that
    \begin{align}
        \|\widetilde T-|\Gamma\rangle\otimes T\|\le 2\delta N.
        \label{eq:peps_circuit_effect_error}
    \end{align}
    Each auxiliary register added at a gate acts as a spectator for all
    subsequent operations.
\end{theorem}

\begin{theorem}[Physical density error]
    \label{thm:peps_source_only_density_error}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le_half_of_count_le}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_wholeParty_density_sub_le_half}
    \lean{TNLean.Lattice.PartyCircuit.sourceGateBudget_eq_of_pos}
    \leanok
    \uses{thm:peps_circuit_effect_error}
    Decompose the original output memory into physical and discarded registers.
    For an input $x$ with $\|x\|\le1$, let $\rho$ be the physical reduced
    density operator of $Tx$ and let $\widetilde\rho$ be the physical reduced
    density operator of $\widetilde Tx$ after tracing out the discarded and
    auxiliary registers. Then
    \begin{align}
        \|\widetilde\rho-\rho\|_1\le4\delta N.
        \label{eq:peps_source_only_density_error}
    \end{align}
    In particular, if $N\le M$ and $\varepsilon>0$, setting
    $\delta=\varepsilon/(8\max\{1,M\})$ (which equals $\varepsilon/(8M)$ when
    $M>0$) gives $\|\widetilde\rho-\rho\|_1\le\varepsilon/2$. This applies in
    particular when $x$ is a tensor product of unit vectors, one in each
    party's private memory.
\end{theorem}

\begin{proof}\leanok

    most one, and (\ref{eq:peps_circuit_effect_error}) gives
    $\|u-v\|\le2\delta N$. Under the partial trace $\operatorname{Tr}_E$ over
    the discarded and auxiliary registers,
    \begin{align}
        \|\operatorname{Tr}_E(|u\rangle\langle u|-|v\rangle\langle v|)\|_1
        \le(\|u\|+\|v\|)\|u-v\|\le4\delta N.
        \label{eq:peps_source_only_pure_error}

    $\delta=\varepsilon/(8\max\{1,M\})$ and $N\le M$ gives the $\varepsilon/2$
    bound.
\end{proof}

\begin{theorem}[Regional dimensions without padding]
    \label{thm:peps_family_regional_dimensions}

    \begin{align}
        \dim\mathcal H_A=\prod_{p\in A}d_p\le D^{|A|}

    replacement density within $\varepsilon/2$ of the original regional density
    in trace norm, for the whole-party product input and error parameter
    $\delta$ of Theorem~\ref{thm:peps_source_only_density_error}.
\end{theorem}

\begin{theorem}[Resources of the constructed source-only circuit]
    \label{thm:peps_source_only_resources}
    \lean{TNLean.Lattice.PartyCircuit.sum_norm_replacementCoefficients_le}

    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.participationCount_replacement}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.participationCount_replacement_le}
    \leanok
    \uses{thm:peps_circuit_effect_error,thm:peps_replacement_participants}
    Suppose each original nonprivate gate has at most $K$ monomials,
    absolute coefficient sum at most $S$, and at most $r$ pair effects
    per monomial. Suppose also that each gate involves at most $b$
    parties and each party participates in at most $b$ nonprivate
    gates over the entire circuit.

    With $m=\lceil(rS/\delta)^2\rceil+1$, each replacement gate has at
    most $Km^r$ monomials and absolute coefficient sum at most $S$, with
    \begin{align}
        m    & \le(rS/\delta)^2+2,                 \\
        Km^r & \le K\bigl((rS/\delta)^2+2\bigr)^r.
        \label{eq:peps_replacement_resource_bound}
    \end{align}
    After combining sources on each unordered pair of parties, a gate with $q$
    participating parties has $\binom q2$ pair-source slots (including trivial
    one-dimensional slots), so the replacement circuit has at most
    $N\binom b2$ source slots in total. Each party participates in the same
    number of expanded gates as in the original circuit, at most $b$ in total
    (counted with multiplicity).
\end{theorem}

    participating parties, so summing $\binom q2\le\binom b2$ over the $N$
    nonprivate gates bounds the total number of source slots. Finally, the
    bijection of Theorem~\ref{thm:peps_replacement_participants} restricts to
    the gates incident to any fixed party, preserving its participation count.
\end{proof}
