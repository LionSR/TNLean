    For an ordered register list with spaces $H_1,\ldots,H_n$ (or one private memory per party) and unit vectors $x_j\in H_j$, set $x=x_1\otimes\cdots\otimes x_n$ ($1$ for $n=0$) and define the preparation isometry $\C\to\bigotimes_{j=1}^n H_j$, $z\mapsto zx$. \end{definition}

    A source-only operation is a composition of local contractions, register permutations, and normalized pair-source preparations $y\mapsto x\otimes y$ ($\|x\|=1$). For an operation $T$ and spectator memory $R$, retaining $R$ acts as $I_R\otimes T$, block exchange acts as $r\otimes s\otimes y\mapsto s\otimes r\otimes y$, and equal register lists are identified isometrically; none of these adds nonprivate gates.
\end{definition}

    An original circuit is an ordered composition of private single-party contractions and $N$ nonprivate contractions $G_j=\sum_\xi c_{j,\xi}M_{j,\xi}$, where each $M_{j,\xi}$ is an allowed monomial on a specified participating party set (with outside registers as spectators) containing at most $r$ pair effects, and $\sum_\xi|c_{j,\xi}|\le S$.
\end{definition}

\begin{lemma}[Common auxiliary vector and operator error]
    \label{thm:peps_circuit_effect_error}
    \lean{QuantumCircuit.Party.EffectCircuit.referenceOp_gate,
        QuantumCircuit.Party.EffectCircuit.referenceOp_comp}
    \lean{QuantumCircuit.Party.EffectCircuit.norm_auxiliaryVector}
    \lean{QuantumCircuit.Party.EffectCircuit.norm_replacement_sub_reference_le}
    \lean{QuantumCircuit.Party.EffectCircuit.isAllowed_replacement}
    \lean{QuantumCircuit.Party.OriginalCircuit.op_toEffectCircuit}
    \lean{QuantumCircuit.Party.OriginalCircuit.numGates_toEffectCircuit}
    \leanok
    \uses{def:peps_original_circuit,cor:peps_pair_effects_rescaling}
    Let $T$ be the operator of an original circuit and $\delta>0$. With $m=\lceil(rS/\delta)^2\rceil+1$, replacing each nonprivate gate by its rescaled source-only expansion yields a contraction $\widetilde T$ and a unit auxiliary vector $\Gamma$ (on spectator auxiliary registers) such that
    \begin{align}
        \|\widetilde T-|\Gamma\rangle\otimes T\|\le 2\delta N.
        \label{eq:peps_circuit_effect_error}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    \lean{QuantumCircuit.Party.EffectCircuit.IsExpansionBounded}
    \lean{QuantumCircuit.Party.EffectCircuit.auxiliary}
    \lean{QuantumCircuit.Party.EffectCircuit.auxiliaryVector}
    \lean{QuantumCircuit.Party.PartyCircuit.numGates}
    \lean{QuantumCircuit.Party.EffectCircuit.replacement}
    \lean{QuantumCircuit.Party.SourceCircuit.op_frame}
    \lean{QuantumCircuit.Party.WeightedGate.exists_prepared_effectReplacement_native}
    \lean{QuantumCircuit.Party.WeightedGate.exists_prepared_effectReplacement_uniform}
    \lean{QuantumCircuit.Party.WeightedGate.replacementCoefficients}
    \lean{QuantumCircuit.Party.EffectCircuit.norm_referenceOp_le_one}
    \lean{QuantumCircuit.Party.EffectCircuit.referenceOp}
    \uses{cor:peps_pair_effects_rescaling}
    By Lemma~\ref{cor:peps_pair_effects_rescaling}, each nonprivate gate $G_j$ has a source-only contraction $\widetilde G_j$ and unit auxiliary vector $\Gamma_j$ with $\|\widetilde G_j-|\Gamma_j\rangle\otimes G_j\|\le 2\delta$. Setting $\Gamma=\bigotimes_{j=1}^N\Gamma_j$ and inducting over the circuit composition via $\|B_2B_1-A_2A_1\|\le\|B_2-A_2\|+\|B_1-A_1\|$ for contractions (while private gates and permutations contribute $0$) yields (\ref{eq:peps_circuit_effect_error}).
\end{proof}

\begin{lemma}[Physical density error]
    \label{thm:peps_source_only_density_error}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le_half_of_count_le}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_wholeParty_density_sub_le_half}
    \lean{QuantumCircuit.Party.sourceGateBudget_eq_of_pos}
    \leanok
    \uses{thm:peps_circuit_effect_error}
    Decompose the original output memory into physical and discarded registers. For $\|x\|\le1$, let $\rho=\operatorname{Tr}_{\rm disc}(|Tx\rangle\langle Tx|)$ and let $\widetilde\rho=\operatorname{Tr}_{{\rm disc},\,{\rm aux}}(|\widetilde Tx\rangle\langle\widetilde Tx|)$. Then
    \begin{align}
        \|\widetilde\rho-\rho\|_1\le4\delta N.
        \label{eq:peps_source_only_density_error}
    \end{align}
    In particular, if $N\le M$ and $\varepsilon>0$, setting $\delta=\varepsilon/(8\max\{1,M\})$ gives $\|\widetilde\rho-\rho\|_1\le\varepsilon/2$ (including when $x$ is a whole-party product of unit vectors).
\end{lemma}

\begin{proof}\leanok
    \lean{Matrix.appendAuxiliaryVector}
    \lean{Matrix.norm_appendAuxiliaryVector}
    \lean{Matrix.partialTraceRight_appendAuxiliaryVector}
    \lean{Matrix.rectangularTraceNorm_appendAuxiliaryVector_error_le_two}
    \lean{QuantumCircuit.Party.EffectCircuit.originalPhysicalDensity}
    \lean{QuantumCircuit.Party.EffectCircuit.rectangularTraceNorm_replacement_density_sub_le}
    \lean{QuantumCircuit.Party.EffectCircuit.replacementPhysicalDensity}
    \lean{QuantumCircuit.Party.extendedReadout}
    \lean{QuantumCircuit.Party.toEuclideanLin_extendedReadout_prepMatrix}
    \lean{QuantumCircuit.Party.norm_extendedReadout_le_one}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le_half}
    \lean{QuantumCircuit.Party.sourceGateBudget}
    \lean{QuantumCircuit.Party.sourceGateBudget_pos}
    \lean{QuantumCircuit.Party.sourceGateBudget_spec}
    \uses{thm:peps_circuit_effect_error}
    Set $u=\widetilde T x$ and $v=\Gamma\otimes Tx$, so $\|u\|,\|v\|\le1$ and $\|u-v\|\le2\delta N$ by (\ref{eq:peps_circuit_effect_error}). Since $\operatorname{Tr}_E(|v\rangle\langle v|)=\|\Gamma\|^2\rho=\rho$ under the partial trace $\operatorname{Tr}_E$ over discarded and auxiliary registers,
    \begin{align}
        \|\widetilde\rho-\rho\|_1
        =\|\operatorname{Tr}_E(|u\rangle\langle u|-|v\rangle\langle v|)\|_1
        \le(\|u\|+\|v\|)\|u-v\|\le4\delta N,
        \label{eq:peps_source_only_pure_error}
    \end{align}
    giving (\ref{eq:peps_source_only_density_error}) and the $\varepsilon/2$ bound for $\delta=\varepsilon/(8\max\{1,M\})$.
\end{proof}

\begin{lemma}[Regional dimensions without padding]
    \label{thm:peps_family_regional_dimensions}
    \lean{QuantumCircuit.Party.OriginalCircuit.rectangularTraceNorm_sourceOnly_regional_density_sub_le_half}
    \leanok
    \uses{def:peps_family_physical_output,
        thm:peps_source_only_density_error,
        thm:peps_basis_physical_dimensions,
        thm:peps_basis_regional_readouts}
    If $d_p\le D$ for all $p\in A$, the physical memory $\mathcal H_A$ of $A$ satisfies
    \begin{align}
        \dim\mathcal H_A=\prod_{p\in A}d_p\le D^{|A|},
        \label{eq:peps_family_regional_dimension}
    \end{align}
    and for $N\le M$ and $\delta=\varepsilon/(8\max\{1,M\})$ as in Lemma~\ref{thm:peps_source_only_density_error}, the regional reduced densities on $A$ satisfy $\|\widetilde\rho_A-\rho_A\|_1\le\varepsilon/2$.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_source_only_density_error,
        def:peps_basis_physical_readouts,
        def:peps_basis_region_coordinates,
        thm:peps_basis_physical_dimensions,
        def:peps_family_physical_output,
        thm:peps_basis_regional_readouts}
    Lemma~\ref{thm:peps_basis_physical_dimensions} gives (\ref{eq:peps_family_regional_dimension}), and applying Lemma~\ref{thm:peps_source_only_density_error} to the regional readout decomposing the physical registers into $A$ and $A^c$ (Lemma~\ref{thm:peps_basis_regional_readouts}) yields $\|\widetilde\rho_A-\rho_A\|_1\le\varepsilon/2$.
\end{proof}

\begin{lemma}[Resources of the constructed source-only circuit]
    \label{thm:peps_source_only_resources}
    \lean{QuantumCircuit.Party.WeightedGate.sum_norm_replacementCoefficients_le}
    \lean{QuantumCircuit.Party.WeightedGate.card_replacementLabels_le}
    \lean{QuantumCircuit.Party.WeightedGate.stackLength_le}
    \lean{QuantumCircuit.Party.WeightedGate.card_replacementLabels_stackLength_le}
    \lean{QuantumCircuit.Party.OriginalCircuit.isExpansionBounded_replacement}
    \lean{QuantumCircuit.Party.PreparedSourceGate.length_sources_branchWord_eq_choose}
    \lean{QuantumCircuit.Party.SourceCircuit.slotCount_eq_choose}
    \lean{QuantumCircuit.Party.OriginalCircuit.card_sourceLocations_replacement_le}
    \lean{QuantumCircuit.Party.OriginalCircuit.participationCount_replacement}
    \lean{QuantumCircuit.Party.OriginalCircuit.participationCount_replacement_le}
    \leanok
    \uses{thm:peps_circuit_effect_error,thm:peps_replacement_participants}
    Suppose each original nonprivate gate has at most $K$ monomials, absolute coefficient sum at most $S$, at most $r$ pair effects per monomial, and at most $b$ participating parties, and each party participates in at most $b$ nonprivate gates. With $m=\lceil(rS/\delta)^2\rceil+1$, each replacement gate has absolute coefficient sum at most $S$ and at most $Km^r$ monomials, where
    \begin{align}
        m    & \le(rS/\delta)^2+2,                 \\
        Km^r & \le K\bigl((rS/\delta)^2+2\bigr)^r.
        \label{eq:peps_replacement_resource_bound}
    \end{align}
    Combining sources per unordered party pair gives $\binom q2\le\binom b2$ pair-source slots at a $q$-party gate, at most $N\binom b2$ source slots in total, and at most $b$ expanded gates incident to any party.
\end{lemma}

    \lean{QuantumCircuit.Party.OriginalCircuit.natCard_nonprivateLocations}
    \lean{QuantumCircuit.Party.OriginalCircuit.participation}

    \lean{QuantumCircuit.Party.PartyCircuit.participationCount}
    \lean{QuantumCircuit.Party.SourceInventory.length_eq_choose}
    \lean{QuantumCircuit.Party.PreparedSourceGate.length_sources_branchWord}
    \lean{QuantumCircuit.Party.PreparedSourceGate.length_sources_branchWord_le_choose}

    \uses{cor:peps_pair_effects_rescaling,thm:peps_replacement_participants}
    By Lemma~\ref{cor:peps_pair_effects_rescaling}, cyclic insertion averaging and rescaling by $(1+\delta)^{-1}<1$ keep the coefficient $\ell_1$-norm at most $S$ and expand each of the $K$ monomials into $m^r$ terms, where $\lceil x\rceil+1\le x+2$ gives (\ref{eq:peps_replacement_resource_bound}). Summing $\binom q2\le\binom b2$ over the $N$ gates bounds the total source slots by $N\binom b2$, and Lemma~\ref{thm:peps_replacement_participants} preserves each party's gate participation count.
\end{proof}
