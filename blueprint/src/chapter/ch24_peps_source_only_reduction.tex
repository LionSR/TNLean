\begin{theorem}[Physical density error]
    \label{thm:peps_source_only_density_error}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_density_sub_le_half_of_count_le}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.rectangularTraceNorm_sourceOnly_wholeParty_density_sub_le_half}
    \lean{TNLean.Lattice.PartyCircuit.sourceGateBudget_eq_of_pos}
    \leanok
    \uses{thm:peps_circuit_effect_error}
    Decompose the original output memory into physical and discarded registers. For $\|x\|\le1$, let $\rho=\operatorname{Tr}_{\rm disc}(|Tx\rangle\langle Tx|)$ and let $\widetilde\rho=\operatorname{Tr}_{{\rm disc},\,{\rm aux}}(|\widetilde Tx\rangle\langle\widetilde Tx|)$. Then
    \begin{align}
        \|\widetilde\rho-\rho\|_1\le4\delta N.
        \label{eq:peps_source_only_density_error}
    \end{align}
    In particular, if $N\le M$ and $\varepsilon>0$, setting $\delta=\varepsilon/(8\max\{1,M\})$ gives $\|\widetilde\rho-\rho\|_1\le\varepsilon/2$ (including when $x$ is a whole-party product of unit vectors).
\end{theorem}

\begin{theorem}[Resources of the constructed source-only circuit]

With $m=\lceil(rS/\delta)^2\rceil+1$, each replacement gate has absolute coefficient sum at most $S$ and at most $Km^r$ monomials, where
    \begin{align}
        m    & \le(rS/\delta)^2+2,                 \\
        Km^r & \le K\bigl((rS/\delta)^2+2\bigr)^r.
        \label{eq:peps_replacement_resource_bound}
    \end{align}
    Combining sources per unordered party pair gives $\binom q2\le\binom b2$ pair-source slots at a $q$-party gate, at most $N\binom b2$ source slots in total, and at most $b$ expanded gates incident to any party.
