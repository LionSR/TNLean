\begin{lemma}[Horizontal sections of sampled polygons]
    \label{lem:area_law_sample_rows}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.latticeRow,
        TNLean.PEPS.AreaLaw.Geometry.mem_latticeRow,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.convex_region,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.isCompact_region,
        TNLean.PEPS.AreaLaw.Geometry.horizontal_mem_of_mem_of_le,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.isCompact_horizontalSection,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.horizontalSection_eq_Icc,
        TNLean.PEPS.AreaLaw.Geometry.Template.latticeRow_eq_Icc_ceil_floor,
        TNLean.PEPS.AreaLaw.Geometry.Template.mem_sample_of_row_between,
        TNLean.PEPS.AreaLaw.Geometry.Template.latticeRow_eq_Icc,
        TNLean.PEPS.AreaLaw.Geometry.ceil_affine_row,
        TNLean.PEPS.AreaLaw.Geometry.floor_affine_row
    }
    \leanok
    \uses{def:area_law_template}
    Let $P$ be one of the real polygons of a template and put
    $S=P\cap\mathbb Z^2$. For $y\in\mathbb Z$, write
    $R_y(S)=\{x\in\mathbb Z:(x,y)\in S\}$.
    Every nonempty real horizontal section of $P$ is a closed interval
    $[a,b]$, and
    \[
        R_y(S)=[\lceil a\rceil,\lfloor b\rfloor]\cap\mathbb Z.
    \]
    This integer interval may be empty. If it is nonempty, its endpoints
    are its attained minimum and maximum. For every real $c$ and integers
    $m,y$, rounding commutes with the integer shift:
    \[
        \lceil c+my\rceil=\lceil c\rceil+my,
        \qquad
        \lfloor c+my\rfloor=\lfloor c\rfloor+my.
    \]
\end{lemma}
\begin{proof}
    \leanok
    A finite convex hull is compact and convex. Its intersection with a
    horizontal line is compact, and convexity fills the interval between
    its attained extrema. Exact lattice sampling then rounds the endpoints.
    Translation by an integer commutes with both floor and ceiling.
    This is the initial row argument in the proof of Lemma~9.4 of
    \cite{OpenAI2026AreaLaw}.
\end{proof}

\begin{lemma}[Area of a template]
    \label{lem:area_law_template_area}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.Template.sample_subset_box,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_sample_le,
        TNLean.PEPS.AreaLaw.Geometry.template_card_le
    }
    \leanok
    \uses{def:area_law_template}
    If $C_{\rm tpl}\ge1$, a template satisfies $|T|\le9ns_0$.
\end{lemma}
\begin{proof}
    \leanok
    Each nonempty sample lies in an integer square of radius $s_0$ about
    any one of its sites, so its cardinality is at most $(2s_0+1)^2$.
    Empty samples contribute zero. Sum over the pieces and use
    $(2s_0+1)^2\le9s_0(s_0+1)$ and $N_0(s_0+1)\le n$.
\end{proof}

\begin{lemma}[Rows and unions under ambient dilation]
    \label{lem:area_law_dilation_rows_union}
    \lean{
        TNLean.PEPS.AreaLaw.mem_ambientDilation_iff,
        TNLean.PEPS.AreaLaw.ambientDilation_mono,
        TNLean.PEPS.AreaLaw.ambientDilation_mono_radius,
        TNLean.PEPS.AreaLaw.ambientDilation_add,
        TNLean.PEPS.AreaLaw.ambientDilation_biUnion_sdiff_subset,
        TNLean.PEPS.AreaLaw.card_ambientDilation_biUnion_sdiff_le,
        TNLean.PEPS.AreaLaw.Geometry.mem_latticeRow_ambientDilation,
        TNLean.PEPS.AreaLaw.Geometry.Template.mem_latticeRow_dilation_iff
    }
    \leanok
    \uses{def:area_law_template,lem:area_law_sample_rows}
    For a finite $S\subset\mathbb Z^2$, let
    $S_r=S+([-r,r]^2\cap\mathbb Z^2)$. Membership in $S_r$ means that
    both coordinate differences from some point of $S$ are at most $r$
    in absolute value. Dilation is monotone in both $S$ and $r$, satisfies
    $(S_r)_t=S_{r+t}$, and
    \[
        R_y(S_r)=
        \bigcup_{z=y-r}^{y+r}\ \bigcup_{a\in R_z(S)}
        ([a-r,a+r]\cap\mathbb Z).
    \]
    For a sampled polygon, each nonempty row in this union contributes the
    interval from its minimum minus $r$ to its maximum plus $r$.
    If $S=\bigcup_{i\in I}S_i$ is a finite union, then, for $r,t\ge0$,
    \[
        S_r\setminus S_t
        \subseteq\bigcup_{i\in I}((S_i)_r\setminus(S_i)_t),
        \qquad
        |S_r\setminus S_t|
        \le\sum_{i\in I}|(S_i)_r\setminus(S_i)_t|.
    \]
    The pieces may be empty, overlapping, or disconnected.
\end{lemma}
\begin{proof}
    \leanok
    The coordinate inequalities give the row formula. A point newly present
    in the dilation of a union has a witness in one constituent, and cannot
    belong to the old dilation of that constituent. Counting the union gives
    the cardinality inequality without requiring disjointness.
\end{proof}

The one-step endpoint bounds for the allowed slopes, their preservation
under dilation, and the estimate $|T_j\setminus T_{j-1}|\le n$ in
Lemma~9.4 of \cite{OpenAI2026AreaLaw} remain to be established here.
