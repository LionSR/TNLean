\begin{lemma}[Horizontal sections of sampled polygons]
    \label{thm:area_law_sample_rows}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.latticeRow,
        TNLean.PEPS.AreaLaw.Geometry.mem_latticeRow,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.convex_region,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.isCompact_region,
        TNLean.PEPS.AreaLaw.Geometry.horizontal_mem_of_mem_of_le,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.isCompact_horizontalSection,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.horizontalSection_eq_Icc,
        TNLean.PEPS.AreaLaw.Geometry.Template.latticeRow_eq_Icc_ceil_floor,
        TNLean.PEPS.AreaLaw.Geometry.Template.mem_sample_of_row_between,
        TNLean.PEPS.AreaLaw.Geometry.Template.latticeRow_eq_Icc,
        TNLean.PEPS.AreaLaw.Geometry.ceil_affine_row,
        TNLean.PEPS.AreaLaw.Geometry.floor_affine_row
    }
    \leanok
    \uses{def:area_law_template}
    For a template polygon $P$, set $S=P\cap\mathbb Z^2$ and $R_y(S)=\{x\in\mathbb Z:(x,y)\in S\}$ ($y\in\mathbb Z$). Every nonempty horizontal section of $P$ is a compact interval $[a,b]$ with
    \begin{align}
        R_y(S) & =[\lceil a\rceil,\lfloor b\rfloor]\cap\mathbb Z,
    \end{align}
    attaining its extrema when nonempty, and for all $c\in\mathbb R$ and $m,y\in\mathbb Z$,
    \begin{align}
        \lceil c+my\rceil   & =\lceil c\rceil+my,   \\
        \lfloor c+my\rfloor & =\lfloor c\rfloor+my.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Since $P$ is compact and convex, its horizontal section at $y$ is a compact interval $[a,b]$; intersecting with $\mathbb Z$ gives $[\lceil a\rceil,\lfloor b\rfloor]\cap\mathbb Z$, and integer shifts commute with $\lceil\cdot\rceil$ and $\lfloor\cdot\rfloor$ (\cite[Lemma~9.4]{OpenAI2026AreaLaw}).
\end{proof}

\begin{lemma}[Area of a template]
    \label{thm:area_law_template_area}
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.Template.sample_subset_box,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_sample_le,
        TNLean.PEPS.AreaLaw.Geometry.template_card_le
    }
    \leanok
    \uses{def:area_law_template}
    If $C_{\rm tpl}\ge1$, a template $T$ satisfies $|T|\le9ns_0$.
\end{lemma}
\begin{proof}
    \leanok
    Each nonempty piece sample lies in a $\ell_\infty$-ball of radius $s_0$ and thus has cardinality at most $(2s_0+1)^2\le9s_0(s_0+1)$; summing over the $N_0$ pieces with $N_0(s_0+1)\le C_{\rm tpl}N_0(s_0+1)\le n$ gives $|T|\le9ns_0$.
\end{proof}

\begin{lemma}[Rows and unions under ambient dilation]
    \label{thm:area_law_dilation_rows_union}
    \lean{
        TNLean.Lattice.mem_ambientDilation_iff,
        TNLean.PEPS.AreaLaw.ambientDilation_mono,
        TNLean.PEPS.AreaLaw.ambientDilation_mono_radius,
        TNLean.PEPS.AreaLaw.ambientDilation_add,
        TNLean.PEPS.AreaLaw.ambientDilation_biUnion_sdiff_subset,
        TNLean.PEPS.AreaLaw.card_ambientDilation_biUnion_sdiff_le,
        TNLean.PEPS.AreaLaw.Geometry.mem_latticeRow_ambientDilation,
        TNLean.PEPS.AreaLaw.Geometry.Template.mem_latticeRow_dilation_iff
    }
    \leanok
    \uses{def:area_law_template,thm:area_law_sample_rows}
    For finite $S\subset\mathbb Z^2$, set $S_r=S+([-r,r]^2\cap\mathbb Z^2)$. Ambient dilation is monotone in $(S,r)$, satisfies $(S_r)_t=S_{r+t}$, and has rows
    \begin{align}
        R_y(S_r) & =
        \bigcup_{z=y-r}^{y+r}\ \bigcup_{a\in R_z(S)}
        ([a-r,a+r]\cap\mathbb Z).
    \end{align}
    For any finite union $S=\bigcup_{i\in I}S_i$ and $r,t\ge0$,
    \begin{align}
        S_r\setminus S_t   & \subseteq\bigcup_{i\in I}((S_i)_r\setminus(S_i)_t), \\
        |S_r\setminus S_t| & \le\sum_{i\in I}|(S_i)_r\setminus(S_i)_t|.
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Decomposing $[-r,r]^2\cap\mathbb Z^2$ by vertical and horizontal coordinates gives the formula for $R_y(S_r)$, and for $S=\bigcup_{i\in I}S_i$, any $x\in S_r\setminus S_t$ lies in $(S_i)_r\setminus(S_i)_t$ for some $i\in I$, yielding the inclusion and subadditivity bound.
\end{proof}

\begin{lemma}[Supporting strips of allowed polygons]
    \label{thm:area_law_polygon_strips}
    \leanok
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.vertices,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.region_eq_convexHull_vertices,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.mem_region_iff_normal_bounds,
        TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.exists_four_strip_bounds
    }
    \uses{def:area_law_template}
    Every allowed polygon equals the intersection of its supporting strips for the four linear forms $x,y,x+y,x-y$ (equivalently, all closed supporting slabs with allowed normal directions).
\end{lemma}
\begin{proof}
    \leanok
    Each supporting slab contains the convex hull of the vertices; conversely, normalizing the side functionals of a triangle (by its nonzero signed determinant) or a rectangle gives nonnegative barycentric or product-interval coordinates whenever the four strip bounds for $x,y,x+y,x-y$ hold.
\end{proof}

\begin{lemma}[Finite integer points in allowed polygons]
    \label{thm:area_law_polygon_integer_points_finite}
    \lean{TNLean.PEPS.AreaLaw.Geometry.TemplatePolygon.finite_integerPoints}
    \leanok
    \uses{def:area_law_template}
    For any closed nondegenerate triangle or rectangle $P\subseteq\mathbb R^2$ with side slopes in $\{0,\infty,1,-1\}$, the set $P\cap\mathbb Z^2$ is finite.
\end{lemma}

\begin{proof}
    \leanok
    \uses{thm:area_law_polygon_strips}
    By Lemma~\ref{thm:area_law_polygon_strips}, $P\subseteq[l_x,u_x]\times[l_y,u_y]$ for some $l_x,u_x,l_y,u_y\in\mathbb R$, so
    \begin{align}
        P\cap\mathbb Z^2 & \subseteq
        ([\lceil l_x\rceil,\lfloor u_x\rfloor]\cap\mathbb Z)
        \times([\lceil l_y\rceil,\lfloor u_y\rfloor]\cap\mathbb Z),\notag
    \end{align}
    which is a subset of a finite Cartesian product.
\end{proof}

\begin{lemma}[A template for the integer points of a fan run]
    \label{thm:area_law_fan_run_template}
    \lean{TNLean.PEPS.AreaLaw.Geometry.CellFanRun.exists_template}
    \leanok
    \uses{def:area_law_template,def:area_law_dyadic_cells,def:area_law_cell_fan,
        def:area_law_fan_runs}
    Let $C_{\mathrm{tpl}}\ge1$ and $C_B\ge16C_{\mathrm{tpl}}$. Fix $o\in\mathbb R^2$, $\ell\in\mathbb N$, $z\in\mathbb Z^2$, and midpoint subdivisions and a two-coloring of the fan triangles of the closed dyadic square of side $u=2^\ell$. For any run $R$ with $T_R\cap\mathbb Z^2\ne\varnothing$, there exists a template $T$ with $s_0=2^\ell$, $n=\lceil C_Bu\rceil$, and at most $8$ polygonal pieces such that $T=T_R\cap\mathbb Z^2$.
\end{lemma}

\begin{proof}
    \leanok
    \uses{def:area_law_template,def:area_law_fan_runs,
        thm:area_law_cell_fan_count,thm:area_law_cell_fan_cover,
        thm:area_law_fan_run_cover,
        thm:area_law_polygon_integer_points_finite,
        thm:area_law_dyadic_cell_distance}
    By Lemmas~\ref{thm:area_law_cell_fan_count}, \ref{thm:area_law_fan_run_cover}, and~\ref{thm:area_law_polygon_integer_points_finite}, the $m\in\{1,\ldots,8\}$ fan triangles $P_e$ ($e\in R$) have finite samples $P_e\cap\mathbb Z^2$ with union $T_R\cap\mathbb Z^2$.     \begin{align}
        C_{\mathrm{tpl}}m(s_0+1) & \le8C_{\mathrm{tpl}}(u+1)
        \le16C_{\mathrm{tpl}}u\le C_Bu\le n.\notag
    \end{align}
\end{proof}

\begin{lemma}[Rounded affine row profiles]
    \label{thm:area_law_affine_row_profiles}
    \leanok
    \lean{
        TNLean.PEPS.AreaLaw.Geometry.Template.exists_sample_four_strip_bounds,
        TNLean.PEPS.AreaLaw.Geometry.Template.exists_latticeRow_profile,

    For each sampled polygon $S$ there exist $l_x,u_x,l_y,u_y,l_+,u_+,l_-,u_-\in\mathbb Z$ such that $R_y(S)=\varnothing$ for $y\notin[l_y,u_y]$ and, for $y\in[l_y,u_y]$,
    \begin{align}
        R_y(S) & =
        [\max(l_x,l_+-y,l_-+y),\min(u_x,u_+-y,u_-+y)]\cap\mathbb Z.
    \end{align}
    The $y\in\mathbb Z$ with $R_y(S)\ne\varnothing$ form an integer interval on which $\min R_y(S)$ and $\max R_y(S)$ are $1$-Lipschitz in $y$.
\end{lemma}
\begin{proof}
    \leanok
    Applying Lemma~\ref{thm:area_law_sample_rows} to the four strip bounds of Lemma~\ref{thm:area_law_polygon_strips} rounds the lower and upper bounds to $\max(l_x,l_+-y,l_-+y)$ and $\min(u_x,u_+-y,u_-+y)$, which are $1$-Lipschitz in $y$ and satisfy the lower$\le$upper condition on an interval of $y\in\mathbb Z$.
\end{proof}

\begin{lemma}[Single-piece ambient dilation layers]
    \label{thm:area_law_piece_depth_layers}
    \leanok
    \lean{

        TNLean.PEPS.AreaLaw.Geometry.Template.latticeRow_dilation_eq_Icc,

    }
    \uses{thm:area_law_affine_row_profiles,thm:area_law_dilation_rows_union,
        thm:area_law_template_area}
    For any sampled piece $S$, each $R_y(S_r)$ is an integer interval of length at most $2s_0+2r+1$ with $|R_y(S_{r+1})\setminus R_y(S_r)|\le4$ whenever $R_y(S_r)\ne\varnothing$. When $S\ne\varnothing$, the occupied rows of $S_r$ form $[a-r,b+r]\cap\mathbb Z$ for some $a\le b$ with $b-a\le2s_0$, and for every $1\le j\le s_0$ (including $S=\varnothing$),
    \begin{align}
        |S_j\setminus S_{j-1}| & \le24(s_0+1).
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    By Lemmas~\ref{thm:area_law_dilation_rows_union} and~\ref{thm:area_law_affine_row_profiles}, $R_y(S_r)$ is an interval of width at most $2s_0+2r+1$, and incrementing $r=j-1\mapsto j$ shifts each row endpoint by at most $2$ (adding at most $4$ sites on each of the at most $2s_0+2j-1$ old rows) while adding two extreme rows of width at most $2s_0+2j+1$, giving $|S_j\setminus S_{j-1}|\le 4(2s_0+2j-1)+2(2s_0+2j+1)=12s_0+12j-2\le24(s_0+1)$.
\end{proof}
