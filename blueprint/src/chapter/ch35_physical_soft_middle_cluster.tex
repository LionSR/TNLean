\section{A native soft middle with its closed spectral cluster}

The scalar parameters in this construction are selected before the chain.
The regional minima precede the estimated centers, and the native operator
approximation precedes the choice of an original ground spectral value.
The construction supports \cite[Proposition~5.7 and Theorem~5.8]{Arad2016RigorousRG},
with the corrected estimates of
\cite{gap:alvv16_cluster_approximation,gap:alvv16_soft_regional_residuals}.

\begin{lemma}[A constructed middle approximation and its own low cluster]
    \label{lem:physical_soft_middle_cluster}
    \lean{TNLean.MPS.AreaLaw.ALVV.exists_physical_soft_middle_cluster}
    \leanok
    Let $g>0$ and $0<\tau\le1$. One can choose $m\ge2$, $t=3m$ and
    a common cutoff $T$ before every chain and estimated center. They
    satisfy $t,T\ge156+2g$, $T\ge g/40+24$, and
    \begin{align}
        3\left(32e^{-(T-g/40-24)/8}+\frac{t}{m}(T/t)^m\right)
         & \le\tau g/10^6,\notag \\
        3\left(32e^{-(T-156-2g)/8}+\frac{t}{m}(T/t)^m\right)
         & \le\tau g/10^6.
        \label{eq:physical_soft_middle_parameters}
    \end{align}

    Consider $N+1$ sites of dimension $q\ge2$, with positive-contraction
    nearest-neighbour terms $h_i$. Let $\ell\ge1$ and suppose that
    $a+2\ell\le b$ and $b+2\ell\le N$. Use the five site regions with
    endpoints $0,a+1,a+2\ell,b+1,b+2\ell,N+1$. Let $A_j$ be their
    internal Hamiltonians and $Y$ the crossing terms. There are actual
    minima $e_j$, one common unit vector $x$, and $E\in[0,4]$, with
    $A_jx=e_jx$ and $A_j-e_jI\ge0$, chosen before all following centers.

    Put $S=\{0,2,4\}$. For any centers $e_j-10\le p_j\le e_j$ on $S$, define
    \begin{align}
        F_j & =p_jI+h_{m,t}(A_j-p_jI),\notag                                   \\
        G   & =\sum_{j=0}^4
        \bigl(\mathbf1_{j\in S}F_j+\mathbf1_{j\notin S}A_j-e_jI\bigr)+Y,\notag \\
        M   & =3t(1+\log m)+4\ell+34,\qquad
        \xi=\min\left\{1,\frac{\tau g^2}{20000(M+g+8)}\right\}.
        \label{eq:physical_soft_middle_precision}
    \end{align}
    Then $G$ is Hermitian, $\|G\|\le M$, $0<\xi\le1$, and $p_2I\le A_2$.
    Set
    \begin{align}
        c_* & =2(e^{1/3}-1),\qquad W=t2^{m+1},\notag  \\
        r_* & =\max\left\{1,
        \left\lceil\frac{\log\max\{1,N^2\}-\log\log(1+\xi/W)}
        {-\log c_*}\right\rceil\right\},\notag        \\
        B   & =2\bigl(1+Nq^2+(m+1)r_*^2q^{r_*}\bigr).
        \label{eq:physical_soft_middle_bond}
    \end{align}
    There are native operator tensors on $[a+2\ell,b+1)$ with bond $B$
    and a closing matrix, representing a Hermitian matrix $Z$.
    Write $Z_{\rm phys}$ for its
    placement on the full chain and $(F_2)_{\rm nat}$ for the principal
    slice with zero exterior configuration. They satisfy
    \begin{align}
        F_2        & =\iota\bigl((F_2)_{\rm nat}\bigr),\qquad
        \|(F_2)_{\rm nat}-Z\|\le\xi,\qquad
        \|F_2-Z_{\rm phys}\|\le\xi,\notag                     \\
        \widehat G & =G-F_2+Z_{\rm phys},\qquad
        \|G-\widehat G\|\le\xi,\qquad
        \|\widehat G\|\le M+\xi.
        \label{eq:physical_soft_middle_error}
    \end{align}
    Both $Z_{\rm phys}$ and $\widehat G$ are Hermitian.

    These tensors are chosen before every $\mu_0\in\operatorname{spec}(H_0)$
    satisfying $\operatorname{spec}(H_0)\subseteq\{\mu_0\}\cup[\mu_0+g,\infty)$,
    where $H_0=\sum_i h_i$. For every such value,
    $E+\sum_j e_j=\mu_0$. If $G_0$ is the full ground space of $H_0$, then
    \begin{align}
        \widehat T
                       & =\mathbf1_{(-\infty,E+g/4]}(\widehat G)\mathcal H
        =\mathbf1_{(-\infty,E+3g/4]}(\widehat G)\mathcal H,\notag          \\
        \dim\widehat T & =\dim G_0.
        \label{eq:physical_soft_middle_cluster}
    \end{align}
    Each of $\widehat T,G_0$ is $3\tau/200$-close to the other.
    The operator tensors may depend on the centers; no assertion of
    independence from those centers or of running time is made.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:alvv_five_soft_uniform_clusters,lem:alvv_soft_actual_window_losses,
        lem:native_soft_middle_mpo}
    Choose $T$ to control the exponential residual and then $m$ to
    control the power residual, with $t=3m$. Apply
    Lemma~\ref{lem:alvv_five_soft_uniform_clusters} once. The minimum
    certificate and $p_2\le e_2$ imply $p_2I\le A_2$.
    Lemma~\ref{lem:alvv_soft_actual_window_losses} gives $0<\xi\le1$.
    Apply Lemma~\ref{lem:native_soft_middle_mpo} once at this precision,
    retaining its native tensors, closing matrix and both errors.

    Since $G-\widehat G=F_2-Z_{\rm phys}$, the native placement error gives
    $\|G-\widehat G\|\le\xi$; the triangle inequality gives
    $\|\widehat G\|\le M+\xi$. The arbitrary Hermitian approximation
    conclusion of Lemma~\ref{lem:alvv_five_soft_uniform_clusters}, applied
    to this very $\widehat G$, yields~(\ref{eq:physical_soft_middle_cluster})
    and both directed errors.
\end{proof}
