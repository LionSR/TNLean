\subsection{Staircase patch geometry}
\label{subsec:peps_torus_staircase_patch_geometry}

\begin{definition}[Horizontal staircase patch and its corner]
    \label{def:peps_horizontal_staircase_patch}
    \lean{TNLean.PEPS.horizontalStaircasePatch}
    \lean{TNLean.PEPS.horizontalStaircaseLeftWindow}
    \lean{TNLean.PEPS.horizontalStaircaseRightWindow}
    \lean{TNLean.PEPS.horizontalStaircaseEndPair}
    \lean{TNLean.PEPS.horizontalStaircaseCorner}
    \lean{TNLean.PEPS.horizontalStaircaseCompletedCorner}
    \leanok
    \uses{def:peps_torus_geometry_translation}
    For the cyclic rectangles used in the staircase chaining and
    corner-stripping step of
    \cite[proof sketch, lines~2320--2445]{Molnar2018NormalPEPS}, fix a
    horizontal reference edge and write the staircase corner as
    $s=(a,b)$ and $I(u,v)=\{t\mid u\le t<v\}$. The two end windows, their end
    pair $S$, the staircase patch $P$, the corner $C$, and the completed
    $L\times K$ corner $\widehat C$ are
    \begin{align}
        W_{\mathrm{left}}
         & =I(a,a+L)\times I(b,b+K),
         &
        W_{\mathrm{right}}
         & =I(a+L,a+2L)\times I(b+K-1,b+2K-1),
        \notag \\
        S & =W_{\mathrm{left}}\cup W_{\mathrm{right}},
         &
        C & =I(a,a+L)\times I(b+K,b+2K-1),
        \notag \\
        \widehat C & =I(a,a+L)\times I(b+K,b+2K),
         &
        P & =(I(a,a+2L)\times I(b+K-1,b+2K-1))\cup(I(a,a+L)\times I(b,b+2K-1)).
        \notag
    \end{align}
\end{definition}

\begin{lemma}[The staircase end windows are disjoint]
    \label{lem:peps_horizontal_staircase_end_pair_disjoint}
    \lean{TNLean.PEPS.horizontalStaircaseEndPair_disjoint}
    \leanok
    \uses{def:peps_horizontal_staircase_patch}
    If $2L\le W$, then $W_{\mathrm{left}}\cap W_{\mathrm{right}}=\varnothing$.
\end{lemma}
\begin{proof}
    \leanok
    For $2L\le W$, the horizontal intervals $I(a,a+L)$ and $I(a+L,a+2L)$ are
    disjoint modulo $W$.
\end{proof}

\begin{lemma}[Injectivity of the staircase end windows and completed corner]
    \label{lem:peps_horizontal_staircase_window_injectivity}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.horizontalStaircaseLeftWindow_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.horizontalStaircaseRightWindow_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.horizontalStaircaseCompletedCorner_injective}
    \leanok
    \uses{def:peps_horizontal_staircase_patch}
    Under the $L\times K$ torus-window injectivity hypotheses,
    $\inj(W_{\mathrm{left}})$, $\inj(W_{\mathrm{right}})$, and
    $\inj(\widehat C)$.
\end{lemma}
\begin{proof}
    \leanok
    By Definition~\ref{def:peps_horizontal_staircase_patch}, each of
    $W_{\mathrm{left}}$, $W_{\mathrm{right}}$, and $\widehat C$ is an
    $L\times K$ cyclic rectangle.
\end{proof}

\begin{lemma}[Patch decomposition by the end pair and the corner]
    \label{lem:peps_horizontal_staircase_patch_decomposition}
    \lean{TNLean.PEPS.horizontalStaircaseCorner_disjoint_endPair}
    \lean{TNLean.PEPS.horizontalStaircasePatch_eq_endPair_union_corner}
    \lean{TNLean.PEPS.horizontalStaircasePatch_sdiff_endPair}
    \leanok
    \uses{def:peps_horizontal_staircase_patch, lem:peps_horizontal_staircase_end_pair_disjoint}
    If $L,K>0$, $2L\le W$, and $2K\le H$, then $C\cap S=\varnothing$,
    $P=S\cup C$, and $P\setminus S=C$.
\end{lemma}
\begin{proof}
    \leanok
    Under $2L\le W$ and $2K\le H$, expanding the coordinate intervals of
    Definition~\ref{def:peps_horizontal_staircase_patch} shows
    $C\cap W_{\mathrm{left}}=C\cap W_{\mathrm{right}}=\varnothing$ and
    $P=W_{\mathrm{left}}\cup W_{\mathrm{right}}\cup C=S\cup C$.
\end{proof}

\begin{definition}[Torus rectangular injectivity hypotheses]%
    \label{def:peps_torusRectangleInjectivityHypotheses}
    \lean{TNLean.PEPS.NormalTorusRectangleInjectivityHypotheses}
    \leanok
    \uses{def:peps_regionInjectivityData, def:peps_torus_geometry_translation}
    On the discrete $n\times m$ torus
    (Definition~\ref{def:peps_torus_geometry_translation}), the \emph{torus
    rectangular injectivity hypotheses} for $\inj$
    (Definition~\ref{def:peps_regionInjectivityData}) assert $\inj(R)$ for
    every contiguous $2\times 3$ and $3\times 2$ coordinate rectangle $R$.
\end{definition}

\begin{definition}[Normal torus tensor with injective rectangles]%
    \label{def:peps_normalTorusRectanglePEPS}
    \lean{TNLean.PEPS.IsNormalTorusRectanglePEPS}
    \leanok
    \uses{def:peps_torusRectangleInjectivityHypotheses, def:peps_regionInjectivityDataOf}
    A PEPS tensor $A$ on the discrete torus is \emph{normal with injective
    rectangles} if $\inj_A$ (Definition~\ref{def:peps_regionInjectivityDataOf})
    satisfies Definition~\ref{def:peps_torusRectangleInjectivityHypotheses} and
    $D_A(e)>0$ for every edge $e$ \cite[Theorem~3]{Molnar2018NormalPEPS}
    (with bond positivity recorded explicitly
    \cite{gap:peps_injective_ft_section3_route}).
\end{definition}

\begin{lemma}[Torus reference-edge blocking data]%
    \label{lem:peps_torusRectangleReferenceBlockingData}
    \lean{TNLean.PEPS.torusHorizontalReferenceEdge}
    \lean{TNLean.PEPS.torusVerticalReferenceEdge}
    \lean{TNLean.PEPS.torusHorizontalRectangleBlockingDatum}
    \lean{TNLean.PEPS.torusVerticalRectangleBlockingDatum}
    \lean{TNLean.PEPS.isCrossingEdge_torusHorizontalRectangleBlockingDatum}
    \lean{TNLean.PEPS.isCrossingEdge_torusVerticalRectangleBlockingDatum}
    \leanok
    \uses{def:peps_torusRectangleInjectivityHypotheses, def:peps_regionInjectivityData, def:peps_torus_geometry_translation}
    At a frame origin $(x_0,y_0)$ with $2\le x_0$, $1\le y_0$ (or $1\le x_0$,
    $2\le y_0$ vertically) and either $x_0+5=n$, $y_0+5=m$ or $x_0+7\le n$,
    $y_0+7\le m$, let $e_h=e_{\mathrm R}(x_0+1,y_0+2)$ and
    $e_v=e_{\mathrm U}(x_0+2,y_0+1)$. Under
    Definition~\ref{def:peps_torusRectangleInjectivityHypotheses} and union
    closure (Definition~\ref{def:peps_regionInjectivityData}), the frames at
    $e_h$ and $e_v$ yield one-edge blocking data $(R_{e_h},B_{e_h},C_{e_h})$
    and $(R_{e_v},B_{e_v},C_{e_v})$ satisfying
    \begin{align}
        \{\,g\in E:\text{$g$ joins $R_{e_h}$ and $B_{e_h}$}\,\}
         & =\{e_h\},
         &
        \{\,g\in E:\text{$g$ joins $R_{e_v}$ and $B_{e_v}$}\,\}
         & =\{e_v\}.
        \notag
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    Rectangular injectivity gives $\inj(R_{e_h}),\inj(B_{e_h})$ and
    $\inj(R_{e_v}),\inj(B_{e_v})$, union closure gives $\inj(C_{e_h})$ and
    $\inj(C_{e_v})$ from their rectangular band decompositions, and direct
    coordinate enumeration verifies the unique-crossing identities.
\end{proof}

\begin{theorem}[Bond-dimension equality from normal torus rectangular injectivity]%
    \label{thm:peps_bondDim_eq_of_normalTorusRectangularInjectivity}
    \lean{TNLean.PEPS.bondDim_eq_of_normalTorusRectangle}
    \leanok
    \uses{def:peps_torus_translation_invariant, def:peps_same_state,
        def:peps_normalTorusRectanglePEPS}
    Let $A$ and $B$ be translationally invariant PEPS tensors
    (Definition~\ref{def:peps_torus_translation_invariant}) on the $n\times m$
    torus with $n,m\ge 7$ and $d>0$, generating the same state
    (Definition~\ref{def:peps_same_state}), and both normal with injective
    rectangles (Definition~\ref{def:peps_normalTorusRectanglePEPS}). Then
    $D_A=D_B$ \cite[Section~3, Lemma \emph{inj\_isomorph}, lines~560--583 and
        1407--1500]{Molnar2018NormalPEPS}.
    \begin{proof}
        \leanok
        \uses{lem:peps_injectiveRegion_union_pos,
            lem:peps_torusRectangleReferenceBlockingData,
            thm:peps_bondDim_eq_of_normalBlocking,
            thm:peps_torusUniformBondDim_of_translationInvariant,
            def:peps_torus_geometry_translation}
        By Lemma~\ref{lem:peps_injectiveRegion_union_pos}, $D_A,D_B>0$ gives
        union closure for $\inj_A$ and $\inj_B$. At $(x_0,y_0)=(n-5,m-5)$
        ($n,m\ge 7$), Lemma~\ref{lem:peps_torusRectangleReferenceBlockingData}
        and the one-edge form of Theorem~\ref{thm:peps_bondDim_eq_of_normalBlocking}
        at $e_h$ and $e_v$ give $D_A(e_h)=D_B(e_h)$ and $D_A(e_v)=D_B(e_v)$.
        For any horizontal edge $e=\tau e_h$ (and similarly any vertical edge
        $e=\tau e_v$), Theorem~\ref{thm:peps_torusUniformBondDim_of_translationInvariant}
        gives
        \begin{align}
            D_A(e)=D_A(\tau e_h)=D_A(e_h)=D_B(e_h)=D_B(\tau e_h)=D_B(e).
            \notag
        \end{align}
    \end{proof}
\end{theorem}

\begin{theorem}[Fundamental Theorem for normal PEPS on the torus]%
    \label{thm:fundamentalTheorem_normalTorusPEPS}
    \leanok
    \uses{thm:fundamentalTheorem_normalSquarePEPS, thm:peps_oneVertexComplementComparison, def:peps_torus_translation_invariant, thm:peps_torusUniformBondDim_of_translationInvariant, def:GaugeEquivPEPS, thm:fundamentalTheorem_normalTorusPEPS_unconditional, thm:torusAbsorbedGauge_unique_scalar, thm:torusCovariantAbsorbedGauge_unique_classScalar, thm:torusGauge_unique_scalar_of_perVertex}
    Let $A$ and $B$ be translationally invariant normal PEPS on the $n\times m$
    torus ($n,m\ge 7$) whose $2\times 3$ and $3\times 2$ regions are injective.
    If $A$ and $B$ generate the same state, then they are related by horizontal
    and vertical gauges $X,Y$ (unique up to scalar) and a scalar $\lambda$ with
    $\lambda^{nm}=1$ \cite[Theorem~3]{Molnar2018NormalPEPS}.
    This is proved in translation-covariant form in
    Theorem~\ref{thm:fundamentalTheorem_normalTorusPEPS_unconditional} (where
    seam edges carry the transposed inverse of the class matrix), with gauge
    uniqueness proved in Theorem~\ref{thm:torusAbsorbedGauge_unique_scalar} (per
    edge), Theorem~\ref{thm:torusCovariantAbsorbedGauge_unique_classScalar} (per
    orientation class), and Theorem~\ref{thm:torusGauge_unique_scalar_of_perVertex}
    (from the per-vertex relation).
\end{theorem}

\begin{theorem}[Fundamental Theorem for normal PEPS on the torus,
        translation-covariant form]%
    \label{thm:fundamentalTheorem_normalTorusPEPS_unconditional}
    \lean{TNLean.PEPS.fundamentalTheorem_normalTorusPEPS_unconditional}
    \leanok
    \uses{thm:fundamentalTheorem_normalSquarePEPS,
        thm:peps_oneVertexComplementComparison, thm:peps_sameState_transport,
        thm:peps_stateCoeff_translationInvariant,
        thm:peps_torusUniformBondDim_of_translationInvariant,
        def:peps_normalTorusRectanglePEPS, def:GaugeEquivPEPS}
    Let $A$ and $B$ be translationally invariant PEPS tensors on the discrete
    $n\times m$ torus ($n,m\ge 7$) with $d>0$, generating the same state, and
    both normal with injective rectangles
    (Definition~\ref{def:peps_normalTorusRectanglePEPS}). Then $D_A=D_B$, and
    there exist a translation-covariant family of invertible edge matrices
    $X=(X_e)_{e\in E}$ (generated from one horizontal reference matrix
    $X_{e_h}$ and one vertical reference matrix $X_{e_v}$, taking the
    transposed inverse on wraparound seam edges where the stored endpoint order
    is reversed) and a scalar $\lambda\in\C$ with $\lambda^{nm}=1$ such that,
    with $\widetilde B=B^X$:
    \begin{align}
        A_v & =\lambda\widetilde B_v,
            &
        C_A(e,M;\sigma) & =C_{\widetilde B}(e,M;\sigma),
        \label{eq:peps_torus_ft_relations}
    \end{align}
    for all $v\in V$, $e\in E$, $M$, and $\sigma$.
    Keeping $\lambda$ separate from $X$ realizes the translation-invariance
    remark of \cite[Section~4]{Molnar2018NormalPEPS}; uniqueness of $X$ up to
    scalar is given in
    Theorems~\ref{thm:torusAbsorbedGauge_unique_scalar}--\ref{thm:torusGauge_unique_scalar_of_perVertex}.
    \begin{proof}
        \leanok
        \uses{lem:peps_injectiveRegion_union_pos, thm:peps_bondDim_eq_of_normalTorusRectangularInjectivity}
        Lemma~\ref{lem:peps_injectiveRegion_union_pos} gives union closure of
        injective regions, and
        Theorem~\ref{thm:peps_bondDim_eq_of_normalTorusRectangularInjectivity}
        gives $D_A=D_B$. Anchoring the reference blocking frames against the
        far seam ($(x_0,y_0)=(n-5,m-5)$) decomposes their complements into
        wraparound-free rectangular bands for $n,m\ge 7$. Translating the
        reference gauges $X_{e_h}$ and $X_{e_v}$ produces a
        translation-covariant family $X$ whose absorption $\widetilde B=B^X$
        satisfies $C_A(e,M;\sigma)=C_{\widetilde B}(e,M;\sigma)$ at every
        $e\in E$. Since gauge absorption preserves region injectivity, the
        one-site comparison (Theorem~\ref{thm:peps_oneVertexComplementComparison})
        between the punctured $3\times 3$ square and its completion yields
        scalars $c_R,c_S\in\C^\times$ and $\lambda=c_S/c_R$ satisfying
        $A_v=\lambda\widetilde B_v$ in (\ref{eq:peps_torus_ft_relations}),
        carried to every $v\in V$ by translation covariance. Contracting over
        all $nm$ sites against the nonzero state gives $\lambda^{nm}=1$.
    \end{proof}
\end{theorem}

\begin{corollary}[Torus scalar condition from the per-vertex two-block proportionalities]%
    \label{cor:peps_lambda_pow_card_torus_eq_one_of_twoBlockProportional}
    \lean{TNLean.PEPS.lambda_pow_card_torus_eq_one_of_twoBlockProportional}
    \leanok
    \uses{cor:peps_component_eq_gaugeVertex_of_twoBlockProportional}
    Let $A$ and $B$ be PEPS tensors on the $n\times m$ torus ($n,m>1$) with
    $D_A=D_B>0$, generating the same state, and let $X$ be a family of
    invertible edge matrices with $\widetilde B=B^X$. Suppose at each $v\in V$
    there is a region $R_v\not\ni v$ with $\inj_{\widetilde B}(R_v)$ such that
    the two-block proportionalities hold on $R_v$ and $S_v=R_v\cup\{v\}$ with
    ratios $c_R(v)\ne 0$ and $c_S(v)$ satisfying $c_S(v)/c_R(v)=\lambda$
    independently of $v$. If there exists a region $R_0$ with $\inj_A(R_0)$ and
    $\inj_A(V\setminus R_0)$, then $\lambda^{nm}=1$.
\end{corollary}
\begin{proof}\leanok
    By Corollary~\ref{cor:peps_component_eq_gaugeVertex_of_twoBlockProportional},
    $A_v(\eta,\sigma)=\lambda\widetilde B_v(\eta,\sigma)$ at each of the $nm$
    sites $v$. Contracting over the torus yields
    \begin{align}
        \Coeff_A(\sigma)
         & =
        \lambda^{nm}\Coeff_{\widetilde B}(\sigma)
        =\lambda^{nm}\Coeff_A(\sigma).
        \notag
    \end{align}
    Since $\inj_A(R_0)$ and $\inj_A(V\setminus R_0)$ imply $\Coeff_A\ne 0$,
    $\lambda^{nm}=1$.
\end{proof}

\begin{theorem}[Per-edge uniqueness of the torus gauge up to a constant]%
    \label{thm:torusAbsorbedGauge_unique_scalar}
    \lean{TNLean.PEPS.torusAbsorbedGauge_unique_scalar}
    \leanok
    \uses{def:applyGauge, def:peps_edgeInsertedCoeff, def:peps_normalTorusRectanglePEPS}
    Let $A$ and $B$ be translationally invariant PEPS tensors on the $n\times m$
    torus ($n,m\ge 7$) with $d>0$ and $D_A=D_B$, generating the same state, and
    both normal with injective rectangles
    (Definition~\ref{def:peps_normalTorusRectanglePEPS}). If two families of
    invertible edge matrices $X,X'$ both satisfy the absorbed insertion
    equality $C_A(e,M;\sigma)=C_{B^X}(e,M;\sigma)=C_{B^{X'}}(e,M;\sigma)$ at an
    edge $e$ for all $M$ and $\sigma$
    (Definitions~\ref{def:applyGauge} and~\ref{def:peps_edgeInsertedCoeff}),
    then there exists $c\in\C^\times$ such that $X'_e=c X_e$
    \cite[Theorem~3]{Molnar2018NormalPEPS}.
    \begin{proof}
        \leanok
        \uses{lem:peps_injectiveRegion_union_pos}
        By Lemma~\ref{lem:peps_injectiveRegion_union_pos} and rectangular
        injectivity, there is a region $R_1$ with $\inj_B(R_1)$,
        $\inj_B(V\setminus R_1)$, and $e$ as its unique crossing edge to the
        adjacent block. Lifting the bare-edge insertion equalities to $R_1$
        shows that the orientation-adapted absorbing matrices
        $X_{\mathrm{abs}},X'_{\mathrm{abs}}$ induce the same conjugation on
        $\MN{D_A(e)}$:
        \begin{align}
            X_{\mathrm{abs}}\,N\,X_{\mathrm{abs}}^{-1}
             & =
            X'_{\mathrm{abs}}\,N
            (X'_{\mathrm{abs}})^{-1}
            \notag
        \end{align}
        for every $N\in\MN{D_A(e)}$. Hence
        $X'_{\mathrm{abs}}=c X_{\mathrm{abs}}$ for some $c\in\C^\times$, and
        unwinding the orientation adaptation gives $X'_e=c X_e$.
    \end{proof}
\end{theorem}

\begin{theorem}[Class constancy of the gauge comparison scalars]%
    \label{thm:torusCovariantAbsorbedGauge_unique_classScalar}
    \lean{TNLean.PEPS.torusCovariantAbsorbedGauge_unique_classScalar}
    \leanok
    \uses{def:applyGauge, def:peps_edgeInsertedCoeff}
    In the setting of Theorem~\ref{thm:torusAbsorbedGauge_unique_scalar},
    suppose $X$ and $X'$ realize $C_A(e,M;\sigma)=C_{B^X}(e,M;\sigma)=C_{B^{X'}}(e,M;\sigma)$
    at every $e\in E$ and are both translation covariant (meaning $X_{\tau e}=X_e$
    when $\tau$ preserves the stored endpoint order and $X_{\tau e}=X_e^{-\top}$
    when $\tau$ swaps it on the seam). Then there exist $c_h,c_v\in\C^\times$
    such that $X'_e=c_h X_e$ on interior horizontal edges ($e_{1,x}+1=e_{2,x}$),
    $X'_e=c_h^{-1}X_e$ on seam horizontal edges, and analogously with $c_v$ on
    interior and seam vertical edges \cite[Theorem~3]{Molnar2018NormalPEPS}.
    \begin{proof}
        \leanok
        \uses{thm:torusAbsorbedGauge_unique_scalar}
        Let $c_h,c_v\in\C^\times$ be the scalars of
        Theorem~\ref{thm:torusAbsorbedGauge_unique_scalar} at the reference
        edges $e_h,e_v$, so $X'_{e_h}=c_h X_{e_h}$ and $X'_{e_v}=c_v X_{e_v}$.
        For any horizontal edge $e=\tau e_h$, translation covariance gives
        \begin{align}
            X'_e
             & =\tau\cdot X'_{e_h}
            =\tau\cdot(c_h X_{e_h})
            =c_h X_e
            \notag
        \end{align}
        when $e_{1,x}+1=e_{2,x}$, and on a seam edge
        \begin{align}
            X'_e
             & =(X'_{e_h})^{-\top}
            =(c_h X_{e_h})^{-\top}
            =c_h^{-1}(X_{e_h})^{-\top}
            =c_h^{-1}X_e,
            \notag
        \end{align}
        and identically for the vertical class with $c_v$.
    \end{proof}
\end{theorem}

\begin{theorem}[Uniqueness of the gauges against the per-vertex relation]%
    \label{thm:torusGauge_unique_scalar_of_perVertex}
    \lean{TNLean.PEPS.torusGauge_unique_scalar_of_perVertex}
    \leanok
    \uses{def:gaugeVertex, def:applyGauge, def:peps_normalTorusRectanglePEPS}
    Let $A$ and $B$ be as in
    Theorem~\ref{thm:torusAbsorbedGauge_unique_scalar}. If two families of
    invertible edge matrices $X,X'$ and scalars $\lambda,\lambda'$ with
    $\lambda^{nm}={\lambda'}^{nm}=1$ satisfy
    $A_v=\lambda(B^X)_v=\lambda'(B^{X'})_v$ at every $v\in V$
    (Definitions~\ref{def:gaugeVertex} and~\ref{def:applyGauge}), then for
    every $e\in E$ there exists $c_e\in\C^\times$ such that $X'_e=c_e X_e$
    \cite[Theorem~3]{Molnar2018NormalPEPS}.
    \begin{proof}
        \leanok
        \uses{thm:torusAbsorbedGauge_unique_scalar, def:peps_edgeInsertedCoeff}
        By multilinearity of $C_A(e,M;\sigma)$
        (Definition~\ref{def:peps_edgeInsertedCoeff}) over the $nm$ sites and
        $\lambda^{nm}={\lambda'}^{nm}=1$,
        $C_A(e,M;\sigma)=\lambda^{nm}C_{B^X}(e,M;\sigma)=C_{B^X}(e,M;\sigma)$
        and similarly $C_A(e,M;\sigma)=C_{B^{X'}}(e,M;\sigma)$; apply
        Theorem~\ref{thm:torusAbsorbedGauge_unique_scalar}.
    \end{proof}
\end{theorem}
