\section{Changes of ownership in encoded frames}
\label{sec:peps_ownership_changes}
Changes of site and hole ownership in encoded frames
(\cite[Lemmas~6.5 and~6.6]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 347--570) take the mutual-information bounds as
hypotheses and realize the maps constructed in their proofs as bounded
changes: allowed monomials of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}
(Section~\ref{sec:peps_pair_effects}) on the registers of the frames, one per
site and per tag (Definition~\ref{def:peps_frame_registers}). The monomials are
first constructed on registers in which the site registers of a region held by
one party are grouped into one register, and then placed on the registers of
the frames by reorderings of tensor factors and grouping unitaries.

We keep the setting of Section~\ref{sec:peps_encoded_frames}. For a set
$S\subseteq\Lambda$ write $\mathcal H_S=\bigotimes_{x\in S}\C^q$. For a vector
$\Omega\in\mathcal H_\Lambda$ and disjoint sets $T,E\subseteq\Lambda$, the
mutual information $I_\Omega(T:E)$ is
$S(\rho_T)+S(\rho_E)-S(\rho_{TE})$ with natural logarithms, where the $\rho$
are the reduced density matrices of $\ket\Omega\bra\Omega$.

\begin{lemma}[Operators on parts of a sheet]

        TNLean.PEPS.EncodedFrame.threeSplit_snd,

        TNLean.PEPS.EncodedFrame.exists_threeSplit_eq_kronecker_one,
        TNLean.PEPS.EncodedFrame.exists_threeSplit_eq_one_kronecker_kronecker_one}
    \leanok
    Under the identifications $\mathcal H_\Lambda\simeq\mathcal H_S\otimes\mathcal H_{\Lambda\setminus S}$
    and, for disjoint $T,E\subseteq\Lambda$ with $U=\Lambda\setminus(T\cup E)$,
    $\mathcal H_\Lambda\simeq\mathcal H_T\otimes\mathcal H_E\otimes\mathcal H_U$
    via $c\mapsto((c|_T,c|_E),c|_U)$, an operator $A$ supported on $S$ has the
    form $A'\otimes\id_{\Lambda\setminus S}$, on a set disjoint from $S$ has
    the form $\id_S\otimes A'$, on $T\cup E$ has the form $A'\otimes\id_U$, and
    on a subset of $E$ has the form $\id_T\otimes A'\otimes\id_U$.
\end{lemma}

\begin{proof}\leanok
    Each product operator $\bigotimes_{x\in\Lambda}m_x$ with $m_x=\id$ off the
    support factors across the tensor decomposition, and the claim extends by
    linearity.
\end{proof}

\begin{lemma}[Splitting consequence on a sheet]
    \label{lem:peps_frame_splitting}
    \lean{TNLean.PEPS.EncodedFrame.SplittingData,
        TNLean.PEPS.EncodedFrame.SplittingData.error,
        TNLean.PEPS.EncodedFrame.splitVec,
        TNLean.PEPS.EncodedFrame.partialTraceRight_splitVec,
        TNLean.PEPS.EncodedFrame.quantumRelativeEntropy_splitVec_eq_mutualInformation,
        TNLean.PEPS.EncodedFrame.exists_sheetSplitting,
        TNLean.PEPS.EncodedFrame.exists_sheetSplitting_zpow,
        TNLean.PEPS.EncodedFrame.exists_sheetSplitting_rpow}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $T,E\subseteq\Lambda$ be disjoint, $U=\Lambda\setminus(T\cup E)$,
    $\lVert\Omega\rVert=1$, and $b=I_\Omega(T:E)$. There exist
    finite-dimensional spaces $\mathcal B_T,\mathcal B_E$, an isometry
    $V\colon\mathcal H_U\to\mathcal B_T\otimes\mathcal B_E$, and unit vectors
    $s\in\mathcal H_T\otimes\mathcal B_T$, $s'\in\mathcal H_E\otimes\mathcal B_E$
    such that
    \begin{align}
        \lVert(\id_{TE}\otimes V)\Omega-s\otimes s'\rVert
        \le\sqrt{2(1-e^{-b/2})}\le\sqrt b .
        \label{eq:peps_frame_splitting}
    \end{align}
    In particular, for the sheet splitting of
    \cite[Lemma~6.4]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 352--368), $b\le L^{-60}$ gives an error of at most $L^{-30}$; more
    generally, for $L\ge0$ and real $p$, $b\le L^{-2p}$ gives an error of at
    most $L^{-p}$.
\end{lemma}

\begin{proof}\leanok
    Since $D(\rho_{TE}\,\|\,\rho_T\otimes\rho_E)=I_\Omega(T:E)=b$, applying the
    tripartite pure-state splitting bound to $\Omega\in\mathcal H_T\otimes\mathcal H_E\otimes\mathcal H_U$
    (Lemma~\ref{lem:peps_frame_split_forms}) with $\mathcal B_T=\mathcal H_T$
    and $\mathcal B_E=\mathcal H_E\oplus\mathcal H_U$ yields~(\ref{eq:peps_frame_splitting}).
\end{proof}

\begin{definition}[Birth map]
    \label{def:peps_frame_birth_map}
    \lean{TNLean.PEPS.EncodedFrame.pairSource,
        TNLean.PEPS.EncodedFrame.birthKernel,
        TNLean.PEPS.EncodedFrame.birthEffect,
        TNLean.PEPS.EncodedFrame.birthOp,
        TNLean.PEPS.EncodedFrame.sheetBirthOp,
        TNLean.PEPS.EncodedFrame.birthOp_eq,
        TNLean.PEPS.EncodedFrame.pairSource_mul_conjTranspose_apply}
    \leanok
    \uses{lem:peps_frame_splitting}
    With the splitting data $(V,s,s')$ of Lemma~\ref{lem:peps_frame_splitting},
    set $\mathcal V=\id_{TE}\otimes V$ and let
    $S\colon\mathcal H_E\otimes\mathcal B_E\to\mathcal H_{TE}\otimes\mathcal B_T\otimes\mathcal B_E$
    act by $x\mapsto s\otimes x$. The birth map is
    \begin{align}
        B=\mathcal V^\dagger SS^\dagger\mathcal V
        =\mathcal V^\dagger\bigl(\ket s\bra s_{T\mathcal B_T}
        \otimes\id_{E\mathcal B_E}\bigr)\mathcal V
        \label{eq:peps_frame_birth_map}
    \end{align}
    (\texttt{05-frames.tex}, equation (birth-map), lines 414--426).
\end{definition}

\begin{lemma}[Properties of the birth map]
    \label{lem:peps_frame_birth_map}
    \lean{TNLean.PEPS.EncodedFrame.pairSource_conjTranspose_mul_self,
        TNLean.PEPS.EncodedFrame.pairSource_mulVec,
        TNLean.PEPS.EncodedFrame.birthOp_isHermitian,
        TNLean.PEPS.EncodedFrame.norm_birthOp_le_one,
        TNLean.PEPS.EncodedFrame.norm_act_birthOp_sub_le,
        TNLean.PEPS.EncodedFrame.birthEffect_eq_submatrix,
        TNLean.PEPS.EncodedFrame.birthOp_eq_submatrix,
        TNLean.PEPS.EncodedFrame.commute_birthOp,
        TNLean.PEPS.EncodedFrame.sheetBirthOp_isHermitian,
        TNLean.PEPS.EncodedFrame.norm_sheetBirthOp_le_one,
        TNLean.PEPS.EncodedFrame.norm_act_sheetBirthOp_sub_le,
        TNLean.PEPS.EncodedFrame.commute_sheetBirthOp}
    \leanok
    \uses{def:peps_frame_birth_map}
    The map $B$ (Definition~\ref{def:peps_frame_birth_map}) satisfies
    $B^\dagger=B$, $\|B\|\le 1$, $B=\id_E\otimes K^\dagger K$ with
    $K=\bra s(\id_T\otimes V)$ (so $[B,A]=0$ for any $A$ supported in $E$), and
    for every $\psi\in\mathcal H_\Lambda$,
    \begin{align}
        \lVert(B-\id)\psi\rVert\le\lVert\mathcal V\psi-s\otimes s'\rVert .
        \label{eq:peps_frame_birth_map_error}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $\lVert s\rVert=1$, $S^\dagger S=\id$, so $\Pi=SS^\dagger$ is an
    orthogonal projection with $\Pi(s\otimes s')=s\otimes s'$. As
    $\mathcal V^\dagger\mathcal V=\id$,
    $(B-\id)\psi=\mathcal V^\dagger(\Pi-\id)(\mathcal V\psi-s\otimes s')$,
    giving~(\ref{eq:peps_frame_birth_map_error}). On
    $\mathcal H_E\otimes(\mathcal H_T\otimes\mathcal H_U)$,
    $S^\dagger\mathcal V=\id_E\otimes K$, so $B=\id_E\otimes K^\dagger K$.
\end{proof}

\begin{lemma}[Operators passing through an encoding]
    \label{lem:peps_frame_pass_encoding}
    \lean{TNLean.PEPS.EncodedFrame.one_kronecker_mul_stack,
        TNLean.PEPS.EncodedFrame.one_kronecker_mul_frameEncoder,
        TNLean.PEPS.EncodedFrame.norm_act_one_kronecker_refVec_sub_le}
    \leanok
    \uses{def:peps_frame}
    If a sheet operator $B$ commutes with the sheet part of $K_{\mathcal F}$
    (Definition~\ref{def:peps_frame}) on every tag configuration, then
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F}B$ and
    $\lVert(\id_{\rm tags}\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F}\rVert\le\lVert B\Omega-\Omega\rVert$.
\end{lemma}

\begin{proof}\leanok
    Commutation on each tag configuration gives
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F}B$, and
    $\|K_{\mathcal F}\|\le 1$ gives the norm bound on
    $\Omega_{\mathcal F}=K_{\mathcal F}\Omega$.
\end{proof}

\begin{theorem}[Homogeneous birth and death: sheet map]
    \label{thm:peps_frame_birth}
    \lean{TNLean.PEPS.EncodedFrame.Frame.birth,
        TNLean.PEPS.EncodedFrame.Frame.death}
    \leanok
    \uses{lem:peps_frame_birth_map,lem:peps_frame_pass_encoding,
        def:peps_frame_birth_partition,lem:peps_frame_birth_partition_owners}
    In an encoded frame $\mathcal F$ with $\lVert\Omega\rVert=1$, let $T$ be
    owned by $P_\circ$ with $T\cap H^+(\mathcal F)=\varnothing$, and let
    $\mathcal F',E,U$ be as in Definition~\ref{def:peps_frame_birth_partition}.
    If $I_\Omega(T:E)\le L^{-60}$, the birth map $B$ associated with splitting
    data of error at most $L^{-30}$ satisfies $B^\dagger=B$, $\|B\|\le 1$, acts
    as $\id$ on $E$, commutes with $K_{\mathcal F}=K_{\mathcal F'}$ via
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F'}B$, and obeys
    \begin{align}
        \lVert(\id_{\rm tags}\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F'}\rVert
        \le L^{-30}
        \label{eq:peps_frame_birth_bound}
    \end{align}
    (\cite[Lemma~6.5]{OpenAI2026PolynomialPEPS}). The death map is the same
    operator $B$ from $\mathcal F'$ to $\mathcal F$ when the hypotheses hold in
    the post-death frame $\mathcal F$.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:peps_frame_birth_map,
        lem:peps_frame_pass_encoding,def:peps_frame_birth_partition,
        lem:peps_frame_birth_partition_owners}
    By Lemma~\ref{lem:peps_frame_birth_partition_owners}, $T\cap E=\varnothing$,
    so Lemma~\ref{lem:peps_frame_splitting} provides splitting data with error
    $\le L^{-30}$. Since $H^+(\mathcal F)\subseteq E$, every hole encoder acts
    inside $E$ and commutes with $B$ by Lemma~\ref{lem:peps_frame_birth_map}.
    Applying Lemma~\ref{lem:peps_frame_pass_encoding} and~(\ref{eq:peps_frame_birth_map_error})
    with $\Omega_{\mathcal F'}=\Omega_{\mathcal F}$ yields~(\ref{eq:peps_frame_birth_bound}).
\end{proof}

\begin{theorem}[Composition without additional pair effects]
    \label{thm:peps_monomial_composition_counts}
    \lean{TNLean.PEPS.PairEffect.PartyChain.precomp_props,
        TNLean.PEPS.PairEffect.PartyChain.postcomp_props,
        TNLean.PEPS.PairEffect.PartyChain.precomp,
        TNLean.PEPS.PairEffect.PartyChain.postcomp,
        TNLean.PEPS.PairEffect.PartyChain.eval_precomp,
        TNLean.PEPS.PairEffect.PartyChain.eval_postcomp}
    \leanok
    \uses{def:peps_monomial_parties}
    Let $M$ be an allowed monomial using only parties in $S$, and let $W_0,W_1$
    be effect-free allowed words using only parties in $S$ with compatible
    boundary registers. Then $M_0=M\circ W_0$ and $M_1=W_1\circ M$ are allowed
    monomials using only parties in $S$, with operators
    $F_{M_0}=F_M\circ F_{W_0}$ and $F_{M_1}=F_{W_1}\circ F_M$ and resource
    counts
    \begin{align}
        s(M_0) & =s(W_0)+s(M),\notag \\
        s(M_1) & =s(M)+s(W_1),\notag \\
        e(M_0) & =e(M_1)=e(M).
        \label{eq:peps_monomial_composition_counts}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    Absorb $W_0$ into the initial effect-free segment of $M$ and $W_1$ into its
    final effect-free segment by induction on the pair effects of $M$.
\end{proof}

              ($\mathcal B_E\to\mathcal H_T\otimes\mathcal H_U$) (lines 439--446).
    \end{itemize}
    For $Q_\circ=P_\circ$, both are the single private contraction
    $K^\dagger K$ at $P_\circ$ (lines 447--449).
\end{definition}

\subsection{Exchanging assignments between two sheets}

\begin{definition}[Swapping two sheets]
    \label{def:peps_frame_sheet_swap}
    \lean{TNLean.Lattice.sheetSwap,
        TNLean.Lattice.sheetSwapOp}
    \leanok
    For $S\subseteq\Lambda$, let $F_S$ be the unitary swapping the two sheets
    at the sites in $S$ (\texttt{05-frames.tex}, lines 476--477).
\end{definition}

\begin{theorem}[Permutation matrices are unitary]
    \label{thm:permutation_matrix_unitary}
    \lean{Equiv.Perm.permMatrix_mem_unitaryGroup}
    \leanok
    For a permutation $\sigma$ of a finite set, its complex permutation matrix
    $P_\sigma$ satisfies $P_\sigma P_\sigma^\dagger=P_\sigma^\dagger P_\sigma=I$.
\end{theorem}

\begin{proof}\leanok
    Since $P_\sigma^\dagger=P_{\sigma^{-1}}$,
    $P_\sigma P_{\sigma^{-1}}=P_{\sigma\sigma^{-1}}=I$.
\end{proof}

\begin{theorem}[Sheet swaps are contractions]
    \label{thm:peps_frame_sheet_swap_norm}
    \lean{TNLean.Lattice.norm_sheetSwapOp_le_one}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    $\|F_S\|\leq1$.
\end{theorem}

\begin{proof}\leanok
    By Definition~\ref{def:peps_frame_sheet_swap} and
    Theorem~\ref{thm:permutation_matrix_unitary}, $F_S$ is unitary, hence
    $\|F_S\|\le 1$.
\end{proof}

\begin{lemma}[Composing and commuting sheet swaps]
    \label{lem:peps_frame_sheet_swap_compose}
    \lean{TNLean.Lattice.sheetSwapOp_mul_sheetSwapOp,
        TNLean.Lattice.commute_sheetSwapOp_kronecker}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    If $Y=T\sqcup A$, then $F_AF_Y=F_T$. Moreover, $[F_S,R_1\otimes R_2]=0$
    whenever $R_1,R_2$ are supported in $\Lambda\setminus S$
    (\texttt{05-frames.tex}, lines 512--520).
\end{lemma}

\begin{proof}\leanok
    Swapping at $Y$ and then at $A\subseteq Y$ swaps on $Y\setminus A=T$, so
    $F_AF_Y=F_T$. When $R_1,R_2$ are supported in $\Lambda\setminus S$,
    $F_S$ and $R_1\otimes R_2$ act on disjoint tensor factors.
\end{proof}

\begin{theorem}[Exchanging supported operators]
    \label{thm:peps_sheet_swap_supported_exchange}
    \lean{TNLean.Lattice.sheetSwapOp_mul_kronecker_of_mem_supportedOperators}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    For $R\subseteq\Lambda$ and operators $A,B$ on $\mathcal H_\Lambda$
    supported on $R$, the replica-term exchange identity of
    \cite{OpenAI2026PolynomialPEPS} (\texttt{02-information.tex}, lines 416--424)
    is
    \begin{align}
        F_R(A\otimes B)=(B\otimes A)F_R.
        \label{eq:peps_sheet_swap_supported_exchange}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_split_forms,lem:peps_frame_exchange_intertwining}
    By Lemma~\ref{lem:peps_frame_split_forms}, $A=A_R\otimes I_{R^c}$ and
    $B=B_R\otimes I_{R^c}$, while $F_R$ swaps the two copies of $\mathcal H_R$
    and acts as $I$ on $\mathcal H_{R^c}^{\otimes 2}$, giving~(\ref{eq:peps_sheet_swap_supported_exchange}).
\end{proof}

\begin{theorem}[Cancellation of a noncrossing doubled interaction]
    \label{thm:peps_sheet_swap_doubled_cancellation}
    \lean{TNLean.Lattice.sheetSwapOp_conj_doubledHamiltonian_eq_of_mem_supportedOperators}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    For $R,S\subseteq\Lambda$ and an operator $A$ supported on $S$, if
    $S\subseteq R$ or $S\cap R=\varnothing$, then
    \begin{align}
        F_R(A\otimes I+I\otimes A)F_R^\dagger
        =A\otimes I+I\otimes A
        \label{eq:peps_sheet_swap_doubled_cancellation}
    \end{align}
    (\cite{OpenAI2026PolynomialPEPS}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_supported_exchange,lem:peps_frame_sheet_swap_compose,
        thm:permutation_matrix_unitary}
    If $S\subseteq R$, Theorem~\ref{thm:peps_sheet_swap_supported_exchange}
    gives $F_R(A\otimes I)=(I\otimes A)F_R$ and $F_R(I\otimes A)=(A\otimes I)F_R$;
    if $S\cap R=\varnothing$, Lemma~\ref{lem:peps_frame_sheet_swap_compose}
    gives $[F_R,A\otimes I]=[F_R,I\otimes A]=0$. Multiplying on the right by
    $F_R^\dagger=F_R^{-1}$ (Theorem~\ref{thm:permutation_matrix_unitary})
    yields~(\ref{eq:peps_sheet_swap_doubled_cancellation}).
\end{proof}

\begin{theorem}[Restricting a swap to an interaction support]
    \label{thm:peps_sheet_swap_interaction_restriction}
    \lean{TNLean.Lattice.sheetSwapOp_conj_kronecker_eq_inter,
        TNLean.Lattice.sheetSwapOp_conj_doubledHamiltonian_eq_inter}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    Let $q$ be a natural local dimension, let $\Lambda$ be a finite set of
    sites, and let $R,S\subseteq\Lambda$. Suppose that $A$ and $B$ are
    operators supported on $S$. If $F_R$ exchanges the two copies at the
    sites in $R$, then
    \begin{align}
        F_R(A\otimes B)F_R^\dagger
        =F_{R\cap S}^\dagger(A\otimes B)F_{R\cap S}.
    \end{align}
    In particular, writing $T(A)=A\otimes I+I\otimes A$, one has
    \begin{align}
        F_R T(A) F_R^\dagger = F_{R\cap S}^\dagger T(A) F_{R\cap S}.
        \label{eq:peps_swap_interaction_restriction}
    \end{align}
    This is the support reduction used in the boundary perturbation argument
    of \cite{OpenAI2026PolynomialPEPS}. Neither Hermiticity nor a positive
    local dimension is required.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_frame_sheet_swap_compose,thm:permutation_matrix_unitary}
    Put $K=R\cap S$, $O=R\setminus S$, and $M=A\otimes B$. Composition
    of the swaps gives $F_K F_R=F_O$, hence $F_R=F_K^\dagger F_O$ by
    unitarity. Since $O$ is disjoint from $S$, the swap $F_O$ commutes
    with $M$, and $F_O M F_O^\dagger=M$. Consequently,
    \begin{align}
        F_R M F_R^\dagger
        =F_K^\dagger(F_O M F_O^\dagger)F_K
        =F_K^\dagger M F_K.
    \end{align}
    Apply this identity to $(A,I)$ and $(I,A)$ and add the two equations
    to obtain the assertion for $T(A)$.
\end{proof}

\begin{theorem}[Crossing-edge expansion of the doubled perturbation]
    \label{thm:peps_sheet_swap_boundary_expansion}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.sheetSwapOp_conj_doubled_operator_sub_eq_sum_crossing}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_frame_sheet_swap}
    For an open-square Hamiltonian $H=\sum_v h_v+\sum_e h_e$
    (Definition~\ref{def:polynomial_peps_model}), $T(A)=A\otimes I+I\otimes A$,
    and $R\subseteq V$ with crossing-edge set $\partial R$,
    \begin{align}
        F_RT(H)F_R^\dagger-T(H)
        =\sum_{e\in\partial R}\bigl(F_RT(h_e)F_R^\dagger-T(h_e)\bigr)
        \label{eq:peps_sheet_swap_boundary_expansion}
    \end{align}
    (\cite{OpenAI2026PolynomialPEPS}).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_sheet_swap_doubled_cancellation}
    By Theorem~\ref{thm:peps_sheet_swap_doubled_cancellation},
    $F_RT(h_v)F_R^\dagger-T(h_v)=0$ for all $v$ and
    $F_RT(h_e)F_R^\dagger-T(h_e)=0$ for all $e\notin\partial R$, leaving~(\ref{eq:peps_sheet_swap_boundary_expansion}).
\end{proof}

\begin{theorem}[Norm of a swapped doubled interaction]
    \label{thm:peps_sheet_swap_doubled_term_norm}
    \lean{TNLean.PEPS.EncodedFrame.norm_sheetSwapOp_conj_doubledHamiltonian_sub_le}

    \end{align}
\end{theorem}

\begin{proof}\leanok

    $\|T(A)\|\le\|A\otimes I\|+\|I\otimes A\|\le 2\|A\|$; combining this with
    $\|F_R\|\le 1$ (Theorem~\ref{thm:peps_frame_sheet_swap_norm}) and the
    triangle inequality gives~(\ref{eq:peps_sheet_swap_doubled_term_norm}).
\end{proof}

\begin{theorem}[Support of a swapped tensor product]
    \label{thm:peps_sheet_swap_conjugate_support}
    \lean{TNLean.Lattice.reindex_sheetSwapOp_conj_kronecker_mem_supportedOperators,
        TNLean.Lattice.reindex_sheetSwapOp_conj_doubledHamiltonian_mem_supportedOperators}
    \leanok

    Let $\Lambda$ be a finite site set, let $q$ be a natural local dimension,
    and let $R,S\subseteq\Lambda$. Suppose that $A$ and $B$ are supported
    on $S$. Express the two-copy operators in paired-site coordinates,
    and write $\widetilde F_R$ and $\widetilde M$ for the matrices of
    $F_R$ and $A\otimes B$ in these coordinates. Then
    $\widetilde F_R\widetilde M\widetilde F_R^\dagger$ is supported on $S$.
    In particular this holds for the paired matrix of
    $T(A)=A\otimes I+I\otimes A$, as used in the boundary perturbation
    argument of \cite{OpenAI2026PolynomialPEPS}. Zero local dimension
    and empty site sets are included.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_interaction_restriction,
        thm:peps_sheet_swap_paired_site_support,thm:qc_site_pair_tensor_support}
    Put $K=R\cap S$. Restricting the swap gives
    $F_R(A\otimes B)F_R^\dagger=F_K^\dagger(A\otimes B)F_K$.
    In paired-site coordinates, $F_K$ is supported on $K\subseteq S$,
    and $A\otimes B$ is supported on $S$. The operators supported on
    $S$ are closed under adjoints and products, which proves the first
    assertion. Apply it to $(A,I)$ and $(I,A)$ and use closure under
    sums to obtain the assertion for $T(A)$.
\end{proof}

\begin{theorem}[Support and norm of the doubled boundary perturbation]
    \label{thm:peps_swapped_doubled_boundary_bound}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.swapped_doubled_boundary_bound}
    \leanok
    \uses{def:polynomial_peps_model,def:qc_site_pair_coordinates,def:peps_frame_sheet_swap}
    Let $H=\sum_{v\in V}h_v+\sum_e h_e$ on $V=\{0,\ldots,L-1\}^2$ be a
    Hermitian open-square Hamiltonian (Definition~\ref{def:polynomial_peps_model})
    with interaction norms $\le J$. For $R\subseteq V$ with crossing-edge set
    $\partial R$, set $Z=\bigcup_{e\in\partial R}\{\text{endpoints of }e\}$,
    and let $K$ and $F$ be the matrices of $H\otimes I+I\otimes H$ and $F_R$ in
    paired-site coordinates (Definition~\ref{def:qc_site_pair_coordinates}).
    Then $FKF^\dagger-K$ is supported on $Z$ (with local dimension $q^2$) and
    satisfies
    \begin{align}
        \|FKF^\dagger-K\|\leq 8JL^2
        \label{eq:peps_swapped_doubled_boundary_bound}
    \end{align}
    (\cite{OpenAI2026PolynomialPEPS}).
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_boundary_expansion,
        thm:peps_sheet_swap_conjugate_support,
        thm:peps_sheet_swap_doubled_cancellation,
        thm:peps_sheet_swap_doubled_term_norm,thm:peps_square_edge_count}
    By~(\ref{eq:peps_sheet_swap_boundary_expansion}) and
    Theorem~\ref{thm:peps_sheet_swap_conjugate_support}, $FKF^\dagger-K$ is a
    sum over $e\in\partial R$ of terms supported on $e\subseteq Z$. By~(\ref{eq:peps_sheet_swap_doubled_term_norm})
    and $|\partial R|\le 2L^2$ (Theorem~\ref{thm:peps_square_edge_count}),
    $\|FKF^\dagger-K\|\le\sum_{e\in\partial R}4\|h_e\|\le 4J|\partial R|\le 8JL^2$.
\end{proof}

    Under the hypotheses of Theorem~\ref{thm:peps_frame_exchange} (or
    Theorem~\ref{thm:peps_frame_exchange_any_order} for arbitrary hole
    orderings), the two-sheet exchange is realized by an allowed monomial
    between the registers of $(\mathcal F_1,\mathcal F_2)$ and
    $(\mathcal F_1',\mathcal F_2')$ using only $\{P_\circ\}$ and $(s,e)=(0,0)$,
    with operator $C=D_UF_A\mathcal R$ (or $CG$) and reference-vector error
    $\le 4L^{-30}$ (\cite[Lemma~6.6 and Theorem~5.2]{OpenAI2026PolynomialPEPS},
    \texttt{05-frames.tex}, lines 460--479).
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_frame_exchange,thm:peps_frame_exchange_any_order,
        lem:peps_exchange_renaming_word,lem:peps_exchange_correction_word,
        lem:peps_frame_exchange_correction_monomial,
        lem:peps_frame_exchange_owners,lem:peps_register_reorderings}
    Compose the tag-reordering word of Lemma~\ref{lem:peps_register_reorderings}
    (if needed), the renaming word of Lemma~\ref{lem:peps_exchange_renaming_word},
    and the correction word of Lemma~\ref{lem:peps_exchange_correction_word};
    the operator and error bound follow from Theorem~\ref{thm:peps_frame_exchange}
    and Theorem~\ref{thm:peps_frame_exchange_any_order}.
\end{proof}

\subsection{Sheets without holes}

None of the monomials above acts on the tag registers: the birth and death
monomials and the renaming of the sites act behind the tag registers, which may be
replaced by any block of registers in front of the site registers. With the
empty block they act on a sheet without holes, an owner for each site of a
finite set $\Lambda$, whose reference vector is $\Omega$ placed in the site
registers. This is the setting of the one-dimensional protocol, where no
three owners meet and no encoded holes are needed.

\begin{definition}[Sheets without holes]
    \label{def:lattice_sheet}
    \lean{TNLean.Lattice.Sheet,
        TNLean.Lattice.Sheet.regs,
        TNLean.Lattice.Sheet.regIso,
        TNLean.Lattice.Sheet.regRef,
        TNLean.Lattice.Sheet.twoRegIso,
        TNLean.Lattice.Sheet.changeOwner,
        TNLean.Lattice.Sheet.birthEnv,
        TNLean.Lattice.Sheet.swapIn,
        TNLean.Lattice.Sheet.exchangeEnv,
        TNLean.Lattice.Sheet.exchangeOp,
        TNLean.Lattice.unitProdIso₂_act_liftFront,
        TNLean.Lattice.Sheet.regIso_eq_frontLayoutIso,
        TNLean.Lattice.Sheet.twoRegIso_eq_twoFrontLayoutIso}
    \leanok
    \uses{def:peps_frame_registers,def:peps_frame_sheet_swap,
        def:peps_frame_buffer_correction}
    A sheet on $\Lambda$ is an owner for each site; its registers are one
    register $\C^q$ per site, held by its owner, identified with
    $\mathcal H_\Lambda$ in product coordinates, and its reference vector is
    $\Omega$. These coordinates are those of the empty block of registers in
    front of the site registers. For a party $P_\circ$ let $E$ be
    the set of sites not owned by $P_\circ$. For two sheets and a region $Y$,
    the exchange gives the first sheet the owners of the second inside $Y$ and
    conversely; with $Z$ the set of sites whose two owners are not both
    $P_\circ$, $T=Z\cap Y$, $E=Z\setminus Y$ and $A=Y\setminus Z$, the
    implemented map is $C=D_UF_AF_Y$.
\end{definition}

\begin{lemma}[Sheets as encoded frames without holes]
    \label{lem:lattice_sheet_frame}
    \lean{TNLean.Lattice.Sheet.toFrame,
        TNLean.Lattice.Sheet.toFrame_regs,
        TNLean.Lattice.Sheet.toFrame_regIso,
        TNLean.Lattice.Sheet.toFrame_regRef,
        TNLean.Lattice.Sheet.toFrame_changeOwner}
    \leanok
    \uses{def:lattice_sheet,def:peps_frame_registers}
    Placed in the plane, a sheet is the encoded frame without holes with the
    same owners. Its registers are those of the frame, its product coordinates
    are those of the frame without the trivial tag coordinate, and handing a set
    of sites to another party commutes with the passage to frames. For $q\ge1$,
    where the reference vector of an encoded frame is defined, the two
    reference vectors agree.
\end{lemma}

\begin{proof}\leanok
    With no holes the encoder is the identity, so the frame places $\Omega$ in
    the site registers.
\end{proof}

\begin{theorem}[Birth and death of sheets: bounded change]
    \label{thm:lattice_sheet_birth_bounded}
    \lean{TNLean.Lattice.Sheet.birth_bounded,
        TNLean.Lattice.Sheet.death_bounded,
        TNLean.Lattice.Sheet.changeOwner_birth_owners}
    \leanok
    \uses{def:lattice_sheet,lem:peps_frame_splitting,def:peps_frame_birth_map,
        lem:peps_frame_birth_map,def:peps_monomial_parties}
    In a sheet, let $T$ be a set of sites owned by $P_\circ$, let $E$ be the set
    of sites not owned by $P_\circ$, let $\lVert\Omega\rVert=1$, let $L\ge0$ and
    let $p$ be real, and suppose $I_\Omega(T:E)\le L^{-2p}$. There are splitting data and an allowed
    monomial from the registers of the sheet to those of the sheet in which $T$
    is owned by $Q_\circ$, using only $P_\circ$ and $Q_\circ$ and at most one
    normalized pair source, whose operator in product coordinates is the birth
    map $B$, and which maps the reference vector within $L^{-p}$ of the
    reference vector of the new sheet. The reversed death obeys the same bound
    when these conditions hold in the sheet after the death, using one
    normalized pair effect and no pair source. For $p=30$ this is the case
    without holes of \cite[Lemma~6.5]{OpenAI2026PolynomialPEPS}
    (\texttt{05-frames.tex}, lines 396--410); in one dimension the exponent is
    a free parameter, used for instance with $2p=24$.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,thm:peps_frame_birth_bounded}
    Every site outside $T\cup E$ is owned by $P_\circ$ before and after the
    birth, and the sites of $E$ keep their owners. The monomials of
    Theorem~\ref{thm:peps_frame_birth_bounded}, placed behind the
    empty block of registers, act in product coordinates as $B$, and
    $\lVert(B-\id)\Omega\rVert$ is at most the splitting error, at most
    $L^{-p}$ by Lemma~\ref{lem:peps_frame_splitting}.
\end{proof}

\begin{theorem}[Exchange of two sheets: bounded change]
    \label{thm:lattice_sheet_exchange_bounded}
    \lean{TNLean.Lattice.Sheet.exchange_bounded,
        TNLean.Lattice.Sheet.renameWord,
        TNLean.Lattice.Sheet.twoRegIso_renameWord,
        TNLean.Lattice.Sheet.exchangeOp_eq}
    \leanok
    \uses{def:lattice_sheet,lem:peps_frame_splitting,
        lem:peps_frame_exchange_net_map,def:peps_monomial_parties}
    For two sheets, a region $Y$, a party $P_\circ$ with $Z$, $T$, $E$, $U$ as
    above, $\lVert\Omega\rVert=1$, $L\ge0$ and real $p$ with
    $I_\Omega(T:E)\le L^{-2p}$, there
    are splitting data and an allowed monomial from the registers of the two
    sheets to those of the two exchanged sheets, consisting of exchanges of
    tensor factors and one private contraction at $P_\circ$, so using only
    $P_\circ$ and no pair resource, whose operator in product coordinates is
    $C=D_UF_AF_Y=D_UF_T$, and which maps $\Omega\otimes\Omega$ within
    $4L^{-p}$ of $\Omega\otimes\Omega$. For $p=30$ this is the case without
    holes of \cite[Lemma~6.6]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 460--479), where the renaming acts in product coordinates as $F_Y$;
    in one dimension the exponent is a free parameter.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:peps_exchange_renaming_word,
        lem:peps_exchange_correction_word,
        lem:peps_frame_exchange_correction_monomial,lem:peps_frame_exchange_error,
        lem:peps_register_reorderings}
    The renaming of Lemma~\ref{lem:peps_exchange_renaming_word} with empty
    blocks of tags exchanges the configurations of the two sheets inside $Y$,
    so its operator is $F_Y$; the corrections act as $D_UF_A$. Since $Y$ is
    the disjoint union of $T$ and $A$, the product is $D_UF_T$, and
    Lemma~\ref{lem:peps_frame_exchange_error} bounds the reference error.
\end{proof}

\begin{definition}[Stacks of sheets]
    \label{def:lattice_sheet_stack}
    \lean{TNLean.Lattice.Sheet.refFamily,
        TNLean.Lattice.Sheet.HasBoundedChange,
        TNLean.Lattice.Sheet.HasSourceBoundedChange}
    \leanok
    \uses{def:lattice_sheet,def:peps_sheet_stack,def:peps_bounded_change}
    The memory of a list of sheets and its reference vector are those of
    Definition~\ref{def:peps_sheet_stack}, with the reference vector $\Omega$ of
    each sheet placed in its site registers. A bounded change of a list of
    sheets is a bounded change, in the sense of
    Definition~\ref{def:peps_bounded_change}, between these reference vectors;
    the variant with bounded pair sources also bounds the number of normalized
    pair sources.
\end{definition}

\begin{lemma}[Reference vectors of stacks of sheets]
    \label{lem:lattice_sheet_stack_reference}
    \lean{TNLean.Lattice.Sheet.norm_refFamily_le,
        TNLean.Lattice.Sheet.norm_framesVec,
        TNLean.Lattice.Sheet.pairRef_eq_twoRegIso_symm,
        TNLean.Lattice.Sheet.finiteDimensional_of_mem_stackRegs}
    \leanok
    \uses{def:lattice_sheet_stack}
    If $\lVert\Omega\rVert\le1$, the reference vector of every sheet has norm at
    most one; if $\lVert\Omega\rVert=1$, the reference vector of a list of sheets
    without further registers has norm one. The reference vector of two adjacent
    sheets is $\Omega\otimes\Omega$ in the product coordinates of the two sheets.
    Every register of a list of sheets is $\C^q$, so it is finite-dimensional.
\end{lemma}

\begin{proof}\leanok
    The reference vector of a sheet is the image of $\Omega$ under an isometry,
    and the norm of a tensor product of vectors is the product of their norms.
\end{proof}

\begin{theorem}[The initial step for sheets]
    \label{thm:lattice_sheet_initialization}
    \lean{TNLean.Lattice.Sheet.exists_originalCircuit_initial_of_owner}
    \leanok
    \uses{lem:lattice_sheet_stack_reference,thm:peps_sheet_initialization}
    Let $\lVert\Omega\rVert=1$ and let every sheet of a list be held entirely by
    one party. Then the product of the reference vectors of the sheets of every
    party has norm one, and one private map per party followed by a reordering
    carries the product of these vectors exactly to the reference vector of the
    list, with no nonprivate gate. This is the case without holes of
    Theorem~\ref{thm:peps_sheet_initialization}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:lattice_sheet_stack_reference,thm:peps_sheet_initialization}
    Every register of a sheet held by one party belongs to that party, and the
    reference vector of every list of sheets has norm one by
    Lemma~\ref{lem:lattice_sheet_stack_reference}. Apply
    Theorem~\ref{thm:peps_sheet_initialization}.
\end{proof}
