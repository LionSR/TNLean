\section{Changes of ownership in encoded frames}
\label{sec:peps_ownership_changes}
Changes of site and hole ownership in encoded frames
(\cite[Lemmas~6.5 and~6.6]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 347--570) take the mutual-information bounds as
hypotheses and realize the maps constructed in their proofs as bounded
changes: allowed monomials of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}
(Section~\ref{sec:peps_pair_effects}) on the registers of the frames, one per
site and per tag (Definition~\ref{def:peps_frame_registers}). The monomials are
first constructed on registers in which the site registers of a region held by
one party are grouped into one register, and then placed on the registers of
the frames by reorderings of tensor factors and grouping unitaries.

We keep the setting of Section~\ref{sec:peps_encoded_frames}. For a set
$S\subseteq\Lambda$ write $\mathcal H_S=\bigotimes_{x\in S}\C^q$. For a vector
$\Omega\in\mathcal H_\Lambda$ and disjoint sets $T,E\subseteq\Lambda$, the
mutual information $I_\Omega(T:E)$ is
$S(\rho_T)+S(\rho_E)-S(\rho_{TE})$ with natural logarithms, where the $\rho$
are the reduced density matrices of $\ket\Omega\bra\Omega$.

\begin{lemma}[Support-intersection bound for an operator on two regions]
    \label{thm:peps_supported_left_factor_rank}
    \lean{TNLean.Lattice.Sheet.exists_threeSplit_eq_kronecker_one_rank_le}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $\Lambda$ be a finite site set, let $q$ be any natural number, and
    let $T,E,D\subseteq\Lambda$ satisfy $T\cap E=\varnothing$ and
    $D\subseteq T\cup E$. Put $U=\Lambda\setminus(T\cup E)$, and let
    $A$ be an operator supported on $D$. There is an operator $B$ on
    $\mathcal H_T\otimes\mathcal H_E$ such that, in the three-region
    coordinates of Lemma~\ref{lem:peps_frame_split_forms},
    \begin{align}
        A                     & = B\otimes\id_U,\notag                         \\
        \operatorname{OSR}(B) & \le q^{2|D\cap T|}.\label{eq:peps_left_factor}
    \end{align}
    The rank in~(\ref{eq:peps_left_factor}) is taken across $\mathcal H_T:\mathcal H_E$, allowing an empty
    sum in the operator Schmidt decomposition. The statement includes
    $q=0$ and empty regions. In particular, $|D\cap T|=1$ gives the bound
    $q^2$. This support estimate applies to the local terms in the
    nearest-neighbour Hamiltonian decomposition of
    \cite[Section~4.1]{Arad2016RigorousRG}.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:peps_frame_split_forms,thm:area_law_supported_identity}
    If there is no global configuration, then $A=0$, and take $B=0$.
    Otherwise choose a global configuration and let $u_0$ be its restriction
    to $U$. Take $B$ from Lemma~\ref{lem:peps_frame_split_forms}.
    The support criterion of Lemma~\ref{thm:area_law_supported_identity}
    and the matrix-unit decomposition across a cut~\cite[``Matrix-unit
        decomposition across a cut'']{QICLean2026Blueprint} give
    \begin{align}
        A & =\sum_{\alpha=1}^{M}L_\alpha\otimes R_\alpha,\notag                \\
        M & \le\bigl(q^{|D\cap T|}\bigr)^2,\label{eq:peps_left_support_decomp}
    \end{align}
    in the coordinates
    $\mathcal H_T\otimes\mathcal H_{\Lambda\setminus T}$.
    For a configuration $e$ on $E$, let $\gamma(e)$ be the configuration on
    $\Lambda\setminus T=E\sqcup U$ equal to $e$ on $E$ and to $u_0$ on $U$.
    Restricting~(\ref{eq:peps_left_support_decomp}) at this fixed $u_0$ yields
    \begin{align}
        (\widehat R_\alpha)_{e,e'}
          & =(R_\alpha)_{\gamma(e),\gamma(e')},\notag                                                   \\
        B & =\sum_{\alpha=1}^{M}L_\alpha\otimes\widehat R_\alpha.\label{eq:peps_left_restricted_decomp}
    \end{align}
    The identity on $U$ has entry $1$ at $(u_0,u_0)$, so the restriction
    in~(\ref{eq:peps_left_restricted_decomp}) gives the same $B$. Its operator Schmidt rank is at most $M$.
\end{proof}

\begin{lemma}[Support-intersection bound for a complementary operator]
    \label{thm:peps_supported_right_factor_rank}
    \lean{TNLean.Lattice.Sheet.exists_threeSplit_eq_reindex_one_kronecker_rank_le}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $\Lambda$ be a finite site set, let $q$ be any natural number, and
    let $T,E,D\subseteq\Lambda$ satisfy $T\cap E=D\cap T=\varnothing$.
    Put $U=\Lambda\setminus(T\cup E)$, and let $A$ be an operator supported
    on $D$. There is an operator $V$ on $\mathcal H_E\otimes\mathcal H_U$
    such that, in the three-region coordinates of
    Lemma~\ref{lem:peps_frame_split_forms},
    \begin{align}
        A                     & =\id_T\otimes V,\notag                          \\
        \operatorname{OSR}(V) & \le q^{2|D\cap E|}.\label{eq:peps_right_factor}
    \end{align}
    The tensor product in~(\ref{eq:peps_right_factor}) is read after canonical reassociation into
    $(\mathcal H_T\otimes\mathcal H_E)\otimes\mathcal H_U$.
    Here the operator Schmidt rank is the least number of simple operator
    tensors in a decomposition, allowing the empty sum. The statement
    includes $q=0$ and empty regions. In particular, $|D\cap E|=1$ gives
    the bound $q^2$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:peps_frame_split_forms,thm:area_law_supported_identity}
    If there is no global configuration, then $A=0$, and take $V=0$.
    Otherwise choose a global configuration and let $t_0$ be its restriction
    to $T$. The support criterion of
    Lemma~\ref{thm:area_law_supported_identity} and the matrix-unit
    decomposition across a cut~\cite[``Matrix-unit decomposition across a
        cut'']{QICLean2026Blueprint} give, in the coordinates
    $\mathcal H_E\otimes\mathcal H_{\Lambda\setminus E}$,
    \begin{align}
        A & =\sum_{\alpha=1}^{M}L_\alpha\otimes R_\alpha,\notag                 \\
        M & \le\bigl(q^{|D\cap E|}\bigr)^2.\label{eq:peps_right_support_decomp}
    \end{align}
    The count in~(\ref{eq:peps_right_support_decomp}) is the square of the number of configurations on $D\cap E$;
    no factor-norm estimate is needed.
    Take $V$ from Lemma~\ref{lem:peps_frame_split_forms}.
    For a configuration $u$ on $U$, let $\gamma(u)$ be the configuration
    on $\Lambda\setminus E=T\sqcup U$ equal to $t_0$ on $T$ and to $u$ on $U$.
    Evaluating both tensor decompositions at the same first-region
    configuration $t_0$ gives
    \begin{align}
        (\widehat R_\alpha)_{u,u'}
          & =(R_\alpha)_{\gamma(u),\gamma(u')},\notag                                                    \\
        V & =\sum_{\alpha=1}^{M}L_\alpha\otimes\widehat R_\alpha.\label{eq:peps_right_restricted_decomp}
    \end{align}
    Thus~(\ref{eq:peps_right_restricted_decomp}) gives a decomposition of $V$ with the same $M$ summands and bounds its
    operator Schmidt rank. No injectivity of $\gamma$ is required.
\end{proof}

\begin{lemma}[Splitting consequence on a sheet]
    \label{lem:peps_frame_splitting}
    \lean{TNLean.Lattice.Sheet.SplittingData,
        TNLean.Lattice.Sheet.SplittingData.error,
        TNLean.Lattice.Sheet.splitVec,
        TNLean.Lattice.Sheet.partialTraceRight_splitVec,
        TNLean.Lattice.Sheet.quantumRelativeEntropy_splitVec_eq_mutualInformation,
        TNLean.Lattice.Sheet.exists_sheetSplitting,
        TNLean.Lattice.Sheet.exists_sheetSplitting_zpow,
        TNLean.Lattice.Sheet.exists_sheetSplitting_rpow}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $T,E\subseteq\Lambda$ be disjoint, $U=\Lambda\setminus(T\cup E)$,
    $\lVert\Omega\rVert=1$, and $b=I_\Omega(T:E)$. There exist
    finite-dimensional spaces $\mathcal B_T,\mathcal B_E$, an isometry
    $V\colon\mathcal H_U\to\mathcal B_T\otimes\mathcal B_E$, and unit vectors
    $s\in\mathcal H_T\otimes\mathcal B_T$, $s'\in\mathcal H_E\otimes\mathcal B_E$
    such that
    \begin{align}
        \lVert(\id_{TE}\otimes V)\Omega-s\otimes s'\rVert
        \le\sqrt{2(1-e^{-b/2})}\le\sqrt b .
        \label{eq:peps_frame_splitting}
    \end{align}
    In particular, for the sheet splitting of
    \cite[Lemma~6.4]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 352--368), $b\le L^{-60}$ gives an error of at most $L^{-30}$; more
    generally, for $L\ge0$ and real $p$, $b\le L^{-2p}$ gives an error of at
    most $L^{-p}$.
\end{lemma}

\begin{proof}\leanok
    Since $D(\rho_{TE}\,\|\,\rho_T\otimes\rho_E)=I_\Omega(T:E)=b$, applying the
    tripartite pure-state splitting bound to $\Omega\in\mathcal H_T\otimes\mathcal H_E\otimes\mathcal H_U$
    (Lemma~\ref{lem:peps_frame_split_forms}) with $\mathcal B_T=\mathcal H_T$
    and $\mathcal B_E=\mathcal H_E\oplus\mathcal H_U$ yields~(\ref{eq:peps_frame_splitting}).
\end{proof}

\begin{definition}[Birth map]
    \label{def:peps_frame_birth_map}
    \lean{TNLean.Lattice.Sheet.pairSource,
        TNLean.Lattice.Sheet.birthKernel,
        TNLean.Lattice.Sheet.birthEffect,
        TNLean.Lattice.Sheet.birthOp,
        TNLean.Lattice.Sheet.sheetBirthOp,
        TNLean.Lattice.Sheet.birthOp_eq,
        TNLean.Lattice.Sheet.pairSource_mul_conjTranspose_apply}
    \leanok
    \uses{lem:peps_frame_splitting}
    With the splitting data $(V,s,s')$ of Lemma~\ref{lem:peps_frame_splitting},
    set $\mathcal V=\id_{TE}\otimes V$ and let
    $S\colon\mathcal H_E\otimes\mathcal B_E\to\mathcal H_{TE}\otimes\mathcal B_T\otimes\mathcal B_E$
    act by $x\mapsto s\otimes x$. The birth map is
    \begin{align}
        B=\mathcal V^\dagger SS^\dagger\mathcal V
        =\mathcal V^\dagger\bigl(\ket s\bra s_{T\mathcal B_T}
        \otimes\id_{E\mathcal B_E}\bigr)\mathcal V
        \label{eq:peps_frame_birth_map}
    \end{align}
    (\texttt{05-frames.tex}, equation (birth-map), lines 414--426).
\end{definition}

\begin{lemma}[Properties of the birth map]
    \label{lem:peps_frame_birth_map}
    \lean{TNLean.Lattice.Sheet.pairSource_conjTranspose_mul_self,
        TNLean.Lattice.Sheet.pairSource_mulVec,
        TNLean.Lattice.Sheet.birthOp_isHermitian,
        TNLean.Lattice.Sheet.norm_birthOp_le_one,
        TNLean.Lattice.Sheet.norm_act_birthOp_sub_le,
        TNLean.Lattice.Sheet.birthEffect_eq_submatrix,
        TNLean.Lattice.Sheet.birthOp_eq_submatrix,
        TNLean.Lattice.Sheet.commute_birthOp,
        TNLean.Lattice.Sheet.sheetBirthOp_isHermitian,
        TNLean.Lattice.Sheet.norm_sheetBirthOp_le_one,
        TNLean.Lattice.Sheet.norm_act_sheetBirthOp_sub_le,
        TNLean.Lattice.Sheet.commute_sheetBirthOp}
    \leanok
    \uses{def:peps_frame_birth_map}
    The map $B$ (Definition~\ref{def:peps_frame_birth_map}) satisfies
    $B^\dagger=B$, $\|B\|\le 1$, $B=\id_E\otimes K^\dagger K$ with
    $K=\bra s(\id_T\otimes V)$ (so $[B,A]=0$ for any $A$ supported in $E$), and
    for every $\psi\in\mathcal H_\Lambda$,
    \begin{align}
        \lVert(B-\id)\psi\rVert\le\lVert\mathcal V\psi-s\otimes s'\rVert .
        \label{eq:peps_frame_birth_map_error}
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $\lVert s\rVert=1$, $S^\dagger S=\id$, so $\Pi=SS^\dagger$ is an
    orthogonal projection with $\Pi(s\otimes s')=s\otimes s'$. As
    $\mathcal V^\dagger\mathcal V=\id$,
    $(B-\id)\psi=\mathcal V^\dagger(\Pi-\id)(\mathcal V\psi-s\otimes s')$,
    giving~(\ref{eq:peps_frame_birth_map_error}). On
    $\mathcal H_E\otimes(\mathcal H_T\otimes\mathcal H_U)$,
    $S^\dagger\mathcal V=\id_E\otimes K$, so $B=\id_E\otimes K^\dagger K$.
\end{proof}

\begin{lemma}[Operators passing through an encoding]
    \label{lem:peps_frame_pass_encoding}
    \lean{TNLean.Lattice.Sheet.one_kronecker_mul_stack,
        TNLean.PEPS.EncodedFrame.one_kronecker_mul_frameEncoder,
        TNLean.PEPS.EncodedFrame.norm_act_one_kronecker_refVec_sub_le}
    \leanok
    \uses{def:peps_frame}
    If a sheet operator $B$ commutes with the sheet part of $K_{\mathcal F}$
    (Definition~\ref{def:peps_frame}) on every tag configuration, then
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F}B$ and
    $\lVert(\id_{\rm tags}\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F}\rVert\le\lVert B\Omega-\Omega\rVert$.
\end{lemma}

\begin{proof}\leanok
    Commutation on each tag configuration gives
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F}B$, and
    $\|K_{\mathcal F}\|\le 1$ gives the norm bound on
    $\Omega_{\mathcal F}=K_{\mathcal F}\Omega$.
\end{proof}

\begin{lemma}[Homogeneous birth and death: sheet map]
    \label{thm:peps_frame_birth}
    \lean{TNLean.PEPS.EncodedFrame.Frame.birth,
        TNLean.PEPS.EncodedFrame.Frame.death}
    \leanok
    \uses{lem:peps_frame_birth_map,lem:peps_frame_pass_encoding,
        def:peps_frame_birth_partition,lem:peps_frame_birth_partition_owners}
    In an encoded frame $\mathcal F$ with $\lVert\Omega\rVert=1$, let $T$ be
    owned by $P_\circ$ with $T\cap H^+(\mathcal F)=\varnothing$, and let
    $\mathcal F',E,U$ be as in Definition~\ref{def:peps_frame_birth_partition}.
    If $I_\Omega(T:E)\le L^{-60}$, the birth map $B$ associated with splitting
    data of error at most $L^{-30}$ satisfies $B^\dagger=B$, $\|B\|\le 1$, acts
    as $\id$ on $E$, commutes with $K_{\mathcal F}=K_{\mathcal F'}$ via
    $(\id_{\rm tags}\otimes B)K_{\mathcal F}=K_{\mathcal F'}B$, and obeys
    \begin{align}
        \lVert(\id_{\rm tags}\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F'}\rVert
        \le L^{-30}
        \label{eq:peps_frame_birth_bound}
    \end{align}
    (\cite[Lemma~6.5]{OpenAI2026PolynomialPEPS}). The death map is the same
    operator $B$ from $\mathcal F'$ to $\mathcal F$ when the hypotheses hold in
    the post-death frame $\mathcal F$.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:peps_frame_birth_map,
        lem:peps_frame_pass_encoding,def:peps_frame_birth_partition,
        lem:peps_frame_birth_partition_owners}
    By Lemma~\ref{lem:peps_frame_birth_partition_owners}, $T\cap E=\varnothing$,
    so Lemma~\ref{lem:peps_frame_splitting} provides splitting data with error
    $\le L^{-30}$. Since $H^+(\mathcal F)\subseteq E$, every hole encoder acts
    inside $E$ and commutes with $B$ by Lemma~\ref{lem:peps_frame_birth_map}.
    Applying Lemma~\ref{lem:peps_frame_pass_encoding} and~(\ref{eq:peps_frame_birth_map_error})
    with $\Omega_{\mathcal F'}=\Omega_{\mathcal F}$ yields~(\ref{eq:peps_frame_birth_bound}).
\end{proof}

\subsection{Sheets without holes}

None of the monomials above acts on the tag registers: the birth and death
monomials and the renaming of the sites act behind the tag registers, which may be
replaced by any block of registers in front of the site registers. With the
empty block they act on a sheet without holes, an owner for each site of a
finite set $\Lambda$, whose reference vector is $\Omega$ placed in the site
registers. This is the setting of the one-dimensional protocol, where no
three owners meet and no encoded holes are needed.

\begin{definition}[Sheets without holes]
    \label{def:lattice_sheet}
    \lean{TNLean.Lattice.Sheet,
        TNLean.Lattice.Sheet.regs,
        TNLean.Lattice.Sheet.regIso,
        TNLean.Lattice.Sheet.regRef,
        TNLean.Lattice.Sheet.twoRegIso,
        TNLean.Lattice.Sheet.changeOwner,
        TNLean.Lattice.Sheet.birthEnv,
        TNLean.Lattice.Sheet.swapIn,
        TNLean.Lattice.Sheet.exchangeEnv,
        TNLean.Lattice.Sheet.exchangeOp,
        TNLean.Lattice.unitProdIso₂_act_liftFront,
        TNLean.Lattice.Sheet.regIso_eq_frontLayoutIso,
        TNLean.Lattice.Sheet.twoRegIso_eq_twoFrontLayoutIso}
    \leanok
    \uses{def:lattice_site_registers,def:peps_frame_sheet_swap,def:peps_frame_buffer_correction}
    A sheet on $\Lambda$ is an owner for each site; its registers are one
    register $\C^q$ per site, held by its owner, identified with
    $\mathcal H_\Lambda$ in product coordinates, and its reference vector is
    $\Omega$. These coordinates are those of the empty block of registers in
    front of the site registers. For a party $P_\circ$ let $E$ be
    the set of sites not owned by $P_\circ$. For two sheets and a region $Y$,
    the exchange gives the first sheet the owners of the second inside $Y$ and
    conversely; with $Z$ the set of sites whose two owners are not both
    $P_\circ$, $T=Z\cap Y$, $E=Z\setminus Y$ and $A=Y\setminus Z$, the
    implemented map is $C=D_UF_AF_Y$.
\end{definition}

\begin{lemma}[Sheets as encoded frames without holes]
    \label{lem:peps_sheet_frame}
    \lean{TNLean.Lattice.Sheet.toFrame,
        TNLean.Lattice.Sheet.toFrame_regs,
        TNLean.Lattice.Sheet.toFrame_regIso,
        TNLean.Lattice.Sheet.toFrame_regRef,
        TNLean.Lattice.Sheet.toFrame_changeOwner}
    \leanok
    \uses{def:lattice_sheet,def:peps_frame_registers}
    Placed in the plane, a sheet is the encoded frame without holes with the
    same owners. Its registers are those of the frame, its product coordinates
    are those of the frame without the trivial tag coordinate, and handing a set
    of sites to another party commutes with the passage to frames. For $q\ge1$,
    where the reference vector of an encoded frame is defined, the two
    reference vectors agree.
\end{lemma}

\begin{proof}\leanok
    With no holes the encoder is the identity, so the frame places $\Omega$ in
    the site registers.
\end{proof}

\begin{lemma}[Birth and death of sheets: bounded change]
    \label{thm:lattice_sheet_birth_bounded}
    \lean{TNLean.Lattice.Sheet.birth_bounded,
        TNLean.Lattice.Sheet.death_bounded,
        TNLean.Lattice.Sheet.changeOwner_birth_owners}
    \leanok
    \uses{def:lattice_sheet,lem:peps_frame_splitting,def:peps_frame_birth_map,
        lem:peps_frame_birth_map,def:peps_monomial_parties}
    In a sheet, let $T$ be a set of sites owned by $P_\circ$, let $E$ be the set
    of sites not owned by $P_\circ$, let $\lVert\Omega\rVert=1$, let $L\ge0$ and
    let $p$ be real, and suppose $I_\Omega(T:E)\le L^{-2p}$. There are splitting data and an allowed
    monomial from the registers of the sheet to those of the sheet in which $T$
    is owned by $Q_\circ$, using only $P_\circ$ and $Q_\circ$ and at most one
    normalized pair source, whose operator in product coordinates is the birth
    map $B$, and which maps the reference vector within $L^{-p}$ of the
    reference vector of the new sheet. The reversed death obeys the same bound
    when these conditions hold in the sheet after the death, using one
    normalized pair effect and no pair source. For $p=30$ this is the case
    without holes of \cite[Lemma~6.5]{OpenAI2026PolynomialPEPS}
    (\texttt{05-frames.tex}, lines 396--410); in one dimension the exponent is
    a free parameter, used for instance with $2p=24$.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:lattice_front_births}
    Every site outside $T\cup E$ is owned by $P_\circ$ before and after the
    birth, and the sites of $E$ keep their owners. The monomials of
    Lemma~\ref{lem:lattice_front_births}, placed behind the
    empty block of registers, act in product coordinates as $B$, and
    $\lVert(B-\id)\Omega\rVert$ is at most the splitting error, at most
    $L^{-p}$ by Lemma~\ref{lem:peps_frame_splitting}.
\end{proof}

\begin{lemma}[Exchange of two sheets: bounded change]
    \label{thm:lattice_sheet_exchange_bounded}
    \lean{TNLean.Lattice.Sheet.exchange_bounded,
        TNLean.Lattice.Sheet.renameWord,
        TNLean.Lattice.Sheet.twoRegIso_renameWord,
        TNLean.Lattice.Sheet.exchangeOp_eq}
    \leanok
    \uses{def:lattice_sheet,lem:peps_frame_splitting,lem:lattice_exchange_net_map,
        def:peps_monomial_parties}
    For two sheets, a region $Y$, a party $P_\circ$ with $Z$, $T$, $E$, $U$ as
    above, $\lVert\Omega\rVert=1$, $L\ge0$ and real $p$ with
    $I_\Omega(T:E)\le L^{-2p}$, there
    are splitting data and an allowed monomial from the registers of the two
    sheets to those of the two exchanged sheets, consisting of exchanges of
    tensor factors and one private contraction at $P_\circ$, so using only
    $P_\circ$ and no pair resource, whose operator in product coordinates is
    $C=D_UF_AF_Y=D_UF_T$, and which maps $\Omega\otimes\Omega$ within
    $4L^{-p}$ of $\Omega\otimes\Omega$. For $p=30$ this is the case without
    holes of \cite[Lemma~6.6]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 460--479), where the renaming acts in product coordinates as $F_Y$;
    in one dimension the exponent is a free parameter.
\end{lemma}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:lattice_block_renaming_word,
        lem:lattice_front_correction_word,lem:lattice_exchange_correction_matrix,
        lem:peps_frame_exchange_error,lem:lattice_register_reorderings}
    The renaming of Lemma~\ref{lem:lattice_block_renaming_word} with empty
    blocks of tags exchanges the configurations of the two sheets inside $Y$,
    so its operator is $F_Y$; the corrections act as $D_UF_A$. Since $Y$ is
    the disjoint union of $T$ and $A$, the product is $D_UF_T$, and
    Lemma~\ref{lem:peps_frame_exchange_error} bounds the reference error.
\end{proof}

\begin{definition}[Bounded changes of a stack of sheets]
    \label{def:lattice_bounded_change}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.norm_eval_le_norm,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange,
        TNLean.Lattice.stackRegs,
        TNLean.Lattice.stackVec,
        TNLean.Lattice.norm_stackVec,
        TNLean.Lattice.norm_stackVec_le,
        TNLean.Lattice.framesVec,
        TNLean.Lattice.framesSpace,
        TNLean.Lattice.pairRef,
        TNLean.Lattice.HasBoundedChange,
        TNLean.Lattice.HasSourceBoundedChange}
    \leanok
    \uses{def:peps_circuit_source_bound}
    An original circuit is a bounded change from a vector $x$ to a vector $y$, with
    participants in a set $S$, bounds $r$ and $K$ and error $\varepsilon$, if it has
    at most one nonprivate gate, every participant of that gate lies in $S$, every
    monomial of its expansion has at most $r$ pair effects, its coefficient sum and
    its number of monomials are at most $K$, and it takes $x$ within $\varepsilon$ of
    $y$; every original circuit is a contraction. Let each sheet $F$ of some type
    carry a list of registers and a reference vector $\omega_F$ on them. The memory
    of a list $F_1,\dots,F_m$ of sheets followed by further registers $\ell$ is the
    tensor product of their registers, in order, and for a vector $\xi$ of $\ell$
    its reference vector is $\omega_{F_1}\otimes\cdots\otimes\omega_{F_m}\otimes\xi$,
    whose norm is the product of the norms of the factors; if every $\omega_F$ has
    norm at most one, this norm is at most $\lVert\xi\rVert$. Without further
    registers, the memory of the sheets is the whole private memory of a party
    holding them, and the reference vector of two sheets is
    $\omega_{F_1}\otimes\omega_{F_2}$ on their concatenated registers. A bounded
    change of the sheets $F_1,\dots,F_m$ into $F'_1,\dots,F'_n$ is an original
    circuit between the two memories, with the same further registers, that is a
    bounded change from one reference vector to the other; the variant with bounded
    pair sources also requires at most a given number of pair sources in each
    monomial.
\end{definition}

\begin{proof}\leanok
    The norm of a tensor product of vectors is the product of their norms, and the
    operator of an original circuit is a composition of contractions.
\end{proof}

\begin{lemma}[Operations that preserve bounded changes of a stack]
    \label{thm:lattice_bounded_change_closure}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.mono,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.comp_right,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.comp_left,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.isBoundedChange_id,
        TNLean.Lattice.PartyCircuit.Word.isBoundedChange_toOriginalCircuit,
        TNLean.Lattice.PartyCircuit.PartyChain.isBoundedChange_toOriginalCircuit,
        TNLean.Lattice.norm_eval_liftOne_stackVec_sub_le,
        TNLean.Lattice.nonprivateCount_liftTwo,
        TNLean.Lattice.participants_liftTwo_subset,
        TNLean.Lattice.liftOne,
        TNLean.Lattice.isMonomialBounded_liftTwo,
        TNLean.Lattice.participants_liftOne_subset,
        TNLean.Lattice.isExpansionBounded_liftTwo,
        TNLean.Lattice.liftTwo,
        TNLean.Lattice.norm_eval_liftTwo_stackVec_sub_le,
        TNLean.Lattice.isMonomialBounded_liftOne,
        TNLean.Lattice.isExpansionBounded_liftOne,
        TNLean.Lattice.nonprivateCount_liftOne,
        TNLean.Lattice.hasBoundedChange_activate,
        TNLean.Lattice.hasBoundedChange_liftTwo,
        TNLean.Lattice.hasBoundedChange_liftOne,
        TNLean.Lattice.exists_isBoundedChange_of_eq,
        TNLean.Lattice.HasBoundedChange.perm_left,
        TNLean.Lattice.HasBoundedChange.perm_right,
        TNLean.Lattice.hasBoundedChange_refl,
        TNLean.Lattice.isBoundedChange_liftTwo,
        TNLean.Lattice.hasBoundedChange_perm_liftOne,
        TNLean.Lattice.HasBoundedChange.mono,
        TNLean.Lattice.isBoundedChange_liftOne,
        TNLean.Lattice.hasBoundedChange_of_perm,
        TNLean.Lattice.stackVec_append,
        TNLean.Lattice.exists_reordering_stackVec,
        TNLean.Lattice.stackRegs_append,
        TNLean.Lattice.exists_originalCircuit_of_eq,
        TNLean.Lattice.exists_originalCircuit_stackVec_of_perm,
        TNLean.Lattice.hasSourceBoundedChange_refl,
        TNLean.Lattice.HasSourceBoundedChange.perm_right,
        TNLean.Lattice.hasSourceBoundedChange_activate,
        TNLean.Lattice.hasSourceBoundedChange_liftTwo,
        TNLean.Lattice.hasSourceBoundedChange_liftOne,
        TNLean.Lattice.HasSourceBoundedChange.mono,
        TNLean.Lattice.hasSourceBoundedChange_of_perm,
        TNLean.Lattice.isSourceBounded_liftOne,
        TNLean.Lattice.hasSourceBoundedChange_perm_liftOne,
        TNLean.Lattice.HasSourceBoundedChange.perm_left,
        TNLean.Lattice.HasSourceBoundedChange.hasBoundedChange,
        TNLean.Lattice.isSourceBounded_liftTwo}
    \leanok
    \uses{def:lattice_bounded_change,thm:peps_original_produced_resource_bounds}
    Let every reference vector of a sheet and the vector $\xi$ of the further
    registers have norm at most one.
    \begin{enumerate}
        \item A bounded change remains one after enlarging its participant set, its
              bounds or its error. Composing it, on either side, with a circuit without
              nonprivate gate that satisfies the same pair-effect, coefficient-sum and
              monomial bounds gives a bounded change with the same data, between the
              transported vectors. The circuit without operations is a bounded change
              between equal vectors, for every participant set and all bounds, and so is
              the change of a sheet into an equal sheet.
        \item The memory and the reference vector of a concatenation of two lists of
              sheets are the tensor products of those of the two lists. A permutation of
              the sheets is implemented by a word of exchanges of tensor factors, without
              nonprivate gate, carrying the reference vector of the first list exactly to
              that of the second; so it is a bounded change without error and without pair
              sources, and a permutation before or after a bounded change of the sheets
              gives a bounded change with the same data.
        \item A change of one sheet, or of two adjacent sheets, lifted to the whole
              stack with the other sheets and the further registers as spectators, has the
              same nonprivate gates, participants, pair-effect, coefficient-sum, monomial
              and pair-source bounds, and its reference error on the stack is at most its
              error on the sheets it acts on. Hence a bounded change of these sheets is a
              bounded change of the stack with the same data, also after a permutation of
              the stack that brings these sheets into place.
        \item An allowed word using only the parties of $S$ with at most one pair source
              is a bounded change with its own error, for every $r$ and every $K\ge1$. An
              allowed chain using only the parties of $S$, without pair sources and with at
              most one pair effect, is a bounded change for every $r\ge1$ and $K\ge1$.
    \end{enumerate}
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_original_produced_resource_bounds}
    The spectator sheets of a lifted change have reference vectors of norm at most
    one, so the error on the stack is the error on the acting sheets times norms at
    most one. Composing with a circuit without nonprivate gate adds no gate and
    keeps the bounds. A permutation of the sheets is a composition of exchanges of
    adjacent blocks of registers. The resource bounds of the original circuit of a
    word or a chain follow from its counts of pair sources and pair effects.
\end{proof}

\begin{lemma}[The initial private maps of a stack of sheets]
    \label{thm:lattice_stack_initialization}
    \lean{TNLean.Lattice.splitMap,

    \leanok

    to that party. Give each party, as its whole initial private memory, the
    registers of its sheets in the order of the list, and as its initial vector the
    product of their reference vectors, assumed of norm one. Then one private map per
    party, followed by a reordering of the sheets, carries the product of the initial
    vectors of all parties exactly to the reference vector of the list, with no
    nonprivate gate.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:lattice_bounded_change_closure}
    Each private map identifies the whole memory of a party with the memory of its
    sheets. The concatenation, over the parties, of their lists of sheets is a
    permutation of the list, which a reordering implements.
\end{proof}
