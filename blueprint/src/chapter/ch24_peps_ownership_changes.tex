\section{Changes of ownership in encoded frames}
\label{sec:peps_ownership_changes}
Changes of site and hole ownership in encoded frames
(\cite[Lemmas~6.5 and~6.6]{OpenAI2026PolynomialPEPS}, September 24, 2026,
\texttt{05-frames.tex}, lines 347--570) take the mutual-information bounds as
hypotheses and realize the maps constructed in their proofs as bounded
changes: allowed monomials of \cite[Theorem~5.2]{OpenAI2026PolynomialPEPS}
(Section~\ref{sec:peps_pair_effects}) on the registers of the frames, one per
site and per tag (Definition~\ref{def:peps_frame_registers}). The monomials are
first constructed on registers in which the site registers of a region held by
one party are grouped into one register, and then placed on the registers of
the frames by reorderings of tensor factors and grouping unitaries.

We keep the setting of Section~\ref{sec:peps_encoded_frames}. For a set
$S\subseteq\Lambda$ write $\mathcal H_S=\bigotimes_{x\in S}\C^q$. For a vector
$\Omega\in\mathcal H_\Lambda$ and disjoint sets $T,E\subseteq\Lambda$, the
mutual information $I_\Omega(T:E)$ is
$S(\rho_T)+S(\rho_E)-S(\rho_{TE})$ with natural logarithms, where the $\rho$
are the reduced density matrices of $\ket\Omega\bra\Omega$.

\subsection{Splitting a sheet}

\begin{lemma}[Operators on parts of a sheet]
    \label{lem:peps_frame_split_forms}
    \lean{TNLean.Lattice.sheetSplit,
        TNLean.Lattice.Sheet.threeSplit,
        TNLean.Lattice.Sheet.threeSplit_fst,
        TNLean.Lattice.Sheet.threeSplit_snd,
        TNLean.Lattice.rectKronecker_submatrix_sheetSplit,
        TNLean.Lattice.Sheet.rectKronecker_submatrix_threeSplit,
        TNLean.Lattice.exists_kronecker_one_of_mem_supportedOperators,
        TNLean.Lattice.exists_one_kronecker_of_mem_supportedOperators,
        TNLean.Lattice.Sheet.exists_threeSplit_eq_kronecker_one,
        TNLean.Lattice.Sheet.exists_threeSplit_eq_one_kronecker_kronecker_one,
        TNLean.Lattice.Sheet.exists_threeSplit_eq_reindex_one_kronecker}
    \leanok
    Let $\Lambda$ be a finite site set and $q$ any natural number. For
    $S\subseteq\Lambda$, identify $\mathcal H_\Lambda$ with $\mathcal H_S\otimes\mathcal H_{\Lambda\setminus S}$
    and, for disjoint $T,E\subseteq\Lambda$ with
    $U=\Lambda\setminus(T\cup E)$, with $(\mathcal H_T\otimes\mathcal H_E)\otimes
        \mathcal H_U$. The latter identification reads a configuration $c$ as
    $((c|_T,c|_E),c|_U)$. An operator acting on the sites of $S$ is
    $A'\otimes\id$, an
    operator acting on a set disjoint from $S$ is $\id\otimes A'$, an operator
    acting on $T\cup E$ is $A'\otimes\id_U$, and an operator acting on a subset
    of $E$ is $\id_T\otimes A'\otimes\id_U$. An operator supported on a set
    disjoint from $T$ has the form $\id_T\otimes V$, where $V$ acts on
    $\mathcal H_E\otimes\mathcal H_U$, after canonical reassociation into
    the same coordinates. These assertions include $q=0$ and empty regions.
\end{lemma}

\begin{proof}\leanok
    A product operator $\bigotimes_xm_x$ is, in these coordinates, the tensor
    product of its restrictions to the parts, and $m_x=\id$ off the support.
    Both sides are linear in the operator.
    For an operator supported disjointly from $T$, first split off $T$ to
    write it as $\id_T\otimes B$ on
    $\mathcal H_T\otimes\mathcal H_{\Lambda\setminus T}$.
    Since $\Lambda\setminus T=E\sqcup U$, express $B$ in the coordinates
    $\mathcal H_E\otimes\mathcal H_U$ and call the resulting operator $V$.
    For configurations $t,t'$ on $T$, $e,e'$ on $E$ and $u,u'$ on $U$,
    the entries in the three-region coordinates are
    \begin{align}
        A_{((t,e),u),((t',e'),u')}
         & =\delta_{t,t'}V_{(e,u),(e',u')}.\notag
    \end{align}
    This is precisely $\id_T\otimes V$ after reassociation. The coordinate
    identities remain valid when a configuration set is empty.
\end{proof}

\begin{theorem}[Support-intersection bound for an operator on two regions]
    \label{thm:peps_supported_left_factor_rank}
    \lean{TNLean.Lattice.Sheet.exists_threeSplit_eq_kronecker_one_rank_le}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $\Lambda$ be a finite site set, let $q$ be any natural number, and
    let $T,E,D\subseteq\Lambda$ satisfy $T\cap E=\varnothing$ and
    $D\subseteq T\cup E$. Put $U=\Lambda\setminus(T\cup E)$, and let
    $A$ be an operator supported on $D$. There is an operator $B$ on
    $\mathcal H_T\otimes\mathcal H_E$ such that, in the three-region
    coordinates of Lemma~\ref{lem:peps_frame_split_forms},
    \begin{align}
        A                     & = B\otimes\id_U,\notag    \\
        \operatorname{OSR}(B) & \le q^{2|D\cap T|}.\notag
    \end{align}
    The rank is taken across $\mathcal H_T:\mathcal H_E$, allowing an empty
    sum in the operator Schmidt decomposition. The statement includes
    $q=0$ and empty regions. In particular, $|D\cap T|=1$ gives the bound
    $q^2$. This support estimate applies to the local terms in the
    nearest-neighbour Hamiltonian decomposition of
    \cite[Section~5]{Arad2016RigorousRG}.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:peps_frame_split_forms,thm:area_law_supported_identity}
    If there is no global configuration, then $A=0$, and take $B=0$.
    Otherwise choose a global configuration and let $u_0$ be its restriction
    to $U$. Take $B$ from Lemma~\ref{lem:peps_frame_split_forms}.
    The support criterion of Theorem~\ref{thm:area_law_supported_identity}
    and the matrix-unit decomposition across a cut~\cite[``Matrix-unit
        decomposition across a cut'']{QICLean2026Blueprint} give
    \begin{align}
        A & =\sum_{\alpha=1}^{M}L_\alpha\otimes R_\alpha,\notag \\
        M & \le\bigl(q^{|D\cap T|}\bigr)^2,\notag
    \end{align}
    in the coordinates
    $\mathcal H_T\otimes\mathcal H_{\Lambda\setminus T}$.
    For a configuration $e$ on $E$, let $\gamma(e)$ be the configuration on
    $\Lambda\setminus T=E\sqcup U$ equal to $e$ on $E$ and to $u_0$ on $U$.
    Restricting the outside factors at this fixed $u_0$ yields
    \begin{align}
        (\widehat R_\alpha)_{e,e'}
          & =(R_\alpha)_{\gamma(e),\gamma(e')},\notag                   \\
        B & =\sum_{\alpha=1}^{M}L_\alpha\otimes\widehat R_\alpha.\notag
    \end{align}
    The identity on $U$ has entry $1$ at $(u_0,u_0)$, so this restriction
    gives the same $B$. Its operator Schmidt rank is at most $M$.
\end{proof}

\begin{lemma}[Splitting consequence on a sheet]
    \label{lem:peps_frame_splitting}
    \lean{TNLean.Lattice.Sheet.SplittingData,
        TNLean.Lattice.Sheet.SplittingData.error,
        TNLean.Lattice.Sheet.splitVec,
        TNLean.Lattice.Sheet.partialTraceRight_splitVec,
        TNLean.Lattice.Sheet.quantumRelativeEntropy_splitVec_eq_mutualInformation,
        TNLean.Lattice.Sheet.exists_sheetSplitting,
        TNLean.Lattice.Sheet.exists_sheetSplitting_zpow,
        TNLean.Lattice.Sheet.exists_sheetSplitting_rpow}
    \leanok
    \uses{lem:peps_frame_split_forms}
    Let $T,E\subseteq\Lambda$ be disjoint, $U=\Lambda\setminus(T\cup E)$,
    $\lVert\Omega\rVert=1$ and $b=I_\Omega(T:E)$. There are finite-dimensional
    spaces $\mathcal B_T,\mathcal B_E$, an isometry
    $V\colon\mathcal H_U\to\mathcal B_T\otimes\mathcal B_E$ and unit vectors
    $s\in\mathcal H_T\otimes\mathcal B_T$, $s'\in\mathcal H_E\otimes\mathcal
        B_E$ with
    \begin{align}
        \lVert(\id_{TE}\otimes V)\Omega-s\otimes s'\rVert
        \le\sqrt{2(1-e^{-b/2})}\le\sqrt b .
    \end{align}
    In particular, for the sheet splitting of
    \cite[Lemma~6.4]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 352--368), $b\le L^{-60}$ gives an error of at most $L^{-30}$; more
    generally, for $L\ge0$ and real $p$, $b\le L^{-2p}$ gives an error of at
    most $L^{-p}$.
\end{lemma}

\begin{proof}\leanok
    The relative entropy of $\rho_{TE}$ with respect to $\rho_T\otimes\rho_E$
    is $b$. The splitting estimate for a unit vector on three systems, with
    $\mathcal B_T=\mathcal H_T$ and $\mathcal B_E=\mathcal H_E\oplus\mathcal
        H_U$, applies in the coordinates $\mathcal H_T\otimes\mathcal
        H_E\otimes\mathcal H_U$.
\end{proof}

\begin{definition}[Birth map]
    \label{def:peps_frame_birth_map}
    \lean{TNLean.Lattice.Sheet.pairSource,
        TNLean.Lattice.Sheet.birthKernel,
        TNLean.Lattice.Sheet.birthEffect,
        TNLean.Lattice.Sheet.birthOp,
        TNLean.Lattice.Sheet.sheetBirthOp,
        TNLean.Lattice.Sheet.birthOp_eq,
        TNLean.Lattice.Sheet.pairSource_mul_conjTranspose_apply}
    \leanok
    \uses{lem:peps_frame_splitting}
    With the data of the splitting lemma put $\mathcal V=\id_{TE}\otimes V$ and
    let $S\colon\mathcal H_E\otimes\mathcal B_E\to\mathcal
        H_{TE}\otimes\mathcal B_T\otimes\mathcal B_E$ insert $s$ on
    $T\mathcal B_T$. The birth map is
    \begin{align}
        B=\mathcal V^\dagger SS^\dagger\mathcal V
        =\mathcal V^\dagger\bigl(\ket s\bra s_{T\mathcal B_T}
        \otimes\id_{E\mathcal B_E}\bigr)\mathcal V
    \end{align}
    (\texttt{05-frames.tex}, equation (birth-map), lines 414--426). It is the
    effect $S^\dagger\mathcal V$, which applies $V$ on $U$ and contracts
    $T\mathcal B_T$ with $\bra s$, followed by its adjoint, which inserts
    $s$ and applies $V^\dagger$.
\end{definition}

\begin{lemma}[Properties of the birth map]
    \label{lem:peps_frame_birth_map}
    \lean{TNLean.Lattice.Sheet.pairSource_conjTranspose_mul_self,
        TNLean.Lattice.Sheet.pairSource_mulVec,
        TNLean.Lattice.Sheet.birthOp_isHermitian,
        TNLean.Lattice.Sheet.norm_birthOp_le_one,
        TNLean.Lattice.Sheet.norm_act_birthOp_sub_le,
        TNLean.Lattice.Sheet.birthEffect_eq_submatrix,
        TNLean.Lattice.Sheet.birthOp_eq_submatrix,
        TNLean.Lattice.Sheet.commute_birthOp,
        TNLean.Lattice.Sheet.sheetBirthOp_isHermitian,
        TNLean.Lattice.Sheet.norm_sheetBirthOp_le_one,
        TNLean.Lattice.Sheet.norm_act_sheetBirthOp_sub_le,
        TNLean.Lattice.Sheet.commute_sheetBirthOp}
    \leanok
    \uses{def:peps_frame_birth_map}
    The map $B$ is a Hermitian contraction, it acts as the identity on
    $\mathcal H_E$, so it commutes with every operator acting on a subset of
    $E$, and for every vector $\psi$
    \begin{align}
        \lVert(B-\id)\psi\rVert\le\lVert\mathcal V\psi-s\otimes s'\rVert .
    \end{align}
\end{lemma}

\begin{proof}\leanok
    Since $s$ is a unit vector, $S^\dagger S=\id$, so $\Pi=SS^\dagger$ is an
    orthogonal projection with $\Pi(s\otimes s')=s\otimes s'$. As
    $\mathcal V^\dagger\mathcal V=\id$,
    $(B-\id)\psi=\mathcal V^\dagger(\Pi-\id)(\mathcal V\psi-s\otimes s')$, and
    $\id-\Pi$ is a projection. In the coordinates
    $\mathcal H_E\otimes(\mathcal H_T\otimes\mathcal H_U)$ the effect
    $S^\dagger\mathcal V$ is $\id_E\otimes K$ with $K=\bra s(\id_T\otimes V)$,
    so $B=\id_E\otimes K^\dagger K$.
\end{proof}

\begin{lemma}[Operators passing through an encoding]
    \label{lem:peps_frame_pass_encoding}
    \lean{TNLean.Lattice.Sheet.one_kronecker_mul_stack,
        TNLean.PEPS.EncodedFrame.one_kronecker_mul_frameEncoder,
        TNLean.PEPS.EncodedFrame.norm_act_one_kronecker_refVec_sub_le}
    \leanok
    \uses{def:peps_frame}
    If a sheet operator $B$ commutes with the sheet part of the encoding of a
    frame on every tag configuration, then $(\id_{\rm tags}\otimes
        B)K_{\mathcal F}=K_{\mathcal F}B$ and
    $\lVert(\id\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F}\rVert
        \le\lVert B\Omega-\Omega\rVert$.
\end{lemma}

\begin{proof}\leanok
    Comparing the two sides on each tag gives the intertwining identity, and
    contractivity of the encoding gives the norm bound.
\end{proof}

\begin{definition}[Birth partition]
    \label{def:peps_frame_birth_partition}
    \lean{TNLean.PEPS.EncodedFrame.Frame.changeOwner,

    \leanok

    For an encoded frame $\mathcal F$, a party $P_\circ$ and a set $T$ of
    sites, let $\mathcal F'$ be the frame in which the site registers of $T$
    are owned by $Q_\circ$, with every hole, encoding and tag owner kept, and
    \begin{align}
        E=\{x:\text{the site owner of }x\neq P_\circ\}\cup H^+(\mathcal F),
        \qquad U=\Lambda\setminus(T\cup E)
    \end{align}
    (\texttt{05-frames.tex}, equation (birth-partition), lines 400--405).
\end{definition}

\begin{lemma}[Owners in the birth partition]
    \label{lem:peps_frame_birth_partition_owners}
    \lean{TNLean.PEPS.EncodedFrame.Frame.owner_eq_of_notMem_birthEnv,
        TNLean.PEPS.EncodedFrame.Frame.changeOwner_owner_of_notMem_birthEnv,
        TNLean.PEPS.EncodedFrame.Frame.disjoint_birthEnv}
    \leanok
    \uses{def:peps_frame_birth_partition}
    Every register outside $E$ is owned by $P_\circ$ in $\mathcal F$, and by
    $P_\circ$ or $Q_\circ$ in $\mathcal F'$. If $T$ is owned by $P_\circ$ and
    avoids every outer hole footprint, then $T$ and $E$ are disjoint
    (\texttt{05-frames.tex}, lines 397--405).
\end{lemma}

\begin{proof}\leanok
    A register outside $E$ has old owner $P_\circ$ by the definition of $E$;
    in $\mathcal F'$ its owner is $Q_\circ$ if it lies in $T$ and its old
    owner otherwise. A register of $T$ has old owner $P_\circ$ and lies in no
    outer hole footprint, so it is in neither part of $E$.
\end{proof}

\begin{theorem}[Homogeneous birth and death: sheet map]
    \label{thm:peps_frame_birth}
    \lean{TNLean.PEPS.EncodedFrame.Frame.birth,
        TNLean.PEPS.EncodedFrame.Frame.death}
    \leanok
    \uses{lem:peps_frame_birth_map,lem:peps_frame_pass_encoding,
        def:peps_frame_birth_partition,lem:peps_frame_birth_partition_owners}
    In an encoded frame $\mathcal F$, let $T$ be owned by $P_\circ$ and
    disjoint from every outer hole footprint, let $E$ be as in the birth
    partition, and let $\lVert\Omega\rVert=1$. If $I_\Omega(T:E)\le L^{-60}$,
    the birth map $B$ of splitting data with error at most $L^{-30}$ is a
    Hermitian contraction, acts as the identity on $E$ and on all tags,
    satisfies $(\id\otimes B)K_{\mathcal F}=K_{\mathcal F'}B$, and
    \begin{align}
        \lVert(\id\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal F'}\rVert
        \le L^{-30}.
    \end{align}
    Since $\mathcal F$ and $\mathcal F'$ differ only in site owners,
    $K_{\mathcal F'}=K_{\mathcal F}$ and $\Omega_{\mathcal
            F'}=\Omega_{\mathcal F}$, so the intertwining identity says that $B$
    commutes with the encoding; the change of owner is the content of the
    bounded-change assertion below. In the operator statement of
    \cite[Lemma~6.5]{OpenAI2026PolynomialPEPS}, the death is the same map with the
    reverse input and output ownership, subject to the conditions of the frame
    after the death.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:peps_frame_birth_map,
        lem:peps_frame_pass_encoding,def:peps_frame_birth_partition,
        lem:peps_frame_birth_partition_owners}
    Every hole encoder acts on $H^+(\mathcal F)\subseteq E$, so it commutes
    with $B$. Hence $\lVert(\id\otimes B)\Omega_{\mathcal F}-\Omega_{\mathcal
            F}\rVert\le\lVert(B-\id)\Omega\rVert\le\lVert\mathcal V\Omega-s\otimes
        s'\rVert\le L^{-30}$, and $\Omega_{\mathcal F'}=\Omega_{\mathcal F}$ in
    product coordinates, since only site owners change.
\end{proof}

\begin{definition}[Parties used by a monomial]
    \label{def:peps_monomial_parties}
    \lean{TNLean.Lattice.PartyCircuit.Word.UsesOnly,
        TNLean.Lattice.PartyCircuit.Word.sourceCount,
        TNLean.Lattice.PartyCircuit.PartyChain.UsesOnly,
        TNLean.Lattice.PartyCircuit.PartyChain.sourceCount}
    \leanok
    \uses{def:peps_party_layout,def:peps_party_monomial}
    An allowed monomial on parties uses only the parties of a set $S$ when
    each of its local maps acts at a party of $S$ and each of its pair sources
    and pair effects joins two parties of $S$; reorderings of tensor factors
    use no party. Its numbers of pair sources and of pair effects are counted
    along the composition.
\end{definition}

\begin{theorem}[Composition without additional pair effects]
    \label{thm:peps_monomial_composition_counts}
    \lean{TNLean.Lattice.PartyCircuit.PartyChain.precomp_props,
        TNLean.Lattice.PartyCircuit.PartyChain.postcomp_props,
        TNLean.Lattice.PartyCircuit.PartyChain.precomp,
        TNLean.Lattice.PartyCircuit.PartyChain.postcomp,
        TNLean.Lattice.PartyCircuit.PartyChain.eval_precomp,
        TNLean.Lattice.PartyCircuit.PartyChain.eval_postcomp}
    \leanok
    \uses{def:peps_monomial_parties}
    Let $M$ be an allowed monomial using only parties in $S$. Let $W_0$ and
    $W_1$ be allowed compositions containing no pair effects, also using only
    parties in $S$, whose output and input registers respectively match the
    input and output registers of $M$. Let $M_0$ be the monomial formed by
    preceding $M$ by $W_0$, and $M_1$ the monomial formed by following $M$
    by $W_1$. Both are allowed and use only parties in $S$. If $F_N$ denotes
    the operator of a monomial $N$, then $F_{M_0}=F_M\circ F_{W_0}$ and
    $F_{M_1}=F_{W_1}\circ F_M$. Writing $s(N)$ and $e(N)$ for its numbers
    of pair sources and pair effects, the counts satisfy
    \begin{align}
        s(M_0) & =s(W_0)+s(M),\notag \\
        s(M_1) & =s(M)+s(W_1),\notag \\
        e(M_0) & =e(M_1)=e(M).
    \end{align}
\end{theorem}

\begin{proof}\leanok
    For the preceding composition, combine $W_0$ with the first segment
    containing no pair effects. For the following composition, induct along the
    effects of $M$ and combine its final segment with $W_1$. The local
    contraction and normalized-source conditions are preserved by
    composition, the source counts add, and every original effect is kept
    exactly once. The operators compose in the stated order.
\end{proof}

    pair effect, and the death one pair effect and no pair source. With the
    registers $\mathcal H_T$ and $\mathcal H_U$ identified with $\mathcal

    the untouched registers, and on the sheet $B=\id_E\otimes K^\dagger K$.
\end{lemma}

\begin{proof}\leanok

    $\lVert K\rVert\le1$, and $\lVert V\rVert\le1$. For the birth, on $t\otimes
        u\otimes w$ the first contraction gives $k=K(t\otimes u)$; preparing
    $s=\sum t'\otimes b$ and applying $V^\dagger$ to $b\otimes k$ gives
    $\sum t'\otimes V^\dagger(b\otimes k)=K^\dagger k$, by comparing
    coordinates and using linearity in the prepared vector. For the death,
    $V u=\sum b\otimes e$ and the effect gives $\sum\langle s,t\otimes
        b\rangle e=K(t\otimes u)$, after which $K^\dagger$ is applied.
\end{proof}

\begin{lemma}[Reordering registers]
    \label{lem:peps_register_reorderings}
    \lean{TNLean.PEPS.EncodedFrame.tagSplit,
        TNLean.PEPS.EncodedFrame.tagMerge,
        TNLean.PEPS.EncodedFrame.eval_tagSplit,
        TNLean.PEPS.EncodedFrame.eval_tagMerge,
        TNLean.PEPS.EncodedFrame.exists_tagWord_of_perm,
        TNLean.PEPS.EncodedFrame.isReordering_tagSplit,
        TNLean.PEPS.EncodedFrame.isReordering_tagMerge}
    \leanok
    \uses{def:peps_frame_registers,lem:peps_frame_registers_coordinates,def:peps_monomial_parties,
        lem:lattice_register_reorderings}
    For a list of sites $\ell$ and a condition $f$, let $\ell_+$ and
    $\ell_-$ retain the sites satisfying $f$ and not satisfying $f$,
    respectively, in their original order. Thus $x\in\ell_+$ exactly when
    $x\in\ell$ and $f(x)$, and $x\in\ell_-$ exactly when $x\in\ell$ and
    $\neg f(x)$. If $\ell$ has no repeated site, neither part has one.
    The following compositions of exchanges of adjacent tensor factors are
    allowed monomials that use no party and no pair resource, and each sends
    product basis vectors to the indicated product basis vectors: moving the
    site registers of the sites satisfying a condition in front of the others,
    keeping the order within both parts, and back; reassociating and
    interchanging blocks of registers; splitting the tag registers of a
    concatenated list of holes into those of the two parts, and merging them.
    These reorderings may be performed before arbitrary untouched registers.
    One step of the site partition starts with a register $x$ followed by
    the already separated lists $A,B$: it leaves $x$ first in $A$ if selected,
    and otherwise moves it past $A$ to the front of $B$. The inverse step
    reverses this move.
    Adjoining or removing an empty trailing list is an identity and uses no
    party or pair resource. Naming a register with an equal owner label moves
    nothing. If a second list of holes is a reordering of the first, with
    pairwise disjoint outer
    footprints, there are a relabelling of the tags that preserves the sheet
    part of the encoding and such a composition sending $\ket\tau$ to the
    relabelled $\ket\tau$ (the identification of tag orderings in
    \texttt{05-frames.tex}, lines 84--85).

    More generally, adjoining untouched registers before any composition
    containing no pair effects preserves whether it is allowed, whether it
    uses only the parties of a given set, and its number of pair sources.
    Its operator is the identity on the added registers tensored with the
    original operator, under the prescribed tensor identifications.
\end{lemma}

\begin{proof}\leanok
    \uses{thm:peps_register_exchange}
    Each composition is built by induction on the lists. For the partition
    step, let $A,B$ be two site lists and $C$ an arbitrary untouched register
    list. In concatenations below, site lists and sites denote their site
    register lists and individual site registers. Write $J_{A,C}$ for the
    isometry separating concatenated
    lists, and let $(A,C)$ denote their concatenation. For the product basis
    vectors $u=\ket{c(x)}$, $a=\ket c_A$, $b=\ket c_B$ and any $y$ in the
    memory of $C$, put $t=J_{B,C}^{-1}(b\otimes y)$ and
    $z=J_{A,(B,C)}^{-1}(a\otimes t)$. The selected and unselected steps satisfy
    \begin{align}
        \Pi_+(u\otimes z) & =u\otimes z,\notag                         \\
        \Pi_-(u\otimes z) & =J_{A,(x,B,C)}^{-1}(a\otimes(u\otimes t)).
        \label{eq:peps_register_partition_step}
    \end{align}
    The inverse steps read (\ref{eq:peps_register_partition_step}) from right
    to left. They are the identity and the move of $x$ back past $A$,
    respectively. This proves the partition identities by induction; the
    two lists are obtained by retaining sites with the indicated condition,
    which gives the membership and non-repetition assertions. For a
    reordering of the holes, induct on the reordering: an exchange of two
    adjacent holes is an exchange of their tag registers, and the sheet parts of
    the encodings of the two holes commute by disjointness of supports.
    Adjoining untouched registers changes none of the original operations,
    so induction on the added list preserves each of the stated properties.
\end{proof}

\begin{theorem}[Coordinates of ordered region registers]
    \label{thm:peps_region_register_coordinates}
    \lean{TNLean.Lattice.Sheet.mem_listT,

    to obtain $\ell_U$, with the remaining sites forming $\ell_E$.

    \begin{align}
        x\in\ell_T & \iff x\in T,\notag    \\
        x\in\ell_R & \iff x\notin T,\notag \\
        x\in\ell_U & \iff x\in U.
    \end{align}
    If also $T\cap E=\varnothing$, then $x\in\ell_E$ if and only if $x\in E$.

    position gives bijections from the positions of $\ell_T$ to $T$ and of
    $\ell_U$ to $U$. For every configuration $c$ and any assignment of
    owners to the sites, the corresponding grouping isometries $G_T,G_U$
    satisfy
    \begin{align}
        G_T\ket c_{\ell_T} & =\ket{c|_T},\notag \\
        G_U\ket c_{\ell_U} & =\ket{c|_U}
        \label{eq:peps_region_grouping_basis}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_register_reorderings,lem:peps_frame_registers_coordinates}
    Filtering gives the membership assertions and preserves the absence of
    repetitions. Outside $T$, membership in $T\cup E$ is membership in $E$;
    disjointness ensures that all sites of $E$ remain. Completeness and
    non-repetition make the position maps bijective. The product-coordinate
    identification sends a product basis vector to the configuration read
    along that position map, giving (\ref{eq:peps_region_grouping_basis}).
\end{proof}

\begin{theorem}[Reconstructing a configuration from its restrictions]
    \label{thm:peps_region_configuration}
    \lean{TNLean.Lattice.Sheet.merge_restrict_T,
        TNLean.Lattice.Sheet.merge_restrict_U,

    coordinate identification applied to $((a,c|_E),b)$. Its restrictions are
    \begin{align}
        m(a,b;c)|_T & =a,\notag    \\
        m(a,b;c)|_E & =c|_E,\notag \\
        m(a,b;c)|_U & =b.
        \label{eq:peps_region_reconstruction}
    \end{align}

    $c'=m(a,b;c)$ if and only if
    $(c'|_T,c'|_U)=(a,b)$ and $c'|_E=c|_E$.

\end{theorem}

\begin{proof}\leanok
    The inverse coordinate map assigns the prescribed value on each of the
    three disjoint regions, proving (\ref{eq:peps_region_reconstruction}).
    Every site belongs to exactly one region, so equality of the restrictions
    is equivalent to equality of the configurations. Taking $a=c|_T$ gives
    the last assertion.
\end{proof}

\begin{lemma}[Composing and commuting sheet swaps]
    \label{lem:peps_frame_sheet_swap_compose}
    \lean{TNLean.Lattice.sheetSwapOp_mul_sheetSwapOp,
        TNLean.Lattice.commute_sheetSwapOp_kronecker}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    If $Y$ is the disjoint union of $T$ and $A$, then $F_AF_Y=F_T$. The swap
    $F_S$ commutes with $R_1\otimes R_2$ whenever $R_1$ and $R_2$ act on a
    set disjoint from $S$ (\texttt{05-frames.tex}, lines 512--520).
\end{lemma}

\begin{proof}\leanok
    The swaps are permutation matrices of the two-sheet configurations. On a
    configuration, swapping at $Y$ and then at $A\subseteq Y$ swaps exactly
    at the sites of $Y$ outside $A$, that is, at $T$. For the second claim,
    $R_1$ and $R_2$ are the identity on $S$ tensored with operators $R_1'$
    and $R_2'$ on the complement, so in the coordinates splitting off the two
    copies of $S$ both $F_S$ and $R_1\otimes R_2$ act on different factors.
\end{proof}

\begin{theorem}[Exchanging supported operators]
    \label{thm:peps_sheet_swap_supported_exchange}
    \lean{TNLean.Lattice.sheetSwapOp_mul_kronecker_of_mem_supportedOperators}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    Let $R\subseteq\Lambda$, and let $A$ and $B$ be operators on
    $\mathcal H_\Lambda$ supported on $R$. Then the replica-term exchange identity of
    \cite{OpenAI2026PolynomialPEPS} (\texttt{02-information.tex},
    lines 416--424) holds:
    \begin{align*}
        F_R(A\otimes B)=(B\otimes A)F_R.
    \end{align*}
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_split_forms,lem:lattice_sheet_split_swap}
    Splitting each sheet into $R$ and its complement expresses
    $A=A_R\otimes I$ and $B=B_R\otimes I$ by the support assumptions. The swap interchanges the two
    copies of $R$ and fixes the complementary factors. Writing each
    configuration as $x=(x_R,x_{R^c})$, both sides have the entry
    \begin{align*}
        A_R\big((y_2)_R,(x_1)_R\big)\,B_R\big((y_1)_R,(x_2)_R\big)\,
        \delta_{(y_1)_{R^c},(x_1)_{R^c}}\,\delta_{(y_2)_{R^c},(x_2)_{R^c}}
    \end{align*}
    in row $(y_1,y_2)$ and column $(x_1,x_2)$.
\end{proof}

\begin{theorem}[Cancellation of a noncrossing doubled interaction]
    \label{thm:peps_sheet_swap_doubled_cancellation}
    \lean{TNLean.Lattice.sheetSwapOp_conj_doubledHamiltonian_eq_of_mem_supportedOperators}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    Let $q\geq0$ be a local dimension and $R,S\subseteq\Lambda$. Let $A$
    be an operator supported on $S$, and let
    $F_R$ exchange two copies at the sites in $R$. If $S\subseteq R$ or $S\cap R=\varnothing$, then the noncrossing
    cancellation of the boundary perturbation argument in
    \cite{OpenAI2026PolynomialPEPS} gives
    \begin{align*}
        F_R(A\otimes I+I\otimes A)F_R^\dagger
        =A\otimes I+I\otimes A.
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_supported_exchange,lem:peps_frame_sheet_swap_compose,
        thm:permutation_matrix_unitary}
    If $S\subseteq R$, the exchange identity applied with $B=I$ on both
    sides gives
    \begin{align*}
        F_R(A\otimes I)=(I\otimes A)F_R,\qquad F_R(I\otimes A)=(A\otimes I)F_R.
    \end{align*}
    If $S\cap R=\varnothing$, the swap commutes with $A\otimes I$ and with
    $I\otimes A$. In either case
    $F_R(A\otimes I+I\otimes A)=(A\otimes I+I\otimes A)F_R$, and
    multiplying on the right by $F_R^\dagger=F_R^{-1}$ gives the stated
    identity.
\end{proof}

\begin{theorem}[Restricting a swap to an interaction support]
    \label{thm:peps_sheet_swap_interaction_restriction}
    \lean{TNLean.Lattice.sheetSwapOp_conj_kronecker_eq_inter,
        TNLean.Lattice.sheetSwapOp_conj_doubledHamiltonian_eq_inter}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    Let $q$ be a natural local dimension, let $\Lambda$ be a finite set of
    sites, and let $R,S\subseteq\Lambda$. Suppose that $A$ and $B$ are
    operators supported on $S$. If $F_R$ exchanges the two copies at the
    sites in $R$, then
    \begin{align}
        F_R(A\otimes B)F_R^\dagger
        =F_{R\cap S}^\dagger(A\otimes B)F_{R\cap S}.
    \end{align}
    In particular, writing $T(A)=A\otimes I+I\otimes A$, one has
    \begin{align}
        F_R T(A) F_R^\dagger = F_{R\cap S}^\dagger T(A) F_{R\cap S}.
        \label{eq:peps_swap_interaction_restriction}
    \end{align}
    This is the support reduction used in the boundary perturbation argument
    of \cite{OpenAI2026PolynomialPEPS}. Neither Hermiticity nor a positive
    local dimension is required.
\end{theorem}
\begin{proof}
    \leanok
    \uses{lem:peps_frame_sheet_swap_compose,thm:permutation_matrix_unitary}
    Put $K=R\cap S$, $O=R\setminus S$, and $M=A\otimes B$. Composition
    of the swaps gives $F_K F_R=F_O$, hence $F_R=F_K^\dagger F_O$ by
    unitarity. Since $O$ is disjoint from $S$, the swap $F_O$ commutes
    with $M$, and $F_O M F_O^\dagger=M$. Consequently,
    \begin{align}
        F_R M F_R^\dagger
        =F_K^\dagger(F_O M F_O^\dagger)F_K
        =F_K^\dagger M F_K.
    \end{align}
    Apply this identity to $(A,I)$ and $(I,A)$ and add the two equations
    to obtain the assertion for $T(A)$.
\end{proof}

\begin{theorem}[Crossing-edge expansion of the doubled perturbation]
    \label{thm:peps_sheet_swap_boundary_expansion}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.sheetSwapOp_conj_doubled_operator_sub_eq_sum_crossing}
    \leanok
    \uses{def:polynomial_peps_model,def:peps_frame_sheet_swap}
    Let $H=\sum_v h_v+\sum_e h_e$ be an open-square Hamiltonian with the
    original bounded site and edge interactions, and let $R$ be any set of
    vertices. For an operator $A$, write $T(A)=A\otimes I+I\otimes A$.
    Let $F_R$ exchange the two copies at the sites of $R$, and let
    $\partial R$ be the set of edges with exactly one endpoint in $R$.
    Then the crossing-edge reduction in the boundary perturbation argument of
    \cite{OpenAI2026PolynomialPEPS} gives
    \begin{align*}
        F_RT(H)F_R^\dagger-T(H)
        =\sum_{e\in\partial R}\bigl(F_RT(h_e)F_R^\dagger-T(h_e)\bigr).
    \end{align*}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_sheet_swap_doubled_cancellation}
    Write $\delta(A)=F_RT(A)F_R^\dagger-T(A)$, which is linear in $A$. By
    Theorem~\ref{thm:peps_sheet_swap_doubled_cancellation}, $\delta(h_v)=0$
    for every site $v$, since $\{v\}\subseteq R$ or
    $\{v\}\cap R=\varnothing$, and $\delta(h_e)=0$ for every edge
    $e\notin\partial R$, since both endpoints of $e$ lie in $R$ or neither
    does. Hence
    \begin{align*}
        \delta(H)=\sum_v\delta(h_v)+\sum_e\delta(h_e)=\sum_{e\in\partial R}\delta(h_e).
    \end{align*}
\end{proof}

\begin{theorem}[Norm of a swapped doubled interaction]
    \label{thm:peps_sheet_swap_doubled_term_norm}
    \lean{TNLean.PEPS.EncodedFrame.norm_sheetSwapOp_conj_doubledHamiltonian_sub_le}
    \leanok
    \uses{def:peps_frame_sheet_swap}
    Let $q\geq0$ be a local dimension and $R\subseteq\Lambda$. For an
    arbitrary operator $A$ on $\mathcal H_\Lambda$, write $T(A)=A\otimes I+I\otimes A$.
    Then the single-interaction estimate for the boundary perturbation in
    \cite{OpenAI2026PolynomialPEPS} gives
    \begin{align*}
        \|F_RT(A)F_R^\dagger-T(A)\|\leq4\|A\|.
    \end{align*}
\end{theorem}

\begin{proof}\leanok
    \uses{def:peps_frame_sheet_swap,thm:peps_frame_sheet_swap_norm,
        thm:peps_frame_identity_tensor_norm}
    The two identity-factor estimates give
    $\|T(A)\|\leq\|A\otimes I\|+\|I\otimes A\|\leq2\|A\|$.
    Since $\|F_R\|\leq1$ and $\|F_R^\dagger\|=\|F_R\|$, submultiplicativity
    gives $\|F_RT(A)F_R^\dagger\|\leq\|T(A)\|$. The triangle inequality
    proves the assertion.
\end{proof}

\begin{theorem}[Support of the swap in paired coordinates]
    \label{thm:peps_sheet_swap_paired_site_support}
    \lean{TNLean.Lattice.reindex_sheetSwapOp_mem_supportedOperators}
    \leanok
    \uses{def:qc_site_pair_coordinates,def:peps_frame_sheet_swap}
    Let $q\geq0$ be a local dimension and $R\subseteq\Lambda$. Identify two configurations of local dimension $q$
    with one configuration of local dimension $q^2$ by pairing the two values
    at each site. In these coordinates for the doubled Hamiltonian in
    \cite{OpenAI2026PolynomialPEPS}, the sheet swap $F_R$ is supported on $R$.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:qc_permutation_layers}
    In paired coordinates the swap is the permutation $\sigma$ of
    configurations $x=(x_i)_{i\in\Lambda}$, $x_i=(a_i,b_i)$, given by
    \begin{align*}
        \sigma(x)_i=
        \begin{cases}
            (b_i,a_i), & i\in R,    \\
            (a_i,b_i), & i\notin R.
        \end{cases}
    \end{align*}
    Thus $\sigma$ fixes every coordinate outside $R$, and its coordinates in
    $R$ depend only on the coordinates of $x$ in $R$. The permutation matrix
    of such a permutation is supported on $R$.
\end{proof}

\begin{theorem}[Support of a swapped tensor product]
    \label{thm:peps_sheet_swap_conjugate_support}
    \lean{TNLean.Lattice.reindex_sheetSwapOp_conj_kronecker_mem_supportedOperators,
        TNLean.Lattice.reindex_sheetSwapOp_conj_doubledHamiltonian_mem_supportedOperators}
    \leanok
    \uses{def:qc_site_pair_coordinates,def:peps_frame_sheet_swap}
    Let $\Lambda$ be a finite site set, let $q$ be a natural local dimension,
    and let $R,S\subseteq\Lambda$. Suppose that $A$ and $B$ are supported
    on $S$. Express the two-copy operators in paired-site coordinates,
    and write $\widetilde F_R$ and $\widetilde M$ for the matrices of
    $F_R$ and $A\otimes B$ in these coordinates. Then
    $\widetilde F_R\widetilde M\widetilde F_R^\dagger$ is supported on $S$.
    In particular this holds for the paired matrix of
    $T(A)=A\otimes I+I\otimes A$, as used in the boundary perturbation
    argument of \cite{OpenAI2026PolynomialPEPS}. Zero local dimension
    and empty site sets are included.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_interaction_restriction,
        thm:peps_sheet_swap_paired_site_support,thm:qc_site_pair_tensor_support}
    Put $K=R\cap S$. Restricting the swap gives
    $F_R(A\otimes B)F_R^\dagger=F_K^\dagger(A\otimes B)F_K$.
    In paired-site coordinates, $F_K$ is supported on $K\subseteq S$,
    and $A\otimes B$ is supported on $S$. The operators supported on
    $S$ are closed under adjoints and products, which proves the first
    assertion. Apply it to $(A,I)$ and $(I,A)$ and use closure under
    sums to obtain the assertion for $T(A)$.
\end{proof}

\begin{theorem}[Support and norm of the doubled boundary perturbation]
    \label{thm:peps_swapped_doubled_boundary_bound}
    \lean{TNLean.PEPS.Approximation.SquareHamiltonian.swapped_doubled_boundary_bound}
    \leanok
    \uses{def:polynomial_peps_model,def:qc_site_pair_coordinates,def:peps_frame_sheet_swap}
    Let $V=\{0,\ldots,L-1\}^2$, where $L\geq0$ is an integer, and let
    the local dimension be an integer $q\geq0$.
    Let $H=\sum_{v\in V}h_v+\sum_e h_e$ be a Hermitian open-square
    Hamiltonian. Each original site interaction is supported on its site,
    each original edge interaction is supported on its endpoints, and all
    these interactions have norm at most $J\in\R$.
    For $R\subseteq V$, let $\partial R$ consist of the edges with exactly
    one endpoint in $R$, and put
    \begin{align*}
        Z=\bigcup_{e\in\partial R}\{\text{endpoints of }e\}.
    \end{align*}
    In paired-site coordinates, let $K$ be the matrix of
    $H\otimes I+I\otimes H$, and let $F$ be the matrix of the operator
    exchanging the two copies at the sites in $R$.
    Then the boundary perturbation of \cite{OpenAI2026PolynomialPEPS},
    $FKF^\dagger-K$, is supported on $Z$ with local dimension $q^2$ and satisfies
    \begin{align*}
        \|FKF^\dagger-K\|\leq 8JL^2.
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:peps_sheet_swap_boundary_expansion,
        thm:peps_sheet_swap_conjugate_support,
        thm:peps_sheet_swap_doubled_cancellation,
        thm:peps_sheet_swap_doubled_term_norm,thm:peps_square_edge_count}
    The crossing-edge expansion expresses the perturbation as a sum over
    $e\in\partial R$. Each doubled interaction, before and after the
    regional swap, is supported on the endpoints of $e$. Closure of the
    supported operators under subtraction and finite sums proves the
    support assertion.

    If $L=0$, both original interaction sums vanish and the estimate is
    $0\leq0$. Otherwise, any site $v\in V$ gives
    $0\leq\|h_v\|\leq J$. Each crossing-edge contribution has norm at most
    $4\|h_e\|\leq4J$. The triangle inequality and the square-edge count give
    \begin{align*}
        \|FKF^\dagger-K\|
        \leq \sum_{e\in\partial R}4J
        =4J|\partial R|
        \leq 4J(2L^2)
        =8JL^2.
    \end{align*}
    Relabelling the two-copy configurations by paired-site configurations
    preserves the operator norm.
\end{proof}

\begin{lemma}[The corrections of an exchange on the two copies of the remainder]

\begin{theorem}[Coordinates on two copies of a region]
    \label{thm:peps_two_region_coordinates}
    \lean{TNLean.Lattice.Sheet.pairUIso_siteVec,
        TNLean.Lattice.Sheet.pairUIso_symm_single,
        TNLean.Lattice.Sheet.mem_of_mem_sitesR,
        TNLean.Lattice.Sheet.sitesU,
        TNLean.Lattice.Sheet.sitesR,
        TNLean.Lattice.Sheet.pairUIso}
    \leanok
    \uses{def:peps_frame_registers,lem:peps_frame_registers_coordinates,
        thm:peps_region_configuration}
    Let $T,E\subseteq\Lambda$ and $U=\Lambda\setminus(T\cup E)$, with two
    arbitrary assignments of site owners. Retain the sites of $U$ in the
    fixed site order; every site in the complementary list belongs to
    $T\cup E$. Let $\ket c_{U,j}$ be the product basis vector on these site
    registers on sheet $j$, and let $\iota_U$ be the inverse of the concatenation
    isometry separating their two consecutive lists. Grouping both lists
    gives an isometry $Q_U$ to $\C^{[q]^U\times[q]^U}$ satisfying
    \begin{align}
        Q_U\iota_U(\ket{c_1}_{U,1}\otimes\ket{c_2}_{U,2})
         & =\ket{c_1|_U,c_2|_U}.
        \label{eq:peps_two_region_basis}
    \end{align}
    If $T\cap E=\varnothing$, take any configurations $c_1,c_2$ and
    $u_1,u_2\in[q]^U$, and set $c_j^{u_j}=m(c_j|_T,u_j;c_j)$. Then
    \begin{align}
        Q_U^{-1}\ket{u_1,u_2}
         & =\iota_U(\ket{c_1^{u_1}}_{U,1}\otimes\ket{c_2^{u_2}}_{U,2}).
        \label{eq:peps_two_region_inverse}
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_region_register_coordinates,thm:peps_region_configuration,
        lem:peps_frame_registers_coordinates}
    Grouping the two lists by their position bijections and identifying the tensor
    product of their coordinate spaces with the space indexed by pairs of
    configurations proves (\ref{eq:peps_two_region_basis}), while the
    complementary list contains only sites outside $U$. For the inverse,
    $c_j^{u_j}|_U=u_j$ by (\ref{eq:peps_region_reconstruction}), so substituting
    into (\ref{eq:peps_two_region_basis}) and applying $Q_U^{-1}$ gives the
    inverse formula.
\end{proof}

\begin{theorem}[Two-sheet exchange: bounded change]

\subsection{Sheets without holes}

None of the monomials above acts on the tag registers: the birth and death
monomials and the renaming of the sites act behind the tag registers, which may be
replaced by any block of registers in front of the site registers. With the
empty block they act on a sheet without holes, an owner for each site of a
finite set $\Lambda$, whose reference vector is $\Omega$ placed in the site
registers. This is the setting of the one-dimensional protocol, where no
three owners meet and no encoded holes are needed.

\begin{definition}[Sheets without holes]
    \label{def:lattice_sheet}

    A sheet on $\Lambda$ is an owner for each site; its registers are one
    register $\C^q$ per site, held by its owner, identified with
    $\mathcal H_\Lambda$ in product coordinates, and its reference vector is
    $\Omega$. These coordinates are those of the empty block of registers in
    front of the site registers. For a party $P_\circ$ let $E$ be
    the set of sites not owned by $P_\circ$. For two sheets and a region $Y$,
    the exchange gives the first sheet the owners of the second inside $Y$ and
    conversely; with $Z$ the set of sites whose two owners are not both
    $P_\circ$, $T=Z\cap Y$, $E=Z\setminus Y$ and $A=Y\setminus Z$, the

\begin{proof}\leanok

    $P_\circ$ and no pair resource, whose operator in product coordinates is
    $C=D_UF_AF_Y=D_UF_T$, and which maps $\Omega\otimes\Omega$ within
    $4L^{-p}$ of $\Omega\otimes\Omega$. For $p=30$ this is the case without
    holes of \cite[Lemma~6.6]{OpenAI2026PolynomialPEPS} (\texttt{05-frames.tex},
    lines 460--479), where the renaming acts in product coordinates as $F_Y$;
    in one dimension the exponent is a free parameter.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:peps_frame_splitting,lem:lattice_block_renaming_word,
        lem:lattice_front_correction_word,lem:lattice_exchange_correction_matrix,
        lem:peps_frame_exchange_error,lem:lattice_register_reorderings}
    The renaming of Lemma~\ref{lem:lattice_block_renaming_word} with empty
    blocks of tags exchanges the configurations of the two sheets inside $Y$,
    so its operator is $F_Y$; the corrections act as $D_UF_A$. Since $Y$ is
    the disjoint union of $T$ and $A$, the product is $D_UF_T$, and
    Lemma~\ref{lem:peps_frame_exchange_error} bounds the reference error.
\end{proof}

\begin{definition}[Bounded changes of a stack of sheets]
    \label{def:lattice_bounded_change}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.norm_eval_le_norm,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange,
        TNLean.Lattice.stackRegs,
        TNLean.Lattice.stackVec,
        TNLean.Lattice.norm_stackVec,
        TNLean.Lattice.norm_stackVec_le,
        TNLean.Lattice.framesVec,
        TNLean.Lattice.framesSpace,
        TNLean.Lattice.pairRef,
        TNLean.Lattice.HasBoundedChange,
        TNLean.Lattice.HasSourceBoundedChange}
    \leanok
    \uses{def:peps_circuit_source_bound}
    An original circuit is a bounded change from a vector $x$ to a vector $y$, with
    participants in a set $S$, bounds $r$ and $K$ and error $\varepsilon$, if it has
    at most one nonprivate gate, every participant of that gate lies in $S$, every
    monomial of its expansion has at most $r$ pair effects, its coefficient sum and
    its number of monomials are at most $K$, and it takes $x$ within $\varepsilon$ of
    $y$; every original circuit is a contraction. Let each sheet $F$ of some type
    carry a list of registers and a reference vector $\omega_F$ on them. The memory
    of a list $F_1,\dots,F_m$ of sheets followed by further registers $\ell$ is the
    tensor product of their registers, in order, and for a vector $\xi$ of $\ell$
    its reference vector is $\omega_{F_1}\otimes\cdots\otimes\omega_{F_m}\otimes\xi$,
    whose norm is the product of the norms of the factors; if every $\omega_F$ has
    norm at most one, this norm is at most $\lVert\xi\rVert$. Without further
    registers, the memory of the sheets is the whole private memory of a party
    holding them, and the reference vector of two sheets is
    $\omega_{F_1}\otimes\omega_{F_2}$ on their concatenated registers. A bounded
    change of the sheets $F_1,\dots,F_m$ into $F'_1,\dots,F'_n$ is an original
    circuit between the two memories, with the same further registers, that is a
    bounded change from one reference vector to the other; the variant with bounded
    pair sources also requires at most a given number of pair sources in each
    monomial.
\end{definition}

\begin{proof}\leanok
    The norm of a tensor product of vectors is the product of their norms, and the
    operator of an original circuit is a composition of contractions.
\end{proof}

\begin{theorem}[Operations that preserve bounded changes of a stack]
    \label{thm:lattice_bounded_change_closure}
    \lean{TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.mono,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.comp_right,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.IsBoundedChange.comp_left,
        TNLean.Lattice.PartyCircuit.OriginalCircuit.isBoundedChange_id,
        TNLean.Lattice.PartyCircuit.Word.isBoundedChange_toOriginalCircuit,
        TNLean.Lattice.PartyCircuit.PartyChain.isBoundedChange_toOriginalCircuit,
        TNLean.Lattice.norm_eval_liftOne_stackVec_sub_le,
        TNLean.Lattice.nonprivateCount_liftTwo,
        TNLean.Lattice.participants_liftTwo_subset,
        TNLean.Lattice.liftOne,
        TNLean.Lattice.isMonomialBounded_liftTwo,
        TNLean.Lattice.participants_liftOne_subset,
        TNLean.Lattice.isExpansionBounded_liftTwo,
        TNLean.Lattice.liftTwo,
        TNLean.Lattice.norm_eval_liftTwo_stackVec_sub_le,
        TNLean.Lattice.isMonomialBounded_liftOne,
        TNLean.Lattice.isExpansionBounded_liftOne,
        TNLean.Lattice.nonprivateCount_liftOne,
        TNLean.Lattice.hasBoundedChange_activate,
        TNLean.Lattice.hasBoundedChange_liftTwo,
        TNLean.Lattice.hasBoundedChange_liftOne,
        TNLean.Lattice.exists_isBoundedChange_of_eq,
        TNLean.Lattice.HasBoundedChange.perm_left,
        TNLean.Lattice.HasBoundedChange.perm_right,
        TNLean.Lattice.hasBoundedChange_refl,
        TNLean.Lattice.isBoundedChange_liftTwo,
        TNLean.Lattice.hasBoundedChange_perm_liftOne,
        TNLean.Lattice.HasBoundedChange.mono,
        TNLean.Lattice.isBoundedChange_liftOne,
        TNLean.Lattice.hasBoundedChange_of_perm,
        TNLean.Lattice.stackVec_append,
        TNLean.Lattice.exists_reordering_stackVec,
        TNLean.Lattice.stackRegs_append,
        TNLean.Lattice.exists_originalCircuit_of_eq,
        TNLean.Lattice.exists_originalCircuit_stackVec_of_perm,
        TNLean.Lattice.hasSourceBoundedChange_refl,
        TNLean.Lattice.HasSourceBoundedChange.perm_right,
        TNLean.Lattice.hasSourceBoundedChange_activate,
        TNLean.Lattice.hasSourceBoundedChange_liftTwo,
        TNLean.Lattice.hasSourceBoundedChange_liftOne,
        TNLean.Lattice.HasSourceBoundedChange.mono,
        TNLean.Lattice.hasSourceBoundedChange_of_perm,
        TNLean.Lattice.isSourceBounded_liftOne,
        TNLean.Lattice.hasSourceBoundedChange_perm_liftOne,
        TNLean.Lattice.HasSourceBoundedChange.perm_left,
        TNLean.Lattice.HasSourceBoundedChange.hasBoundedChange,
        TNLean.Lattice.isSourceBounded_liftTwo}
    \leanok
    \uses{def:lattice_bounded_change,thm:peps_original_produced_resource_bounds}
    Let every reference vector of a sheet and the vector $\xi$ of the further
    registers have norm at most one.
    \begin{enumerate}
        \item A bounded change remains one after enlarging its participant set, its
              bounds or its error. Composing it, on either side, with a circuit without
              nonprivate gate that satisfies the same pair-effect, coefficient-sum and
              monomial bounds gives a bounded change with the same data, between the
              transported vectors. The circuit without operations is a bounded change
              between equal vectors, for every participant set and all bounds, and so is
              the change of a sheet into an equal sheet.
        \item The memory and the reference vector of a concatenation of two lists of
              sheets are the tensor products of those of the two lists. A permutation of
              the sheets is implemented by a word of exchanges of tensor factors, without
              nonprivate gate, carrying the reference vector of the first list exactly to
              that of the second; so it is a bounded change without error and without pair
              sources, and a permutation before or after a bounded change of the sheets
              gives a bounded change with the same data.
        \item A change of one sheet, or of two adjacent sheets, lifted to the whole
              stack with the other sheets and the further registers as spectators, has the
              same nonprivate gates, participants, pair-effect, coefficient-sum, monomial
              and pair-source bounds, and its reference error on the stack is at most its
              error on the sheets it acts on. Hence a bounded change of these sheets is a
              bounded change of the stack with the same data, also after a permutation of
              the stack that brings these sheets into place.
        \item An allowed word using only the parties of $S$ with at most one pair source
              is a bounded change with its own error, for every $r$ and every $K\ge1$. An
              allowed chain using only the parties of $S$, without pair sources and with at
              most one pair effect, is a bounded change for every $r\ge1$ and $K\ge1$.
    \end{enumerate}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_original_produced_resource_bounds}
    The spectator sheets of a lifted change have reference vectors of norm at most
    one, so the error on the stack is the error on the acting sheets times norms at
    most one. Composing with a circuit without nonprivate gate adds no gate and
    keeps the bounds. A permutation of the sheets is a composition of exchanges of
    adjacent blocks of registers. The resource bounds of the original circuit of a
    word or a chain follow from its counts of pair sources and pair effects.
\end{proof}

\begin{theorem}[The initial private maps of a stack of sheets]
    \label{thm:lattice_stack_initialization}
    \lean{TNLean.Lattice.splitMap,
        TNLean.Lattice.splitMap_tmul_one,
        TNLean.Lattice.norm_splitMap_le,
        TNLean.Lattice.splitCircuit,
        TNLean.Lattice.nonprivateCount_splitCircuit,
        TNLean.Lattice.isExpansionBounded_splitCircuit,
        TNLean.Lattice.isMonomialBounded_splitCircuit,
        TNLean.Lattice.participants_splitCircuit_subset,
        TNLean.Lattice.eval_splitCircuit,
        TNLean.Lattice.exists_originalCircuit_initial}
    \leanok
    \uses{def:lattice_bounded_change,thm:lattice_bounded_change_closure}
    Let every sheet of a list be held entirely by one party: all its registers belong
    to that party. Give each party, as its whole initial private memory, the
    registers of its sheets in the order of the list, and as its initial vector the
    product of their reference vectors, assumed of norm one. Then one private map per
    party, followed by a reordering of the sheets, carries the product of the initial
    vectors of all parties exactly to the reference vector of the list, with no
    nonprivate gate.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:lattice_bounded_change_closure}
    Each private map identifies the whole memory of a party with the memory of its
    sheets. The concatenation, over the parties, of their lists of sheets is a
    permutation of the list, which a reordering implements.
\end{proof}

\begin{definition}[Stacks of sheets]
    \label{def:lattice_sheet_stack}
    \lean{TNLean.Lattice.Sheet.refFamily,
        TNLean.Lattice.Sheet.HasBoundedChange,
        TNLean.Lattice.Sheet.HasSourceBoundedChange}
    \leanok
    \uses{def:lattice_sheet,def:lattice_bounded_change}
    The memory of a list of sheets and its reference vector are those of
    Definition~\ref{def:lattice_bounded_change}, with the reference vector $\Omega$ of
    each sheet placed in its site registers. A bounded change of a list of
    sheets is a bounded change, in the sense of
    Definition~\ref{def:lattice_bounded_change}, between these reference vectors;
    the variant with bounded pair sources also bounds the number of normalized
    pair sources.
\end{definition}

\begin{lemma}[Reference vectors of stacks of sheets]
    \label{lem:lattice_sheet_stack_reference}
    \lean{TNLean.Lattice.Sheet.norm_refFamily_le,
        TNLean.Lattice.Sheet.norm_framesVec,
        TNLean.Lattice.Sheet.pairRef_eq_twoRegIso_symm,
        TNLean.Lattice.Sheet.finiteDimensional_of_mem_stackRegs}
    \leanok
    \uses{def:lattice_sheet_stack}
    If $\lVert\Omega\rVert\le1$, the reference vector of every sheet has norm at
    most one; if $\lVert\Omega\rVert=1$, the reference vector of a list of sheets
    without further registers has norm one. The reference vector of two adjacent
    sheets is $\Omega\otimes\Omega$ in the product coordinates of the two sheets.
    Every register of a list of sheets is $\C^q$, so it is finite-dimensional.
\end{lemma}

\begin{proof}\leanok
    The reference vector of a sheet is the image of $\Omega$ under an isometry,
    and the norm of a tensor product of vectors is the product of their norms.
\end{proof}

\begin{theorem}[The initial step for sheets]
    \label{thm:lattice_sheet_initialization}
    \lean{TNLean.Lattice.Sheet.exists_originalCircuit_initial_of_owner}
    \leanok
    \uses{lem:lattice_sheet_stack_reference,thm:lattice_stack_initialization}
    Let $\lVert\Omega\rVert=1$ and let every sheet of a list be held entirely by
    one party. Then the product of the reference vectors of the sheets of every
    party has norm one, and one private map per party followed by a reordering
    carries the product of these vectors exactly to the reference vector of the
    list, with no nonprivate gate. This is the case without holes of
    Theorem~\ref{thm:lattice_stack_initialization}.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:lattice_sheet_stack_reference,thm:lattice_stack_initialization}
    Every register of a sheet held by one party belongs to that party, and the
    reference vector of every list of sheets has norm one by
    Lemma~\ref{lem:lattice_sheet_stack_reference}. Apply
    Theorem~\ref{thm:lattice_stack_initialization}.
\end{proof}
