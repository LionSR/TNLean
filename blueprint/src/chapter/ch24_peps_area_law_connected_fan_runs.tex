\section{Connected runs after radial deletion}
\label{sec:area_law_fan_cut_runs}

The connected pieces of
Lemma~\ref{thm:area_law_fan_cut_piece_connected} meet across every
retained adjacency by
Lemma~\ref{thm:area_law_fan_inactive_radial_intersection}.
This glues the pieces along each run of
Definition~\ref{def:area_law_fan_runs} in the fan construction of
\cite[Section~11]{OpenAI2026AreaLaw}.

\begin{lemma}[Connectedness of a run after radial deletion]
    \label{thm:area_law_fan_cut_run_connected}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanRunRegion_cut_inter_ball_isConnected}
    \leanok
    \uses{def:area_law_fan_cut_piece,def:area_law_fan_runs}
    Consider the fan, colouring, cut set $L$ of active radials, and pieces
    $S_i=(P_i\setminus L)\cap B_\infty(c,r)$ of
    Definition~\ref{def:area_law_fan_cut_piece}.
    Join consecutive triangles $P_i$ and $P_j$ by an edge whenever their
    colours agree. For each connected component $R$ of this graph,
    put $T_R=\bigcup_{i\in R}P_i$. Then
    $(T_R\setminus L)\cap B_\infty(c,r)$ is nonempty and connected.
\end{lemma}
\begin{proof}
    \leanok
    \uses{def:area_law_fan_cut_piece,def:area_law_fan_runs,
        thm:area_law_fan_cut_piece_connected,
        thm:area_law_fan_inactive_radial_intersection}
    Each piece $S_i$ is nonempty and connected by
    Lemma~\ref{thm:area_law_fan_cut_piece_connected}.
    Every edge of $R$ joins consecutive triangles of the same colour, so
    Lemma~\ref{thm:area_law_fan_inactive_radial_intersection} gives
    $S_i\cap S_j\ne\varnothing$ along that edge.

    Because $R$ is a connected component, it is nonempty and any two of
    its vertices are joined by a finite path of edges in $R$.
    Therefore $\bigcup_{i\in R}S_i$ is nonempty and connected, and
    \begin{align}
        (T_R\setminus L)\cap B_\infty(c,r)
         & =\bigcup_{i\in R}S_i.
    \end{align}
\end{proof}
