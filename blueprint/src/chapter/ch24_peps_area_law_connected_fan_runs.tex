\section{Connected runs after radial deletion}
\label{sec:area_law_fan_cut_runs}

The connected pieces of
Theorem~\ref{thm:area_law_fan_cut_piece_connected} meet across every
retained adjacency by
Theorem~\ref{thm:area_law_fan_inactive_radial_intersection}.
The following gluing assertion applies to the runs of
Definition~\ref{def:area_law_fan_runs} in the fan construction of
\cite[Section~11]{OpenAI2026AreaLaw}.

\begin{theorem}[Connectedness of a run after radial deletion]
    \label{thm:area_law_fan_cut_run_connected}
    \lean{TNLean.PEPS.AreaLaw.Geometry.cellFanRunRegion_cut_inter_ball_isConnected}
    \leanok
    \uses{def:area_law_cell_fan,def:area_law_cell_fan_successor,
        def:area_law_fan_runs}
    Let $o\in\R^2$, $\ell\in\N$ and $z\in\Z^2$ be arbitrary, and let
    $Q=o+2^\ell(z+[0,1]^2)$ be the corresponding closed dyadic square.
    Write $c$ for its center and $r=2^\ell/2>0$ for its half-side.
    Equip the plane with the maximum norm.
    Subdivide every side at its midpoint, obtaining eight triangles
    $P_i=\operatorname{conv}\{c,a_i,b_i\}$ indexed in perimeter order
    by $\mathcal J$. Let $\tau$ be the cyclic successor and assign
    arbitrary colors $f_i\in\{0,1\}$. Define
    \begin{align}
        A & =\{t\in\mathcal J:f_t\ne f_{\tau(t)}\}, \\
        L & =\bigcup_{t\in A}[c,b_t].
    \end{align}
    Retain the adjacency between consecutive triangles precisely when
    their colors agree. For each connected component $R$ of this graph,
    put $T_R=\bigcup_{i\in R}P_i$. Then
    $(T_R\setminus L)\cap B_\infty(c,r)$ is nonempty and connected.
    The change set $A$ may be empty. Components with the same color
    remain distinct runs.
\end{theorem}
\begin{proof}\leanok
    \uses{def:area_law_fan_runs,
        thm:area_law_fan_cut_piece_connected,
        thm:area_law_fan_inactive_radial_intersection}
    For each $i\in\mathcal J$, set
    $S_i=(P_i\setminus L)\cap B_\infty(c,r)$.
    Each $S_i$ is nonempty and connected by
    Theorem~\ref{thm:area_law_fan_cut_piece_connected}.
    A retained edge joins consecutive triangles with equal colors, so
    Theorem~\ref{thm:area_law_fan_inactive_radial_intersection} gives
    $S_i\cap S_j\ne\varnothing$ along every such edge.

    The component $R$ is nonempty, and any two of its vertices are
    joined by a finite path of retained edges within $R$.
    Gluing the connected sets along these paths shows that
    $\bigcup_{i\in R}S_i$ is connected. It is nonempty because $R$
    contains a vertex and every $S_i$ is nonempty.
    Distributing deletion and intersection over the union gives
    \begin{align}
        (T_R\setminus L)\cap B_\infty(c,r)
         & =\bigcup_{i\in R}S_i.
    \end{align}
    This proves the assertion. If $A$ is empty, no radial is removed
    and the center is retained.
\end{proof}
