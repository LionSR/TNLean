\begin{lemma}[Physical support isometry uniform in inserted bonds]%
    \label{thm:peps_inserted_physical_support_transport}
    \lean{TNLean.PEPS.graphPhysicalProductMap_graphInsertedBondNetwork,
        TNLean.PEPS.graphInsertedBondNetwork_mem_range_productSiteMap,
        TNLean.PEPS.graphInsertedBondNetwork_mem_euclideanRange_productSiteMap,
        TNLean.PEPS.exists_graphGIsometricInsertedPhysicalSupportTransport}
    \leanok
    Let $A_v,B_v$ be G-isometric tensors for the same unitary virtual
    representation at each vertex $v$ of a finite simple graph. There exist
    $c_{A,v},c_{B,v}>0$, local physical maps $F_v$, and one isometry $I$ on
    $\bigotimes_v\operatorname{ran}\mathcal P(A_v)$ (acting there by
    $\bigotimes_v F_v$, independently of $K$) such that every inserted state
    $\Psi_A(K)$ belongs to $\bigotimes_v\operatorname{ran}\mathcal P(A_v)$ and,
    for every family of inserted bond matrices $K=(K_e)_e$,
    \begin{align}
        I\Psi_A(K)=\left(\prod_v\frac{\sqrt{c_{A,v}}}{\sqrt{c_{B,v}}}\right)
        \Psi_B(K).
        \label{eq:peps_inserted_support_transport}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_g_isometric_physical_transport}
    Lemma~\ref{thm:peps_g_isometric_physical_transport} provides the local
    normalized-adjoint maps $F_v$ and constants $c_{A,v},c_{B,v}>0$ before
    choosing $K$, whose column Gram identities make $\bigotimes_v F_v$
    isometric on $\bigotimes_v\operatorname{ran}\mathcal P(A_v)$. Expanding
    $\Psi_A(K)$ over independent endpoint labels and commuting $\bigotimes_v F_v$
    with the sum gives (\ref{eq:peps_inserted_support_transport}).
\end{proof}

\begin{definition}[Permutation bonds under two-by-two grouping]%
    \label{def:peps_twisted_two_by_two_coordinates}
    \lean{TNLean.PEPS.twoByTwoHorizontalPerm,
        TNLean.PEPS.twoByTwoVerticalPerm,
        TNLean.PEPS.twoByTwoPermutationBoundary,
        TNLean.PEPS.pairedLeftRegularMatrix}
    \leanok
    Group the fine torus into disjoint $2\times2$ squares, assign the same
    permutation $s_h$ to each pair of horizontal crossing bonds and $s_v$ to
    each pair of vertical crossing bonds, and leave internal bonds unchanged.
    The paired regular action is $L_g^{(2)}=L_g\otimes L_g$, acting on $(a,b)$
    by $(ga,gb)$, with horizontal insertions acting on the receiving left leg
    and vertical insertions on the top leg in the native downward orientation.
\end{definition}

\begin{lemma}[Exact twisted geometric grouping]%
    \label{thm:peps_twisted_two_by_two_geometry}
    \lean{TNLean.PEPS.twoByTwoSiteLegs_permutation,
        TNLean.PEPS.torusBondNetwork_perm_eq_twoByTwoTiledSum,
        TNLean.PEPS.torusBondNetwork_perm_eq_twoByTwoBlocked,
        TNLean.PEPS.kitaevDoubleCoordinateEquiv_seam,
        TNLean.PEPS.twoByTwoHorizontalPerm_seam,
        TNLean.PEPS.twoByTwoVerticalPerm_seam,
        TNLean.PEPS.pairedLeftRegularMatrix_apply,
        TNLean.PEPS.pairedLeftRegularMatrix_eq_kronecker,
        TNLean.PEPS.torusGClosure_leftRegular_eq_twoByTwoBlocked}
    \leanok
    For all coarse periods $w,h>0$ (including $1$ and $2$), the fine
    contraction with the crossing permutations of
    Definition~\ref{def:peps_twisted_two_by_two_coordinates} equals the coarse
    contraction of the four-site tensors with paired permutations, with no
    scalar factor. In particular, a native closure $(g_0,h_0)$ (without
    assuming symmetry or $g_0h_0=h_0g_0$) becomes the closure $(g_0,h_0)$ for
    $L^{(2)}$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_twisted_two_by_two_coordinates}
    Reindexing the fine bonds of Definition~\ref{def:peps_twisted_two_by_two_coordinates}
    into four internal bonds per $2\times2$ block and two labels per coarse
    crossing bond, summing the internal labels gives the four-site tensor. For
    a doubled periodic coordinate $2x+b$, the outgoing bond crosses the seam
    iff $b=1$ and $x$ is the final coarse coordinate, so exactly the two
    crossing bonds receive each seam insertion.
\end{proof}

\begin{definition}[Ordered labels for native torus seams]%
    \label{def:peps_twisted_torus_ordered_labels}
    \lean{TNLean.PEPS.torusGraphRegularLabels,
        TNLean.PEPS.torusClosureRegularLabels}
    \leanok
    For a simple torus graph with periods at least $3$, orient every native
    bond in the ordered-edge orientation, replacing $u$ by $u^{-1}$ on reversed
    arrows, with $h_0$ on the horizontal periodic seam and $g_0$ on the
    downward vertical seam.
\end{definition}

\begin{lemma}[Native, paired, and ordered inserted contractions]%
    \label{thm:peps_twisted_torus_graph_coordinates}
    \lean{TNLean.PEPS.graphInsertedBondNetwork_equiv,
        TNLean.PEPS.graphInsertedBondNetwork_torusIncidentFamily,
        TNLean.PEPS.graphInsertedBondNetwork_twoByTwoBundledGraphSite,
        TNLean.PEPS.torusGraphBondMatrix_leftRegular,
        TNLean.PEPS.torusGraphBondMatrix_regularBundle,
        TNLean.PEPS.graphInsertedBondNetwork_torusClosureRegularLabels,
        TNLean.PEPS.regularBundleMatrix_unit_eq_paired,
        TNLean.PEPS.torusGClosure_leftRegular_eq_twoByTwoBundledGraphSite}
    \leanok
    Native torus contractions equal their ordered-edge inserted graph
    contractions for arbitrary site-dependent physical alphabets, and bijective
    endpoint relabeling preserves every coefficient. Hence the native fine
    closure equals the inserted coarse contraction of its four-site tensors
    with two regular labels on each bond, and reversal acts by
    $L_u^{\mathsf T}=L_{u^{-1}}$ on both single and bundled regular bonds.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_twisted_torus_ordered_labels,
        thm:peps_twisted_two_by_two_geometry}
    Under the incidence-coordinate and endpoint bijections of
    Definition~\ref{def:peps_twisted_torus_ordered_labels} and
    Lemma~\ref{thm:peps_twisted_two_by_two_geometry}, left-multiplication
    permutation matrices satisfy $L_u^{\mathsf T}=L_{u^{-1}}$ on both single
    and paired labels, matching native and ordered contraction factors at every
    endpoint.
\end{proof}

\begin{lemma}[Inserted bonds preserve the surplus Bell state]%
    \label{thm:peps_inserted_regular_surplus}
    \lean{TNLean.PEPS.regularBundleMatrix_transpose,
        TNLean.PEPS.regularSurplusCoordinates_twisted,
        TNLean.PEPS.graphInsertedBondNetwork_regularGraphSurplusSite,
        TNLean.PEPS.regularGraphOriginalOutputRegrouping,
        TNLean.PEPS.exists_regularGraphInsertedOriginalTensorIsometry}
    \leanok
    In relative bundle coordinates $(a,(a^{-1}b_k)_k)$ for $(a,(b_k)_k)$, an
    oriented insertion $u_e$ acts only on $a$, so the inserted contraction of
    the original tensor with surplus identity registers equals its original
    inserted contraction tensored with the untwisted surplus equality vectors.
    A single physical-support isometry maps every bundled inserted state to
    this form with the same local normalization factors and normalized Bell
    product.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_inserted_physical_support_transport,
        thm:peps_regular_graph_surplus_factorization}
    At a twisted head, $(ua)^{-1}(ub_k)=a^{-1}b_k$ leaves the relative labels
    invariant; summing over $a$ gives the original inserted contraction while
    the independent relative sums give the surplus equality vectors as in
    Lemma~\ref{thm:peps_regular_graph_surplus_factorization}, and
    Lemma~\ref{thm:peps_inserted_physical_support_transport} supplies the
    uniform physical-support isometry.
\end{proof}

\begin{lemma}[Nonzero native closures in one fine physical support]%
    \label{thm:peps_twisted_two_by_two_support}
    \lean{TNLean.PEPS.IsGIsometric.torusLegRep_of_regularFourLeg,
        TNLean.PEPS.IsGIsometric.regularFourLegClosure_ne_zero,
        TNLean.PEPS.twoByTwoFineClosure_mem_physicalSupport,
        TNLean.PEPS.twoByTwoSupportedClosure}
    \leanok
    Let $A$ be a regular G-isometric four-leg tensor. Every native closure is
    nonzero, and on a doubled torus with coarse periods at least $3$, all fine
    closure vectors lie in the pullback of the product block physical support
    under the physical grouping unitary.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_twisted_torus_graph_coordinates,
        thm:peps_inserted_physical_support_transport}
    The closure self-overlap equals a positive site factor times a positive
    power of $|G|$ times $|\{k\in G:kg_0=g_0k,\ kh_0=h_0k\}|\ge1$, so the
    closure is nonzero; support membership follows from
    Lemmas~\ref{thm:peps_twisted_torus_graph_coordinates} and
    \ref{thm:peps_inserted_physical_support_transport}.
\end{proof}

\begin{lemma}[Uniform twisted fixed point and coherent superpositions]%
    \label{thm:peps_twisted_two_by_two_fixed_point}
    \lean{TNLean.PEPS.physicalStateProduct_sum,
        TNLean.PEPS.LinearIsometry.coherent_physicalStateProduct,
        TNLean.PEPS.exists_regularTwoByTwoTwistedTensorIsometry}
    \leanok
    Let $A$ be a homogeneous regular G-isometric tensor on a doubled torus with
    coarse periods at least $3$, and write $\Psi_f(g,h)$ and $\Psi_c(g,h)$ for
    its fine and coarse native closure vectors (with coarse tensor $A$). There
    exist $R>0$, a unit vector $\Omega$, and one isometry $I$ on the full fine
    block physical support (acting by physical grouping and fixed local
    physical maps) such that for all $g,h\in G$,
    \begin{align}
        I\Psi_f(g,h) & =R\,\Psi_c(g,h)\otimes\Omega,
                     & I\widehat{\Psi_f(g,h)}        & =\widehat{\Psi_c(g,h)}\otimes\Omega.
        \label{eq:peps_twisted_fixed_point_pair}
    \end{align}
    For any $z_{g,h}\in\C$, setting $x=\sum_{g,h}z_{g,h}\Psi_f(g,h)$ and
    $y=\sum_{g,h}z_{g,h}\Psi_c(g,h)$ gives $Ix=R\,y\otimes\Omega$,
    $x\ne0\iff y\ne0$, and $I\widehat x=\widehat y\otimes\Omega$ whenever
    $x\ne0$ (in particular for coherent sums over commuting pairs $gh=hg$).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_twisted_two_by_two_support,
        thm:peps_inserted_regular_surplus,
        thm:peps_two_by_two_original_scale,
        thm:peps_physical_state_normalization}
    Combining Lemmas~\ref{thm:peps_twisted_two_by_two_support},
    \ref{thm:peps_inserted_regular_surplus},
    \ref{thm:peps_two_by_two_original_scale}, and
    \ref{thm:peps_physical_state_normalization} gives
    (\ref{eq:peps_twisted_fixed_point_pair}) with a single $I$ and $R>0$.
    Linearity yields $Ix=R\,y\otimes\Omega$, and $\|x\|=R\|y\|$ gives
    $x\ne0\iff y\ne0$ and $I\widehat x=\widehat y\otimes\Omega$.
\end{proof}

In the native periodic, physical-support specialization of
\cite[Observation~6.6]{Schuch2010PEPS}, the all-pairs algebraic identity
(\ref{eq:peps_twisted_fixed_point_pair}) includes noncommuting labels while the
parent-ground-state interpretation uses commuting closures $gh=hg$; the
physical map $I$ is chosen before the seam labels and coefficients, and the
remaining period and boundary restrictions (coarse periods $1$ and $2$, other
boundary conditions, and ambient physical unitary extension) are recorded in
\cite{gap:scp10_twisted_two_by_two_regular_fixed_point}.
