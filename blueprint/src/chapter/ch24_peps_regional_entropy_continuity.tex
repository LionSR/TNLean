\section{Regional entropy continuity}

\begin{theorem}[Entropy continuity on a fixed lattice region]
    \label{thm:area_law_regional_entropy_continuity}
    \lean{TNLean.PEPS.AreaLaw.abs_regionalEntropy_sub_le_sqrt_traceDistance}
    \leanok
    \uses{def:area_law_ground_entropy}
    Let $\Lambda\subset\mathbb Z^2$ be finite, let $q\in\mathbb N$, and let
    $\Omega,\Phi\in\bigotimes_{v\in\Lambda}\mathbb C^q$ be unit vectors.
    Set $\delta=\frac12\bigl\||\Omega\rangle\langle\Omega|
        -|\Phi\rangle\langle\Phi|\bigr\|_1$.
    For every region $A\subseteq\Lambda$,
    \begin{align*}
        |S_\Omega(A)-S_\Phi(A)| & \le 4\sqrt\delta\,(1+|A|\log q).
    \end{align*}
    (When $q=0$, the existence of unit vectors forces $\Lambda=A=\varnothing$
    and $|A|\log q=0$.)
    This is the regional form of \cite[Lemma~2.1]{OpenAI2026AreaLaw} used
    in the scanner entropy estimate.
\end{theorem}
\begin{proof}\leanok
    Partial trace contracts trace distance, so the reduced states on $A$ have
    trace distance at most $\delta$. Since each reduced state has trace one, the
    subsystem space of dimension $q^{|A|}$ is nontrivial. Applying the
    Fannes--Audenaert continuity bound on a space of dimension $q^{|A|}$ bounds
    the entropy difference by
    $4\sqrt\delta\log(e q^{|A|}) = 4\sqrt\delta\,(1+|A|\log q)$.
\end{proof}
