\begin{theorem}[Matching weighted sums of periodic blocks]%
    \label{thm:periodic_block_families_unitary_matching}
    \lean{MPSTensor.exists_unitary_matching_of_periodic_block_families}
    \leanok
    \uses{def:periodic_tensor, def:same_mpv2_pos, def:block_diagonal_tensor}
    Let $A_j$ and $B_k$ be left-canonical periodic tensors with periods
    $m_j$ and $n_k$, and let all weights $\mu_j$ and $\nu_k$ be nonzero.
    Repeated blocks are allowed. If
    $\bigoplus_j\mu_jA_j$ and $\bigoplus_k\nu_kB_k$ generate the same vectors
    at every positive length, then there are complex numbers $c_j$ with
    $c_j^{m_j}=1$ and a matrix $W$ with $WW^\dagger=\Id$ and
    $W^\dagger W=\Id$ such that, for every $i$,
    \begin{align}
        \bigoplus_jc_j\mu_jA_j^i & =W\Bigl(\bigoplus_k\nu_kB_k^i\Bigr)W^\dagger. \notag
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:periodic_grouped_unitary, thm:periodic_ft_equal_case_trace_preserving}
    Theorem~\ref{thm:periodic_grouped_unitary} groups both sums into
    multiplicity forms $P$ and $Q$ over bases of periodic tensors without
    repetition, with unitaries $Y_A$ and $Y_B$. The two forms generate the
    same vectors at every positive length, so
    Theorem~\ref{thm:periodic_ft_equal_case_trace_preserving} gives
    multiplicity scalars $z_{j,q}$ with $z_{j,q}^{m_{\rho_j}}=1$ and a unitary
    $Y$ with $Z_zP^i=YQ^iY^\dagger$. Conjugating $Z_z$ back by $Y_A$ as in
    Theorem~\ref{thm:periodic_grouped_unitary} gives the scalars
    $c_{e(j,q)}=z_{j,q}$ and $W=Y_AYY_B^\dagger$. Each original block has
    the period of its representative, so $c_k^{m_k}=1$.
\end{proof}

\begin{theorem}[Physical symmetries of a tensor in irreducible form~II]%
    \label{thm:irreducible_form_physical_symmetry}
    \lean{MPSTensor.exists_diagonal_unitary_of_irreducibleForm_physical_symmetry,
        MPSTensor.exists_blockScalar_unitary_of_periodic_physical_symmetry,
        MPSTensor.toTensorFromBlocks_sum_smul,
        MPSTensor.blockScalarMatrix_eq_diagonal,
        MPSTensor.blockScalarMatrix_mem_unitaryGroup,
        MPSTensor.sameMPV₂_toTensorFromBlocks_of_periodic_phases}
    \leanok
    \uses{def:block_diagonal_tensor, def:same_mpv2, def:same_mpv2_pos}
    Let $A^i=\bigoplus_j\mu_jA_j^i$ with nonzero weights $\mu_j$, where every
    $A_j$ is irreducible with spectral radius one and has a trace-preserving
    transfer map. Let $u$ be a unitary $d\times d$ matrix with
    $u^{\otimes N}\ket{V_N(A)}=\ket{V_N(A)}$ for every $N>0$. Then there are a
    diagonal unitary $Z$ with $[A^i,Z]=0$ for all $i$ and
    $\mc V(A)=\mc V(ZA)$, and a unitary $U$, such that for every $i'$
    \begin{align}
        \sum_iu^{i',i}A^i & =ZUA^{i'}U^\dagger. \notag
    \end{align}
    Here $Z=\bigoplus_jz_j\Id$ with $z_j^{m_j}=1$, where $m_j$ is the
    period of $A_j$; this establishes the symmetry corollary of
    \cite[lines~834--845]{DeLasCuevas2017Irreducible} without requiring the
    fixed points in irreducible form~II to be diagonal.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:spectral_periodic_period_existence,
        thm:periodic_block_families_unitary_matching,
        thm:pgvwc07_normalized_periodic_state_vector_decomposition}
    Theorem~\ref{thm:spectral_periodic_period_existence} gives the periods
    $m_j$. Mixing the physical index commutes with the direct sum:
    $\sum_iu^{i',i}A^i=\bigoplus_j\mu_jC_j^{i'}$ with
    $C_j^{i'}=\sum_iu^{i',i}A_j^i$. Since $u$ is unitary, $\E_{C_j}=\E_{A_j}$,
    so $C_j$ is periodic with the same period. By hypothesis
    $\bigoplus_j\mu_jC_j$ and $A$ generate the same vectors at every positive
    length, so Theorem~\ref{thm:periodic_block_families_unitary_matching} gives
    $c_j$ with $c_j^{m_j}=1$ and a unitary $U$ with
    $\bigoplus_jc_j\mu_jC_j^{i'}=UA^{i'}U^\dagger$. With $z_j=c_j^{-1}$ and
    $Z=\bigoplus_jz_j\Id$ this is $\sum_iu^{i',i}A^i=ZUA^{i'}U^\dagger$.
    The matrix $Z$ is diagonal with entries of modulus one and commutes with
    every $A^i$. Finally,
    $V_N(ZA)=\sum_jz_j^N\mu_j^NV_N(A_j)$: if $m_j\mid N$ then $z_j^N=1$, and
    otherwise $V_N(A_j)=0$
    (Theorem~\ref{thm:pgvwc07_normalized_periodic_state_vector_decomposition}).
    Hence $\mc V(A)=\mc V(ZA)$.
\end{proof}

\begin{theorem}[Conditional group-action $Z$-gauge reformulation of the symmetry corollary]%
    \label{cor:cor_4_1_physical_symmetry_zgauge}
    \lean{MPSTensor.cor_4_1_physical_symmetry_zgauge}
    \lean{MPSTensor.PeriodicEqualCaseFT}
    \leanok
    \uses{def:irreducible_form, def:z_gauge_equiv, def:on_site_symmetric, def:twisted_tensor, thm:periodic_ft}
    Let $A$ be an MPS tensor in irreducible form~II and let
    $U : G \to \GL_d(\C)$ be a representation of a group $G$ on the
    physical leg, acting unitarily ($U(g)\, U(g)^{\dagger} = \Id$).
    Assume the periodic equal-case Fundamental Theorem
    (Theorem~\ref{thm:periodic_ft}) for the dimensions $(d, D)$.
    If $A$ is on-site symmetric under~$U$ --- that is, the twisted tensor
    $\widetilde{A}_g$ has the same MPV family as~$A$ for every $g \in G$
    --- then for each $g \in G$ there exists a positive period $m_g$ and
    a $\Z_{m_g}$-gauge equivalence between~$A$ and~$\widetilde{A}_g$.

    Concretely there are matrices $Z_g$ and~$Y_g$, with $Z_g^{m_g} = \Id$
    and $[A^i, Z_g] = 0$, satisfying
    $Z_g A^i = Y_g \widetilde{A}_g^{\,i}Y_g^{-1}$ for every physical
    index~$i$.

    % Source comparison: the source corollary in
    % \cite[lines~834--845]{DeLasCuevas2017Irreducible} is stated for a single
    % local unitary. It concludes that, for a local unitary $u$, there are a
    % diagonal unitary $Z$ and a virtual unitary $Y$ satisfying, for every
    % physical index $i'$,
    % \[
    %     \sum_i u^{i',i} A^i = Z\,Y\,A^{i'}Y^{\dagger}.
    % \]
    % The statement here is a pointwise group-action reformulation using the
    % periodic equal-case Fundamental Theorem as an explicit hypothesis. It does
    % not assert the diagonal normalization of $Z_g$ or the exact displayed
    % source equation; the source corollary itself is
    % Theorem~\ref{thm:irreducible_form_physical_symmetry}.
\end{theorem}

\begin{proof}\leanok
    \uses{def:rotate_physical, thm:periodic_ft, thm:irreducible_form_rotate_physical, cor:symmetry_periodic_assembly}
    The twisted tensor $\widetilde{A}_g$ coincides with the physical-index
    rotation $A_{U(g)}$ of Definition~\ref{def:rotate_physical}.
    By Theorem~\ref{thm:irreducible_form_rotate_physical} the rotated
    tensor remains in irreducible form~II, and the on-site symmetry
    hypothesis gives the equality $\mc{V}(A) = \mc{V}(A_{U(g)})$.
    The periodic equal-case Fundamental Theorem
    (Theorem~\ref{thm:periodic_ft}) then yields the asserted
    $\Z_{m_g}$-gauge equivalence.
\end{proof}

\begin{definition}[Conditional periodic projective-rigidity hypothesis]%
    \label{def:periodic_projective_rigidity}
    \lean{MPSTensor.PeriodicProjectiveRigidity}
    \leanok
    \uses{def:twisted_tensor, def:on_site_symmetric}
    Let $A$ be an MPS tensor with physical dimension $d$ and bond dimension $D$,
    and assume it is on-site symmetric under a group action
    $U : G \to \GL_d(\C)$. The tensor $A$ satisfies the
    \emph{periodic projective-rigidity hypothesis} if one may choose a
    family of virtual gauges $Y : G \to \GL_D(\C)$ and a scalar
    $2$-cochain $u : G \times G \to \C^{\times}$ such that

    \begin{enumerate}
        \item every $Y_g$ realizes the pointwise $Z$-gauge intertwining
              modeled on the symmetry corollary with the twisted tensor
              $\widetilde{A}_g$ for some accompanying
              $Z_g$ (with $Z_g^{m_g} = \Id$ and $[A^i, Z_g] = 0$); and
        \item the family obeys the projective multiplication law on the
              bond space: $Y_h\, Y_g = u(g, h)\, Y_{gh}$.
    \end{enumerate}

    The source proves the corresponding intertwining for a single local unitary.
    The group-indexed coherent choice and the projective multiplication law are
    additional assumptions: after proving the symmetry corollary, the source
    paper says that its consequences will be explored elsewhere. The injective
    case reduces this coherence to scalar uniqueness of gauges
    (cf.~\S\ref{sec:injective_symmetry_cocycle}); in the periodic
    setting there is a non-trivial $\Z_m$-gauge ambiguity, so coherence of the
    factor system $u$ is an independent analytic assumption.
\end{definition}

\begin{theorem}[Conditional projective representation from symmetry gauges]%
    \label{thm:cor_4_1_projective_rep}
    \lean{MPSTensor.cor_4_1_projective_rep}
    \leanok
    \uses{def:periodic_projective_rigidity}
    Assume the periodic projective-rigidity hypothesis
    (Definition~\ref{def:periodic_projective_rigidity}) for the tensor $A$ and
    the physical action $U$. Then the virtual gauge family from that
    hypothesis defines a projective representation of $G$ on the bond space:
    there exist a scalar $2$-cochain
    $\omega : G \times G \to \C^{\times}$ and virtual gauges
    $X : G \to \GL_D(\C)$ satisfying
    $X(g)X(h) = \omega(g,h)X(gh)$, while for each $g$ the matrix
    $X(g^{-1})$ still realizes the pointwise
    $\Z_{m_g}$-intertwining, encoded in the rigidity hypothesis, between $A$
    and the twisted tensor $\widetilde{A}_g$.
\end{theorem}

\begin{proof}\leanok
    \uses{def:periodic_projective_rigidity}
    By Definition~\ref{def:periodic_projective_rigidity}, let $(Y, u)$ be the
    virtual gauge family and factor system. Set
    $X(g) := Y(g^{-1})$ and $\omega(g, h) := u(h^{-1}, g^{-1})$. The
    projective multiplication law $X(g)X(h) = \omega(g,h)X(gh)$
    is then exactly the $(h^{-1}, g^{-1})$-instance of the factor-system
    identity, using $(gh)^{-1} = h^{-1}g^{-1}$. The intertwining
    relation between $A$ and $\widetilde{A}_g$ carries over unchanged.
\end{proof}

\begin{theorem}[Cocycle identity for the upgraded factor system]%
    \label{thm:cor_4_1_projective_rep_cocycle}
    \lean{MPSTensor.cor_4_1_projective_rep_cocycle}
    \leanok
    \uses{thm:cor_4_1_projective_rep}
    If the bond dimension $D$ is positive, the factor system $\omega$
    produced by Theorem~\ref{thm:cor_4_1_projective_rep} satisfies the
    multiplicative $2$-cocycle identity
    $\omega(g,h)\omega(gh,k) = \omega(g,hk)\omega(h,k)$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:cor_4_1_projective_rep}
    Associativity of matrix multiplication applied to
    $X(g)X(h)X(k)$, combined with the projective multiplication
    law, gives
    \begin{align}
        \omega(g,h)\omega(gh,k)X((gh)k)
         & =\omega(g,hk)\omega(h,k)X(g(hk)).
        \label{eq:pft_assoc_scalar_multiple}
    \end{align}
    Since $(gh)k = g(hk)$, both sides of
    (\ref{eq:pft_assoc_scalar_multiple}) are scalar multiples of the
    same matrix $X((gh)k)$. Since $X((gh)k) \in \GL_D(\C)$ and $D>0$,
    the matrix $X((gh)k)$ is nonzero. Cancelling $X((gh)k)$ from both sides of
    (\ref{eq:pft_assoc_scalar_multiple}) yields
    $\omega(g,h)\omega(gh,k) = \omega(g,hk)\omega(h,k)$ in $\C^{\times}$.
\end{proof}

\begin{remark}[External cohomological interpretation]%
    \label{rem:cor_4_1_projective_rep_spt}
    \uses{thm:cor_4_1_projective_rep_cocycle}
    Theorem~\ref{thm:cor_4_1_projective_rep_cocycle} establishes a $\C^\times$-valued
    $2$-cocycle under the conditional periodic projective-rigidity hypothesis of
    Definition~\ref{def:periodic_projective_rigidity}. In the broader SPT
    literature \cite{PerezGarcia2008StringOrder,Schuch2011Phases}, this factor
    system is given an external physical interpretation as defining a
    cohomology class in $H^2(G,\mathrm{U}(1))$; no such $\mathrm{U}(1)$ reduction is asserted
    here.
\end{remark}

\begin{theorem}[Conditional projective representation and cocycle from symmetry]%
    \label{thm:projectiveRep_of_symmetry}
    \lean{MPSTensor.projectiveRep_of_symmetry}
    \leanok
    \uses{thm:cor_4_1_projective_rep, thm:cor_4_1_projective_rep_cocycle}
    Under the periodic projective-rigidity hypothesis
    (Definition~\ref{def:periodic_projective_rigidity}), if $D>0$, then the
    virtual gauge family defines a projective representation of $G$ on the bond
    space whose factor system is a $2$-cocycle.
\end{theorem}
