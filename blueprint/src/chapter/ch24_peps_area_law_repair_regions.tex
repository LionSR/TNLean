\section{Replacement regions and the recursive repair family}
\label{sec:area_law_repair_regions}

The replacement construction of \cite[Section~11]{OpenAI2026AreaLaw}
uses the eight triangles of a square divided at its side midpoints.
We retain the removal half-side $d(S)$ and square $Q(v,S)$ of
Definition~\ref{def:area_law_repair_square}. Distances in the plane are
maximum-norm distances, and colors are represented by $0$ and $1$.

\begin{definition}[The normalized fan and its colored runs]
    \label{def:area_law_repair_fan}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairFanSplit,
        TNLean.PEPS.AreaLaw.Geometry.RepairFanSlot,
        TNLean.PEPS.AreaLaw.Geometry.repairExteriorColor,
        TNLean.PEPS.AreaLaw.Geometry.RepairFanRun,
        TNLean.PEPS.AreaLaw.Geometry.repairFanRunExteriorColor,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementColor}
    \leanok
    \uses{def:area_law_fan_runs,thm:area_law_fan_run_color}
    Divide each side of $[-1,1]^2$ at its midpoint. Index the eight
    resulting perimeter segments by
    $J=\{0,1,2,3\}\times\{0,1\}$ in their cyclic order, and write
    $[a_s,b_s]$ for segment $s$. Its closed fan triangle is
    $T_s=\operatorname{conv}\{0,a_s,b_s\}$.

    Let $I$ be a set of region identifiers, $\xi:I\to\{0,1\}$ their
    colors, and $\sigma:J\to I$ an assignment. Give triangle $s$ the
    exterior color $\lambda_\sigma(s)=\xi(\sigma(s))$. Join consecutive
    triangles exactly when their exterior colors agree. Write
    $\mathcal R(\xi,\sigma)$ for the connected components of this graph.
    For each run $H\in\mathcal R(\xi,\sigma)$ and any $s\in H$, set
    \begin{align}
        T_H & =\bigcup_{s\in H}T_s,
            & \lambda_\sigma(H)                & =\lambda_\sigma(s),
            & \lambda_\sigma^{\mathrm{new}}(H) & =1-\lambda_\sigma(H).
        \label{eq:area_law_repair_run}
    \end{align}
    The common exterior color is well defined by color constancy on a
    run. Distinct runs retain distinct identifiers when their colors agree.
\end{definition}

\begin{definition}[Birth and current region of a replacement]
    \label{def:area_law_repair_birth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairAffineMap,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementBirthRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementOpenRegion}
    \leanok
    \uses{def:area_law_repair_fan,def:area_law_repair_square}
    For a center $v\in\mathbb R^2$, scale $S\in\mathbb R$ and run $H$,
    define
    \begin{align}
        F_{v,S}(x) & =v+d(S)x,                                     \\
        B(v,S,H)   & =F_{v,S}(T_H),
                   & U(v,S,H)       & =\operatorname{int}B(v,S,H).
        \label{eq:area_law_repair_birth}
    \end{align}
    The birth region is the affine image of the closed triangle union $T_H$,
    and its interior is taken after forming the union.
\end{definition}

\begin{theorem}[Disjoint replacement regions at one parent]
    \label{thm:area_law_repair_replacement_disjoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairReplacementOpenRegion_pairwise_disjoint}
    \leanok
    \uses{def:area_law_repair_birth,thm:area_law_fan_run_disjoint_interiors}
    Fix a center $v$, a scale $S$ and an exterior assignment.
    Distinct runs $H\ne K$ give disjoint current replacement regions,
    $U(v,S,H)\cap U(v,S,K)=\varnothing$.
    The assertion includes distinct runs of the same color and reflex runs.
\end{theorem}
\begin{proof}\leanok
    The map $x\mapsto v+d(S)x$ is a homeomorphism since $d(S)>0$.
    A homeomorphism carries the interior of a set to the interior of its
    image and preserves disjointness. Apply these facts to the whole-run
    interiors in Theorem~\ref{thm:area_law_fan_run_disjoint_interiors}.
\end{proof}

\begin{definition}[Assignments from the current working wedges]
    \label{def:area_law_repair_current_assignment}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairCurrentFanTriangle,
        TNLean.PEPS.AreaLaw.Geometry.repairCurrentSectorAssignment,
        TNLean.PEPS.AreaLaw.Geometry.repairCurrentAssignments}
    \leanok
    \uses{def:area_law_repair_fan,def:area_law_repair_square}
    Let $I$ be nonempty and let $(R_i)_{i\in I}$ be the current regions.
    For $v\in\mathbb R^2$, $r\in\mathbb R$ and $s\in J$, put
    $W_s(v,r)=\{v+rx:x\in T_s\}$. Choose
    \begin{align}
        \sigma^R_{v,r}(s)
         & =\varepsilon\{i\in I:\operatorname{int}W_s(v,r)\subseteq R_i\},
        \\
        \sigma^R_p(s)
         & =\sigma^R_{v_p,\,6d(S_p)}(s).
        \label{eq:area_law_repair_assignment}
    \end{align}
    Here $\varepsilon$ selects an element of the indicated set when it
    is nonempty, and an arbitrary element of $I$ otherwise. Thus the
    assignment at a parent uses the current regions in the working
    square of half-side $6d(S_p)$, before any removal.
\end{definition}

\begin{definition}[Simultaneous replacement of the current family]
    \label{def:area_law_repair_successor_regions}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairReplacementIndex,
        TNLean.PEPS.AreaLaw.Geometry.RepairRegionIndex,
        TNLean.PEPS.AreaLaw.Geometry.repairNextBirthRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairNextCurrentRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairNextRegionColor}
    \leanok
    \uses{def:area_law_repair_birth,def:area_law_simultaneous_remnants}
    Let $j\ge0$ be the current generation. Let $B_i,R_i$ and $\xi_i$
    be the birth sets, current regions and colors indexed by $I$.
    Let $K$ index the current parent occurrences, with locations $v_p$,
    scales $S_p$ and assignments $\sigma_p:J\to I$. For a threshold
    $S_*\in\mathbb R$, set
    \begin{align}
        K_+ & =\{p\in K:S_*\le d(S_p)\},                                                             \\
        I_{\mathrm{new}}
            & =\{j+1\}\times\coprod_{p\in K_+}\mathcal R(\xi,\sigma_p),
            & I'                                                        & =I\sqcup I_{\mathrm{new}}, \\
        Q_K & =\bigcup_{p\in K}Q(v_p,S_p).
        \label{eq:area_law_repair_indices}
    \end{align}
    With old and new identifiers distinguished by this disjoint union,
    for each $i\in I$ and each new identifier
    $q=(j+1,p,H)$, define
    \begin{align}
        B'_i & =B_i,
             & R'_i           & =R_i\setminus Q_K,
             & \xi'_i         & =\xi_i,                   \\
        B'_q & =B(v_p,S_p,H),
             & R'_q           & =U(v_p,S_p,H),
             & \xi'_q         & =1-\lambda_{\sigma_p}(H).
        \label{eq:area_law_repair_successor}
    \end{align}
    Every old current region loses the full union $Q_K$ of closed squares,
    including those with $d(S_p)<S_*$, while only parents in $K_+$ contribute
    replacement regions.
\end{definition}

\begin{theorem}[Disjointness after simultaneous repair]
    \label{thm:area_law_repair_next_disjoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairNextCurrentRegion_pairwise_disjoint}
    \leanok
    \uses{def:area_law_repair_successor_regions}
    Use the notation of
    Definition~\ref{def:area_law_repair_successor_regions} for one
    simultaneous repair, as in \cite[Section~11]{OpenAI2026AreaLaw}.
    Suppose that the following hold:
    \begin{enumerate}
        \item The old current regions are pairwise disjoint:
              $R_i\cap R_k=\varnothing$ for distinct $i,k\in I$.
        \item The closed removal squares are pairwise disjoint:
              $Q(v_p,S_p)\cap Q(v_q,S_q)=\varnothing$ for distinct
              $p,q\in K$.
        \item For every $p\in K_+$ and every
              $H\in\mathcal R(\xi,\sigma_p)$,
              $U(v_p,S_p,H)\subseteq Q(v_p,S_p)$.
    \end{enumerate}
    Then the next current regions $(R'_x)_{x\in I'}$ are pairwise
    disjoint: $R'_x\cap R'_y=\varnothing$ whenever $x\ne y$.
    Distinct runs of the same color and reflex runs are included.
\end{theorem}

\begin{proof}\leanok
    \uses{def:area_law_repair_successor_regions,
        thm:area_law_repair_replacement_disjoint}
    Two distinct old identifiers $i,k\in I$ satisfy
    $R'_i\subseteq R_i$ and $R'_k\subseteq R_k$, so their next
    current regions are disjoint by the first assumption.

    For an old identifier $i$ and a fresh identifier
    $x=(j+1,p,H)$, the third assumption gives
    $R'_x=U(v_p,S_p,H)\subseteq Q(v_p,S_p)\subseteq Q_K$.
    Since $R'_i=R_i\setminus Q_K$, these two regions are disjoint.
    Here $Q_K$ includes every parent square, including terminal squares.

    Two fresh identifiers with distinct parents $p\ne q$ have current
    regions contained in $Q(v_p,S_p)$ and $Q(v_q,S_q)$, respectively.
    The second and third assumptions therefore give disjointness.

    Finally, two distinct fresh identifiers with the same parent have
    distinct runs, since both have generation $j+1$.
    Theorem~\ref{thm:area_law_repair_replacement_disjoint} gives
    disjointness of their current regions, including equal-colored or
    reflex runs. These cases exhaust all pairs of distinct identifiers
    in $I'$.
\end{proof}

\begin{definition}[The recursive repair family]
    \label{def:area_law_repair_stage}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairStage,
        TNLean.PEPS.AreaLaw.Geometry.initialRepairStage,
        TNLean.PEPS.AreaLaw.Geometry.repairNextStage,
        TNLean.PEPS.AreaLaw.Geometry.repairStage}
    \leanok
    \uses{def:area_law_repair_current_assignment,
        def:area_law_repair_successor_regions,
        def:area_law_repair_terminal_children,
        thm:area_law_uniform_separated_initial_marks,
        def:area_law_initial_open_regions,def:area_law_initial_region_colors}
    A stage is a tuple
    $\mathcal D=(I,*,B,R,\xi,K,v,S)$ consisting of a set $I$ of region
    identifiers, a distinguished identifier $*\in I$, families
    $B_i,R_i\subseteq\mathbb R^2$ and $\xi_i\in\{0,1\}$, a set $K$
    of mark occurrences, and functions $v:K\to\mathbb R^2$ and
    $S:K\to\mathbb R$.

    Fix a finite domain $\Lambda$, a cut $A\subseteq\Lambda$, an origin
    $o$, an integer width $C_0\ge2$ and a lower layer
    $k_0\ge50{,}000{,}000$. Let $Z$ be the set of endpoints of
    $\partial_\Lambda A$. Choose once, jointly, the sparse residues
    $(a_k,b_k)$, their finite union $M$ of initial marks and its least
    incident side function $s$ from
    Theorem~\ref{thm:area_law_uniform_separated_initial_marks}.
    Use these same residues for the initial identifier family
    $\mathcal I$, its birth regions $X_i^0$ and its colors $\lambda(i)$.
    Define the initial stage by
    \begin{align}
        I_0       & =\mathcal I,
                  & *_0          & =*,
                  & K_0          & =M,                       \\
        B_i^{(0)} & =X_i^0,
                  & R_i^{(0)}    & =\operatorname{int}X_i^0,
                  & \xi_i^{(0)}  & =\lambda(i),              \\
        v_p^{(0)} & =p,
                  & S_p^{(0)}    & =s(p).
        \label{eq:area_law_repair_initial}
    \end{align}
    Fix a threshold $S_*\in\mathbb R$. For a stage $\mathcal D$ and
    generation $j$, let
    $\Phi_{j,S_*}(\mathcal D)$ use the assignments
    $\sigma^R_p$ from~(\ref{eq:area_law_repair_assignment}), the region
    families~(\ref{eq:area_law_repair_successor}), and the marks
    (\ref{eq:area_law_repair_next_marks}). Its distinguished identifier
    is the retained copy of $*$. The recursive family is
    \begin{align}
        \mathcal D_0 & =\mathcal D_{\mathrm{init}},
                     & \mathcal D_{j+1}             & =\Phi_{j,S_*}(\mathcal D_j).
        \label{eq:area_law_repair_iteration}
    \end{align}
    At each generation the assignments are selected from that stage's
    current regions before simultaneous removal.
\end{definition}

\begin{theorem}[Decrease of retained current regions]
    \label{thm:area_law_repair_current_inclusion_subset}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_current_oldInclusion_subset}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_repair_inclusions}
    Let $R_i^{(j)}$ be the current regions of the actual repair family.
    For all $j,n\ge0$ and $i\in I_j$,
    \begin{align}
        R_{\iota_{j,j+n}(i)}^{(j+n)} & \subseteq R_i^{(j)}.\notag
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:area_law_repair_successor_regions,def:area_law_repair_inclusions}
    Induct on $n$. At $n=0$ the inclusion is the identity. At each
    successor step, the retained current region loses the removal union,
    and is therefore contained in the current region at the preceding
    generation. The induction hypothesis completes the proof.
\end{proof}

\begin{theorem}[Current regions under successive inclusions]
    \label{thm:area_law_repair_current_inclusion_add}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_current_oldInclusion_add}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_repair_inclusions}
    Let $R_i^{(j)}$ be the current regions of the actual repair family.
    For all $j,n,m\ge0$ and $i\in I_j$,
    \begin{align}
        R_{\iota_{j,j+(n+m)}(i)}^{(j+(n+m))}
         & =R_{\iota_{j+n,(j+n)+m}(\iota_{j,j+n}(i))}^{((j+n)+m)}.\notag
    \end{align}
\end{theorem}

\begin{proof}\leanok
    \uses{def:area_law_repair_successor_regions,def:area_law_repair_inclusions}
    Induct on $m$. The case $m=0$ follows from the identity inclusion.
    At the next step, each retained region loses the removal union at
    its current generation. The induction hypothesis identifies the two
    current regions, and $j+(n+m)=(j+n)+m$ identifies the removal unions.
    Subtracting these equal unions gives the claimed equality.
\end{proof}
