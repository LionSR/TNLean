\section{Replacement regions and the recursive repair family}
\label{sec:area_law_repair_regions}

The replacement construction of \cite[Section~11]{OpenAI2026AreaLaw}
uses the eight triangles of a square divided at its side midpoints.
We retain the removal half-side $d(S)$ and square $Q(v,S)$ of
Definition~\ref{def:area_law_repair_square}. Distances in the plane are
maximum-norm distances, and colors are represented by $0$ and $1$.

\begin{definition}[The normalized fan and its colored runs]
    \label{def:area_law_repair_fan}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairFanSplit,
        TNLean.PEPS.AreaLaw.Geometry.RepairFanSlot,
        TNLean.PEPS.AreaLaw.Geometry.repairExteriorColor,
        TNLean.PEPS.AreaLaw.Geometry.RepairFanRun,
        TNLean.PEPS.AreaLaw.Geometry.repairFanRunExteriorColor,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementColor}
    \leanok
    \uses{def:area_law_fan_runs,thm:area_law_fan_run_color}
    Divide each side of $[-1,1]^2$ at its midpoint. Index the eight
    resulting perimeter segments by
    $J=\{0,1,2,3\}\times\{0,1\}$ in their cyclic order, and write
    $[a_s,b_s]$ for segment $s$. Its closed fan triangle is
    $T_s=\operatorname{conv}\{0,a_s,b_s\}$.

    Let $I$ be a set of region identifiers, $\xi:I\to\{0,1\}$ their
    colors, and $\sigma:J\to I$ an assignment. Give triangle $s$ the
    exterior color $\lambda_\sigma(s)=\xi(\sigma(s))$. Join consecutive
    triangles exactly when their exterior colors agree. Write
    $\mathcal R(\xi,\sigma)$ for the connected components of this graph.
    For each run $H\in\mathcal R(\xi,\sigma)$ and any $s\in H$, set
    \begin{align}
        T_H & =\bigcup_{s\in H}T_s,
            & \lambda_\sigma(H)                & =\lambda_\sigma(s),
            & \lambda_\sigma^{\mathrm{new}}(H) & =1-\lambda_\sigma(H).
        \label{eq:area_law_repair_run}
    \end{align}
    The common exterior color is well defined by color constancy on a
    run. Distinct runs retain distinct identifiers when their colors agree.
\end{definition}

\begin{definition}[Birth and current region of a replacement]
    \label{def:area_law_repair_birth}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairAffineMap,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementBirthRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairReplacementOpenRegion}
    \leanok
    \uses{def:area_law_repair_fan,def:area_law_repair_square}
    For a center $v\in\mathbb R^2$, scale $S\in\mathbb R$ and run $H$,
    define
    \begin{align}
        F_{v,S}(x) & =v+d(S)x,                                     \\
        B(v,S,H)   & =F_{v,S}(T_H),
                   & U(v,S,H)       & =\operatorname{int}B(v,S,H).
        \label{eq:area_law_repair_birth}
    \end{align}
    The birth region is the affine image of the closed triangle union $T_H$,
    and its interior is taken after forming the union.
\end{definition}

\begin{theorem}[Disjoint replacement regions at one parent]
    \label{thm:area_law_repair_replacement_disjoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairReplacementOpenRegion_pairwise_disjoint}
    \leanok
    \uses{def:area_law_repair_birth,thm:area_law_fan_run_disjoint_interiors}
    Fix a center $v$, a scale $S$ and an exterior assignment.
    Distinct runs $H\ne K$ give disjoint current replacement regions,
    $U(v,S,H)\cap U(v,S,K)=\varnothing$.
    The assertion includes distinct runs of the same color and reflex runs.
\end{theorem}
\begin{proof}\leanok
    The map $x\mapsto v+d(S)x$ is a homeomorphism since $d(S)>0$.
    A homeomorphism carries the interior of a set to the interior of its
    image and preserves disjointness. Apply these facts to the whole-run
    interiors in Theorem~\ref{thm:area_law_fan_run_disjoint_interiors}.
\end{proof}

\begin{theorem}[Lattice points in a replacement square]
    \label{thm:area_law_repair_replacement_lattice_cover}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_mem_repairReplacementOpenRegion_of_mem_repairSquare}
    \leanok
    \uses{def:area_law_dyadic_origin,def:area_law_affine_mesh,
        def:area_law_repair_birth}
    Set $o=(\sqrt2,\sqrt3)$. Let $v\in\mathcal M(o,1)=o+\mathbb Z^2$
    and $S\ge1$. For an arbitrary identifier set $I$, binary labels
    $\xi:I\to\{0,1\}$ and assignment $\sigma:J\to I$, every integer
    lattice point of $Q(v,S)$ lies in the current region of a run:
    \begin{align}
        Q(v,S)\cap\mathbb Z^2
         & \subseteq\bigcup_{H\in\mathcal R(\xi,\sigma)}U(v,S,H).\notag
    \end{align}
    Lattice points are identified with their points in $\mathbb R^2$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_square,thm:area_law_repair_halfside_integral,
        thm:area_law_dyadic_origin_lines,lem:area_law_repair_fan_cones,
        def:area_law_repair_birth}
    Write $v=o+m$ with $m\in\mathbb Z^2$, and put $d=d(S)$.
    By Theorem~\ref{thm:area_law_repair_halfside_integral}, $d$ is a positive integer.
    Let $z\in Q(v,S)\cap\mathbb Z^2$. The four sides of $Q(v,S)$
    and all its fan rays lie on integer translates of the four
    supporting-line families in
    Theorem~\ref{thm:area_law_dyadic_origin_lines}. They contain no
    lattice point. Consequently $\|z-v\|_\infty<d$, and $z-v$
    lies on none of the fan rays.

    By Lemma~\ref{lem:area_law_repair_fan_cones}, $z-v$ lies in a
    closed fan cone. Since it lies on neither boundary ray, it is in
    the corresponding open cone. Together with $\|z-v\|_\infty<d$,
    this places $z$ in $\operatorname{int}(v+dT_s)$ for some $s\in J$.
    If $H$ is the run containing $s$, then $v+dT_s\subseteq B(v,S,H)$.
    Monotonicity of the interior gives $z\in U(v,S,H)$.
\end{proof}

\begin{definition}[Assignments from the current working wedges]
    \label{def:area_law_repair_current_assignment}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairCurrentFanTriangle,
        TNLean.PEPS.AreaLaw.Geometry.repairCurrentSectorAssignment,
        TNLean.PEPS.AreaLaw.Geometry.repairCurrentAssignments}
    \leanok
    \uses{def:area_law_repair_fan,def:area_law_repair_square}
    Let $I$ be nonempty and let $(R_i)_{i\in I}$ be the current regions.
    For $v\in\mathbb R^2$, $r\in\mathbb R$ and $s\in J$, put
    $W_s(v,r)=\{v+rx:x\in T_s\}$. Choose
    \begin{align}
        \sigma^R_{v,r}(s)
         & =\varepsilon\{i\in I:\operatorname{int}W_s(v,r)\subseteq R_i\},
        \\
        \sigma^R_p(s)
         & =\sigma^R_{v_p,\,6d(S_p)}(s).
        \label{eq:area_law_repair_assignment}
    \end{align}
    Here $\varepsilon$ selects an element of the indicated set when it
    is nonempty, and an arbitrary element of $I$ otherwise. Thus the
    assignment at a parent uses the current regions in the working
    square of half-side $6d(S_p)$, before any removal.
\end{definition}

\begin{definition}[Simultaneous replacement of the current family]
    \label{def:area_law_repair_successor_regions}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairReplacementIndex,
        TNLean.PEPS.AreaLaw.Geometry.RepairRegionIndex,
        TNLean.PEPS.AreaLaw.Geometry.repairNextBirthRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairNextCurrentRegion,
        TNLean.PEPS.AreaLaw.Geometry.repairNextRegionColor}
    \leanok
    \uses{def:area_law_repair_birth,def:area_law_simultaneous_remnants}
    Let $j\ge0$ be the current generation. Let $B_i,R_i$ and $\xi_i$
    be the birth sets, current regions and colors indexed by $I$.
    Let $K$ index the current parent occurrences, with locations $v_p$,
    scales $S_p$ and assignments $\sigma_p:J\to I$. For a threshold
    $S_*\in\mathbb R$, set
    \begin{align}
        K_+ & =\{p\in K:S_*\le d(S_p)\},                                                             \\
        I_{\mathrm{new}}
            & =\{j+1\}\times\coprod_{p\in K_+}\mathcal R(\xi,\sigma_p),
            & I'                                                        & =I\sqcup I_{\mathrm{new}}, \\
        Q_K & =\bigcup_{p\in K}Q(v_p,S_p).
        \label{eq:area_law_repair_indices}
    \end{align}
    With old and new identifiers distinguished by this disjoint union,
    for each $i\in I$ and each new identifier
    $q=(j+1,p,H)$, define
    \begin{align}
        B'_i & =B_i,
             & R'_i           & =R_i\setminus Q_K,
             & \xi'_i         & =\xi_i,                   \\
        B'_q & =B(v_p,S_p,H),
             & R'_q           & =U(v_p,S_p,H),
             & \xi'_q         & =1-\lambda_{\sigma_p}(H).
        \label{eq:area_law_repair_successor}
    \end{align}
    Every old current region loses the full union $Q_K$ of closed squares,
    including those with $d(S_p)<S_*$, while only parents in $K_+$ contribute
    replacement regions.
\end{definition}

\begin{theorem}[Disjointness after simultaneous repair]
    \label{thm:area_law_repair_next_disjoint}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairNextCurrentRegion_pairwise_disjoint}
    \leanok
    \uses{def:area_law_repair_successor_regions}
    Use the notation of
    Definition~\ref{def:area_law_repair_successor_regions} for one
    simultaneous repair, as in \cite[Section~11]{OpenAI2026AreaLaw}.
    Suppose that the following hold:
    \begin{enumerate}
        \item The old current regions are pairwise disjoint:
              $R_i\cap R_k=\varnothing$ for distinct $i,k\in I$.
        \item The closed removal squares are pairwise disjoint:
              $Q(v_p,S_p)\cap Q(v_q,S_q)=\varnothing$ for distinct
              $p,q\in K$.
        \item For every $p\in K_+$ and every
              $H\in\mathcal R(\xi,\sigma_p)$,
              $U(v_p,S_p,H)\subseteq Q(v_p,S_p)$.
    \end{enumerate}
    Then the next current regions $(R'_x)_{x\in I'}$ are pairwise
    disjoint: $R'_x\cap R'_y=\varnothing$ whenever $x\ne y$.
    Distinct runs of the same color and reflex runs are included.
\end{theorem}

\begin{proof}\leanok
    \uses{def:area_law_repair_successor_regions,
        thm:area_law_repair_replacement_disjoint}
    Two distinct old identifiers $i,k\in I$ satisfy
    $R'_i\subseteq R_i$ and $R'_k\subseteq R_k$, so their next
    current regions are disjoint by the first assumption.

    For an old identifier $i$ and a fresh identifier
    $x=(j+1,p,H)$, the third assumption gives
    $R'_x=U(v_p,S_p,H)\subseteq Q(v_p,S_p)\subseteq Q_K$.
    Since $R'_i=R_i\setminus Q_K$, these two regions are disjoint.
    Here $Q_K$ includes every parent square, including terminal squares.

    Two fresh identifiers with distinct parents $p\ne q$ have current
    regions contained in $Q(v_p,S_p)$ and $Q(v_q,S_q)$, respectively.
    The second and third assumptions therefore give disjointness.

    Finally, two distinct fresh identifiers with the same parent have
    distinct runs, since both have generation $j+1$.
    Theorem~\ref{thm:area_law_repair_replacement_disjoint} gives
    disjointness of their current regions, including equal-colored or
    reflex runs. These cases exhaust all pairs of distinct identifiers
    in $I'$.
\end{proof}

\begin{definition}[Terminal interiors and next-generation marks]
    \label{def:area_law_repair_terminal_children}

    Definition~\ref{def:area_law_repair_successor_regions}, define
    \begin{align}
        E_{\mathrm{term}}
                            & =\bigcup_{\substack{p\in K                               \\d(S_p)<S_*}}
        \operatorname{int}Q(v_p,S_p),                                                  \\
        P                   & =\{-1,0,1\}^2,
                            & K'                         & =\{j+1\}\times K_+\times P, \\

    midpoints of a parent's square. Each nonterminal parent contributes nine
    child marks, whereas a terminal parent contributes neither replacement
    regions nor child marks.
\end{definition}

\begin{theorem}[Lattice coverage after one repair]
    \label{thm:area_law_repair_next_lattice_cover}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_mem_repairNextCurrentRegion_or_terminalInterior}
    \leanok
    \uses{def:area_law_repair_successor_regions,
        def:area_law_repair_terminal_children,def:area_law_affine_mesh,
        def:area_law_dyadic_origin}
    Consider one simultaneous repair at generation $j\in\mathbb N$,
    with old regions $(R_i)_{i\in I}$, parent set $K$, arbitrary binary
    labels and assignments, and stopping scale $S_*\in\mathbb R$.
    Set $o=(\sqrt2,\sqrt3)$, and suppose that every parent $p\in K$
    satisfies $v_p\in\mathcal M(o,1)$ and $S_p\ge1$.
    Every integer lattice point covered by an old region belongs to a
    next current region or to a terminal-square interior:
    \begin{align}
        \Bigl(\bigcup_{i\in I}R_i\Bigr)\cap\mathbb Z^2
         & \subseteq\Bigl(\bigcup_{x\in I'}R'_x\Bigr)\cup E_{\mathrm{term}}.\notag
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_successor_regions,
        def:area_law_repair_terminal_children,def:area_law_repair_birth,
        thm:area_law_repair_replacement_lattice_cover}
    Let $z\in\mathbb Z^2$ belong to an old region $R_i$.
    If $z$ lies outside every removal square, it belongs to the retained
    region $R'_i=R_i\setminus Q_K$.
    Otherwise choose $p\in K$ with $z\in Q(v_p,S_p)$.
    By Theorem~\ref{thm:area_law_repair_replacement_lattice_cover},
    $z\in U(v_p,S_p,H)$ for some run $H$ of the assignment at $p$.
    If $S_*\le d(S_p)$, this is a next current region.
    If $d(S_p)<S_*$, the containment
    $B(v_p,S_p,H)\subseteq Q(v_p,S_p)$ and monotonicity of the interior
    give $z\in\operatorname{int}Q(v_p,S_p)\subseteq E_{\mathrm{term}}$.
\end{proof}

\begin{definition}[The recursive repair family]
    \label{def:area_law_repair_stage}
    \lean{TNLean.PEPS.AreaLaw.Geometry.RepairStage,
        TNLean.PEPS.AreaLaw.Geometry.initialRepairStage,
        TNLean.PEPS.AreaLaw.Geometry.repairNextStage,
        TNLean.PEPS.AreaLaw.Geometry.repairStage}
    \leanok
    \uses{def:area_law_repair_current_assignment,
        def:area_law_repair_successor_regions,
        def:area_law_repair_terminal_children,
        thm:area_law_uniform_separated_initial_marks,
        def:area_law_initial_open_regions,def:area_law_initial_region_colors}
    A stage is a tuple
    $\mathcal D=(I,*,B,R,\xi,K,v,S)$ consisting of a set $I$ of region
    identifiers, a distinguished identifier $*\in I$, families
    $B_i,R_i\subseteq\mathbb R^2$ and $\xi_i\in\{0,1\}$, a set $K$
    of mark occurrences, and functions $v:K\to\mathbb R^2$ and
    $S:K\to\mathbb R$.

    Fix a finite domain $\Lambda$, a cut $A\subseteq\Lambda$, an origin
    $o$, an integer width $C_0\ge2$ and a lower layer
    $k_0\ge50{,}000{,}000$. Let $Z$ be the set of endpoints of
    $\partial_\Lambda A$. Choose once, jointly, the sparse residues
    $(a_k,b_k)$, their finite union $M$ of initial marks and its least
    incident side function $s$ from
    Theorem~\ref{thm:area_law_uniform_separated_initial_marks}.
    Use these same residues for the initial identifier family
    $\mathcal I$, its birth regions $X_i^0$ and its colors $\lambda(i)$.
    Define the initial stage by
    \begin{align}
        I_0       & =\mathcal I,
                  & *_0          & =*,
                  & K_0          & =M,                       \\
        B_i^{(0)} & =X_i^0,
                  & R_i^{(0)}    & =\operatorname{int}X_i^0,
                  & \xi_i^{(0)}  & =\lambda(i),              \\
        v_p^{(0)} & =p,
                  & S_p^{(0)}    & =s(p).
        \label{eq:area_law_repair_initial}
    \end{align}
    Fix a threshold $S_*\in\mathbb R$. For a stage $\mathcal D$ and
    generation $j$, let
    $\Phi_{j,S_*}(\mathcal D)$ use the assignments
    $\sigma^R_p$ from~(\ref{eq:area_law_repair_assignment}), the region
    families~(\ref{eq:area_law_repair_successor}), and the marks
    (\ref{eq:area_law_repair_next_marks}). Its distinguished identifier
    is the retained copy of $*$. The recursive family is
    \begin{align}
        \mathcal D_0 & =\mathcal D_{\mathrm{init}},
                     & \mathcal D_{j+1}             & =\Phi_{j,S_*}(\mathcal D_j).
        \label{eq:area_law_repair_iteration}
    \end{align}
    At each generation the assignments are selected from that stage's
    current regions before simultaneous removal.
\end{definition}

\begin{theorem}[Properties of the chosen initial repair marks]
    \label{thm:area_law_initial_repair_mark_facts}
    \lean{TNLean.PEPS.AreaLaw.Geometry.initialRepairStage_mark_spec}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_belt_cells,
        def:area_law_belt_cell_marks,def:area_law_fine_belt_scales}
    Fix a finite domain $\Lambda\subset\mathbb Z^2$, a cut
    $A\subseteq\Lambda$, an origin $o\in\mathbb R^2$, and integers
    $C_0\ge2$ and $k_0\ge50{,}000{,}000$. Let $F_k$ be the selected
    belt-cell indices for the residues used in the initial repair stage,
    and write $t_k=2^{\ell_k}$. Let $K_0$ be its mark set, with
    locations $v_p^{(0)}$ and scales $S_p^{(0)}$. The choices in
    Definition~\ref{def:area_law_repair_stage} have the following properties,
    as in the belt and initial-mark construction of
    \cite[Section~11]{OpenAI2026AreaLaw}.
    \begin{enumerate}
        \item For every $k\ge0$,
              \begin{align}
                  |F_k| & \le64(2C_0+1)^2|\partial_\Lambda A|
                  \,2^{-\delta_0 k/2}.\notag
              \end{align}
        \item The set $\{k\ge0:F_k\ne\varnothing\}$ is finite.
        \item For every $p\in K_0$, the scale $S_p^{(0)}$ is positive,
              and there are $k\ge k_0$ and $z\in F_k$ such that
              $v_p^{(0)}\in M_{\ell_k,z}$ and $S_p^{(0)}=t_k$.
        \item For every $p\in K_0$, $k\ge k_0$ and $z\in F_k$,
              the inclusion $v_p^{(0)}\in M_{\ell_k,z}$ implies
              $S_p^{(0)}\le t_k$.
        \item For distinct $p,q\in K_0$,
              \begin{align}
                  \operatorname{dist}_\infty(v_p^{(0)},v_q^{(0)})
                   & \ge\frac14\max(S_p^{(0)},S_q^{(0)}).\notag
              \end{align}
        \item The location map $p\mapsto v_p^{(0)}$ is injective.
    \end{enumerate}
\end{theorem}

\begin{proof}\leanok
    \uses{thm:area_law_uniform_separated_initial_marks,
        def:area_law_repair_stage}
    The joint choice defining the initial stage supplies the first five
    assertions. Its mark set is the selected finite set of points, and
    the location map is its inclusion into $\mathbb R^2$, which is
    injective.
\end{proof}

\begin{theorem}[Uniform weighted bound for the initial repair marks]
    \label{thm:area_law_initial_repair_weighted_sum}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_uniform_initialRepairStage_weighted_sum_bound}
    \leanok
    \uses{def:area_law_repair_stage}
    For every $P\in\mathbb R$, there is $B>0$, depending only on $P$
    and the fixed exponent $\delta_0$, with the following property.
    For every $\lambda\ge0$, origin $o\in\mathbb R^2$, finite domain
    $\Lambda\subseteq\mathbb Z^2$, cut $A\subseteq\Lambda$, and integers
    $C_0\ge2$ and $k_0\ge50{,}000{,}000$, take the actual initial
    repair marks $K_0$ and their scales $S_p^{(0)}$ from
    Definition~\ref{def:area_law_repair_stage}.
    Let $w:K_0\to\mathbb R$ satisfy
    $w(p)\le\lambda(k+1)^P$ for every $p\in K_0$ and every integer
    $k\ge k_0$ for which $S_p^{(0)}\le2^k$.
    Then, with $b=|\partial_\Lambda A|$,
    \begin{align}
        \sum_{p\in K_0}w(p)
         & \le576\lambda B(2C_0+1)^2b.\notag
    \end{align}
    The weights may have either sign.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_initial_repair_mark_facts,
        thm:area_law_belt_mark_count,thm:area_law_fine_belt_scales,
        thm:area_law_polynomial_finite_sums}
    Choose $B$ first from
    Theorem~\ref{thm:area_law_polynomial_finite_sums} at exponent $P$.
    The actual initial mark set is finite by its construction.
    For each root $p$, choose one incident selected cell at a layer
    $\kappa(p)\ge k_0$. These choices retain the existing marks,
    scales and belt residues. Theorem~\ref{thm:area_law_initial_repair_mark_facts}
    and the bound $\ell_k\le k$ give

\begin{theorem}[Counting retained regions in finitely many generations]
    \label{thm:area_law_repair_retained_prefix_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.finite_and_card_repairStage_retainedRegionIndex_le}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_repair_region_sites}

    stopping scale $S_*\in\mathbb R$, $C_0\ge2$, and
    $k_0\ge50000000$. Let $\Lambda\subset\mathbb Z^2$ be finite and
    $A\subseteq\Lambda$. Write $K_j$ for the mark occurrences at
    generation $j$, and let
    $I_A^{(j)}=\{i\in I_j\setminus\{*_j\}:A\cap R_i^{(j)}\ne\varnothing\}$.
    For every $N\in\mathbb N$, the set $I_A^{(N)}$ is finite and
    \begin{align}
        |I_A^{(N)}| & \le |I_A^{(0)}|+8\sum_{0\le j<N}|K_j|.\notag

\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:area_law_initial_open_disjoint,
        lem:area_law_repair_region_sites_finite,
        thm:area_law_repair_finite_marks,thm:area_law_cell_fan_run_count,
        def:area_law_repair_successor_regions}

    is finite. At step $j$, every old current region shrinks. An old
    non-dummy identifier meeting $A$ after this step therefore already
    belonged to $I_A^{(j)}$. Every fresh identifier records a
    nonterminal parent mark and one connected run. There are at most
    eight runs per parent, so there are at most $8|K_j|$ fresh
    identifiers. It follows that $I_A^{(j+1)}$ is finite and
    \begin{align}
        |I_A^{(j+1)}| & \le |I_A^{(j)}|+8|K_j|.\notag
    \end{align}
    Induction gives the stated bound. The argument counts identifiers;
    overlapping later regions or coincident parent locations do not
    affect the estimate.
\end{proof}
