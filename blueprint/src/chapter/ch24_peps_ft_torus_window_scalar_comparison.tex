\subsection{Corner comparison at the minimal torus sizes}
\label{subsec:peps_window_scalar_comparison}

\begin{lemma}[Enlarging an injective cyclic window]%
    \label{thm:peps_arc_rectangle_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.arcRectangle_injective}
    \leanok
    Let $\mathcal I$ be closed under finite nonempty unions and satisfy the
    $L\times K$ cyclic-window hypothesis (Definition~\ref{def:peps_normal_arc_windows}),
    where $L,K>0$. If $L\le a\le w$ and $K\le b\le h$, then
    $\mathcal R(s;a,b)\in\mathcal I$ for every $s$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_normal_arc_windows, lem:peps_regionInjectivityUnionClosure_finite}
    Since $\mathcal R(s;a,b)=\bigcup_{(i,j)\in\{0,\ldots,a-L\}\times\{0,\ldots,b-K\}}\mathcal R(s+(i,j);L,K)$,
    Lemma~\ref{lem:peps_regionInjectivityUnionClosure_finite} gives
    $\mathcal R(s;a,b)\in\mathcal I$.
\end{proof}

\begin{lemma}[Complement of a cyclic rectangle]%
    \label{thm:peps_arc_rectangle_complement}
    \lean{TNLean.PEPS.compl_torusArcRectangle}
    \leanok
    For $a<w$ and $b<h$, the complement of $\mathcal R(s;a,b)$ is
    \begin{align}
        \mathcal R(s;a,b)^c
         & =\mathcal R((s_1+a,s_2);w-a,h)
        \cup\mathcal R((s_1,s_2+b);a,h-b).
        \label{eq:peps_arc_complement}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    A vertex outside $\mathcal R(s;a,b)$ either has its first coordinate outside
    the cyclic interval of length $a$ or has its first coordinate inside that
    interval and its second coordinate outside the cyclic interval of length $b$,
    giving (\ref{eq:peps_arc_complement}).
\end{proof}

\begin{definition}[The two corner comparison regions]%
    \label{def:peps_window_corner_regions}
    \lean{TNLean.PEPS.windowCornerRectangle}
    \lean{TNLean.PEPS.windowCornerRegion}
    \lean{TNLean.PEPS.windowCornerVertex}
    \leanok
    Writing ${\lbrack a,b)}$ for the vertices with coordinate representative in
    $\{a,\ldots,b-1\}$, define the one-site comparison regions of
    \cite[proof of Theorem~3]{Molnar2018NormalPEPS} for $L,K>0$ by
    \begin{align}
        S   & ={\lbrack L,2L+1)}\times{\lbrack K,2K+1)},\notag            \\
        R   & =\bigl({\lbrack L+1,2L+1)}\times{\lbrack K,2K+1)}\bigr)
        \cup\bigl({\lbrack L,2L+1)}\times{\lbrack K+1,2K+1)}\bigr),\notag \\
        v_0 & =(L,K).
        \label{eq:peps_window_corner_regions}
    \end{align}
\end{definition}

\begin{lemma}[Restoring the omitted corner]%
    \label{thm:peps_window_corner_geometry}
    \lean{TNLean.PEPS.mem_windowCornerRegion}
    \lean{TNLean.PEPS.eq_windowCornerVertex_iff}
    \lean{TNLean.PEPS.windowCornerVertex_notMem}
    \lean{TNLean.PEPS.insert_windowCornerVertex_windowCornerRegion}
    \lean{TNLean.PEPS.compl_windowCornerRegion_eq_union}
    \leanok
    If $2L+1\le w$ and $2K+1\le h$, then $v=v_0$ in
    (\ref{eq:peps_window_corner_regions}) iff its coordinate representatives
    are $(L,K)$, $v_0\notin R$, $R\cup\{v_0\}=S$, and for $L,K>0$,
    \begin{align}
        R^c=S^c\cup\bigl({\lbrack 1,L+1)}\times{\lbrack 1,K+1)}\bigr).
        \label{eq:peps_window_corner_complement}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{def:peps_window_corner_regions}
    By (\ref{eq:peps_window_corner_regions}), $S\setminus R=\{v_0\}$ and
    $({\lbrack 1,L+1)}\times{\lbrack 1,K+1)})\cap S=\{v_0\}$, which yields
    $R\cup\{v_0\}=S$ and (\ref{eq:peps_window_corner_complement}).
\end{proof}

\begin{lemma}[Injectivity of the corner regions and their complements]%
    \label{thm:peps_window_corner_injectivity}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.windowCornerRegion_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.windowCornerRectangle_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.compl_windowCornerRectangle_injective}
    \lean{TNLean.PEPS.NormalTorusArcWindowInjectivityHypotheses.compl_windowCornerRegion_injective}
    \leanok
    Let $\mathcal I$ be closed under finite nonempty unions and satisfy the
    $L\times K$ cyclic-window hypothesis. If $L,K>0$, $2L+1\le w$, and
    $2K+1\le h$, then $R,S,R^c,S^c\in\mathcal I$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_arc_rectangle_injective, thm:peps_arc_rectangle_complement,
        thm:peps_window_corner_geometry}
    Lemma~\ref{thm:peps_arc_rectangle_injective} places $S$ (of size
    $(L+1)\times(K+1)$) and the two rectangles of $R$ (of sizes $L\times(K+1)$
    and $(L+1)\times K$) in $\mathcal I$. Since $w-L-1\ge L$ and $h-K-1\ge K$,
    Lemma~\ref{thm:peps_arc_rectangle_complement} and
    Lemma~\ref{thm:peps_arc_rectangle_injective} place $S^c\in\mathcal I$, and
    (\ref{eq:peps_window_corner_complement}) of
    Lemma~\ref{thm:peps_window_corner_geometry} then gives $R^c\in\mathcal I$.
\end{proof}

\begin{lemma}[Boundary edges of the corner regions]%
    \label{thm:peps_window_corner_boundary}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_windowCornerRegion}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_windowCornerRectangle}
    \leanok
    If $w,h>1$, $K>0$, $2L+1\le w$, and $2K+1\le h$, the upward edge from
    $(L,K)$ crosses $\partial R$ and the upward edge from $(L,K-1)$ crosses
    $\partial S$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_window_corner_geometry}
    By Lemma~\ref{thm:peps_window_corner_geometry} and $2L+1\le w$, $2K+1\le h$,
    $(L,K)\notin R$ with $(L,K+1)\in R$, and $(L,K-1)\notin S$ with $(L,K)\in S$.
\end{proof}

\begin{lemma}[Injectivity under identification of coordinates]%
    \label{thm:peps_region_injective_identification}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_transport}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_translate}
    \lean{TNLean.PEPS.regionBlockedTensorInjective_reindexTensor}
    \leanok
    For any graph isomorphism $\phi$, $A^\phi$ is injective on $\phi(R)$ iff
    $A$ is injective on $R$; replacing bond dimensions by equal dimensions
    preserves injectivity on $R$, and for translation-invariant $A$ every
    translate $tR$ has the same injectivity as $R$.
\end{lemma}
\begin{proof}\leanok
    Bijections of boundary and physical configurations induce invertible linear
    coordinate changes on the blocked tensor families, preserving linear
    independence.
\end{proof}

\begin{lemma}[Block proportionality from absorbed edge identities]%
    \label{thm:peps_block_proportional_absorbed}
    \lean{TNLean.PEPS.twoBlockProportional_of_edgeAbsorbed}
    \leanok
    Let $A,B$ be tensors on a finite graph with matched bond dimensions, and
    put $C=X\cdot B$ for an invertible edge gauge family $X$. If $A$ and $C$
    are injective on $Q$ and $Q^c$, $\partial Q\ne\varnothing$, and
    $C_A(e,M;\sigma)=C_C(e,M;\sigma)$ for all $e,M,\sigma$, then $A_Q=c_Q C_Q$
    for some $c_Q\ne0$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_twoInjectiveTensorInsertionComparison}
    Equality of $C_A(e,M;\sigma)=C_C(e,M;\sigma)$ on $e\in\partial Q$ and
    Lemma~\ref{thm:peps_twoInjectiveTensorInsertionComparison} yield
    $A_Q=c_Q C_Q$ with $c_Q\ne0$.
\end{proof}

\begin{lemma}[Translation preserves the block proportionality scalar]%
    \label{thm:peps_block_proportional_translate}
    \lean{TNLean.PEPS.twoBlockScalarProportional_translate}
    \leanok
    Let $w,h>1$, let $A,B$ be translation invariant on the torus with matched
    bond dimensions, and put $C=X\cdot B$ for a translation-covariant invertible
    edge gauge family $X$. If $A_Q=c C_Q$, then $A_{tQ}=c C_{tQ}$ for every
    translation $t$.
\end{lemma}
\begin{proof}\leanok
    Translation invariance of $A,B$ and covariance of $X$ identify the blocked
    weights on $Q$ and $tQ$ under the translation bijections of boundary and
    physical configurations, preserving $c$.
\end{proof}

\begin{lemma}[A common scalar from a translated one-site comparison]%
    \label{thm:peps_translated_scalar_comparison}
    \lean{TNLean.PEPS.component_eq_gaugeVertex_of_translatedProportional}
    \leanok
    Under the hypotheses of Lemma~\ref{thm:peps_block_proportional_translate},
    assume positive bond dimensions of $A$, injectivity of $B$ on $Q$, and
    $v_0\notin Q$. If $A_Q=c_Q C_Q$ and
    $A_{Q\cup\{v_0\}}=c_S C_{Q\cup\{v_0\}}$ with $c_Q\ne0$, then
    $A_v=(c_S/c_Q)C_v$ at every vertex $v$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_block_proportional_translate,
        thm:peps_region_injective_identification,
        thm:peps_regionBlockedTensorInjective_applyGauge,
        cor:peps_component_eq_gaugeVertex_of_twoBlockProportional}
    For the translation $t$ with $tv_0=v$,
    Lemma~\ref{thm:peps_block_proportional_translate} gives
    $A_{tQ}=c_Q C_{tQ}$ and $A_{tQ\cup\{v\}}=c_S C_{tQ\cup\{v\}}$, while
    Lemmas~\ref{thm:peps_region_injective_identification} and
    \ref{thm:peps_regionBlockedTensorInjective_applyGauge} give injectivity of
    $C$ on $tQ$. Lemma~\ref{cor:peps_component_eq_gaugeVertex_of_twoBlockProportional}
    then yields $A_v=(c_S/c_Q)C_v$.
\end{proof}

\begin{lemma}[The volume power of a common site scalar]%
    \label{thm:peps_common_scalar_volume}
    \lean{TNLean.PEPS.lambda_pow_card_torus_eq_one}
    \leanok
    Let $w,h>1$, and suppose $A,B$ generate the same torus state, have matched
    bond dimensions, have positive bond dimensions for $A$, and $A$ is
    injective on $Q$ and $Q^c$ for some $Q$. If an invertible edge gauge family
    $X$ satisfies $A_v=\lambda(X\cdot B)_v$ for all $v$, then $\lambda^{wh}=1$.
\end{lemma}
\begin{proof}\leanok
    Injectivity of $A$ on $Q$ and $Q^c$ gives a nonzero state coefficient, and
    since gauge action preserves the closed state while $A_v=\lambda(X\cdot B)_v$
    scales it by $\lambda^{wh}$, equality of the states gives $\lambda^{wh}=1$.
\end{proof}

\begin{lemma}[Scalar normalization from injective windows and absorbed gauges]%
    \label{thm:peps_window_absorbed_scalar}
    \lean{TNLean.PEPS.exists_scalar_of_normalArcWindows_absorbedGauge}
    \leanok
    Let $A,B$ be translation-invariant tensors on a $w\times h$ torus with
    positive matched bond dimensions, generating the same state, such that
    every cyclic $L\times K$ window is injective for $A$ and $B$, where
    $L,K>0$, $2L+1\le w$, and $2K+1\le h$. If an invertible
    translation-covariant edge gauge family $X$ satisfies
    $C_A(e,M;\sigma)=C_{X\cdot B}(e,M;\sigma)$ for all $e,M,\sigma$, then there
    exists $\lambda\in\C$ with $\lambda^{wh}=1$ and $A_v=\lambda(X\cdot B)_v$
    for all $v$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_regionInjectivityUnionClosure_of_overlap,
        thm:peps_window_corner_injectivity, thm:peps_window_corner_geometry,
        thm:peps_window_corner_boundary, thm:peps_region_injective_identification,
        thm:peps_regionBlockedTensorInjective_applyGauge,
        thm:peps_block_proportional_absorbed, thm:peps_translated_scalar_comparison,
        thm:peps_common_scalar_volume}
    Lemmas~\ref{thm:peps_regionInjectivityUnionClosure_of_overlap},
    \ref{thm:peps_window_corner_injectivity},
    \ref{thm:peps_region_injective_identification}, and
    \ref{thm:peps_regionBlockedTensorInjective_applyGauge} give injectivity of
    $A$ and $X\cdot B$ on $R,S,R^c,S^c$ of
    (\ref{eq:peps_window_corner_regions}). By
    Lemmas~\ref{thm:peps_window_corner_boundary} and
    \ref{thm:peps_block_proportional_absorbed}, $A_R=c_R(X\cdot B)_R$ and
    $A_S=c_S(X\cdot B)_S$ with $c_R\ne0$. Since $S=R\cup\{v_0\}$ by
    Lemma~\ref{thm:peps_window_corner_geometry},
    Lemma~\ref{thm:peps_translated_scalar_comparison} gives
    $A_v=\lambda(X\cdot B)_v$ with $\lambda=c_S/c_R$, and
    Lemma~\ref{thm:peps_common_scalar_volume} gives $\lambda^{wh}=1$.
\end{proof}
