\begin{definition}[Region two-block tensor]%
    \label{def:peps_regionTwoBlock}
    \lean{TNLean.PEPS.regionTwoBlock}
    \leanok
    For a finite region $R\subseteq V$, the region two-block tensor on the
    shared crossing bonds $\partial R$ with one-point external boundary $\{*\}$
    (\cite[Section~3]{Molnar2018NormalPEPS}) is
    $B_{A,R}(*,\eta,\sigma)=T_{A,R}^{\eta}(\sigma)$.
\end{definition}

\begin{definition}[Complement-boundary identification]%
    \label{def:peps_regionComplementBoundaryConfig}
    \lean{TNLean.PEPS.regionBoundaryEdgeToCompl,
        TNLean.PEPS.regionBoundaryEdgeComplEquiv,
        TNLean.PEPS.regionComplementBoundaryConfig}
    \leanok
    The equality $\partial R=\partial(V\setminus R)$ identifies each boundary
    configuration $\eta$ of $R$ with a boundary configuration $\eta^c$ of
    $V\setminus R$.
\end{definition}

\begin{definition}[Complement-region two-block tensor]%
    \label{def:peps_regionComplementTwoBlock}
    \lean{TNLean.PEPS.regionComplementTwoBlock}
    \leanok
    \uses{def:peps_regionComplementBoundaryConfig}
    For a finite region $R\subseteq V$, the complement-region two-block tensor
    over $\partial R$ (\cite[Section~3]{Molnar2018NormalPEPS}) is
    $B_{A,R^c}(*,\eta,\tau)=T_{A,V\setminus R}^{\eta^c}(\tau)$.
\end{definition}

\begin{lemma}[Injectivity of the complement-region two-block tensor]%
    \label{thm:peps_isTwoBlockInjective_regionComplementTwoBlock}
    \lean{TNLean.PEPS.isTwoBlockInjective_regionComplementTwoBlock}
    \leanok
    \uses{def:peps_regionComplementTwoBlock, def:peps_regionInjectivityData}
    For any finite $R\subseteq V$,
    $\inj(T_{A,V\setminus R})\Longrightarrow\inj(B_{A,R^c})$.
\end{lemma}
\begin{proof}\leanok
    By Definitions~\ref{def:peps_regionComplementBoundaryConfig}
    and~\ref{def:peps_regionComplementTwoBlock}, $(*,\eta)\mapsto\eta^c$ is a
    bijection $\partial R\xrightarrow{\sim}\partial(V\setminus R)$, which
    preserves linear independence of the blocked tensor family.
\end{proof}

\begin{definition}[Region insertion coefficient]%
    \label{def:peps_regionInsertedCoeff}
    \lean{TNLean.PEPS.regionInsertedCoeff}
    \leanok
    \uses{def:peps_regionTwoBlock, def:peps_regionComplementTwoBlock}
    For $f\in\partial R$ and $M\in\MN{D_f}$, the region insertion coefficient is
    \begin{align}
        C_A(R,f,M;\sigma,\tau)
         & =
        \sum_{\substack{\mu,\nu\in\partial R \\
                \mu|_{\partial R\setminus\{f\}}
                =\nu|_{\partial R\setminus\{f\}}}}
        M_{\mu_f,\nu_f}
        T_{A,R}^{\mu}(\sigma)
        T_{A,V\setminus R}^{\nu^c}(\tau).
        \label{eq:peps_regionInsertedCoeff_def}
    \end{align}
\end{definition}

\begin{lemma}[Region insertion coefficients determine the inserted matrix]%
    \label{thm:peps_regionInsertedCoeff_injective}
    \lean{TNLean.PEPS.regionInsertedCoeff_injective}
    \leanok
    \uses{def:peps_regionInsertedCoeff}
    Assume $\inj(T_{A,R})$, $\inj(T_{A,V\setminus R})$, and
    $\forall e\in E,\ D_A(e)>0$. If $f\in\partial R$ and $M,M'\in\MN{D_A(f)}$
    satisfy $\forall\sigma,\tau,\ C_A(R,f,M;\sigma,\tau)=C_A(R,f,M';\sigma,\tau)$,
    then $M=M'$.
\end{lemma}
\begin{proof}\leanok
    Applying the left inverses of $T_{A,R}$ and $T_{A,V\setminus R}$ to
    $C_A(R,f,M-M';\sigma,\tau)=0$ in (\ref{eq:peps_regionInsertedCoeff_def})
    (using $D_A(e)>0$ to fix off-$f$ boundary labels) gives $(M-M')_{i,j}=0$
    for all $i,j$, so $M=M'$.
\end{proof}

\begin{definition}[Bond-inserted region insert]%
    \label{def:peps_bond_inserted_region_insert}
    \lean{TNLean.PEPS.bondInsertedRegionInsert}
    \leanok
    For $f\in\partial R$ and $M\in\MN{D_f}$, the bond-inserted region insert
    $I_{A,R,f,M}$ on $R$ (\cite[Section~3]{Molnar2018NormalPEPS}) is
    \begin{align}
        (I_{A,R,f,M})_{\nu,\sigma}
         & =
        \sum_{\substack{\mu\in\partial R \\
                \mu|_{\partial R\setminus\{f\}}
                =\nu|_{\partial R\setminus\{f\}}}}
        M_{\mu_f,\nu_f}\,T_{A,R}^{\mu}(\sigma).
        \label{eq:peps_bond_inserted_region_insert_def}
    \end{align}
\end{definition}

\begin{lemma}[Bond insertion as a deformed region state]
    \label{thm:peps_deformedRegionState_bondInsertedRegionInsert}
    \lean{TNLean.PEPS.deformedRegionState_bondInsertedRegionInsert}
    \leanok
    \uses{def:peps_bond_inserted_region_insert, def:peps_regionInsertedCoeff}
    For all physical configurations $\sigma$ on $R$ and $\tau$ on $V\setminus R$,
    $\Psi_{A,R,I_{A,R,f,M}}(\sigma,\tau)=C_A(R,f,M;\sigma,\tau)$.
\end{lemma}
\begin{proof}\leanok
    Substituting (\ref{eq:peps_bond_inserted_region_insert_def}) into
    $\Psi_{A,R,I_{A,R,f,M}}(\sigma,\tau)=\sum_{\nu\in\partial R}(I_{A,R,f,M})_{\nu,\sigma}T_{A,V\setminus R}^{\nu^c}(\tau)$
    yields (\ref{eq:peps_regionInsertedCoeff_def}).
\end{proof}

For $R\subseteq V$, $\sigma$ on $R$, and $\tau$ on $V\setminus R$, define
$\omega_{R,\sigma,\tau}$ on $V$ by $\omega_{R,\sigma,\tau}|_R=\sigma$ and
$\omega_{R,\sigma,\tau}|_{V\setminus R}=\tau$.

\begin{lemma}[Region insertion as an edge insertion]
    \label{thm:peps_regionInsertedCoeff_eq_smul_edgeInsertedCoeff}
    \lean{TNLean.PEPS.regionInsertedCoeff_eq_smul_edgeInsertedCoeff}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_edgeInsertedCoeff}
    For $f\in\partial R$, $N\in\MN{D_B(f)}$, and
    $p_B(R)=\prod_{g\notin\partial R}D_B(g)$,
    \begin{align}
        C_B(R,f,N;\sigma,\tau)
         & =
        p_B(R)
        E_B(f,\omega_{R,\sigma,\tau},\mathcal O_{R,f}(N)),
        \label{eq:peps_region_as_edge_insertion}
    \end{align}
    where $\mathcal O_{R,f}(N)=N$ if the left endpoint of $f$ lies in $R$ and
    $\mathcal O_{R,f}(N)=N^T$ otherwise.
\end{lemma}
\begin{proof}\leanok
    Expanding $T_{B,R}^{\mu}$ and $T_{B,V\setminus R}^{\nu^c}$ in
    (\ref{eq:peps_regionInsertedCoeff_def}) and grouping by the open-edge
    configuration $\xi\in\mathcal C_{R,f}(\sigma,\tau)$ gives a fibre of size
    $\prod_{g\notin\partial R}D_B(g)=p_B(R)$ over each $\xi$, so
    \begin{align}
        C_B(R,f,N;\sigma,\tau)
         & =
        p_B(R)
        \sum_{\xi\in\mathcal C_{R,f}(\sigma,\tau)}
        N_{\xi_{R,f}^{(1)},\xi_{R,f}^{(2)}}
        \prod_{v\in V}
        B_v(\xi_v,\omega_{R,\sigma,\tau}(v))
        =p_B(R)E_B(f,\omega_{R,\sigma,\tau},\mathcal O_{R,f}(N))
        \notag
    \end{align}
    by Definition~\ref{def:peps_edgeInsertedCoeff}.
\end{proof}

\begin{lemma}[Bond insertion through a corner extension]
    \label{thm:peps_regionInsertedCoeff_eq_extendInsert_bondInserted}
    \lean{TNLean.PEPS.regionInsertedCoeff_eq_extendInsert_bondInserted}
    \leanok
    \uses{thm:peps_deformedRegionState_bondInsertedRegionInsert, thm:peps_corner_extension_state_preserves}
    If $R\subseteq S$ and $\forall e\in E,\ D_A(e)>0$, then for every global
    physical configuration $\sigma$,
    \begin{align}
        C_A(R,f,M;\sigma|_R,\sigma|_{V\setminus R})
         & =
        \Psi_{A,S,
            \Ext_{R\subseteq S}(I_{A,R,f,M})}(\sigma).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Combine Lemma~\ref{thm:peps_deformedRegionState_bondInsertedRegionInsert}
    with Lemma~\ref{thm:peps_corner_extension_state_preserves}.
\end{proof}

\begin{lemma}[Region insertion as a two-block insertion]%
    \label{thm:peps_twoBlockInsertedCoeff_eq_regionInsertedCoeff}
    \lean{TNLean.PEPS.twoBlockInsertedCoeff_eq_regionInsertedCoeff}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_regionTwoBlock, def:peps_regionComplementTwoBlock}
    For $(B_{A,R},B_{A,R^c})$ from Definitions~\ref{def:peps_regionTwoBlock}
    and~\ref{def:peps_regionComplementTwoBlock},
    $C_{B_{A,R},B_{A,R^c}}(f,M;*,*,\sigma,\tau)=C_A(R,f,M;\sigma,\tau)$.
\end{lemma}
\begin{proof}\leanok
    Unfold Definitions~\ref{def:peps_regionTwoBlock},
    \ref{def:peps_regionComplementTwoBlock},
    and~\ref{def:peps_regionInsertedCoeff}.
\end{proof}

\begin{lemma}[Region insertion under bond-dimension identification]%
    \label{thm:peps_regionInsertedCoeff_reindexTensor}
    \lean{TNLean.PEPS.regionInsertedCoeff_reindexTensor}
    \leanok
    \uses{def:peps_regionInsertedCoeff}
    If $B^h$ is $B$ reindexed along a bond-dimension equality $h$ with induced
    matrix isomorphism $\phi_f$ on $f\in\partial R$, then
    $C_{B^h}(R,f,N;\sigma,\tau)=C_B(R,f,\phi_f(N);\sigma,\tau)$.
\end{lemma}
\begin{proof}\leanok
    Reindexing $\mu,\nu\in\partial R$ along $h$ in
    (\ref{eq:peps_regionInsertedCoeff_def}) preserves the off-$f$ agreement
    condition and blocked weights while replacing $N_{\mu_f,\nu_f}$ by
    $\phi_f(N)$.
\end{proof}

\begin{lemma}[Same two-block insertions from region insertions]%
    \label{thm:peps_sameTwoBlockInsertions_of_regionInsertedCoeff_eq}
    \lean{TNLean.PEPS.sameTwoBlockInsertions_of_regionInsertedCoeff_eq}
    \leanok
    \uses{thm:peps_twoBlockInsertedCoeff_eq_regionInsertedCoeff, thm:peps_regionInsertedCoeff_reindexTensor}
    Let $B^h$ be $B$ reindexed along $h:D_A=D_B$ with induced bond isomorphism
    $\phi_f$. If $C_A(R,f,N;\sigma,\tau)=C_B(R,f,\phi_f(N);\sigma,\tau)$ for all
    $f\in\partial R,N,\sigma,\tau$, then
    $C_{B_{A,R},B_{A,R^c}}(f,N;*,*,\sigma,\tau)
        =C_{B_{B^h,R},B_{B^h,R^c}}(f,N;*,*,\sigma,\tau)$.
\end{lemma}
\begin{proof}\leanok
    By Lemmas~\ref{thm:peps_twoBlockInsertedCoeff_eq_regionInsertedCoeff}
    and~\ref{thm:peps_regionInsertedCoeff_reindexTensor},
    $C_{B_{A,R},B_{A,R^c}}(f,N;*,*,\sigma,\tau)
        =C_A(R,f,N;\sigma,\tau)
        =C_B(R,f,\phi_f(N);\sigma,\tau)
        =C_{B_{B^h,R},B_{B^h,R^c}}(f,N;*,*,\sigma,\tau)$.
\end{proof}

\begin{lemma}[Swapping the two blocks in a one-bond insertion]
    \label{thm:peps_twoBlockInsertedCoeff_swap}
    \lean{TNLean.PEPS.twoBlockInsertedCoeff_swap}
    \leanok
    For any two-block pair $(B_1,B_2)$ and shared bond $f$,
    \begin{align}
        C_{B_1,B_2}(f,X;\eta_1,\eta_2,\sigma_1,\sigma_2)
         & =
        C_{B_2,B_1}(f,X^T;\eta_2,\eta_1,\sigma_2,\sigma_1).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Swapping the summation variables $(\mu,\nu)\leftrightarrow(\nu,\mu)$
    preserves $\mu|_{\setminus\{f\}}=\nu|_{\setminus\{f\}}$ and rewrites
    $X_{\mu_f,\nu_f}=(X^T)_{\nu_f,\mu_f}$.
\end{proof}

\begin{lemma}[Complement-side reading of a region insertion]
    \label{thm:peps_regionInsertedCoeff_complement_reading}
    \lean{TNLean.PEPS.regionInsertedCoeff_eq_complementTwoBlock}
    \leanok
    \uses{thm:peps_twoBlockInsertedCoeff_swap, thm:peps_twoBlockInsertedCoeff_eq_regionInsertedCoeff}
    For $f\in\partial R$, with $C_{A,R^c;R}$ the insertion coefficient of
    $(B_{A,R^c},B_{A,R})$,
    $C_A(R,f,X;\sigma,\tau)=C_{A,R^c;R}(f,X^T;\tau,\sigma)$.
\end{lemma}
\begin{proof}\leanok
    Combine Lemma~\ref{thm:peps_twoBlockInsertedCoeff_eq_regionInsertedCoeff}
    with Lemma~\ref{thm:peps_twoBlockInsertedCoeff_swap}.
\end{proof}

\begin{lemma}[Incident form gives the transferred endpoint form]
    \label{thm:peps_hform_of_region_incident_form}
    \lean{TNLean.PEPS.hform_of_isIncidentMatrixForm}
    \leanok
    \uses{thm:peps_physical_to_virtual_insertion}
    Assume $\inj(A_v)$, $\inj(B_v)$ at the in-region endpoint $v$ of $f$, and
    $\forall e\in E,\ D_B(e)>0$. If the virtual pullback $W$ at $v$ of the
    physical operator for $X^T$ on $f$ has incident-matrix form
    $W=\mathcal I_f(P)$, then the transfer matrix extracted from $W$ reinserts
    to $W$.
\end{lemma}
\begin{proof}\leanok
    Extracting the incident matrix from $\mathcal I_f(P)$ yields $P$, whose
    reinsertion is $\mathcal I_f(P)=W$.
\end{proof}

\begin{lemma}[Virtual pullback realizes the endpoint operator]
    \label{thm:peps_region_insertion_virtual_pullback_realizes}
    \lean{TNLean.PEPS.localTensorMap_localVirtualOpOfPhysicalOpAt_regionInsertionOp}
    \leanok
    \uses{def:peps_same_state, def:IsVertexInjective, def:peps_localTensorMap, thm:peps_physical_to_virtual_insertion}
    If $A,B$ are vertex-injective with positive bond dimensions and
    $\Coeff_A=\Coeff_B$, and $W$ is the virtual pullback for $B$ at the
    in-region endpoint $v$ of $f$ of the physical operator $O_{A,R,f}(X^T)$,
    then $\mathcal T_{B,v}(Wc)=O_{A,R,f}(X^T)\mathcal T_{B,v}(c)$ for all $c$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_physical_to_virtual_insertion}, $O_{A,R,f}(X^T)$
    preserves $\operatorname{im}\mathcal T_{B,v}$, so the left-inverse pullback
    $W$ satisfies $\mathcal T_{B,v}W=O_{A,R,f}(X^T)\mathcal T_{B,v}$.
\end{proof}

\begin{lemma}[Region resonate identity and blocked read-offs]
    \label{thm:peps_region_resonate_identity_readoffs}
    \lean{TNLean.PEPS.regionInsertionOp_regionStateVec_pin_compl,
        TNLean.PEPS.region_resonate_identity,
        TNLean.PEPS.regionComplementRow,
        TNLean.PEPS.regionInsertedCoeff_eq_complement_blockedMap,
        TNLean.PEPS.regionRegionRow,
        TNLean.PEPS.regionInsertedCoeff_eq_region_blockedMap,
        TNLean.PEPS.regionBlockedLeftInverse_complement_regionInsertedCoeff,
        TNLean.PEPS.regionBlockedLeftInverse_region_regionInsertedCoeff}
    \leanok
    \uses{def:peps_region_out_endpoint_block_inverse, thm:peps_regionInsertedCoeff_complement_cast}
    Assume $\inj(A_v)$ at both endpoints of $f\in\partial R$ and
    $\Coeff_A=\Coeff_B$. With interior bond products $P_R,P_{R^c}$, endpoint
    physical operators $O^{\mathrm{in}}_f(X),O^{\mathrm{out}}_f(X)$, and row
    functions $\rho^{R^c}_{X,\sigma},\rho^{R}_{X,\tau}$,
    \begin{align}
        P_R\,O^{\mathrm{in}}_f(X) \Psi_B(R;\sigma,\tau)
         & =
        P_{R^c}\,O^{\mathrm{out}}_f(X)
        \Psi_B(R^c;\tau,\sigma^{cc}),
        \notag \\
        C_A(R,f,X;\sigma,-)
         & =
        T_{A,R^c}\!\left(\rho^{R^c}_{X,\sigma}\right),
         &
        C_A(R,f,X;-,\tau)
         & =
        T_{A,R}\!\left(\rho^{R}_{X,\tau}\right),
        \notag \\
        L_{A,R^c}\!\left(C_A(R,f,X;\sigma,-)\right)
         & =
        \rho^{R^c}_{X,\sigma},
         &
        L_{A,R}\!\left(C_A(R,f,X;-,\tau)\right)
         & =
        \rho^{R}_{X,\tau}.
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Both $P_R\,O^{\mathrm{in}}_f(X)\Psi_B(R;\sigma,\tau)$ and
    $P_{R^c}\,O^{\mathrm{out}}_f(X)\Psi_B(R^c;\tau,\sigma^{cc})$ evaluate to
    $C_A(R,f,X;\sigma,\tau)$ by Lemma~\ref{thm:peps_regionInsertedCoeff_complement_cast}.
    Factoring (\ref{eq:peps_regionInsertedCoeff_def}) through $T_{A,R^c}$ and
    $T_{A,R}$ and applying $L_{A,R^c}$ and $L_{A,R}$ from
    Definition~\ref{def:peps_region_out_endpoint_block_inverse} gives the
    remaining identities.
\end{proof}
