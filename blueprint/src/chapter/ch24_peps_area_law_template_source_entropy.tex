\section{Original-state scanner entropy from safe boxes}
\label{sec:template-source-entropy}

Following \emph{A two-dimensional area law from a global spectral gap}
(\texttt{08-scanner.tex}, lines 353--357, 574--590, 641--667, and 828--844,
at revision \texttt{adc7f1241b42e322a6451854ab7e4b4c146bf78a}), the entropy
coverings of Lemma 9.4 now give the scanner's original-state
target and prefix inputs for a separated template.  The safe-box
estimate is needed only at the single exponent in use, without yet deriving
the improved exponent of Proposition 9.5.

\begin{theorem}[One uniform coefficient for template entropy inputs]
    \label{thm:template-source-entropy}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_template_source_entropy_bounds,TNLean.PEPS.AreaLaw.Scan.templateSourceEntropyConstant,TNLean.PEPS.AreaLaw.Scan.template_source_entropy_bounds}
    \leanok
    Fix $q\geq1$, an exponent $e>0$ and a safe-box coefficient $C_{\rm box}\geq0$.
    There is $C_e\geq0$, chosen before every finite domain, unit vector,
    cut, safety parameter, template and collar scale, with the following
    property.  Let $T$ be a template with $C_{\rm tpl}\geq24$,
    separated from the cut at safety parameter $D_0\geq1$, and suppose
    \[
        S_\Omega(A\cap Q)\leq C_{\rm box}|Q|_{\rm side}^{1+e}
    \]
    for every safe integer rectangle $Q$.  Then
    \[
        S_\Omega(A\cap T)\leq C_e n^{1+e},\qquad
        S_\Omega(Q_j)\leq C_e(nL^e+n)
        \quad(0\leq j\leq L\leq s_0,\ L\geq1).
    \]
    The same shell bound holds after replacing a complete prefix by all
    preceding depth rows and an arbitrary subset of the final row.
\end{theorem}

\begin{proof}
    \leanok
    The template scale and nonempty piece family imply $s_0\leq n$.
    The safe dyadic coverings give core and shell coefficients
    \[
        C_{\rm core}=C_{\rm box}(18+4/(2^e-1)),\qquad
        C_{\rm shell}=C_{\rm box}(2+14/(2^e-1)).
    \]
    With $C_e=\max(0,C_{\rm core},C_{\rm shell},\log q)$ fixed before any
    instance and the vector held fixed throughout, the core bound
    $C_{\rm core}ns_0^e$ is at most $C_e n^{1+e}$, the shell estimate
    applies to the exact scanner region because the ambient-depth positive
    prefix equals $A\cap(T_j\setminus T)$ (including $j=0$), and a partial
    last row contributes at most $n\log q\leq C_en$.
\end{proof}

\begin{theorem}[Physical entropy inputs conditional on the improved safe-box bound]
    \label{thm:template-final-source-entropy}
    \lean{TNLean.PEPS.AreaLaw.Scan.exists_template_final_source_entropy_bounds_of_boxEntropy,TNLean.PEPS.AreaLaw.Scan.template_final_source_entropy_bounds_of_boxEntropy}
    \leanok
    Fix $q\geq1$, $D_0\geq1$ and $C_{\rm box}\geq0$.  Suppose the
    uniform safe-box function at fixed $q,R,J,\Delta,D_0$ satisfies
    $F_{\rm box}(r)\leq C_{\rm box}r^{1+e_*}$ for all $r\geq1$,
    where $e_*=2\cdot10^{-6}$.
    One coefficient $C_e\geq0$, chosen before the finite physical system,
    gives the two displayed entropy inputs for every original gapped
    ground vector and every separated template target of a scanner whose
    depth is the ambient target distance.  In particular one may
    use $L=\lfloor n^{1-10^{-5}}\rfloor$ whenever $1\leq L\leq s_0$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:template-source-entropy}
    Without introducing a new ground vector, truncated-vector entropy, or
    replica or interpolation parameter, each safe-rectangle entropy is bounded
    by the supremum defining $F_{\rm box}$ at its maximal side length, so the
    preceding theorem at $e=e_*$ and the exact depth identification yield both
    displayed entropy bounds.
\end{proof}

The improved safe-box bound remains an input.  Its physical one-step
bootstrap, the numerical scanner and geometric prerequisites not stated
here, and the full Proposition 9.6 remain separate obligations.
