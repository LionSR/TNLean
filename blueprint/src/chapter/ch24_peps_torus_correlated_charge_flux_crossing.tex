\begin{definition}[The correlated charge state]
    \label{def:peps_regular_correlated_charge_state}
    \lean{TNLean.PEPS.regularCorrelatedChargeState}
    \leanok
    \uses{def:peps_regular_edge_charge_contraction}
    For ordered edges $e_0,e_1$ of a finite graph and bond assignment $u$, define the correlated state $\Psi_z(u;\chi,p)$ (\cite{Schuch2010PEPS}, lines 2505--2535) by summing over $q\in G$ the closed tensor state with indicator $\mathbf1_{\eta_{e_1}=q}$ on $e_1$ and weight $\chi(pq^{-1}z\eta_{e_0})$ on $e_0$.
\end{definition}

\begin{lemma}[Correlated weights in the physical cut]
    \label{thm:peps_regular_correlated_charge_cut}
    \lean{TNLean.PEPS.openRegionWeight_regularEdgeCharacterSite_eq_weighted,
        TNLean.PEPS.regularCorrelatedChargeState_apply,
        TNLean.PEPS.regionLocalTerm_mulVec_regularCorrelatedChargeState}
    \leanok
    \uses{def:peps_regular_correlated_charge_state,
        def:peps_regular_weighted_open_contraction,
        thm:peps_regular_coherent_global_transport}
    The state $\Psi_z(u;\chi,p)$ of Definition~\ref{def:peps_regular_correlated_charge_state} has components
    \begin{align}
        \Psi_z(u;\chi,p)(\sigma)
        =\sum_{\eta:E\to G}\chi(p\eta_{e_1}^{-1}z\eta_{e_0})
        \prod_v a_v\bigl(\eta^{u}|_v,\sigma_v\bigr). \label{eq:correlated_charge_state_coeff}
    \end{align}
    If $t(e_0)\in R$, $t(e_1)\notin R$, and a regional operator $W_R$ transforms every weighted open column via $\chi(pq^{-1}\eta_{e_0})\mapsto\chi(pq^{-1}k^{-1}\eta_{e_0})$ uniformly in $(\chi,p,q,\theta)$, then (\cite{Schuch2010PEPS}, lines 2505--2535 and 2560--2581)
    \begin{align}
        (W_R\otimes I)\Psi_1(u;\chi,p)=\Psi_{k^{-1}}(u;\chi,p). \label{eq:correlated_charge_cut_global}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Summing $\mathbf1_{\eta_{e_1}=q}$ over $q\in G$ substitutes $q=\eta_{e_1}$ to give (\ref{eq:correlated_charge_state_coeff}).
    Since $t(e_1)\notin R$, for each $q$ the indicator $\mathbf1_{\eta_{e_1}=q}$ lies in $R^c$, so applying Lemma~\ref{thm:peps_regular_coherent_global_transport} at each $q$ and summing over $q$ yields (\ref{eq:correlated_charge_cut_global}).
\end{proof}

\begin{definition}[The native remote charge edge]
    \label{def:peps_torus_charge_flux_partner_edge}
    \lean{TNLean.PEPS.torusChargeFluxPartnerEdge}
    \leanok
    \uses{def:peps_torus_charge_flux_crossing_strip}
    At $v=(x,y)$, define the remote reference edge $e_1$ as the native horizontal bond joining $(x+1,y)$ to $(x+2,y)$ below the eight-site crossing strip $R_v$ of Definition~\ref{def:peps_torus_charge_flux_crossing_strip} (\cite{Schuch2010PEPS}, lines 2505--2581, retaining both seam orientations).
\end{definition}

\begin{lemma}[A physical crossing with the remote partner]
    \label{thm:peps_torus_correlated_charge_flux_crossing_global}
    \lean{TNLean.PEPS.torusChargeFluxPartnerEdge_tail_not_mem,
        TNLean.PEPS.IsGIsometric.exists_unitary_torusCorrelatedChargeFluxCrossing_global}
    \leanok
    \uses{def:peps_torus_charge_flux_partner_edge,
        thm:peps_regular_correlated_charge_cut,
        thm:peps_torus_charge_flux_crossing}
    For finite $G$, torus periods $w\ge 5$ and $H\ge 4$, and a regular $G$-isometric four-leg tensor, on every translated strip $R=R_v$ with charge bond $e_0$, remote bond $e_1$ of Definition~\ref{def:peps_torus_charge_flux_partner_edge}, and initial flux string $u^{(0)}_k$, there exists a regional unitary $W_R$ (chosen before $k,\chi,p$, with global unitary extension $W_R\otimes I$ and no extra tree-gauge, factorization, Gram, or state-action hypotheses) such that
    \begin{align}
        (W_R\otimes I)\Psi_1(u^{(0)}_k;\chi,p)
        =\Psi_{k^{-1}}(u^{(0)}_k;\chi,p), \label{eq:torus_correlated_crossing_global}
    \end{align}
    transforming the joint weight by $\chi(p\eta_1^{-1}\eta_0)\mapsto\chi(p\eta_1^{-1}k^{-1}\eta_0)$ (as an auxiliary local crossing step; the complete geometric braid and reunion experiment in \cite{Schuch2010PEPS}, lines 2560--2615, remain separate).
\end{lemma}
\begin{proof}\leanok
    Since $H\ge 4$, the vertical coordinate $y$ of $t(e_1)$ is distinct from the strip heights $y+1,y+2$, so $t(e_1)\notin R_v$ across either seam.
    Combining the uniform column unitary $W_R$ of Lemma~\ref{thm:peps_torus_charge_flux_crossing} with (\ref{eq:correlated_charge_cut_global}) of Lemma~\ref{thm:peps_regular_correlated_charge_cut} gives (\ref{eq:torus_correlated_crossing_global}).
\end{proof}
