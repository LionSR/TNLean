\section{Common scalar recovery of two marked coefficients}

These are the interpolation and ordered-word identities used in
\cite[Lemma~4.2]{Arad2013AreaLaw} and
\cite[Proposition~5.7]{Arad2016RigorousRG}.
Only a spanning family of evaluations is required; the coefficient operators
need not be linearly independent. The distinction between spanning and a
basis in the source interpolation argument is recorded in
\cite{gap:aklv13_interpolation_and_word_count}.

\begin{definition}[Ordinary labels]
    \label{def:two_marker_ordinary_labels}
    \lean{MvPolynomial.twoMarkerOrdinaryLabels}
    \leanok
    For a finite alphabet $I$ and distinguished labels $u,v\in I$, let
    \begin{align}
        \mathcal O_{u,v}=I\setminus\{u,v\}.
        \label{eq:two_marker_ordinary_labels}
    \end{align}
    The definition does not require the two labels to be distinct.
\end{definition}

\begin{definition}[The full-frequency ordered-word coefficient]
    \label{def:full_frequency_word_coefficient}
    \lean{WordMultiplicity.wordCoefficient}
    \leanok
    Let $I$ be finite, let $R$ be a semiring, and let $H_a\in R$ for $a\in I$.
    For $r\ge0$ and a count vector $\nu:I\to\N$, put
    \begin{align}
        \mathcal W_{H,r}(\nu)=
        \sum_{\substack{w:\{0,\ldots,r-1\}\to I \\
                |w^{-1}(a)|=\nu(a)\ (a\in I)}}
        H_{w(0)}H_{w(1)}\cdots H_{w(r-1)}.
        \label{eq:full_frequency_word_coefficient}
    \end{align}
    The factors retain their displayed order. An empty product is $1$, and a
    sum with no words is zero.
\end{definition}

\begin{lemma}[One sample family for every degree and frequency slice]
    \label{thm:two_window_common_scalar_recovery}
    \lean{Polynomial.exists_twoWindow_wordCoefficient_recovery}
    \leanok
    \uses{def:two_marker_ordinary_labels,def:full_frequency_word_coefficient}
    Let $\ell,K$ be nonnegative integers, let
    \begin{align}
        I=(\{0,\ldots,2\ell-1\}\sqcup\{0,\ldots,2\ell-1\})
        \sqcup\{0,1,2\},
        \label{eq:two_window_scalar_alphabet}
    \end{align}
    and choose $0\le p,s<2\ell$.
    Let $u$ be the label $p$ in the first window and $v$ the label $s$ in the
    second window. Let $R$ be a ring and a complex algebra, without any
    commutativity assumption on $R$, and put $N=(K+1)^{4\ell+1}$.

    There are complex sample points $z_t:I\to\C$, $0\le t<N$, with the
    following property. For every $i,j\ge0$ and every scalar mask
    $\mu:(I\to\N)\to\C$, there are complex coefficients $\gamma_t$ such
    that, for every family $H:I\to R$ and every $0\le r\le K$,
    \begin{align}
         & \sum_{\substack{\nu:I\to\N         \\ \sum_a\nu(a)=r\\
                \nu(u)=i,\ \nu(v)=j}}
        \mu(\nu)\,\mathcal W_{H,r}(\nu)\notag \\
         & \qquad=\sum_{t=0}^{N-1}\gamma_t
        [x^iy^j]\left(D_t+yH_v+xH_u\right)^r,
        \qquad D_t=\sum_{a\in\mathcal O_{u,v}}z_t(a)H_a.
        \label{eq:two_window_common_scalar_recovery}
    \end{align}
    Here $x,y$ are central formal variables and scalar products mean the
    complex algebra action. The selected terms $H_u,H_v$ remain unweighted.
    The points precede $i,j,\mu$, while the recovery coefficients precede $H$
    and every degree $r\le K$. Points for different selected pairs need not
    coincide. The case $K=0$ is included.
\end{lemma}
\begin{proof}\leanok
    \uses{def:two_marker_ordinary_labels,def:full_frequency_word_coefficient}
    The distinguished labels are distinct, so $\mathcal O=\mathcal O_{u,v}$
    has $4\ell+1$ members. Put $G=\{0,\ldots,K\}^{\mathcal O}$.
    For $0\le s,t\le K$, let $V_{s,t}=s^t$.
    The nodes $0,\ldots,K$ are distinct, so the Vandermonde matrix $V$ is
    invertible. For $\alpha,g\in G$, define
    \begin{align}
        E_{g,\alpha} & =\prod_{a\in\mathcal O}g(a)^{\alpha(a)}, \notag   \\
        R_{\alpha,g} & =\prod_{a\in\mathcal O}(V^{-1})_{\alpha(a),g(a)}.
        \label{eq:two_marker_grid_weights}
    \end{align}
    Multiplying the one-variable recovery identities over the ordinary labels gives
    \begin{align}
        \sum_{g\in G}R_{\alpha,g}E_{g,\beta}=\delta_{\alpha,\beta}.
        \label{eq:two_marker_grid_recovery}
    \end{align}
    Take $z_g:I\to\C$ to agree with $g$ on $\mathcal O$ and to vanish
    elsewhere. These points depend only on the alphabet, the distinguished
    labels and $K$.

    Fix $i,j,\mu$. For $\alpha\in G$, let $\nu_{i,j,\alpha}$ be the
    full count vector obtained by assigning $i$ at $u$, $j$ at $v$, and
    $\alpha(a)$ at each ordinary label $a$. Define
    \begin{align}
        \gamma_g=\sum_{\alpha\in G}\mu(\nu_{i,j,\alpha})R_{\alpha,g}.
        \label{eq:two_marker_mask_weights}
    \end{align}
    These coefficients precede the operator family and the degree.
    Fix a degree $r\le K$, and let $S_{r,i,j}$ be the prescribed frequency
    slice, the set of $\nu:I\to\N$ with $\sum_a\nu(a)=r$, $\nu(u)=i$ and
    $\nu(v)=j$. Every ordinary coordinate of $\nu\in S_{r,i,j}$ is at most
    $r\le K$. Restriction to the ordinary labels and restoration of the two
    distinguished counts therefore satisfy
    \begin{align}
        \nu|_{\mathcal O}\in G,\quad
        \nu_{i,j,\nu|_{\mathcal O}}=\nu
        \quad(\nu\in S_{r,i,j}),\qquad
        (\nu_{i,j,\alpha})|_{\mathcal O}=\alpha
        \quad(\alpha\in G).
        \label{eq:two_marker_slice_restriction}
    \end{align}
    By (\ref{eq:two_marker_slice_restriction}), restriction is a bijection from $S_{r,i,j}$ onto the set of
    $\alpha\in G$ with $\nu_{i,j,\alpha}\in S_{r,i,j}$, and for every
    function $F$ on count vectors the slice sum is the zero extension
    \begin{align}
        \sum_{\nu\in S_{r,i,j}}F(\nu)
        =\sum_{\alpha\in G}\mathbf 1_{\nu_{i,j,\alpha}\in S_{r,i,j}}
        F(\nu_{i,j,\alpha}).
        \label{eq:two_marker_slice_zero_extension}
    \end{align}
    Substituting
    (\ref{eq:two_marker_grid_recovery}) into
    (\ref{eq:two_marker_slice_zero_extension}) and exchanging the two finite sums
    gives recovery with the weights in (\ref{eq:two_marker_mask_weights}).

    Apply this recovery identity to $\mathcal W_{H,r}(\nu)$.
    The counted-word identity identifies each evaluated ordinary monomial sum
    with $[x^iy^j](D_g+yH_v+xH_u)^r$, preserving the order of the factors
    $H_a$. The box has exactly $(K+1)^{4\ell+1}=N$ elements.
    Reindexing it by $0\le t<N$ gives
    (\ref{eq:two_window_common_scalar_recovery}). The formulas include $K=0$.
\end{proof}
