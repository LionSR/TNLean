\section{Final sharp error absorption at the source parameters}
\label{sec:arealaw-final-sharp-scalar}

The final collar calculation uses the fixed values in
\cite{OpenAI2026AreaLaw}, \path{08-scanner.tex}, lines~839--942:
\begin{align}
    e_*&=\frac{2}{1000000},&
    \mu&=\frac14,&
    \ell&=\frac1{100000},&
    \omega&=\frac1{20000},&
    \kappa&=\frac1{1000000}.
    \label{eq:final_scalar_parameters}
\end{align}
The source target and prefix entropy estimates at exponent $e_*$
are antecedents throughout this section. This calculation does not
construct the preceding safe-box estimate.

\begin{theorem}[Literal rounded scales and scalar losses]
    \label{thm:arealaw-final-sharp-scales}
    \lean{TNLean.PEPS.AreaLaw.Scan.ScannerExponents.eventually_bandCount_le_rpow,TNLean.PEPS.AreaLaw.Scan.exists_finalSharp_scale_bounds,TNLean.PEPS.AreaLaw.Scan.finalSharp_neg_log_mass_le,TNLean.PEPS.AreaLaw.Scan.finalSharp_binEntropy_le,TNLean.PEPS.AreaLaw.Scan.finalSharp_binary_cost_le}
    Retain the actual scales
    \begin{align}
        L&=\lfloor n^{1-\ell}\rfloor,&
        m&=\lfloor n^\mu\rfloor,&
        K&=\left\lfloor\frac{L}{8m}\right\rfloor,&
        W&=n^\omega,&
        a&=\frac WK,&
        \tau&=n^{-1/4000}.
        \label{eq:final_scalar_scales}
    \end{align}
    There are fixed $C_a\geq1$ and $N\geq2$ such that, for all
    integers $n\geq N$, one has $K\geq1$, $a>0$, $0<\tau\leq1/2$,
    and
    \begin{align}
        a&\leq C_a n^{-0.74994},&
        a^{-1}&\leq n^{0.74994}.
        \label{eq:final_scalar_a}
    \end{align}
    For $n\geq2$ and $1-n^{-100}\leq z\leq1$, $z>0$,
    the mass estimate below holds. The binary estimate additionally
    assumes $a>0$ and $a^{-1}\leq n^{0.74994}$, as supplied above.
    \begin{align}
        0\leq-\log z&\leq2n^{-100},&
        (3+a^{-1})h(\tau)
          &\leq4n^{0.74969}(1+\log n/4000).
        \label{eq:final_scalar_mass_binary}
    \end{align}
    The binary estimate uses $h(t)\leq t(1-\log t)$ for $0<t<1$.
\end{theorem}

\begin{proof}
    The existing lower bound for the rounded band count gives the
    first inequality in (\ref{eq:final_scalar_a}). For the second,
    $n^\mu/2\leq m$ and $8Km\leq L\leq n^{1-\ell}$ give
    $K\leq n^{1-\ell-\mu}$. The negative logarithm is bounded by
    $z^{-1}-1\leq2n^{-100}$. Applying the same logarithmic inequality
    to $1-t$ gives $h(t)\leq t(1-\log t)$. Substitute
    $\tau=n^{-1/4000}$ and (\ref{eq:final_scalar_a}).
\end{proof}

\begin{definition}[Complete post-replica scalar error]
    \label{def:arealaw-final-sharp-error}
    \lean{TNLean.PEPS.AreaLaw.Scan.finalSharpError,TNLean.PEPS.AreaLaw.Scan.finalSharpPowerEnvelope}
    Set $D=\lceil n^\kappa\rceil$, $w=n^{3/5}$,
    $c=w-\log z$, and
    \begin{align}
        H&=4S_t+2c+2B_{\rm sh}+6nD\log q.
        \label{eq:final_scalar_H}
    \end{align}
    With the actual final choices of $W$ and $\tau$, define
    \begin{align}
        \mathcal E_n
          &=\frac{4S_t+4c-2\log z}{W}
            +(\tau+n^{-1/2000}/\tau)H
            +(3+a^{-1})h(\tau) \notag\\
          &\quad+37C_0\mathcal B a
            -\frac{\log z}{a}+2c+16\sqrt t\,d_X.
        \label{eq:final_scalar_error}
    \end{align}
    Here $t$ is the original-to-truncated trace distance and
    $d_X=\log(e\dim\mathcal H_X)$. No far-region dimension occurs.
\end{definition}

\begin{theorem}[Pointwise source envelope]
    \label{thm:arealaw-final-sharp-envelope}
    \lean{TNLean.PEPS.AreaLaw.Scan.finalSharp_source_entropy_bounds,TNLean.PEPS.AreaLaw.Scan.finalSharpError_le_powerEnvelope}
    Fix nonnegative $C_e,C_0,C_B,C_a,C_{\rm tr},C_{\dim}$
    and $\log q$. In addition to (\ref{eq:final_scalar_a}) and
    the selected-mass bounds, assume
    \begin{align}
        S_t&\leq C_e n^{1+e_*}+n^{-247},&
        B_{\rm sh}&\leq C_e(nL^{e_*}+n)+n^{-247}, \notag\\
        \mathcal B&\leq C_B nD(\log n)^{12},&
        t&\leq C_{\rm tr}n^{-500},&
        d_X&\leq1+C_{\dim}n^2.
        \label{eq:final_scalar_inputs}
    \end{align}
    The complete error is at most
    \begin{align}
        \mathcal E_n
          &\leq(4C_e+20)n^{0.999952}
            +2(8C_e+14+12\log q)n^{0.999752} \notag\\
          &\quad+4n^{0.74969}(1+\log n/4000)
            +74C_aC_0C_B n^{0.250061}(\log n)^{12} \notag\\
          &\quad+2n^{-99.25006}+6n^{0.6}
            +16\sqrt{C_{\rm tr}}(1+C_{\dim})n^{-248}.
        \label{eq:final_scalar_envelope}
    \end{align}
    The decimal powers in this display denote exact terminating
    rationals, not numerical approximations.
\end{theorem}

\begin{proof}
    \uses{thm:arealaw-final-sharp-scales,def:arealaw-final-sharp-error}
    The source entropy inputs give
    $S_t\leq(C_e+1)n^{1+e_*}$ and
    $B_{\rm sh}\leq2(C_e+1)n^{1+e_*}$.
    Also $nD\leq2n^{1+e_*}$ since $\kappa\leq e_*$, and
    $c\leq3n^{3/5}$. These give the first two terms.
    The third is (\ref{eq:final_scalar_mass_binary}).
    The marginal budget and $a$ upper bound give the fourth.
    The quantitative mass loss gives the fifth, while $2c$
    gives the sixth. Finally $\sqrt t\leq\sqrt{C_{\rm tr}}n^{-250}$
    and $d_X\leq(1+C_{\dim})n^2$ give the trace correction.
\end{proof}

\begin{theorem}[Uniform final power absorption]
    \label{thm:arealaw-final-sharp-power}
    \lean{TNLean.PEPS.AreaLaw.Scan.finalSharp_error_exponents,TNLean.PEPS.AreaLaw.Scan.eventually_finalSharp_log_cost_le,TNLean.PEPS.AreaLaw.Scan.eventually_finalSharp_powerEnvelope_le,TNLean.PEPS.AreaLaw.Scan.exists_finalSharp_error_le}
    For the fixed coefficients in (\ref{eq:final_scalar_inputs}),
    one threshold works for every remaining scalar datum at the
    literal rounded scales. With
    \begin{align}
        C&=20C_e+62+24\log q
            +16\sqrt{C_{\rm tr}}(1+C_{\dim}),
        \label{eq:final_scalar_coefficient}
    \end{align}
    it gives
    \begin{align}
        \mathcal E_n&\leq Cn^{62497/62500}
          \leq Cn^{1-1/100000}.
        \label{eq:final_scalar_output}
    \end{align}
    The exact maximal power is $62497/62500=0.999952$.
    Its margin to $1-1/100000$ is $19/500000>0$.
\end{theorem}

\begin{proof}
    \uses{thm:arealaw-final-sharp-envelope}
    The binary logarithmic term and the marginal logarithmic term
    each become at most $n^{62497/62500}$ after fixing their
    coefficients and enlarging the threshold. Every remaining
    exponent is at most $62497/62500$. Adding their coefficients
    gives (\ref{eq:final_scalar_coefficient}).
\end{proof}

\begin{theorem}[Exact trace budget and transport smallness]
    \label{thm:arealaw-final-sharp-source}
    \lean{TNLean.PEPS.AreaLaw.Scan.finalSharp_traceDistance_power,TNLean.PEPS.AreaLaw.Scan.exists_finalSharp_source_error_le,TNLean.PEPS.AreaLaw.Scan.exists_finalSharp_transport_smallness}
    For fixed $q\geq1$, $g>0$ and $C_{\rm vol}\geq0$, the source estimate
    \begin{align}
        t&\leq\sqrt{2\min(n^{-1000},g/4)/g}
          \leq\sqrt{2/g}\,n^{-500}
        \label{eq:final_scalar_trace}
    \end{align}
    is covered by $C_{\rm tr}=\sqrt{2/g}$ in the preceding theorem.
    If $|X|\leq C_{\rm vol}n^2$, its first-region dimension bound
    is covered by $C_{\dim}=C_{\rm vol}\log q$.
    Furthermore, for fixed $C_r\geq0$ and $c_t>0$, a uniform
    threshold gives
    \begin{align}
        a\,\ell_q(\lceil C_r\log^2n\rceil)&\leq c_t.
        \label{eq:final_scalar_transport}
    \end{align}
\end{theorem}

\begin{proof}
    \uses{thm:arealaw-final-sharp-scales,thm:arealaw-final-sharp-power}
    The first estimate follows by the monotonicity and product
    rules for the square root. The accepted transport estimate is
    $\ell_q(\lceil C_r\log^2n\rceil)\leq C_d\log^4n$.
    Multiply it by $a\leq C_an^{-0.74994}$ and absorb the fixed
    logarithmic factor into any strictly smaller negative power.
\end{proof}

These are scalar implications. Applying them to the actual comparison
requires its same-point finite-copy inequalities and the entropy-to-information
bridge. All replica remainders must first tend to zero with $n,W,a,\tau$
and the physical instance fixed. The chosen round and interpolation parameter
may vary with the replica count. No uniform replica rate in the system size
is asserted. The earlier bootstrap recurrence and the safe-box source
entropy antecedent remain separate obligations.
