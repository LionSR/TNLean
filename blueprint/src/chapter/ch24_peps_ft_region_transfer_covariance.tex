\begin{lemma}[Relabelling of the complement-side blocked weight]%
    \label{thm:peps_regionComplementWeight_transport}
    \lean{TNLean.PEPS.regionComplementWeight_transport}
    \leanok
    \uses{def:peps_tensor_transport, def:peps_regionComplementTwoBlock}
    For any graph isomorphism $\phi:G\simeq G'$,
    \begin{align}
        T_{A^\phi,V'\setminus\phi(R)}^{(\phi\cdot\eta)^c}(\phi\cdot\tau)
         & =T_{A,V\setminus R}^{\eta^c}(\tau).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Identifying $\phi(V\setminus R)=V'\setminus\phi(R)$ and reindexing virtual
    configurations along $\phi$ per Definition~\ref{def:peps_tensor_transport}
    matches the two blocked sums termwise.
\end{proof}

\begin{lemma}[Boundary relabelling preserves agreement away from a bond]%
    \label{thm:peps_sameAwayFromBond_regionBoundaryConfigMap}
    \lean{TNLean.PEPS.sameAwayFromBond_regionBoundaryConfigMap}
    \leanok
    \uses{thm:peps_regionComplementWeight_transport}
    For a graph isomorphism $\phi:G\simeq G'$, a crossing edge $f\in\partial R$,
    and boundary labels $\mu,\nu\in\partial R$,
    \begin{align}
        \mu|_{\partial R\setminus\{f\}}=\nu|_{\partial R\setminus\{f\}}
         & \iff
        (\phi\cdot\mu)|_{\partial\phi(R)\setminus\{\phi(f)\}}
        =(\phi\cdot\nu)|_{\partial\phi(R)\setminus\{\phi(f)\}}.
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    The boundary bijection induced by $\phi$ restricts to a bijection
    $\partial R\setminus\{f\}\xrightarrow{\sim}\partial\phi(R)\setminus\{\phi(f)\}$.
\end{proof}

\begin{lemma}[Covariance of the region insertion coefficient]%
    \label{thm:peps_regionInsertedCoeff_transport}
    \lean{TNLean.PEPS.regionInsertedCoeff_transport}
    \leanok
    \uses{def:peps_regionInsertedCoeff, def:peps_tensor_transport, thm:peps_regionComplementWeight_transport, thm:peps_sameAwayFromBond_regionBoundaryConfigMap}
    For a graph isomorphism $\phi:G\simeq G'$ and $f\in\partial R$,
    \begin{align}
        C_{A^\phi}(\phi(R),\phi(f),M;\phi\cdot\sigma,\phi\cdot\tau)
         & =C_A(R,f,\phi^{*}M;\sigma,\tau).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Reindexing $\mu,\nu\in\partial R$ by $\phi$ in
    Definition~\ref{def:peps_regionInsertedCoeff} preserves the off-$f$
    agreement condition by
    Lemma~\ref{thm:peps_sameAwayFromBond_regionBoundaryConfigMap}, transforms
    $T_{A,R}^{\mu}$ and $T_{A,V\setminus R}^{\nu^c}$ by
    Definition~\ref{def:peps_tensor_transport} and
    Lemma~\ref{thm:peps_regionComplementWeight_transport}, and carries
    $M_{(\phi\cdot\mu)_{\phi(f)},(\phi\cdot\nu)_{\phi(f)}}$ to
    $(\phi^{*}M)_{\mu_f,\nu_f}$.
\end{proof}

\begin{lemma}[Region insertion coefficient under an equality of tensors]%
    \label{thm:peps_regionInsertedCoeff_congrTensor}
    \lean{TNLean.PEPS.regionInsertedCoeff_congrTensor}
    \leanok
    \uses{def:peps_regionInsertedCoeff}
    If $A_1=A_2$ with induced bond-dimension transport $\psi$, then
    \begin{align}
        A_1=A_2
        \quad\Longrightarrow\quad
        C_{A_1}(R,f,M;\sigma,\tau)
         & =C_{A_2}(R,f,\psi M;\sigma,\tau).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Substitution along $A_1=A_2$ identifies the blocked weights and acts as the
    identity $\psi$ on bond dimensions.
\end{proof}

\begin{lemma}[Bond dimension at a translated crossing edge]%
    \label{thm:peps_bondDim_boundaryEdgeMap_translate}
    \lean{TNLean.PEPS.bondDim_boundaryEdgeMap_translate}
    \leanok
    \uses{def:peps_torus_translation_invariant, def:peps_torus_geometry_translation}
    If $A$ is translation invariant on the discrete torus, then for every
    $f\in\partial R$ and translation $T_{a,b}$,
    $D_A(T_{a,b}(f))=D_A(f)$.
\end{lemma}
\begin{proof}\leanok
    By Definition~\ref{def:peps_torus_translation_invariant}, $A^{T_{a,b}}=A$,
    so $D_A(T_{a,b}(f))=D_{A^{T_{a,b}}}(T_{a,b}(f))=D_A(f)$.
\end{proof}

\begin{lemma}[Translation covariance of the coefficient identity]%
    \label{thm:peps_regionInsertedCoeff_translate_coeffIdentity}
    \lean{TNLean.PEPS.regionInsertedCoeff_translate_coeffIdentity}
    \leanok
    \uses{thm:peps_regionInsertedCoeff_transport, thm:peps_regionInsertedCoeff_congrTensor, thm:peps_bondDim_boundaryEdgeMap_translate, def:peps_torus_translation_invariant}
    Let $A,B$ be translation invariant on the discrete torus and $f\in\partial R$.
    If $C_A(R,f,M;\sigma,\tau)=C_B(R,f,g(M);\sigma,\tau)$ for all $M,\sigma,\tau$,
    then for every translation $T_{a,b}$,
    \begin{align}
        C_A\bigl(T_{a,b}(R),T_{a,b}(f),M';
        T_{a,b}\cdot\sigma,T_{a,b}\cdot\tau\bigr)
         & =
        C_B\bigl(T_{a,b}(R),T_{a,b}(f),
        \widehat{g}(M');
        T_{a,b}\cdot\sigma,T_{a,b}\cdot\tau\bigr),
        \label{eq:peps_translate_coeff}
    \end{align}
    where $\widehat{g}(M')=(T_{a,b}^{-1})^*g(T_{a,b}^*M')$.
\end{lemma}
\begin{proof}\leanok
    Combining $A^{T_{a,b}}=A$ and $B^{T_{a,b}}=B$ with
    Lemmas~\ref{thm:peps_regionInsertedCoeff_transport},
    \ref{thm:peps_regionInsertedCoeff_congrTensor},
    and~\ref{thm:peps_bondDim_boundaryEdgeMap_translate} rewrites both sides of
    (\ref{eq:peps_translate_coeff}) as the reference identity evaluated at
    $M=T_{a,b}^*M'$.
\end{proof}

\begin{definition}[Single-vertex two-block tensor]%
    \label{def:peps_vertexTwoBlock}
    \lean{TNLean.PEPS.vertexTwoBlock}
    \leanok
    \uses{def:peps_tensor}
    For $v\in V$, the single-vertex two-block tensor on the shared bonds
    $\operatorname{Inc}(v)$ with one-point external boundary $\{*\}$ is
    $T_v(*,\eta,\sigma)=A_v(\eta,\sigma)$.
\end{definition}

\begin{lemma}[Injectivity of the single-vertex two-block tensor]%
    \label{thm:peps_isTwoBlockInjective_vertexTwoBlock}
    \lean{TNLean.PEPS.isTwoBlockInjective_vertexTwoBlock}
    \leanok
    \uses{def:IsVertexInjective, def:peps_vertexTwoBlock}
    If $A$ is vertex-injective, then $\inj(T_v)$ holds for every $v\in V$.
\end{lemma}
\begin{proof}\leanok
    In Definition~\ref{def:peps_vertexTwoBlock}, the bijection $(*,\eta)\mapsto\eta$
    identifies $(T_v(*,\eta,-))_\eta$ with $(A_v(\eta,-))_\eta$, which is
    linearly independent by Definition~\ref{def:IsVertexInjective}.
\end{proof}

\begin{definition}[Complement two-block tensor]%
    \label{def:peps_complementTwoBlock}
    \lean{TNLean.PEPS.complementTwoBlock}
    \leanok
    \uses{def:peps_tensor}
    For $v\in V$, the complement two-block tensor on $\operatorname{Inc}(v)$
    with one-point external boundary $\{*\}$ is
    $T_{v^c}(*,\eta,\tau)=A_{v^c}(\eta,\tau)$, the contraction of $A$ over
    $V\setminus\{v\}$.
\end{definition}

\begin{definition}[Complement-region index identifications]%
    \label{def:peps_vertexComplementRegionEquivs}
    \lean{TNLean.PEPS.isRegionBoundaryEdge_vertexComplementVertices_iff}
    \lean{TNLean.PEPS.vertexComplementBoundaryEdgeEquiv}
    \lean{TNLean.PEPS.vertexComplementBoundaryConfigEquiv}
    \lean{TNLean.PEPS.vertexComplementVertexEquiv}
    \lean{TNLean.PEPS.vertexComplementPhysicalConfigEquiv}
    \leanok
    \uses{def:peps_complementTwoBlock}
    For $v\in V$, $\partial(V\setminus\{v\})=\operatorname{Inc}(v)$ induces a
    boundary-configuration bijection $\Psi_v:\partial(V\setminus\{v\})\xrightarrow{\sim}\mc V_v$,
    and $V\setminus\{v\}=\{w\in V:w\ne v\}$ induces a physical-configuration
    bijection $\Phi_v$.
\end{definition}

\begin{lemma}[Vertex-complement contraction as a blocked region]%
    \label{thm:peps_regionBlockedWeight_vertexComplement}
    \lean{TNLean.PEPS.vertexComplementBoundaryConfigEquiv_regionBoundaryLabel}
    \lean{TNLean.PEPS.regionBlockedWeight_vertexComplement}
    \leanok
    \uses{def:peps_vertexComplementRegionEquivs, def:peps_complementTwoBlock}
    Under $(\Psi_v,\Phi_v)$ from Definition~\ref{def:peps_vertexComplementRegionEquivs},
    \begin{align}
        T_{A,V\setminus\{v\}}^{\rho}(\tau)
         & =
        A_{v^c}(\Psi_v\rho,\Phi_v\tau).
        \notag
    \end{align}
\end{lemma}
\begin{proof}\leanok
    Reindexing the sum over complement virtual configurations $\zeta$ with
    $\zeta|_{\partial}=\rho$ by $\Psi_v$ and the product over $w\ne v$ by
    $\Phi_v$ gives
    \begin{align}
        T_{A,V\setminus\{v\}}^{\rho}(\tau)
         & =
        \textstyle
        \sum_{\zeta:\,\zeta|_{\partial}=\rho}\;
        \prod_{w\neq v} A_w(\zeta|_w,\tau_w)
        =
        A_{v^c}(\Psi_v\rho,\Phi_v\tau).
        \notag
    \end{align}
\end{proof}

\begin{lemma}[Injectivity of the vertex-complement contraction]%
    \label{thm:peps_vertexComplementTensorInjective}
    \lean{TNLean.PEPS.vertexComplementTensorInjective_of_isVertexInjective}
    \leanok
    \uses{def:IsVertexInjective, def:peps_tensor}
    If $A$ is vertex-injective and $\forall e\in E,\ D_A(e)>0$, then
    $\eta\mapsto A_{v^c}(\eta,-)$ is linearly independent for every $v\in V$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_regionBlockedWeight_vertexComplement, thm:peps_regionBlockedTensorInjective_of_isVertexInjective}
    Lemma~\ref{thm:peps_regionBlockedTensorInjective_of_isVertexInjective}
    gives $\inj(T_{A,V\setminus\{v\}})$, which transports across the bijections
    $(\Psi_v,\Phi_v)$ of Lemma~\ref{thm:peps_regionBlockedWeight_vertexComplement}
    to linear independence of $\eta\mapsto A_{v^c}(\eta,-)$.
\end{proof}

\begin{lemma}[Injectivity of the complement two-block tensor]%
    \label{thm:peps_isTwoBlockInjective_complementTwoBlock}
    \lean{TNLean.PEPS.isTwoBlockInjective_complementTwoBlock}
    \leanok
    \uses{def:IsVertexInjective, def:peps_complementTwoBlock, thm:peps_vertexComplementTensorInjective}
    If $A$ is vertex-injective and $\forall e\in E,\ D_A(e)>0$, then
    $\inj(T_{v^c})$ holds for every $v\in V$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_vertexComplementTensorInjective}
    Reindexing $(*,\eta)\leftrightarrow\eta$ in
    Definition~\ref{def:peps_complementTwoBlock} reduces $\inj(T_{v^c})$ to
    Lemma~\ref{thm:peps_vertexComplementTensorInjective}.
\end{proof}

\begin{lemma}[One-vertex versus complement comparison]%
    \label{thm:peps_oneVertexComplementComparison}
    \lean{TNLean.PEPS.one_vertex_complement_comparison}
    \leanok
    \uses{thm:peps_postAbsorptionEdgeInsertionEquality, thm:peps_twoInjectiveTensorInsertionComparison}
    Suppose the shared bond type and external boundary spaces are nonempty, and
    $A_v,A_{v^c},\widetilde B_v,\widetilde B_{v^c}$ are injective two-block
    tensors satisfying, for all $b,X,\eta_v,\eta_{v^c},\sigma_v,\sigma_{v^c}$,
    \begin{align}
        C_{A_v,A_{v^c}}(b,X;\eta_v,\eta_{v^c},\sigma_v,\sigma_{v^c})
         & =
        C_{\widetilde B_v,\widetilde B_{v^c}}
        (b,X;\eta_v,\eta_{v^c},\sigma_v,\sigma_{v^c}).
        \notag
    \end{align}
    Then there exists $\lambda_v\in\C^\times$ such that
    $A_v=\lambda_v\widetilde B_v$.
\end{lemma}
\begin{proof}\leanok
    Apply Lemma~\ref{thm:peps_twoInjectiveTensorInsertionComparison} to
    $(A_1,A_2)=(A_v,A_{v^c})$ and $(B_1,B_2)=(\widetilde B_v,\widetilde B_{v^c})$.
\end{proof}

\begin{remark}[Normal-region form]
    The normal-region application in \cite[Section~4]{Molnar2018NormalPEPS}
    uses the same scalar comparison after comparing two injective regions
    $R\subset S=R\cup\{v\}$. If the region comparisons give
    $A_R=\lambda_R\widetilde B_R$ and
    $A_S=\lambda_S\widetilde B_S$, the source-paper comparison is
    % Formula: A_R A_v=A_S\propto\widetilde B_S=\widetilde B_R\widetilde B_v.
    % Source verdict: Molnar et al., arXiv:1804.04964, page 16, lines
    %   1544--1570 of paper_normal.tex.  The source labels the tensors at v
    %   simply by A and \widetilde B.
    % Ink-to-index map: the horizontal bond in each two-tensor panel is the
    %   contracted boundary index between R and v.  The upper stubs are free
    %   physical indices, and the outer horizontal stubs are free virtual
    %   boundary indices.
    % Type verdict: the single upper leg of A_S or \widetilde B_S denotes the
    %   fused physical space H_R\otimes H_v; the source draws that composite
    %   leg once, while the factored panels expose its H_R and H_v factors.
    % Contraction audit: the two-tensor panels have one internal virtual
    %   contraction; the one-tensor panels have none.
    % Two-tensor boundary signature:
    %   boundary|virtual-west=1|virtual-east=1|physical-up=2|physical-down=0.
    % One-tensor boundary signature:
    %   boundary|virtual-west=1|virtual-east=1|physical-up=1|physical-down=0.
    \begin{center}
        \tenkzkernel
        \begin{tenkz}[rows={wire}, cols=2, west=open, east=open, physical=up]
            \tn[label pos=s]{A_R} & \tn[label pos=s]{A}
        \end{tenkz}
        \quad$=$\quad
        \begin{tenkz}[rows={wire}, cols=1, west=open, east=open, physical=up]
            \tn[label pos=s]{A_S}
        \end{tenkz}
        \quad$\propto$\quad
        \begin{tenkz}[rows={wire}, cols=1, west=open, east=open, physical=up]
            \tn[label pos=s]{\widetilde B_S}
        \end{tenkz}
        \quad$=$\quad
        \begin{tenkz}[rows={wire}, cols=2, west=open, east=open, physical=up]
            \tn[label pos=s]{\widetilde B_R} & \tn[label pos=s]{\widetilde B}
        \end{tenkz}.
    \end{center}
    Applying the left inverse given by injectivity of the block
    $A_R=\lambda_R\widetilde B_R$ gives
    $A_v=(\lambda_S/\lambda_R)\widetilde B_v$.
\end{remark}

\begin{definition}[Injective PEPS tensor]%
    \label{def:peps_injective_tensor}
    \lean{TNLean.PEPS.Tensor.IsInjective}
    \leanok
    \uses{def:IsVertexInjective}
    A PEPS tensor $A$ is \emph{injective} if it is vertex-injective
    (\cite[Section~3]{Molnar2018NormalPEPS}) and $\forall e\in E,\ D_A(e)>0$
    (the standing convention of that section, recorded in
    \cite{gap:peps_injective_ft_section3_route}).
\end{definition}

\begin{lemma}[Fundamental Theorem for injective PEPS, conditional on
        bond-dimension equality]%
    \label{thm:fundamentalTheorem_PEPS_of_bondDim}
    \lean{TNLean.PEPS.fundamentalTheorem_PEPS_of_bondDim}
    \leanok
    \uses{thm:gaugeConsistency, thm:peps_oneVertexComplementComparison, def:GaugeEquivPEPS, def:peps_injective_tensor}
    On a connected graph with identified virtual bond spaces $D_A=D_B$, if $A,B$
    are injective PEPS tensors with $\Coeff_A=\Coeff_B$, then $A$ and $B$ are
    gauge equivalent via invertible edge matrices $(X_e)_{e\in E}$:
    $B=A^X$.
\end{lemma}
\begin{proof}\leanok
    Combine Lemma~\ref{thm:gaugeConsistency} and
    Lemma~\ref{thm:peps_oneVertexComplementComparison} across the identified
    bond spaces.
\end{proof}

\begin{lemma}[Equal bond dimensions for injective PEPS]%
    \label{thm:peps_injective_sameState_bondDim_eq}
    \lean{TNLean.PEPS.bondDim_eq_of_matrixAlgEquiv}
    \lean{TNLean.PEPS.bondDim_eq_of_isInjective_sameState}
    \leanok
    \uses{def:peps_injective_tensor, def:peps_same_state}
    If $A,B$ are injective PEPS tensors on the same finite simple graph with
    $\Coeff_A=\Coeff_B$, then $D_A(e)=D_B(e)$ for every $e\in E$
    (\cite[lines~279--282 and 571--582]{Molnar2018NormalPEPS}, without
    requiring connectedness).
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_edgeBlockedThreeSiteInjective,
        thm:peps_edgeBlockedInsertionAlgebraIsomorphism}
    For each $e\in E$, Lemmas~\ref{thm:peps_edgeBlockedThreeSiteInjective}
    and~\ref{thm:peps_edgeBlockedInsertionAlgebraIsomorphism} give an algebra
    isomorphism $\MN{D_A(e)}\simeq\MN{D_B(e)}$, so
    $D_A(e)^2=D_B(e)^2$ and hence $D_A(e)=D_B(e)$.
\end{proof}

\begin{theorem}[Fundamental Theorem for injective PEPS]%
    \label{thm:fundamentalTheorem_PEPS}
    \lean{TNLean.PEPS.fundamentalTheorem_PEPS}
    \leanok
    % Local fix: the nonzero virtual bond spaces of Theorem 2 in
    % \cite{Molnar2018NormalPEPS} are part of the injectivity hypothesis,
    % as recorded in
    % \path{docs/paper-gaps/peps_injective_ft_section3_route.tex}.
    \uses{thm:fundamentalTheorem_PEPS_of_bondDim, def:GaugeEquivPEPS, def:peps_injective_tensor}
    Let $A,B$ be injective PEPS tensors on a connected finite simple graph. If
    $\Coeff_A=\Coeff_B$, then $\forall e\in E,\ D_A(e)=D_B(e)$ and, after
    identifying these bond spaces, there exist invertible edge matrices
    $(X_e)_{e\in E}$ such that $B=A^X$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:peps_injective_sameState_bondDim_eq}
    Lemma~\ref{thm:peps_injective_sameState_bondDim_eq} gives $D_A(e)=D_B(e)$
    for all $e\in E$, after which
    Lemma~\ref{thm:fundamentalTheorem_PEPS_of_bondDim} applies.
\end{proof}
