\section{The scanner estimate with four norm comparisons}
\label{sec:al_scanner_source}

The scanner estimate of \cite[Proposition~9.2]{OpenAI2026AreaLaw} selects a charge round and interpolation parameter satisfying the density, energy, and all four norm comparisons (with output shell and density coefficients allowed to grow after fixing the input entropy coefficient $C_e$; see~\cite{gap:arealaw2d_scanner_inputs}).

\begin{definition}[Statement of the scanner estimate]
    \label{def:al_scanner_source_statement}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.ScannerHypotheses,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.typicalWindow,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.scanPrevector,
        TNLean.PEPS.AreaLaw.Scan.CollarScan.ScannerStatement}
    \leanok
    \uses{thm:al_scan_truncation_inputs,thm:arealaw-supplied-target-norm-bounds}
    Fix exponents $0<\ell<1$, $0<\kappa<\mu<1-\ell$, $\kappa\le e<1$, $\nu>0$, interaction range $R$, term bound $J$, gap $\Delta$, target-size coefficient $C_T$, and $g=\Delta/\max\{1,J(\|f_{1,\Delta/2}\|_1+1)\}$.
    At scale $n$, set
    \begin{align}
        L =\lfloor n^{1-\ell}\rfloor,\quad
        m =\lfloor n^\mu\rfloor,\quad
        K =\lfloor L/(8m)\rfloor,\quad
        D =\lceil n^\kappa\rceil,\quad
        \epsilon =n^{-\nu},\quad
        a =W/K.\notag
    \end{align}
    The scale-$n$ source hypotheses specify a range-$R$, bound-$J$, gap-$\Delta$ Hamiltonian on $\Lambda$ with unit ground vector $\Omega$; a collar scan (measuring depth by $\operatorname{dist}_\infty(\cdot,T)$ with anchor-containing supports, $n$ slots per depth row, truncation radius $r_0=\lceil C_{\rm tr}(\log n)^2\rceil$, and charge-list constant $C_1$); a finite target $T$ with $|T|\le C_Tn^2$, $|T_j\setminus T_{j-1}|\le n$ ($1\le j\le L$), and $\operatorname{dist}_\infty(Z_A,T)>2L+10r_0$; and nonempty $X=A\cap T$ and shells $Q_j=A\cap(T_j\setminus T)$ ($0\le j\le L$) satisfying
    \begin{align}
        S_\Omega(X)   \le C_en^{1+e},\qquad
        S_\Omega(Q_j) \le C_e(nL^e+n).\notag
    \end{align}

    The scanner statement asserts that there exists $C_{\rm tr}>0$ such that for all integers $q\ge2$ and reals $C_e$, there exist constants $C_1,c_a,C,C_\delta>0$, $C_b,C_{l,b},C_l,C_{l,\delta}$, and $N_0$ such that for every $n\ge N_0$ and finite system obeying the source hypotheses:
    \begin{enumerate}
        \item The truncated Hamiltonian has a unit ground vector $\widetilde\Omega$ with energy $0\le e_t\le n^{-1000}$, gap $g/2$, and trace distance $D_{\rm tr}(\rho,\sigma)=\frac12\|\rho-\sigma\|_1$ obeying
              \begin{align}
                  D_{\rm tr}(\widetilde\Omega\widetilde\Omega^*,\Omega\Omega^*)
                   & \le\sqrt{2\min\{n^{-1000},g/4\}/g};
                  \label{eq:al_scanner_source_trace_distance}
              \end{align}
              its width-$n^{3/5}$ target typical set $E$ is nonempty of size $d_*=|E|$ and mass $0<1-n^{-100}\le z\le1$; and a single label sequence $\lambda_k$ defines replica vectors $\mathrm{pre}_k=\pi_R^{\lambda_k}(\widetilde\Omega\otimes\mathrm{Bell}_{CR})^{\otimes k}$ with $0<\|\mathrm{pre}_k\|\le1$ ($\forall k>0$, prior to choosing $W$).
        \item For every $W\ge1$ with $ar_0^2\le c_a$, comparator parameters
              \begin{align}
                  B_{\rm sh}\le C_b(nL^e+n),\qquad
                  B_{\rm exc}\le C_bnD,\qquad
                  1\le\mathcal B\le C_bnD(\log n)^{C_{l,b}} \notag
              \end{align}
              bound the labelled cut parameters of all completed and pre-charge status cuts by $\mathcal B$ (with marginal moment bounds for $\widetilde\Omega$ using each region's labelled cut bound), and for $k\gg1$ the terminal charge at parameter $1$ satisfies (with $\rho_{\rm end,k}\to0$)
              \begin{align}
                  -\frac{\log N_{\rm end,k}^2}{k}
                   & \le2\log d_*-\log z
                  +W(B_{\rm sh}/z+B_{\rm exc})+\rho_{\rm end,k}.\notag
              \end{align}
        \item With $\delta_n=C_\delta\epsilon^{-1}(n^{e-\mu}+a^{1/4}(\log n)^{C_{l,\delta}})$, there exists $\rho_k\to0$ such that for $k\gg1$, a selected charge round and $p_k\in[\epsilon/2,\epsilon]$ with filtered norm $N_k>0$, normalized vector $v_k$, and $E_{{\rm def},k}=(\langle v_k,\overline{\widetilde H}_k v_k\rangle-e_t)/(g/2)\ge0$ satisfy
              \begin{align}
                  \frac{\mathcal Q_k(p_k)}{KnD} & \le\delta_n+\rho_k,\notag        \\
                  E_{{\rm def},k}               & \le Cn^\ell D^2W^2(\log n)^{C_l}
                  \left(\delta_n^{1/8}+\epsilon+n^{-100}\right)
                  +Cn^{-1000}+\rho_k,
                  \label{eq:al_scanner_source_energy}
              \end{align}
              together with the rough upper bound $-\frac{1}{k}\log N_k^2\le2\log d_*-\log z+W(B_{\rm sh}/z+B_{\rm exc})+R_k^{\rm RU}$, the rough lower bound (\ref{eq:arealaw-supplied-rough-lower}) for all $0<\tau\le1/2$, and—when $a\sqrt{\mathcal B}\le\min\{c_0,1\}/10$ for $C_0=1024\mathrm e/g$ and $c_0=[32\sqrt{1+2/g}]^{-1}$—both sharp bounds (\ref{eq:arealaw-supplied-sharp-bounds}) of Theorem~\ref{thm:arealaw-supplied-target-norm-bounds} with remainders vanishing uniformly in the selected charge and $p_k$ (while keeping input $C_e$ fixed; see~\cite{gap:arealaw2d_scanner_inputs}).
    \end{enumerate}
\end{definition}

\begin{theorem}[The scanner estimate]
    \label{thm:al_scanner_source}
    \lean{TNLean.PEPS.AreaLaw.Scan.CollarScan.scannerStatement}
    \leanok
    \uses{def:al_scanner_source_statement}
    Definition~\ref{def:al_scanner_source_statement} holds for $J\ge0$, $\Delta>0$, and $C_T\ge0$, establishing \cite[Proposition~9.2]{OpenAI2026AreaLaw} under~\cite{gap:arealaw2d_scanner_inputs}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:arealaw-source-selected-energy,thm:arealaw-supplied-target-norm-bounds,
        thm:al_scan_common_status_budget,thm:al_positive_prefix_entropy_transfer,
        thm:al_scan_actual_rough_upper,thm:al_augmented_prevector_norm,thm:al_lattice_counts}
    Choose $C_{\rm tr}$ from Theorem~\ref{thm:arealaw-source-selected-energy} before fixing $q$.
    By Theorem~\ref{thm:al_lattice_counts}, range-$R$ anchor multiplicity is at most $\mu_*=2^{2R(R+1)}$; setting $C_1=3\mu_*+1$ and applying Theorem~\ref{thm:arealaw-source-selected-energy} with entropy coefficient $\max\{C_e,1\}$ yields $\widetilde\Omega$, $E$, $\lambda_k$, and a selected point satisfying (\ref{eq:al_scanner_source_trace_distance}), $1-n^{-100}\le z\le1$, $\|\mathrm{pre}_k\|\le1$ (Theorem~\ref{thm:al_augmented_prevector_norm}), and (\ref{eq:al_scanner_source_energy}).

    Applying Theorem~\ref{thm:arealaw-supplied-target-norm-bounds} and Theorem~\ref{thm:al_scan_common_status_budget} with cut and norm coefficients $C_{B,\rm cut},C_{B,\rm norm}$, set
    \begin{align}
        C_b        =\max\{C_{B,\rm cut},C_{B,\rm norm},
        \max\{C_e,1\}+1,1+3\log q\},\qquad
        \mathcal B =C_bnD(\log n)^{12}. \notag
    \end{align}
    Theorem~\ref{thm:al_positive_prefix_entropy_transfer} gives $B_{\rm sh}=\max\{C_e,1\}(nL^e+n)+n^{-247}$ and $B_{\rm exc}=1+3nD\log q\le C_bnD$, while Theorem~\ref{thm:al_scan_actual_rough_upper} gives the terminal-round bound at parameter $1$ and the rough upper bound at the selected point.
\end{proof}
