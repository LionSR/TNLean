\begin{definition}[A strip beside a flux endpoint]
    \label{def:peps_torus_charge_flux_crossing_strip}
    \lean{TNLean.PEPS.torusChargeFluxCrossingRegion,
        TNLean.PEPS.torusChargeFluxCrossingChargeBond}
    \leanok
    \uses{def:peps_three_plaquette_geometry,
        def:peps_torus_charge_string_edge}
    For $v=(x,y)$, let $R_v$ be the translated $4\times 2$ strip with lower-left vertex $(x-1,y+1)$, charge bond $e_c$ joining $(x+1,y+1)$ to $(x+2,y+1)$, initial flux-string endpoint in its leftmost plaquette, and spanning tree consisting of all six horizontal bonds and the rightmost vertical bond (\cite{Schuch2010PEPS}, lines 2560--2581).
\end{definition}

\begin{lemma}[A physical charge crossing beside the flux endpoint]
    \label{thm:peps_torus_charge_flux_crossing}
    \lean{TNLean.PEPS.IsGIsometric.exists_unitary_torusChargeFluxCrossing,
        TNLean.PEPS.IsGIsometric.exists_unitary_torusCorrelatedChargeFluxCrossing}
    \leanok
    \uses{def:peps_torus_charge_flux_crossing_strip,
        thm:peps_regular_charge_flux_physical_transport,
        thm:peps_torus_swept_string_transport}
    For finite $G$, torus periods $w\ge 5$ and $H\ge 4$, and a regular $G$-isometric four-leg tensor, on each translated strip $R_v$ of Definition~\ref{def:peps_torus_charge_flux_crossing_strip} (including across either seam, without prior tree-gauge, Gram, or state hypotheses) there exists an original-spin unitary $W_R$ (chosen before $k\in G$, scalar function $\chi$, $p,q\in G$, and crossing labels $\theta$) acting on the open contraction with initial flux string $u^{(0)}_k$ to transform the single and partner-correlated charge weights via
    \begin{align}
        \chi(p\eta_c)\longmapsto\chi(pk^{-1}\eta_c),\qquad
        \chi(pq^{-1}\eta_c)\longmapsto\chi(pq^{-1}k^{-1}\eta_c). \label{eq:torus_charge_flux_crossing}
    \end{align}
    This auxiliary local operation does not identify the complete braid or reunion experiment of \cite{Schuch2010PEPS}, lines 2560--2615.
\end{lemma}
\begin{proof}\leanok
    Under $u^{(0)}_k$ (Lemma~\ref{thm:peps_torus_swept_string_transport}), the seven tree bonds carry $1$ except the rightmost upward bond $k$, so the normalized tree gauge is $1$ on the lower row (including $t(e_c)$) and $k$ on the upper row, and the leftmost chord residual is $k^{-1}$ (or $k$ when reversed at a seam).
    Applying Lemma~\ref{thm:peps_regular_charge_flux_physical_transport} (or its adjoint) gives the first map in (\ref{eq:torus_charge_flux_crossing}), and substituting $\chi(\cdot)\mapsto\chi(pq^{-1}\cdot)$ gives the second.
\end{proof}
