\begin{lemma}[Exterior replacement with prescribed winding]%
    \label{thm:peps_torus_complement_path_replacement}
    \lean{TNLean.PEPS.exists_complementWalk_of_integerExteriorCollarGraph_connected,
        TNLean.PEPS.exists_complementBoundaryLoops_of_integerExteriorCollarGraph_connected}
    \leanok
    \uses{def:peps_integer_exterior_collar_graph}
    Let both torus periods be at least three. Suppose a region has an
    integer unit-step lift, its safe exterior graph is connected, and
    its exterior collar projects outside the torus region.
    Every route which enters through one crossing bond, follows an
    interior path, and leaves through another crossing bond has a
    complementary replacement with the same two seam crossing numbers.

    In the auxiliary geometric construction for
    \cite[proof of Theorem~6.9]{Schuch2010PEPS}, fix in particular rooted
    spanning trees in the region and its complement and a reference crossing
    bond. For each crossing bond there is a loop based at the complementary
    root, wholly within the induced complement, with the winding of the
    corresponding two-tree boundary route, chosen independently of all
    group-valued closure labels.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_integer_exterior_collar_paths,
        thm:peps_torus_region_boundary_lift,
        thm:peps_torus_exterior_path_approximation}
    Lifting the two exterior crossing endpoints places them in the
    integer band, with deck-coordinate difference equal to the
    crossing numbers of the interior route. A safe graph walk gives
    a continuous collar path between them, and projection exclusion together with
    lattice-path approximation gives a native complementary walk with
    that same deck-coordinate difference. Prepending the complementary
    tree path from the root to the first crossing and appending the
    tree path from the second crossing back to the root yields the loop, since crossing
    numbers add under concatenation.
\end{proof}

\begin{lemma}[Complementary boundary loops of a simply connected region]%
    \label{thm:peps_torus_simply_connected_boundary_loops}
    \lean{TNLean.PEPS.exists_complementBoundaryLoops_of_isSimplyConnected}\leanok
    In the exterior geometric construction of
    \cite[proof of Theorem~6.9]{Schuch2010PEPS}, let both torus periods be at
    least three, and suppose the closed-cell realization of $R$ is simply
    connected. Choose rooted spanning trees in $R$ and its induced complement
    and fix a reference crossing bond. There is a complementary root loop for
    every crossing bond with the same winding as its two-tree boundary route,
    chosen before all group-valued closure labels without any additional
    exterior-path, collar-connectivity, or winding-identity premise.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_torus_complement_path_replacement,
        thm:peps_integer_boundary_contour_connected,
        thm:peps_torus_integer_cell_lift}
    Choosing one covering lift whose planar integer-cell union is simply
    connected and whose exterior collar projects outside the torus
    region gives a safe exterior graph that is connected by the boundary contour
    theorem, so the preceding replacement construction applies to the
    chosen two-tree routes.
\end{proof}

\begin{lemma}[Avoiding a simply connected region with prescribed winding]
    \label{lem:peps_simply_connected_walk_avoidance}
    \lean{TNLean.PEPS.exists_complementWalk_same_winding_of_isSimplyConnected}
    \leanok
    Let both torus periods be at least three, let $R$ have simply connected
    closed-cell realization, and let $a,b\notin R$. Every finite torus
    graph walk $p:a\to b$ has a walk $q:a\to b$ wholly in the induced
    complement, with the same integer seam crossing numbers:
    \begin{align}
        \operatorname{wind}(q)=\operatorname{wind}(p).
        \label{eq:peps_walk_avoidance_winding}
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:peps_torus_complement_path_replacement,
        thm:peps_integer_boundary_contour_connected,
        thm:peps_torus_integer_cell_lift}
    Induct on the length of $p$. An exterior first edge is retained and
    the remaining walk is treated inductively. If the first edge enters
    $R$, split at the first subsequent exit. The finite interior segment
    has a complementary detour with the same winding, by the integer-cell
    lift, connected exterior collar, and exterior replacement lemma.
    Apply the induction hypothesis to the shorter remaining walk and
    concatenate. Additivity of winding gives
    (\ref{eq:peps_walk_avoidance_winding}).
\end{proof}


\begin{definition}[Native dual path as a graph walk]
    \label{def:peps_dual_path_graph_walk}
    \lean{TNLean.PEPS.torusDualPathGraphWalk}\leanok
    Replace each labelled east, west, north, or south dual step by its
    corresponding native graph edge and concatenate in path order.
\end{definition}

\begin{lemma}[Native dual avoidance with fixed displacement]
    \label{lem:peps_simply_connected_dual_avoidance}
    \lean{TNLean.PEPS.exists_torusDualPath_avoiding_of_isSimplyConnected}\leanok
    Under the preceding hypotheses on the forbidden plaquette set $R$,
    every four-labelled dual path $p:a\to b$ with $a,b\notin R$ admits
    a dual path $q:a\to b$ with the same lifted integer displacement,
    all of whose visited plaquettes lie outside $R$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:peps_simply_connected_walk_avoidance,
        def:peps_dual_path_graph_walk}
    Encode $p$ as a graph walk and use the complementary detour. Every
    native graph edge has a corresponding direction label, so decode
    the detour as $q$. Its displacement equals the endpoint coordinate
    difference plus the period times each seam crossing number.
    Equal seam crossing numbers therefore give equal displacements.
\end{proof}

\begin{definition}[Dual plaquettes incident to a physical region]
    \label{def:peps_physical_forbidden_dual_plaquettes}
    \lean{TNLean.PEPS.torusIncidentDualPlaquettes}\leanok
    For a physical observation region $R$, let $B(R)$ consist of dual
    plaquettes with at least one of their four physical corners in $R$.
\end{definition}

\begin{lemma}[A detour with no incident physical bond crossings]
    \label{lem:peps_physical_dual_avoidance}
    \lean{TNLean.PEPS.exists_torusDualPath_zero_crossings_on_incident}\leanok
    Let both torus periods be at least three. Suppose the closed-cell
    realization of $B(R)$ is simply connected and the endpoints of a
    native dual path $p$ lie outside $B(R)$. There exists a native dual
    path $q$ with those endpoints and the same lifted displacement, whose
    signed crossing counts vanish on every physical bond incident to $R$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:peps_simply_connected_dual_avoidance,
        def:peps_physical_forbidden_dual_plaquettes}
    Apply the native detour lemma to the computed forbidden plaquettes.
    All four physical corners of every visited plaquette lie outside $R$.
    Each step crosses a bond between two corners of one of its endpoint
    plaquettes. Its contribution to every incident bond count therefore
    vanishes, and addition of the step counts proves the claim.
\end{proof}

\begin{lemma}[Identity insertions on incident physical bonds]
    \label{lem:peps_physical_dual_identity_insertions}
    \lean{TNLean.PEPS.exists_torusDualPath_trivial_on_incident}\leanok
    Under the preceding geometric hypotheses, for any group $G$ and
    $g\in G$ the constructed detour has identity native group insertions
    on every physical bond incident to $R$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:peps_physical_dual_avoidance}
    Native insertions are powers of $g$ with the signed crossing counts
    as exponents. The preceding lemma makes every relevant exponent zero.
\end{proof}
