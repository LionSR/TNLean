\section{Separation of births from earlier actual regions}
\label{sec:area_law_birth_separation}

A replacement born at a mark is compared with the final actual regions of
the older and peer identifiers of the same label, not with their births. The
comparison near the mark is reduced to a fixed normalized square, where a
compactness argument handles points away from the marks and an explicit
angular estimate handles points near a mark. The trim that follows the birth
then removes a square of radius comparable with $u^{\rho_0}$ about every mark
from every compared region.

Throughout, $\|\cdot\|$ is the maximum norm on $\mathbb R^2$. Normalize a
removal square to $Q=[-1,1]^2$, so that its center, corners and side
midpoints are the points of $\{-1,0,1\}^2$.

\begin{theorem}[Angular estimate in the maximum norm]
    \label{thm:area_law_sup_angular_estimate}
    \lean{TNLean.PEPS.AreaLaw.Geometry.half_max_le_norm_smul_sub_smul,
        TNLean.PEPS.AreaLaw.Geometry.unitBox,
        TNLean.PEPS.AreaLaw.Geometry.one_le_norm_sub_of_disjoint_unitBox,
        TNLean.PEPS.AreaLaw.Geometry.norm_eq_one_of_mem_unitBox}
    \leanok
    Let $d,e$ be unit vectors with $\|d-e\|\ge1$ and let $r,s\ge0$. Then
    \begin{align}
        \|rd-se\| & \ge\tfrac12\max(r,s).
        \label{eq:area_law_sup_angular_estimate}
    \end{align}
    For $d\in\mathbb R^2$ let $B(d)$ be the box with opposite corners
    $(\lfloor d_1\rfloor,\lfloor d_2\rfloor)$ and
    $(\lceil d_1\rceil,\lceil d_2\rceil)$. If $B(d)\cap B(e)=\emptyset$, then
    $\|d-e\|\ge1$. If $\|d\|=1$, then $B(d)$ lies on the unit sphere; it is
    the closed elementary half-side containing $d$, or $d$ itself when $d$
    is one of the eight points of $\{-1,0,1\}^2\setminus\{0\}$.
\end{theorem}

\begin{proof}
    \leanok
    Assume $s\le r$ and put $D=\|rd-se\|$. The reverse triangle inequality
    gives $D\ge r-s$, and
    $r\le r\|d-e\|\le D+(r-s)$ gives $D\ge s$. Hence $D\ge\max(r-s,s)\ge
        r/2$. If $\|d-e\|<1$, then each coordinate satisfies
    $\lfloor d_i\rfloor\le\lceil e_i\rceil$ and
    $\lfloor e_i\rfloor\le\lceil d_i\rceil$, so the point with coordinates
    $\max(\lfloor d_i\rfloor,\lfloor e_i\rfloor)$ lies in both boxes. A unit
    vector has an integer coordinate of absolute value one, which the box
    fixes, while the other coordinate stays in $[-1,1]$.
\end{proof}

\begin{theorem}[Quarter sections of integer half-planes]
    \label{thm:area_law_quarter_sections}
    \lean{TNLean.PEPS.AreaLaw.Geometry.intHalfPlane,
        TNLean.PEPS.AreaLaw.Geometry.quarterRadialPoint,
        TNLean.PEPS.AreaLaw.Geometry.IsQuarterSectorial,
        TNLean.PEPS.AreaLaw.Geometry.isQuarterSectorial_intHalfPlane,
        TNLean.PEPS.AreaLaw.Geometry.IsQuarterSectorial.iUnion,
        TNLean.PEPS.AreaLaw.Geometry.IsQuarterSectorial.iInter}
    \leanok
    \uses{thm:area_law_sup_angular_estimate}
    Call a set $P\subseteq\mathbb R^2$ quarter-sectorial at $q$ if, for
    $0<\|x-q\|\le1/4$, membership of $x$ in $P$ agrees with membership of
    $q+(x-q)/(4\|x-q\|)$, and if, for every unit vector $d$ with
    $q+d/4\in P$, also $q+e/4\in P$ for every $e\in B(d)$. Arbitrary unions
    and intersections of sets quarter-sectorial at $q$ are
    quarter-sectorial at $q$. Every closed half-plane
    $\{x:ax_1+bx_2\le n\}$ with $a,b,n\in\mathbb Z$ and $|a|+|b|\le2$ is
    quarter-sectorial at every point of $\mathbb Z^2$.
\end{theorem}

\begin{proof}
    \leanok
    Put $f(x)=ax_1+bx_2-n$ and $k=f(q)\in\mathbb Z$. On the closed
    quarter-square about $q$, $|f(x)-k|\le(|a|+|b|)/4\le1/2$. If $k\ne0$,
    membership is therefore constant there. If $k=0$, then $f(q+w)=aw_1+bw_2$
    is homogeneous, so membership is unchanged along rays from $q$. On an
    open elementary half-side, say $d=(\pm1,t)$ with $m<t<m+1$ and
    $m\in\{-1,0\}$, the integer bound $|a|+|b|\le2$ excludes a sign change of
    $\pm a+bt$, so a homogeneous inequality at $d$ persists at both endpoints
    $t=m,m+1$ and hence on $B(d)$.
\end{proof}

\begin{theorem}[Angular estimate at a contact mark]
    \label{thm:area_law_contact_angular_estimate}
    \lean{TNLean.PEPS.AreaLaw.Geometry.half_max_le_norm_sub_of_isQuarterSectorial}
    \leanok
    \uses{thm:area_law_sup_angular_estimate, thm:area_law_quarter_sections}
    Let $P,P'$ be quarter-sectorial at $q$, and suppose every common point
    $z\in P\cap P'$ with $\|z-q\|\le1/4$ equals $q$. Then every $x\in P$ and
    $y\in P'$ with $\|x-q\|,\|y-q\|\le1/4$ satisfy
    \begin{align}
        \|x-y\| & \ge\tfrac12\max(\|x-q\|,\|y-q\|).
        \label{eq:area_law_contact_angular_estimate}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    If $x=q$ or $y=q$ the bound is immediate. Otherwise write
    $x=q+rd$ and $y=q+se$ with unit vectors $d,e$. Radial extension puts
    $q+d/4\in P$ and $q+e/4\in P'$. A common point of $B(d)$ and $B(e)$
    would give a common point of $P$ and $P'$ at distance $1/4$ from $q$.
    Thus the boxes are disjoint, $\|d-e\|\ge1$, and
    (\ref{eq:area_law_sup_angular_estimate}) applies.
\end{proof}

\begin{definition}[Normalized replacement fan and exterior trapezoids]
    \label{def:area_law_normalized_repair_fan}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairSlotFacets,
        TNLean.PEPS.AreaLaw.Geometry.repairNormTriangle,
        TNLean.PEPS.AreaLaw.Geometry.repairExteriorTrapezoid,
        TNLean.PEPS.AreaLaw.Geometry.RepairSlotConsecutive}
    \leanok
    \uses{def:area_law_repair_square}
    For each of the eight elementary wedges $s$ of $Q$, let $o_s$ be the
    outward unit normal of its side and let $T_s$ be the closed triangle
    joining the center to its side segment. For $L\in\{1,2\}$, $LT_s$ is cut
    out by $o_s\cdot x\le L$ and the two homogeneous inequalities of its
    radial edges. The exterior trapezoid of $s$ is
    $E_s=\{x\in2T_s:o_s\cdot x\ge1\}$, the part of $2T_s$ outside the open
    square $\operatorname{int}Q$. Two wedges are consecutive when they share
    a radial edge.
\end{definition}

    linear inequalities, one for each pair of wedges. The actual triangle is
    the convex hull of its three vertices, each of which satisfies the three
    inequalities.
\end{proof}

\begin{theorem}[Compactness away from the marks]
    \label{thm:area_law_contact_dichotomy}
    \lean{TNLean.PEPS.AreaLaw.Geometry.exists_pos_le_norm_sub_away_from_intPoints,
        TNLean.PEPS.AreaLaw.Geometry.exists_pos_contact_dichotomy,
        TNLean.PEPS.AreaLaw.Geometry.exists_pos_common_contact_dichotomy,
        TNLean.PEPS.AreaLaw.Geometry.exists_pos_forall_of_finite}
    \leanok
    \uses{thm:area_law_contact_angular_estimate}
    Let $X,Y$ be compact, quarter-sectorial at every point of
    $\mathbb Z^2$, and suppose $X\cap Y\subseteq\mathbb Z^2$. There is
    $\eta>0$ such that for all $x\in X$ and $y\in Y$, either
    $\|x-y\|\ge\eta$, or there is $q\in\mathbb Z^2$ with
    $\|x-q\|,\|y-q\|<1/4$ and $\|x-y\|\ge\|y-q\|/2$.
\end{theorem}

\begin{proof}
    \leanok
    The pairs $(x,y)\in X\times Y$ that do not lie together in an open
    quarter-square about a point of $\mathbb Z^2$ form a compact set. If it
    is nonempty, $\|x-y\|$ attains a minimum there. A zero minimum would give
    a common point $x=y\in\mathbb Z^2$, which lies in its own open
    quarter-square. Pairs inside such a square about $q$ meet only at $q$
    there, and Theorem~\ref{thm:area_law_contact_angular_estimate} applies.
\end{proof}

\begin{theorem}[Explicit separation of belt polygons]
    \label{thm:area_law_belt_polygon_separation}
    \lean{TNLean.PEPS.AreaLaw.Geometry.facetPolygon,
        TNLean.PEPS.AreaLaw.Geometry.exists_mem_facetPolygon_of_le_add,
        TNLean.PEPS.AreaLaw.Geometry.birthSeparationConstant,
        TNLean.PEPS.AreaLaw.Geometry.exists_mem_facetPolygon_inter_of_norm_sub_lt,
        TNLean.PEPS.AreaLaw.Geometry.facetPolygon_contact_dichotomy}
    \leanok
    \uses{thm:area_law_quarter_sections,
        thm:area_law_contact_angular_estimate}
    Call a closed half-plane $\{a x_1+b x_2\le n\}$ a belt half-plane if
    $n\in\mathbb Z$ and $(a,b)$ is one of $(\pm1,0)$, $(0,\pm1)$, or if
    $(a,b)=(\pm1,\pm1)$ and $n$ is even.
    \begin{enumerate}
        \item Let $F$ be a set of belt half-planes, let
              $0\le\varepsilon<1/7$, and let $z$ satisfy
              $a z_1+b z_2\le n+\varepsilon$ for every member of $F$. Then
              the intersection of $F$ contains a point within distance
              $3\varepsilon$ of $z$.
        \item Let $P$ and $Q$ be intersections of belt half-planes with
              $P\cap Q\subseteq\mathbb Z^2$. For all $x\in P$ and
              $y\in Q$, either $\|x-y\|\ge1/28$, or there is
              $q\in P\cap Q$ with $\|x-q\|,\|y-q\|<1/4$ and
              $\|x-y\|\ge\|y-q\|/2$.
    \end{enumerate}
\end{theorem}

\begin{proof}
    \leanok
    For (1), suppose some member is violated at $z$, with normal $(a,b)$ and
    excess $c\in(0,\varepsilon]$, and consider the members whose value at
    $z$ exceeds their level minus $2\varepsilon$. If one of them is not
    parallel to the violated one, the two boundary lines meet in a point
    $v\in\mathbb Z^2$, and Cramer's rule, with a nonzero integer
    determinant, gives $\|z-v\|\le3\varepsilon$. Each value is
    $2$-Lipschitz for the sup norm, so every member has value less than its
    level plus $7\varepsilon<1$ at $v$; this value is an integer, so $v$
    satisfies every member. Otherwise move $z$ by $-c(a,b)/(a^2+b^2)$, a
    vector of length at most $\varepsilon$. The members parallel to the
    violated one then hold, because their levels differ from its level by
    integers; the others had value at most their level minus
    $2\varepsilon$ and change by at most $2\varepsilon$.

    For (2), the values at $x$ of the half-planes of $Q$ exceed their levels
    by at most $2\|x-y\|$. If $\|x-y\|<1/28$, part (1) with
    $\varepsilon=2\|x-y\|$ gives $q\in P\cap Q$ with
    $\|x-q\|\le6\|x-y\|<1/4$ and $\|y-q\|\le7\|x-y\|<1/4$. Then
    $q\in\mathbb Z^2$, belt half-planes are quarter-sectorial at every point
    of $\mathbb Z^2$ by Theorem~\ref{thm:area_law_quarter_sections}, and
    Theorem~\ref{thm:area_law_contact_angular_estimate} applies.
\end{proof}

The constant $1/28$ replaces the positive minimum over the finitely many
rescaled local patterns in the source's compactness argument.

        def:area_law_repair_children}
    Let $c_1=1/28$. The following holds. Let the exterior labels of
    the eight wedges be given, let $C$ be one of their maximal cyclic runs of
    equal label, and let $B_C=\bigcup_{s\in C}T_s$. Let $W_C$ be the union
    of the exterior trapezoids $E_t$ of the wedges whose label differs from
    that of $C$, together with the sets $B_{C'}$ of the other runs $C'$ with
    the label of $C$. Let $v\in\mathbb R^2$ and $u\ge1$, and let $Y$ be a
    set such that
    \begin{enumerate}
        \item every $y\in Y$ with $\|y-v\|<2u$ satisfies
              $(y-v)/u\in W_C$;
        \item $Y$ avoids the nine squares $Q(v+up,u)$, $p\in\{-1,0,1\}^2$.
    \end{enumerate}
    Then $\|x-y\|\ge c_1u^{\rho_0}$ for all $x\in v+uB_C$ and $y\in Y$.
\end{theorem}

\begin{proof}
    \leanok
    Apply part (2) of Theorem~\ref{thm:area_law_belt_polygon_separation} to
    each pair of a triangle $T_s$, $s\in C$, and a set $E_t$ or $T_t$ in
    $W_C$; all of them are intersections of belt half-planes. A wedge
    $t$ of different label is distinct from $s$, and a wedge of another run
    with the same label is neither $s$ nor consecutive to $s$, so
    Theorem~\ref{thm:area_law_normalized_fan_contacts} gives the contact
    hypothesis. This gives $\eta=1/28$, and $c_1=\min(\eta,1/4)=1/28$. If $\|y-v\|\ge2u$, then $\|x-y\|\ge u\ge
        u^{\rho_0}$. Otherwise rescale: either $\|x-y\|\ge\eta u$, or there
    is a mark $q=v+up$ with $\|x-y\|\ge\|y-q\|/2$, where $p$ lies in
    $\{-1,0,1\}^2$ because $\|(x-v)/u\|\le1$. The second hypothesis and
    Theorem~\ref{thm:area_law_repair_margins} give
    $\|y-q\|>d(u)\ge u^{\rho_0}/2$.
\end{proof}

description supplied by the isolated star of the parent mark, after its
removal square has been cut out; the second is the removal of the next
generation's squares, including terminal ones, from every old region.

Theorem~\ref{thm:area_law_birth_separation}. The births of initial belt tiles
are treated next, without any assumption beyond the construction.

For an initial belt tile in a belt cell of side $u=t_k$, normalize that cell
to $[0,4]^2$, so that lengths are measured in units of $u/4$. The cells of
the same and adjacent layers have sides $2$, $4$ or $8$ and corners in

\begin{align}
    x_1=m,\qquad x_2=m,\qquad x_1+x_2=2m,\qquad x_1-x_2=2m,
    \qquad m\in\mathbb Z.

Two of these lines that are not parallel meet in a point of

Definition~\ref{def:area_law_normalized_repair_fan}; we call them atoms.

    (\ref{eq:area_law_belt_arrangement}).

              frontier lie on the lines (\ref{eq:area_law_belt_arrangement}).

        \item Two atoms with a common point outside $\mathbb Z^2$ share a
              nondegenerate segment.
        \item If $q\in\mathbb Z^2$ has $q_1+q_2$ odd, then every atom

\begin{proof}
    \leanok
    An atom is cut out by three half-planes of the arrangement, with integer
    coefficients and $|a|+|b|\le2$, so

    of the arrangement meet in an integer point, by Cramer's rule and the
    parity of the diagonal levels. In the interior of a normalized triangle
    $T_s$ the coordinates lie strictly between consecutive integers and
    $x_1\pm x_2$ strictly between consecutive even integers.
    For (1), pick $c$ in the interior of $P$ and $q\in P$. A ray from $q$ stays
    for a short time in one atom, because the atoms of a square of side two
    are finitely many closed convex sets covering it. This atom meets the
    interior of $P$ near $q$. Its interior is connected and avoids the
    frontier of $P$, so it lies in the interior of $P$. For (2), all facets of

\end{proof}

\begin{theorem}[Birth separation at a common generation]
    \label{thm:area_law_birth_separation_common_stage}
    \lean{TNLean.PEPS.AreaLaw.Geometry.repairStage_birth_separation_of_rank_le}
    \leanok
    \uses{def:area_law_repair_stage,def:area_law_repair_inclusions,
        def:area_law_repair_birth_rank,def:area_law_repair_characteristic_length,
        thm:area_law_initial_belt_birth_separation,
        thm:area_law_replacement_birth_separation_conditional}
    Consider the actual repair family $\mathcal D_j$ of
    Definition~\ref{def:area_law_repair_stage}. Write $I_j$ for its
    identifier set, $*_j$ for its dummy, $B_i^{(j)}$ for a closed birth,
    and $R_i^{(\infty,j)}$ for the final region, obtained by intersecting
    all later retained current regions. Let $\tau_j$ be its characteristic
    length and $\operatorname{rk}_j$ its birth rank.

    Fix $c_1\in\mathbb R$ and assume the following two pointwise bounds.
    For every initial belt birth of side $u=t_k$, every pair of points
    in that birth and in the final region of any other nondummy initial
    identifier of the same label has distance at least $c_1u^{\rho_0}$.
    For every fresh replacement with removal half-side $u$, every pair
    of points in that birth and in the final region of any other
    identifier of the same label present at its birth generation has
    distance at least $c_1u^{\rho_0}$.

    Let $N\ge0$ and $x,y\in I_N$. Suppose that $x$ has positive rank,
    that $y\ne x$ and $y\ne *_N$, and that
    $\operatorname{rk}_N(y)\le\operatorname{rk}_N(x)$ and
    $\xi_y^{(N)}=\xi_x^{(N)}$. Then every $p\in B_x^{(N)}$ and
    $q\in R_y^{(\infty,N)}$ satisfy
    \begin{align}
        c_1\tau_N(x)^{\rho_0} & \le\|p-q\|_\infty.\notag
    \end{align}
    Equal-rank peers are included.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:area_law_repair_birth_rank,
        def:area_law_repair_characteristic_length,
        thm:area_law_repair_final_retention}
    The rank definition gives $\operatorname{rk}_j(i)\le j+1$ for every
    $i\in I_j$, by induction on $j$: initial ranks are zero or one,
    old ranks are retained, and new ranks at generation $j+1$ are $j+2$.

    By Theorem~\ref{thm:area_law_repair_final_retention},
    retention preserves the final region:
    $R_i^{(\infty,j)}=R_{\iota_{j,j+1}(i)}^{(\infty,j+1)}$.

    Induct on $N$. At $N=0$, positive rank makes $x$ an initial belt
    identifier. Its characteristic length is its fine-cell side, so
    the assumed initial bound applies. At $N=j+1$, if $x$ is fresh,
    its characteristic length is the removal half-side, and the
    assumed replacement bound applies whether $y$ is old or fresh.
    If both identifiers are old, their births, labels, ranks and
    characteristic lengths are retained. The final-region identity
    permits the induction hypothesis at generation $j$.
    Finally, if $x$ is old and $y$ is fresh, then
    $\operatorname{rk}_{j+1}(x)\le j+1<j+2
        =\operatorname{rk}_{j+1}(y)$,
    contradicting the assumed rank inequality.
\end{proof}

If the marks at generation $N$ are empty,
Theorem~\ref{thm:area_law_repair_final_current_empty} identifies
$R_y^{(\infty,N)}$ with the current region $R_y^{(N)}$.
