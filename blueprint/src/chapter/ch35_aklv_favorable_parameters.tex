\section{Favorable scalar parameters for the AKLV filter}

\begin{lemma}[A uniform square-degree tradeoff]
    \label{thm:aklv_favorable_scalar_parameters}
    \lean{Real.exists_aklv_favorable_scalar_parameters}
    \leanok
    \uses{thm:chain_polynomial_filter_parameters}
    There is a universal real constant $A\ge21$ such that the following
    holds for every integer $q\ge2$ and every $0<g\le1$.
    Put
    \begin{align*}
        u                & =1+\log q+\log(1/g),                            &
        s                & =\left\lceil Au^2/g\right\rceil,                & \ell & =s^2,    \\
        t_0              & =1024+16\log(1/g),                              & R    & =2t_0+s, \\
        D                & =\bigl(q(s^2+1)\bigr)^{32(2s+1)},               &
        D_{\mathrm{aux}} & =(qs^2)^{22s},                                                    \\
        \Delta           & =4\exp\left(-4\ell\sqrt{\frac{9g}{10R}}\right).
    \end{align*}
    Then $Au^2/g\le s\le(A+1)u^2/g$, $s\ge21$, $s\ge\log(1/g)$,
    $\ell/s=s$, $\lfloor\sqrt\ell\rfloor=s$, and
    \begin{align*}
        0<R     & \le2081s,         & D & \le D_{\mathrm{aux}}^9, & \Delta & >0, \\
        D\Delta & \le10^{-5}<1/256,                                              \\
        \log D  & \le
        198(A+1)\bigl(6+2\log(A+1)\bigr)g^{-1}u^3.
    \end{align*}
    These are scalar estimates supporting the choice following
    \cite[Theorem~6.1]{Arad2013AreaLaw}. They assert neither that a finite
    chain contains the chosen window nor an entropy conclusion.
\end{lemma}
\begin{proof}
    \leanok
    Apply Lemma~\ref{thm:chain_polynomial_filter_parameters} with
    $C_H=2081/2$ and $M=0$. Its scale inequalities give $s\ge21$ and
    $s\ge\log(1/g)$, hence
    \begin{align*}
        R=2048+32\log(1/g)+s\le2048+33s\le2081s.
    \end{align*}
    Since $s\ge1$ and $q\ge2$,
    \begin{align*}
        q(s^2+1) & \le(qs^2)^2, & 64(2s+1) & \le198s.
    \end{align*}
    Thus $D\le D_{\mathrm{aux}}^9\le D_{\mathrm{aux}}^{12}$.
    Replacing $9g/10$ by $4g/5$ increases the exponential shrinking
    coefficient. The joint scalar tradeoff therefore gives
    $D\Delta\le10^{-5}$.
    Finally, $\log D\le9\log D_{\mathrm{aux}}$ gives the displayed
    logarithmic estimate.
\end{proof}


\begin{lemma}[Square-degree parameters for frustration-free clipping]
    \label{lem:aklv_frustration_free_scalar_parameters}
    \lean{Real.exists_aklv_frustrationFree_scalar_parameters}
    \notready
    \uses{thm:aklv_favorable_scalar_parameters}
    There is a universal $A\ge21$ such that the following holds for every
    integer $q\ge2$ and $0<g\le1$. Put $g_0=g/8192$,
    $u=1+\log q+\log(1/g)$, $u_0=1+\log q+\log(1/g_0)$, and
    \begin{align}
        s      & =\left\lceil Au_0^2/g_0\right\rceil,          &
        \ell   & =s^2,                                         & t & =1024/g, & R & =2t+s,\notag \\
        D      & =\bigl(q(s^2+1)\bigr)^{32(2s+1)},             &
        \Delta & =4\exp\left(-4\ell\sqrt{\frac{g}{2R}}\right).
        \label{eq:aklv_ff_scalar_choice}
    \end{align}
    Then $s\ge21$, $s\ge1/g$, $\ell/s=s$, $\lfloor\sqrt\ell\rfloor=s$,
    $R>0$, $\Delta>0$, and $D\Delta\le10^{-5}<1/256$.
    Moreover,
    \begin{align}
        s      & \le\frac{8192(A+1)(1+\log8192)^2}{g}u^2,\notag \\
        \log D & \le
        \frac{8192\cdot198(A+1)(6+2\log(A+1))(1+\log8192)^3}{g}u^3.
        \label{eq:aklv_ff_scalar_cost}
    \end{align}
    The logarithm of the inverse gap is retained. These are scalar
    parameters for the frustration-free clipping step in
    \cite[Section~4, frustration-free area law]{Arad2013AreaLaw}; neither an
    available physical window nor the printed sharper entropy rate is asserted.
\end{lemma}
\begin{proof}
    \notready
    \uses{thm:aklv_favorable_scalar_parameters}
    Apply the uniform square-degree estimate at $g_0=g/8192$ and use the
    same $s,\ell,D$. Write $R_0=2(1024+16\log(1/g_0))+s$ for its ceiling
    and $\Delta_0$ for its shrinking coefficient. The lower scale bound
    implies $s\ge1/g$. Hence
    \begin{align}
        R                  & \le2049s\le2049R_0, &
        \frac{9g_0}{10R_0} & \le\frac{g}{2R}.
        \notag
    \end{align}
    Therefore $\Delta\le\Delta_0$ and the same favorable tradeoff holds.
    Finally $u\ge1$ and
    $u_0=u+\log8192\le(1+\log8192)u$.
    Substitute this comparison and $g_0=g/8192$ into the existing upper
    scale and logarithmic-rank bounds to obtain
    (\ref{eq:aklv_ff_scalar_cost}).
\end{proof}
