\section{Choice of the patch energy parameter}
\label{sec:peps_patch_energy_choice}

The patch energy estimate has the form
$\langle\phi,H\phi\rangle-E_0\le C_E\kappa^2$, where
$C_E=36q^2J$ depends only on the fixed physical parameters.
The following scalar choice depends only on $q$, $J$ and $\Delta$, and is
made before the regions, radius and minimizing densities are chosen. It does
not establish the energy estimate itself.

\begin{definition}[The filter exponent]
    \label{def:peps_patch_filter_exponent}
    \lean{TNLean.PEPS.patchFilterExponent}
    \leanok
    For an integer $q\ge0$ and reals $J,\Delta$, put $C_E=36q^2J$ and
    \begin{align}
        \kappa(q,J,\Delta)=\min\left\{\frac12,
        \sqrt{\frac{\Delta}{1024C_E}}\right\},
        \label{eq:peps_patch_filter_exponent}
    \end{align}
    where the square root is read as zero when $C_E=0$.
\end{definition}

\begin{lemma}[A uniformly small energy parameter]
    \label{lem:peps_patch_energy_parameter_choice}
    \lean{TNLean.PEPS.patchFilterExponent_le_half,
        TNLean.PEPS.patchFilterExponent_pos,
        TNLean.PEPS.patchFilterExponent_pos_of_pos,
        TNLean.PEPS.mul_sq_le_div_1024_of_le_patchFilterExponent,
        TNLean.PEPS.mul_patchFilterExponent_sq_le}
    \leanok
    \uses{def:peps_patch_filter_exponent}
    Let $J\ge0$ and $\Delta\ge0$, and let $\kappa=\kappa(q,J,\Delta)$ be the
    filter exponent (\ref{eq:peps_patch_filter_exponent}). Then
    $\kappa\le1/2$, and every $0\le\kappa'\le\kappa$ satisfies
    \begin{align}
        C_E\kappa'^2\le\frac{\Delta}{1024}.\notag
    \end{align}
    If $C_E>0$ and $\Delta>0$, in particular if $q\ge1$ and $J,\Delta>0$, then
    $\kappa>0$.
\end{lemma}
\begin{proof}
    \leanok
    The first entry of the minimum gives $\kappa\le1/2$. If $C_E=0$, the
    energy bound reads $0\le\Delta/1024$. If $C_E>0$, then
    $\kappa'^2\le\kappa^2\le\Delta/(1024C_E)$ by the second entry. Both
    entries are positive when $C_E,\Delta>0$.
\end{proof}

For interactions of range $R$, an extension beyond the source, the same
choice is made with the coefficient $C_E(q,R,J)$ of
Definition~\ref{def:peps_range_patch_energy_constant} in place of $36q^2J$.

\begin{definition}[The filter exponent at range $R$]
    \label{def:peps_range_patch_filter_exponent}
    \lean{TNLean.PEPS.Approximation.rangePatchFilterExponent}
    \leanok
    \uses{def:peps_range_patch_energy_constant}
    For integers $q,R\ge0$ and reals $J,\Delta$, put
    \begin{align}
        \kappa_R(q,J,\Delta)=\min\left\{\frac12,
        \sqrt{\frac{\Delta}{1024C_E(q,R,J)}}\right\},
        \label{eq:peps_range_patch_filter_exponent}
    \end{align}
    where the square root is read as zero when $C_E(q,R,J)=0$.
\end{definition}

\begin{lemma}[A uniformly small energy parameter at range $R$]
    \label{lem:peps_range_patch_energy_parameter_choice}
    \lean{TNLean.PEPS.Approximation.rangePatchFilterExponent_le_half,
        TNLean.PEPS.Approximation.rangePatchFilterExponent_pos,
        TNLean.PEPS.Approximation.mul_sq_le_div_1024_of_le_rangePatchFilterExponent}
    \leanok
    \uses{def:peps_range_patch_filter_exponent,
        lem:peps_range_patch_energy_constant_sign}
    Let $J\ge0$ and $\Delta\ge0$. Then $\kappa_R\le1/2$, and every
    $0\le\kappa'\le\kappa_R$ satisfies $C_E(q,R,J)\kappa'^2\le\Delta/1024$.
    If $q,R\ge1$ and $J,\Delta>0$, then $\kappa_R>0$.
\end{lemma}
\begin{proof}
    \leanok
    As in Lemma~\ref{lem:peps_patch_energy_parameter_choice}, with
    $C_E(q,R,J)\ge0$ in place of $36q^2J$.
\end{proof}
