\section{Invertible physical comparison of G-injective tensors}%
\label{sec:peps_g_injective_physical_equivalence}

\begin{theorem}[Ambient physical comparison]%
    \label{thm:peps_g_injective_ambient_physical_comparison}
    \lean{TNLean.PEPS.IsGInjective.exists_physicalEquiv}
    \leanok
    \uses{def:peps_g_injective}
    Let a finite group $G$ act on a virtual space $\mathcal V$.
    Suppose $T,S:\mathcal V\to\mathcal P$ are $G$-injective for
    that same representation and their physical ranges are
    finite-dimensional. Then an invertible physical map
    $F:\mathcal P\to\mathcal P$ satisfies $FT=S$.
    This is a local consequence of
    \cite[Definition~5.1]{Schuch2010PEPS}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_g_injective_physical_range_equivalence}
    The physical range isomorphism sends $Tx$ to $Sx$ for every
    virtual vector $x$. Extend this isomorphism of finite-dimensional
    subspaces to an automorphism of $\mathcal P$ by choosing
    complementary subspaces. Its restriction has the required identity.
\end{proof}

\begin{theorem}[Comparison between physical spaces of equal dimension]%
    \label{thm:peps_g_injective_equal_dim_physical_comparison}
    \lean{TNLean.PEPS.IsGInjective.exists_physicalEquiv_of_finrank_eq}
    \leanok
    \uses{def:peps_g_injective}
    Let $T:\mathcal V\to\mathcal P$ and
    $S:\mathcal V\to\mathcal Q$ be $G$-injective for the same
    representation of a finite group. If $\mathcal P$ and
    $\mathcal Q$ are finite-dimensional and have equal dimensions,
    there is a linear isomorphism $F:\mathcal P\to\mathcal Q$
    satisfying $FT=S$.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:peps_g_injective_ambient_physical_comparison,
        thm:peps_g_injective_physical_map}
    Choose a linear isomorphism $E:\mathcal P\to\mathcal Q$.
    The map $E^{-1}S$ is $G$-injective. Apply the ambient comparison
    to $T$ and $E^{-1}S$ to obtain an automorphism $L$ of
    $\mathcal P$ with $LT=E^{-1}S$. Then $F=EL$ satisfies $FT=S$.
\end{proof}
