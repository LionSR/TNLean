\section{Group matrix product operators from a three-cocycle}%
\label{sec:asymex_group_cocycle_mpo}

Garre-Rubio, Lootens and Molnár~\cite{GarreRubio2023MPOSym} build, from a
finite group $G$ and a three-cocycle $\omega$, a matrix product operator
representation of $G$ on a periodic chain whose sites carry the regular
representation $\C[G]$. Each periodic operator acts as
$U_g=\bigl(\bigotimes_iL_g\bigr)\prod_iW_g^{i,i+1}$
(Theorem~\ref{thm:asymex_group_cocycle_kernel}), combining on-site left shifts
$L_g$ with diagonal two-site phase gates $W_g$ determined by $\omega$; its
place among matrix product operator symmetries, and the anomaly carried by
the class of $\omega$, are described in
Remark~\ref{rem:mposym_group_cocycle}. For cyclic groups,
Garre-Rubio and Schuch~\cite{GarreRubioSchuch2024DomainWall} give one explicit
cocycle in each cohomology class.

The periodic display of~\cite{GarreRubio2023MPOSym} places two left shifts
on each pair of neighboring sites, which read literally acts on every site
by the shift by $g^2$. The local tensor of that construction, and its
example for the group of order two, carry one left shift per site; the
operator below follows the tensor, as recorded
in~\cite{gap:glm23_pbc_group_mpo_single_shift}.

\begin{definition}[The tensor of a three-cochain]
    \label{def:asymex_group_cocycle_tensor}
    \lean{MPOTensor.GroupCocycle.wGate,
        MPOTensor.GroupCocycle.leftShift,
        MPOTensor.GroupCocycle.leftShift_apply,
        MPOTensor.GroupCocycle.siteShift,
        MPOTensor.GroupCocycle.siteShift_apply,
        MPOTensor.GroupCocycle.tensor,
        MPOTensor.GroupCocycle.tensor_apply,
        MPOTensor.GroupCocycle.family,
        MPOTensor.GroupCocycle.shift,
        MPOTensor.GroupCocycle.shift_apply,
        MPOTensor.GroupCocycle.phase}
    \leanok
    \uses{def:mpo_tensor, def:mpo_family, def:mpug_scalar_cochains}
    Let $G$ be a finite group and let
    $\omega\colon G^3\to\C^\times$ be a scalar three-cochain. On
    $\C[G]\otimes\C[G]$ define the diagonal gate and the left shift by
    \begin{align}
        W_g\ket{k,l} & =\omega(g,l,l^{-1}k)\ket{k,l}, &
        L_g\ket{k}   & =\ket{gk}.
        \notag
    \end{align}
    The tensor $\hat T_g$ has physical and bond indices in $G$, with
    \begin{align}
        (\hat T_g^{\,a,b})_{l,r}
         & =\delta_{a,gb}\,\delta_{r,b}\,\omega(g,b,b^{-1}l),
        \label{eq:asymex_group_cocycle_tensor}
    \end{align}
    where $a$ is the output, $b$ the input, and $l,r$ the incoming and
    outgoing bonds. For a configuration $t=(t_0,\ldots,t_{N-1})\in G^N$
    on a periodic chain of $N>0$ sites, with $t_N=t_0$, write
    $gt=(gt_0,\ldots,gt_{N-1})$ and
    \begin{align}
        \phi_g(t)
         & =\prod_{i=0}^{N-1}\omega(g,t_{i+1},t_{i+1}^{-1}t_i).
        \label{eq:asymex_group_cocycle_phase}
    \end{align}
    The group elements are identified with the basis labels
    $\{0,\ldots,|G|-1\}$ by a fixed bijection, and $U_g=O_N(\hat T_g)$
    denotes the periodic operator.
\end{definition}

Following~\cite{GarreRubio2023MPOSym}, write
$\omega_g(l,b)=\omega(g,b,b^{-1}l)$ for the diagonal entries of $W_g$.
By~(\ref{eq:asymex_group_cocycle_tensor}), the tensor $\hat T_g$ copies its
input $b$ to the outgoing bond, weights the pair formed by the incoming
bond and the input by $\omega_g$, and applies $L_g$ to the input to give
the output.

The boundary is the ordinary trace, with no additional boundary matrix.
The monomial kernel is given in
Theorem~\ref{thm:asymex_group_cocycle_kernel}. The cocycle equation implies
the multiplication law (Theorem~\ref{thm:asymex_group_cocycle_mul}), while
unit-modulus values imply unitarity
(Theorem~\ref{thm:asymex_group_cocycle_unitary}). For the chosen fusion
tensors, the associator is $\omega^{-1}$
(Theorem~\ref{thm:asymex_group_cocycle_inverse_cocycle}). This does not
assert that all the tensors are normal: for a nontrivial group and a
normalized cocycle, the identity tensor is not normal
(Theorem~\ref{thm:asymex_group_cocycle_identity_not_normal}).

\begin{theorem}[The periodic operator is monomial]
    \label{thm:asymex_group_cocycle_kernel}
    \lean{MPOTensor.GroupCocycle.mpo_tensor_apply,
        MPOTensor.GroupCocycle.mpo_tensor,
        MPOTensor.GroupCocycle.mpo_tensor_apply_eq_leftShift_wGate}
    \leanok
    \uses{def:asymex_group_cocycle_tensor}
    For every scalar three-cochain $\omega$, every $g\in G$, every $N>0$,
    and all $s,t\in G^N$,
    \begin{align}
        U_g(s,t) & =\delta_{s,gt}\,\phi_g(t).
        \notag
    \end{align}
    Equivalently,
    $U_g=\bigl(\bigotimes_iL_g\bigr)\prod_iW_g^{i,i+1}$, the gates acting on
    the neighboring pairs $(t_i,t_{i+1})$ before the shifts.
\end{theorem}

\begin{proof}
    \leanok
    Expand the trace over cyclic bond configurations $b$. The factor
    $\delta_{r,b}$ in~(\ref{eq:asymex_group_cocycle_tensor}) forces
    $b_{i+1}=t_i$, so only $b_i=t_{i-1}$ contributes. Its contribution
    vanishes unless $s_i=gt_i$ at every site, and otherwise equals
    $\prod_i\omega(g,t_i,t_i^{-1}t_{i-1})$, which is $\phi_g(t)$ after the
    cyclic reindexing $i\mapsto i+1$. The gates are diagonal, so the entry
    of $\bigl(\bigotimes_iL_g\bigr)\prod_iW_g^{i,i+1}$ at $(s,t)$ is the
    same product.
\end{proof}

The product law is the following identity of periodic operators, with
$U_g$ applied after $U_h$.

\begin{theorem}[Product law]
    \label{thm:asymex_group_cocycle_mul}
    \lean{MPOTensor.GroupCocycle.mpo_tensor_mul,
        MPOTensor.GroupCocycle.phase_mul,
        MPOTensor.GroupCocycle.shift_mul}
    \leanok
    \uses{thm:asymex_group_cocycle_kernel, def:mpug_three_cocycle}
    If $\omega$ is a three-cocycle, then $U_gU_h=U_{gh}$ for all
    $g,h\in G$ and every $N>0$.
\end{theorem}

\begin{proof}
    \leanok
    Both sides are monomial with the shift by $gh$, so it suffices that
    $\phi_g(ht)\phi_h(t)=\phi_{gh}(t)$. The cocycle
    equation~(\ref{eq:mpug_three_cocycle}) at $(g,h,a,a^{-1}x)$ gives, at
    each site,
    \begin{align}
        \omega(g,ha,a^{-1}x)\,\omega(h,a,a^{-1}x)
         & =\omega(gh,a,a^{-1}x)\,
        \frac{\omega(g,h,x)}{\omega(g,h,a)}.
        \notag
    \end{align}
    With $a=t_{i+1}$ and $x=t_i$ the correction factors telescope around
    the ring.
\end{proof}

\begin{theorem}[Unit law]
    \label{thm:asymex_group_cocycle_one}
    \lean{MPOTensor.GroupCocycle.mpo_tensor_one,
        MPOTensor.GroupCocycle.wGate_one,
        MPOTensor.GroupCocycle.shift_one}
    \leanok
    \uses{thm:asymex_group_cocycle_kernel, def:mpug_scalar_cochains}
    If $\omega$ is normalized, then $W_e=1$ and $U_e=1$ for every $N>0$.
\end{theorem}

\begin{proof}
    \leanok
    Normalization in the first argument makes every factor of $W_e$ and of
    $\phi_e(t)$ equal to one, and the shift by $e$ is the identity.
\end{proof}

\begin{theorem}[Unitarity and the adjoint law]
    \label{thm:asymex_group_cocycle_unitary}
    \lean{MPOTensor.GroupCocycle.mpo_tensor_mem_unitaryGroup,
        MPOTensor.GroupCocycle.star_phase_mul_phase,
        MPOTensor.GroupCocycle.isMPUPos,
        MPOTensor.GroupCocycle.mpo_tensor_conjTranspose,
        MPOTensor.GroupCocycle.family_operator_laws}
    \leanok
    \uses{thm:asymex_group_cocycle_kernel, thm:asymex_group_cocycle_mul,
        thm:asymex_group_cocycle_one, def:mpug_positive_mpu}
    If every value of $\omega$ has modulus one, then $U_g$ is unitary for
    every $g\in G$ and every $N>0$, so $\hat T_g$ is a matrix product
    unitary on nonempty chains. If moreover $\omega$ is a normalized
    three-cocycle, then $U_g^\dagger=U_{g^{-1}}$ for every $N>0$.
\end{theorem}

\begin{proof}
    \leanok
    A monomial matrix whose nonzero entries have modulus one is unitary.
    For the adjoint law, $U_g^\dagger=U_g^\dagger U_gU_{g^{-1}}=U_{g^{-1}}$
    by Theorems~\ref{thm:asymex_group_cocycle_mul}
    and~\ref{thm:asymex_group_cocycle_one}.
\end{proof}

The single-site tensors are not injective: their $|G|^2$ physical matrices
span a space of dimension at most $|G|$. The operator laws above are
therefore stated directly rather than through an injective representation.

\subsection{Fusion tensors and the inverse cocycle}

The source equips the tensors $\hat T_g$ with explicit fusion tensors and
states that they realize the inverse cocycle~\cite{GarreRubio2023MPOSym}.
The bond of $\hat T_g$ carries the input label of the preceding site, so
along the stacked product $\hat T_g\hat T_h$ the bond pair over a label
$l$ of $\hat T_{gh}$ is $(hl,l)$.

\begin{definition}[Fusion tensors of the cocycle construction]
    \label{def:asymex_group_cocycle_fusion_tensors}
    \lean{MPOTensor.GroupCocycle.pairLabel,
        MPOTensor.GroupCocycle.fusionV, MPOTensor.GroupCocycle.fusionW,
        MPOTensor.GroupCocycle.fusionData}
    \leanok
    \uses{def:asymex_group_cocycle_tensor, def:mpug_anomaly_three_cochain}
    For $g,h\in G$ let $F^<_{g,h}$ send the bond pair $(hl,l)$ of
    $\hat T_g\hat T_h$ to the bond label $l$ of $\hat T_{gh}$ with the
    factor $\omega(g,h,l)^{-1}$, and vanish on every other bond pair, and
    let $F^>_{g,h}$ send $l$ to $(hl,l)$ with the factor $\omega(g,h,l)$.
\end{definition}

\begin{theorem}[The fusion tensors reduce the product]
    \label{thm:asymex_group_cocycle_fusion}
    \lean{MPOTensor.GroupCocycle.pairLabel_injective,
        MPOTensor.GroupCocycle.fusionV_mul_apply,
        MPOTensor.GroupCocycle.mul_fusionV_apply,
        MPOTensor.GroupCocycle.fusionV_mul_mulTensor,
        MPOTensor.GroupCocycle.fusionV_mul_fusionW,
        MPOTensor.GroupCocycle.isReduction_fusion}
    \leanok
    \uses{def:asymex_group_cocycle_fusion_tensors, def:mps_rectangular_reduction,
        def:mpdo_operator_product, thm:asymex_group_cocycle_mul}
    If $\omega$ is a three-cocycle, then $F^<_{g,h}F^>_{g,h}=\mathbf 1$ and
    $F^<_{g,h}\,(\hat T_g\hat T_h)^{ij}=\hat T_{gh}^{ij}\,F^<_{g,h}$ for every
    pair of physical indices. Hence
    $F^<_{g,h}\,(\hat T_g\hat T_h)^{\mathbf w}\,F^>_{g,h}=\hat T_{gh}^{\mathbf w}$
    for every word $\mathbf w$, including the empty word.
\end{theorem}

\begin{proof}
    \leanok
    Both sides of the letter identity vanish unless the output bond pair is
    $(hj,j)$ and the ket index is $ghj$. On that support the left side is
    $\omega(g,h,l)^{-1}\omega(g,hj,j^{-1}l)\,\omega(h,j,j^{-1}l)$ and the right
    side is $\omega(gh,j,j^{-1}l)\,\omega(g,h,j)^{-1}$, which agree by the
    sitewise cocycle identity in the proof of
    Theorem~\ref{thm:asymex_group_cocycle_mul}. The word identity follows by
    moving $F^<_{g,h}$ through the word and using $F^<_{g,h}F^>_{g,h}=\mathbf 1$.
\end{proof}

\begin{theorem}[The construction carries the inverse cocycle]
    \label{thm:asymex_group_cocycle_inverse_cocycle}
    \lean{MPOTensor.GroupCocycle.fusionV_mul_kronId,
        MPOTensor.GroupCocycle.castMat_family_mul,
        MPOTensor.GroupCocycle.fusionData_leftV,
        MPOTensor.GroupCocycle.fusionData_isAssociator,
        MPOTensor.GroupCocycle.exists_evalWord_tensor_ne_zero,
        MPOTensor.GroupCocycle.fusionData_omega}
    \leanok
    \uses{thm:asymex_group_cocycle_fusion, def:mpug_anomaly_three_cochain,
        thm:asymex_group_cocycle_kernel}
    Let $\omega$ be a three-cocycle. For the fusion tensors of
    Definition~\ref{def:asymex_group_cocycle_fusion_tensors}, the left
    boundaries of the two fusion trees of $(g,h,k)$ satisfy
    $V^L=\omega(g,h,k)^{-1}\,V^R$, and the anomaly three-cochain of these
    fusion tensors is $\omega^{-1}$.
\end{theorem}

\begin{proof}
    \leanok
    Both boundaries send the triple bond label $(hkx,kx,x)$ to $x$ and vanish
    elsewhere. Their factors are
    $\omega(gh,k,x)^{-1}\omega(g,h,kx)^{-1}$ and
    $\omega(g,hk,x)^{-1}\omega(h,k,x)^{-1}$, whose ratio is $\omega(g,h,k)^{-1}$
    by the cocycle equation at $(g,h,k,x)$. The scalar is unique because
    every $\hat T_g$ has nonzero words of every length, its periodic operator
    being a shift with nonzero phases.
\end{proof}

\begin{theorem}[The identity member is not normal]
    \label{thm:asymex_group_cocycle_identity_not_normal}
    \lean{MPOTensor.GroupCocycle.tensor_one_apply_row,
        MPOTensor.GroupCocycle.evalWord_tensor_one_apply_row,
        MPOTensor.GroupCocycle.not_isNormal_tensor_one,
        MPOTensor.GroupCocycle.not_isNormalRepresentation}
    \leanok
    \uses{def:asymex_group_cocycle_tensor, def:normal,
        def:mpug_anomaly_three_cochain}
    If $G$ is nontrivial and $\omega$ is normalized, then $\hat T_e$ is not
    normal. Hence the tensors $\hat T_g$ do not form a representation with
    normal tensors, and the three-cochain of
    Theorem~\ref{thm:asymex_group_cocycle_inverse_cocycle} is that of the
    fusion tensors of Definition~\ref{def:asymex_group_cocycle_fusion_tensors}.
\end{theorem}

\begin{proof}
    \leanok
    Every letter of $\hat T_e$ has all rows equal, so every nonempty product
    of letters does too, while the matrix unit at a diagonal position has
    distinct rows.
\end{proof}

\subsection{The group of order two and the decorated CZX operator}

For $G=\Z_2=\{e,s\}$ with the nontrivial cocycle $\omega_1$ of
Definition~\ref{def:asymex_z2_cocycles}, the source identifies the gate as
$CZ(1\otimes Z)$, the shift as $X$, and the operator as
$\prod_iCZ_{i,i+1}Z_i\prod_iX_i$~\cite{GarreRubio2023MPOSym}. Identify
$e,s$ with the qubit labels $0,1$ and write $\bar t$ for the bitwise
complement of $t\in\{0,1\}^N$ and $q(s)=\sum_is_is_{i+1}$.

\begin{theorem}[The decorated CZX operator]
    \label{thm:asymex_group_cocycle_z2}
    \lean{MPOTensor.GroupCocycle.bitEquiv,
        MPOTensor.GroupCocycle.czxDecorated,
        MPOTensor.GroupCocycle.wGate_cyclicTwo,
        MPOTensor.GroupCocycle.mpo_tensor_cyclicTwo}
    \leanok
    \uses{def:asymex_z2_cocycles, thm:asymex_group_cocycle_kernel}
    For $\omega=\omega_1$ and $g=s$, the gate is
    $W_s\ket{k,l}=(-1)^{kl+l}\ket{k,l}$, which is $CZ(1\otimes Z)$. For
    every $N>0$,
    \begin{align}
        U_s(u,t) & =\delta_{u,\bar t}\,(-1)^{q(u)+\sum_iu_i},
        \notag
    \end{align}
    so that $U_s=\prod_iCZ_{i,i+1}Z_i\prod_iX_i$.
\end{theorem}

\begin{proof}
    \leanok
    On bits, $\omega_1(s,l,l^{-1}k)=(-1)^{kl+l}$, checked on the four
    pairs $(k,l)$. By Theorem~\ref{thm:asymex_group_cocycle_kernel} the
    kernel is supported on $u=\bar t$ with phase
    $\prod_i(-1)^{t_it_{i+1}+t_{i+1}}$. On complemented bits,
    $(-1)^{kl+l}=(-1)^{\bar k\bar l+\bar k}$, and the product becomes
    $(-1)^{q(u)+\sum_iu_i}$ after reindexing the second factor.
\end{proof}

\subsubsection{Hamiltonians symmetric under the CZX operators}

Write $X_j$ and $Z_j$ for the Pauli operators on qubit $j$ of a periodic
chain of $N\geq2$ qubits, with sites taken modulo $N$, and
$K_j=Z_{j-1}X_jZ_{j+1}$. The source records the Hamiltonian
$H_{\mathrm R}(\mu)=\sum_iCZ_{i,i+2}X_{i+1}-\mu\sum_iZ_iZ_{i+1}$ of Roose
et al.\ as invariant under the operator
$U=\prod_iCZ_{i,i+1}Z_i\prod_iX_i$ of
Theorem~\ref{thm:asymex_group_cocycle_z2}~\cite{GarreRubio2023MPOSym}. The
review states that the cluster Hamiltonian in a critical transverse field
commutes with the operator of the nontrivial three-cocycle, which it writes,
as the virtual symmetry of the CZX state, as
$O_Z=\prod_iCZ_{i,i+1}\prod_iX_i$, without the factors
$Z_i$~\cite{Cirac2021Matrix}. Let
$H(\lambda)=-\sum_jK_j-\lambda\sum_jX_j$ be the cluster Ising Hamiltonian.

\begin{definition}[Pauli terms and the two Hamiltonians]%
    \label{def:asymex_group_cocycle_z2_hamiltonians}
    \lean{MPOTensor.GroupCocycle.flipAt, MPOTensor.GroupCocycle.flipAt_apply,
        MPOTensor.GroupCocycle.spinFlip_mul_flipAt, MPOTensor.GroupCocycle.pauliXAt,
        MPOTensor.GroupCocycle.clusterTerm, MPOTensor.GroupCocycle.czXTerm,
        MPOTensor.GroupCocycle.zzTerm, MPOTensor.GroupCocycle.clusterIsingHamiltonian,
        MPOTensor.GroupCocycle.rooseHamiltonian}
    \leanok
    The operators $X_j$, $K_j$, $CZ_{i,i+2}X_{i+1}$ and $Z_iZ_{i+1}$ are
    monomial in the computational basis: the first three flip one bit, with
    the phases $1$, $(-1)^{t_{j-1}+t_{j+1}}$ and $(-1)^{t_it_{i+2}}$ on the
    input $t$, and the last is diagonal with the phase
    $(-1)^{t_i+t_{i+1}}$. The Hamiltonians are $H(\lambda)$ and
    $H_{\mathrm R}(\mu)$ above.
\end{definition}

\begin{theorem}[Hamiltonians invariant under the CZX operators]%
    \label{thm:asymex_group_cocycle_z2_hamiltonians}
    \lean{MPOTensor.GroupCocycle.mpo_czxTensor_mul_pauliXAt,
        MPOTensor.GroupCocycle.mpo_czxTensor_mul_clusterTerm,
        MPOTensor.GroupCocycle.mpo_czxTensor_conj_clusterIsingHamiltonian,
        MPOTensor.GroupCocycle.mpo_czxTensor_conj_clusterIsingHamiltonian_one,
        MPOTensor.GroupCocycle.czxDecorated_mul_pauliXAt,
        MPOTensor.GroupCocycle.czxDecorated_mul_clusterTerm,
        MPOTensor.GroupCocycle.czxDecorated_phase_flipAt,
        MPOTensor.GroupCocycle.czxDecorated_mul_czXTerm,
        MPOTensor.GroupCocycle.czxDecorated_mul_zzTerm,
        MPOTensor.GroupCocycle.czxDecorated_mul_rooseHamiltonian,
        MPOTensor.GroupCocycle.czxDecorated_mul_clusterIsingHamiltonian,
        MPOTensor.GroupCocycle.czxDecorated_conj_clusterIsingHamiltonian,
        MPOTensor.GroupCocycle.czxDecorated_conj_clusterIsingHamiltonian_eq_smul,
        MPOTensor.GroupCocycle.czxDecorated_conj_clusterIsingHamiltonian_neg_one,
        MPOTensor.GroupCocycle.czxDecorated_conj_clusterIsingHamiltonian_one}
    \leanok
    \uses{thm:asymex_group_cocycle_z2, def:asymex_czx_tensors,
        def:asymex_group_cocycle_z2_hamiltonians}
    On a periodic chain of $N\geq2$ qubits the following hold.
    \begin{enumerate}
        \item $O_ZX_jO_Z^\dagger=K_j$ and $O_ZK_jO_Z^\dagger=X_j$, so
              $O_ZH(\lambda)O_Z^\dagger=-\sum_jX_j-\lambda\sum_jK_j$, and
              $O_Z$ commutes with $H(1)$.
        \item $UX_jU^\dagger=-K_j$ and $UK_jU^\dagger=-X_j$, so
              $UH(\lambda)U^\dagger=\lambda\sum_jK_j+\sum_jX_j$, which is
              $-\lambda H(\lambda^{-1})$ for $\lambda\neq0$. In particular $U$
              commutes with $H(-1)$, while $UH(1)U^\dagger=-H(1)$.
        \item $U$ commutes with $H_{\mathrm R}(\mu)$ for every $\mu$.
    \end{enumerate}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_czx_kernel, thm:asymex_czx_unitarity,
        thm:mpug_monomial_matrix_laws}
    All operators are monomial in the computational basis. The periodic
    operator of the undecorated CZX tensor is $O_Z$ up to the global sign
    $(-1)^N$ of its gate order~\cite{gap:mpu_czx_tensor_normalization},
    which cancels in every conjugation. The operator
    $O_Z$ flips every bit of the input $t$ and multiplies by $(-1)^{q(t)}$,
    and $U$ multiplies instead by $(-1)^{q(\bar t)+\sum_i\bar t_i}$.
    Flipping bit $j$ of $t$ changes $q(t)$ by $t_{j-1}+t_{j+1}$ and
    $\sum_i\bar t_i$ by one, modulo two. Since $X_j$, $K_j$ and
    $CZ_{i,i+2}X_{i+1}$ flip one bit and $O_Z$, $U$ flip all bits, the two
    products in each relation have the same permutation, and the phases
    agree by this parity count together with
    $(-1)^{\bar a+\bar b}=(-1)^{a+b}$ and
    $(-1)^{\bar a\bar b}=-(-1)^{a+b+ab}$ for bits $a,b$. The operators
    $Z_iZ_{i+1}$ are diagonal with a phase invariant under flipping all bits.
    Summing the relations over the sites and using unitarity gives the
    statements about the Hamiltonians.
\end{proof}

For $\lambda\neq0$ the conjugate in item~1 is
$-\sum_jX_j-\lambda\sum_jK_j=\lambda H(\lambda^{-1})$, so $O_Z$ exchanges
$H(\lambda)$ with a multiple of $H(\lambda^{-1})$.

The review prints no formula for the cluster Hamiltonian in a critical field.
The factors $Z_i$ of $U$ reverse the sign of the transverse field at the
symmetric point: $O_Z$ fixes $H(1)$ and $U$ fixes $H(-1)$.

\subsection{The cyclic groups}

For $n\geq1$ and $j\in\N$, Garre-Rubio and
Schuch~\cite[Section~IV.D]{GarreRubioSchuch2024DomainWall} write the
three-cocycles of $\Z_n$ as $\omega_j$ below, with $j=0,\ldots,n-1$
labelling the cohomology classes.

\begin{definition}[The cocycles of the cyclic groups]
    \label{def:asymex_group_cocycle_cyclic}
    \lean{TNLean.Algebra.ScalarThreeCochain.cyclicCocycle,
        TNLean.Algebra.ScalarThreeCochain.rootOfUnity,
        TNLean.Algebra.ScalarThreeCochain.rootOfUnity_pow,
        MPOTensor.GroupCocycle.residueEquiv}
    \leanok
    \uses{def:mpug_scalar_cochains}
    Represent residues of $\Z_n$ by $\{0,\ldots,n-1\}$ and write
    $[b+c]$ for the representative of $b+c$. Define
    \begin{align}
        \omega_j(a,b,c)
         & =\exp\Bigl\{\frac{2\pi\mathrm{i}\,j}{n^2}\,a\bigl(b+c-[b+c]\bigr)\Bigr\},
        \notag
    \end{align}
    and let $\zeta=e^{2\pi\mathrm{i}/n}$.
\end{definition}

\begin{theorem}[The cyclic instances]
    \label{thm:asymex_group_cocycle_cyclic}
    \lean{TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_val,
        TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_isCocycle,
        TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_isNormalized,
        TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_norm,
        MPOTensor.GroupCocycle.mpo_tensor_cyclic_apply,
        MPOTensor.GroupCocycle.cyclic_operator_laws}
    \leanok
    \uses{def:asymex_group_cocycle_cyclic, def:mpug_three_cocycle,
        thm:asymex_group_cocycle_kernel, thm:asymex_group_cocycle_mul,
        thm:asymex_group_cocycle_one, thm:asymex_group_cocycle_unitary}
    For every $n\geq1$ and $j\in\N$, the value is
    $\omega_j(a,b,c)=\zeta^{ja\lfloor(b+c)/n\rfloor}$, and $\omega_j$ is a
    normalized three-cocycle of $\Z_n$ with values of modulus one. For every
    $g\in\Z_n$, every $N>0$ and all $s,t\in\Z_n^N$,
    \begin{align}
        U_g(s,t)
         & =\delta_{s,g+t}\prod_{i=0}^{N-1}
        \zeta^{jg\lfloor(t_{i+1}+[t_i-t_{i+1}])/n\rfloor}.
        \notag
    \end{align}
    Every $\hat T_g$ is a matrix product unitary on nonempty chains, and
    $U_0=1$, $U_gU_h=U_{g+h}$ and $U_g^\dagger=U_{-g}$ for every $N>0$.
\end{theorem}

\begin{proof}
    \leanok
    Since $b+c-[b+c]=n\lfloor(b+c)/n\rfloor$, the exponential is the stated
    power of $\zeta$. The cocycle equation reduces modulo $n$ to the carry
    identity
    \begin{align}
        \Bigl\lfloor\frac{k+l}{n}\Bigr\rfloor
        +\Bigl\lfloor\frac{h+[k+l]}{n}\Bigr\rfloor
         & =\Bigl\lfloor\frac{h+k}{n}\Bigr\rfloor
        +\Bigl\lfloor\frac{[h+k]+l}{n}\Bigr\rfloor,
        \notag
    \end{align}
    both sides counting the multiples of $n$ in $h+k+l$. An identity
    argument makes the exponent zero, and every power of $\zeta$ has
    modulus one. The kernel is
    Theorem~\ref{thm:asymex_group_cocycle_kernel} evaluated on these
    values, and the operator laws are
    Theorems~\ref{thm:asymex_group_cocycle_mul},
    \ref{thm:asymex_group_cocycle_one}
    and~\ref{thm:asymex_group_cocycle_unitary}.
\end{proof}

Garre-Rubio and Schuch~\cite[Section~IV.C]{GarreRubioSchuch2024DomainWall}
show that interchanging the domain walls created by a symmetry-breaking
element $g$ of order $o(g)$ multiplies the state by the phase
$\prod_{k=1}^{o(g)}\omega^{-1}(g,g^k,g)$. Since $g^{o(g)}=e$ this phase is
the inverse of the cyclic invariant $I_{g,o(g)}(\omega)$ of
Definition~\ref{def:mpug_cyclic_invariant}, so it depends only on the
class of $\omega$.

\begin{definition}[The domain-wall phase]
    \label{def:asymex_group_cocycle_domain_wall_phase}
    \lean{TNLean.Algebra.ScalarThreeCochain.domainWallPhase,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_eq_inv_cyclicInvariant}
    \leanok
    \uses{def:mpug_cyclic_invariant}
    For a scalar three-cochain $\omega$ of a group $G$ and $g\in G$ of order
    $o(g)$, set
    \begin{align}
        P_g(\omega) & =\prod_{k=1}^{o(g)}\omega(g,g^k,g)^{-1}
        =I_{g,o(g)}(\omega)^{-1}.
        \notag
    \end{align}
\end{definition}

\begin{theorem}[Gauge invariance of the domain-wall phase]
    \label{thm:asymex_group_cocycle_domain_wall_phase_gauge}
    \lean{TNLean.Algebra.ScalarThreeCochain.CohomologousTo.domainWallPhase_eq}
    \leanok
    \uses{def:asymex_group_cocycle_domain_wall_phase}
    Cohomologous three-cochains have the same phase $P_g$ for every $g\in G$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:mpug_cyclic_invariant_gauge}
    Since $P_g(\omega)=I_{g,o(g)}(\omega)^{-1}$, this is the gauge invariance
    of the cyclic invariant in
    Theorem~\ref{thm:mpug_cyclic_invariant_gauge}.
\end{proof}

\begin{theorem}[Domain-wall phases of the cyclic cocycles]
    \label{thm:asymex_group_cocycle_cyclic_domain_wall}
    \lean{TNLean.Algebra.ScalarThreeCochain.cyclicInvariant_cyclicCocycle,
        TNLean.Algebra.ScalarThreeCochain.val_toAdd_pow,
        TNLean.Algebra.ScalarThreeCochain.sum_carry,
        TNLean.Algebra.ScalarThreeCochain.orderOf_eq_div_gcd,
        TNLean.Algebra.ScalarThreeCochain.orderOf_mul_val,
        TNLean.Algebra.ScalarThreeCochain.cyclicInvariant_cyclicCocycle_one,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_cyclicCocycle_eq_exp,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_cyclicCocycle_eq_exp_sum,
        TNLean.Algebra.ScalarThreeCochain.sum_carry_orderOf,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_cyclicCocycle,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_cyclicCocycle_three,
        TNLean.Algebra.ScalarThreeCochain.domainWallPhase_cyclicCocycle_four,
        TNLean.Algebra.ScalarThreeCochain.not_cohomologousTo_cyclicCocycle}
    \leanok
    \uses{def:asymex_group_cocycle_domain_wall_phase,
        thm:asymex_group_cocycle_cyclic}
    Let $n\geq1$, $j\in\N$ and $a\in\{0,\ldots,n-1\}$, with order
    $o(a)=n/\gcd(n,a)$ in $\Z_n$. For every $m\in\N$,
    $I_{a,m}(\omega_j)=\zeta^{ja\lfloor ma/n\rfloor}$. The domain-wall phase is
    \begin{align}
        P_a(\omega_j)
         & =\exp\Bigl\{-\frac{2\pi\mathrm{i}\,j}{n^2}\,a
        \sum_{k=0}^{o(a)-1}\bigl([ka]+a-[ka+a]\bigr)\Bigr\}
        =\exp\Bigl\{-\frac{2\pi\mathrm{i}\,j\,a^2}{n\gcd(n,a)}\Bigr\},
        \notag
    \end{align}
    the carry sum being equal to $o(a)\,a$. For $n=3$ the phases are
    $e^{-2\pi\mathrm{i}/3}$ for $j=1$ and $e^{2\pi\mathrm{i}/3}$ for $j=2$, at
    both $a=1$ and $a=2$. For $n=4$ and $a=1,2,3$ they are
    $-\mathrm{i},-1,-\mathrm{i}$ for $j=1$, then $-1,1,-1$ for $j=2$, and
    $\mathrm{i},-1,\mathrm{i}$ for $j=3$. Both tables agree with those printed
    in~\cite[Section~IV.D]{GarreRubioSchuch2024DomainWall}. If $n>1$ and
    $j\not\equiv j'\pmod n$, then $\omega_j$ and $\omega_{j'}$ are not
    cohomologous.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:mpug_cyclic_invariant_gauge, thm:asymex_group_cocycle_cyclic}
    By Theorem~\ref{thm:asymex_group_cocycle_cyclic},
    $\omega_j(a,[ka],a)=\zeta^{ja\lfloor([ka]+a)/n\rfloor}$. Writing
    $[ka]+a=[(k+1)a]+n\lfloor([ka]+a)/n\rfloor$ and summing over $k<m$, the
    carries add up to $\lfloor ma/n\rfloor$, which gives $I_{a,m}(\omega_j)$.
    At $m=o(a)$ one has $o(a)\,a=n\,(a/\gcd(n,a))$, which gives the carry sum
    and the closed form. The tables follow by reducing the exponent modulo
    $n$, with $\zeta=\mathrm{i}$ for $n=4$. At the generator $a=1$ and $m=n$
    the cyclic invariant of $\omega_j$ is $\zeta^j$, and $\zeta$ is a
    primitive $n$-th root of unity, so the gauge invariance of
    Theorem~\ref{thm:mpug_cyclic_invariant_gauge} separates the classes.
\end{proof}

\begin{theorem}[The classes of the cyclic cocycles]
    \label{thm:asymex_group_cocycle_cyclic_classes}
    \lean{TNLean.Algebra.ScalarThreeCochain.cyclicCocycleSubtype,
        TNLean.Algebra.ScalarThreeCochain.isPrimitiveRoot_rootOfUnity,
        TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_eq_of_modEq,
        TNLean.Algebra.ScalarThreeCochain.cyclicCocycle_zero,
        TNLean.Algebra.ScalarThreeCochain.isTrivialGaugeClass_cyclicCocycle_iff,
        TNLean.Algebra.ScalarThreeCochain.anomalyClass_cyclicCocycle_eq_zero_iff,
        TNLean.Algebra.ScalarThreeCochain.anomalyClass_cyclicCocycle_eq_iff,
        TNLean.Algebra.ScalarThreeCochain.mem_zpowers_ofAdd_one,
        TNLean.Algebra.ScalarThreeCochain.norm_scalarH2Representation_apply,
        TNLean.Algebra.ScalarThreeCochain.applyAsHom_ofAdd_one_sub_id,
        TNLean.Algebra.ScalarThreeCochain.normKerEquivRootsOfUnity,
        TNLean.Algebra.ScalarThreeCochain.nat_card_groupCohomology_three_cyclic,
        TNLean.Algebra.ScalarThreeCochain.exists_anomalyClass_cyclicCocycle_eq,
        TNLean.Algebra.ScalarThreeCochain.exists_cohomologousTo_cyclicCocycle,
        TNLean.Algebra.ScalarThreeCochain.ofAdd_one_pow_self,
        TNLean.Algebra.ScalarThreeCochain.cohomologousTo_of_cyclicInvariant_eq,
        TNLean.Algebra.ScalarThreeCochain.orderOf_ofAdd_one,
        TNLean.Algebra.ScalarThreeCochain.cohomologousTo_iff_domainWallPhase_eq,
        TNLean.Algebra.ScalarThreeCochain.anomalyClass_eq_iff_domainWallPhase_eq}
    \leanok
    \uses{thm:asymex_group_cocycle_cyclic_domain_wall,
        thm:mpug_h3_group_cohomology}
    Let $n>1$. The group $H^3(\Z_n,\C^\times_{\mathrm{triv}})$ has $n$
    elements and consists of the classes $[\omega_0],\ldots,[\omega_{n-1}]$,
    with $[\omega_j]=[\omega_{j'}]$ exactly when $j\equiv j'\pmod n$. In
    particular $[\omega_j]=0$ exactly when $n\mid j$, and every
    three-cocycle of $\Z_n$ is cohomologous to some $\omega_j$ with
    $0\leq j<n$. Two three-cocycles $\omega$ and $\eta$ of
    $\Z_n$ are cohomologous exactly when $P_1(\omega)=P_1(\eta)$, the
    domain-wall phases at the generator $1$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:asymex_group_cocycle_cyclic_domain_wall,
        thm:mpug_h3_group_cohomology,
        thm:asymex_group_cocycle_domain_wall_phase_gauge}
    The value $\omega_j(a,b,c)=\zeta^{ja\lfloor(b+c)/n\rfloor}$ depends on $j$
    only modulo $n$, and $\omega_0=1$. By
    Theorem~\ref{thm:asymex_group_cocycle_cyclic_domain_wall} the classes of
    $\omega_j$ and $\omega_{j'}$ differ when $j\not\equiv j'\pmod n$, so
    $[\omega_0],\ldots,[\omega_{n-1}]$ are $n$ distinct classes. The
    periodic resolution of the finite cyclic group $\Z_n$ identifies
    $H^3(\Z_n,A)$ with $\ker N/\operatorname{im}(\rho(1)-1)$, where
    $N=\sum_{g\in\Z_n}\rho(g)$. For the trivial action on $\C^\times$ one has
    $\rho(1)-1=0$ and $N(z)=z^n$, so $H^3(\Z_n,\C^\times_{\mathrm{triv}})$ is
    the group of $n$-th roots of unity and has $n$ elements. Hence the $n$
    classes of the $\omega_j$ exhaust it. If $\omega\sim\omega_j$ and
    $\eta\sim\omega_{j'}$ with $P_1(\omega)=P_1(\eta)$, then
    $\zeta^{-j}=P_1(\omega_j)=P_1(\omega_{j'})=\zeta^{-j'}$, so
    $j\equiv j'\pmod n$ and $\omega\sim\eta$; the converse is
    Theorem~\ref{thm:asymex_group_cocycle_domain_wall_phase_gauge}.
\end{proof}

\subsection{The group \texorpdfstring{$\Z_2\times\Z_2$}{Z2 x Z2}}

For $\Z_2\times\Z_2$ the construction is applied to the mixed three-cocycle
of type II whose exponent couples the first factor to the second, the
cocycle targeted in Section~\ref{sec:asymex_z2z2}. It is the type-II
component in the decomposition of the three-cocycles of $\Z_2\times\Z_2$
printed in~\cite[Section~6.1]{GarreRubio2023MPOSym}, where it takes the
value $-1$ exactly on the triples $(ab^i,a^jb,a^kb)$ with $a=(1,0)$ and
$b=(0,1)$. That source does not apply the periodic construction to it.

\begin{definition}[The mixed cocycle of type II]

\begin{theorem}[The instance for the group of order four]

    The function $\omega$ is a normalized three-cocycle with values $\pm1$.

    \begin{align}
        U_g(s,t)
         & =\delta_{s,g+t}\,
        (-1)^{\sum_ig_1x_{i+1,2}(x_{i,2}-x_{i+1,2})}.
        \notag
