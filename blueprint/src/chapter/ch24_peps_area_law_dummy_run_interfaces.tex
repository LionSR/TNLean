\section{Contacts with the initial dummy region}
\label{sec:area_law_dummy_run_interfaces}

The initial dummy region has label $k_0-1$ modulo two in
\cite[Section~11]{OpenAI2026AreaLaw}. The results below pass from
a contact with this region to the elementary perimeter assignment and
then to the label of an entire belt run. Write $N_0=N_{k_0}$ for the
half-open initial neighborhood and $\ell_k$ for the fine exponent at
layer $k$.

\begin{lemma}[An interior elementary contact determines the whole dummy side]
    \label{thm:area_law_elementary_open_dummy_contact}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayer_elementarySide_subset_dummy_of_mem_openSegment}
    \leanok
    \uses{def:area_law_dyadic_layers,def:area_law_cell_fan}
    Fix a common origin, a finite endpoint set and a width $C\ge 2$.
    Let $k\ge k_0$ and let $z\in F_{k,\ell_k}$ be a fine-cell index.
    Choose any optional midpoint subdivision of its sides, and let
    $e_i=[a_i,b_i]$ be an elementary base of the resulting fan.
    If $x\in(a_i,b_i)\cap\overline{N_0}$, then
    $e_i\subseteq\overline{N_0}$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dummy_corner_elementary_side,
        thm:area_law_dummy_layer_disjoint}
    Because $x\in\overline{N_0}$, some coarse cell in $N_0$ contains $x$ in its closure, and its interior is
    disjoint from the interior of the reference fine cell. After
    interchanging coordinates if necessary, $e_i$ is vertical. Since
    $x\in(a_i,b_i)$, disjointness forces the normal coordinate of $e_i$
    to be a boundary coordinate of the coarse cell. If a tangential
    boundary coordinate of the coarse cell lay strictly between $a_i$
    and $b_i$, the corresponding coarse-cell corner would lie in
    $(a_i,b_i)$, contradicting the endpoint rule for coarse-cell corners on
    elementary sides. Thus the
    coarse-cell closure contains $e_i$ and lies in $\overline{N_0}$.
\end{proof}

\begin{lemma}[A triangle meets the dummy closure through its base]
    \label{thm:area_law_fan_dummy_contact}
    \lean{TNLean.PEPS.AreaLaw.Geometry.fineLayer_cellFanPolygon_inter_dummy_eq}
    \leanok
    \uses{def:area_law_dyadic_layers,def:area_law_cell_fan}
    Fix a common origin, a finite endpoint set and an integer width $C\ge0$.
    Let $k\ge k_0$ and let $z\in F_{k,\ell_k}$ be a fine-cell index.
    Choose any optional midpoint subdivision of its sides. Every resulting
    fan triangle $P_i$, with elementary base $e_i$, satisfies
    \begin{align}
        P_i\cap\overline{N_0}=e_i\cap\overline{N_0}.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_dummy_layer_disjoint,thm:area_law_cell_fan_cover,
        thm:area_law_fan_own_side_contact,thm:area_law_unsplit_side_endpoints}
    The open interior
    of the reference fine cell is disjoint from $N_0$ and hence from
    $\overline{N_0}$. Any point of $P_i\cap\overline{N_0}$ belongs to the
    closed reference square and therefore lies on its boundary, which is
    the union of its four sides; the own-side contact identity
    then places the point on $e_i$. The reverse inclusion follows from
    $e_i\subseteq P_i$.
\end{proof}

\begin{lemma}[Opposite dummy parity along a belt-run contact]
    \label{thm:area_law_belt_run_dummy_interface}
    \lean{TNLean.PEPS.AreaLaw.Geometry.beltCellFanRun_dummy_interface}
    \leanok
    \uses{def:area_law_belt_fan_color,def:area_law_fan_runs}
    Fix a common origin, a finite endpoint set, a width $C\ge 2$ and an
    initial layer $k_0\ge 50{,}000{,}000$, with valid pitch residues in
    every layer. Let $R$ be a run in a belt cell of layer $k\ge k_0$. If
    a closed segment with distinct endpoints lies in both the run region
    and $\overline{N_0}$, then every constituent triangle of $R$ has label
    $k_0\ne k_0-1$ modulo two.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_fan_dummy_contact,
        thm:area_law_elementary_open_dummy_contact,
        thm:area_law_belt_fan_dummy_color,thm:area_law_fan_run_color}
    Since the shared segment has infinitely many points and the fan has
    finitely many elementary endpoints, the shared segment contains a point $x$
    that is not an elementary endpoint and lies in some constituent
    triangle $P_i$ of $R$. Because $P_i\cap\overline{N_0}=e_i\cap\overline{N_0}$,
    $x$ lies on the open base
    segment $(a_i,b_i)$ of $P_i$, which gives $e_i\subseteq\overline{N_0}$.
    The dummy-contact color rule assigns $P_i$ label $k_0$
    modulo two, and constancy of color along a run extends this
    label to every triangle in $R$.
\end{proof}
