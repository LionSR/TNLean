\begin{definition}[Normal one-site separation hypotheses]%
    \label{def:peps_normalOneSiteSeparationHypotheses}
    \lean{TNLean.PEPS.NormalOneSiteSeparationHypotheses}
    \leanok
    \uses{def:peps_regionInjectivityData, def:peps_regionComplement}
    For a region-injectivity assignment $\inj$
    (Definition~\ref{def:peps_regionInjectivityData}), the \emph{normal one-site
        separation hypotheses} assign to each $v\in V$ two regions
    $R_v,S_v\subseteq V$ such that $v\notin R_v$, $v\in S_v$,
    $S_v\setminus\{v\}=R_v\setminus\{v\}$ (so $S_v=R_v\cup\{v\}$), and
    $\inj(R_v)$, $\inj(V\setminus R_v)$, $\inj(S_v)$, $\inj(V\setminus S_v)$
    (Definition~\ref{def:peps_regionComplement}).
    In the square-lattice proof this compares the source regions $R$ and
    $S=R\cup\{v\}$:
    \begin{tenkzequation}
        % Source verdict: paper_normal.tex:1519--1544 compares the injective
        % regions R and S=R\cup\{v\}; their complements are injective as well.
        % Type verdict: this is a lattice/set schematic, not a tensor
        % contraction. The marked site v remains present with all its bonds.
        % Cell-set audit: R=(2-4,2-4)-(4,4) has eight selected cells, while
        % S=(2-4,2-4) has nine; v=(4,4) is marked in both pictures.
        % Contraction/boundary audit: there are no tensor contractions and no
        % tensor boundary signature; the boundary legs are cut lattice edges.
        \tenkzkernel
        \begin{tenkz}[rows={ket,ket,ket,ket,ket}, cols=5,
                west=open, east=open, north=open, south=open]
            \tn[at=(4,4), species=marked, name=vSite, label pos=sw]{v}
            \tnmark[form=enclosure, species=selected, tint, label pos=nw]
            {(2,2) .. (4,4) - (4,4)}{$R$}
        \end{tenkz}
        $\qquad$
        \begin{tenkz}[rows={ket,ket,ket,ket,ket}, cols=5,
                west=open, east=open, north=open, south=open]
            \tn[at=(4,4), species=marked, name=vSite, label pos=sw]{v}
            \tnmark[form=enclosure, species=selected, tint, label pos=nw]
            {(2,2) .. (4,4)}{$S$}
        \end{tenkz}
        $\qquad R\subset S=R\cup\{v\}$
    \end{tenkzequation}
\end{definition}

\begin{lemma}[Membership in a one-site separated region]%
    \label{thm:peps_normalOneSiteSeparation_mem_withSite_iff}
    \lean{TNLean.PEPS.NormalOneSiteSeparationHypotheses.mem_withSite_iff}
    \leanok
    \uses{def:peps_normalOneSiteSeparationHypotheses}
    Under Definition~\ref{def:peps_normalOneSiteSeparationHypotheses}, for every
    $v,w\in V$, $w\in S_v\Longleftrightarrow w=v\lor w\in R_v$.
\end{lemma}
\begin{proof}\leanok
    For $w=v$ use $v\in S_v$ and $v\notin R_v$; for $w\ne v$ use
    $S_v\setminus\{v\}=R_v\setminus\{v\}$.
\end{proof}

\begin{lemma}[Translationally invariant normal gauge absorption]%
    \label{thm:peps_tiNormalGaugeAbsorption}
    \notready
    \uses{thm:peps_normalEdgeBlockingToInjectiveChain, thm:peps_edge_gauge_absorption}

    \cite[Theorem~3]{Molnar2018NormalPEPS}, the edge gauges from
    Lemma~\ref{thm:peps_normalEdgeBlockingToInjectiveChain} are represented by
    one horizontal matrix $X$ and one vertical matrix $Y$, unique up to scalar;
    absorbing them into $B$ via Lemma~\ref{thm:peps_edge_gauge_absorption}
    yields $\widetilde B$ and the local gauge relation:
    % Formula (paper_normal.tex:1451--1471):

    % Ink-to-index: the four gauge capsules act on the four virtual indices of A;
    % the diagonal north-east stub is the unchanged physical index.
    % Boundary signature: both sides have (V,P)=(4,1).  Source verdict: exact
    % translation-invariant square-lattice gauge absorption in house style.
    \begin{tenkzequation}
        \tenkzkernel
        % Each unbonded typed port renders its own stub, so the four virtual
        % faces and the diagonal physical index need no boundary atoms.
        \begin{tenkz}[rows={wire}, cols=1, bonds=none]
            \tn[label pos=sw,
                ports={180:virtual, 0:virtual, 90:virtual, 270:virtual,
                        45:physical}]{B}
        \end{tenkz}
        $ = \lambda $
        \begin{tenkz}[rows={wire}, cols=1, bonds=none]
            \tn[name=tiAbsA, label pos=sw,
                ports={180:virtual, 0:virtual, 90:virtual, 270:virtual,
                        45:physical}]{A}

\begin{theorem}[Fundamental Theorem for normal square-lattice PEPS]%
    \label{thm:fundamentalTheorem_normalSquarePEPS}
    \notready
    \uses{thm:peps_tiNormalGaugeAbsorption, thm:peps_oneVertexComplementComparison, def:GaugeEquivPEPS}
    Let $A$ and $B$ be translationally invariant normal square-lattice PEPS
    tensors whose $2\times 3$ and $3\times 2$ regions are injective. If they
    generate the same state on an $n\times m$ lattice with $n,m\ge 7$, then they
    are related by horizontal and vertical gauges $X,Y$ up to a scalar $\lambda$
    with $\lambda^{nm}=1$ \cite[Theorem~3]{Molnar2018NormalPEPS}.
    The periodic torus theorem is
    Corollary~\ref{thm:fundamentalTheorem_normalTorusPEPS_unconditional}, the
    open-rectangle interior-window theorem (without translation invariance and
    with per-vertex scalars $\lambda_v$) is
    Corollary~\ref{thm:fundamentalTheorem_normalSquarePEPS_unconditional}, and the
    connected-graph theorem is
    Corollary~\ref{thm:fundamentalTheorem_normalPEPS_connected}.
\end{theorem}

\begin{corollary}[Fundamental Theorem for normal square-lattice PEPS,
        interior-window form]%
    \label{thm:fundamentalTheorem_normalSquarePEPS_unconditional}
    \lean{TNLean.PEPS.fundamentalTheorem_normalSquarePEPS_unconditional}
    \leanok
    \uses{thm:fundamentalTheorem_normalSquarePEPS,
        def:peps_normalSquareLatticeRectanglePEPS, def:applyGauge,
        def:peps_edgeInsertedCoeff}
    Let $A$ and $B$ be PEPS tensors on the open $n\times m$ square lattice
    with physical dimension $d>0$ and $D_A=D_B$, generating the same state, and
    both normal with injective rectangles
    (Definition~\ref{def:peps_normalSquareLatticeRectanglePEPS}). Then there
    exists a family of invertible edge matrices $X=(X_e)_{e\in E}$ such that,
    with $\widetilde B=B^X$ (Definition~\ref{def:applyGauge}):
    \begin{enumerate}
        \item for every edge $e$ with interior margins ($3\le x\le n-7$,
              $4\le y\le m-6$ for horizontal $e$ at $(x,y)$, and $4\le x\le n-6$,
              $3\le y\le m-7$ for vertical $e$ at $(x,y)$), every matrix $M$,
              and every physical configuration $\sigma$,
              $C_A(e,M;\sigma)=C_{\widetilde B}(e,M;\sigma)$
              (Definition~\ref{def:peps_edgeInsertedCoeff}); and
        \item for every vertex $v=(x,y)$ in the interior window
              $4\le x\le n-9$, $4\le y\le m-9$, there exists
              $\lambda_v\in\C^\times$ such that $A_v=\lambda_v\widetilde B_v$.
    \end{enumerate}
    This open-lattice counterpart of \cite[Theorem~3]{Molnar2018NormalPEPS} is
    restricted to interior margins and uses site-dependent scalars $\lambda_v$
    (since without translation invariance, rescaling $A$ at two interior sites
    by $\mu$ and $\mu^{-1}$ preserves the state and rectangular injectivity
    while separating the scalars).
    \begin{proof}
        \leanok
        \uses{lem:peps_injectiveRegion_union_pos}
        By Lemma~\ref{lem:peps_injectiveRegion_union_pos}, $D_A,D_B>0$ gives
        union closure of injective regions. The translated blocking frames at
        interior-margin edges yield per-edge gauges $X_e$ whose absorption
        $\widetilde B=B^X$ satisfies $C_A(e,M;\sigma)=C_{\widetilde B}(e,M;\sigma)$.
        Since gauge absorption preserves region injectivity, comparing the
        punctured $3\times 3$ square $R_v$ at an interior-window vertex $v$ with
        $S_v=R_v\cup\{v\}$ (all of whose boundary edges carry interior margins)
        yields proportionality scalars $c_R,c_S\in\C^\times$ and
        $A_v=\lambda_v\widetilde B_v$ with $\lambda_v=c_S/c_R$.
    \end{proof}
\end{corollary}
