\section{Proof of the amplification proposition}

This section assembles the proof of Proposition~10.2 of
\cite{OpenAI2026AreaLaw} from the estimates of the preceding sections. Three
connections remain. The restoring operator acts on two ancillary copies
$s',e'$ of the target and on the physical system, with the target written in
an eigenbasis of its reduced state, while the constraint products act on the
physical system with the ancillas as spectators. The omission estimate is
stated for operators applied to the ground vector and the truncation estimate
for products over the retained letters of one clock word. The final entropy
comparison treats the two copies as additional sites.

\begin{theorem}[One clock realization for the three products]
    \label{thm:al_amp_constraint_word_lists}
    \lean{Matrix.contractionWord_eq_prod_reverse,
        Matrix.contractionWord_partitionWords_fst,
        Matrix.contractionWord_mul_vecMulVec_self,
        Matrix.conjTranspose_contractionWord_mulVec_self}
    \leanok
    Let $0\le k_i\le I$ on a finite-dimensional Hilbert space and
    $G_i=(I-k_i)^{1/2}$. For a word $w=(i_1,\dots,i_m)$, the chronological
    product $G(w)=G_{i_m}\cdots G_{i_1}$ equals the product over the reversed
    list $w^{\mathrm{rev}}$, and the product over the letters of $w$ satisfying
    a predicate $P$ equals $G(w|_P)$ for the filtered subword $w|_P$. If
    $k_i\Omega=0$ for all $i$, then every such product $D=G(u)$ satisfies
    $D|\Omega\rangle\langle\Omega|=|\Omega\rangle\langle\Omega|$ and
    $D^\dagger\Omega=\Omega$.
\end{theorem}
\begin{proof}
    \leanok
    Induct on $|w|$; since $k_i\Omega=0$ and $G_i^\dagger=G_i$, each factor
    satisfies $G_i\Omega=G_i^\dagger\Omega=\Omega$.
\end{proof}

\begin{definition}[Coordinate change placing the ancillas as spectators]
    \label{def:al_amp_ancilla_transport}
    \lean{TNLean.PEPS.AreaLaw.ancillaRotation, TNLean.PEPS.AreaLaw.ancillaTransport,
        TNLean.PEPS.AreaLaw.ancillaPhysical, TNLean.PEPS.AreaLaw.restoringEigenbasis,
        TNLean.PEPS.AreaLaw.restoringTransport}
    \leanok
    \uses{def:al_amp_restoring_coordinates}
    Let $U$ be the unitary whose columns form an eigenbasis $\{|x\rangle\}$ of
    $\rho_X$. The unitary $\mathcal W$ maps the restoring coordinates
    $((s',e')\times(X\times Y))\times Z'$ to $\Lambda\times(s',e')$ by
    reordering factors, conjugating $X$ by $U$, and conjugating $e'$ by $U$;
    an operator $M$ on $X\otimes Y$ in the eigenbasis lifts to the physical
    operator $L(M)=((U\otimes I_Y)M(U\otimes I_Y)^\dagger)\otimes I_{Z'}$.
\end{definition}

\begin{theorem}[Restoring data under $\mathcal W$]
    \label{thm:al_amp_ancilla_transport}
    \lean{TNLean.PEPS.AreaLaw.ancillaTransport_mul_conjTranspose,
        TNLean.PEPS.AreaLaw.conjTranspose_mul_ancillaTransport,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_physicalLift,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_kronecker,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_restored,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_initial,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_copy,
        TNLean.PEPS.AreaLaw.restoringGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringColumnGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_restoringGlobal,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_restoringColumnGlobal}
    \leanok
    \uses{def:al_amp_ancilla_transport}
    The map $\mathcal W$ is unitary and satisfies
    \begin{align}
        \mathcal W\bigl((A\otimes M)\otimes I_{Z'}\bigr)\mathcal W^\dagger
         & =L(M)\otimes\bigl((I_{s'}\otimes U)A(I_{s'}\otimes U)^\dagger\bigr), \notag \\
        (\mathcal W\varphi_{\rm copy})(c,(s,e))
         & =\langle s|0\rangle\,\langle c_X|U|0\rangle\,\Omega(c|_{X^c},e), \notag     \\
        \mathcal W\eta_{\rm rest}
         & =\Omega\otimes(I_{s'}\otimes U)\sum_x\sqrt{p_x}\,|x,x\rangle,
        \qquad
        \mathcal W\psi_{\rm init}=\Omega\otimes(|0\rangle\otimes U|0\rangle),
        \label{eq:al_amp_ancilla_transport}
    \end{align}
    where $(c|_{X^c},e)$ agrees with $c$ off $X$ and equals $e$ on $X$. Under
    conjugation by $\mathcal W$, the restoring operator $R_0$ and column
    operator $B_0$ expand as sums over the selected indices of physical
    operators supported on $X\cup Y$ tensored with
    $p_x^{-1/2}|x\rangle\langle0|_{s'}\otimes I_{e'}$ and with ancilla
    operators on $(s',e')$, respectively.
\end{theorem}
\begin{proof}
    \leanok
    Since $\mathcal W$ is a coordinate permutation composed with a Kronecker
    product of unitaries, (\ref{eq:al_amp_ancilla_transport}) and the sum
    expansions follow from the mixed-product property, $UU^\dagger=I$, and
    direct evaluation on basis configurations.
\end{proof}

\begin{theorem}[The restoring estimates with the ancillas as spectators]
    \label{thm:al_amp_restoring_transport}
    \lean{TNLean.PEPS.AreaLaw.restoringTransport_estimates,
        TNLean.PEPS.AreaLaw.restoringTransport_copy,
        TNLean.PEPS.AreaLaw.restoringTransport_restored,
        TNLean.PEPS.AreaLaw.restoringTransport_restoringGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringTransport_restoringColumnGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringTransport_unitary,
        TNLean.PEPS.AreaLaw.restoringEigenbasis_unitary,
        TNLean.PEPS.AreaLaw.ancillaPhysical_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.localLift_eq_kronecker_one_submatrix,
        TNLean.PEPS.AreaLaw.l2_opNorm_le_one_of_conjTranspose_mul_self,
        TNLean.PEPS.AreaLaw.norm_toLp_mulVec_of_conjTranspose_mul_self,
        TNLean.PEPS.AreaLaw.l2_opNorm_conj_le,
        TNLean.PEPS.AreaLaw.mulVec_mulVec_eq_conj}
    \leanok
    \uses{thm:al_amp_restoring_estimates}
    Let $\|\Omega\|=1$, let $X,Y$ be disjoint with $S_0=X\cup Y$, and suppose
    the typical projections of $X$, $Y$, and $S_0$ fail with probability at
    most $\delta$. In the coordinates $\Lambda\times(s',e')$, with
    $\xi=|0\rangle\otimes U|0\rangle$, the transported vectors $\varphi,\eta$
    and operators $R_0,B_0$ satisfy
    \begin{align}
        \|\varphi\|=\|\eta\|=1,\qquad
        \|R_0\|,\|B_0\|\le M_0=e^{(\mathcal I+3w)/2},\qquad
        R_0\varphi=B_0(\Omega\otimes\xi),\qquad
        \|(|\Omega\rangle\langle\Omega|\otimes I)R_0\varphi\|\le 1,
        \label{eq:al_amp_restoring_transport_norms}
    \end{align}
    and, for every physical operator $L'$,
    \begin{align}
        \|((L'\otimes I)R_0)^\dagger\eta-\varphi\|
         & \le3\sqrt\delta+\|(L')^\dagger\Omega-\Omega\|.
        \label{eq:al_amp_restoring_transport_adj}
    \end{align}
    Moreover, $B_0$ acts on $S_0\times(s',e')$ and $R_0$ acts on $S_0\times s'$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_amp_ancilla_transport}
    Conjugating Theorem~\ref{thm:al_amp_restoring_estimates} by the unitary
    $\mathcal W$ via (\ref{eq:al_amp_ancilla_transport}) preserves vector and
    operator norms, giving (\ref{eq:al_amp_restoring_transport_norms}) and
    (\ref{eq:al_amp_restoring_transport_adj}). Since $U$ acts only on $X$,
    each summand $L(M)$ is supported on $S_0=X\cup Y$.
\end{proof}

\begin{definition}[Relabelled and product vectors on finite site sets]
    \label{def:al_amp_site_relabel}
    \lean{FiniteProduct.relabelSites, FiniteProduct.regionRelabel,
        FiniteProduct.complRelabel, FiniteProduct.sumSitesProduct,
        FiniteProduct.disjSumRegionEquiv, FiniteProduct.disjSumComplEquiv}
    \leanok
    For a bijection $\sigma:A\to B$ between finite site sets with a common local
    alphabet, $(\sigma_*\psi)(c)=\psi(c\circ\sigma)$. For vectors $f$ on $A$
    and $g$ on $B$, the product vector $f\otimes g$ lives on $A\sqcup B$.
\end{definition}

\begin{theorem}[Regional entropies of relabelled and product vectors]
    \label{thm:al_amp_site_entropy}
    \lean{FiniteProduct.splitEquiv_symm_comp_relabel,
        FiniteProduct.reducedPure_relabelSites, FiniteProduct.entropy_relabelSites,
        FiniteProduct.mutualInformation_relabelSites,
        FiniteProduct.splitEquiv_symm_comp_inl, FiniteProduct.splitEquiv_symm_comp_inr,
        FiniteProduct.reducedPure_sumSitesProduct, FiniteProduct.entropy_sumSitesProduct,
        FiniteProduct.mutualInformation_sumSitesProduct_inl,
        FiniteProduct.mutualInformation_sumSitesProduct_inr_inl}
    \leanok
    \uses{def:al_amp_site_relabel}
    For any bijection $\sigma:A\to A'$ and $R\subseteq A$,
    $S_{\sigma_*\psi}(\sigma(R))=S_\psi(R)$. For unit vectors $f$ on $A$ and
    $g$ on $B$, and regions $R,R'\subseteq A$ and $T\subseteq B$,
    \begin{align}
        S_{f\otimes g}(R\sqcup T)=S_f(R)+S_g(T),\qquad
        I_{f\otimes g}(R:R')=I_f(R:R'),\qquad
        I_{f\otimes g}(T:R)=0. \notag
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Relabelling conjugates the reduced density matrix by the induced
    configuration bijection, while $\rho_{f\otimes g,R\sqcup T}=\rho_{f,R}\otimes\rho_{g,T}$,
    so additivity of von Neumann entropy on tensor products gives all three
    identities.
\end{proof}

\begin{definition}[The physical sites enlarged by two copies of the target]
    \label{def:al_amp_ancilla_sites}
    \lean{TNLean.PEPS.AreaLaw.AncillaSites, TNLean.PEPS.AreaLaw.ancillaSplit,
        TNLean.PEPS.AreaLaw.ancillaCopyRegion,
        TNLean.PEPS.AreaLaw.ancillaPhysicalRegion,
        TNLean.PEPS.AreaLaw.ancillaSwapFun, TNLean.PEPS.AreaLaw.ancillaSwap}
    \leanok
    The enlarged site set is $\widetilde\Lambda=\Lambda\sqcup(s'\sqcup e')$,
    where $s',e'$ are two copies of $X$ with local dimension $q$. The involution
    $\tau:\widetilde\Lambda\to\widetilde\Lambda$ swaps each site of $X\subseteq\Lambda$
    with its copy in $e'$ and fixes $\Lambda\setminus X$ and $s'$.
\end{definition}

\begin{theorem}[The enlarged sites]
    \label{thm:al_amp_ancilla_sites}
    \lean{TNLean.PEPS.AreaLaw.ancillaSplit_apply,
        TNLean.PEPS.AreaLaw.kronecker_one_submatrix_ancillaSplit,
        TNLean.PEPS.AreaLaw.one_kronecker_submatrix_ancillaSplit,
        TNLean.PEPS.AreaLaw.kronecker_submatrix_ancillaSplit_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.kronecker_one_submatrix_ancillaSplit_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.restoringSplit_symm_replace,
        TNLean.PEPS.AreaLaw.ancillaSwapFun_involutive,
        TNLean.PEPS.AreaLaw.mutualInformation_ancillaProduct_copy_eq_zero,
        TNLean.PEPS.AreaLaw.mutualInformation_ancillaCopy_eq}
    \leanok
    \uses{def:al_amp_ancilla_sites,thm:al_amp_site_entropy}
    On $\widetilde\Lambda=\Lambda\sqcup(s'\sqcup e')$, if $P$ is supported on
    $S\subseteq\Lambda$ and $b$ acts on $s'$, then $P\otimes(b\otimes I_{e'})$
    is supported on $S\sqcup s'$ and $P\otimes I_{s'\sqcup e'}$ is supported on
    $S$. For unit vectors $\Omega$ on $\Lambda$ and $\chi$ on $s'\sqcup e'$,
    $I_{\Omega\otimes\chi}(e':J)=0$ for every $J\subseteq\Lambda$. If
    $\varphi(c,(s,e))=a(s)b(c_X)\Omega(c|_{X^c},e)$ with $\|a\|=\|b\|=\|\Omega\|=1$,
    then for every $X'\subseteq X$ (with copy $X'_{e'}\subseteq e'$) and
    $J\subseteq\Lambda\setminus X$,
    \begin{align}
        I_\varphi(X'_{e'}:J)=I_\Omega(X':J).
        \label{eq:al_amp_ancilla_copy_mi}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Since $\varphi=\tau_*(\Omega\otimes a\otimes b)$ with $\tau(X')=X'_{e'}$ and
    $\tau(J)=J$, Theorem~\ref{thm:al_amp_site_entropy} yields
    $I_{\Omega\otimes\chi}(e':J)=0$ and (\ref{eq:al_amp_ancilla_copy_mi}).
\end{proof}

\begin{theorem}[The local unitary and the entropy comparison on a region]
    \label{thm:al_amp_supported_polar}
    \lean{QuantumCircuit.embedOp_val_eq_kronecker_one_submatrix,
        QuantumCircuit.kronecker_one_submatrix_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.exists_supported_unitary_of_good_realization,
        TNLean.Lattice.mutualInformation_le_inv_pow_of_supported_unitary}
    \leanok
    \uses{thm:area_law_amplification_unitary_error,thm:al_amp_entropy_comparison_scale}
    On a finite site set with local dimension $q$, suppose $R_0$ and $L'$ are
    supported on $S$ and satisfy the hypotheses of
    Theorem~\ref{thm:area_law_amplification_unitary_error}. Then there is a
    unitary $U$ supported on $S$ with
    $\|U\varphi-\eta\|\le a+\sqrt{2a+6\varepsilon+9\varepsilon^2}$. If $U$ is a
    unitary supported on a region disjoint from $X\cup J$, $I_\eta(X:J)=0$,
    $\|U\varphi-\eta\|\le C_1n^{-K/2}$ with $K\ge4(M+2)$, and
    $\log\dim\mathcal H_X\le C_2n^2$, then every $X'\subseteq X$ satisfies
    $I_\varphi(X':J)\le 8\sqrt{C_1}\,(1+C_2)\,n^{-M}$.
\end{theorem}
\begin{proof}
    \leanok
    In the tensor decomposition of $S$ and its complement (respectively
    $X\cup J$ and its complement), an operator supported on $S$ has the form
    $A\otimes I$ (respectively $I\otimes A$). Applying
    Theorem~\ref{thm:area_law_amplification_unitary_error} in the first
    decomposition and Theorem~\ref{thm:al_amp_entropy_comparison_scale} in the
    second gives both claims.
\end{proof}

\begin{theorem}[The local unitary and the entropy comparison on the enlarged sites]
    \label{thm:al_amp_enlarged_steps}
    \lean{TNLean.PEPS.AreaLaw.exists_ancilla_unitary_of_good_realization,
        TNLean.PEPS.AreaLaw.mutualInformation_le_of_ancilla_unitary}
    \leanok
    \uses{thm:al_amp_ancilla_sites,thm:al_amp_supported_polar}
    On $\widetilde\Lambda=\Lambda\sqcup(s'\sqcup e')$, suppose $R_0$ is
    supported on $N\sqcup s'$, $L'\otimes I_{s'\sqcup e'}$ is supported on
    $N\subseteq\Lambda$, and the hypotheses of
    Theorem~\ref{thm:area_law_amplification_unitary_error} hold. Then there is a
    unitary $U$ supported on $N\sqcup s'$ with
    $\|U\varphi-\eta\|\le a+\sqrt{2a+6\varepsilon+9\varepsilon^2}$. If
    $X\subseteq N$, $\varphi=\mathcal W\varphi_{\rm copy}$ and
    $\eta=\mathcal W\eta_{\rm rest}$ have unit norm,
    $\|U\varphi-\eta\|\le C_1n^{-K/2}$ with $K\ge4(M+2)$, and
    $|X|\log q\le C_2n^2$, then for every $X'\subseteq X$ and
    $J\subseteq\Lambda\setminus N$,
    \begin{align}
        I_\Omega(X':J)\le 8\sqrt{C_1}\,(1+C_2)\,n^{-M}.
        \label{eq:al_amp_enlarged_mi_bound}
    \end{align}
\end{theorem}
\begin{proof}
    \leanok
    Applying Theorem~\ref{thm:al_amp_supported_polar} on $\widetilde\Lambda$
    with $S=N\sqcup s'$ gives $U$. Since $N\sqcup s'$ is disjoint from
    $e'\sqcup J$, $I_\eta(e':J)=0$ by (\ref{eq:al_amp_ancilla_transport}) and
    Theorem~\ref{thm:al_amp_ancilla_sites}, and $I_\varphi(X'_{e'}:J)=I_\Omega(X':J)$
    by (\ref{eq:al_amp_ancilla_copy_mi}), Theorem~\ref{thm:al_amp_supported_polar}
    applied to $e'$ and $J$ yields (\ref{eq:al_amp_enlarged_mi_bound}).
\end{proof}

\begin{theorem}[The amplification radius is a power of the scale]
    \label{thm:al_amp_radius_bound}
    \lean{TNLean.PEPS.AreaLaw.exists_amplificationRadius_le}
    \leanok
    \uses{def:area_law_amplification_time_radius}
    Let $g>0$, $C'\ge0$, $C_0\ge0$, $\alpha>0$, $0<\varepsilon\le2/5$, and
    $K\in\mathbb Z$. There exists $C$ such that for all $n\ge1$ and
    $0\le\mathcal I\le C_0n^{1-\varepsilon}$, the radius $r$ of
    Definition~\ref{def:area_law_amplification_time_radius} with $w=n^{3/5}$
    satisfies $r\le Cn^{(1-\varepsilon)/\alpha}$.
\end{theorem}
\begin{proof}
    \leanok
    Since $w=n^{3/5}\le n^{1-\varepsilon}$ and $\log n\le n^{1-\varepsilon}$ for
    $n\ge1$, $C'(t_*+K\log n)\le C'Bn^{1-\varepsilon}$, and raising to the
    power $1/\alpha$ gives $r\le Cn^{(1-\varepsilon)/\alpha}$.
\end{proof}

\begin{theorem}[Proposition 10.2: amplification]
    \label{thm:area_law_amplification}
    \lean{TNLean.PEPS.AreaLaw.amplificationBound}
    \leanok
    \uses{def:area_law_amplification_bound}
    In Proposition~10.2 of \cite{OpenAI2026AreaLaw}, for every $q,R,J$ and every
    gap $\Delta>0$ the amplification bound holds.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_amplification_of_radius_output,thm:al_amp_restoring_transport,
        thm:al_amp_constraint_word_lists,thm:al_lattice_excited_decay,
        thm:al_lattice_constraint_omission,thm:al_lattice_product_truncation,
        thm:area_law_amplification_scale_bounds,thm:al_amp_typical_projectors,
        thm:al_amp_ancilla_sites,thm:al_amp_supported_polar,thm:al_amp_enlarged_steps,
        thm:al_amp_radius_bound}
    If $q=0$ or $J<0$, a gapped ground vector exists only on $\Lambda=\varnothing$.
    Otherwise fix $p=10^6$, $\alpha=p/(p+1)$, and $K=4(M+2)$. At scale $n$, set
    $w=n^{3/5}$, $\mathcal I=I_\Omega(X:\Lambda\setminus S_0)$, and $\delta=n^{-2K}$,
    and choose $t_*$ and $r$ as in
    Definition~\ref{def:area_law_amplification_time_radius}, so
    $r\le Cn^{(1-\varepsilon)/\alpha}$ by Theorem~\ref{thm:al_amp_radius_bound}.
    Theorem~\ref{thm:al_amp_typical_projectors} and
    Theorem~\ref{thm:al_amp_restoring_transport} provide $R_0,B_0,\varphi,\eta$
    satisfying (\ref{eq:al_amp_restoring_transport_norms}) and
    (\ref{eq:al_amp_restoring_transport_adj}). On clock words at time $t_*$,
    Theorem~\ref{thm:al_lattice_excited_decay} bounds the excited component of
    $\zeta=R_0\varphi$, Theorem~\ref{thm:al_lattice_constraint_omission} applied
    via $R_0\varphi=B_0(\Omega\otimes\xi)$ bounds the omitted events,
    Theorem~\ref{thm:al_lattice_product_truncation} bounds the radius-$r$
    truncation $L'$ supported on $N_{2r}^\Lambda(S_0)$, and
    Theorem~\ref{thm:al_amp_constraint_word_lists} matches the filtered clock
    products. By Theorem~\ref{thm:area_law_amplification_scale_bounds}, a good
    realization makes all three errors at most $n^{-K}$. Applying
    Theorem~\ref{thm:al_amp_enlarged_steps} with $N=N_{2r}^\Lambda(S_0)$,
    $C_1=9$, and $C_2=C_0\log q$ gives a unitary $U$ on
    $N_{2r}^\Lambda(S_0)\sqcup s'$ with $\|U\varphi-\eta\|\le 9n^{-K/2}$ and,
    via (\ref{eq:al_amp_enlarged_mi_bound}),
    $I_\Omega(X':J)\le 24(1+C_0\log q)\,n^{-M}$ for every $X'\subseteq X$ and
    $J\subseteq\Lambda\setminus N_{2r}^\Lambda(S_0)$. Finally,
    Theorem~\ref{thm:area_law_amplification_of_radius_output} yields the
    cut-clearance form.
\end{proof}
