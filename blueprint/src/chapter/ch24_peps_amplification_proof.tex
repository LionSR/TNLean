\section{Proof of the amplification proposition}

This section assembles the proof of Proposition~10.2 of
\cite{OpenAI2026AreaLaw} from the estimates of the preceding sections. Three
connections remain. The restoring operator acts on two ancillary copies
$s',e'$ of the target and on the physical system, with the target written in
an eigenbasis of its reduced state, while the constraint products act on the
physical system with the ancillas as spectators. The omission estimate is
stated for operators applied to the ground vector and the truncation estimate
for products over the retained letters of one clock word. The final entropy
comparison treats the two copies as additional sites.

\begin{theorem}[One clock realization for the three products]
    \label{thm:al_amp_constraint_word_lists}
    \lean{Matrix.contractionWord_eq_prod_reverse,
        Matrix.contractionWord_partitionWords_fst,
        Matrix.contractionWord_mul_vecMulVec_self,
        Matrix.conjTranspose_contractionWord_mulVec_self}
    \leanok
    Let $k_i$ be positive contractions on a finite-dimensional Hilbert space and
    $G_i=(I-k_i)^{1/2}$. For a finite word $w=(i_1,\dots,i_m)$ the chronological
    product $G_{i_m}\cdots G_{i_1}$ is the product over the reversed list of
    letters. The product over the letters of $w$ satisfying a predicate $P$
    equals the product over the list of letters of $w$ filtered by $P$. If
    every $k_i$ annihilates a vector $\Omega$, then every such product $D$
    satisfies $D|\Omega\rangle\langle\Omega|=|\Omega\rangle\langle\Omega|$
    and $D^\dagger\Omega=\Omega$.
\end{theorem}
\begin{proof}
    \leanok
    We induct on the length of the word. Each factor is self-adjoint and fixes
    $\Omega$ because $k_i\Omega=0$.
\end{proof}

\begin{definition}[Coordinate change placing the ancillas as spectators]
    \label{def:al_amp_ancilla_transport}
    \lean{TNLean.PEPS.AreaLaw.ancillaRotation, TNLean.PEPS.AreaLaw.ancillaTransport,
        TNLean.PEPS.AreaLaw.ancillaPhysical, TNLean.PEPS.AreaLaw.restoringEigenbasis,
        TNLean.PEPS.AreaLaw.restoringTransport}
    \leanok
    \uses{def:al_amp_restoring_coordinates}
    Let $U$ be the unitary whose columns are a full eigenbasis $\{|x\rangle\}$
    of $\rho_X$. The unitary $\mathcal W$ maps the coordinates
    $((s',e'),(X,Y)),Z'$ of the restoring operator to the coordinates
    $\Lambda\times(s',e')$: it reorders the factors, undoes the basis change on
    the physical target, and applies the same basis change on $e'$. An operator
    $M$ on $X\otimes Y$ in the eigenbasis gives the physical operator
    $((U\otimes I)M(U\otimes I)^\dagger)\otimes I_{Z'}$.
\end{definition}

\begin{theorem}[Restoring data under $\mathcal W$]
    \label{thm:al_amp_ancilla_transport}
    \lean{TNLean.PEPS.AreaLaw.ancillaTransport_mul_conjTranspose,
        TNLean.PEPS.AreaLaw.conjTranspose_mul_ancillaTransport,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_physicalLift,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_kronecker,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_restored,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_initial,
        TNLean.PEPS.AreaLaw.ancillaTransport_mulVec_copy,
        TNLean.PEPS.AreaLaw.restoringGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringColumnGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_restoringGlobal,
        TNLean.PEPS.AreaLaw.ancillaTransport_conj_restoringColumnGlobal}
    \leanok
    \uses{def:al_amp_ancilla_transport}
    The map $\mathcal W$ is unitary. It maps a physical operator written in
    the eigenbasis of the target to the operator $L\otimes I$ on the original
    physical coordinates, and $(A\otimes M)\otimes I_{Z'}$ to the physical operator
    obtained from $M$ tensored with $A$ rotated on $e'$. The restored vector
    becomes $\Omega$ tensored with the ancillary vector
    $(I\otimes U)\sum_x\sqrt{p_x}|x,x\rangle$, the vector with blanks on both
    ancillas becomes $\Omega$ tensored with the blanks, and the copied vector
    becomes the vector whose value at a configuration $c$ and an ancillary pair
    $(s,e)$ is $|0\rangle(s)\,(U|0\rangle)(c_X)\,\Omega(c')$, with $c'$ equal to $c$
    off $X$ and to $e$ on $X$. The restoring operator is a sum over the selected
    indices of physical operators on $X\cup Y$ tensored with
    $p_x^{-1/2}|x\rangle\langle0|_{s'}\otimes I_{e'}$, and the column operator is a
    sum of physical operators on $X\cup Y$ tensored with operators on the
    ancillas.
\end{theorem}
\begin{proof}
    \leanok
    The map $\mathcal W$ is a reindexing followed by a Kronecker product of unitaries.
    Each image follows from the mixed-product rule for Kronecker products, the
    identity $UU^\dagger=I$, and direct substitution in the copied vector.
\end{proof}

\begin{theorem}[The restoring estimates with the ancillas as spectators]
    \label{thm:al_amp_restoring_transport}
    \lean{TNLean.PEPS.AreaLaw.restoringTransport_estimates,
        TNLean.PEPS.AreaLaw.restoringTransport_copy,
        TNLean.PEPS.AreaLaw.restoringTransport_restored,
        TNLean.PEPS.AreaLaw.restoringTransport_restoringGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringTransport_restoringColumnGlobal_eq_sum,
        TNLean.PEPS.AreaLaw.restoringTransport_unitary,
        TNLean.PEPS.AreaLaw.restoringEigenbasis_unitary,
        TNLean.PEPS.AreaLaw.ancillaPhysical_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.localLift_eq_kronecker_one_submatrix,
        TNLean.PEPS.AreaLaw.l2_opNorm_le_one_of_conjTranspose_mul_self,
        TNLean.PEPS.AreaLaw.norm_toLp_mulVec_of_conjTranspose_mul_self,
        TNLean.PEPS.AreaLaw.l2_opNorm_conj_le,
        TNLean.PEPS.AreaLaw.mulVec_mulVec_eq_conj}
    \leanok
    \uses{thm:al_amp_restoring_estimates}
    Let $\Omega$ be a unit vector, let $X,Y$ be disjoint with $S_0=X\cup Y$, and
    suppose that the typical projections of $X$, $Y$ and $S_0$ fail with
    probability at most $\delta$. In the coordinates $\Lambda\times(s',e')$ the
    vectors $\varphi$ and $\eta$ are unit vectors,
    $\|R_0\|,\|B_0\|\le M_0=e^{(\mathcal I+3w)/2}$, the column identity
    $R_0\varphi=B_0(\Omega\otimes\xi)$ holds for the image $\xi$ of the blank pair,
    the physical ground component of $R_0\varphi$ has norm at most one, and
    \begin{align*}
        \|((L'\otimes I)R_0)^\dagger\eta-\varphi\| & \le3\sqrt\delta+
        \|(L')^\dagger\Omega-\Omega\|
    \end{align*}
    for every physical operator $L'$. The column operator acts on $S_0$ and the
    ancillas, and the restoring operator on $S_0$ and $s'$.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_amp_ancilla_transport}
    Transferring the estimates of Theorem~\ref{thm:al_amp_restoring_estimates} along
    $\mathcal W$ gives the bounds, since unitary equivalence preserves norms of
    vectors and of operators. An operator
    on $X\otimes Y$ in the eigenbasis tensored with $I_{Z'}$ acts on $X\cup Y$ in
    the original coordinates, since the basis change acts on $X$ alone.
\end{proof}

\begin{definition}[Relabelled and product vectors on finite site sets]
    \label{def:al_amp_site_relabel}
    \lean{FiniteProduct.relabelSites, FiniteProduct.regionRelabel,
        FiniteProduct.complRelabel, FiniteProduct.sumSitesProduct,
        FiniteProduct.disjSumRegionEquiv, FiniteProduct.disjSumComplEquiv}
    \leanok
    For a bijection $\sigma$ between finite site sets with a common local
    alphabet, a vector $\psi$ is moved along $\sigma$ by reading each
    configuration through $\sigma^{-1}$. For vectors $f$ on the sites $A$ and $g$
    on the sites $B$, the product vector $f\otimes g$ lives on the sites
    $A\sqcup B$.
\end{definition}

\begin{theorem}[Regional entropies of relabelled and product vectors]
    \label{thm:al_amp_site_entropy}
    \lean{FiniteProduct.splitEquiv_symm_comp_relabel,
        FiniteProduct.reducedPure_relabelSites, FiniteProduct.entropy_relabelSites,
        FiniteProduct.mutualInformation_relabelSites,
        FiniteProduct.splitEquiv_symm_comp_inl, FiniteProduct.splitEquiv_symm_comp_inr,
        FiniteProduct.reducedPure_sumSitesProduct, FiniteProduct.entropy_sumSitesProduct,
        FiniteProduct.mutualInformation_sumSitesProduct_inl,
        FiniteProduct.mutualInformation_sumSitesProduct_inr_inl}
    \leanok
    \uses{def:al_amp_site_relabel}
    Relabelling the sites along $\sigma$ moves every regional entropy along
    $\sigma$. For unit vectors $f$ and $g$, the region $R\sqcup T$ of $f\otimes g$
    has entropy $S_f(R)+S_g(T)$. Consequently adjoining sites in a product state
    does not change the mutual information of two regions of $A$, and a region of
    $B$ has no mutual information with a region of $A$.
\end{theorem}
\begin{proof}
    \leanok
    The reduced state of the relabelled vector is the reduced state of the
    original vector on the image region, after relabelling configurations. The
    reduced state of $f\otimes g$ on $R\sqcup T$ is the Kronecker product of the
    two reduced states, and von Neumann entropy is additive on Kronecker products
    of density matrices.
\end{proof}

\begin{definition}[The physical sites enlarged by two copies of the target]
    \label{def:al_amp_ancilla_sites}
    \lean{TNLean.PEPS.AreaLaw.AncillaSites, TNLean.PEPS.AreaLaw.ancillaSplit,
        TNLean.PEPS.AreaLaw.ancillaCopyRegion,
        TNLean.PEPS.AreaLaw.ancillaPhysicalRegion,
        TNLean.PEPS.AreaLaw.ancillaSwapFun, TNLean.PEPS.AreaLaw.ancillaSwap}
    \leanok
    The enlarged site set is $\Lambda\sqcup(X\sqcup X)$, where the two copies of
    $X$ are the ancillas $s'$ and $e'$, each site carrying the local dimension
    $q$. Its configurations are a physical configuration together with
    configurations of $s'$ and of $e'$. The swap exchanges each site of $X$ with
    its copy in $e'$ and fixes all other sites.
\end{definition}

\begin{theorem}[The enlarged sites]
    \label{thm:al_amp_ancilla_sites}
    \lean{TNLean.PEPS.AreaLaw.ancillaSplit_apply,
        TNLean.PEPS.AreaLaw.kronecker_one_submatrix_ancillaSplit,
        TNLean.PEPS.AreaLaw.one_kronecker_submatrix_ancillaSplit,
        TNLean.PEPS.AreaLaw.kronecker_submatrix_ancillaSplit_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.kronecker_one_submatrix_ancillaSplit_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.restoringSplit_symm_replace,
        TNLean.PEPS.AreaLaw.ancillaSwapFun_involutive,
        TNLean.PEPS.AreaLaw.mutualInformation_ancillaProduct_copy_eq_zero,
        TNLean.PEPS.AreaLaw.mutualInformation_ancillaCopy_eq}
    \leanok
    \uses{def:al_amp_ancilla_sites,thm:al_amp_site_entropy}
    On the enlarged sites, an operator $P\otimes(b\otimes I_{e'})$ with $P$
    acting on a physical region $S$ and $b$ acting on $s'$ acts on $S\sqcup s'$,
    and a physical operator on $S$ tensored with the identity on both copies acts
    on $S$. For unit vectors $\Omega$ and $\chi$, the vector $\Omega\otimes\chi$
    has no mutual information between $e'$ and any physical region. Let $\varphi$
    be the vector whose value at a configuration $c$ and an ancillary pair
    $(s,e)$ is $a(s)\,b(c_X)\,\Omega(c')$, with $c'$ equal to $c$ off $X$ and to
    $e$ on $X$, for unit vectors $a,b$ and a unit vector $\Omega$. For every
    $X'\subseteq X$ and every physical region $J$ disjoint from $X$, the mutual
    information of $\varphi$ between the copy of $X'$ in $e'$ and $J$ is
    $I_\Omega(X':J)$.
\end{theorem}
\begin{proof}
    \leanok
    The vector $\varphi$ is $\Omega\otimes a\otimes b$ relabelled by the swap, and
    the swap maps the copy of $X'$ to $X'$ and fixes $J$, so
    Theorem~\ref{thm:al_amp_site_entropy} gives the equality.
\end{proof}

\begin{theorem}[The local unitary and the entropy comparison on a region]
    \label{thm:al_amp_supported_polar}
    \lean{QuantumCircuit.embedOp_val_eq_kronecker_one_submatrix,
        QuantumCircuit.kronecker_one_submatrix_mem_supportedOperators,
        TNLean.PEPS.AreaLaw.exists_supported_unitary_of_good_realization,
        TNLean.PEPS.AreaLaw.mutualInformation_le_inv_pow_of_supported_unitary}
    \leanok
    \uses{thm:area_law_amplification_unitary_error,thm:al_amp_entropy_comparison_scale}
    On a finite set of sites with local dimension $q$, let $R_0$ and $L'$ act on
    a region $S$. Under the hypotheses of the deterministic steps after the
    choice of a good realization, there is a unitary acting on $S$ with
    $\|U\varphi-\eta\|\le a+\sqrt{2a+6\varepsilon+9\varepsilon^2}$. If a unitary acts
    on a region disjoint from $X\cup J$, $I_\eta(X:J)=0$,
    $\|U\varphi-\eta\|\le C_1n^{-K/2}$ with $K\ge4(M+2)$ and
    $\log\dim\mathcal H_X\le C_2n^2$, then every part of $X$ has mutual
    information at most $8\sqrt{C_1}\,(1+C_2)\,n^{-M}$ with $J$ in $\varphi$.
\end{theorem}
\begin{proof}
    \leanok
    An operator acting on $S$ is $A\otimes I$ in the coordinates of $S$ and its
    complement, and an operator acting on a region disjoint from $X\cup J$ is
    $I\otimes A$ in the coordinates of $X\cup J$ and its complement. Applying the
    polar correction in the first coordinates and the entropy comparison in the
    second gives both assertions.
\end{proof}

\begin{theorem}[The local unitary and the entropy comparison on the enlarged sites]
    \label{thm:al_amp_enlarged_steps}
    \lean{TNLean.PEPS.AreaLaw.exists_ancilla_unitary_of_good_realization,
        TNLean.PEPS.AreaLaw.mutualInformation_le_of_ancilla_unitary}
    \leanok
    \uses{thm:al_amp_ancilla_sites,thm:al_amp_supported_polar}
    Work on the physical sites enlarged by the two copies $s',e'$ of $X$. Let the
    restoring operator $R_0$ act on $N\sqcup s'$ and the truncated product $L'$ act
    on a physical region $N$. Under the hypotheses of the deterministic steps after
    the choice of a good realization, with the physical operators tensored with
    the identity on both copies, there is a unitary acting on $N\sqcup s'$ with
    $\|U\varphi-\eta\|\le a+\sqrt{2a+6\varepsilon+9\varepsilon^2}$. If $X\subseteq N$,
    $\varphi$ is the copied vector built from unit blank vectors, $\varphi$ and
    $\eta$ have unit norm, $\|U\varphi-\eta\|\le C_1n^{-K/2}$ with $K\ge4(M+2)$ and
    $|X|\log q\le C_2n^2$, then every part of $X$ has mutual information at most
    $8\sqrt{C_1}\,(1+C_2)\,n^{-M}$ in $\Omega$ with every physical region avoiding
    $N$.
\end{theorem}
\begin{proof}
    \leanok
    Relabelling the enlarged configurations as physical configurations together with
    the configurations of $s'$ and $e'$ makes
    Theorem~\ref{thm:al_amp_supported_polar} applicable on the enlarged sites. For the entropy
    comparison, $\eta$ is $\Omega$ tensored with a unit ancillary vector, so it has
    no mutual information between $e'$ and a physical region, and the copy of
    each $X'\subseteq X$ in $e'$ carries $I_\Omega(X':J)$ by
    Theorem~\ref{thm:al_amp_ancilla_sites}.
\end{proof}

\begin{theorem}[The amplification radius is a power of the scale]
    \label{thm:al_amp_radius_bound}
    \lean{TNLean.PEPS.AreaLaw.exists_amplificationRadius_le}
    \leanok
    \uses{def:area_law_amplification_time_radius}
    Let $g>0$, $C'\ge0$, $C_0\ge0$, $0<\alpha$, $0<\varepsilon\le2/5$ and let $K$ be
    an integer. There is $C$ such that at every scale $n\ge1$ and for every
    $0\le\mathcal I\le C_0n^{1-\varepsilon}$ the radius of
    Definition~\ref{def:area_law_amplification_time_radius}, chosen with
    $w=n^{3/5}$, is at most $Cn^{(1-\varepsilon)/\alpha}$.
\end{theorem}
\begin{proof}
    \leanok
    Bounding $n^{3/5}$ and $\log n$ by multiples of $n^{1-\varepsilon}$ gives
    $C'(t_*+K\log n)\le C'Bn^{1-\varepsilon}$, and raising to the power $1/\alpha$
    absorbs the rounding into $n^{(1-\varepsilon)/\alpha}\ge1$.
\end{proof}

\begin{theorem}[Proposition 10.2: amplification]
    \label{thm:area_law_amplification}
    \lean{TNLean.PEPS.AreaLaw.amplificationBound}
    \leanok
    \uses{def:area_law_amplification_bound}
    In Proposition~10.2 of \cite{OpenAI2026AreaLaw}, for every $q,R,J$ and every
    gap $\Delta>0$ the amplification bound holds.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:area_law_amplification_of_radius_output,thm:al_amp_restoring_transport,
        thm:al_amp_constraint_word_lists,thm:al_lattice_excited_decay,
        thm:al_lattice_constraint_omission,thm:al_lattice_product_truncation,
        thm:area_law_amplification_scale_bounds,thm:al_amp_typical_projectors,
        thm:al_amp_ancilla_sites,thm:al_amp_supported_polar,thm:al_amp_enlarged_steps,
        thm:al_amp_radius_bound}
    If $q=0$ or $J<0$, a gapped ground vector exists only on the empty domain,
    where every part of the target is empty. Otherwise fix $p=10^6$,
    $\alpha=p/(p+1)$ and $K=4(M+2)$, and take the constants of the excited-state
    decay, the omission and truncation estimates, the good realization and the
    typical projectors. At the scale $n$, let $w=n^{3/5}$ and
    $\mathcal I=I_\Omega(X:\Lambda\setminus S_0)$, and choose $t_*$ and $r$ as in
    Definition~\ref{def:area_law_amplification_time_radius}; the radius bound holds by
    Theorem~\ref{thm:al_amp_radius_bound}. With blank vectors on the ancillas,
    Theorem~\ref{thm:al_amp_restoring_transport} gives the restoring operator,
    the column operator and the vectors $\varphi,\eta$ with the ancillas as
    spectators, at failure probability $\delta=n^{-2K}$. On the clock words of
    time $t_*$, the excited-state decay bounds the excited component of
    $\zeta=R_0\varphi$, the omission estimate applied through the column identity
    bounds the omitted events, and the truncation estimate bounds the radius
    truncation; Theorem~\ref{thm:al_amp_constraint_word_lists} identifies the
    retained products of the two estimates. A good realization makes all three
    at most $n^{-K}$. On the enlarged sites, $R_0$ acts on $S_0\sqcup s'$ and the
    truncated product on $N_{2r}^\Lambda(S_0)$, so
    Theorem~\ref{thm:al_amp_supported_polar} gives a unitary on
    $N_{2r}^\Lambda(S_0)\sqcup s'$ with $\|U\varphi-\eta\|\le9n^{-K/2}$. For
    $J$ avoiding $N_{2r}^\Lambda(S_0)$, the vector $\eta$ has no mutual
    information between $e'$ and $J$, and the copy of each $X'\subseteq X$ in
    $e'$ carries $I_\Omega(X':J)$ by Theorem~\ref{thm:al_amp_ancilla_sites}.
    Since $\log\dim e'=|X|\log q\le C_0n^2\log q$, the entropy comparison gives
    $I_\Omega(X':J)\le24(1+C_0\log q)\,n^{-M}$. The cut-clearance clause is
    Theorem~\ref{thm:area_law_amplification_of_radius_output}.
\end{proof}
