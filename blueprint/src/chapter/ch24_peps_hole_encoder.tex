\section{Tag encodings with regional zero resets}
\label{sec:peps_hole_encoder}

Let $V$ be a finite site set with finite coordinate sets $A_v$ and fixed
$z\in\prod_{v\in V}A_v$. Write $\mathcal H_R=\C^{\prod_{v\in R}A_v}$ and
$\ket{0_R}=\bigotimes_{v\in R}\ket{z_v}$. For nested regions
$R_0\subseteq\cdots\subseteq R_{n-1}$ and subspaces $S_j\subseteq\mathcal H_{R_j}$,
let $\widetilde S_j$ be the projected images of
Definition~\ref{def:peps_nested_cylinder_images} and set
$\mathcal E=\sum_{j<n}\mathcal C_{R_j}(S_j)$. This tag-and-reset map realizes the
\href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/05-frames.tex\#L50-L69}{adaptive hole encoding}
of the September 24, 2026 polynomial-PEPS manuscript.

\begin{definition}[Coordinate reset]
    \label{def:peps_coordinate_reset}
    \lean{TNLean.PEPS.coordinateResetMatrix}
    \leanok
    For finite $A$, $a_0\in A$, and $v\in\C^A$, define
    $Z_{a_0}(v)=\ket{a_0}\bra v$, with $(a,b)$-entry
    $\delta_{a,a_0}\overline{v(b)}$.
\end{definition}

\begin{lemma}[Reset Gram operators and basis projectors]
    \label{thm:peps_coordinate_reset_gram}
    \lean{
        TNLean.PEPS.coordinateResetMatrix_gram,
        TNLean.PEPS.coordinateRangeProjector_eq_sum_basis
    }
    \leanok
    \uses{def:peps_coordinate_reset}
    For any (not necessarily normalized) $v\in\C^A$ and any orthonormal basis
    $(b_k)_{k\in B}$ of a subspace $S\subseteq\C^A$,
    \begin{align}
        Z_{a_0}(v)^\dagger Z_{a_0}(v) & =\ketbra v v,
        \label{eq:peps_reset_gram}                                       \\
        P_S                           & =\sum_{k\in B}\ketbra{b_k}{b_k}.
        \label{eq:peps_inside_basis_projector}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    By Definition~\ref{def:peps_coordinate_reset},
    $(Z_{a_0}(v)^\dagger Z_{a_0}(v))_{a,b}=v(a)\overline{v(b)}=(\ketbra v v)_{a,b}$,
    giving (\ref{eq:peps_reset_gram}); orthonormal expansion
    $P_S\psi=\sum_{k\in B}\langle b_k,\psi\rangle b_k$ gives
    (\ref{eq:peps_inside_basis_projector}).
\end{proof}

\begin{definition}[Tag register and zero-reset encoding]
    \label{def:peps_hole_encoder}
    \lean{
        TNLean.PEPS.HoleEncoderTag,
        TNLean.PEPS.holeEncoderBasis,
        TNLean.PEPS.holeEncoder,
        TNLean.PEPS.holeEncoder_apply,
        TNLean.PEPS.holeEncoder_eq_zero_of_not_reset,
        TNLean.PEPS.holeEncoder_eq_zero_of_complement_ne
    }
    \leanok
    \uses{def:peps_nested_cylinder_images,def:peps_coordinate_reset}
    Set $d_j=\dim\widetilde S_j$, fix an orthonormal basis
    $(v_{j\ell})_{0\leq\ell<d_j}$ of each $\widetilde S_j$, and set
    $T=\{(j,\ell):0\leq j<n,\ 0\leq\ell<d_j\}$ and $\mathcal J=\C^T$. Define
    $K\colon\mathcal H_V\to\mathcal J\otimes\mathcal H_V$ by
    \begin{align}
        K & =\sum_{j<n}\sum_{\ell<d_j}\ket{j,\ell}\otimes
        \left[(\ket{0_{R_j}}\bra{v_{j\ell}})_{R_j}
            \otimes I_{V\setminus R_j}\right],
        \label{eq:peps_hole_encoder}
    \end{align}
    with matrix entries for $x,y\in\prod_{v\in V}A_v$ given by
    \begin{align}
        K_{((j,\ell),x),y}
        \label{eq:peps_hole_encoder_coefficient}
         & =
        \begin{cases}
            \overline{v_{j\ell}(y\vert_{R_j})},
               & x\vert_{R_j}=z\vert_{R_j},\quad
            x\vert_{V\setminus R_j}=y\vert_{V\setminus R_j}, \\
            0, & \text{otherwise}.
        \end{cases}
    \end{align}
    At a fixed tag $t=(j,\ell)$, its summand is depicted below:
    \input{figures/peps_hole_encoder}
\end{definition}

\begin{lemma}[Dimension of the tag register]
    \label{thm:peps_hole_encoder_tag_dimension}
    \lean{
        TNLean.PEPS.finrank_coordinateSubspaceES,
        TNLean.PEPS.card_holeEncoderTag,
        TNLean.PEPS.card_holeEncoderTag_le
    }
    \leanok
    \uses{def:peps_hole_encoder}
    Identifying coordinate spaces with their Euclidean Hilbert spaces preserves
    inside-subspace dimensions, and
    \begin{align}
        \dim\mathcal J=|T|
         & =\sum_{j<n}\dim\widetilde S_j
        \leq\sum_{j<n}\dim S_j.
        \label{eq:peps_hole_encoder_tag_dimension}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_nested_cylinder_orthogonalization}
    Since $T=\bigsqcup_{j<n}\{0,\dots,d_j-1\}$ with $d_j=\dim\widetilde S_j\leq\dim S_j$
    (Lemma~\ref{thm:peps_nested_cylinder_orthogonalization}), summing over
    $j<n$ yields (\ref{eq:peps_hole_encoder_tag_dimension}).
\end{proof}

\begin{lemma}[The Gram operator of the encoding]
    \label{thm:peps_hole_encoder_gram}
    \lean{TNLean.PEPS.holeEncoder_gram}
    \leanok
    \uses{def:peps_hole_encoder}
    The encoding $K$ of (\ref{eq:peps_hole_encoder}) satisfies
    \begin{align}
        K^\dagger K
         & =\sum_{j<n}P_{\widetilde S_j}\otimes I_{V\setminus R_j}
        =P_{\mathcal E}.
        \label{eq:peps_hole_encoder_gram}
    \end{align}
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_coordinate_reset_gram,thm:peps_nested_cylinder_projector_sum}
    Tag orthogonality eliminates cross terms in (\ref{eq:peps_hole_encoder}); by
    (\ref{eq:peps_reset_gram})--(\ref{eq:peps_inside_basis_projector}), each
    $(j,\ell)$-summand has Gram operator
    $\ketbra{v_{j\ell}}{v_{j\ell}}\otimes I_{V\setminus R_j}$, summing over
    $\ell<d_j$ to $P_{\widetilde S_j}\otimes I_{V\setminus R_j}$ and over $j<n$
    to $P_{\mathcal E}$ by Lemma~\ref{thm:peps_nested_cylinder_projector_sum}.
\end{proof}

\begin{lemma}[Contraction and projected norm]
    \label{thm:peps_hole_encoder_norm}
    \lean{
        TNLean.PEPS.holeEncoder_opNorm_le_one,
        TNLean.PEPS.holeEncoder_norm_le,
        TNLean.PEPS.holeEncoder_norm_eq_projector
    }
    \leanok
    \uses{def:peps_hole_encoder}
    For every $\psi\in\mathcal H_V$,
    \begin{align}
        \|K\psi\| & =\|P_{\mathcal E}\psi\|\leq\|\psi\|,
        \label{eq:peps_hole_encoder_norm}
    \end{align}
    and $\|K\|\leq1$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_hole_encoder_gram}
    By (\ref{eq:peps_hole_encoder_gram}) and $P_{\mathcal E}^2=P_{\mathcal E}^\dagger=P_{\mathcal E}$,
    \begin{align}
        \|K\psi\|^2
         & =\langle\psi,K^\dagger K\psi\rangle
        =\langle\psi,P_{\mathcal E}\psi\rangle
        =\|P_{\mathcal E}\psi\|^2\leq\|\psi\|^2,\notag
    \end{align}
    whence (\ref{eq:peps_hole_encoder_norm}) and $\|K\|\leq1$.
\end{proof}

\begin{lemma}[Norm preservation on the cylinder span]
    \label{thm:peps_hole_encoder_norm_on_span}
    \lean{TNLean.PEPS.holeEncoder_norm_eq_of_mem}
    \leanok
    \uses{def:peps_hole_encoder}
    For $K$ as in (\ref{eq:peps_hole_encoder}), every $\psi\in\mathcal E$
    satisfies $\|K\psi\|=\|\psi\|$, matching the
    \href{https://github.com/openai/math/blob/adc7f1241b42e322a6451854ab7e4b4c146bf78a/preprints/Polynomial-PEPS-approximation-of-gapped-square-grid-ground-states-September-24-2026/build/sections/05-frames.tex\#L62-L64}{one-hole encoding construction}.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_hole_encoder_norm,def:peps_coordinate_projector}
    For $\psi\in\mathcal E$, $P_{\mathcal E}\psi=\psi$
    (Definition~\ref{def:peps_coordinate_projector}), so (\ref{eq:peps_hole_encoder_norm})
    gives $\|K\psi\|=\|P_{\mathcal E}\psi\|=\|\psi\|$.
\end{proof}

\begin{definition}[Identity encoding]
    \label{def:peps_identity_hole_encoder}
    \lean{
        TNLean.PEPS.identityHoleEncoder,
        TNLean.PEPS.identityHoleEncoder_mulVec
    }
    \leanok
    For a singleton tag set $\{\ast\}$ (used when an outer square has empty
    physical sample), define $K_{\mathrm{id}}\psi=\ket{\ast}\otimes\psi$.
\end{definition}

\begin{lemma}[Identity Gram operator and norm]
    \label{thm:peps_identity_hole_encoder}
    \lean{
        TNLean.PEPS.identityHoleEncoder_gram,
        TNLean.PEPS.identityHoleEncoder_norm
    }
    \leanok
    \uses{def:peps_identity_hole_encoder}
    The map $K_{\mathrm{id}}$ satisfies $K_{\mathrm{id}}^\dagger K_{\mathrm{id}}=I$
    and $\|K_{\mathrm{id}}\psi\|=\|\psi\|$ for all $\psi\in\mathcal H_V$.
\end{lemma}
\begin{proof}
    \leanok
    By Definition~\ref{def:peps_identity_hole_encoder},
    $K_{\mathrm{id}}^\dagger K_{\mathrm{id}}=\braket{\ast}{\ast}I^\dagger I=I$, so
    $\|K_{\mathrm{id}}\psi\|^2=\langle\psi,\psi\rangle=\|\psi\|^2$.
\end{proof}

\begin{lemma}[Zero encodings]
    \label{thm:peps_hole_encoder_zero}
    \lean{
        TNLean.PEPS.holeEncoder_empty,
        TNLean.PEPS.holeEncoder_zero_spaces
    }
    \leanok
    \uses{def:peps_hole_encoder}
    If $n=0$ or $\forall j<n,\ S_j=0$, then $T=\emptyset$ and $K=0$.
\end{lemma}
\begin{proof}
    \leanok
    \uses{thm:peps_hole_encoder_tag_dimension}
    By (\ref{eq:peps_hole_encoder_tag_dimension}), either hypothesis gives
    $|T|=0$, so $T=\emptyset$ and $K=0$.
\end{proof}

\begin{remark}[Empty families and empty regions]
    When $n=0$, $T=\emptyset$, $K=0$, and $\mathcal E=0$, consistent with
    (\ref{eq:peps_hole_encoder_gram}); any $\widetilde S_j=0$ contributes
    $d_j=0$ tags; for $R_j=\emptyset$, $\mathcal H_{R_j}=\C$ and
    $V\setminus R_j=V$; whereas $K_{\mathrm{id}}$ has $|T|=1$ and acts as $I$ on
    $\mathcal H_V$.
\end{remark}
