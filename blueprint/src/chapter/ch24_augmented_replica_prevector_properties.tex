\section{The initial vector in augmented site coordinates}

Let $V$ be a finite set of physical sites, let $q,d,k\ge0$ be integers,
and put $X=\{0,\ldots,q-1\}^V$ and $D=\{0,\ldots,d-1\}$.
The augmented site set is $J=V\sqcup\{\mathrm{false},\mathrm{true}\}$, with
local dimension $q$ at each physical site and $d$ at each auxiliary site.
Write $\mathcal X_J$ for the one-copy configuration space.
Let
\begin{align*}
    E_k:\mathcal X_J^k\longrightarrow X^k\times(D^k\times D^k)
\end{align*}
be the coordinate bijection of Definition~\ref{def:al_scan_auxiliary_coordinates}
identifying the auxiliary sites $\mathrm{false}$ and $\mathrm{true}$ with the registers $C$ and $R$.
For $\Omega\in\mathbb C^X$ and an irreducible representation $\lambda$ of $S_k$, put
\begin{align*}
    v_{k,\lambda}               & =\Omega^{\otimes k}\otimes
    (I_C\otimes\Pi_\lambda)U_k, &
    \xi_{k,\lambda}             & =v_{k,\lambda}\circ E_k,
\end{align*}
where $U_k$ is the $k$-fold tensor power of the unnormalized maximally entangled pair on $D\times D$, following the conventions of \cite[Section~7]{OpenAI2026AreaLaw}.

\begin{theorem}[Full copy symmetry of the augmented initial vector]
    \label{thm:al_augmented_prevector_symmetry}
    \lean{TNLean.PEPS.AreaLaw.Scan.symProj_mulVec_augmentedReplicaPrevector}
    \leanok
    \uses{def:al_scan_auxiliary_coordinates}
    Let $P_{\mathrm{sym}}$ be the average over simultaneous permutations of the
    $k$ copies of $\mathcal X_J$. Then
    \begin{align*}
        P_{\mathrm{sym}}\xi_{k,\lambda}=\xi_{k,\lambda}.
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    Simultaneous permutation of the $k$ copies fixes $\Omega^{\otimes k}$ and $U_k$,
    and commutes with the central projection $\Pi_\lambda$ on the $R$ factor.
    Thus $v_{k,\lambda}$ is invariant under the diagonal $S_k$-action.
    Because $E_k$ is equivariant with respect to copy permutations,
    $\xi_{k,\lambda}$ is invariant under every permutation of the $k$ copies,
    and averaging over $S_k$ gives $P_{\mathrm{sym}}\xi_{k,\lambda}=\xi_{k,\lambda}$.
\end{proof}

\begin{theorem}[Isotypic projection at each auxiliary site]
    \label{thm:al_augmented_prevector_auxiliary_label}
    \lean{TNLean.PEPS.AreaLaw.Scan.labelProj_auxiliary_mulVec_augmentedReplicaPrevector}
    \leanok
    \uses{def:al_scan_auxiliary_coordinates}
    For each auxiliary site $b\in\{\mathrm{false},\mathrm{true}\}$, let
    $\Pi_{\{b\},\lambda}$ denote the central projection for the
    $S_k$-action on the $k$ copies at site $b$. Then
    \begin{align*}
        \Pi_{\{b\},\lambda}\xi_{k,\lambda}=\xi_{k,\lambda}.
    \end{align*}
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:al_augmented_subsystem_action}
    Put $w=(I_C\otimes\Pi_\lambda)U_k$.
    Because each pair in $U_k$ is symmetric between $C$ and $R$, the central
    projections $\Pi_\lambda\otimes I_R$ and $I_C\otimes\Pi_\lambda$ agree on $U_k$.
    Since they commute and are idempotent, each fixes $w$.
    Tensoring with $\Omega^{\otimes k}$ shows that both projections fix $v_{k,\lambda}$.
    By Theorem~\ref{thm:al_augmented_subsystem_action}, the bijection $E_k$
    identifies the two auxiliary copy actions with the single-site actions at
    $\mathrm{false}$ and $\mathrm{true}$, transporting both fixed-vector identities to $\xi_{k,\lambda}$.
\end{proof}
