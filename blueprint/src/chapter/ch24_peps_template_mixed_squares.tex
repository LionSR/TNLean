\subsection{Mixed-square counts for template shells}

% This is the mixed-square prerequisite, not the complete entropy lemma.
\begin{lemma}[Mixed squares of finite unions and differences]
    \label{thm:area_law_mixed_dyadic_boolean}
    \lean{TNLean.PEPS.AreaLaw.Geometry.mem_latticeDyadicCell_iff_bounds,
        TNLean.PEPS.AreaLaw.Geometry.mixedDyadicIndices_empty,
        TNLean.PEPS.AreaLaw.Geometry.mixedDyadicIndices_biUnion_subset,
        TNLean.PEPS.AreaLaw.Geometry.card_mixedDyadicIndices_biUnion_le,
        TNLean.PEPS.AreaLaw.Geometry.mixedDyadicIndices_sdiff_subset,
        TNLean.PEPS.AreaLaw.Geometry.card_mixedDyadicIndices_sdiff_le}
    \leanok
    \uses{def:area_law_lattice_dyadic_square,def:area_law_mixed_dyadic_square}
    Set $u=2^k$. The integer sites of $C_{k,z}$ satisfy
    $uz_i\le x_i<uz_i+u$ in each coordinate. For any finite lattice
    sets and finite unions,
    \begin{align}
        M_k(\varnothing)             & =\varnothing,                \\
        M_k\Bigl(\bigcup_a S_a\Bigr) & \subseteq\bigcup_a M_k(S_a), \\
        M_k(S\setminus U)            & \subseteq M_k(S)\cup M_k(U).
    \end{align}
    In particular, the corresponding mixed-square counts are subadditive.
\end{lemma}
\begin{proof}\leanok
    A site of $C_{k,z}$ inside a union belongs to some $S_a$, whereas a site
    outside the union lies outside every $S_a$, so $C_{k,z}$ is mixed for $S_a$.
    For $S\setminus U$, a site inside $S\setminus U$ lies in $S$ and outside
    $U$, while a site outside $S\setminus U$ either lies outside $S$ or belongs
    to $U$; hence $C_{k,z}$ is mixed for either $S$ or $U$.
\end{proof}

\begin{lemma}[Row displacement under integer dilation]
    \label{thm:area_law_template_dilated_row_displacement}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.exists_sample_in_row_within,
        TNLean.PEPS.AreaLaw.Geometry.Template.exists_dilation_in_row_within}
    \leanok
    For a sampled template piece $S_a=P_a\cap\Z^2$, let
    $(x,y)\in(S_a)_j$. If row $z$ of $(S_a)_j$ is nonempty and $|z-y|\le d$,
    then there is $(w,z)\in(S_a)_j$ with $|w-x|\le d$. The same holds for
    the undilated sample $S_a$.
\end{lemma}
\begin{proof}\leanok
    The four pairs of integer supporting inequalities describe the rows of
    $S_a$. Since their slopes lie in $\{-1,0,1\}$, clamping $x$ into a nonempty
    row $z$ shifts the horizontal coordinate by at most $d$. For the dilated
    piece $(S_a)_j$, first clamp the source row of $S_a$ into the window of
    radius $j$ around $z$; because the nonempty rows of $S_a$ form a
    consecutive interval, this row is nonempty. Applying the undilated estimate
    and clamping the horizontal coordinate within radius $j$ yields $(w,z)$.
\end{proof}

\begin{lemma}[Mixed squares of one dilated piece]
    \label{thm:area_law_template_piece_mixed_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.card_mixedDyadicIndices_sample_dilation_le}
    \leanok
    \uses{def:area_law_mixed_dyadic_square,thm:area_law_template_dilated_row_displacement}
    For every sampled template piece $S_a$ of diameter at most $s_0$,
    every integer $j\ge0$, and $u=2^k$,
    \begin{align}
        u|M_k((S_a)_j)|\le24(s_0+j+u).
    \end{align}
    This is the single-piece mixed-square bound in the proof of
    \cite[Lemma~9.4]{OpenAI2026AreaLaw}.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_template_dilated_row_displacement}
    In an interior block of $u$ rows, every row of $(S_a)_j$ is nonempty.
    Moving an inside site and the two endpoints of a reference row by at most $u$
    shows that any mixed square has column index within $2$ of an endpoint's
    column index, leaving at most $10$ columns in each interior row block.
    In the first and last occupied row blocks, every mixed square meets the
    bounding box of $(S_a)_j$. If $N_x$ and $N_y$ denote the numbers of column
    and row blocks meeting this bounding box, then
    \begin{align}
        |M_k((S_a)_j)| & \le2N_x+10N_y, \\
        uN_x,\ uN_y    & \le2s_0+2j+2u,
    \end{align}
    which gives the stated bound (and when $S_a$ is empty, $M_k((S_a)_j)$ is
    empty).
\end{proof}

\begin{lemma}[Mixed squares of a template and its shell]
    \label{thm:area_law_template_shell_mixed_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.card_mixedDyadicIndices_dilation_pieceCount_le,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_mixedDyadicIndices_le,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_mixedDyadicIndices_dilation_le,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_mixedDyadicIndices_shell_le}
    \leanok
    \uses{thm:area_law_template_piece_mixed_count,thm:area_law_mixed_dyadic_boolean}
    Let $T$ be a template with $N_0$ pieces. For every $j\ge0$,
    \begin{align}
        u|M_k(T_j)|\le24N_0(s_0+j+u).
    \end{align}
    If $C_{\rm tpl}\ge24$ and $0\le j\le s_0$, then
    \begin{align}
        u|M_k(T)|              & \le n+24N_0u, \\
        u|M_k(T_j)|            & \le2n+24N_0u, \\
        u|M_k(T_j\setminus T)| & \le3n+48N_0u.
    \end{align}
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_template_piece_mixed_count,thm:area_law_mixed_dyadic_boolean}
    Integer dilation commutes with the finite union of sampled
    pieces. Summing the single-piece mixed-square bounds and using
    $24N_0(s_0+1)\le n$ gives the bounds for $T$ and $T_j$; the bound for
    $T_j\setminus T$ follows by subadditivity.
\end{proof}

\begin{lemma}[Selected shell squares below the cap]
    \label{thm:area_law_template_shell_selected_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.card_cappedDyadicPartition_shell_below_cap_le}
    \leanok
    \uses{thm:area_law_capped_mixed_parents,thm:area_law_template_shell_mixed_count}
    Suppose $C_{\rm tpl}\ge24$, $0\le j\le s_0$, $k<K$, and $u=2^k\le s_0$.
    The number $N_k$ of selected squares of side $u$ in the capped partition
    of $T_j\setminus T$ satisfies $uN_k\le14n$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_capped_mixed_parents,thm:area_law_template_shell_mixed_count}
    Each selected square at level $k<K$ has a mixed parent at level $k+1$, and
    each parent has four children, so $uN_k\le2(2u)|M_{k+1}(T_j\setminus T)|$.
    The shell mixed-square bound and $24N_0u\le n$ give
    $uN_k\le2(3n+96N_0u)\le14n$.
\end{proof}

\begin{lemma}[Selected core squares at and below the cap]
    \label{thm:area_law_template_core_selected_count}
    \lean{TNLean.PEPS.AreaLaw.Geometry.Template.card_cappedDyadicPartition_at_cap_le,
        TNLean.PEPS.AreaLaw.Geometry.Template.card_cappedDyadicPartition_below_cap_le}
    \leanok
    \uses{thm:area_law_template_area,thm:area_law_capped_dyadic_partition,
        thm:area_law_capped_mixed_parents,thm:area_law_template_shell_mixed_count}
    Let $T$ be a template with $C_{\rm tpl}\ge24$, and suppose
    $2^K\le s_0<2^{K+1}$. Write $N_k$ for the number of selected squares
    of side $2^k$ in the capped partition of $T$. Then
    \begin{align}
        2^K N_K & \le18n,             \\
        2^k N_k & \le4n \qquad (k<K).
    \end{align}
    This establishes the core-covering counts in the proof of Lemma~9.4.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:area_law_template_area,thm:area_law_capped_dyadic_partition,
        thm:area_law_capped_mixed_parents,thm:area_law_template_shell_mixed_count}
    The selected squares at the cap $K$ are pairwise disjoint inside $T$, so
    $4^K N_K\le |T|\le9ns_0\le18n2^K$.
    For $k<K$, every selected square has a mixed parent with four children, and
    $2^{k+1}\le2^K\le s_0$, so
    \begin{align}
        2^k N_k
         & \le2\bigl(2^{k+1}|M_{k+1}(T)|\bigr)   \\
         & \le2\bigl(n+24N_0 2^{k+1}\bigr)\le4n,
    \end{align}
    using $24N_0s_0\le n$.
\end{proof}
