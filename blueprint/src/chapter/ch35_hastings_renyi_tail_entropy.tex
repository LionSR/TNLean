\section{Rényi entropy under geometric tail constraints}
\label{sec:hastings_renyi_tail_entropy}

% Hastings, arXiv:0705.2024v4, law16.tex, lines 392--410 (end of the proof of
% Lemma 2).

The proof of \cite[Lemma~2]{Hastings2007AreaLaw} ends by extending the entropy
bound of Theorem~\ref{thm:hastings_geometric_tail_entropy} to the Rényi
entropies $S_\alpha=(1-\alpha)^{-1}\ln\sum_ap_a^\alpha$. Since
$S_\alpha\le S$ for $\alpha\ge1$, the nontrivial regime is $0<\alpha<1$ with
convergent geometric ratio $r=(D^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$.

\begin{theorem}[Power sums on a finite set]
    \label{thm:hastings_power_sum_card}
    \lean{TNLean.MPS.AreaLaw.Hastings.sum_rpow_le_card_rpow_mul_sum_rpow}
    \leanok
    Let $p_1,\dots,p_c\ge0$ and $0\le\alpha\le1$. Then
    \begin{align}
        \sum_{i=1}^cp_i^\alpha\le c^{1-\alpha}\Bigl(\sum_{i=1}^cp_i\Bigr)^\alpha.
        \label{eq:hastings_power_sum_card}
    \end{align}
\end{theorem}
\begin{proof}\leanok
    Concavity of $t\mapsto t^\alpha$ on $[0,\infty)$ gives
    $c^{-1}\sum_{i=1}^cp_i^\alpha\le\bigl(c^{-1}\sum_{i=1}^cp_i\bigr)^\alpha$,
    and multiplying by $c$ yields~(\ref{eq:hastings_power_sum_card}).
\end{proof}

\begin{theorem}[Rényi entropy under geometric tail constraints]
    \label{thm:hastings_renyi_tail_entropy}
    \lean{
        TNLean.MPS.AreaLaw.Hastings.sum_rpow_le_of_geometric_tails,
        TNLean.MPS.AreaLaw.Hastings.renyiEntropy_le_of_geometric_tails,
        TNLean.MPS.AreaLaw.Hastings.renyiEntropy_le_of_scale_tails
    }
    \leanok
    Let $(p_0,\dots,p_{N-1})$ be a probability vector and let $K\ge1$, $D\ge2$,
    and $x\ge0$ satisfy $\sum_{a\ge KD^{2n}}p_a\le x^{n+1}$ for all $n\ge0$.
    For $0<\alpha<1$ with $r=(D^2)^{1-\alpha}x^\alpha<1$,
    \begin{align}
        \frac1{1-\alpha}\ln\sum_ap_a^\alpha\le\ln K-\frac{\ln(1-r)}{1-\alpha}.
        \label{eq:hastings_renyi_tail_entropy}
    \end{align}
    In particular, for $x=e^{-2/\xi'}$ and $K=kD^{2m+2}$ ($k\ge1$, $m\in\N$)
    with $r=(D^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$,
    \begin{align}
        \frac1{1-\alpha}\ln\sum_ap_a^\alpha\le\ln k+(2m+2)\ln D
        -\frac{\ln\bigl(1-(D^2)^{1-\alpha}e^{-2\alpha/\xi'}\bigr)}{1-\alpha}.
        \label{eq:hastings_renyi_scale_tails}
    \end{align}
    This is the Rényi extension in \cite[Lemma~2, lines 392--410]{Hastings2007AreaLaw}
    under the convergence condition $r<1$ of line 406, with $F(\xi',D)$ replaced
    by the last term of~(\ref{eq:hastings_renyi_scale_tails}) (dropping the
    saturating-block factor $(1-e^{-2/\xi'})^\alpha$ of lines 400--402).
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_power_sum_card}
    Partition $\{0,\dots,N-1\}$ into $B_0=[0,K)$ and
    $B_n=[KD^{2(n-1)},KD^{2n})$ for $n\ge1$. Each $B_n$ has
    $|B_n|\le KD^{2n}$ and $\sum_{a\in B_n}p_a\le x^n$. By
    Theorem~\ref{thm:hastings_power_sum_card},
    \begin{align}
        \sum_ap_a^\alpha=\sum_{n\ge0}\sum_{a\in B_n}p_a^\alpha
        \le\sum_{n\ge0}(KD^{2n})^{1-\alpha}x^{n\alpha}
        =\frac{K^{1-\alpha}}{1-r},
        \notag
    \end{align}
    and taking $(1-\alpha)^{-1}\ln(\cdot)$ gives~(\ref{eq:hastings_renyi_tail_entropy}).
    Substituting $x=e^{-2/\xi'}$ and $\ln K=\ln k+(2m+2)\ln D$
    gives~(\ref{eq:hastings_renyi_scale_tails}).
\end{proof}

\begin{theorem}[Rényi entropy from mixtures of bounded Schmidt rank]
    \label{thm:hastings_renyi_schmidt_rank_stage}
    \lean{TNLean.MPS.AreaLaw.Hastings.renyiEntropy_le_of_schmidtRank_mixtures,
        TNLean.MPS.AreaLaw.Hastings.renyiEntropy_eq_log_sum_eigenvalues₀_rpow}
    \leanok
    Let $\Psi$ be a bipartite unit vector with first-factor marginal $\rho$,
    and let $k\ge1$, $D\ge2$, $\xi'>0$, and $c\ge1$. Suppose that for every
    $s\in\N$ with $ce^{-2s/\xi'}<1$ there is a positive-weight mixture $\sigma_s$
    of pure states of Schmidt rank at most $kD^{2s}$ with
    $(\operatorname{tr}\sigma_s)^{-1}\langle\Psi,\sigma_s\Psi\rangle\ge1-ce^{-2s/\xi'}$.
    Then for $0<\alpha<1$ with $r=(D^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$,
    \begin{align}
        S_\alpha(\rho)\le\ln k+\xi'\ln(c)\ln D+4\ln D-\frac{\ln(1-r)}{1-\alpha}.
        \label{eq:hastings_renyi_schmidt_stage}
    \end{align}
    This is the single-cut Rényi stage of \cite[Lemma~2, lines 333--356 and
        392--410]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_renyi_tail_entropy, thm:hastings_bootstrap_schmidt_tail}
    By Theorem~\ref{thm:hastings_bootstrap_schmidt_tail}, the decreasing
    eigenvalues of $\rho$ satisfy the scale-$m'+1+n$ tail constraints of
    Theorem~\ref{thm:hastings_renyi_tail_entropy} for some $m'\in\N$ with
    $2m'<\xi'\ln c+2$, so~(\ref{eq:hastings_renyi_scale_tails}) yields
    (\ref{eq:hastings_renyi_schmidt_stage}).
\end{proof}

\begin{theorem}[The Rényi bound from the factorization at one cut]
    \label{thm:hastings_renyi_bootstrap_factorization_at}
    \lean{TNLean.MPS.AreaLaw.Hastings.renyiEntropy_le_of_overlap_schmidtRank_mixture_of_isFactorizationAt,
        TNLean.MPS.AreaLaw.Hastings.renyiEntropy_cutMatrix_nonpos_of_one}
    \leanok
    \uses{def:hastings_factorization_at}
    Let $\Omega$ be a unit vector on a chain of $L$ sites with local dimension
    $D\ge2$, satisfying Definition~\ref{def:hastings_factorization_at} at cut
    $j$ and every length $m$ with error $C_1e^{-m/\xi'}$ ($\xi'>0$, $C_1\ge1$).
    Let $\rho$ be a unit-trace mixture of pure states of Schmidt rank at most
    $k\ge1$ across $j$ with $P=\langle\Omega,\rho\Omega\rangle>0$. For
    $0<\alpha<1$ with $r=(D^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$,
    \begin{align}
        S_\alpha(\rho^0_{1,j})\le\ln k+\xi'\ln\bigl(2C_1^2/P\bigr)\ln D+4\ln D
        -\frac{\ln(1-r)}{1-\alpha},
        \label{eq:hastings_renyi_bootstrap}
    \end{align}
    while for $D=1$ and $0<\alpha<1$, $S_\alpha(\rho^0_{1,j})\le0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_renyi_schmidt_rank_stage,
        thm:hastings_bootstrap_chain_stage}
    As in Theorem~\ref{thm:hastings_bootstrap_chain_stage}, the factorization
    gives mixtures of Schmidt rank at most $kD^{2s}$ with normalized overlap at
    least $1-ce^{-2s/\xi'}$ for $c=2C_1^2/P\ge1$, so
    Theorem~\ref{thm:hastings_renyi_schmidt_rank_stage} gives
    (\ref{eq:hastings_renyi_bootstrap}). For $D=1$, $\rho^0_{1,j}$ has
    single eigenvalue $1$, so $S_\alpha(\rho^0_{1,j})=0$.
\end{proof}

\begin{theorem}[The Rényi form of the bootstrap lemma]
    \label{thm:hastings_renyi_bootstrap}
    \lean{TNLean.MPS.AreaLaw.exists_renyiEntropy_le_of_overlap_schmidtRank_mixture}
    \leanok
    \uses{def:chain_model, thm:hastings_lemma1_endpoint}
    For every range $R\in\N$ there is a constant $K\ge1$ depending only on $R$
    with the following property. Let $H$ be a range-$R$ Hamiltonian on a chain
    of $L$ sites with local dimension $D$ and term norms at most $J\ge0$, with
    unique ground state $\Psi_0$ and gap $\Delta E>0$. Let $\tilde\xi$ be the
    length of Theorem~\ref{thm:hastings_lemma1_endpoint}, and set
    $\xi'=6\tilde\xi$ and $C_1=K\tilde\xi^3$. For $0<\alpha<1$ with
    $r=(D^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$ and any unit-trace mixture $\rho$
    of pure states of Schmidt rank at most $k\ge1$ across cut $j$ with
    $P=\langle\Psi_0,\rho\Psi_0\rangle>0$, the bound
    (\ref{eq:hastings_renyi_bootstrap}) holds.

    This is an auxiliary Rényi estimate at the physical pair of
    Theorem~\ref{thm:hastings_source_lengths}, whose velocity includes a
    gap-dependent term. Theorem~\ref{thm:hastings_renyi_bootstrap_source}
    gives the source form of \cite[Lemma~2, lines 392--410]{Hastings2007AreaLaw}
    with an interaction-proportional velocity and one common length.
    Order $1$ is Theorem~\ref{thm:hastings_bootstrap}, and orders $\alpha>1$
    are Theorem~\ref{thm:hastings_renyi_bootstrap_above_one}
    \cite{gap:hastings07_renyi_bootstrap_orders}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_lemma1_endpoint,
        thm:hastings_renyi_bootstrap_factorization_at,
        thm:hastings_factorization_mono}
    Theorem~\ref{thm:hastings_lemma1_endpoint} and
    Theorem~\ref{thm:hastings_factorization_mono} provide the factorization at
    every $(j,m)$ with $K=\max(K_0,1)$ and $C_1=K\tilde\xi^3\ge1$ (as
    $\tilde\xi\ge20$). Theorem~\ref{thm:hastings_renyi_bootstrap_factorization_at}
    then yields~(\ref{eq:hastings_renyi_bootstrap}) for $D\ge2$, and
    $S_\alpha(\rho^0_{1,j})\le0$ (while the right-hand side is nonnegative) for
    $D=1$.
\end{proof}

\begin{theorem}[The Rényi form of the bootstrap lemma, orders above one]
    \label{thm:hastings_renyi_bootstrap_above_one}
    \lean{TNLean.MPS.AreaLaw.exists_renyiEntropy_le_of_overlap_schmidtRank_mixture_of_one_lt,
        TNLean.MPS.AreaLaw.Hastings.renyiEntropy_cutMatrix_le_regionEntropy_of_one_lt}
    \leanok
    \uses{thm:hastings_bootstrap}
    In the setting of Theorem~\ref{thm:hastings_bootstrap}, with the same
    constant $K$, for every $\alpha>1$,
    \begin{align}
        S_\alpha(\rho^0_{1,j})\le\ln k+\xi'\ln\bigl(2C_1^2/P\bigr)\ln D+F(\xi',D).
        \notag
    \end{align}
    For $\alpha>1$ the convergence condition of
    \cite[line 406]{Hastings2007AreaLaw} holds unconditionally with remainder
    $F(\xi',D)$; together with Theorem~\ref{thm:hastings_renyi_bootstrap} this
    covers all orders of \cite[lines 392--410]{Hastings2007AreaLaw}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_bootstrap}
    For eigenvalues $(p_i)$ and $\alpha>1$, concavity of $\ln$ with weights
    $p_i$ gives $\ln\sum_ip_i^\alpha\ge(\alpha-1)\sum_ip_i\ln p_i$; dividing by
    $1-\alpha<0$ yields $S_\alpha(\rho^0_{1,j})\le S(\rho^0_{1,j})$
    \cite[``Rényi entropy of order above one'']{QICLean2026Blueprint}, and
    Theorem~\ref{thm:hastings_bootstrap} bounds $S(\rho^0_{1,j})$.
\end{proof}


\begin{theorem}[The Rényi bootstrap inequalities at the source length]
    \label{thm:hastings_renyi_bootstrap_source}
    \lean{TNLean.MPS.AreaLaw.exists_renyiEntropy_le_of_overlap_schmidtRank_mixture_source,
        TNLean.MPS.AreaLaw.exists_renyiEntropy_le_of_overlap_schmidtRank_mixture_of_one_lt_source}
    \leanok
    \uses{thm:hastings_source_factorization,thm:hastings_bootstrap_source,
        thm:hastings_renyi_bootstrap_factorization_at}
    Under the Hamiltonian and mixture hypotheses of
    Theorem~\ref{thm:hastings_bootstrap_source}, there is $K\ge1$ depending
    only on $R$ such that, with $s=\max(80c_RJ/\Delta E,80)$,
    $\xi'=6s$ and $C_1=Ks^3$, the following hold.
    For $0<\alpha<1$ and $r=(q^2)^{1-\alpha}e^{-2\alpha/\xi'}<1$,
    \begin{align}
        S_\alpha(\rho^0_{[0,j)})
         & \le\ln k+\xi'\ln(2C_1^2/P)\ln q+4\ln q
        -\frac{\ln(1-r)}{1-\alpha}.
        \label{eq:hastings_renyi_bootstrap_source}
    \end{align}
    For $\alpha>1$, a constant $K\ge1$ depending only on $R$ gives
    the bound (\ref{eq:hastings_bootstrap_source}) for $S_\alpha$ as well.
    These are the conclusions of~\cite[lines 392--410]{Hastings2007AreaLaw}
    at the common source length.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:hastings_source_factorization,thm:hastings_bootstrap_source,
        thm:hastings_renyi_bootstrap_factorization_at,
        thm:hastings_renyi_bootstrap_above_one}
    For $0<\alpha<1$, apply the factorization form of the Rényi
    inequality with $C_1=Ks^3\ge1$. For $\alpha>1$, use $S_\alpha\le S$
    and Theorem~\ref{thm:hastings_bootstrap_source}. For $q=1$ the Rényi
    entropy is zero and both bounds follow directly.
\end{proof}
