\begin{definition}[Edge-blocked insertion algebra correspondence]%
    \label{def:peps_edgeBlockedInsertionAlgebraIsomorphism_property}
    \lean{TNLean.PEPS.IsEdgeBlockedInsertionAlgebraIsomorphism}
    \leanok
    \uses{def:peps_edgeInsertedCoeff}
    For two PEPS tensors $A$ and $B$ and an edge $e$, this property is
    \begin{align}
        \exists \Phi_e:\MN{D_A(e)}
        \simeq_{\mathrm{alg}}
        \MN{D_B(e)},\qquad
        \forall \sigma,\ \forall X\in\MN{D_A(e)},\quad
        C_A(e,\sigma,X)=C_B(e,\sigma,\Phi_e(X)).
        \label{eq:peps_edge_blocked_insertion_alg_iso}
    \end{align}
\end{definition}

\begin{lemma}[Edge-blocked insertion algebra isomorphism]%
    \label{thm:peps_edgeBlockedInsertionAlgebraIsomorphism}
    \lean{TNLean.PEPS.isEdgeBlockedInsertionAlgebraIsomorphism}
    \leanok
    \uses{def:peps_edgeBlockedInsertionAlgebraIsomorphism_property, thm:peps_edgeBlockedThreeSiteInjective, thm:peps_virtualInsertionPhysicalRealization, thm:peps_edgeInsertedCoeff_endpointPhysicalRealization, thm:peps_edgeInsertedCoeff_injective, thm:peps_physical_to_virtual_insertion, thm:peps_sameState_edgeBlockedCoeff_eq, def:peps_edgeInsertedCoeff}
    If the edge-blocked three-site chains of $A$ and $B$ at $e$ are injective, $D_A,D_B>0$, and $A$ and $B$ define the same PEPS state, then (\ref{eq:peps_edge_blocked_insertion_alg_iso}) holds.
    % Scope restriction: the unrestricted Section 3 statement requires deriving
    % positive bond dimensions from injectivity; see
    % docs/paper-gaps/peps_injective_ft_section3_route.tex.
    This is the edge-blocked form of the algebra isomorphism $X\mapsto Y$ in
    \cite[Section~3]{Molnar2018NormalPEPS}:
    % Formula: C_A(e,X;sigma)=C_B(e,Y;sigma) for the corresponding three-site
    % blocked chains.  Ink-to-index: X and Y split the distinguished (1,2) bond;
    % the periodic return closes the remaining virtual contraction, and the three
    % upper legs are sigma_1,sigma_2,sigma_3.
    % Boundary signature: both panels have (V,P)=(0,3).  Source verdict: exact
    % reproduction of paper_normal.tex:255--274 and 565--580 in house style.
    \begin{center}
        \tenkzkernel
        \begin{tenkz}[rows={wire}, cols=4, boundary=periodic]
            \tn[label pos=s, ports={90:physical:$\sigma_1$}]{A_1} &
            \tn[skin=ring]{X} &
            \tn[label pos=s, ports={90:physical:$\sigma_2$}]{A_2} &
            \tn[label pos=s, ports={90:physical:$\sigma_3$}]{A_3}
        \end{tenkz}
        $=$
        \begin{tenkz}[rows={wire}, cols=4, boundary=periodic]
            \tn[label pos=s, ports={90:physical:$\sigma_1$}]{B_1} &
            \tn[skin=ring]{Y} &
            \tn[label pos=s, ports={90:physical:$\sigma_2$}]{B_2} &
            \tn[label pos=s, ports={90:physical:$\sigma_3$}]{B_3}
        \end{tenkz}
    \end{center}
\end{lemma}
\begin{proof}\leanok
    For $X\in\MN{D_A(e)}$, realize the insertion of $X$ at the endpoints $u,v$ of $A$'s edge-blocked chain by physical operators $O_1,O_2$ (Lemma~\ref{thm:peps_edgeInsertedCoeff_endpointPhysicalRealization}). With open middle weights $C_{\mathrm{mid}}^A,C_{\mathrm{mid}}^B$, equality of PEPS states (Lemma~\ref{thm:peps_sameState_edgeBlockedCoeff_eq}) gives
    \begin{align}
        C_A(e,\sigma,X)
         & =\sum_{\beta}O_1(B_u(\beta_u))_{\sigma_u}C_{\mathrm{mid}}^B(\sigma;\beta_{\mathrm{L}},\beta_{\mathrm{R}})(B_v(\beta_v))_{\sigma_v}
        =\sum_{\beta}(B_u(\beta_u))_{\sigma_u}C_{\mathrm{mid}}^B(\sigma;\beta_{\mathrm{L}},\beta_{\mathrm{R}})O_2(B_v(\beta_v))_{\sigma_v}.
        \notag
    \end{align}
    Lemma~\ref{thm:peps_physical_to_virtual_insertion} on $B$'s chain yields $Y=T(X)\in\MN{D_B(e)}$ with $C_A(e,\sigma,X)=C_B(e,\sigma,T(X))$. Linearity and two-endpoint realization give
    \begin{align}
        C_B(e,\sigma,T(X+X')) & =C_B(e,\sigma,T(X)+T(X')), &
        C_B(e,\sigma,T(XX'))  & =C_B(e,\sigma,T(X)T(X')),  &
        C_B(e,\sigma,T(1))    & =C_B(e,\sigma,1),
        \notag
    \end{align}
    and $T(cX)=cT(X)$ by Lemma~\ref{thm:peps_edgeInsertedCoeff_injective}; exchanging $A$ and $B$ supplies $T^{-1}$.
\end{proof}

\begin{lemma}[Matrix-algebra isomorphisms preserve size]%
    \label{thm:matrixAlgEquiv_fin_eq}
    \lean{Matrix.matrixAlgEquiv_fin_eq}
    \leanok
    If $\MN{D}\simeq_{\mathrm{alg}}\MN{E}$ as $\C$-algebras, then $D=E$.
\end{lemma}
\begin{proof}\leanok
    $D^2=\dim_{\C}\MN{D}=\dim_{\C}\MN{E}=E^2$, so $D=E$.
\end{proof}

\begin{lemma}[Matrix-algebra isomorphisms are inner after identifying size]%
    \label{thm:matrixAlgEquiv_inner_of_fin}
    \lean{Matrix.matrixAlgEquiv_inner_of_fin}
    \leanok
    \uses{thm:matrixAlgEquiv_fin_eq}
    For any $\C$-algebra isomorphism $\varphi:\MN{D}\to\MN{E}$, one has $h:D=E$, and with the canonical reindexing $\iota_h:\MN{D}\to\MN{E}$ there exists $X\in\GL_E(\C)$ such that $\varphi(M)=X\iota_h(M)X^{-1}$ for all $M\in\MN{D}$.
\end{lemma}
\begin{proof}\leanok
    Lemma~\ref{thm:matrixAlgEquiv_fin_eq} gives $h:D=E$, and applying Skolem--Noether (Theorem~\cite[``Skolem--Noether'']{QICLean2026Blueprint}) to $\varphi\circ\iota_h^{-1}\in\mathrm{Aut}_{\mathrm{alg}}(\MN{E})$ yields $X\in\GL_E(\C)$ with $\varphi(M)=X\iota_h(M)X^{-1}$.
\end{proof}

\begin{lemma}[Edge gauge from the insertion algebra isomorphism]%
    \label{thm:peps_edgeGaugeFromInsertionAlgebraIsomorphism}
    \lean{TNLean.PEPS.edgeGaugeFromInsertionAlgebraIsomorphism}
    \leanok
    \uses{thm:matrixAlgEquiv_inner_of_fin}
    In \cite[Section~3]{Molnar2018NormalPEPS}, if the edge insertion correspondence on $e$ is a $\C$-algebra isomorphism $\MN{D_A(e)}\simeq_{\mathrm{alg}}\MN{D_B(e)}$, then $h_e:D_A(e)=D_B(e)$ and there exists $Z_e\in\GL(D_B(e))$ such that, with the induced isomorphism $\phi_{h_e}:\MN{D_A(e)}\to\MN{D_B(e)}$, the correspondence takes the form
    % Formula/source: Papers/1804.04964/paper_normal.tex, lines 254--279
    % and 576--582, identifies the bond insertion map as an algebra
    % isomorphism and writes its inner-conjugation form Y=ZXZ^{-1}.
    $Y=Z_e\phi_{h_e}(X)Z_e^{-1}$.
\end{lemma}
\begin{proof}\leanok
    Apply Lemma~\ref{thm:matrixAlgEquiv_inner_of_fin} to the bond matrix algebras on $e$.
\end{proof}

\begin{lemma}[Per-edge gauge family from insertion algebra isomorphisms]%
    \label{thm:peps_exists_edgeGaugeFamily}
    \lean{TNLean.PEPS.exists_edgeGaugeFamily}
    \leanok
    \uses{thm:peps_edgeBlockedThreeSiteInjective, thm:peps_edgeBlockedInsertionAlgebraIsomorphism, thm:peps_edgeGaugeFromInsertionAlgebraIsomorphism}
    Let $A$ and $B$ be vertex-injective PEPS tensors defining the same state with $D_A=D_B>0$. Then there exist $X_e\in\GL(D_A(e))$ and $\C$-algebra isomorphisms $\Phi_e:\MN{D_A(e)}\to\MN{D_B(e)}$ for every edge $e$ such that $C_A(e,\sigma,M)=C_B(e,\sigma,\Phi_e(M))$ and $\Phi_e(M)=\phi_{h_e}(X_eMX_e^{-1})$ for all $\sigma$ and $M$, where $\phi_{h_e}$ is induced by $h_e:D_A(e)=D_B(e)$.
\end{lemma}
\begin{proof}\leanok
    Combine Lemma~\ref{thm:peps_edgeBlockedThreeSiteInjective}, Lemma~\ref{thm:peps_edgeBlockedInsertionAlgebraIsomorphism}, and Lemma~\ref{thm:peps_edgeGaugeFromInsertionAlgebraIsomorphism} on each edge $e$.
\end{proof}

\begin{lemma}[Per-edge gauge at a translated horizontal interior edge]%
    \label{thm:peps_exists_regionEdgeGauge_normalSquareHorizontalTranslatedEdge}
    \lean{TNLean.PEPS.exists_regionEdgeGauge_normalSquareHorizontalTranslatedEdge}
    \leanok
    \uses{lem:peps_normalSquareHorizontalTranslatedEdge_blockingData}
    Let $A$ and $B$ be normal PEPS on the open $W\times H$ square lattice satisfying the rectangular-injectivity hypotheses with union closure, with $d>0$, $D_A=D_B>0$, and the same PEPS state. For any translated horizontal interior frame with $2\le x_0$, $2\le y_0$, $x_0+8\le W$, $y_0+8\le H$ and distinguished horizontal edge $e$, one has $h_e:D_A(e)=D_B(e)$ and there exist $Z\in\GL(D_B(e))$ and a forward map $\mathrm{fwd}(M)=Z\phi_{h_e}(M)Z^{-1}$, where $\phi_{h_e}:\MN{D_A(e)}\to\MN{D_B(e)}$ is induced by $h_e$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{lem:peps_normalSquareHorizontalTranslatedEdge_blockingData}, the translated horizontal frame supplies a common one-edge interior blocking datum $(R_A,B_A,C_A)=(R_B,B_B,C_B)$ with unique crossing edge $e$ between the red and blue blocks, and the one-edge blocking gauge construction yields $h_e:D_A(e)=D_B(e)$ and $Z\in\GL(D_B(e))$.
\end{proof}

\begin{definition}[Factorized local gauge datum]%
    \label{def:peps_hasFactorizedLocalGauge}
    \lean{TNLean.PEPS.HasFactorizedLocalGauge}
    \leanok
    \uses{def:peps_localTensorMap, def:peps_localLeftInverse}
    In \cite[Section~3]{Molnar2018NormalPEPS}, for $A,B$ with $D_A=D_B$ at a vertex $v$, let $\mathcal T^A_v$ be the local tensor map of $A$, $L^A_v$ its left inverse, and $\Pi^A_v=\mathcal T^A_vL^A_v$. A factorized local gauge datum at $v$ consists of $X_e\in\GL(D_A(e))$ ($e\ni v$) such that for all local virtual configurations $\eta,\eta'$,
    \begin{align}
        \Pi^A_v B_v(\eta,-)
         & =B_v(\eta,-),                       &
        L^A_vB_v(\eta,-)(\eta')
         & =\prod_{e\ni v}X_e(\eta_e,\eta'_e).
        \label{eq:peps_has_factorized_local_gauge}
    \end{align}
\end{definition}

\begin{definition}[Blocked-middle local gauge formula]%
    \label{def:peps_blockedMiddleGaugeFormula}
    \lean{TNLean.PEPS.BlockedMiddleGaugeFormula}
    \leanok
    The blocked-middle local gauge formula at $v$ asserts the existence of $X_e\in\GL(D_A(e))$ ($e\ni v$) such that for all local virtual configurations $\eta$ and physical indices $\sigma$,
    \begin{align}
        B_v(\eta,\sigma)
        =
        \sum_{\eta'}
        \left(\prod_{e\ni v}X_e(\eta_e,\eta'_e)\right)
        A_v(\eta',\sigma).
        \label{eq:peps_blocked_middle_gauge_formula}
    \end{align}
\end{definition}

\begin{lemma}[Blocked-middle formula implies factorized local gauge]%
    \label{thm:peps_hasFactorizedLocalGauge_of_blockedMiddleGaugeFormula}
    \lean{TNLean.PEPS.hasFactorizedLocalGauge_of_localGaugeFormula}
    \leanok
    \uses{def:peps_blockedMiddleGaugeFormula, def:peps_hasFactorizedLocalGauge}
    If (\ref{eq:peps_blocked_middle_gauge_formula}) holds at $v$ with matrices $X_e$, then (\ref{eq:peps_has_factorized_local_gauge}) holds at $v$ with the same $X_e$.
    % Open: derive the displayed blocked-middle formula from equality of PEPS
    % states by the edge-centred three-site reduction.
\end{lemma}
\begin{proof}\leanok
    With $c_\eta(\eta'):=\prod_{e\ni v}X_e(\eta_e,\eta'_e)$, (\ref{eq:peps_blocked_middle_gauge_formula}) reads $B_v(\eta,-)=\mathcal T^A_vc_\eta$, so $\Pi^A_vB_v(\eta,-)=\mathcal T^A_vc_\eta=B_v(\eta,-)$ and $L^A_vB_v(\eta,-)(\eta')=c_\eta(\eta')$.
\end{proof}

\begin{lemma}[Blocked-middle formula gives a local gauge formula]%
    \label{thm:peps_localGauge_exists_of_blockedMiddleGaugeFormula}
    \lean{TNLean.PEPS.localGauge_exists_of_blockedMiddleGaugeFormula}
    \leanok
    \uses{def:peps_blockedMiddleGaugeFormula, thm:peps_hasFactorizedLocalGauge_of_blockedMiddleGaugeFormula, thm:localGauge_exists}
    If $X_e\in\GL(D_A(e))$ ($e\ni v$) satisfy (\ref{eq:peps_blocked_middle_gauge_formula}), then $(X_e)_{e\ni v}$ is a local gauge at $v$.
\end{lemma}
\begin{proof}\leanok
    By Lemma~\ref{thm:peps_hasFactorizedLocalGauge_of_blockedMiddleGaugeFormula}, (\ref{eq:peps_has_factorized_local_gauge}) holds, and expanding $B_v(\eta,-)=\Pi^A_vB_v(\eta,-)=\mathcal T^A_v(L^A_vB_v(\eta,-))$ recovers (\ref{eq:peps_blocked_middle_gauge_formula}).
\end{proof}
