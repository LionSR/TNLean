% Regression: t2_lhsrhs1 — equates the two tensor-network forms of an overlapping parent-term ground space.
% t2_lhsrhs1 — RMP arXiv:2011.12127, Eq. (eq:4:initiate-intersection-proof),
% files fig4_fig4-pepe_LHS1.pdf / RHS1.pdf: the joint ground space of two
% overlapping parent-Hamiltonian terms.
%
% Formula:  the states of the intersection
%   I = G_{1..L0+1} cap G_{2..L0+2}  are exactly those with
%   sum A^{i_1} B^{i_2} L(i_3)  |i>  =  sum B^{i_2} C^{i_3} R(i_1) |i>,
% where A, C are single-site tensors, B the injective blocked tensor of
% sites 2..L0+1, and L, R are 3-leg boundary tensors (two virtual ends
% plus the physical index of the site NOT covered by the respective
% ground-space term).
%
% Declarative reading per side (rows={ket:nopair, bra}):
%   * row 1: the tensor word; a \tnghost carries the bare wire across
%     the uncovered site (the reference draws the same plain wire);
%   * row 2: the boundary tensor as ONE wide pill (\tn[pill, wide=3]);
%   * west=cup + east=cup: the two cups tie the word's virtual ends
%     into the boundary tensor's ends — the L/R contraction.
%
% Noted gap (layout, not content): the reference routes the boundary
% tensor's physical leg UP past the word, escaping over the east (LHS)
% or west (RHS) cup; a tenkz cup cannot carry an open physical stub over
% its bend, so here that leg is emitted DOWN from the bra-typed boundary
% row.  Same tensor, same open-index count, different routing.
%
% Signature both sides: (vw 0, ve 0, pu 2, pd 1) — balanced across `=`.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: A B (wire) over the boundary tensor L
  \begin{tenkz}[rows={ket:nopair, bra}, west=cup, east=cup,
                tensor style=box]
    \tn{A} & \tn{B} & \tnghost{}\\
    \tn[pill, wide=3]{L} & &
  \end{tenkz}
  \;=\;
  % RHS: (wire) B C over the boundary tensor R
  \begin{tenkz}[rows={ket:nopair, bra}, west=cup, east=cup,
                tensor style=box]
    \tnghost{} & \tn{B} & \tn{C}\\
    \tn[pill, wide=3]{R} & &
  \end{tenkz}
\]
\end{document}
