% Regression: hard12_v3 — imported current-behavior fixture from the handoff corpus.
% T12 v3 — RMP ZCL-MPDO (zero correlation length).  Previously BLOCKED
% on closing a site's up leg into its own down leg; now expressed with
% `trace=physical`.
%
% Formula:  with E := tr_phys(M) = sum_s M^{ss} (the MPDO tensor M with
% its own bra (up) and ket (down) physical legs closed by a per-site
% trace), the zero-correlation-length condition reads
%     E . E = E        (the physically-traced tensor is idempotent).
%
% Declarative reading:
%   physical=updown — each M carries its own up/down physical pair;
%   trace=physical  — every site's up leg arcs around the glyph's east
%   side into its own down leg (Tr_phys per site); traced legs are
%   CLOSED indices and leave the boundary signature, so both sides
%   read (virtual-west, virtual-east, up, down) = (1,1,0,0).
% Styling note: the reference draws the trace loop dotted; tenkz's
% `trace` style is deliberately solid with rounded corners (the
% library's dotted-free doctrine), same as its periodic racetrack.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: E . E — two physically-traced tensors sharing one virtual bond
  \begin{tenkz}[physical=updown, trace=physical, tensor style=box]
    \tn[mpo]{M} & \tn[mpo]{M}
  \end{tenkz}
  \;=\;
  % RHS: E — one physically-traced tensor
  \begin{tenkz}[physical=updown, trace=physical, tensor style=box]
    \tn[mpo]{M}
  \end{tenkz}
\]
\end{document}
