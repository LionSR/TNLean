% Regression: t2_isometry — renders A^i_{alpha,beta} = <i, beta| V |alpha>.
% T2 isometry — RMP arXiv:2011.12127, fig. newfig13: the MPS tensor read
% as an isometry (the sequential-generation / fine-graining view).
%
% Formula:  A^i_{alpha,beta} = <i, beta| V |alpha>,
%   V : H_alpha -> H_i (x) H_beta  an isometry (V^dagger V = 1):
% the 3-leg MPS tensor IS the rounded gate V with ONE input leg alpha
% below and TWO output legs i (physical) and beta (next virtual) above.
%
% Declarative reading: the hard08 vocabulary — an op-type row whose side
% stubs are sealed (rows={op:none}: the gate has no horizontal indices),
% one pill glyph wide=2 with `up at={1,2}` for the two output legs and the
% default single down leg for alpha.  The up=/down= lists label the legs
% leg-by-leg.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  A^{i}_{\alpha,\beta} \;=\;
  % V: one down leg alpha in, two up legs (i, beta) out
  \begin{tenkz}[rows={op:none}, tensor style=box]
    \tn[pill, wide=2, up at={1,2}, up={$i$,$\beta$}, down={$\alpha$}]{V} &
  \end{tenkz}
\]
\end{document}
