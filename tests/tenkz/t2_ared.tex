% T2 — RMP arXiv:2011.12127, fig. fig3-pepe_Ared (boundary algebra of a
% G-injective PEPS / MPO layer): the boundary-to-bulk map of one MPO tensor.
% Formula:
%   X  |->  Gamma(X)^{s}_{s'} = Tr_D[ X A^{(s,s')} ]
%         = sum_{alpha,beta} X_{alpha beta} A^{(s,s')}_{beta alpha} ,
% the physical operator obtained by inserting the boundary matrix X on the
% virtual ring of the MPO tensor A and tracing the virtual (red) wire; the
% physical up/down legs of A stay open.  The source figure carries the
% ": X" annotation of the map's argument on the right; it is kept here.
%
% Declarative reading: one operator-hued op row (rows={op:operator});
% the virtual ring is the row's `periodic` racetrack; X is the on-wire
% matrix \tnX (tenkz's canonical glyph for a matrix ON a wire -- the
% source draws an open circle, same role).
% Noted gap: the periodic closure strokes in the neutral `trace` style and
% does not inherit the traced row's operator hue; recolored here through
% the documented \tikzset restyle door.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % Gamma(X) = Tr_D[X A]: X on the traced virtual wire, physical legs open
  {\tikzset{trace/.append style={draw=tenkzOperator}}%
  \begin{tenkz}[rows={op:operator}, periodic, tensor style=box]
    \tnX{X} & \tn{}
  \end{tenkz}}
  \;\colon\; X
\]
\end{document}
