% Regression: t2_mpucond — renders sum_{s, alpha alpha', beta beta'} a_{alpha alpha'} U^{i s}_{alpha beta} conj(…
% t2_mpucond — RMP arXiv:2011.12127, fig. sec2fig7b(a) first panel, file
% fig2_fig2-pepe_sec2fig8a1.pdf: the unitarity condition on the blocked
% MPU tensor, written with the gauge pair (a, b).
%
% Formula:
%   sum_{s, alpha alpha', beta beta'}  a_{alpha alpha'}
%     U^{i s}_{alpha beta}  conj( U^{i' s}_{alpha' beta'} )  b_{beta beta'}
%   =  delta_{i i'}                                (identity wire)
% — U sits over its conjugate, the facing physical indices s are
% contracted (the middle vertical bond), the two west virtual indices
% close through the matrix a, the two east ones through b, and the
% result is the identity on the remaining physical pair (i, i').
%
% Declarative reading:
%   rows={op,op} — an op row pairs onto the op row below, so the facing
%   legs contract automatically; U keeps its north leg, conj(U) its
%   south leg.
%   west={cup=$a$} / east={cup=$b$} — the virtual cups carry the
%   gauge matrices as on-bend dots (house style for the reference's
%   circled letters riding the elbows).
%   RHS — the reference draws a bare identity wire.  VERIFIED GAP: no
%   wire-only atom exists (WHATS-NEW-0.5 lists the bare identity-wire
%   atom as a 0.6 gap), and the nearest spelling — an all-empty wire row
%   under frame=vertical — renders NOTHING and logs the scalar
%   signature (vw 0, ve 0, pu 0, pd 0), i.e. no ink and the wrong
%   object.  Until the atom lands, the identity is written as algebra
%   between pictures (the \longrightarrow idiom): the symbol I.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: a (west cup) [U over conj(U), physical pair contracted] b (east cup)
  \begin{tenkz}[rows={op,op}, tensor style=box,
                west={cup=$a$}, east={cup=$b$}]
    \tn{U} \\
    \tn*{U}
  \end{tenkz}
  \;=\;
  % RHS: the identity on the physical pair.  The reference's bare
  % vertical wire is unrenderable (see header); the symbol stands in.
  \mathbb{I}
\]
\end{document}
