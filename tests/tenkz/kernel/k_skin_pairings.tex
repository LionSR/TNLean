%% Kernel fixture: declared skins resolve their base and ink ordered own-port
%% pairings over the glyph with their declared species hues.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\ExplSyntaxOff
\makeatother
\begin{document}
\tenkzkernel
\tndeclare{species}{warm}{hue=source:red}
\tndeclare{species}{cool}{hue=source:blue}
\tndeclare{skin}{shift-right}{
  base=box,
  pairings={90@1 > 0@1 : cool, 270@1 > 180@1 : warm}
}
\tndeclare{skin}{shift-left}{
  base=box,
  pairings={90@1 > 180@1 : cool, 270@1 > 0@1 : warm}
}
\tndeclare{skin}{slotted}{
  base=shift-right,
  pairings={
    180@2 > 0@1 : warm,
    90@2 > 270@1 : cool,
    180@1 > 0@1 : cool
  }
}
\tndeclare{skin}{alias-dot}{
  base=dot,
  pairings={90@1 > 0@1 : cool}
}
\tndeclare{skin}{alias-ring}{base=ring}
\tndeclare{skin}{alias-tri}{base=tri}
\tndeclare{skin}{alias-dots}{
  base=dots,
  pairings={180@1 > 0@1 : warm}
}
\tndeclare{skin}{alias-none}{
  base=none,
  pairings={90@1 > 0@1 : cool}
}
\begin{tenkz}[rows={wire}, cols=3, frame=circle, bonds=none]
  \tn[at=(1,1), skin=shift-right]{}
  \tn[at=(1,2), skin=shift-left]{}
  \tn[at=(1,3), skin=shift-right]{}
\end{tenkz}
\qquad
\begin{tenkz}[rows={top,bottom}, cols=4, frame=circle, bonds=none]
  \tn[at=(1,2), skin=slotted, species=warm, wires=2, wide=2]{X}
  \tn[at=(1,1), skin=none, void=sealed]{}
  \tn[at=(1,4), skin=none, void=sealed]{}
  \tn[at=(2,1), skin=none, void=sealed]{}
  \tn[at=(2,4), skin=none, void=sealed]{}
\end{tenkz}
\qquad
\begin{tenkz}[rows={wire}, cols=6, bonds=none]
  \tn[at=(1,1), skin=alias-dot]{}
  \tn[at=(1,2), skin=shift-right]{}
  \tn[at=(1,3), skin=alias-ring]{R}
  \tn[at=(1,4), skin=alias-tri]{T}
  \tn[at=(1,5), skin=alias-dots]{}
  \tn[at=(1,6), skin=alias-none]{}
\end{tenkz}
\qquad
\begin{tenkz}[rows={wire}, cols=1, bonds=none]
  \tn[skin=dot, name=anchor]{}
  \tn[skin=shift-left, at=1 n of anchor]{}
\end{tenkz}
\qquad
\begin{tenkz}[rows={wire}, cols=3, bonds=grid]
  \tn[at=(1,1), skin=dot]{}
  \tn[at=(1,2), skin=alias-none]{}
  \tn[at=(1,3), skin=dot]{}
\end{tenkz}
\qquad
\begin{tenkz}[rows={wire}, cols=1, bonds=none]
  \tn[at=(1,1), skin=shift-right]{}
  \tn[skin=dot, at=on skin-atom-1-1 0.5]{}
\end{tenkz}
\end{document}
