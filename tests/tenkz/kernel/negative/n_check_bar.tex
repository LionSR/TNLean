% Negative regression: a reason is prose, and the one character prose cannot
% hold is the record stream's own field separator, which would split the
% recorded reason into fields no reader can put back together.
\documentclass{standalone}
\usepackage{tenkz}
\begin{document}
\tenkzkernel
\begin{tenkzeq}[check={signature, off={1: compare A | B}}]
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open]
    \tn[ports={180:virtual}]{A}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open]
    \tn[ports={180:virtual}]{B}
  \end{tenkz}
\end{tenkzeq}
\end{document}
