% Negative: a physical index leaving one panel cannot equal one entering
% the other; the boundary orientation is part of the signature.
\documentclass{standalone}
\usepackage{tenkz}
\begin{document}
\tenkzkernel
\begin{tenkzeq}[check={signature}]
  \begin{tenkz}[rows={wire}, cols=1, bonds=none]
    \tn[at=(1,1), name=A, ports={90:physical}]{A}
    \tnwire[dir=to]{A.90}{open n}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire}, cols=1, bonds=none]
    \tn[at=(1,1), name=B, ports={90:physical}]{B}
    \tnwire[dir=from]{B.90}{open n}
  \end{tenkz}
\end{tenkzeq}
\end{document}
