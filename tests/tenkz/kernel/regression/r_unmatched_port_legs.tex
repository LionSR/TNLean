                      {
                        abs
                          (
                            \l__tenkz_test_port_dx_fp
                              * \__tenkz_metric_ratio:n {planerise}
                            - \l__tenkz_test_port_dy_fp
                              * \__tenkz_metric_ratio:n {planeslant}

                      }
                      < {0.000001}
                      {\errmessage{plane~north~port~lost~the~plane~normal}}

                          )
                      }
                      < {0.000001}
                      {\errmessage{plane~diagonal~port~lost~E+N}}
                  }
              }
          }
      }
