    \str_case:enF { \__tenkz_kernel_r_frame_word: }
      {
        {flat}{\int_gincr:N \g__tenkz_test_port_open_flat_int}
        {plane}{\int_gincr:N \g__tenkz_test_port_open_plane_int}
      }
      {\int_gincr:N \g__tenkz_test_port_open_wired_int}
    \__tenkz_model_get:nnN {#1} {port-type} \l_tmpa_tl
    \fp_set:Nn \l_tmpa_fp
      {
        \str_if_eq:VnTF \l_tmpa_tl {physical}
          {\__tenkz_metric_ratio:n {physleg}}
          {\__tenkz_metric_ratio:n {stub}}
      }
    \fp_compare:nNnF
      {
        abs
          (
            sqrt
              (
                (\l__tenkz_kernel_r_xb_fp-\l__tenkz_kernel_r_x_fp)^2
                +(\l__tenkz_kernel_r_yb_fp-\l__tenkz_kernel_r_y_fp)^2
              )
            -\l_tmpa_fp
          )
      }
      < {0.000001}
      {\errmessage{unmatched~port~lost~its~type-owned~reach}}
    \__tenkz_model_get:nnN {#1} {port-face} \l__tenkz_test_port_face_tl

      { \__tenkz_kernel_r_frame_word: }
    \fp_set:Nn \l__tenkz_test_port_dx_fp
      { \l__tenkz_kernel_r_xb_fp - \l__tenkz_kernel_r_x_fp }
    \fp_set:Nn \l__tenkz_test_port_dy_fp
      { \l__tenkz_kernel_r_yb_fp - \l__tenkz_kernel_r_y_fp }
    \str_case:Vn \l__tenkz_test_port_frame_tl
      {
        {flat}
          {
            \fp_compare:nNnF {\l__tenkz_test_port_dy_fp} > {0}
              {\errmessage{flat~unmatched~port~did~not~point~outward}}
            \str_case:Vn \l__tenkz_test_port_face_tl
              {
                {90}
                  {
                    \fp_compare:nNnF
                      {abs(\l__tenkz_test_port_dx_fp)} < {0.000001}
                      {\errmessage{flat~north~port~lost~its~bearing}}
                  }
                {45}
                  {
                    \fp_compare:nNnF {\l__tenkz_test_port_dx_fp} > {0}
                      {\errmessage{flat~diagonal~port~did~not~point~east}}
                    \fp_compare:nNnF
                      {
                        abs
                          (
                            \l__tenkz_test_port_dx_fp
                            - \l__tenkz_test_port_dy_fp
                          )
                      }
                      < {0.000001}
                      {\errmessage{flat~diagonal~port~lost~its~bearing}}
                  }
              }
          }
        {plane}
          {
            \fp_compare:nNnF {\l__tenkz_test_port_dx_fp} > {0}
              {\errmessage{plane~unmatched~port~did~not~point~east}}
            \fp_compare:nNnF {\l__tenkz_test_port_dy_fp} > {0}
              {\errmessage{plane~unmatched~port~did~not~point~north}}
            \str_case:Vn \l__tenkz_test_port_face_tl
              {
                {90}
                  {
                    \fp_compare:nNnF
                      {
                        abs
                          (
                            \l__tenkz_test_port_dx_fp
                              * \__tenkz_metric_ratio:n {planerise}
                            - \l__tenkz_test_port_dy_fp
                              * \__tenkz_metric_ratio:n {planeslant}
                          )
                      }
                      < {0.000001}
                      {\errmessage{plane~north~port~lost~the~plane~normal}}
                  }
                {45}
                  {
                    \fp_compare:nNnF
                      {
                        abs
                          (
                            \l__tenkz_test_port_dx_fp
                              * \__tenkz_metric_ratio:n {planerise}
                            - \l__tenkz_test_port_dy_fp
                              * (1+\__tenkz_metric_ratio:n {planeslant})
                          )
                      }
                      < {0.000001}
                      {\errmessage{plane~diagonal~port~lost~E+N}}
                  }
              }
          }
      }
