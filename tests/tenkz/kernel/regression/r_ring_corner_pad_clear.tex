% Regression (issue #5958 follow-up): the ring corner-pad registry
% (\g__tenkz_kernel_eq_ringpad_prop) must be cleared at the start of every
% equation, exactly like the shared-extent registry
% (\g__tenkz_kernel_eq_extent_prop) it mirrors.  Left uncleared, a
% superscripted ring in one equation leaves a stadium-width entry keyed by
% size class alone, and a later, unrelated equation whose own ring carries
% no superscript inherits that stale width instead of reading its own
% baseline.
%
% The first equation is a pristine control: a lone plain ring, no
% superscript anywhere in scope, pinning the true baseline width.  The
% second equation reproduces the corner-safe pad (X vs X^{-1}, as
% r_ring_corner_inscription.tex).  The third equation is a lone plain ring
% again, immediately after the second: it must render at the SAME width as
% the first, pristine equation -- not at the second equation's padded
% width.
\documentclass{standalone}
\usepackage{tenkz}
\begin{document}
\tenkzkernel
\begin{tenkzeq}[check={signature}]
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}
\end{tenkzeq}
\begin{tenkzeq}[check={signature}]

    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}

  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X^{-1}}
  \end{tenkz}
\end{tenkzeq}
\begin{tenkzeq}[check={signature}]

    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=ring, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}
\end{tenkzeq}
\end{document}
