% Regression: an atom claims every cell of its rectangle, so a wire leaving
% one of the rows below its anchor reserves that face too.  A station read
% from the anchor row alone leaves the lower rows looking free and stands
% the name on the wire.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\int_new:N \l__tenkz_test_side_count_int
\tl_new:N \l__tenkz_test_side_label_tl
\cs_new_eq:NN
  \__tenkz_test_side_auto:nN \__tenkz_kernel_r_label_auto_side:nN
\cs_set_protected:Npn \__tenkz_kernel_r_label_auto_side:nN #1#2
  {
    \__tenkz_test_side_auto:nN {#1} #2
    \int_gincr:N \l__tenkz_test_side_count_int
    \__tenkz_model_get:nnN {#1} {label} \l__tenkz_test_side_label_tl
    \str_case:VnF \l__tenkz_test_side_label_tl
      {
        {S}
          {
            \str_if_eq:VnF #2 {n}
              { \errmessage{a~spanned~row~left~the~atom's~south~free} }
          }
        {W}
          {
            \str_if_eq:VnF #2 {n}
              { \errmessage{a~spanned~column~left~the~atom's~south~free} }
          }
      }
      { \errmessage{unexpected~label~in~row-span~fixture} }
  }
\ExplSyntaxOff
\makeatother
\begin{document}
\tenkzkernel
% The wire leaves row 2, the lower of the two rows S stands on.
\begin{tenkz}[lattice={3x1}, bonds=none]
  \tn[at=(1,1), wires=2, skin=dot]{S}
  \tn[at=(3,1)]{}
  \tnwire{(2,1)}{(3,1)}
\end{tenkz}
% The same reading across a column span, from the second of two columns.
\begin{tenkz}[lattice={2x2}, bonds=none]
  \tn[at=(1,1), wide=2, skin=dot]{W}
  \tn[at=(2,2)]{}
  \tnwire{(1,2)}{(2,2)}
\end{tenkz}
\ExplSyntaxOn
\int_compare:nNnF { \l__tenkz_test_side_count_int } = {2}
  { \errmessage{row-span~fixture~did~not~station~two~labels} }
\ExplSyntaxOff
\end{document}
