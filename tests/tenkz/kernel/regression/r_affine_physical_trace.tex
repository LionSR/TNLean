\cs_new_protected:Npn \__tenkz_test_phtrace_close:nnn #1#2#3

% Pin the three public topology paths' boundary effects.  The one-row trace
% consumes both directions of its single host's axis, so the boundary closes.
% The two-host trace consumes the outer transverse ends only, leaving the
% facing pair as the boundary -- exactly the flat frame's signature for the
% same picture.  The row-zero opening exposes exactly one physical end.
\cs_new_eq:NN \__tenkz_test_phtrace_event:n \__tenkz_kernel_event:n
\cs_set_protected:Npn \__tenkz_kernel_event:n #1
  {
    \tl_set:Ne \l__tenkz_test_phtrace_event_tl {#1}
    \exp_args:NnV \regex_match:nnT
      { \A \s* kernel-boundary } \l__tenkz_test_phtrace_event_tl
      {
        \int_gincr:N \g__tenkz_test_phboundary_int
        \int_case:nn { \g__tenkz_test_phboundary_int }
          {
            { 1 }
              {
                \exp_args:NnV \regex_match:nnF
                  { \A \s* kernel-boundary \s* \| \s* signature \s* = \s* \Z }
                  \l__tenkz_test_phtrace_event_tl
                  {
                    \errmessage{a~plane~physical~trace~left~an~open~boundary}
                  }
              }
            { 2 }
              {
                \exp_args:NnV \regex_match:nnF
                  {
                    \A \s* kernel-boundary \s* \| \s* signature \s* =
                    \s* phys:n \s* , \s* phys:s \s* \Z
                  }
                  \l__tenkz_test_phtrace_event_tl
                  {
                    \errmessage
                      {a~plane~physical~trace~changed~its~facing~boundary}
                  }
              }
            { 3 }
              {
                \exp_args:NnV \regex_match:nnF
                  {
                    \A \s* kernel-boundary \s* \| \s* signature \s* =
                    \s* phys:n \s* \Z
                  }
                  \l__tenkz_test_phtrace_event_tl
                  {
                    \errmessage
                      {a~plane~physical~opening~lost~its~typed~boundary}
                  }
              }
          }
      }
    \__tenkz_test_phtrace_event:n {#1}
  }

\cs_new_eq:NN
  \__tenkz_test_phtrace_plane:n
  \__tenkz_kernel_r_physical_trace_plane:n
\cs_set_protected:Npn \__tenkz_kernel_r_physical_trace_plane:n #1
  {
    \__tenkz_test_phtrace_plane:n {#1}
    \__tenkz_model_get:nnN {#1} {physical-axis}
      \l__tenkz_test_phtrace_tl
    \str_if_eq:VnF \l__tenkz_test_phtrace_tl {transverse}
      { \errmessage{a~plane~physical~trace~lost~its~transverse~identity} }
    \__tenkz_model_get:nnN {#1} {from-axis} \l__tenkz_test_phtrace_tl
    \str_if_eq:VnF \l__tenkz_test_phtrace_tl {n}
      { \errmessage{a~plane~physical~trace~lost~its~north~identity} }
    \__tenkz_model_get:nnN {#1} {to-axis} \l__tenkz_test_phtrace_tl
    \str_if_eq:VnF \l__tenkz_test_phtrace_tl {s}
      { \errmessage{a~plane~physical~trace~lost~its~south~identity} }
    \__tenkz_model_get:nnN {#1} {from-host} \l__tenkz_test_phtrace_tl
    \__tenkz_model_get:nnN {#1} {to-host} \l__tenkz_test_phtrace_other_tl
    \quark_if_no_value:NT \l__tenkz_test_phtrace_tl
      { \errmessage{a~plane~physical~trace~lost~its~north~host} }
    \quark_if_no_value:NT \l__tenkz_test_phtrace_other_tl
      { \errmessage{a~plane~physical~trace~lost~its~south~host} }
    \__tenkz_test_phtrace_close:nnn
      { \l__tenkz_kernel_phtrace_from_tip_x_fp }
      { \l__tenkz_kernel_phtrace_from_x_fp }
      {a~plane~physical~trace~north~leg~left~the~transverse~axis}
    \__tenkz_test_phtrace_close:nnn
      { \l__tenkz_kernel_phtrace_to_tip_x_fp }
      { \l__tenkz_kernel_phtrace_to_x_fp }
      {a~plane~physical~trace~south~leg~left~the~transverse~axis}
    \__tenkz_test_phtrace_close:nnn
      {
        \l__tenkz_kernel_phtrace_from_tip_y_fp
        - \l__tenkz_kernel_phtrace_from_y_fp
      }
      { \__tenkz_metric_ratio:n {physleg} }
      {a~plane~physical~trace~lost~its~named~north~excursion}
    \__tenkz_test_phtrace_close:nnn
      {
        \l__tenkz_kernel_phtrace_to_y_fp
        - \l__tenkz_kernel_phtrace_to_tip_y_fp
      }
      { \__tenkz_metric_ratio:n {physleg} }
      {a~plane~physical~trace~lost~its~named~south~excursion}
    \str_if_eq:VVTF
      \l__tenkz_test_phtrace_tl \l__tenkz_test_phtrace_other_tl
      {
        \tl_set:Ne \l__tenkz_test_phtrace_tl
          { \__tenkz_metric_ratio:n {phtracereach} }
      }
      {
        \tl_set:Ne \l__tenkz_test_phtrace_tl
          { \__tenkz_metric_ratio:n {pairtracereach} }
      }
    \__tenkz_test_phtrace_close:nnn
      {
        \l__tenkz_kernel_phtrace_outer_x_fp
        - max(
            \l__tenkz_kernel_phtrace_from_east_x_fp,
            \l__tenkz_kernel_phtrace_to_east_x_fp,
            \l__tenkz_kernel_phtrace_from_tip_x_fp,
            \l__tenkz_kernel_phtrace_to_tip_x_fp )
      }
      { \l__tenkz_test_phtrace_tl }
      {a~plane~physical~trace~lost~its~named~page~clearance}
    \int_gincr:N \g__tenkz_test_phtrace_int
  }

\begin{tenkz}[
    rows={wire}, cols=1, bonds=none, physical=updown, trace=physical,
    frame={plane,basis={wire at (0,0)}}]
  \tn[at=(1,1,1), skin=box]{P}

% The row-zero cell-set spelling joins distinct top and bottom hosts; the
% facing physical pair stays open, matching the flat frame's boundary.
\begin{tenkz}[
    rows={wire,wire}, cols=1, bonds=none, physical=updown,
    trace={(physical,1)}, frame=plane]
  \tn[skin=box]{Q} \\
  \tn[skin=box]{R}
\end{tenkz}

\begin{tenkz}[

  \tn[skin=box]{S}
