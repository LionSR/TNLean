\begin{document}

  \tn[at=(2,2), name=B, skin=box]{B}
  \tn[at=midway (1,1) and (2,1), name=e, skin=pill, wires=2]{e}
  \tnwire{e.0@1}{A.180}
  \tnwire{e.0@2}{B.180}
\end{tenkz}
