% resolving a port carried by another port preserves both authored fractions.

      {
        {A}
          {
            \fp_gset:Nn \g__tenkz_test_nested_ax_fp
              {\l__tenkz_test_nested_x_tl}
            \fp_gset:Nn \g__tenkz_test_nested_ay_fp
              {\l__tenkz_test_nested_y_tl}
          }
        {B}
          {
            \fp_gset:Nn \g__tenkz_test_nested_bx_fp
              {\l__tenkz_test_nested_x_tl}
            \fp_gset:Nn \g__tenkz_test_nested_by_fp
              {\l__tenkz_test_nested_y_tl}

            \fp_gset:Nn \g__tenkz_test_nested_cx_fp

  }

  \tn[at=(1,1), name=A, ports={90:physical}]{A}

  \tn[skin=dot, name=C, at=on port-open-2 0.75]{}
  \tn[skin=dot, name=B, at=on port-open-1 0.25,
      ports={0:virtual}]{}
