% Regression: a plane frame owns one physical axis transverse to its two
% in-plane directions.  The policy legs remain distinct from all four
% authored virtual faces, and up, down, and updown use the signed axis.
\documentclass{standalone}
\usepackage{tenkz}
\begin{document}
\tenkzkernel
\begin{tenkz}[
    rows={wire}, cols=1, frame=plane, bonds=none, physical=up]
  \tn[at=(1,1),
    ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
\end{tenkz}
\begin{tenkz}[
    rows={wire}, cols=1, frame=plane, bonds=none, physical=down]
  \tn[at=(1,1),
    ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
\end{tenkz}
\begin{tenkz}[
    rows={wire}, cols=1, frame=plane, bonds=none, physical=updown]
  \tn[at=(1,1),
    ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
\end{tenkz}
\end{document}
