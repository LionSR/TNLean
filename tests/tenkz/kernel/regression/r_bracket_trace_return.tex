%% Regression: a bracket under a traced row steps outside the row's closure
%% return (issue #5631).  On a two-row ladder the lower row's return runs
%% one trace reach below the row; a south bracket on that row previously
%% put its arc between the row and the rail while its caption stood beyond
%% the rail, so the generated return split the annotation.  The bracket now
%% takes concentric order against the returns routed on the side it speaks
%% from: the rail keeps its pinned run, the band between the row and the
%% rail carries no annotation ink, the bar stands two clearances outside
%% the rail -- the turns that close the arc rise one clearance from the
%% bar, keeping one clearance of daylight between their tips and the rail
%% -- and the two paths never touch.  The single-row picture adds labelled
%% physical legs through the return: the bar steps past the leg tips and
%% their label bands as well, so it never lands in the label text beyond
%% the rail.  Each expectation counts intersections of a probe segment
%% against the saved rail and bracket paths, so the layering is pinned
%% structurally as well as by the exact-toolchain pixel ledger.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\int_new:N \l__tenkz_test_brk_cross_int
\fp_new:N \l__tenkz_test_brk_x_fp
\fp_new:N \l__tenkz_test_brk_ya_fp
\fp_new:N \l__tenkz_test_brk_yb_fp
% saved path #1 must meet the probe segment (#2,#3)--(#4,#5) exactly #6 times
\cs_new_protected:Npn \__tenkz_test_brk_probe:nnnnnn #1#2#3#4#5#6
  {
    \path [ spath/save = tenkz-test-brk-probe ]
      \__tenkz_kernel_r_pair:nn {#2} {#3}
      --
      \__tenkz_kernel_r_pair:nn {#4} {#5} ;
    \__tenkz_string_count:nnN {#1} {tenkz-test-brk-probe}
      \l__tenkz_test_brk_cross_int
    \int_compare:nNnF { \l__tenkz_test_brk_cross_int } = {#6}
      {
        \errmessage
          { bracket~layering~#1~expected~#6~probe~crossings,~found~
            \int_use:N \l__tenkz_test_brk_cross_int }
      }
  }
\cs_new_protected:Npn \__tenkz_test_brk_audit:
  {
    \int_compare:nNnTF { \l__tenkz_kernel_r_rows_int } = {1}
      { \__tenkz_test_brk_leg_audit: }
      { \__tenkz_test_brk_ladder_audit: }
  }
% The single-row picture: the rail keeps its run, and the bar clears not
% only the rail but the labelled leg tips beyond it, standing below the
% conservative label band at one further clearance.
\cs_new_protected:Npn \__tenkz_test_brk_leg_audit:
  {
    \__tenkz_kernel_r_frame_xy:nnNN {1} {1.5}
      \l__tenkz_test_brk_x_fp \l__tenkz_test_brk_ya_fp
    \__tenkz_test_brk_probe:nnnnnn {wrap-1}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 0.35 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 0.45 }
      {1}
    % no bar between the rail and the far side of the leg-label band
    \__tenkz_test_brk_probe:nnnnnn {tenkz-test-bracket}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 0.05 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 0.85 }
      {0}
    \__tenkz_test_brk_probe:nnnnnn {tenkz-test-bracket}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 0.95 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp - 2 }
      {1}
    \__tenkz_string_count:nnN {tenkz-test-bracket} {wrap-1}
      \l__tenkz_test_brk_cross_int
    \int_compare:nNnF { \l__tenkz_test_brk_cross_int } = {0}
      {
        \errmessage
          { bracket~and~trace~return~intersect~
            \int_use:N \l__tenkz_test_brk_cross_int ~time(s) }
      }
  }
\cs_new_protected:Npn \__tenkz_test_brk_ladder_audit:
  {
    \__tenkz_kernel_r_frame_xy:nnNN {1} {1.5}
      \l__tenkz_test_brk_x_fp \l__tenkz_test_brk_ya_fp
    \__tenkz_kernel_r_frame_xy:nnNN {2} {1.5}
      \l__tenkz_test_brk_x_fp \l__tenkz_test_brk_yb_fp
    % the lower row's rail keeps its pinned run, one trace reach below
    \__tenkz_test_brk_probe:nnnnnn {wrap-2}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 0.35 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 0.45 }
      {1}
    % the band between the row and the rail carries no annotation ink
    \__tenkz_test_brk_probe:nnnnnn {tenkz-test-bracket}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 0.05 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 0.35 }
      {0}
    % the bar runs outside the rail; this is the probe the split layering
    % fails, its bar standing between the row and the rail
    \__tenkz_test_brk_probe:nnnnnn {tenkz-test-bracket}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 0.45 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_yb_fp - 2 }
      {1}
    % the upper row's return is not the bracket's side and does not move
    \__tenkz_test_brk_probe:nnnnnn {wrap-1}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp + 0.05 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp + 2 }
      {1}
    \__tenkz_test_brk_probe:nnnnnn {tenkz-test-bracket}
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp + 0.05 }
      { \l__tenkz_test_brk_x_fp }
      { \l__tenkz_test_brk_ya_fp + 2 }
      {0}
    % stepped outside, the bracket and the rail never touch
    \__tenkz_string_count:nnN {tenkz-test-bracket} {wrap-2}
      \l__tenkz_test_brk_cross_int
    \int_compare:nNnF { \l__tenkz_test_brk_cross_int } = {0}
      {
        \errmessage
          { bracket~and~trace~return~intersect~
            \int_use:N \l__tenkz_test_brk_cross_int ~time(s) }
      }
  }
