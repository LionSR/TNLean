% Regression: physical policy refines a consumed endpoint without requiring a
% matching `ports=` descriptor, whether that endpoint is first or second.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\int_new:N \g__tenkz_test_policy_endpoint_stroke_int
\int_new:N \g__tenkz_test_policy_endpoint_leg_int
\cs_new_eq:NN
  \__tenkz_test_policy_endpoint_stroke:nn
  \__tenkz_kernel_r_wire_stroke:nn
\cs_set_protected:Npn \__tenkz_kernel_r_wire_stroke:nn #1#2
  {
    \exp_args:NV \__tenkz_model_get:nnN
      \l__tenkz_kernel_r_wire_record_tl {name} \l_tmpa_tl
    \str_case:Vn \l_tmpa_tl
      {
        {policy-from}
          {
            \str_if_eq:eeF {#1} {physical~leg}
              {\errmessage{first~policy~endpoint~remained~virtual}}
            \int_gincr:N \g__tenkz_test_policy_endpoint_stroke_int
          }
        {policy-to}
          {
            \str_if_eq:eeF {#1} {physical~leg}
              {\errmessage{second~policy~endpoint~remained~virtual}}
            \int_gincr:N \g__tenkz_test_policy_endpoint_stroke_int
          }
      }
    \__tenkz_test_policy_endpoint_stroke:nn {#1}{#2}
  }
\cs_new_eq:NN
  \__tenkz_test_policy_endpoint_leg:nn
  \__tenkz_kernel_r_leg:nn
\cs_set_protected:Npn \__tenkz_kernel_r_leg:nn #1#2
  {
    \__tenkz_model_get:nnN {#1} {name} \l_tmpa_tl
    \str_case:Vn \l_tmpa_tl
      {
        {A}{\int_gincr:N \g__tenkz_test_policy_endpoint_leg_int}
        {B}{\int_gincr:N \g__tenkz_test_policy_endpoint_leg_int}
      }
    \__tenkz_test_policy_endpoint_leg:nn {#1}{#2}
  }
\AtEndDocument
  {
    \int_compare:nNnF
      {\g__tenkz_test_policy_endpoint_stroke_int} = {2}
      {\errmessage{physical-policy~endpoint-order~coverage~changed}}
    \int_compare:nNnF
      {\g__tenkz_test_policy_endpoint_leg_int} = {0}
      {\errmessage{consumed~policy~endpoint~grew~a~duplicate~leg}}
  }
\ExplSyntaxOff
\makeatother
\begin{document}
\tenkzkernel
\begin{tenkz}[rows={wire}, cols=1, bonds=none, physical=up]
  \tn[name=A, ports={180:physical}]{A}
  \tnwire[name=policy-from]{A.90}{A.180}
\end{tenkz}
\begin{tenkz}[rows={wire}, cols=1, bonds=none, physical=up]
  \tn[name=B, ports={0:physical}]{B}
  \tnwire[name=policy-to]{B.0}{B.90}
\end{tenkz}
\end{document}
