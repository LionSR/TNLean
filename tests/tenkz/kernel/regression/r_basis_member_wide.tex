\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\tl_new:N \l__tenkz_test_member_span_x_tl
\tl_new:N \l__tenkz_test_member_span_y_tl
\cs_new_eq:NN \__tenkz_test_member_span_atoms: \__tenkz_kernel_r_atoms:
\cs_set_protected:Npn \__tenkz_kernel_r_atoms:
  {
    \__tenkz_model_map_ids:nn {atom}
      {
        \__tenkz_geom_xy:nNN {##1}
          \l__tenkz_test_member_span_x_tl
          \l__tenkz_test_member_span_y_tl
        \fp_compare:nNnF
          { abs ( \l__tenkz_test_member_span_x_tl - 0.75 ) }
          < {0.000001}
          { \errmessage{member~span~lost~its~offset~midpoint} }
        \fp_compare:nNnF
          { abs ( \l__tenkz_test_member_span_y_tl ) }
          < {0.000001}
          { \errmessage{member~span~left~its~row} }
      }
    \__tenkz_test_member_span_atoms:
  }
