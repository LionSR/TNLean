\newcommand{\tenkzAssertSameExtent}[3]{%
  \ifdim\wd#1=\wd#2\else
    \errmessage{#3 width depends on declarative whitespace}%
  \fi
  \ifdim\ht#1=\ht#2\else
    \errmessage{#3 height depends on declarative whitespace}%
  \fi
  \ifdim\dp#1=\dp#2\else
    \errmessage{#3 depth depends on declarative whitespace}%
  \fi
}
