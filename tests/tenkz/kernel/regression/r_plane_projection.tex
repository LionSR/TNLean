% label bearings on both records and ports use the full local basis.

\file_input:n { tenkz-kernel.code.tex }
\int_new:N \l__tenkz_test_plane_axis_count_int
\int_new:N \l__tenkz_test_plane_label_count_int
\tl_new:N \l__tenkz_test_plane_row_tl
\tl_new:N \l__tenkz_test_plane_pos_tl
\tl_new:N \l__tenkz_test_plane_angle_tl
\cs_new_eq:NN
  \__tenkz_test_plane_axis_place:
  \__tenkz_kernel_r_axis_place:
\cs_set_protected:Npn \__tenkz_kernel_r_axis_place:
  {
    \tl_set_eq:NN
      \l__tenkz_test_plane_row_tl \l__tenkz_kernel_r_axis_tl
    \__tenkz_test_plane_axis_place:
    \str_if_eq:eeT { \__tenkz_kernel_r_frame_word: } {plane}
      {
        \int_gincr:N \l__tenkz_test_plane_axis_count_int
        \fp_compare:nNnF
          {
            abs
              (
                \l__tenkz_kernel_r_axis_tl
                - \str_case_e:nn
                    { \l__tenkz_test_plane_row_tl }
                    { {2}{-0.6} {3}{-1.2} }
              )
          }
          < {0.000001}
          { \errmessage{plane~baseline~did~not~use~the~projected~row} }
      }
  }
\cs_new_eq:NN
  \__tenkz_test_plane_mark_label_ink:n
  \__tenkz_kernel_r_mark_label_ink:n
\cs_set_protected:Npn \__tenkz_kernel_r_mark_label_ink:n #1
  {
    \str_if_eq:eeT { \__tenkz_kernel_r_frame_word: } {plane}
      {
        \int_gincr:N \l__tenkz_test_plane_label_count_int
        \__tenkz_model_get:nnN {#1} {label-pos} \l__tenkz_test_plane_pos_tl
        \tl_set:Ne \l__tenkz_test_plane_angle_tl
          {
            \__tenkz_kernel_r_compass:n
              { \tl_use:N \l__tenkz_test_plane_pos_tl }
          }
        \tl_set:Ne \l__tenkz_test_plane_angle_tl
          {
            \fp_eval:n
              {
                \l__tenkz_test_plane_angle_tl
                + \l__tenkz_kernel_r_tau_tl
              }
          }
        \fp_compare:nNnF
          {
            abs
              (
                \l__tenkz_test_plane_angle_tl
                - \str_case:Vn \l__tenkz_test_plane_pos_tl
                    { {e}{0} {n}{53.130102} }
              )
          }
          < {0.000001}
          { \errmessage{plane~label~did~not~use~the~projected~bearing} }
      }
    \__tenkz_test_plane_mark_label_ink:n {#1}
  }
