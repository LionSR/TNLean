%% Regression: a projected three-by-three cell complex carries one source-shaped
%% route through three affine midpoints.  Its two crossings are geometric atoms,
%% not extra cells.  Physical ownership is exercised separately because the
%% twelve virtual grid bonds occupy the same four cell faces as picture ports.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn

\int_new:N \g__tenkz_test_affine_waypoint_int

\int_new:N \l__tenkz_test_affine_slot_int
\fp_new:N \l__tenkz_test_affine_actual_x_fp
\fp_new:N \l__tenkz_test_affine_actual_y_fp
\fp_new:N \l__tenkz_test_affine_expect_x_fp
\fp_new:N \l__tenkz_test_affine_expect_y_fp

\tl_new:N \l__tenkz_test_affine_geom_y_tl

\cs_new_protected:Npn \__tenkz_test_affine_compare:n #1
  {
    \__tenkz_kernel_r_frame_xy:nnNN
      { \l__tenkz_test_affine_row_fp }
      { \l__tenkz_test_affine_col_fp }
      \l__tenkz_test_affine_expect_x_fp
      \l__tenkz_test_affine_expect_y_fp
    \fp_compare:nNnF
      {
        abs(
          \l__tenkz_test_affine_actual_x_fp
          - \l__tenkz_test_affine_expect_x_fp )
      }
      < {0.00005}
      { \errmessage{affine~#1~lost~its~logical~x~coordinate} }
    \fp_compare:nNnF
      {
        abs(
          \l__tenkz_test_affine_actual_y_fp
          - \l__tenkz_test_affine_expect_y_fp )
      }
      < {0.00005}
      { \errmessage{affine~#1~lost~its~logical~y~coordinate} }
  }

\cs_new_eq:NN
  \__tenkz_test_affine_pts_add:nN
  \__tenkz_kernel_r_pts_add:nN
\cs_set_protected:Npn \__tenkz_kernel_r_pts_add:nN #1#2
  {
    \__tenkz_test_affine_pts_add:nN {#1} #2
    \int_gincr:N \g__tenkz_test_affine_waypoint_int
    \fp_set:Nn \l__tenkz_test_affine_actual_x_fp
      { \l__tenkz_kernel_r_xc_fp }
    \fp_set:Nn \l__tenkz_test_affine_actual_y_fp
      { \l__tenkz_kernel_r_yc_fp }
    \int_set:Nn \l__tenkz_test_affine_slot_int
      { 1 + \int_mod:nn { \g__tenkz_test_affine_waypoint_int - 1 } {3} }
    \int_case:nnF { \l__tenkz_test_affine_slot_int }
      {
        {1}
          {
            \fp_set:Nn \l__tenkz_test_affine_row_fp {2.5}
            \fp_set:Nn \l__tenkz_test_affine_col_fp {1.5}
          }
        {2}
          {
            \fp_set:Nn \l__tenkz_test_affine_row_fp {2.5}
            \fp_set:Nn \l__tenkz_test_affine_col_fp {2.5}
          }
        {3}
          {
            \fp_set:Nn \l__tenkz_test_affine_row_fp {1.5}
            \fp_set:Nn \l__tenkz_test_affine_col_fp {2.5}
          }
      }
      { \errmessage{affine~route~visited~an~unexpected~waypoint} }
    \__tenkz_test_affine_compare:n {route~waypoint}
  }
