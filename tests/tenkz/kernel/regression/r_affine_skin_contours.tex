% the complete basis of a sheared plane carrier.  In particular a box is a
% parallelogram, not a rectangle rotated by the carrier's east-axis angle.
% Flat and circular carriers retain their established scalar-turn path.

\cs_new_protected:Npn \__tenkz_test_affine_box:n #1
  {
    \int_gincr:N \l__tenkz_test_affine_box_int
    \pgfextractx \l__tenkz_test_ne_x_dim { \pgfpointanchor{#1}{north~east} }
    \pgfextracty \l__tenkz_test_ne_y_dim { \pgfpointanchor{#1}{north~east} }
    \pgfextractx \l__tenkz_test_nw_x_dim { \pgfpointanchor{#1}{north~west} }
    \pgfextracty \l__tenkz_test_nw_y_dim { \pgfpointanchor{#1}{north~west} }
    \pgfextractx \l__tenkz_test_se_x_dim { \pgfpointanchor{#1}{south~east} }
    \pgfextracty \l__tenkz_test_se_y_dim { \pgfpointanchor{#1}{south~east} }
    \pgfextractx \l__tenkz_test_sw_x_dim { \pgfpointanchor{#1}{south~west} }
    \pgfextracty \l__tenkz_test_sw_y_dim { \pgfpointanchor{#1}{south~west} }
    \fp_set:Nn \l__tenkz_test_east_x_fp
      {
        \dim_to_fp:n { \l__tenkz_test_ne_x_dim - \l__tenkz_test_nw_x_dim }
        / \dim_to_fp:n { \use:c {tenkz@pitch} }
      }
    \fp_set:Nn \l__tenkz_test_east_y_fp
      {
        \dim_to_fp:n { \l__tenkz_test_ne_y_dim - \l__tenkz_test_nw_y_dim }
        / \dim_to_fp:n { \use:c {tenkz@pitch} }
      }
    \fp_set:Nn \l__tenkz_test_north_x_fp
      {
        \dim_to_fp:n { \l__tenkz_test_nw_x_dim - \l__tenkz_test_sw_x_dim }
        / \dim_to_fp:n { \use:c {tenkz@pitch} }
      }
    \fp_set:Nn \l__tenkz_test_north_y_fp
      {
        \dim_to_fp:n { \l__tenkz_test_nw_y_dim - \l__tenkz_test_sw_y_dim }
        / \dim_to_fp:n { \use:c {tenkz@pitch} }
      }
    \fp_compare:nNnF { abs(\l__tenkz_test_east_y_fp) } < {0.00001}
      { \errmessage{the~plane~box~lost~its~horizontal~east~edge} }
    \fp_compare:nNnF
      {
        abs(
          \l__tenkz_test_north_x_fp
            * \__tenkz_metric_ratio:n {planerise}
          - \l__tenkz_test_north_y_fp
            * \__tenkz_metric_ratio:n {planeslant} )
      }
      < {0.00001}
      { \errmessage{the~plane~box~lost~its~sheared~north~edge} }
    \fp_compare:nNnF
      {
        abs(
          \l__tenkz_test_east_x_fp * \l__tenkz_test_north_y_fp
          - \l__tenkz_test_east_y_fp * \l__tenkz_test_north_x_fp )
      }
      > {0.00001}
      { \errmessage{the~plane~box~contour~collapsed} }
    \dim_compare:nNnF
      { \l__tenkz_test_ne_x_dim - \l__tenkz_test_nw_x_dim }
      =
      { \l__tenkz_test_se_x_dim - \l__tenkz_test_sw_x_dim }
      { \errmessage{the~plane~box~east~edges~are~not~parallel} }
    \dim_compare:nNnF
      { \l__tenkz_test_nw_y_dim - \l__tenkz_test_sw_y_dim }
      =
      { \l__tenkz_test_ne_y_dim - \l__tenkz_test_se_y_dim }
      { \errmessage{the~plane~box~north~edges~are~not~parallel} }
  }
