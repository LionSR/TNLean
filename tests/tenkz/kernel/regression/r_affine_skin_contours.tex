            \prop_get:NnN \l__tenkz_kernel_hull_live_prop {#1/xmax}

              \l__tenkz_test_affine_ymin_tl

              \l__tenkz_test_affine_ymax_tl

              {
                abs(\l__tenkz_kernel_r_basis_ey_tl)
                  * (\l__tenkz_test_affine_xmax_tl
                    - \l__tenkz_test_affine_xmin_tl) / 2
                + abs(\l__tenkz_kernel_r_basis_ny_tl)
                  * (\l__tenkz_test_affine_ymax_tl
                    - \l__tenkz_test_affine_ymin_tl) / 2
              }
            \prop_get:NnN \l__tenkz_kernel_hull_live_prop {#1/box}
              \l__tenkz_kernel_hull_box_tl
            \fp_set:Nn \l__tenkz_test_affine_expected_fp
              {
                \dim_ratio:nn
                  {
                    (\box_ht:c {\l__tenkz_kernel_hull_box_tl}
                      + \box_dp:c {\l__tenkz_kernel_hull_box_tl}) / 2
                  }
                  { \dim_use:c {tenkz@pitch} }
                + \__tenkz_metric_ratio:n {labelclear}
              }
            \fp_compare:nF
              {
                \l__tenkz_test_affine_support_fp + 0.00001
                  >= \l__tenkz_test_affine_expected_fp
              }
              { \errmessage{affine~ellipsis~support~missed~page-space~ink} }
            \int_gincr:N \l__tenkz_test_affine_dots_support_int
          }
      }
  }
