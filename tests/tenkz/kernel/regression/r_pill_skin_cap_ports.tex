% ordinary open legs.

\begin{document}

    ports={12:virtual:$a$, 28:virtual:$b$, 168:virtual, 192:virtual}]{P}
  \tnwire{flatcap.12}{open e} \tnwire{flatcap.28}{open ne}
  \tnwire{open w}{flatcap.192}
\end{tenkz}
\quad
\begin{tenkz}[rows={wire,wire,wire}, cols=4, bonds=none, size=m]
