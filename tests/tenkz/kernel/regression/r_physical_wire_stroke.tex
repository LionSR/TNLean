% Regression: closed index wires, including refinable grid bonds, use the
% stroke selected by their resolved endpoint type; virtual directions remain.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\int_new:N \g__tenkz_test_physical_stroke_int
\int_new:N \g__tenkz_test_physical_open_stroke_int
\int_new:N \g__tenkz_test_physical_policy_stroke_int
\int_new:N \g__tenkz_test_virtual_stroke_int
\int_new:N \g__tenkz_test_grid_physical_stroke_int
\int_new:N \g__tenkz_test_grid_virtual_stroke_int
\cs_new_eq:NN
  \__tenkz_test_wire_stroke:nn
  \__tenkz_kernel_r_wire_stroke:nn
\cs_set_protected:Npn \__tenkz_kernel_r_wire_stroke:nn #1#2
  {
    \exp_args:NV \__tenkz_model_get:nnN
      \l__tenkz_kernel_r_wire_record_tl {name} \l_tmpa_tl
    \str_case:Vn \l_tmpa_tl
      {
        {physical-wire}
          {
            \str_if_eq:eeF {#1} {physical~leg}
              {\errmessage{physical~wire~lost~physical~stroke}}
            \int_gincr:N \g__tenkz_test_physical_stroke_int
          }
        {physical-open}
          {
            \str_if_eq:eeF {#1} {physical~leg}
              {\errmessage{physical~open~wire~lost~physical~stroke}}
            \int_gincr:N \g__tenkz_test_physical_open_stroke_int
          }
        {virtual-wire}
          {
            \str_if_eq:eeF {#1} {bond}
              {\errmessage{virtual~wire~lost~bond~stroke}}
            \int_gincr:N \g__tenkz_test_virtual_stroke_int
          }
      }
    \__tenkz_model_get:nnN
      { \l__tenkz_kernel_r_wire_record_tl } {origin} \l_tmpb_tl
    \str_if_eq:VnT \l_tmpb_tl {grid}
      {
        \str_if_eq:eeTF {#1} {physical~leg}
          { \int_gincr:N \g__tenkz_test_grid_physical_stroke_int }
          {
            \str_if_eq:eeTF {#1} {bond}
              { \int_gincr:N \g__tenkz_test_grid_virtual_stroke_int }
              { \errmessage{grid~wire~used~an~unknown~stroke} }
          }
      }
    \__tenkz_model_get:nnN
      { \l__tenkz_kernel_r_wire_record_tl } {mode} \l_tmpc_tl
    \bool_lazy_and:nnT
      { \str_if_eq_p:Vn \l_tmpb_tl {open} }
      { \str_if_eq_p:Vn \l_tmpc_tl {physical} }
      {
        \str_if_eq:eeF {#1} {physical~leg}
          {\errmessage{physical~open~policy~lost~physical~stroke}}
        \int_gincr:N \g__tenkz_test_physical_policy_stroke_int
      }
    \__tenkz_test_wire_stroke:nn {#1} {#2}
  }
\AtEndDocument
  {
    \int_compare:nNnF { \g__tenkz_test_physical_stroke_int } = {1}
      {\errmessage{physical~wire~stroke~count~changed}}
    \int_compare:nNnF { \g__tenkz_test_physical_open_stroke_int } = {1}
      {\errmessage{physical~open~wire~stroke~count~changed}}
    \int_compare:nNnF { \g__tenkz_test_physical_policy_stroke_int } = {1}
      {\errmessage{physical~open~policy~stroke~count~changed}}
    \int_compare:nNnF { \g__tenkz_test_virtual_stroke_int } = {1}
      {\errmessage{virtual~wire~stroke~count~changed}}
    \int_compare:nNnF { \g__tenkz_test_grid_physical_stroke_int } = {2}
      {\errmessage{physical~grid~wire~stroke~count~changed}}
    \int_compare:nNnF { \g__tenkz_test_grid_virtual_stroke_int } = {2}
      {\errmessage{virtual~grid~wire~stroke~count~changed}}
  }
\ExplSyntaxOff
\makeatother
