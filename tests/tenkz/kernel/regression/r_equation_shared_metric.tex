% Regression: one equation, one measure.  An equation says its two panels are
% the same object written two ways, so the names inscribed in them are set on
% one measure and the glyphs holding them come out the same size.  X and its
% inverse are the test the contract names: the inverse is the wider name, and
% before the shared measure the two operators were drawn a quarter apart in
% width, which reads as two different objects.  The class reference floors
% both axes, so the height was already shared; the width is what the name
% pushes past, and what one measure settles.  The measure only ever grows a
% glyph, so the shorter name keeps its ink and the longer one is never
% squeezed.
\documentclass{standalone}
\usepackage{tenkz}
\begin{document}
\tenkzkernel
\begin{tenkzeq}[check={signature}]
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=box, ports={0:virtual, 180:virtual}]{X}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=open, east=open]
    \tn[skin=box, ports={0:virtual, 180:virtual}]{X^{-1}}
  \end{tenkz}
\end{tenkzeq}
\end{document}
