%% Regression: a declared cell basis is transported by the full plane map.
%% Selecting a cell resolves all of its members; (r,c,k) resolves exactly one.
\documentclass{standalone}
\usepackage{tenkz}
\makeatletter
\ExplSyntaxOn
\file_input:n { tenkz-kernel.code.tex }
\int_new:N \g__tenkz_test_basis_member_int
\tl_new:N \l__tenkz_test_basis_member_tl
\tl_new:N \l__tenkz_test_basis_x_tl
\tl_new:N \l__tenkz_test_basis_y_tl
\cs_new_eq:NN \__tenkz_test_basis_atoms: \__tenkz_kernel_r_atoms:
\cs_set_protected:Npn \__tenkz_kernel_r_atoms:
  {
    \__tenkz_model_map_ids:nn {atom}
      {
        \__tenkz_model_get:nnN {##1} {member}
          \l__tenkz_test_basis_member_tl
        \quark_if_no_value:NF \l__tenkz_test_basis_member_tl
          {
            \int_gincr:N \g__tenkz_test_basis_member_int
            \__tenkz_geom_xy:nNN {##1}
              \l__tenkz_test_basis_x_tl \l__tenkz_test_basis_y_tl
            \fp_compare:nNnF
              {
                abs
                  (
                    \l__tenkz_test_basis_x_tl
                    - \str_case:Vn \l__tenkz_test_basis_member_tl
                        { {1}{0} {2}{1} {3}{0.45} }
                  )
              }
              < {0.000001}
              { \errmessage{basis~member~lost~its~projected~x~coordinate} }
            \fp_compare:nNnF
              {
                abs
                  (
                    \l__tenkz_test_basis_y_tl
                    - \str_case:Vn \l__tenkz_test_basis_member_tl
                        { {1}{0} {2}{0} {3}{0.6} }
                  )
              }
              < {0.000001}
              { \errmessage{basis~member~lost~its~projected~y~coordinate} }
          }
      }
    \__tenkz_test_basis_atoms:
  }
\ExplSyntaxOff
\makeatother
