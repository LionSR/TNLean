% Regression: pentagon_corrected — renders Internal charges: x = ab, y = abc, u = bc, w = bcd, v = cd.
% Formula: Internal charges: x = ab, y = abc, u = bc, w = bcd, v = cd.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
% Pentagon for four charges a,b,c,d fusing to total charge e.
% Internal charges: x = ab, y = abc, u = bc, w = bcd, v = cd.
% Conventions: Bultinck et al., arXiv:1511.08090 eq. (33)
% (F^{abc}_d: fused charges up, total charge down; F-move maps the
% left-associated tree to the right-associated tree), matching
% arXiv:2203.12563 eqs. (5)-(6) up to super/subscript transposition.
% Every F-symbol slot is a simple charge; the mixed slots carry the
% internal charges u, x, v -- not composite labels like "bc".
\begin{tenkzcd}[polygon=5, radius=34mm]
  \tntree{(((a\,b)_x\,c)_y\,d)_e} &
  \tntree{((a\,(b\,c)_u)_y\,d)_e} &
  \tntree{(a\,((b\,c)_u\,d)_w)_e} &
  \tntree{(a\,(b\,(c\,d)_v)_w)_e} &
  \tntree{((a\,b)_x\,(c\,d)_v)_e}
  \tnarrow[from=1, to=2]{F^{abc}_y}
  \tnarrow[from=2, to=3]{F^{aud}_e}
  \tnarrow[from=3, to=4]{F^{bcd}_w}
  \tnarrow*[from=1, to=5]{F^{xcd}_e}
  \tnarrow*[from=5, to=4]{F^{abv}_e}
\end{tenkzcd}
\end{document}
