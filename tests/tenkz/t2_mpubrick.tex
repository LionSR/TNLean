% Regression: t2_mpubrick — renders U^{(N)} = prod_k U(site k) = ( (x)_k u_{2k-1,2k} ) . ( (x)_k v_{2k,2k+1} )…
% t2_mpubrick — RMP arXiv:2011.12127, fig. sec2fig7b(b) second panel,
% file fig2_fig2-pepe_sec2fig8b2.pdf: the MPU chain equals a two-layer
% brick circuit of nearest-neighbour unitaries.
%
% Formula:
%   U^{(N)} = prod_k U(site k)  =  ( (x)_k u_{2k-1,2k} ) . ( (x)_k v_{2k,2k+1} )
% — six simple MPU site tensors on one virtual wire (up output / down
% input legs each) equal a quantum circuit of two layers: unitaries u on
% the even-odd pairs, unitaries v on the odd-even pairs, brick-offset by
% one site (caption of fig. sec2fig7b).
%
% Declarative reading:
%   LHS — rows={op}: the six-site MPU word, virtual stubs west/east,
%   physical legs up and down per site.
%   RHS — the grid's circuit convention (hard11): wires are rows, gates
%   are one-atom bricks spanning two wires (wires=2); frame=vertical
%   rotates the whole grammar 90 degrees so the six wires run
%   vertically (outputs north, inputs south) as in the reference, u
%   bricks on wire pairs (1,2),(3,4),(5,6), v bricks offset on
%   (2,3),(4,5).
%
% House-style deviations (noted, not blockers): each reference gate is
% drawn as a square--circle sub-tensor pair grouped by a dotted ellipse
% and coloured (u red, v blue); the grid states the gate as ONE labelled
% brick — the sub-tensor granularity and the red/blue coding have no
% grid keys (per-atom hue is unexposed).  The reference's two cut
% half-gates at the window edges (west stub bottom-left, east stub
% bottom-right) are likewise not expressible: a brick is whole.
% Audit note: the rotated circuit logs its wire ends as VIRTUAL
% (vw 6 / ve 6) while the LHS logs pu 6 / pd 6 — the circuit
% convention's known signature skew, not an object mismatch.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: six simple MPU tensors on the virtual wire
  \begin{tenkz}[rows={op}, tensor style=box]
    \tn{U} & \tn{U} & \tn{U} & \tn{U} & \tn{U} & \tn{U}
  \end{tenkz}
  \;=\;
  % RHS: the brick circuit, rotated to the reference's frame — wires
  % vertical, u layer first (west column = top layer before rotation)
  \begin{tenkz}[rows={wire,wire,wire,wire,wire,wire}, frame=vertical]
    \tn[box, wires=2]{u} & \\
    & \tn[box, wires=2]{v} \\
    \tn[box, wires=2]{u} & \\
    & \tn[box, wires=2]{v} \\
    \tn[box, wires=2]{u} & \\
    &
  \end{tenkz}
\]
\end{document}
