% Declarative reading per side (rows={ket:nopair, bra}):
%   * word row: surviving tensors, \tnskip the exposed site (virtual

%
% Noted gaps: identical to t2_lhsrhs2 (hole stubs horizontal instead of

%
% Layout note: the hole must sit in an INTERIOR column.  A first draft
% put \tnskip in the last column of the cupped row and the east cup

% boundary loop.  A trailing \tnghost keeps the cup anchored on a real

\[
  % LHS: A B [hole] (wire) over the boundary tensor L
  \begin{tenkz}[rows={ket:nopair, bra}, west=cup, east=cup,
                tensor style=box]
    \tn{A} & \tn{B} & \tnskip & \tnghost{}\\
    \tn[pill, wide=4]{L} & & &
  \end{tenkz}
  \;=\;
  % RHS: (wire) B [hole] C over the boundary tensor R
  \begin{tenkz}[rows={ket:nopair, bra}, west=cup, east=cup,
                tensor style=box]
    \tnghost{} & \tn{B} & \tnskip & \tn{C}\\
    \tn[pill, wide=4]{R} & & &
  \end{tenkz}
\]
