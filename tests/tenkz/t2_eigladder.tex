% Regression: t2_eigladder — renders eig( rho_n ) = eig( rho^R E^n rho^L ) (up to the cut).
% T2 eigladder — RMP arXiv:2011.12127, fig. fig9-2/fig9-3 genre (the
% fig9-3 panel): the entanglement spectrum of a block of n sites written
% through the transfer-matrix ladder between the fixed points.
%
% Formula:  eig( rho_n ) = eig( rho^R  E^n  rho^L )   (up to the cut),
%   E = sum_s A^s (x) conj(A^s)  the transfer matrix: each rung of the
% ladder is one ket tensor over its conjugate joined by the contracted
% physical index; rho^R sits INLINE on the ket rail at the west end,
% rho^L inline on the bra rail at the east end, and the four remaining
% virtual indices stay open (the matrix being diagonalized).
%
% Declarative reading:
%   rows={ket,bra}       — the paired sandwich: each \tn{} over \tn*{}
%                          column IS one rung sum_s A^s (x) conj(A^s).
%   \tn[no legs]{\rho^R} — a fixed-point matrix inline on the rail: a
%                          bead with no physical leg.
%   \tnghost{}           — extends the opposite rail straight through the
%                          fixed-point column so both rails span the word.
%   \tnspan[brace below] — marks the n-site window.
%
% Noted gaps (see verdict): (1) the reference turns the four open ends
% OUT vertically (up at the ket ends, down at the bra ends) — tenkz has
% no bent-stub boundary key, so the ends here are straight west/east
% stubs with the same boundary signature (vw 2, ve 2, pu 0, pd 0);
% (2) the reference marks the window by a red dashed RECTANGLE — the
% grid has \tncut (a full-height dashed cut line) and \tnspan (a brace)
% but no region-overlay primitive; the brace stands in.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  \mathrm{eig}\left(
  % ket rail: rho^R then four A beads; bra rail: four conj(A) beads
  % then rho^L; ghosts carry each rail past the other's fixed point
  \begin{tenkz}[rows={ket,bra}]
    \tn[no legs]{\rho^R} & \tn{} & \tn{} & \tn{} & \tn{} & \tnghost{}\\
    \tnghost{} & \tn*{}\tnspan[brace below]{4}{n} & \tn*{} & \tn*{} &
      \tn*{} & \tn[no legs]{\rho^L}
  \end{tenkz}
  \right)
\]
\end{document}
