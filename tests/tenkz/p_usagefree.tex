\usepackage{tenkz}

  \tnput[ports={west:virtual}]{b}{(14mm,0)}{B}
  \tnjoin{a.east}{b.west}
  \tnjoin[route=curve, out=90, in=90, label=T]{a.north}{b.north}
\end{tenkzfree}
