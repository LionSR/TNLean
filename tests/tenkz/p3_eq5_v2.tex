% Formula: ((a b)_e c)_d  =  sum_{f,chi,eta} (F_{abc}^d)^{e mu nu}_{f chi eta}
% P3 v2 — arXiv:2203.12563 eq (5): the F-symbol between the two fusion
% orders of three MPO blocks,
%   ((a b)_e c)_d  =  sum_{f,chi,eta} (F_{abc}^d)^{e mu nu}_{f chi eta}
%                     (a (b c)_f)_d .
% Paper ink: cascaded chirality fat bars read root-west (d enters, splits
% into e/c, then e splits into a/b), with vertex multiplicities nu, mu
% (left tree) and eta, chi (right tree) beside the bars.
%
% tenkz spelling: \tntree is the sanctioned fusion-tree object — leaves on
% the top line, one junction per fused pair, internal charge on the
% downward edge, total charge on the root leg.  The cascaded-bar ink
% itself is out of the grid grammar (\tnfuse cannot take a fused wire as
% one of its pair inputs), so the tree IS the reproduction, in tenkz's
% root-down normalization of the paper's root-west ink.
% Noted gap: junctions carry no multiplicity label slot; mu, nu, chi, eta
% survive only inside the F coefficient.  The second display shows the
% pragmatic workaround — the multiplicity folded into the edge charge.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}

% plain charges, multiplicities in the coefficient only
\[
  \tntree{((a\,b)_e\,c)_d}
  \;=\;
  \sum_{f,\chi,\eta}
  \left(F_{abc}^{d}\right)_{f\chi\eta}^{e\mu\nu}\,
  \tntree{(a\,(b\,c)_f)_d}
\]

% workaround: multiplicity folded into the internal edge charge
\[
  \tntree{((a\,b)_{e,\mu}\,c)_d}
  \;=\;
  \sum_{f,\chi,\eta}
  \left(F_{abc}^{d}\right)_{f\chi\eta}^{e\mu\nu}\,
  \tntree{(a\,(b\,c)_{f,\chi}\,)_d}
\]

\end{document}
