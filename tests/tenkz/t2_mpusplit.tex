% Regression: t2_mpusplit — imported current-behavior fixture from the handoff corpus.
% t2_mpusplit — RMP arXiv:2011.12127, fig. sec2fig7b(b) first panel,
% file fig2_fig2-pepe_sec2fig8b1.pdf: the simple (normal-form) MPU
% tensor splits into an upper and a lower factor.
%
% Formula:
%   U^{i j}_{alpha beta} = sum_gamma  u^{i}_{gamma beta}  v^{j}_{alpha gamma}
% — the 4-leg unitary site tensor U (west/east virtual, north output i,
% south input j) factorizes through one new internal index gamma into
% the upper factor u (north leg i, EAST virtual leg beta only) and the
% lower factor v (south leg j, WEST virtual leg alpha only); the
% reference then redraws the same equation with round glyphs (the
% reference distinguishes the factors as red/blue fills; here they are
% the labels u, v).
%
% Declarative reading:
%   panel 1 — rows={op}: the 4-leg U box.
%   panels 2, 3 — rows={op,op}: op-onto-op pairing gives the internal
%   gamma bond, u keeps its north leg, v its south leg.
%
% STILL-BLOCKED (leg pattern): u must have ONLY an east virtual stub and
% v ONLY a west one — one open side per row, opposite per row.  The
% rows= suffix (open/none) seals BOTH sides of its row, and the side
% policies west=/east= apply to ALL rows, so the diagonal stub pattern
% {row 1: east only, row 2: west only} is not expressible.  Missing
% primitive: a per-row-per-side rows= suffix, e.g.
% rows={op:west none, op:east none}.  The rendering below uses
% boundary=open and therefore draws TWO spurious stubs (west of u, east
% of v), changing both factors' arity from 3 to 4.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % panel 1: the simple MPU tensor, all four legs open
  \begin{tenkz}[rows={op}, tensor style=box]
    \tn{U}
  \end{tenkz}
  \;=\;
  % panel 2: u over v through the internal bond (spurious west/east
  % stubs, see header)
  \begin{tenkz}[rows={op,op}, tensor style=box]
    \tn{u} \\
    \tn{v}
  \end{tenkz}
  \;=\;
  % panel 3: the same split with round glyphs
  \begin{tenkz}[rows={op,op}, tensor style=dot]
    \tn{u} \\
    \tn{v}
  \end{tenkz}
\]
\end{document}
