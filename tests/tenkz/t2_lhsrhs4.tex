% Regression: t2_lhsrhs4 — applies the four-leg inverse S to a jointly injective A-B tensor pair.
% t2_lhsrhs4 — RMP arXiv:2011.12127, parent-Hamiltonian intersection
% argument, fourth displayed step, files fig4_fig4-pepe_LHS4.pdf /
% RHS4.pdf: applying the inverse S of the jointly injective pair (A,B).
%
% Formula:  S is the 4-leg inverse of the map (virtual x virtual) ->
% (physical x physical) defined by A.B; applying it to the first two
% physical indices of the step-1 equation gives
%   (S (x) 1) [ A B (wire) . L-loop ]  =  (S (x) 1) [ (wire) B C . R-loop ],
% whose LHS collapses to bare open index pairs over L and whose RHS
% keeps S drawn ABOVE the word: S's two down legs attach to (i) the
% boundary tensor R's escaped physical leg at the west cup and (ii) B's
% physical leg; S's two up legs stay open.
%
% Attempted declarative reading:
%   * LHS: two \tnskip holes (all virtual + physical indices of the
%     removed sites open) + ghost wire into the east cup, boundary L as
%     a wide bra-row pill;
%   * RHS: an operator row above the sandwich — rows={op:none,
%     ket:nopair, bra}; S = \tn[wide=2, legs at={1,2}] spanning word
%     columns 1-2; cups tie rows 2-3 (row 1 is boundary-sealed, hence
%     cup-ineligible by the pair-selection rule).
%
% STILL BLOCKED (RHS): S's western down leg must land on the PHYSICAL
% leg that the reference routes up over the west cup from the boundary
% tensor R.  A tenkz cup cannot emit an open physical stub at its bend,
% so that target does not exist in the grid; the leg instead meets the
% \tnghost wire cell below and is dropped with the documented
% `ghost-pair` warning.  Exact missing primitive: an open physical stub
% carried over a `west=cup`/`east=cup` bend (a cup-apex physical
% index) that pairing can attach to.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: two holes and a wire over the boundary tensor L
  \begin{tenkz}[rows={ket:nopair, bra}, east=cup, tensor style=box]
    \tnskip & \tnskip & \tnghost{}\\
    \tn[pill, wide=3]{L} & &
  \end{tenkz}
  \;=\;
  % RHS: S above the word (wire) B [hole] C over the boundary tensor R
  \begin{tenkz}[rows={op:none, ket:nopair, bra}, west=cup, east=cup,
                tensor style=box]
    \tn[wide=2, legs at={1,2}]{S} & & &\\
    \tnghost{} & \tn{B} & \tnskip & \tn{C}\\
    \tn[pill, wide=4]{R} & & &
  \end{tenkz}
\]
\end{document}
