% Regression: t2_rhoR — renders rho_R = Tr_{Rbar}|psi><psi| in lasso and boundary-state forms.
% Formula: e (using Figboundary-theoryc's lassos A^I, B^{Ibar} and
% t2_rhoR — RMP arXiv:2011.12127, fig2-pepe_Figboundary-theoryb: the
% reduced state of region R through the lasso decomposition, and its
% boundary-state form.
%
% Formulae (using Figboundary-theoryc's lassos A^I, B^{Ibar} and
% Figboundary-theorya's polar pieces A^I = U^A_I sigma^A,
% B^{Ibar} = sigma^B U^B_{Ibar}):
%   rho_R = Tr_{Rbar} |psi><psi|
%         = sum_{I,I'} ( A^{I} E conj(A^{I'}) ) |I><I'| ,
%     E := sum_{Ibar} B^{Ibar} (x) conj(B^{Ibar})   [the east cup]
%   rho_R = U ( sigma_{\partial R} ) U^dagger ,
%     sigma_{\partial R} := sigma^A sigma^B (x) conj(sigma^A sigma^B)
%     closed by the same cup — the BOUNDARY STATE on the \partial R bonds.
%
% Declarative rendering: rows={ket:nopair, bra} (ket copy over bra copy,
% physical bundles I / I' stay open), boundary=none (the lasso has no
% wire west of A; the east ends are cup-closed, so sealing both sides is
% exact), east=cup ties the two B-wires — the Tr_{Rbar}.  Second
% picture: the isometry U as a box (up leg = I), then the on-wire scalars
% sigma^A, sigma^B as legless beads inside the cup; the brace names the
% enclosed object sigma_{\partial R}.
%
% Two deliberate deviations from the RMP ink (recorded in the verdict):
%  * the RMP's dotted rounded box around the sigma-dots-plus-cup has no
%    grid-genre spelling (the grid draws no semantic enclosures); the
%    brace span carries the sigma_{\partial R} naming instead;
%  * the RMP draws the ket and bra physical stubs at the west column as
%    one continuous vertical line (they touch); tenkz keeps the honest
%    gap between the ket up leg and the bra down leg — the two bundles
%    are open indices, not a contraction.

\usepackage{tenkz}
\begin{document}
\[
  \rho_R \;=\;
  \begin{tenkz}[rows={ket:nopair, bra}, boundary=none, east=cup]
    \tn{} & \tn[no legs]{}\\
    \tn*{} & \tn*[no legs]{}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={ket:nopair, bra}, boundary=none, east=cup]
    \tn[box]{U^{A}_{I}} &
      \tn[no legs]{\sigma^{A}}\tnspan[brace above]{2}{\sigma_{\partial R}} &
      \tn[no legs]{\sigma^{B}}\\
    \tn*[box]{U^{A}_{I}} & \tn*[no legs]{\sigma^{A}} & \tn*[no legs]{\sigma^{B}}
  \end{tenkz}
\]
\end{document}
