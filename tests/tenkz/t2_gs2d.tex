% Regression: t2_gs2d — renders |psi(X)> = sum_{s} X( {beta_r} ) C( prod_r A^{s_r} ) |s>.
% t2_gs2d — RMP arXiv:2011.12127, fig4-pepe_GS2D: the 2D analog of GS1D.
%
% Formula: |psi(X)> = sum_{s} X( {beta_r} ) C( prod_r A^{s_r} ) |s>,
% i.e. the finite PEPS window whose dangling virtual boundary indices
% {beta_r} are contracted with one boundary operator X living on the rim
% — the 2D generalization of tr( X A^{s_1}..A^{s_N} ).
%
% The RMP draws X as an ANNULUS: the band between two nested closed
% curves around the window; every cut virtual bond ends inside the band.
% Declarative rendering: oblique lattice window, boundary legs for the
% cut virtual indices, physical=up for the on-site physical legs, and two
% outline-only regions over the full window — the inset=1 loop is the
% annulus' inner rim, the inset=0 loop its outer rim, and the label X
% names the band.
%
% Noted gaps (cosmetic, recorded in the verdict):
%  * no annular-operator primitive: the band is spelled as two nested
%    outlines, which do not join into one glyph;
%  * register: boundary legs run 0.30 pitch but the inset=0 rim sits at
%    0.27 pitch, so leg tips overshoot the outer loop by 0.03 pitch;
%  * RMP sites are boxes; lattice sites render as beads (site=dot only).
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  |\psi(X)\rangle \;=\;
  \begin{tenkzlattice}[rows=4, cols=4, frame=oblique,
                       physical=up, boundary legs]
    \tnregion[outline, inset=1]{(1-4,1-4)}
    \tnregion[outline, label=$X$, label at=north east]{(1-4,1-4)}
  \end{tenkzlattice}
\]
\end{document}
