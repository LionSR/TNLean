% one sweep cell: {slant}{rise}{extra body}
\newcommand{\sweepcell}[3]{%
  \begin{minipage}[t]{6.6cm}\centering
    \vspace{0pt}%
    {\planeratios{#1}{#2}%
     \begin{tenkzlattice}[rows=4, cols=4, frame=oblique,
                          physical=up, boundary legs]
       #3
     \end{tenkzlattice}}\\[3pt]
    {\small slant $= #1$, rise $= #2$}%
  \end{minipage}}
