% Regression: free — exercises free-placement pictures through \tnput/\tnjoin.
% Formula: geometry-only; no tensor identity.
\documentclass[border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\newdimen\tenkzTestSmall
\newdimen\tenkzTestMedium
\newdimen\tenkzTestLarge
\newdimen\tenkzTestReference
\newdimen\tenkzTestHeight
\newcommand{\tenkzAssertWidthClasses}[4]{%
  \pgfextractx\tenkzTestSmall{\pgfpointdiff
    {\pgfpointanchor{#1}{west}}{\pgfpointanchor{#1}{east}}}%
  \pgfextractx\tenkzTestMedium{\pgfpointdiff
    {\pgfpointanchor{#2}{west}}{\pgfpointanchor{#2}{east}}}%
  \pgfextractx\tenkzTestLarge{\pgfpointdiff
    {\pgfpointanchor{#3}{west}}{\pgfpointanchor{#3}{east}}}%
  \pgfextractx\tenkzTestReference{\pgfpointdiff
    {\pgfpointanchor{#4}{west}}{\pgfpointanchor{#4}{east}}}%
  \ifdim\tenkzTestSmall<\tenkzTestMedium\else
    \errmessage{small glyph #1 is not narrower than medium glyph #2}\fi
  \ifdim\tenkzTestMedium<\tenkzTestLarge\else
    \errmessage{medium glyph #2 is not narrower than large glyph #3}\fi
  \ifdim\tenkzTestMedium=\tenkzTestReference\else
    \errmessage{medium glyph #2 differs from reference glyph #4}\fi}
\begin{document}
\begin{tenkzfree}
  \tnput[box]{A}{(0,0)}{A}
  \tnput[dot]{B}{(2,0)}{B}
  \tnput[pill]{C}{(4,0)}{C^{-1}}
  \tnput[mpo]{D}{(2,-2)}{O}
  \tnput[ring]{E}{(4,-2)}{X}
  \tnput[circle,size=s]{Cs}{(0,-4)}{}
  \tnput[circle,size=m]{Cm}{(2,-4)}{}
  \tnput[circle,size=l]{Cl}{(4,-4)}{}
  \tnput[circle]{Cref}{(6,-4)}{}
  \tnput[circle]{Clambda}{(8,-4)}{\lambda}
  \tnput[dot,size=s]{Ds}{(0,-6)}{}
  \tnput[dot,size=m]{Dm}{(2,-6)}{}
  \tnput[dot,size=l]{Dl}{(4,-6)}{}
  \tnput[dot]{Dref}{(6,-6)}{}
  \tnput[box,size=s]{Bs}{(0,-8)}{}
  \tnput[box,size=m]{Bm}{(2,-8)}{}
  \tnput[box,size=l]{Bl}{(4,-8)}{}
  \tnput[box]{Bref}{(6,-8)}{}
  \tnput[pill,size=s]{Ps}{(0,-10)}{}
  \tnput[pill,size=m]{Pm}{(2,-10)}{}
  \tnput[pill,size=l]{Pl}{(4,-10)}{}
  \tnput[pill]{Pref}{(6,-10)}{}
  \tnput[ring,size=s]{Rs}{(0,-12)}{}
  \tnput[ring,size=m]{Rm}{(2,-12)}{}
  \tnput[ring,size=l]{Rl}{(4,-12)}{}
  \tnput[ring]{Rref}{(6,-12)}{}
  % The size classes must change the measured hull, while explicit medium
  % is exactly the established reference extent.
  \pgfextractx\tenkzTestSmall{\pgfpointdiff
    {\pgfpointanchor{Cs}{west}}{\pgfpointanchor{Cs}{east}}}
  \pgfextractx\tenkzTestMedium{\pgfpointdiff
    {\pgfpointanchor{Cm}{west}}{\pgfpointanchor{Cm}{east}}}
  \pgfextractx\tenkzTestLarge{\pgfpointdiff
    {\pgfpointanchor{Cl}{west}}{\pgfpointanchor{Cl}{east}}}
  \pgfextractx\tenkzTestReference{\pgfpointdiff
    {\pgfpointanchor{Cref}{west}}{\pgfpointanchor{Cref}{east}}}
  \ifdim\tenkzTestSmall<\tenkzTestMedium\else
    \errmessage{small circle is not narrower than medium circle}\fi
  \ifdim\tenkzTestMedium<\tenkzTestLarge\else
    \errmessage{medium circle is not narrower than large circle}\fi
  \ifdim\tenkzTestMedium=\tenkzTestReference\else
    \errmessage{medium circle differs from the reference circle}\fi
  % A natural inscribed label may enlarge the glyph, but the result remains
  % circular: its measured horizontal and vertical diameters agree.
  \pgfextractx\tenkzTestReference{\pgfpointdiff
    {\pgfpointanchor{Clambda}{west}}{\pgfpointanchor{Clambda}{east}}}
  \pgfextracty\tenkzTestHeight{\pgfpointdiff
    {\pgfpointanchor{Clambda}{south}}{\pgfpointanchor{Clambda}{north}}}
  \ifdim\tenkzTestReference=\tenkzTestHeight\else
    \errmessage{labelled circle has unequal measured diameters}\fi
  \pgfextractx\tenkzTestSmall{\pgfpointdiff
    {\pgfpointanchor{Ds}{west}}{\pgfpointanchor{Ds}{east}}}
  \pgfextractx\tenkzTestMedium{\pgfpointdiff
    {\pgfpointanchor{Dm}{west}}{\pgfpointanchor{Dm}{east}}}
  \pgfextractx\tenkzTestLarge{\pgfpointdiff
    {\pgfpointanchor{Dl}{west}}{\pgfpointanchor{Dl}{east}}}
  \pgfextractx\tenkzTestReference{\pgfpointdiff
    {\pgfpointanchor{Dref}{west}}{\pgfpointanchor{Dref}{east}}}
  \ifdim\tenkzTestSmall<\tenkzTestMedium\else
    \errmessage{small dot is not narrower than medium dot}\fi
  \ifdim\tenkzTestMedium<\tenkzTestLarge\else
    \errmessage{medium dot is not narrower than large dot}\fi
  \ifdim\tenkzTestMedium=\tenkzTestReference\else
    \errmessage{medium dot differs from the reference dot}\fi
  % Every inscribed skin reads the same reference-width size classes.
  \tenkzAssertWidthClasses{Bs}{Bm}{Bl}{Bref}
  \tenkzAssertWidthClasses{Ps}{Pm}{Pl}{Pref}
  \tenkzAssertWidthClasses{Rs}{Rm}{Rl}{Rref}
  \tnjoin[route=straight]{A.east}{B.west}
  \tnjoin[route=hv, label=$D$]{B.east}{C.west}
  \tnjoin[route=curve, out=-90, in=90, fused]{B.south}{D.north}
  \tnjoin[route=vh]{C.south}{D.east}
\end{tenkzfree}
\end{document}
