% Formula: (gauge-fixed, A_L left-canonical, A_R right-canonical):
% T10 v2 — RMP arXiv:2011.12127, Fig. fig-tangent-plane (fig2-pepe_PTA):
% the tangent-space projector of a uniform MPS.
%
% Formula (gauge-fixed, A_L left-canonical, A_R right-canonical):
%   P_{T(A)} = sum_{i=1}^n  P_L^{(i-1)} (x) 1_i (x) P_R^{(i+1)}
%            - sum_{i=1}^n  P_L^{(i)}          (x) P_R^{(i+1)},
% with the block projectors
%   P_L^{(k)} = sum_alpha |Phi^L_alpha><Phi^L_alpha|,
%     |Phi^L_alpha> = A_L-chain on sites 1..k, right bond index alpha;
%   P_R^{(k)} = sum_beta  |Phi^R_beta><Phi^R_beta|,
%     |Phi^R_beta>  = A_R-chain on sites k..n, left bond index beta.
% The RMP draws each summand as ONE two-layer diagram: ket and bra
% chains of canonical triangles with OPEN physical hooks, tied
% ket-to-bra at the block edge by a bracket, a hole (blue identity wire)
% at site i in the first sum, and adjacent brackets with a blue bond
% mark in the second sum.
%
% Declarative status: STILL BLOCKED as a single diagram.  Missing keys:
%   (i)  `east=cup` / `west=cup` — tying the ket row's and bra
%        row's virtual wires to each other at a block edge (the
%        bracket); no rows= combination draws it;
%   (ii) `\tnskip` — a skipped site whose virtual wires stay open and
%        whose physical index is the bare identity wire (\tnghost
%        wrongly routes the bond through and silently drops the partner
%        leg; an absent cell wrongly bridges the bond across);
%   (iii) an unpaired ket-over-bra row pair — rows={ket,bra} always
%        contracts the layers' physical legs, but every physical leg in
%        this figure is open;
%   (iv) `tri` — triangle skins for canonical tensors (cosmetic).
% What IS declarative today: the factored form below — each |Phi> block
% is a one-layer grid chain with a labelled open bond stub, and the
% projector formula assembles them.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}

% |Phi^L_alpha><Phi^L_alpha| : ket chain (physical up) times bra chain
% (physical down, conjugated), sharing the open bond label alpha.  The
% unlabelled outer stubs are the blocks' chain-edge boundaries (the RMP
% draws them as wavy semi-infinite wires).
\[
  P_L \;=\; \sum_{\alpha}\;
  \begin{tenkz}[physical=up, east label={$\alpha$}, tensor style=box]
    \tn{A_L} & \tndots & \tn{A_L}
  \end{tenkz}
  \;
  \begin{tenkz}[physical=down, east label={$\alpha$}, tensor style=box]
    \tn*{A_L} & \tndots & \tn*{A_L}
  \end{tenkz}
\]
\[
  P_R \;=\; \sum_{\beta}\;
  \begin{tenkz}[physical=up, west label={$\beta$}, tensor style=box]
    \tn{A_R} & \tndots & \tn{A_R}
  \end{tenkz}
  \;
  \begin{tenkz}[physical=down, west label={$\beta$}, tensor style=box]
    \tn*{A_R} & \tndots & \tn*{A_R}
  \end{tenkz}
\]

% The projector: hole summand (identity at site i) minus the
% bond-contracted summand.
\[
  P_{T(A)} \;=\;
  \sum_{i=1}^{n} P_L^{(i-1)} \otimes \mathbb{I}_i \otimes P_R^{(i+1)}
  \;-\;
  \sum_{i=1}^{n} P_L^{(i)} \otimes P_R^{(i+1)}
\]
\end{document}
