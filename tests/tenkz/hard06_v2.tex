% Regression: hard06_v2 — renders O^{(s,s')}_{a b}.
% T06 v2 — RMP arXiv:2011.12127, Fig. sec2fig6 (fig2-pepe_sec2fig6-1):
% sufficient local condition for a translationally invariant MPO to be
% positive (local purification of the 4-leg MPO tensor O).
%
% Formula:  O^{(s,s')}_{a b}
%   = sum_{a1 a2 b1 b2 k} (X^{-1})_{a,(a1 a2)}
%       A^{(s,k)}_{a1 b1} conj(A)^{(s',k)}_{a2 b2} X_{(b1 b2), b},
% i.e.  O = X^{-1} [ sum_k A^k (x) conj(A^k) ] X, where A is the 4-leg
% purification tensor (physical s up, purification index k down onto
% conj(A), virtual a1/b1) and X is the 3-LEG gauge transform mapping the
% doubled virtual index (a1,a2) to O's single virtual index a.
%
% Declarative rendering: the 3-leg gauge tensors ARE row-fusing maps, so
% they are drawn with \tnfuse (bar: two layer wires -> one combined
% wire), labelled X^{-1} / X.  rows={op,op} pairs A's down leg with
% conj(A)'s up leg — that vertical bond is the purification index k —
% and leaves s (up) and s' (down) open.  Gap vs the RMP's styling only:
% the paper draws X as a tall rounded box spanning both layers with the
% label inscribed; the grammar's fusion bar carries the label beside the
% combined wire instead (a row-spanning labelled pill skin would close
% this).
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  \begin{tenkz}[physical=updown, tensor style=box]
    \tn{O}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={op,op}, tensor style=box]
    \tnfuse{X^{-1}} & \tn{A} & \tnfuse[combined=east]{X}\\
                    & \tn*{A} &
  \end{tenkz}
\]
\end{document}
