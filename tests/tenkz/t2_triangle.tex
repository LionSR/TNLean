% Formula: psi_{ijk} = sum_{alpha,beta,gamma} A^{i}_{gamma alpha}
% RMP 2011.12127 Fig. sec2fig3 (ref_ fig2_fig2-pepe_sec2fig3.pdf):
% a tensor network on a general graph, the smallest closed example.
%
%   psi_{ijk} = sum_{alpha,beta,gamma} A^{i}_{gamma alpha}
%               B^{j}_{alpha beta} C^{k}_{beta gamma}
%
% Three rank-3 tensors on a triangle: virtual (bond) indices alpha
% (top--right edge), beta (bottom edge), gamma (left edge) are contracted
% around the loop; physical indices i, j, k dangle radially outward.
% An odd ring is genuinely non-grid, so this is a free-tier picture.
\documentclass[varwidth=8cm, border=6pt]{standalone}
\usepackage{amsmath}
\usepackage{tenkz}
\begin{document}
\[
  \psi_{ijk} \;=\;
  \begin{tenkzfree}
    % the three tensors at the triangle corners
    \tnput[dot]{t}{(0,9mm)}{}
    \tnput[dot]{r}{(8mm,-4.5mm)}{}
    \tnput[dot]{l}{(-8mm,-4.5mm)}{}
    % contracted virtual loop: alpha, beta, gamma
    \tnjoin[label=\alpha]{t}{r}
    \tnjoin[label=\beta]{r}{l}
    \tnjoin[label=\gamma]{l}{t}
    % open physical indices i, j, k, radially outward
    \tnjoin[label=i]{t}{0,17mm}
    \tnjoin[label=j]{r}{14mm,-10mm}
    \tnjoin[label=k]{l}{-14mm,-10mm}
  \end{tenkzfree}
\]
\end{document}
