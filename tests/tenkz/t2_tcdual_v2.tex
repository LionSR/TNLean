% Regression: t2_tcdual_v2 — renders physical spins = differences of adjacent plaquette colours.
% Formula: physical spins = differences of adjacent plaquette colours
% RMP 2011.12127, appendix "The Toric Code and quantum double models",
% eq:app:tcode-rep-dual (ref_ fig_app_tc-dual.pdf).  Corrects
% t2_tcdual.tex, which drew the PRIMAL figure's diamond: the dual
% tensor's interior is the dashed corner-to-corner cross.
%
%   Dual quantum-double PEPS: colour variables h in G on the PLAQUETTES,
%   physical spins = differences of adjacent plaquette colours
%   (arXiv:cond-mat/0601075 school; RMP eq:app:tcode-rep-dual).  The four
%   straight legs carry the virtual plaquette colours
%       h_1 (west),  h_2 (depth-up),  h_3 (east),  h_4 (depth-down);
%   the dashed corner-to-corner cross is the original lattice inside the
%   blocked cell, whose edges carry the physical difference spins
%       h_2 h_1^{-1} (NW),  h_3 h_2^{-1} (NE),
%       h_4 h_1^{-1} (SW),  h_3 h_4^{-1} (SE)
%   (squiggle marks in the paper).  The representation is G-injective
%   (G-isometric) w.r.t. the regular representation: shifting every
%   colour h_i -> g h_i leaves all physical differences invariant.
%
% Declarative model: a 3x3 oblique sheet; the shaded blocked tensor is a
% filled region over all cells; corners + centre removed, so the only
% surviving lattice ink is the four mid-side boundary legs (the virtual
% plaquette colours); the original lattice is the corner-to-corner pair
% of distinguished edges.
%
% Honest gaps: \tnedge parses but ignores its option slot, so the cross
% cannot be dashed (the paper draws the original lattice dashed); no
% leg/edge label key, so the h labels cannot be attached; the physical
% squiggle marks have no primitive.
\documentclass[border=6pt]{standalone}
\usepackage{tenkz}
\begin{document}
\begin{tenkzlattice}[rows=3, cols=3, frame=oblique, site=none, boundary legs]
  % the blocked tensor (shaded sheet)
  \tnregion[slot=selected]{(1-3,1-3)}
  % keep only the four mid-side sites: their boundary legs are the four
  % virtual plaquette colours h_1..h_4
  \tnsite[removed]{(1,1)} \tnsite[removed]{(1,3)}
  \tnsite[removed]{(3,1)} \tnsite[removed]{(3,3)}
  \tnsite[removed]{(2,2)}
  % original lattice inside the blocked cell (dashed in the paper; solid
  % here -- see Honest gaps): edges carrying the difference spins
  \tnedge{(1,1)-(3,3)}
  \tnedge{(1,3)-(3,1)}
\end{tenkzlattice}
\end{document}
