\__tenkz_geom_frame_declare_rotation:nnnn {rotA} {30} {0} {0}

\__tenkz_geom_frame_declare_affine:nnnnnnn {shear} {1} {1/2} {0} {1} {2} {0}
\__tenkz_geom_frame_compose:nnn {rotshear} {rotA} {shear}
\__tenkz_geom_apply:nnnNN {rotshear} {1} {1} \l_probe_xa_tl \l_probe_ya_tl
\__tenkz_geom_probe:nn {compose-11}
  { x = \__tenkz_geom_round:n { \l_probe_xa_tl } | y = \__tenkz_geom_round:n { \l_probe_ya_tl } }
% direction transport bends with the shear: north through `shear` leans east
\__tenkz_geom_dir:nnN {shear} {90} \l_probe_ta_tl
\__tenkz_geom_probe:nn {shear-north}
  { angle = \__tenkz_geom_round:n { \l_probe_ta_tl } }
