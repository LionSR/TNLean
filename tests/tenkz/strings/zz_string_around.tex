% Regression: zz_string_around — a detour past a measured rectangle.
% Formula: apex = centre + (support(90) + daylight) * unit(90), pitch units
%   scaled by 1.1 cm/pitch: support of rect hy=0.3 northward is 0.3, plus
%   daylight 0.15, so apex = (0, 0.495) cm.
% Expectation: stringapex event x=0, y=0.495; ink arcs smoothly over the box.
\documentclass[varwidth=8cm, border=6pt]{standalone}
\usepackage{tenkz}
\usetikzlibrary{hobby,spath3,intersections}
\begin{document}
\makeatletter\input{tenkz-string.code.tex}\makeatother
\begin{tikzpicture}
\draw ({-0.5*1.1},{-0.3*1.1}) rectangle ({0.5*1.1},{0.3*1.1});
\ExplSyntaxOn
\__tenkz_string_reset:
\__tenkz_string_declare_around:nnnnnn {det} {-1.5,0} {1.5,0} {90}
  { {rect} {0.5} {0.3} {0} {0} {0} {0} } {1.1}
\__tenkz_string_resolve:
\__tenkz_string_dev_draw:nn {det} { thick }
\ExplSyntaxOff
\end{tikzpicture}
\end{document}
