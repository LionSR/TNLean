% Regression: zz_string_braid — declared under-crossings
% (rmp-iii-b-braid-one topology).
% Formula: strand b passes under strand a exactly once; the under path splits
%   into two components with a crossgap break; over path drawn whole.
% Expectation: stringcross event hits=1; visually one clean gap in b at the
%   crossing, both strands smooth S-curves with no kinks.
\documentclass[varwidth=8cm, border=6pt]{standalone}
\usepackage{tenkz}
\usetikzlibrary{hobby,spath3,intersections}
\begin{document}
\begin{tikzpicture}
\ExplSyntaxOn
\__tenkz_string_require_libs:
\__tenkz_string_reset:
% a: top-left dot column descending to bottom-right (over)
\__tenkz_string_declare_open:nn {sa} { {0,2} {0.9,1.3} {2,0} }
% b: top-right descending to bottom-left (under)
\__tenkz_string_declare_open:nn {sb} { {2,2} {1.1,1.2} {0,0} }
% c: a horizontal strand passing under both descenders -- the
% multi-crossing case; two gaps leave three components.
\__tenkz_string_declare_open:nn {sc} { {-0.2,1.15} {1.0,0.95} {2.2,1.15} }
\__tenkz_string_under:nn {sb} {sa}
\__tenkz_string_under:nn {sc} {sa}
\__tenkz_string_under:nn {sc} {sb}
\__tenkz_string_resolve:
\__tenkz_string_dev_draw:nn {sc} { thick }
\__tenkz_string_dev_draw:nn {sb} { thick }
\__tenkz_string_dev_draw:nn {sa} { thick }
\ExplSyntaxOff
\end{tikzpicture}
\end{document}
