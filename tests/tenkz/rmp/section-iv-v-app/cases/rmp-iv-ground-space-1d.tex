% Target: rmp-iv-ground-space-1d
% Formula: |psi(X)> = sum_{s_1..s_N} tr( X A^{s_1} A^{s_2} ... A^{s_N} )
% Ink: Kernel flat frame; the traced A word returns below the row through corner
%   junctions with the square X tensor inserted on the return.
% Boundary: The virtual chain closes through X below the row; every shown site
%   keeps its upward physical leg.
% Source: paper TN-Review-main.tex line 1995; author source ImagesReview Section
%   IV/ImagesReview Section IV.tex lines 46-54
% Capabilities: kernel, periodic-closure, boundary-insertion
\tenkzkernel
\[
  |\psi(X)\rangle \;=\;
  \begin{tenkz}[rows={wire,wire}, cols=7, bonds=none]
    \tn[at=(1,2), skin=box, name=a1, ports={0:virtual, 180:virtual, 90:physical}]{A}
    \tn[at=(1,3), skin=box, name=a2, ports={0:virtual, 180:virtual, 90:physical}]{A}
    \tn[at=(1,4), skin=box, name=a3, ports={0:virtual, 180:virtual, 90:physical}]{A}
    \tn[at=(1,5), skin=dots, name=d, ports={0:virtual, 180:virtual}]{}
    \tn[at=(1,6), skin=box, name=a4, ports={0:virtual, 180:virtual, 90:physical}]{A}
    % X is the author's square tensor on the closure rectangle, set below the
    % elision, so the trace reads as tr(X A...A).
    \tn[at=(2,5), skin=box, name=x, ports={0:virtual, 180:virtual}]{X}
    \tn[at=(2,1), skin=none, name=jw, ports={0:virtual, 90:virtual}]{}
    \tn[at=(2,7), skin=none, name=je, ports={90:virtual, 180:virtual}]{}
    \tnwire{a1.0}{a2.180}
    \tnwire{a2.0}{a3.180}
    \tnwire{a3.0}{d.180}
    \tnwire{d.0}{a4.180}
    \tnwire[route=arc]{a4.0}{je.90}
    \tnwire{je.180}{x.0}
    \tnwire{x.180}{jw.0}
    \tnwire[route=arc]{jw.90}{a1.180}
  \end{tenkz}
\]
