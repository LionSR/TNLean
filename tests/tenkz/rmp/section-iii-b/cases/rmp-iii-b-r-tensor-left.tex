  \begin{tenkz}[rows={wire,wire,wire}, cols=4, bonds=none, physical=up]
    % the anyon i and the plain lattice site below it; each site carries its

    \tn[at=(1,2), name=i, role=marked, label pos=nw]{i}
    \tn[at=(3,1), name=v]{}
    % lattice rails: two horizontals and the transversal through both sites
    \tnwire[name=uwest]{1 w of i}{i.w} \tnwire[name=ueast]{i.e}{2 e of i}
    \tnwire{1 w of v}{v.w} \tnwire[name=lre]{v.e}{3 e of v}
    \tnwire[name=du]{i.45}{1 n of 1 e of i}
    \tnwire[name=dm]{i.225}{v.45}
    \tnwire{v.225}{1 s of 1 w of v}

    \tn[at=on lre 0.31, name=bL, role=extra, label pos=s]{b}
    \tn[at=on lre 0.48, name=aa, role=marked, label pos=s]{a}

    \tn[at=on uwest 0.5, name=rW, role=extra]{}
    \tn[at=on du 0.5, name=rN, role=extra]{}
    \tn[at=on ueast 0.25, name=rE, role=extra, label pos=ne]{b}
    % the b string winds once around the anyon, passing over its physical leg
    \tnwire[kind=string, species=operator, name=s,
            via={(3,2), on dm 0.3, on uwest 0.5, 0.25 n of i, on du 0.5,
                 on ueast 0.25, (3,3)},
            cross={over at crossing of s and leg n of (1,2)}]
           {1 s of (3,2)}{1 s of (3,3)}
    % the i string leaves the anyon for the a tensor on the lower rail

            via={0.5 e of (3,2)}]{0.15 se of i}{1 s of 0.5 e of (3,2)}
