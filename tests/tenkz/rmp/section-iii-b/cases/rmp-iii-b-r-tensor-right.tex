% Formula: R_{i,b} replaces the crossings of the i-string with the ordered
%   b,a,b strands.
% Ink: Canonical public kernel plane with an R box, strands, and marked endpoints.
% Boundary: Two horizontal rails and the diagonal transversal are open; three
%   resolved strings continue below the lower rail.
% Source: paper TN-Review-main.tex line 1669; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 773-798
% Capabilities: kernel, plane-frame, braid-resolver, crossing-order
\[
  \tenkzkernel
  \tndeclare{species}{operator}{hue=source:red}
  \begin{tenkz}[
      rows={wire,wire,wire}, cols=4,
      frame=plane, bonds=none]
    % The plane frame carries both rails, the transverse direction, and the
    % resolver contour through one shared affine basis.
    \tn[at=(1,1), name=i, role=marked, label pos=nw]{i}
    \tn[at=(2,3), skin=box, name=R]{R_{i,b}}
    \tn[at=(3,1), name=v]{}
    \tn[at=(3,2), name=bL, role=extra, label pos=sw]{b}
    \tn[at=(3,3), name=a, role=marked, label pos=sw]{a}
    \tn[at=(3,4), name=bR, role=extra, label pos=sw]{b}
    % Two horizontal rails and the oblique transverse rail remain open.
    \tnwire{open w}{i.w} \tnwire{i.e}{open e}
    \tnwire{open w}{v.w} \tnwire{v.e}{bL.w}
    \tnwire{bL.e}{a.w} \tnwire{a.e}{bR.w} \tnwire{bR.e}{open e}
    \tnwire{open n}{i.n} \tnwire{i.s}{v.n} \tnwire{v.s}{open s}
    % The i string enters the resolver, which fans out to b, a, b.
    \tnwire[kind=string, species=operator, name=incoming]
      {i.315}{R.45}
    \tnwire[kind=string, species=operator, name=sb, via={bL},
            cross={under at crossing of sb and incoming}]
      {R.s}{open s}
    \tnwire[kind=string, species=operator, name=sa,
            via={a}, cross={over at crossing of sa and sb,
                            under at crossing of sa and incoming}]
      {R.w}{open s}
    \tnwire[kind=string, species=operator, name=sbr,
            via={bR},
            cross={over at crossing of sbr and sb,
                   over at crossing of sbr and sa,
                   under at crossing of sbr and incoming}]{R.e}{open s}
  \end{tenkz}
\]
