% Target: rmp-iii-b-condensation
% Formula: A doubled anyon string joins paired endpoints on ket and bra sheets.
% Ink: Canonical public kernel plane basis with the same marked path on both sheets.
% Boundary: Both 6x4 sheets are open; paired string endpoints stay uncontracted.
% Source: paper TN-Review-main.tex line 1697; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 925-953
% Capabilities: kernel, plane-frame, double-layer-string, marked-endpoints
\ProvideDocumentCommand{\tncondsite}{O{}mm}{
  \tn[at=(#2,#3,1), name=k#2#3, ports={45:physical}, #1]{}
  \tn[at=(#2,#3,2), name=b#2#3, ports={225:physical}, #1]{}
}
\ProvideDocumentCommand{\tncondbond}{mmmm}{
  \tnwire{(#1,#2,1)}{(#3,#4,1)} \tnwire{(#1,#2,2)}{(#3,#4,2)}
}
\ProvideDocumentCommand{\tncondpair}{mm}{
  \tnwire{k#1#2.45}{b#1#2.225}
}

  \tenkzkernel
  \tndeclare{species}{operator}{hue=source:red}
  % One plane frame owns the ket and bra places.  The paired endpoint ports
  % remain open; every other physical interface is an explicit contraction.
  \begin{tenkz}[lattice={6x4}, planes, bonds=none,
                west=none, east=none, north=none, south=none]
