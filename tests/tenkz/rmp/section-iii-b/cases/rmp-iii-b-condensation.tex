% Target: rmp-iii-b-condensation
% Formula: A doubled anyon string joins paired endpoints on ket and bra sheets.
% Ink: Canonical public tenkzplanes with a marked path on the visible sheet.
% Boundary: Both 4x4 sheets are open; paired string endpoints stay uncontracted.
% Source: paper TN-Review-main.tex line 1697; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 925-953
% Capabilities: lattice, double-layer-string, marked-endpoints
\[
  % cols=5, plane slant=0.40: at rows=6 the stock 0.45 slant trips the
  % library's tall-window guard ((rows-1)*slant > (cols-1)/2); slant=0.40
  % is the readable-band value already used elsewhere in the test suite
  % (fig21d_cubic, sheets_test), and cols=5 gives it enough width to clear
  % the guard without warnings.
  \begin{tenkzlattice}[rows=6, cols=5, frame=oblique, plane slant=0.40]
    % the ket-sheet string: two endpoint sites and three intermediate
    % marked sites. The source (lines 925-953) bends the string once,
    % close to its row-1 endpoint, then runs straight through three
    % small sites down to the row-5 endpoint; row 6 is left plain, as
    % in the source's unused near-boundary row.
    \tnsite[role=marked, size=l]{(1,1)}
    \tnsite[role=marked, size=s]{(2,2)}
    \tnsite[role=marked, size=s]{(3,2)}
    \tnsite[role=marked, size=s]{(4,2)}
    \tnsite[role=marked, size=l]{(5,2)}
    % the bra-sheet string: the same zigzag, shifted two columns over
    \tnsite[role=marked, size=l]{(1,3)}
    \tnsite[role=marked, size=s]{(2,4)}
    \tnsite[role=marked, size=s]{(3,4)}
    \tnsite[role=marked, size=s]{(4,4)}
    \tnsite[role=marked, size=l]{(5,4)}
    \tnedge[role=operator]{(1,1)-(2,2)}
    \tnedge[role=operator]{(2,2)-(3,2)}
    \tnedge[role=operator]{(3,2)-(4,2)}
    \tnedge[role=operator]{(4,2)-(5,2)}
    \tnedge[role=operator]{(1,3)-(2,4)}
    \tnedge[role=operator]{(2,4)-(3,4)}
    \tnedge[role=operator]{(3,4)-(4,4)}
    \tnedge[role=operator]{(4,4)-(5,4)}
  \end{tenkzlattice}_{\ a\leftrightarrow\bar a}
\]
