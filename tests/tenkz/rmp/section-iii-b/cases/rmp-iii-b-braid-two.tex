    \tn[at=(1,1), name=b, role=extra, label pos=nw]{b}

    \tnwire[kind=string, species=operator, name=sb,
            via={0.3 w of b, midway b and j, midway j and i,
                 0.3 s of i, 0.3 e of i, 0.3 e of midway i and a}]{open n}{a.315}
    \tnwire[kind=string, species=operator, name=sa,
            via={midway i and a, midway b and a, 0.3 n of a}]{i.45}{open e}

    \tn[at=midway j and i]{}  \tn[at=0.3 s of i]{}
    \tn[at=0.3 e of i]{}      \tn[at=midway i and a]{}
    \tn[at=midway b and a]{}  \tn[at=0.3 n of a]{}
  \end{tenkz}
