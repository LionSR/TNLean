\[
  % The source's three-dimensional surface is represented by the same
  % homotopy class on a lattice with identified sides.
  \begin{tenkz}[lattice={3x3},
      west=trace, east=trace, north=trace, south=trace]
    \tnwire[kind=string, species=operator, name=torus-cycle, closed,
      wind={1,0}, via={s outside}]
    \tn[skin=box, role=marked, at=on torus-cycle 0.55]{i}

  \;\longrightarrow\;
  \begin{tenkz}[lattice={5x5},
      west=trace, east=trace, north=trace, south=trace]
    % O_g starts above the PEPS window and O_h crosses it vertically.  Their
    % policies also order each red MPO over the black lattice bonds.
    \tnwire[kind=string, species=operator, name=og-left, crossing=over]
      {0.5 w of 0.5 n of (1,1)}{0.5 e of 0.5 n of (1,5)}
    \tnwire[kind=string, species=operator, name=oh-left, crossing=under,
      cross={under at crossing of oh-left and og-left}]
      {0.5 n of (1,4)}{0.5 s of (5,4)}
    \tn[skin=box, role=marked,
        at=crossing of og-left and oh-left]{}
    \tn[skin=dot, role=extra, at=on og-left 0.15]{}
    \tn[skin=dot, role=extra, at=on og-left 0.35]{}
    \tn[skin=dot, role=extra, at=on og-left 0.65]{}

    \tn[skin=dot, role=passive, at=on oh-left 0.65]{}
    \tn[skin=dot, role=passive, at=on oh-left 0.85,
        label pos=s]{O_h}
    \tnwire[dir=to, route=arc]
      {0.8 w of 0.5 n of (1,1)}{0.8 w of (3,1)}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[lattice={5x5},
      west=trace, east=trace, north=trace, south=trace]
    % O_g has moved below the crossing.  O_h stays vertical, and the paired

      {0.5 w of 0.5 n of (3,1)}{0.5 e of 0.5 n of (3,5)}

      via={0.4 w of (2,4), 0.4 e of (2,4)}, crossing=over,
      cross={over at crossing of rails and oh-right}]
      {0.4 w of (1,4)}{0.4 e of (1,4)}
    \tn[skin=dot, role=extra, at=on rails 0.1,
        label pos=nw]{g^{-1}}

    \tn[skin=dot, role=extra, at=on og-right 0.35]{}
    \tn[skin=dot, role=extra, at=on og-right 0.65]{}
