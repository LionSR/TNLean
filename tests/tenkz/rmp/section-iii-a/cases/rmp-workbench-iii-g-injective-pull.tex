      \tnwire[kind=index]{A.0}{(3,5)}
      \tnwire[kind=index]{A.270}{(5,3)}

      \tn[skin=box, at=on g 0.37]{}
      \tn[skin=box, at=on g 0.74]{}

      \tnmark[form=label]{(4,4)}{$g$}
    \end{tenkz}
