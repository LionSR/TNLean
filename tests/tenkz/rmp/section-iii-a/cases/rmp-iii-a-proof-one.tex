% Formula: V_alpha reduces O_alpha acting on A to the original MPS tensor A.
% Ink: Kernel flat frames; V_alpha spans the alpha row and the tensor row on each
%   side of the reduction.
% Boundary: Both sides have the same two virtual and one physical open legs.
% Source: paper TN-Review-main.tex line 1403; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 363-375
% Capabilities: kernel, local-intertwiner, equation-composition
\[
  \tenkzkernel
  \begin{tenkzeq}
    % LHS: O_alpha's tensor over the MPS dot, both fed into V_alpha
    \begin{tenkz}[rows={wire,wire}, cols=2, bonds=none, physical=updown]
      \tn[at=(1,1), skin=box, name=O]{\alpha}
      \tn[at=(2,1), skin=dot, name=A, physical=up]{}
      \tn[at=(1,2), skin=box, wires=2, name=V, physical=none,
        ports={180@1:virtual, 180@2:virtual, 0:virtual}]{V_\alpha}
      \tnwire{open w}{O.180} \tnwire{open w}{A.180}
      \tnwire{O.90}{open n}
      \tnwire{O.270}{A.90}
      \tnwire{O.0}{V.180@1} \tnwire{A.0}{V.180@2}
      \tnwire{V.0}{open e}
    \end{tenkz}
    =
    % RHS: V_alpha's fused output feeds the bare MPS dot
    \begin{tenkz}[rows={wire,wire}, cols=2, bonds=none, physical=up]
      \tn[at=(1,1), skin=box, wires=2, name=V, physical=none,
        ports={180@1:virtual, 180@2:virtual, 0@2:virtual}]{V_\alpha}
      \tn[at=(2,2), skin=dot, name=A]{}
      \tnwire{open w}{V.180@1} \tnwire{open w}{V.180@2}
      \tnwire{V.0@2}{A.180}
      \tnwire{A.90}{open n}
      \tnwire{A.0}{open e}
    \end{tenkz}
  \end{tenkzeq}.
\]
