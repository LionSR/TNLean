% Formula: Four PEPS tensors occupy the edges of one renormalization plaquette.
% Ink: Kernel flat frame; four crossing bonds and four edge-centred glyphless
%   sites with angled physical legs.
% Boundary: Eight external half-edges and four physical legs remain open.
% Source: author source ImagesReview Section III/ImagesReview Section III.tex
%   lines 622-630; archival output PEPSrenorm1.pdf
% Capabilities: kernel, four-site-plaquette, renormalization
\[
  \tenkzkernel
  \begin{tenkz}[rows={wire,wire,wire,wire}, cols=4, bonds=none]
    % the bare plaquette: two horizontal and two vertical bonds crossing at
    % the four corners, the verticals painted over the horizontals
    \tnwire[kind=string, name=h1]{(2,1)}{(2,4)}
    \tnwire[kind=string, name=h2]{(3,1)}{(3,4)}
    \tnwire[kind=string, name=v1,
            cross={over at crossing of v1 and h1,
                   over at crossing of v1 and h2}]{(1,2)}{(4,2)}
    \tnwire[kind=string, name=v2,
            cross={over at crossing of v2 and h1,
                   over at crossing of v2 and h2}]{(1,3)}{(4,3)}
    % the four edge-centred PEPS sites are glyphless; each shows only its
    % 50-degree physical leg, as the source draws
    \tn[skin=none, at=midway (2,2) and (2,3), name=PN, ports={50:physical}]{}
    \tn[skin=none, at=midway (3,2) and (3,3), name=PS, ports={50:physical}]{}
    \tn[skin=none, at=midway (2,2) and (3,2), name=PW, ports={50:physical}]{}
    \tn[skin=none, at=midway (2,3) and (3,3), name=PE, ports={50:physical}]{}
  \end{tenkz}.
\]
