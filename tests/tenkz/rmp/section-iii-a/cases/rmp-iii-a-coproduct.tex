% Ink: Canonical public tenkz equation between one-site and two-site words.

\[
  \Delta\!\left(
  \begin{tenkz}[physical=up, tensor style=box]
    \tnX{X} & \tn{A}
  \end{tenkz}\right)
  =
  \begin{tenkz}[physical=up, tensor style=box]
    \tnX{X} & \tn{A} & \tn{A}
  \end{tenkz}.
\]
