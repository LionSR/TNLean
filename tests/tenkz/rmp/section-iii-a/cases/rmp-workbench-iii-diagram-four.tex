\[
  \begin{tenkzfree}
    \tnput[circle]{l}{(-6mm,6mm)}{\lambda}
    \tnput[box]{n}{(0mm,8mm)}{} \tnput[box]{e}{(8mm,0mm)}{}
    \tnput[box]{s}{(0mm,-8mm)}{} \tnput[box]{w}{(-8mm,0mm)}{}
    \tnjoin[route=arc]{n.east}{e.north}
    \tnjoin[route=arc]{e.south}{s.east}
    \tnjoin[route=arc]{s.west}{w.south}
    \tnjoin[route=arc]{w.north}{l.west}
    \tnjoin{l.east}{n.west}
    \tnjoin{n.north}{0mm,16mm} \tnjoin{e.east}{16mm,0mm}
    \tnjoin{0mm,-16mm}{s.south} \tnjoin{-16mm,0mm}{w.west}
    % source Dia4.pdf: each ring box carries a second, shorter leg on its
    % free port, piercing the ring inward and left open (unconnected) --
    % the north/east/south/west ports not already spent on the outward
    % arm or the two ring joins.
    \tnjoin{n.south}{0mm,4mm} \tnjoin{e.west}{4mm,0mm}
    \tnjoin{s.north}{0mm,-4mm} \tnjoin{w.east}{-4mm,0mm}
  \end{tenkzfree}
  =\Delta^3\!\left(
    \begin{tenkz}[boundary=periodic, physical=up, tensor style=box]
      \tnX{\lambda}&\tn{}
    \end{tenkz}
  \right).
