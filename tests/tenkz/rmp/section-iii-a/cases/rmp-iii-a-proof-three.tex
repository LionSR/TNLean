% Formula: V_beta V_alpha = lambda(alpha,beta;gamma) V_gamma F_alpha,beta^gamma.
% Ink: Canonical public tenkz equation comparing sequential and fused reductions.
% Boundary: Both reduction networks expose the same MPO and MPS interfaces.
% Source: paper TN-Review-main.tex line 1412; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 386-411
% Capabilities: grid, local-intertwiner, fusion-map
% The stacked two-layer fusion: alpha sits below beta on one virtual leg;
% V_alpha reduces the dot with alpha first, then V_beta reduces that
% result with beta.  The fused side stacks alpha, beta the same way, with
% F fusing the pair before V_gamma reduces the dot with the fused leg.
\[
  \begin{tenkz}[rows={op:operator, op:operator, op:operator},
                tensor style=box]
    \tn{\beta} & \tnghost{} & \tn[box,wires=2]{V_\beta}\\
    \tn{\alpha} & \tn[box,wires=2]{V_\alpha} & \\
    \tn[dot]{} & &
  \end{tenkz}
  =\lambda(\alpha,\beta;\gamma)
  \begin{tenkz}[rows={op:operator, op:operator, op:operator},
                tensor style=box]
    \tn{\beta} & \tn[box,wires=2]{F_{\alpha\beta}^{\gamma}} & \\
    \tn{\alpha} & & \tn[box,wires=2]{V_\gamma}\\
    \tn[dot]{} & \tnghost{} &
  \end{tenkz}.
\]
