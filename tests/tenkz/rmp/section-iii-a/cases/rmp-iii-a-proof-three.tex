% Formula: V_beta V_alpha = lambda(alpha,beta;gamma) V_gamma F_alpha,beta^gamma.
% Ink: Kernel flat frames; sequential and fused reductions compared through
%   spanning V and F boxes.
% Boundary: Both reduction networks expose the same MPO and MPS interfaces.
% Source: paper TN-Review-main.tex line 1412; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 386-411
% Capabilities: kernel, local-intertwiner, fusion-map
% The stacked two-layer fusion: alpha sits below beta on one physical column
% with the MPS dot at the bottom.  On the left V_alpha reduces the dot with
% alpha first and V_beta reduces that result with beta; on the right F fuses
% the alpha,beta pair before V_gamma reduces the dot with the fused leg.
\[
  \tenkzkernel
  \begin{tenkzeq}
    \begin{tenkz}[rows={wire,wire,wire}, cols=3, bonds=none]
      \tn[at=(1,1), skin=box, name=B,
        ports={180:virtual, 0:virtual, 90:physical, 270:physical}]{\beta}
      \tn[at=(2,1), skin=box, name=O,
        ports={180:virtual, 0:virtual, 90:physical, 270:physical}]{\alpha}
      \tn[at=(3,1), skin=dot, name=A,
        ports={180:virtual, 0:virtual, 90:physical}]{}
      \tn[at=(2,2), skin=box, wires=2, name=Va,
        ports={180@1:virtual, 180@2:virtual, 0@1:virtual}]{V_\alpha}
      \tn[at=(1,3), skin=box, wires=2, name=Vb,
        ports={180@1:virtual, 180@2:virtual, 0:virtual}]{V_\beta}
      \tnwire{open w}{B.180} \tnwire{open w}{O.180} \tnwire{open w}{A.180}
      \tnwire{B.90}{open n}
      \tnwire{B.270}{O.90} \tnwire{O.270}{A.90}
      \tnwire{B.0}{Vb.180@1}
      \tnwire{O.0}{Va.180@1} \tnwire{A.0}{Va.180@2}
      \tnwire{Va.0@1}{Vb.180@2}
      \tnwire{Vb.0}{open e}
    \end{tenkz}
    =\lambda(\alpha,\beta;\gamma)
    \begin{tenkz}[rows={wire,wire,wire}, cols=3, bonds=none]
      \tn[at=(1,1), skin=box, name=B,
        ports={180:virtual, 0:virtual, 90:physical, 270:physical}]{\beta}
      \tn[at=(2,1), skin=box, name=O,
        ports={180:virtual, 0:virtual, 90:physical, 270:physical}]{\alpha}
      \tn[at=(3,1), skin=dot, name=A,
        ports={180:virtual, 0:virtual, 90:physical}]{}
      \tn[at=(1,2), skin=box, wires=2, name=F,
        ports={180@1:virtual, 180@2:virtual, 0@2:virtual}]{F_{\alpha\beta}^{\gamma}}
      \tn[at=(2,3), skin=box, wires=2, name=Vc,
        ports={180@1:virtual, 180@2:virtual, 0:virtual}]{V_\gamma}
      \tnwire{open w}{B.180} \tnwire{open w}{O.180} \tnwire{open w}{A.180}
      \tnwire{B.90}{open n}
      \tnwire{B.270}{O.90} \tnwire{O.270}{A.90}
      \tnwire{B.0}{F.180@1} \tnwire{O.0}{F.180@2}
      \tnwire{F.0@2}{Vc.180@1}
      % The dot's rail runs below the F box into V_gamma, as the source draws it.
      \tnwire{A.0}{Vc.180@2}
      \tnwire{Vc.0}{open e}
    \end{tenkz}
  \end{tenkzeq}.
\]
