% Target: rmp-iii-a-boundary-algebra-n
% Formula: A^(N) = {Tr[X A_1...A_N] : X} is the size-N MPO algebra.
% Ink: Kernel flat frame; the N-site word with the X ring, closed below by the
%   periodic return in the source's red operator hue.
% Boundary: The virtual word is traced; all N physical pairs are open.
% Source: paper TN-Review-main.tex line 1298; author source ImagesReview Section
%   III/ImagesReview Section III.tex lines 136-143
% Capabilities: kernel, periodic-closure, boundary-insertion
\tenkzkernel
\tndeclare{species}{operator}{hue=source:red}
\[
  \mathcal A^{(N)}=\left\{
  \begin{tenkz}[rows={wire,wire}, cols=7, bonds=none]
    \tn[at=(1,2), skin=ring, name=X, ports={0:virtual, 180:virtual}]{X}
    \tn[at=(1,3), skin=box, name=A1,
      ports={0:virtual, 180:virtual, 90:physical, 270:physical}]{A_1}
    \tn[at=(1,4), skin=box, name=A2,
      ports={0:virtual, 180:virtual, 90:physical, 270:physical}]{A_2}
    \tn[at=(1,5), skin=dots, name=D, ports={0:virtual, 180:virtual}]{}
    \tn[at=(1,6), skin=box, name=AN,
      ports={0:virtual, 180:virtual, 90:physical, 270:physical}]{A_N}
    % The traced virtual word returns below the row, one column wider on each
    % side so the descent clears the physical stubs, as the source draws it.
    \tn[at=(2,1), skin=none, name=JW, ports={0:virtual, 90:virtual}]{}
    \tn[at=(2,7), skin=none, name=JE, ports={90:virtual, 180:virtual}]{}
    \tnwire[species=operator]{X.0}{A1.180}
    \tnwire[species=operator]{A1.0}{A2.180}
    \tnwire[species=operator]{A2.0}{D.180}
    \tnwire[species=operator]{D.0}{AN.180}
    \tnwire[species=operator, route=arc]{AN.0}{JE.90}
    \tnwire[species=operator]{JE.180}{JW.0}
    \tnwire[species=operator, route=arc]{JW.90}{X.180}
  \end{tenkz}:X\right\}.
\]
