% Formula: |G|^-1 sum_g of four L_g actions is the G-injective projector.
% Ink: Kernel flat frame; four inward L_g actions and the red projector ring

\[
  \tenkzkernel
  \tndeclare{species}{op}{hue=source:red}
  \frac1{|G|}\sum_{g\in G}
  \begin{tenkz}[rows={wire,wire,wire}, cols=3, bonds=none]
    % four L_g actions point along the four legs toward the centre; each leg
    % keeps its short open end beside the centre
    \tn[skin=ring, at=(1,2), name=Ln, ports={90:virtual, 270:virtual}]{L_g}
    \tn[skin=ring, at=(2,3), name=Le, ports={0:virtual, 180:virtual}]{L_g}
    \tn[skin=ring, at=(3,2), name=Ls, ports={90:virtual, 270:virtual}]{L_g}
    \tn[skin=ring, at=(2,1), name=Lw, ports={0:virtual, 180:virtual}]{L_g}
    \tnwire[dir=to]{open n}{Ln.90}
    \tnwire[dir=to]{open e}{Le.0}
    \tnwire[dir=to]{open s}{Ls.270}
    \tnwire[dir=to]{open w}{Lw.180}
  \end{tenkz}
  =
  \begin{tenkz}[rows={wire,wire,wire,wire,wire}, cols=5, bonds=none]
    % the projector: the red ring through four tensors, with the marked
    % corner dot on the north-west ring wire
    \tn[skin=box, at=(2,3), name=N,
        ports={0:virtual, 90:virtual, 180:virtual, 270:virtual}]{}
    \tn[skin=box, at=(3,4), name=E,
        ports={0:virtual, 90:virtual, 180:virtual, 270:virtual}]{}
    \tn[skin=box, at=(4,3), name=S,
        ports={0:virtual, 90:virtual, 180:virtual, 270:virtual}]{}
    \tn[skin=box, at=(3,2), name=W,
        ports={0:virtual, 90:virtual, 180:virtual, 270:virtual}]{}
    \tnwire[route=arc, species=op, name=markarc]{W.90}{N.180}
    \tnwire[route=arc, species=op]{N.0}{E.90}
    \tnwire[route=arc, species=op]{E.270}{S.0}
    \tnwire[route=arc, species=op]{S.180}{W.270}
    \tn[skin=dot, at=on markarc 0.5, size=s]{}
  \end{tenkz}.
\]
