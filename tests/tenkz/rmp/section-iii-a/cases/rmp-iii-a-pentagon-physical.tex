% Target: rmp-iii-a-pentagon-physical
% Formula: ^2F ^1F = ^1F ^2F ^2F — the physical string zips through the fusion vertex.
% Ink: Kernel flat frame; a directed fusion vertex and a crossing physical string.
% Boundary: One incoming and two outgoing MPO wires and both string ends are open.
% Source: paper TN-Review-main.tex line 1312; paper figure fig3_pentagon3.pdf;
%   no author drawing source
% Capabilities: kernel, coherence, crossing-order
\[
  \tenkzkernel
  \tndeclare{species}{fus}{hue=source:red}
  \tndeclare{species}{vertex}{hue=source:lightgray}
  \begin{tenkzeq}
    \begin{tenkz}[rows={wire,wire,wire}, cols=3, bonds=none]
      % the physical string stands west of the vertex and crosses the
      % incoming wire once
      \tn[at=(2,2), name=V, skin=dot, species=vertex,
          ports={180:virtual, 45:virtual, 315:virtual}]{}
      \tnwire[species=fus, dir=to, name=cin]{(2,1)}{V.180}
      \tnwire[species=fus, dir=to]{V.45}{(1,3)}
      \tnwire[species=fus, dir=to]{V.315}{(3,3)}
      \tnwire[kind=string, dir=to, name=s,
              cross={over at crossing of s and cin}]
        {midway (3,1) and (3,2)}{midway (1,1) and (1,2)}
    \end{tenkz}
    =
    \begin{tenkz}[rows={wire,wire,wire}, cols=4, bonds=none]
      % the same string east of the vertex crosses both outgoing wires
      \tn[at=(2,2), name=V, skin=dot, species=vertex,
          ports={180:virtual, 45:virtual, 315:virtual}]{}
      \tnwire[species=fus, dir=to, name=cin]{(2,1)}{V.180}
      \tnwire[species=fus, dir=to, name=oa]{V.45}{(1,4)}
      \tnwire[species=fus, dir=to, name=ob]{V.315}{(3,4)}
      \tnwire[kind=string, dir=to, name=s,
              cross={over at crossing of s and oa,
                     over at crossing of s and ob}]
        {midway (3,2) and (3,3)}{midway (1,2) and (1,3)}
    \end{tenkz}
  \end{tenkzeq}
\]

  {}^{2}F\,{}^{1}F \;=\; {}^{1}F\,{}^{2}F\,{}^{2}F
