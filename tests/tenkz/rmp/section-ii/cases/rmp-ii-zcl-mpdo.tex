  % LHS: E . E - two physically-traced tensors sharing one virtual bond

    \tn[skin=mpo, name=M1, ports={0:virtual, 180:virtual}]{M} &
    \tn[skin=mpo, name=M2, ports={0:virtual, 180:virtual}]{M}

  \end{tenkz}

  % RHS: E - one physically-traced tensor
  \begin{tenkz}[rows={wire}, cols=1, physical=updown, trace=physical]

    \tnwire{open w}{M.180} \tnwire{M.0}{open e}
  \end{tenkz}
