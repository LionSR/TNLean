% Formula: eig(rho_n) = eig(rho^R E^n rho^L), up to the cut.
% Ink: Kernel flat frame; rho^R and rho^L are small ring atoms tailing each rail's
%   open end.
% Boundary: rho^R and rho^L lie on opposite rails; four virtual ends remain open around
%   the ladder.
% Source: paper TN-Review-main.tex line 427; author source ImagesReview Section
%   II/ImagesReview Section II.tex lines 500-518
% Capabilities: kernel, typed-ports
\tenkzkernel
\[
  \mathrm{eig}\left(
  % ket rail: rho^R then four A beads; bra rail: four conj(A) beads then
  % rho^L.  Each column's physical index is contracted (the transfer
  % operator E), so only the four virtual rail ends stay open.  rho^R and
  % rho^L are each a small ring atom tailing its own rail -- the shared
  % idiom for the two fixed-point tails, resolving the K3 gap-bar residue.
  \begin{tenkz}[rows={wire,wire}, cols=6, bonds=none]
    \tn[at=(1,1), name=R, skin=ring, ports={180:virtual, 0:virtual}]{\rho^R}
    \tn[at=(1,2), name=a1, ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(1,3), name=a2, ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(1,4), name=a3, ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(1,5), name=a4, ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(2,2), name=b1, conjugate,
      ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(2,3), name=b2, conjugate,
      ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(2,4), name=b3, conjugate,
      ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(2,5), name=b4, conjugate,
      ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(2,6), name=L, skin=ring, ports={180:virtual, 0:virtual}]{\rho^L}
    \tnwire{R.180}{open w} \tnwire{R.0}{a1.180}
    \tnwire{a1.0}{a2.180} \tnwire{a2.0}{a3.180}
    \tnwire{a3.0}{a4.180} \tnwire{a4.0}{open e}
    \tnwire{b1.180}{open w} \tnwire{b1.0}{b2.180}
    \tnwire{b2.0}{b3.180} \tnwire{b3.0}{b4.180}
    \tnwire{b4.0}{L.180} \tnwire{L.0}{open e}
    \tnwire{a1.270}{b1.90} \tnwire{a2.270}{b2.90}
    \tnwire{a3.270}{b3.90} \tnwire{a4.270}{b4.90}
    % Neither form=bracket nor the retired brace-below word draws yet in
    % the executable kernel (K1, tracked in #5482); a south label at the
    % midpoint of the four-bead span keeps the n annotation.
    \tnmark[form=label, label pos=s]{midway b1 and b4}{n}
    % The source marks the cut with a dashed red rectangle around the
    % four contracted columns; the house enclosure mark draws the same
    % region.
    \tnmark[form=enclosure, slot=secondary]{(1,2) .. (2,5)}{}
  \end{tenkz}
  \right)
\]
