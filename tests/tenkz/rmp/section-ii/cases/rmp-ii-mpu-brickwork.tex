% Target: rmp-ii-mpu-brickwork
% Formula: U^(N) is the product of alternating u and v brickwork layers.
% Ink: Kernel flat frame; the u,v gates carry explicit typed ports across two
%   columns each, laid out as a genuine brickwork circuit.
% Boundary: The U word and brickwork keep all six circuit wires open at both ends.
% Source: paper TN-Review-main.tex line 387; author source ImagesReview Section
%   II/ImagesReview Section II.tex lines 450-464
% Capabilities: kernel, typed-ports, equation-composition
\tenkzkernel
\[
  \begin{tenkzeq}
  % LHS: six simple MPU tensors, the bond closing trivially at both
  % outer ends so only the six physical wires stay open (Boundary)
  \begin{tenkz}[rows={wire}, cols=6, physical=updown, west=none, east=none]
    \tn[skin=box]{U} & \tn[skin=box]{U} & \tn[skin=box]{U} &
      \tn[skin=box]{U} & \tn[skin=box]{U} & \tn[skin=box]{U}
  \end{tenkz}
  \;=\;
  % RHS: the brick circuit; six wires run top to bottom, the u layer
  % gates wires (1,2)(3,4)(5,6), the v layer gates wires (2,3)(4,5).
  \begin{tenkz}[rows={wire,wire}, cols=6, bonds=none]
    \tn[at=(1,1), name=u1, skin=box, wide=2,
      ports={90@1:physical,90@2:physical,270@1:physical,270@2:physical}]{u}
    \tn[at=(1,3), name=u2, skin=box, wide=2,
      ports={90@1:physical,90@2:physical,270@1:physical,270@2:physical}]{u}
    \tn[at=(1,5), name=u3, skin=box, wide=2,
      ports={90@1:physical,90@2:physical,270@1:physical,270@2:physical}]{u}
    \tn[at=(2,2), name=v1, skin=box, wide=2,
      ports={90@1:physical,90@2:physical,270@1:physical,270@2:physical}]{v}
    \tn[at=(2,4), name=v2, skin=box, wide=2,
      ports={90@1:physical,90@2:physical,270@1:physical,270@2:physical}]{v}
    \tnwire{open n}{u1.90@1} \tnwire{open n}{u1.90@2}
    \tnwire{open n}{u2.90@1} \tnwire{open n}{u2.90@2}
    \tnwire{open n}{u3.90@1} \tnwire{open n}{u3.90@2}
    \tnwire{u1.270@1}{open s}
    \tnwire{u1.270@2}{v1.90@1} \tnwire{u2.270@1}{v1.90@2}
    \tnwire{u2.270@2}{v2.90@1} \tnwire{u3.270@1}{v2.90@2}
    \tnwire{u3.270@2}{open s}
    \tnwire{v1.270@1}{open s} \tnwire{v1.270@2}{open s}
    \tnwire{v2.270@1}{open s} \tnwire{v2.270@2}{open s}
  \end{tenkz}
  \end{tenkzeq}
\]
