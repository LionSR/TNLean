% Formula: The red and blue interleaved paths form a two-layer MPU.
% Ink: Kernel box skins carry the red and blue shifted port pairings.
% Boundary: Each brick contributes one upper and lower endpoint to each path;
%   four lateral endpoints are also open.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   156-170; archival output newfig7.pdf
% Capabilities: kernel, typed-ports, skin-pairings
\tenkzkernel
\tndeclare{species}{blue}{hue=source:blue}
\tndeclare{species}{red}{hue=source:red}
\tndeclare{skin}{two-shift}{
  base=box,
  pairings={
    n@2 > w@2 : red,
    e@2 > s@2 : red,
    w@1 > s@1 : blue,
    n@1 > e@1 : blue
  }
}
\[
  \begin{tenkz}[rows={wire,wire}, cols=8, bonds=none]
    \tn[at=(1,1), name=c1, skin=two-shift, wide=2, wires=2]{}
    \tn[at=(1,4), name=c2, skin=two-shift, wide=2, wires=2]{}
    \tn[at=(1,7), name=c3, skin=two-shift, wide=2, wires=2]{}
    % The red north-to-west and east-to-south turns belong to the brick skin;
    % the commands below attach the open tails and join neighboring bricks.
    \tnwire[species=red]{open n}{c1.n@2}
    \tnwire[species=red]{open n}{c2.n@2}
    \tnwire[species=red]{open n}{c3.n@2}
    \tnwire[species=red]{c1.s@2}{open s}
    \tnwire[species=red]{c2.s@2}{open s}
    \tnwire[species=red]{c3.s@2}{open s}
    \tnwire[species=red]{open w}{c1.w@2}
    \tnwire[species=red]{c1.e@2}{c2.w@2}
    \tnwire[species=red]{c2.e@2}{c3.w@2}
    \tnwire[species=red]{c3.e@2}{open e}
    % The blue path uses the complementary slot on every face.
    \tnwire[species=blue]{open n}{c1.n@1}
    \tnwire[species=blue]{open n}{c2.n@1}
    \tnwire[species=blue]{open n}{c3.n@1}
    \tnwire[species=blue]{c1.s@1}{open s}
    \tnwire[species=blue]{c2.s@1}{open s}
    \tnwire[species=blue]{c3.s@1}{open s}
    \tnwire[species=blue]{open w}{c1.w@1}
    \tnwire[species=blue]{c1.e@1}{c2.w@1}
    \tnwire[species=blue]{c2.e@1}{c3.w@1}
    \tnwire[species=blue]{c3.e@1}{open e}
  \end{tenkz}
\]
