\tndeclare{species}{right}{hue=source:blue}

\tndeclare{skin}{two-shift}{
  base=box,
  pairings={
    80@1 > 190@1 : left,
    350@1 > 280@1 : left,
    170@1 > 260@1 : right,
    100@1 > 10@1 : right
  }
}
