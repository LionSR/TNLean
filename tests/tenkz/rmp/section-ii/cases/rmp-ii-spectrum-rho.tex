\[
  \mathrm{eig}\left(
  % the rho_n racetrack: open legs, both virtual ends closed through the
  % on-bend fixed points rho^R (west) and rho^L (east)
  \begin{tenkz}[rows={wire,wire}, cols=4, west=cup, east=cup]
    \tn[ports={180:virtual, 0:virtual, 90:physical}]{} &
    \tn[ports={180:virtual, 0:virtual, 90:physical}]{} &
    \tn[ports={180:virtual, 0:virtual, 90:physical}]{} &
    \tn[ports={180:virtual, 0:virtual, 90:physical}]{}\\
    \tn[conjugate, ports={180:virtual, 0:virtual, 270:physical}]{} &
    \tn[conjugate, ports={180:virtual, 0:virtual, 270:physical}]{} &
    \tn[conjugate, ports={180:virtual, 0:virtual, 270:physical}]{} &
    \tn[conjugate, ports={180:virtual, 0:virtual, 270:physical}]{}
    % west={cup=$m$}/east={cup=$m$} bracket sugar is not implemented yet
    % (K1); the bare `cup` word plus a hand-placed bead on the canonical
    % cup wire reproduces the same closure and the same on-bend label.
    \tn[skin=ring, at=on cup-west-1-2 0.5]{\rho^R}
    \tn[skin=ring, at=on cup-east-1-2 0.5]{\rho^L}
  \end{tenkz}
  \right)
\]
