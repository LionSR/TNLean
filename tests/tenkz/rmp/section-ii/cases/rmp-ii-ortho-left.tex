    \tn[at=(1,1), skin=tri, name=ket,
      ports={180:virtual, 0:virtual, 270:physical}]{A_L}

  \end{tenkz}

    \tn[at=(1,1), skin=none, name=r1, ports={180:virtual, 0:virtual}]{}
