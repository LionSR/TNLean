% Formula: The archival tangent projector is a difference of two canonical sandwich
%   sums.
% Ink: Canonical public tenkz environment; no raw TikZ.
% Boundary: The first summand seals outer virtual sides; the second closes only its
%   east pair.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   208-236; archival output projectoronPTA.pdf
% Capabilities: grid, typed-ports
\[
  P_{T(A)} \;=\; \sum_{i=1}^{n}
  \begin{tenkz}[rows={ket:nopair,bra},boundary=none]
    \tn[tri=l]{} & \tn[tri=l]{} & \tnX[wires=2]{} &
      \tn[tri=r]{} & \tn[tri=r]{}\\
    \tn*[tri=l]{} & \tn*[tri=l]{} & &
      \tn*[tri=r]{} & \tn*[tri=r]{}
  \end{tenkz}
  \;-\;\sum_{i=1}^{n}
  \begin{tenkz}[rows={ket:nopair,bra},west=none,east=cup]
    \tn[tri=l]{} & \tn[tri=l]{}\\
    \tn*[tri=l]{} & \tn*[tri=l]{}
  \end{tenkz}
\]
