% Formula: psi_ijk = sum_{alpha,beta,gamma} A^i_{gamma alpha} B^j_{alpha beta}
%   C^k_{beta gamma}.
% Ink: Canonical public kernel circle frame; no raw TikZ.
% Boundary: The three physical spokes are open; alpha, beta, and gamma form the
%   contracted triangle.
% Source: paper TN-Review-main.tex line 322; author source ImagesReview Section
%   II/ImagesReview Section II.tex lines 258-268
% Capabilities: kernel, circle-frame, typed-ports
\[
  \psi_{ijk} \;=\;
  \tenkzkernel
  \begin{tenkz}[rows={wire}, cols=3, frame=circle, bonds=none]
    % The circle frame gives the three sites their equilateral placement and
    % transports each physical port along the station's outward radius.
    \tn[at=(1,1), name=t,
      ports={0:virtual,180:virtual,90:physical:$i$}]{}
    \tn[at=(1,2), name=r,
      ports={0:virtual,180:virtual,90:physical:$j$}]{}
    \tn[at=(1,3), name=l,
      ports={0:virtual,180:virtual,90:physical:$k$}]{}
    \tnwire[name=alpha]{t.0}{r.180}
    \tnwire[name=beta]{r.0}{l.180}
    \tnwire[name=gamma]{l.0}{t.180}
    \tnmark[form=label, label pos=e]{on alpha 0.5}{$\alpha$}
    \tnmark[form=label, label pos=s]{on beta 0.5}{$\beta$}
    \tnmark[form=label, label pos=w]{on gamma 0.5}{$\gamma$}
  \end{tenkz}
\]
