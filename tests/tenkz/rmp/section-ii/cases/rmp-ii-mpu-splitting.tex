% Boundary: The U atom has four open legs; each factorization contracts only the

%   II/ImagesReview Section II.tex lines 433-449
% Capabilities: kernel, typed-ports
\[
  \tenkzkernel
  \begin{tenkz}[rows={wire}, cols=1, bonds=none]
    \tn[at=(1,1), skin=box,
      ports={90:physical,270:physical,0:virtual,180:virtual}]{U}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={wire,wire}, cols=1, bonds=none]
    \tn[at=(1,1), skin=box, role=marked, name=u,
      ports={90:physical,270:virtual,0:virtual}]{u}
    \tn[at=(2,1), skin=box, role=extra, name=v,
      ports={90:virtual,270:physical,180:virtual}]{v}
    \tnwire{u.270}{v.90}
  \end{tenkz}
  \;=\;
  % The final factorization mirrors the free side of each coloured bead.
  \begin{tenkz}[rows={wire,wire}, cols=1, bonds=none]
    \tn[at=(1,1), role=marked, label pos=e, name=u,
      ports={90:physical,270:virtual,180:virtual}]{u}
    \tn[at=(2,1), role=extra, label pos=w, name=v,
      ports={90:virtual,270:physical,0:virtual}]{v}
    \tnwire{u.270}{v.90}
  \end{tenkz}
\]
