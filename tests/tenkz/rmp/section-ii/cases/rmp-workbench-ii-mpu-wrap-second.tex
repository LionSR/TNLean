% Formula: Two stacked MPU rows implement opposite local shift orientations.
% Ink: Kernel box skins carry the two opposite local port pairings.
% Boundary: Each row has four upper and four lower open legs and two open
%   virtual endpoints; the turn orientation reverses between rows.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   473-483; archival output sec2fig7c (b).pdf
% Capabilities: kernel, typed-ports, skin-pairings
\tenkzkernel
\tndeclare{species}{shift}{hue=source:black}
\tndeclare{skin}{shift-right}{
  base=box,
  pairings={n@1 > e@1 : shift, s@1 > w@1 : shift}
}
\tndeclare{skin}{shift-left}{
  base=box,
  pairings={n@1 > w@1 : shift, s@1 > e@1 : shift}
}
\[
  \begin{tenkz}[rows={wire,wire}, cols=4, bonds=none, physical=updown]
    % The upper row turns left; the lower row turns right, exactly as the two
    % local drawings at source lines 474-481.
    \tn[at=(1,1), skin=shift-left]{} \tn[at=(1,2), skin=shift-left]{}
    \tn[at=(1,3), skin=shift-left]{} \tn[at=(1,4), skin=shift-left]{}
    \tn[at=(2,1), skin=shift-right]{} \tn[at=(2,2), skin=shift-right]{}
    \tn[at=(2,3), skin=shift-right]{} \tn[at=(2,4), skin=shift-right]{}
    \tnwire{open w}{(1,1)} \tnwire{(1,1)}{(1,2)}
    \tnwire{(1,2)}{(1,3)} \tnwire{(1,3)}{(1,4)}
    \tnwire{(1,4)}{open e}
    \tnwire{open w}{(2,1)} \tnwire{(2,1)}{(2,2)}
    \tnwire{(2,2)}{(2,3)} \tnwire{(2,3)}{(2,4)}
    \tnwire{(2,4)}{open e}
  \end{tenkz}
\]
