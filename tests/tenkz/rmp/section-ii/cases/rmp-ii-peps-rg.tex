% Formula: A renormalized 2x2 PEPS plaquette decomposes into the four source panels
%   (a)-(d).
% Ink: Kernel plane and flat frames; the final bead closes its virtual ring
%   through invisible junctions.
% Boundary: Each PEPS factor keeps its exterior legs open; the final bead closes only
%   its virtual ring.
% Source: paper TN-Review-main.tex line 1025; author source ImagesReview Section
%   II/ImagesReview Section II.tex lines 742-800
% Capabilities: kernel, plane-frame, typed-ports
\tenkzkernel
(a)
\[
  \begin{tenkz}[lattice={2x2}, frame=plane, physical=up,
                west=open, east=open, north=open, south=open]
    \tn{} & \tn{} \\
    \tn{} & \tn{}
  \end{tenkz}
\]

(b)
\[
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up,
                west=open, east=open, north=open, south=open]
    \tn{}
  \end{tenkz}
  \;\otimes\;
  % three disentangled product tensors at half the site size, as the source
  % draws them: up+south, west+up+north, up+east
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=none, east=none, north=none, south=open]
    \tn[size=s]{}
  \end{tenkz}
  \;
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=open, east=none, north=open, south=none]
    \tn[size=s]{}
  \end{tenkz}
  \;
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=none, east=open, north=none, south=none]
    \tn[size=s]{}
  \end{tenkz}
\]

(c)
\[
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up,
                west=open, east=open, north=open, south=open]
    \tn{}
  \end{tenkz}
  \;\otimes\;
  % the generating-MPS factors: west+up+east, up+south, up+north
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=open, east=open, north=none, south=none]
    \tn[size=s]{}
  \end{tenkz}
  \;
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=none, east=none, north=none, south=open]
    \tn[size=s]{}
  \end{tenkz}
  \;
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up, size=s,
                west=none, east=none, north=open, south=none]
    \tn[size=s]{}
  \end{tenkz}
\]

(d)
\[
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up,
                west=open, east=open, north=open, south=open]
    \tn{}
  \end{tenkz}
  \;\otimes\;
  \begin{tenkz}[lattice={1x1}, frame=plane, physical=up,
                west=open, east=open, north=open, south=open]
    \tn{}
  \end{tenkz}
  \;\otimes\;
  % the bead whose virtual wire closes on itself: the ring is drawn through
  % invisible junctions, the house form for the source's identified upward
  % hooks, and carries no physical leg, as the source draws it
  \begin{tenkz}[rows={wire,wire}, cols=2, bonds=none, size=s]
    \tn[at=(2,1), skin=none, name=JW, ports={0:virtual, 90:virtual}]{}
    \tn[at=(2,2), skin=none, name=JE, ports={180:virtual, 90:virtual}]{}
    \tn[at=(1,1), skin=none, name=KW, ports={0:virtual, 270:virtual}]{}
    \tn[at=(1,2), skin=none, name=KE, ports={180:virtual, 270:virtual}]{}
    \tnwire[name=run]{JW.0}{JE.180}
    \tn[at=on run 0.5, skin=dot, size=s]{}
    \tnwire{JW.90}{KW.270}
    \tnwire{JE.90}{KE.270}
    \tnwire{KW.0}{KE.180}
  \end{tenkz}
\]
