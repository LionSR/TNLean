\tenkzkernel
\[
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, west=none]
    \tn[name=A, ports={0:virtual, 90:physical}]{A^I}
    \tnwire{A.0}{open e}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={wire}, cols=2, bonds=none, west=none]
    \tn[skin=box, name=U, ports={0:virtual, 90:physical}]{U^A_I} &
    \tn[skin=ring, name=sig, ports={180:virtual, 0:virtual}]{\sigma^A}
    \tnwire{U.0}{sig.180} \tnwire{sig.0}{open e}
  \end{tenkz}
  \qquad
  \begin{tenkz}[rows={wire}, cols=1, bonds=none, east=none]
    \tn[name=B, ports={180:virtual, 90:physical}]{B^{\overline{I}}}
    \tnwire{open w}{B.180}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={wire}, cols=2, bonds=none, east=none]
    \tn[skin=ring, name=sig, ports={180:virtual, 0:virtual}]{\sigma^B} &
    \tn[skin=box, name=U, ports={180:virtual, 90:physical}]{U^B_{\overline{I}}}
    \tnwire{open w}{sig.180} \tnwire{sig.0}{U.180}
  \end{tenkz}
\]
