% Target: rmp-ii-sv17
% Formula: A many-body state is approximated by a blocked MPS, whose k-site marginal is
%   rho_k.
% Ink: Kernel flat frame; two wide state atoms and a ket-bra racetrack closed by
%   unlabelled cups.
% Boundary: The state panels expose physical legs; rho_k keeps every ket leg open
%   upward and every bra leg open downward and closes both virtual ends.
% Source: paper TN-Review-main.tex line 621; paper-context-only; no author drawing
%   source
% Capabilities: kernel, typed-ports
\tenkzkernel
\[
  % |Phi>: one 9-leg state tensor
  \underset{\lvert\Phi\rangle}{
  \begin{tenkz}[cols=9, bonds=none, west=none, east=none]
    \tn[at=(1,1), skin=pill, wide=9,
      ports={90@1:physical, 90@2:physical, 90@3:physical, 90@4:physical,
             90@5:physical, 90@6:physical, 90@7:physical, 90@8:physical,
             90@9:physical}]{}
  \end{tenkz}}
  \;\longrightarrow\;
  % |phi_D>: blocked MPS of bond dimension D, middle elided
  \underset{\lvert\phi_D\rangle}{
  \begin{tenkz}[cols=9, bonds=none, west=none, east=none]
    \tn[at=(1,1), skin=pill, wide=4,
      ports={90@1:physical, 90@2:physical, 90@3:physical, 90@4:physical}]{}
    \tn[at=(1,5), skin=dots]{}
    \tn[at=(1,6), skin=pill, wide=4,
      ports={90@1:physical, 90@2:physical, 90@3:physical, 90@4:physical}]{}
  \end{tenkz}}
  \;\longrightarrow\;
  % rho_k: the k-site marginal is an operator, so every ket leg stays open
  % upward and every bra leg open downward (the source racetrack has no
  % ket-bra rungs); only the virtual ends close on the environment beads,
  % which the source leaves nameless, so each cup carries an unnamed matrix.
  \underset{\rho_k}{
  \begin{tenkz}[rows={wire,wire}, cols=3, bonds=none,
                west={cup=}, east={cup=}]
    \tn[at=(1,1), name=a1, ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(1,2), name=a2, ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(1,3), name=a3, ports={180:virtual, 0:virtual, 90:physical}]{}
    \tn[at=(2,1), name=b1, conjugate,
      ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(2,2), name=b2, conjugate,
      ports={180:virtual, 0:virtual, 270:physical}]{}
    \tn[at=(2,3), name=b3, conjugate,
      ports={180:virtual, 0:virtual, 270:physical}]{}
    \tnwire{(1,1)}{(1,2)} \tnwire{(1,2)}{(1,3)}
    \tnwire{(2,1)}{(2,2)} \tnwire{(2,2)}{(2,3)}
  \end{tenkz}}
\]
