% Formula: One marked virtual input on A is carried to three marked outputs on B.
% Ink: Kernel plane frame with one projected PEPS star and marked gauge graph.
% Boundary: Both PEPS atoms keep four virtual and one physical leg open; the marked
%   branch has one incoming and three outgoing virtual endpoints.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   358-373; archival output sec2fig7old.pdf
% Capabilities: kernel, plane-frame, typed-ports, open-string
\[
  \tenkzkernel
  \tndeclare{species}{gauge}{hue=source:blue}
  \underset{A}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane]
    \tn[at=(3,3), name=A,
      ports={0:virtual,90:virtual,180:virtual,270:virtual,45:physical}]{}
    \tnwire{A.180}{(3,1)}
    \tnwire{A.0}{(3,5)}
    \tnwire{A.90}{(1,3)}
    \tnwire{A.270}{(5,3)}
    \tnwire{A.45}{midway A and (2,4)}
    \tnwire[kind=string, species=gauge, name=gA]{(1,2)}{(5,2)}
    \tn[at=on gA 0.5, skin=dot, species=gauge]{}
  \end{tenkz}}
  \;=\;
  \underset{B}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane]
    \tn[at=(3,3), name=B,
      ports={0:virtual,90:virtual,180:virtual,270:virtual,45:physical}]{}
    \tnwire{B.180}{(3,1)}
    \tnwire{B.0}{(3,5)}
    \tnwire{B.90}{(1,3)}
    \tnwire{B.270}{(5,3)}
    \tnwire{B.45}{midway B and (2,4)}
    \tn[at=(4,3), name=gS, skin=dot, species=gauge]{}
    \tn[at=(3,4), name=gE, skin=dot, species=gauge]{}
    \tn[at=(2,3), name=gN, skin=dot, species=gauge]{}
    \tnwire[kind=string, species=gauge, name=gInput,
      crossing=over, route=orth, via={(4,1)}]
      {(5,1)}{gS}
    \tnwire[kind=string, species=gauge, name=gEast,
      route=orth, via={(4,4),(3,4)}]
      {gS}{(3,5)}
    \tnwire[kind=string, species=gauge, name=gNorth, route=arc,
      cross={over at crossing of gNorth and gInput,
        over at crossing of gNorth and gEast}]{gS}{gN}
    \tnwire[kind=string, species=gauge, name=gSouth,
      cross={under at crossing of gInput and gSouth,
        under at crossing of gEast and gSouth,
        under at crossing of gNorth and gSouth}]{gS}{(5,3)}
    \tnwire[kind=string, species=gauge]{gN}{(1,3)}
  \end{tenkz}}
\]
