% Formula: A marked route through A becomes a route meeting three virtual legs of B.
% Ink: Kernel plane frame with one projected PEPS star and marked gauge graph.
% Boundary: Both PEPS atoms keep four virtual and one physical leg open; the marked
%   route enters once and meets three distinct virtual-leg markers on B.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   358-373; archival output sec2fig7old.pdf
% Capabilities: kernel, plane-frame, typed-ports, open-string
\[
  \tenkzkernel
  \tndeclare{species}{gauge}{hue=source:blue}
  \underset{A}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane,
      physical=up]
    \tn[at=(3,3), name=A,
      ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
    \tnwire{A.180}{open w}
    \tnwire{A.0}{open e}
    \tnwire{A.90}{open n}
    \tnwire{A.270}{open s}
    \tn[at=(3,2), skin=none, physical=none, name=jgA1]{}
    \tnwire[kind=string, species=gauge, name=gA]{open n}{jgA1}
    \tnwire[kind=string, species=gauge, name=gA-end]{jgA1}{open s}
    \tn[at=on gA 1, skin=dot, species=gauge]{}
  \end{tenkz}}
  \;=\;
  \underset{B}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane,
      physical=up]
    \tn[at=(3,3), name=B,
      ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
    \tnwire{B.180}{open w}
    \tnwire{B.0}{open e}
    \tnwire{B.90}{open n}
    \tnwire{B.270}{open s}
    \tn[at=(4,3), name=gS, skin=dot, species=gauge]{}
    \tn[at=(3,4), name=gE, skin=dot, species=gauge]{}
    \tn[at=(2,3), name=gN, skin=dot, species=gauge]{}
    \tn[at=(4,1), skin=none, physical=none, name=jgInput1]{}
    \tn[at=(4,4), skin=none, physical=none, name=jgEast1]{}
    \tn[at=(2,4), skin=none, physical=none, name=jgNorth1]{}
    \tnwire[kind=string, species=gauge, name=gInput,
      crossing=over, route=orth]{open s}{jgInput1}
    \tnwire[kind=string, species=gauge, name=gInput-end,
      crossing=over, route=orth,
      cross={over at crossing of gInput-end and gInput}]{jgInput1}{gS}
    \tnwire[kind=string, species=gauge, name=gEast,
      crossing=over, route=orth]{gS}{jgEast1}
    \tnwire[kind=string, species=gauge, name=gEast-end,
      crossing=over, route=orth,
      cross={over at crossing of gEast-end and gEast}]{jgEast1}{gE}
    \tnwire[kind=string, species=gauge, name=gNorth,
      crossing=over, route=orth]{gE}{jgNorth1}
    \tnwire[kind=string, species=gauge, name=gNorth-end,
      crossing=over, route=orth,
      cross={over at crossing of gNorth-end and gNorth}]{jgNorth1}{gN}
    \tnwire[kind=string, species=gauge, name=gOutput,
      cross={over at crossing of gOutput and gNorth-end}]{gN}{open n}
  \end{tenkz}}
\]
