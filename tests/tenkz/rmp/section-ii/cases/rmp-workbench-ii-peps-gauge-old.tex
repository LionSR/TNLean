% Formula: A marked route through A becomes a route meeting three virtual legs of B.
% Ink: Kernel plane frame with one projected PEPS star and marked gauge graph.
% Boundary: Both PEPS atoms keep four virtual and one physical leg open; the marked
%   route enters once and meets three distinct virtual-leg markers on B.
% Source: author source ImagesReview Section II/ImagesReview Section II.tex lines
%   358-373; archival output sec2fig7old.pdf
% Capabilities: kernel, plane-frame, typed-ports, open-string
\[
  \tenkzkernel
  \tndeclare{species}{gauge}{hue=source:blue}
  \underset{A}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane,
      physical=up]
    \tn[at=(3,3), name=A,
      ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
    \tnwire{A.180}{open w}
    \tnwire{A.0}{open e}
    \tnwire{A.90}{open n}
    \tnwire{A.270}{open s}
    \tnwire[kind=string, species=gauge, name=gA, via={(3,2)}]{open n}{open s}
    \tn[at=on gA 0.5, skin=dot, species=gauge]{}
  \end{tenkz}}
  \;=\;
  \underset{B}{\begin{tenkz}[
      rows={wire,wire,wire,wire,wire}, cols=5, bonds=none, frame=plane,
      physical=up]
    \tn[at=(3,3), name=B,
      ports={0:virtual,90:virtual,180:virtual,270:virtual}]{}
    \tnwire{B.180}{open w}
    \tnwire{B.0}{open e}
    \tnwire{B.90}{open n}
    \tnwire{B.270}{open s}
    \tn[at=(4,3), name=gS, skin=dot, species=gauge]{}
    \tn[at=(3,4), name=gE, skin=dot, species=gauge]{}
    \tn[at=(2,3), name=gN, skin=dot, species=gauge]{}
    \tnwire[kind=string, species=gauge, name=gInput,
      crossing=over, route=orth, via={(4,1)}]{open s}{gS}
    \tnwire[kind=string, species=gauge, name=gEast,
      crossing=over, route=orth, via={(4,4)}]{gS}{gE}
    \tnwire[kind=string, species=gauge, name=gNorth,
      crossing=over, route=orth, via={(2,4)}]{gE}{gN}
    \tnwire[kind=string, species=gauge, name=gOutput,
      cross={over at crossing of gOutput and gNorth}]{gN}{open n}
  \end{tenkz}}
\]
