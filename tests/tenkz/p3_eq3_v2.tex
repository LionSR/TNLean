% P3 v2 — arXiv:2203.12563 eq (3): fusion-tensor decomposition of the MPO
% product,
%   sum_m T_a^{l,m} T_b^{m,p}
%     = sum_{c,mu} What_{ab}^{c,mu} . T_c^{l,p} . W_{ab}^{c,mu} .
% Paper ink: two red MPO wires (blocks a over b) carrying stacked T dots
% with ONE physical line through both; on the right a chirality pair of
% fat bars (What: pair legs open west; W: pair legs open east) around a
% T_c dot on the fused wire, multiplicity mu atop each bar.
%
% tenkz spelling.  LHS: op-over-op rows pair the facing physical legs
% automatically; `:operator` gives the MPO wire hue.  RHS: each fat bar is
% \tnfuse in its supported orientation (pair wires toward a ghost column =
% the open pair; combined leg toward T_c); the bar pictures and the T_c
% picture are SEPARATE math atoms sharing the math axis — the zipper
% precedent from the manual: a fused wire continuing into a one-row word
% needs the tracked splitting glyph, so the seam between combined stub and
% T_c's west stub is juxtaposition, not one drawn wire.
% Label mapping: block labels ride the atoms (T_a, T_b, T_c) instead of
% the paper's wire labels a, b, c — bond labels are row-1-only and the
% fused stub has no label slot; mu is the \tnfuse bar label.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}

\[
  \begin{tenkz}[rows={op:operator, op:operator}]
    \tn{T_a} \\
    \tn{T_b}
  \end{tenkz}
  \;=\; \sum_{c,\mu}\;
  \begin{tenkz}[rows={wire:operator, wire:operator}]
    \tnghost{} & \tnfuse[combined=east]{\mu} \\
    \tnghost{} &
  \end{tenkz}
  \begin{tenkz}[rows={op:operator}]
    \tn{T_c}
  \end{tenkz}
  \begin{tenkz}[rows={wire:operator, wire:operator}]
    \tnfuse[combined=west]{\mu} & \tnghost{} \\
    & \tnghost{}
  \end{tenkz}
\]

\end{document}
