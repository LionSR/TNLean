% Restyle of v2: the paper draws the gauge X as a TALL ROUNDED CAPSULE

% Formula:  O = X^{-1} [ sum_k A^k (x) conj(A^k) ] X,
% where A is the 4-leg purification tensor (physical s up, purification
% index k down onto conj(A), virtual a1/b1) and X is the gauge mapping
% the doubled virtual index (a1,a2) to O's single virtual index a.
%
% Declarative reading:
%   rows={op:none, op} — A over conj(A); the automatic op-op pairing is
%   the purification index k; s (up) and s' (down) stay open.
%   \tnX[wires=2]{X^{-1}} / {X} — ONE tall on-wire capsule spanning
%   both layer wires and terminating them at its edges (A2 idiom).
%   `op:none` seals the TOP row's boundary, so exactly ONE virtual
%   index leaves west of X^{-1} and one east of X: the signature reads
%   (1,1,1,1) on both sides, matching O = (1,1,1,1).
% Known styling gap: the paper's single outer wire enters the capsule
% at its vertical CENTRE (it is the fused index a); the grid can only
% open ONE ROW's boundary, so the outer wire sits at the lower row's
% height.  A fused/combined external side for wires=k capsules
% (\tnfuse-style boundary consumption, e.g. `combined=west`) would
% close this.

\begin{document}
\[
  % LHS: the 4-leg MPO tensor O (physical s up, s' down, virtual a/b)
  \begin{tenkz}[physical=updown, tensor style=box]
    \tn{O}
  \end{tenkz}
  \;=\;
  % RHS: X^{-1} [ A (x) conj(A) ] X — capsules span both layers; the
  % A–conj(A) bond is the purification index k.  align=2 puts the open
  % row on the math axis, so the outer virtual wire runs collinear
  % with O's across the equation.
  \begin{tenkz}[rows={op:none, op}, align=2, tensor style=box]
    \tnX[wires=2]{X^{-1}} & \tn{A} & \tnX[wires=2]{X}\\
                          & \tn*{A} &
  \end{tenkz}
\]
