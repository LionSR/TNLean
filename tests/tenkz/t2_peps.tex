% Regression: t2_peps — renders the trivalent physical/fusion-wire junction around a PEPS tensor with three…
% Target: RMP 2011.12127 fig3_peps.pdf (the PEPS tensor as a fat graph).
% Formula: the trivalent junction of physical/fusion wires around the
% PEPS tensor —  gamma: m -> k  (vertical, up), then k splits into
% alpha: k -> j (up-west) and beta: k -> n (up-east); fat-strip faces
% A (lower west), B (lower east), C (top); gray wedges at j, k, m, n.
% Genuinely non-grid (trivalent junction): tenkzfree.
% Honest gaps: no fat-strip face outlines or free-tier annotation labels
% (faces A, B, C absent); join labels sit fixed north of the chord
% midpoint, so gamma collides with its vertical wire; no gray-wedge
% terminal skin (house beads instead).
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}

\[
  \begin{tenkzfree}
    \tnput[dot, label pos=south, ports={north:physical}]{m}{(0,-15mm)}{m}
    \tnput[dot, label pos=east,  ports={south:physical}]{k}{(0,0)}{k}
    \tnput[dot, label pos=north]{j}{(-13mm,13mm)}{j}
    \tnput[dot, label pos=north]{n}{(13mm,13mm)}{n}
    % gamma: m -> k, then the split alpha: k -> j and beta: k -> n
    \tnjoin[dir, label=\gamma]{m.north}{k.south}
    \tnjoin[dir, label=\alpha]{k.north west}{j.south east}
    \tnjoin[dir, label=\beta]{k.north east}{n.south west}
  \end{tenkzfree}
\]

\end{document}
