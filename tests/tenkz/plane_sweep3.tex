% Regression: plane_sweep3 — renders rotated chains, a vertical transfer word, and a turning boundary MPO.
% Formula: geometry-only; no tensor identity.
% plane_sweep3 — ROTATED chains and vertical words (the hard05 gap).
% The grid tier (tenkz) lays chains horizontally only; the RMP
% (arXiv:2011.12127) draws O_L(M) as a VERTICAL ring (Fig:MPDO-O_L),
% draws the transfer matrix E as a rotated column in the PEPS
% sections, and boundary MPOs turn corners.  This sweep mocks all
% three in the sanctioned free tier (\tnput/\tnjoin) to find out
% (a) whether current glyph/wire styles carry over to non-horizontal
% chains, and (b) what minimal grid key would make each declarative.
%
% NO file under tex/tenkz/ is touched.  Metric discipline: every
% distance is a multiple of the 11mm pitch —
%   bond 11mm | layersep 8.8mm (0.80) | stub 5mm (0.45)
%   physleg 4.2mm (0.38) | labelclear 1.3mm (0.12)
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}

\textbf{(1a) vertical 3-site MPO word, paper-style truncated side hooks}
(RMP Fig:MPDO-O\(_L\) orientation: virtual bonds vertical, physical
legs west/east, ring closure wraps around the right side):

\medskip
\begin{tenkzfree}[tensor style=box]
  \tnput{m1}{(0,0)}{M}
  \tnput{m2}{(0,-11mm)}{M}
  \tnput{m3}{(0,-22mm)}{M}
  % vertical virtual bonds (the chain axis, rotated)
  \tnjoin{m1.south}{m2.north}
  \tnjoin{m2.south}{m3.north}
  % physical legs, west/east on every site (0.38 pitch)
  \tnjoin{m1.west}{$(m1.west)+(-4.2mm,0)$}
  \tnjoin{m1.east}{$(m1.east)+(4.2mm,0)$}
  \tnjoin{m2.west}{$(m2.west)+(-4.2mm,0)$}
  \tnjoin{m2.east}{$(m2.east)+(4.2mm,0)$}
  \tnjoin{m3.west}{$(m3.west)+(-4.2mm,0)$}
  \tnjoin{m3.east}{$(m3.east)+(4.2mm,0)$}
  % ring closure, paper style: two truncated half-hooks that wrap
  % over the top / under the bottom toward the right side; the tail
  % lands right of the box's east edge, ABOVE the east leg
  \tnjoin[route=curve, out=90, in=90]{m1.north}{$(m1.north)+(5mm,-0.6mm)$}
  \tnjoin[route=curve, out=-90, in=-90]{m3.south}{$(m3.south)+(5mm,0.6mm)$}
\end{tenkzfree}

  \tnjoin{m2.west}{$(m2.west)+(-4.2mm,0)$}
  \tnjoin{m2.east}{$(m2.east)+(4.2mm,0)$}
  \tnjoin{m3.west}{$(m3.west)+(-4.2mm,0)$}
  \tnjoin{m3.east}{$(m3.east)+(4.2mm,0)$}
  % racetrack: up-and-over, down the right at x = 9mm (0.82 pitch,

  \tnjoin[route=vh]{m1.north}{9mm,5mm}
  \tnjoin{9mm,5mm}{9mm,-27mm}
  \tnjoin[route=vh]{m3.south}{9mm,-27mm}
\end{tenkzfree}

\bigskip
\textbf{(2a) transfer matrix \(E\), standard orientation, SANCTIONED
labels} (ket dot over bra dot; the free tier pins every dot label to
the south band, so the label of \(A\) lands ON the physical wire —
collision probe, expected to fail):

\medskip
\begin{tenkzfree}
  \tnput{a}{(0,0)}{A}
  \tnput{ab}{(0,-8.8mm)}{\bar A}
  % physical bond between ket and bra layers
  \tnjoin{a.south}{ab.north}
  % virtual legs, west/east on both layers (0.45 pitch stubs)
  \tnjoin{a.west}{$(a.west)+(-5mm,0)$}
  \tnjoin{a.east}{$(a.east)+(5mm,0)$}
  \tnjoin{ab.west}{$(ab.west)+(-5mm,0)$}
  \tnjoin{ab.east}{$(ab.east)+(5mm,0)$}
\end{tenkzfree}

\bigskip
\textbf{(2b) transfer matrix \(E\) as a 90\(^\circ\)-rotated column}
(two dots side by side, physical bond horizontal, virtual legs
vertical; labels placed with RAW nodes — the unsanctioned workaround
for the south-band collision):

\medskip
\begin{tenkzfree}
  \tnput{a}{(0,0)}{}
  \tnput{ab}{(8.8mm,0)}{}
  \tnjoin{a.east}{ab.west}
  % virtual legs now point north/south
  \tnjoin{a.north}{$(a.north)+(0,5mm)$}
  \tnjoin{a.south}{$(a.south)+(0,-5mm)$}
  \tnjoin{ab.north}{$(ab.north)+(0,5mm)$}
  \tnjoin{ab.south}{$(ab.south)+(0,-5mm)$}
  % workaround: west/east label bands (NOT provided by \tnput)
  % tenkz-lint: allow raw-ink deliberate exhibit of the unsanctioned workaround

  % tenkz-lint: allow raw-ink deliberate exhibit of the unsanctioned workaround

\end{tenkzfree}

\bigskip
\textbf{(3a) L-shaped boundary-MPO chain, bond turns the corner}
(PEPS boundary MPO: chain runs east then north, physical legs point
into the enclosed region; the corner is a rounded elbow of the
virtual wire itself):

\medskip
\begin{tenkzfree}[tensor style=box]
  \tnput{b1}{(0,0)}{M}
  \tnput{b2}{(11mm,0)}{M}
  \tnput{b3}{(22mm,11mm)}{M}
  \tnput{b4}{(22mm,22mm)}{M}
  \tnjoin{b1.east}{b2.west}
  % the corner: horizontal-then-vertical elbow with the grid's
  % rounded-corner ratio (route=hv)
  \tnjoin[route=hv]{b2.east}{b3.south}
  \tnjoin{b3.north}{b4.south}
  % open chain ends
  \tnjoin{b1.west}{$(b1.west)+(-5mm,0)$}
  \tnjoin{b4.north}{$(b4.north)+(0,5mm)$}
  % physical legs into the region (north on the east-arm, west on
  % the north-arm)
  \tnjoin{b1.north}{$(b1.north)+(0,4.2mm)$}
  \tnjoin{b2.north}{$(b2.north)+(0,4.2mm)$}
  \tnjoin{b3.west}{$(b3.west)+(-4.2mm,0)$}
  \tnjoin{b4.west}{$(b4.west)+(-4.2mm,0)$}
\end{tenkzfree}
