% Regression: t2_gs2d_lasso — renders |psi> = sum_{I, Ibar} A^{I} B^{\bar I} |I>_R |Ibar>_{Rbar}.
% t2_gs2d_lasso — RMP arXiv:2011.12127, file fig4_fig4-pepe_GS2D as it now
% stands: the region lasso.  (The earlier t2_gs2d reproduced the annulus-X
% depiction; the present ink is the region-R form with the boundary pair.)
%
% Formula:  |psi> = sum_{I, Ibar} A^{I} B^{\bar I} |I>_R |Ibar>_{Rbar},
%   where I bundles the physical indices of the sites inside region R, Ibar
%   those of the complement Rbar, and the single contracted wire \partial R
%   is the multi-index of the virtual bonds cut by the region boundary.
%   A^I = contraction of the tensors inside R (its open indices: the bundle
%   I upward, \partial R eastward); B^{\bar I} = contraction of the tensors
%   of Rbar (bundle Ibar upward, \partial R westward).
%
% Ink, LHS: a 4 x 6 oblique PEPS window (boundary legs = virtual bonds cut
% by the window, one up physical leg per site), region R marked over the
% interior 2 x 4 block, complement labelled Rbar.  RHS: the two lasso
% tensors joined by the single \partial R bond; each keeps one up physical
% leg (the bundle) and its outer virtual side is SEALED — per-side policy
% west=none / east=none, not boundary=none.
%
% Noted gaps (cosmetic, structure intact):
%  * the RMP marks R by a filled ellipse; \tnregion traces the rounded
%    rectilinear cell boundary — same cell set, different silhouette;
%  * the Rbar label rides the complement region's slot; the RMP floats it
%    in the exterior whitespace;
%  * the RMP's red-vs-black hue split (R side vs Rbar side) has no per-atom
%    colour key; monochrome here.
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % LHS: the PEPS window with region R over the interior 2 x 4 block and
  % the complement labelled Rbar
  \begin{tenkzlattice}[rows=4, cols=6, frame=oblique, boundary legs,
                       physical=up]
    \tnregion[outline, label=$R$, label at=center, name=R]{(2-3,2-5)}
    \tnregion[slot=complement, label={$\bar{R}$},
              label at=north west]{(1-4,1-6) - R}
  \end{tenkzlattice}
  \;=\;
  % RHS: lasso pair A^I --(\partial R)-- B^Ibar; outer sides sealed by the
  % per-side policy, one physical bundle leg each
  \begin{tenkz}[physical=up, west=none, east=none,
                bond label={$\partial R$ at 1-2}]
    \tn[label pos=south]{A^{I}} & \tn[label pos=south]{B^{\bar I}}
  \end{tenkz}
\]
\end{document}
