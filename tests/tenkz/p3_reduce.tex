% Regression: p3_reduce — renders A_1 = red window (2-4, 2-3) — a 3x2 injective block.
% Formula: A_1 = red window  (2-4, 2-3)  — a 3x2 injective block;
% p3_reduce — arXiv:1804.04964 (normal PEPS FT), proof of Theorem 3, p. 15.
% Region-to-MPS reduction: block the 7x7 PEPS into three injective parts
% around one edge, giving a three-partite MPS.
%
% Formulae:
%   A_1 = red window  (2-4, 2-3)  — a 3x2 injective block;
%   A_2 = blue window (4-5, 4-6)  — a 2x3 injective block;
%   A_3 = the rest of the clip (injective for a PEPS at least 5x7);
%   e   = <(4,3),(4,4)>, the horizontal bond joining A_1 to A_2;
%   |Psi> = trace-closed chain A_1 A_2 A_3 (A_3 touches both windows),
%   physical legs up; the A_1--A_2 bond of the chain is the image of e.
%
% Hue mapping: paper red window -> slot=selected (Action), paper blue
% window -> slot=secondary (Marked), paper fat green edge ->
% distinguished edge (Action).  Noted: the edge and the selected window
% share the Action hue, so the paper's three-hue contrast drops to two.
% Noted gaps: chain atoms carry no slot hue (the paper colours the
% reduced chain's DOTS red/blue), and the chain grammar has no
% distinguished-bond primitive (the paper carries the green edge into
% the A_1--A_2 bond); the labels carry the region hues instead.
\documentclass[varwidth=24cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
$
\vcenter{\hbox{%
\begin{tenkzlattice}[rows=7, cols=7, boundary legs]
  \tnregion[slot=selected, outline]{(2-4,2-3)}
  \tnregion[slot=secondary, outline]{(4-5,4-6)}
  \tnedge{(4,3)-(4,4)}
\end{tenkzlattice}}}
\;\Rightarrow\;
\vcenter{\hbox{$
\begin{tenkz}[periodic, physical=up]
  \tn{{\color{tenkzAction}A_1}} & \tn{{\color{tenkzMarked}A_2}} & \tn{A_3}
\end{tenkz}
$}}
$
\end{document}
