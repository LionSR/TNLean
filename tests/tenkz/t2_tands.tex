% Regression: t2_tands — renders there exist trace-preserving completely positive maps.
% t2_tands — RMP arXiv:2011.12127, Eq. (Fig:TandS), file
% fig2_fig2-pepe_TandS.pdf: characterization of RGFP MPDOs by a pair of
% quantum channels.
%
% Formula:  there exist trace-preserving completely positive maps
% (quantum channels) T and S acting on the physical indices such that
%     T( M )   =  M--M      (T grows one MPDO tensor into two, one
%                            virtual bond contracted), and
%     S( M--M ) =  M        (S shrinks two into one),
% where each M is the 4-leg MPDO tensor (virtual west/east, physical
% up/down).
%
% Declarative reading:
%   * both sides are one-row grids, physical=updown (each M keeps its
%     own up and down physical legs open), open virtual stubs at the
%     word ends;
%   * the T/S pair is CHANNEL APPLICATION between the two pictures, not
%     tensor ink: it is typeset as the math relation
%     \overset{T}{\underset{S}{\leftrightarrows}} (top arrow toward the
%     grown two-tensor word, bottom arrow toward the shrunk word),
%     matching the reference's curved arrow pair up to arrow curvature
%     (a math-symbol choice, not a tenkz gap);
%   * signatures intentionally differ across the relation — it is not a
%     diagram equality: LHS (vw 1, ve 1, pu 2, pd 2), RHS (1,1,1,1).
\documentclass[varwidth=19cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % two-site MPDO word: M--M, all four physical legs open
  \begin{tenkz}[physical=updown, tensor style=box]
    \tn{M} & \tn{M}
  \end{tenkz}
  \;\;\overset{\mathcal{T}}{\underset{\mathcal{S}}{\leftrightarrows}}\;\;
  % one-site MPDO tensor M
  \begin{tenkz}[physical=updown, tensor style=box]
    \tn{M}
  \end{tenkz}
\]
\end{document}
