% Formula (per site, MPO virtual spaces):
%     u_gh(1) . X_gh . [u_g (x) u_h](2) [u_g (x) u_h](3)
%   = u_gh(1) u_gh(2) . X_gh . [u_g (x) u_h](3),
% i.e. X_gh : V_gh -> V_g (x) V_h intertwines the gh MPO with the
% stacked g,h MPOs:  X_gh (u_g (x) u_h) = u_gh X_gh.
% Reference ink: gh / g / h are MPO tensors (red virtual wire through
% the box, black physical wire through it vertically); X_gh is a tall
% box with ONE red wire west (V_gh) and TWO red wires east (V_g, V_h).
%
% Declarative encoding: two operator rows (rows={op:operator,...}) give
% the g-row over the h-row with the shared vertical physical wire; the
% 1-west/2-east split of X_gh is exactly a fusion map, so X_gh is a
% \tnfuse bar with `combined=west`.  The gh MPO tensor must sit ON the
% combined wire (between the two rows), which no cell can do — cells
% live at integer rows — so the gh factor is a separate one-row picture
% juxtaposed on the left; the abutting open stubs read as the operator
% product (the contraction is implied, not drawn).

\usepackage{tenkz}
\begin{document}
\[
  % LHS: gh at site 1, then X_gh, then (g over h) at sites 2 and 3
  \begin{tenkz}[rows={op:operator}, tensor style=box]
    \tn[box]{gh}
  \end{tenkz}
  \!
  \begin{tenkz}[rows={op:operator, op:operator}, tensor style=box]
    \tnfuse[combined=west]{X_{gh}} & \tn[box]{g} & \tn[box]{g}\\
                                   & \tn[box]{h} & \tn[box]{h}
  \end{tenkz}
  \;=\;
  % RHS: gh at sites 1 and 2, then X_gh, then (g over h) at site 3
  \begin{tenkz}[rows={op:operator}, tensor style=box]
    \tn[box]{gh} & \tn[box]{gh}
  \end{tenkz}
  \!
  \begin{tenkz}[rows={op:operator, op:operator}, tensor style=box]
    \tnfuse[combined=west]{X_{gh}} & \tn[box]{g}\\
                                   & \tn[box]{h}
  \end{tenkz}
\]
