% Regression: rgn — renders inset=1 nested boundary over a 3x3 window.
% Formula: inset=1 nested boundary over a 3x3 window
\documentclass[border=6pt]{standalone}
\usepackage{tenkz}
\newcommand{\tenkztextlabel}{%
  \ifmmode
    \PackageError{tenkz-test}{Region text label entered math mode}{}%
  \fi
  text%
}
\newcommand{\tenkzmathlabel}{%
  \ifmmode
    R%
  \else
    \PackageError{tenkz-test}{Region mathematical label left math mode}{}%
  \fi
}
\begin{document}
% region tracer probe, public spelling: an outlined region and its
% inset=1 nested boundary over a 3x3 window.  The two labels also assert
% that the caller, rather than the region renderer, determines math mode.
\begin{tenkzlattice}[rows=3, cols=3]
  \tnregion[slot=selected, label=\tenkztextlabel,
            label pos=north west]{(1-3,1-3)}
  \tnregion[outline, inset=1, label={$\tenkzmathlabel$},
            label pos=south]{(1-3,1-3)}
\end{tenkzlattice}
\quad
\begin{tenkzlattice}[rows=2, cols=2]
  \tnregion[outline, label={$\tenkzmathlabel$},
            label pos=center]{(1-2,1-2)}
\end{tenkzlattice}
\end{document}
