% Formula: s (one per panel):
% T2 sv17 — RMP arXiv:2011.12127, fig. 2 (the SV17 panel): approximating
% a state by an MPS and reading off its local reduced blocks.
%
% Formulas (one per panel):
%   |Phi>    = sum_{s_1..s_9} Phi_{s_1..s_9} |s_1..s_9>
%              — a generic state: ONE tensor, all physical legs open,
%                no virtual structure;
%   |phi_D>  = the bond-dimension-D matrix-product approximation: blocked
%              tensors joined by a truncated virtual bond (the middle of
%              the chain elided by dots);
%   rho_k    = Tr_env |phi_D><phi_D| — the k-site reduced block: ket rail
%              over bra rail, physical indices contracted pairwise (the
%              rungs), virtual ends closed through the environment
%              boundary matrices on the bends.
%
% Declarative reading:
%   panel 1: one glyph wide=9, `legs at` all nine columns, boundary=none
%            (a state has no virtual index);
%   panel 2: two wide=4 glyphs with \tndots eliding the middle;
%   panel 3: the sandwich racetrack — west=cup / east=cup with an
%            on-bend matrix; the label is \vphantom'd because the
%            reference leaves the boundary matrices unnamed.
%
% Noted gaps (see verdict):
%   (1) `wide=` grows only the PILL skin — the box skin ignores it (a
%       wide box renders one-column with floating legs), so the sharp-
%       cornered state boxes of the reference render as pills here;
%   (2) the reference draws the on-bend environment matrices as OPEN
%       RINGS; tenkz's on-bend glyph is the filled fixed-point bead — no
%       ring-skin key for west=cup/east.
\documentclass[varwidth=24cm, border=6pt]{standalone}
\usepackage{amsmath, amssymb}
\usepackage{tenkz}
\begin{document}
\[
  % |Phi>: one 9-leg state tensor
  \underset{\lvert\Phi\rangle}{
  \begin{tenkz}[physical=up, boundary=none, compact]
    \tn[pill, wide=9, legs at={1,2,3,4,5,6,7,8,9}]{} & & & & & & & &
  \end{tenkz}}
  \;\longrightarrow\;
  % |phi_D>: blocked MPS of bond dimension D, middle elided
  \underset{\lvert\phi_D\rangle}{
  \begin{tenkz}[physical=up, boundary=none, compact]
    \tn[pill, wide=4, legs at={1,2,3,4}]{} & & & & \tndots &
    \tn[pill, wide=4, legs at={1,2,3,4}]{} & & &
  \end{tenkz}}
  \;\longrightarrow\;
  % rho_k: k-site racetrack closed on the environment matrices
  \underset{\rho_k}{
  \begin{tenkz}[sandwich, compact, west={cup=$\vphantom{\rho}$},
                east={cup=$\vphantom{\rho}$}]
    \tn{} & \tn{} & \tn{}\\
    \tn*{} & \tn*{} & \tn*{}
  \end{tenkz}}
\]
\end{document}
