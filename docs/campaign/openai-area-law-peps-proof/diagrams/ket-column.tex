% PEPS manuscript, Lemma lem:columns (07-assembly.tex): fixing the input
% side of the compressed operator network sigma to one product-basis
% vector |z> gives the ket v = sigma|z>, with the same virtual bonds.
\[
  \begin{tenkz}[rows={wire,wire}, cols=3, bonds=none]
    \tn[at=(1,1), name=a, skin=mpo, ports={0:virtual, 90:physical, 270:physical}]{\sigma_P}
    \tn[at=(1,2), name=b, skin=mpo, ports={180:virtual, 0:virtual, 90:physical, 270:physical}]{\sigma_Q}
    \tn[at=(1,3), name=c, skin=mpo, ports={180:virtual, 90:physical, 270:physical}]{\sigma_R}
    \tn[at=(2,1), name=za, skin=tri, ports={90:physical}]{z_P}
    \tn[at=(2,2), name=zb, skin=tri, ports={90:physical}]{z_Q}
    \tn[at=(2,3), name=zc, skin=tri, ports={90:physical}]{z_R}
    \tnbond{a.0}{b.180} \tnbond{b.0}{c.180}
    \tnwire{a.270}{za.90} \tnwire{b.270}{zb.90} \tnwire{c.270}{zc.90}
  \end{tenkz}
  \;=\;
  \begin{tenkz}[rows={wire}, cols=3, bonds=none]
    \tn[at=(1,1), name=a, skin=box, ports={0:virtual, 90:physical}]{v_P}
    \tn[at=(1,2), name=b, skin=box, ports={180:virtual, 0:virtual, 90:physical}]{v_Q}
    \tn[at=(1,3), name=c, skin=box, ports={180:virtual, 90:physical}]{v_R}
    \tnbond{a.0}{b.180} \tnbond{b.0}{c.180}
  \end{tenkz}
\]
