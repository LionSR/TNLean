\begin{thebibliography}{9}
	\bibitem{CPSV}
	J. I. Cirac, D. P\'erez-Garc\'ia, N. Schuch, and F. Verstraete,
	Matrix product density operators: Renormalization fixed points and
	boundary theories, Annals of Physics \textbf{378}, 100--149 (2017),
	arXiv:1606.00608v4.
	The general fixed-point characterization assumes canonical form
	and positivity of the generated MPDO family.

	\bibitem{Kato}
	K. Kato, Exact renormalization group flow for matrix product density
	operators, arXiv:2410.22696v1 (2024).
	See the definition of MPDOs admitting an exact RG flow and the
	explicit compression and recovery construction.

	\bibitem{Reutter}
	D. Reutter, Uniqueness of unitary structure for unitarizable fusion
	categories, Communications in Mathematical Physics \textbf{397}
	(2023), arXiv:1906.09710.
	See the proposition in the subsection ``Unitary fusion categories''
	asserting that every positive monoidal auto-equivalence of a unitary
	fusion category is monoidally naturally isomorphic to the identity.

	\bibitem{Reviewed}
	Reviewed project notes, ``Physical question and equivalence conventions''
	and ``Categorical fusion paths and regrouping'',
	\path{docs/audits/rfp_intrinsic/sections/01_scope.tex} and
	\path{docs/audits/rfp_intrinsic/sections/03_categories.tex}.
	The input used here is the full-space raw Fibonacci fusion identity,
	its positive horizontal one-letter closures, and the shifted
	multiplicity spectra.

	\bibitem{Checks}
	Exact finite verification script,
	\path{docs/audits/rfp_intrinsic/checks/next_targets.py}.
	It checks short-length Fibonacci moments, compatible path counts,
	and the Pauli coefficient-kernel obstruction. It does not prove
	all-length positivity or existence of a stationary tensor.

	\bibitem{A4report}
	Project report, ``The exact weighted associator test for
	$\operatorname{Rep}(A_4)$: Classification of the two triplet-channel
	weights'',
	\path{Notes/OpenProblemsTN/checks/round4_p6_a4_associator_report.tex}.
	Its coherent-basis calculation excludes positive nonflat weights
	for the standard associator.
\end{thebibliography}
