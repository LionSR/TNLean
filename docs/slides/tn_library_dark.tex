% Dark-theme loader for TNLean's tensor-network calculus.
%
% The shared TikZ library fixes all geometry, typography, and mathematical
% meaning.  This file supplies only the dark palette and loads the six complete
% diagrams used by the slide collection.
\makeatletter
\def\input@path{{../../tex/tn/}}
\makeatother
\usetikzlibrary{tn}

\colorlet{tnDarkInk}{white!94}
\colorlet{tnDarkSecondaryInk}{white!60}
\colorlet{tnDarkAccent}{orange!80!white}
\colorlet{tnDarkSelected}{cyan!80!white}
\colorlet{tnDarkSecondary}{magenta!75!white}

\tikzset{
  tn theme tensor/.append style={
    draw=tnDarkInk, fill=darkbg, text=tnDarkInk
  },
  tn theme component/.append style={
    draw=tnDarkSecondaryInk, fill=darkbg, text=tnDarkInk
  },
  tn theme factor/.append style={
    draw=tnDarkSecondaryInk, fill=darkbg, text=tnDarkInk
  },
  tn theme map/.append style={
    draw=tnDarkSecondaryInk, fill=darkbg, text=tnDarkInk
  },
  tn theme state/.append style={
    draw=tnDarkSecondaryInk, fill=darkbg, text=tnDarkInk
  },
  tn theme expression/.append style={
    draw=tnDarkSecondaryInk, fill=darkbg, text=tnDarkInk
  },
  tn theme insertion/.append style={
    draw=tnDarkAccent, fill=darkbg, text=tnDarkAccent
  },
  tn theme junction/.append style={fill=tnDarkInk},
  tn theme virtual wire/.append style={draw=tnDarkInk},
  tn theme physical wire/.append style={draw=tnDarkInk},
  tn theme factor region/.append style={draw=tnDarkSecondaryInk},
  tn theme grouping region/.append style={draw=tnDarkSecondaryInk},
  tn theme annotation/.append style={draw=tnDarkSecondaryInk},
  tn theme label/.append style={text=tnDarkSecondaryInk},
  tn selected path/.append style={draw=tnDarkSelected},
  tn secondary path/.append style={draw=tnDarkSecondary},
  tn region selected/.append style={
    draw=tnDarkSelected, fill=tnDarkSelected!16!darkbg
  },
  tn region secondary/.append style={
    draw=tnDarkSecondary, fill=tnDarkSecondary!14!darkbg
  },
  tn region label selected/.append style={text=tnDarkSelected},
  tn region label secondary/.append style={text=tnDarkSecondary}
}

\input{tn_slide_catalogue}
