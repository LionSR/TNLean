% A common periodic MPS word.  Omitted sites are represented by two typed
% external terminals and an ellipsis; the trace is a port-to-port closure.
\newcommand{\SlideTNPeriodicMPS}[2]{%
  \begin{tikzpicture}[tn picture]
    \TN@labeledmpssite{sltnAone}{(0,0)}{#1}
    \TN@labeledmpssite{sltnAtwo}{(1.30,0)}{#1}
    \TN@labeledmpssite{sltnApen}{(3.25,0)}{#1}
    \TN@labeledmpssite{sltnAlast}{(4.55,0)}{#1}
    \TN@vconnectports{sltnAoneE}{sltnAtwoW}
    \TN@vopenport{sltnAtwoE}{0.38,0}
    \TN@vopenport{sltnApenW}{-0.38,0}
    \TN@vconnectports{sltnApenE}{sltnAlastW}
    \TN@popenport{sltnAoneN}{0,0.48}
    \TN@popenport{sltnAtwoN}{0,0.48}
    \TN@popenport{sltnApenN}{0,0.48}
    \TN@popenport{sltnAlastN}{0,0.48}
    \TN@label{sltnAoneNOpen}{(0,0.22)}{i_1}
    \TN@label{sltnAtwoNOpen}{(0,0.22)}{i_2}
    \TN@label{sltnApenNOpen}{(0,0.22)}{i_{#2-1}}
    \TN@label{sltnAlastNOpen}{(0,0.22)}{i_{#2}}
    \TN@vtraceportsbelow{sltnAoneW}{sltnAlastE}{0.42}{0.78}
    \node[tn label] at (2.28,0) {$\cdots$};
    \node[tn label] at (2.28,-1.02) {trace; bond dimension $D$};
  \end{tikzpicture}%
}
