% A common periodic MPS word.  Omitted sites are represented by two typed
% external terminals and an ellipsis; the trace is a port-to-port closure.
\newcommand{\SlideTNPeriodicMPS}[2]{%
  \begin{TNDiagram}[normal]
    \foreach \name/\x/\i in {One/0/i_1,Two/1.30/i_2,
        Pen/3.25/i_{#2-1},Last/4.55/i_{#2}} {
      \TNMPSSite{sltnA\name}{(\x,0)}{#1}
      \TNOpenPhysicalNorth{sltnA\name Ket}
      \node[tn label, above] at (sltnA\name Ket) {$\i$};
    }
    \TNConnectVirtual{sltnAOneEast}{sltnATwoWest}
    \TNOpenVirtualEast{sltnATwoEast}
    \TNOpenVirtualWest{sltnAPenWest}
    \TNConnectVirtual{sltnAPenEast}{sltnALastWest}
    \TNTraceVirtualBelow{sltnAOneWest}{sltnALastEast}
    \node[tn label] at (2.28,0) {$\cdots$};
    \node[tn label] at (2.28,-1.02) {trace; bond dimension $D$};
  \end{TNDiagram}%
}

% The left-hand tensor is B, matching B^i = X A^i X^{-1}.
\newcommand{\SlideTNGaugeConjugation}[3]{%
  \begin{TNDiagram}[normal]
    \TNMPSSite{sltnB}{(0,0)}{#1}
    \TNOpenVirtualWest{sltnBWest}
    \TNOpenVirtualEast{sltnBEast}
    \TNOpenPhysicalNorth{sltnBKet}
    \node[tn label, above] at (sltnBKet) {$i$};
    \node[tn label, text=white, font=\Large] at (1.35,0) {$=$};
    \TNInsertion{sltnX}{(2.35,0)}{#3}
    \TNMPSSite{sltnA}{(3.55,0)}{#2}
    \TNInsertion{sltnXi}{(4.80,0)}{#3^{-1}}
    \TNVirtualPort{sltnXWest}{sltnX}{west}
    \TNVirtualPort{sltnXEast}{sltnX}{east}
    \TNVirtualPort{sltnXiWest}{sltnXi}{west}
    \TNVirtualPort{sltnXiEast}{sltnXi}{east}
    \TNOpenVirtualWest{sltnXWest}
    \TNConnectVirtual{sltnXEast}{sltnAWest}
    \TNConnectVirtual{sltnAEast}{sltnXiWest}
    \TNOpenVirtualEast{sltnXiEast}
    \TNOpenPhysicalNorth{sltnAKet}
    \node[tn label, above] at (sltnAKet) {$i$};
  \end{TNDiagram}%
}
