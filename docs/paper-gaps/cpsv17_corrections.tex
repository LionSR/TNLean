\title{Corrections to matrix product density operator fixed-point statements}
\begin{abstract}
	We examine several fixed-point, convergence, and entropy statements in
	Cirac--\allowbreak P\'erez-Garc\'ia--\allowbreak Schuch--\allowbreak Verstraete.
	The counterexamples concern the simple-MPDO implication chain in
	Theorem~\ref{cpsv-thm:main-simple}, the pure-state zero-correlation-length
	equivalences, convergence of canonical-form renormalization flows,
	the thermodynamic limit in Proposition~\ref{cpsv-PropILILp1}, and the
	finite-ring saturation claim in Example~\ref{cpsv-ExSALnoZCL}.
	For the map from two to three sites asserted in Proposition~\ref{cpsv-prop2to5},
	an explicit tensor satisfies all standing hypotheses but has
	$\|K_2(Y)\|_1=337/250<27/20=\|K_3(Y)\|_1=\|K_4(Y)\|_1$,
	excluding any such trace-preserving completely positive map by trace-norm contraction.
	We distinguish these counterexamples from a proof completion for the
	fundamental theorem and from local corrections. The pure
	limit/fixed-point characterization itself remains valid.
\end{abstract}

\section{Scope and conventions}\label{sec:scope}

The statements examined in Ref.~\cite{Cirac2017MPDO_AnnPhys} concern
three distinct questions: fixed-point characterizations, existence of
limits, and exact finite-ring entropy identities.
Table~\ref{tab:source-map} identifies each source claim and its treatment
here. The first counterexample concerns Proposition~\ref{cpsv-prop2to5}
and Theorem~\ref{cpsv-thm:main-simple}; the later counterexamples do not
depend on that example. Section~\ref{sec:matching} instead completes a
proof without changing its intended conclusion, and
Sec.~\ref{sec:local} collects local corrections.
Source references use arXiv:1606.00608v4 numbering;
unqualified equation references are local.

For an MPDO tensor $K$ with physical dimension $d$, bond dimension
$D$, and boundary matrix $X\in M_D(\mathbb C)$, define
\begin{align}
	K_N(X)
	                & =\sum_{\boldsymbol i,\boldsymbol j}
	\tr\!\left(XK^{i_1j_1}\cdots K^{i_Nj_N}\right)
	\ket{\boldsymbol i}\bra{\boldsymbol j},
	\label{eq:closure}                                    \\
	\sigma^{(N)}(K) & =K_N(\Id_D).\nonumber
\end{align}
Here $M_D(\mathbb C)$ denotes the $D\times D$ matrices. The tensor
$K$ generates translationally invariant operators $\sigma^{(N)}(K)\geq0$.
Mixed-state zero correlation length (ZCL) uses the first of two distinct
transfer objects:
\begin{align}
	B_K                    & =\sum_iK^{ii},\qquad
	\mc E_K(H)=\sum_{i,j}K^{ij}H(K^{ij})^\dagger,
	\label{eq:transfer-objects}                   \\
	K\text{ has mixed ZCL} & \iff B_K^2=B_K.
	\label{eq:mixed-zcl-def}
\end{align}
A renormalization fixed point (RFP) requires trace-preserving
completely positive maps (tpCPMs):
\begin{align}
	\exists\,\mc T,\mc S\text{ tpCPM}:\quad
	\begin{cases}
		\mc T[K_1(X)]=K_2(X), \\
		\mc S[K_2(X)]=K_1(X)
	\end{cases}
	\quad\forall X\in M_D(\mathbb C).
	\label{eq:rfp-def}
\end{align}

We use the canonical form (CF), normal tensors, and basis
of normal tensors (BNT) as defined in Ref.~\cite{Cirac2017MPDO_AnnPhys}.
For normal representatives $\mc K_a$, the relevant conditions are
\begin{align}
	 & \{\mc K_a^{ij}\}_{i,j}\text{ irreducible},\qquad
	r(\mc E_{\mc K_a})=1,\nonumber                                       \\
	 & \operatorname{spec}(\mc E_{\mc K_a})\cap\{|z|=1\}=\{1\},\nonumber \\
	 & \text{biCF}:\quad
	\operatorname{span}_{i,j}\{\oplus_a\mc K_a^{ij}\}
	=\oplus_a M_{D_a}(\mathbb C),\nonumber                               \\
	 & 0<|\mu_{a,q}|\leq1,\qquad \max_{a,q}|\mu_{a,q}|=1,\nonumber       \\
	 & \text{simple}:\quad
	B_{\mc K_a}=\sum_i\mc K_a^{ii}\text{ is not nilpotent}
	\quad\forall a.
	\label{eq:canonical-conditions}
\end{align}
Here $r$ is the spectral radius, biCF means block-injective
canonical form, and $\mu_{a,q}$ are the canonical copy weights.
Only active weights are retained. The unit-modulus weight is
required globally, not in each sector.

Entropies use base-two logarithms and the normalized periodic state:
\begin{align}
	\widehat\sigma^{(N)}=\frac{\sigma^{(N)}}{\tr\sigma^{(N)}},
	\qquad
	I_L^{(N)}=S_L^{(N)}+S_{N-L}^{(N)}-S_N^{(N)}.
	\label{eq:entropy-sal-def}
\end{align}
Here $S_L^{(N)}$ is the entropy of a connected $L$-site block.
Saturation of the area law (SAL) means
$I_L^{(N)}=I_1^{(N)}$ for $1\leq L\leq\lfloor N/2\rfloor$.
Entropy normalization does not rescale $K$ in a channel identity.

\section{Local maps for simple tensors with saturation and zero correlation length}\label{sec:blocking}

In Appendix~C.2 of Ref.~\cite{Cirac2017MPDO_AnnPhys}, the standing setup
considers a simple tensor $K$ in block-injective canonical form (biCF) that
satisfies saturation of the area law (SAL) and zero correlation length (ZCL).
Under these assumptions, Proposition~\ref{cpsv-prop2to5} asserts that there
exist two trace-preserving completely positive maps (tpCPMs) $\mc T$ and $\mc S$
acting on the physical indices,
\begin{align}
	\mc T & : M_{d^2}(\mathbb C) \to M_{d^3}(\mathbb C), \nonumber \\
	\mc S & : M_{d^3}(\mathbb C) \to M_{d^2}(\mathbb C),
	\label{eq:TS-domains}
\end{align}
such that for every boundary matrix $X\in M_D(\mathbb C)$,
\begin{align}
	\mc T[K_2(X)] = K_3(X), \qquad \mc S[K_3(X)] = K_2(X).
	\label{eq:C15-statement}
\end{align}
This is the identity represented in diagram~\eqref{cpsv-3to5tcpm}.
Theorem~\ref{cpsv-thm:main-simple} then defines $M$ as the two-site block
of $K$ [Eq.~\eqref{cpsv-K=MM}] and concludes that $M$ is a renormalization
fixed point (RFP) via the implication chain (ii)~$\Rightarrow$~(v).
By Definition~\ref{cpsv-RFPMixedTS}, an RFP map for $M$ would require a
tpCPM mapping $M_1(X) = K_2(X)$ to $M_2(X) = K_4(X)$ for all
$X\in M_D(\mathbb C)$.

The following construction satisfies all standing hypotheses of
Appendix~C.2, but directly tests and excludes these maps.

\subsection{The four-letter tensor}

For $i=0,1,2,3$, set
\begin{align}
	t      & =(1,2,-7,4)^{\mathsf T},\qquad
	s=(-3,-1,1,3)^{\mathsf T},\nonumber                   \\
	l_i    & =\begin{pmatrix}1\\ t_i\end{pmatrix},\qquad
	r_i=\begin{pmatrix}1/4&s_i/100\end{pmatrix},\nonumber \\
	F^{ii} & =F_i=l_ir_i,\qquad F^{ij}=0\quad(i\ne j).
	\label{eq:slices}
\end{align}
Thus $d_F=4$ and $D_F=2$. With $L=(l_0,\ldots,l_3)$ and
$R$ having rows $r_i$,
\begin{align}
	\sum_it_i=\sum_is_i=\sum_is_it_i & =0,\qquad
	LR=P=\diag(1,0),\label{eq:LR}                   \\
	T:=RL                            & =\frac1{100}
	\begin{pmatrix}
		22 & 19 & 46 & 13 \\
		24 & 23 & 32 & 21 \\
		26 & 27 & 18 & 29 \\
		28 & 31 & 4  & 37
	\end{pmatrix}.
	\label{eq:T}
\end{align}
In particular,
\begin{align}
	T_{ij}        & =\frac14+\frac{s_it_j}{100}>0,\qquad
	\boldsymbol1=(1,1,1,1)^{\mathsf T},\nonumber         \\
	T\boldsymbol1 & =\boldsymbol1,\qquad
	\boldsymbol1^{\mathsf T}T=\boldsymbol1^{\mathsf T}.
	\label{eq:T-stochastic}
\end{align}
Writing $Q=\boldsymbol1\boldsymbol1^{\mathsf T}/4$ and
$N_0=st^{\mathsf T}/100$ gives
\begin{align}
	T   & =Q+N_0,\quad Q^2=Q,\quad QN_0=N_0Q=N_0^2=0,\nonumber \\
	T^2 & =Q,\qquad T^m=Q\quad(m\geq2).
	\label{eq:Tpowers}
\end{align}
Rank-one multiplication yields
\begin{align}
	F_{i_1}\cdots F_{i_N}
	 & =l_{i_1}\left(\prod_{a=1}^{N-1}T_{i_ai_{a+1}}\right)r_{i_N},
	\label{eq:rankone}                                              \\
	\tr(F_{i_1}\cdots F_{i_N})
	 & =\prod_{a=1}^NT_{i_ai_{a+1}}>0,\qquad i_{N+1}=i_1.\nonumber
\end{align}
Consequently, without rescaling,
\begin{align}
	\sigma^{(N)}(F)\geq0,\qquad
	\tr\sigma^{(N)}(F) & =\tr T^N=1\quad(N\geq1),\nonumber \\
	B_F=LR             & =P=P^2.
	\label{eq:F-state}
\end{align}

\subsection{Exact finite-ring SAL}

The periodic distribution and its proper connected marginals are
\begin{align}
	p_N(i_1,\ldots,i_N) & =\prod_{a=1}^NT_{i_ai_{a+1}},\nonumber       \\
	p_{N,L}(i_1,\ldots,i_L)
	                    & =\left(\prod_{a=1}^{L-1}T_{i_ai_{a+1}}\right)
	                       (T^{N-L+1})_{i_Li_1}\nonumber \\
	                    & =\frac14\prod_{a=1}^{L-1}T_{i_ai_{a+1}},
	\qquad 1\leq L<N.
	\label{eq:marginal}
\end{align}
The complementary exponent $N-L+1\geq2$ makes
\cref{eq:Tpowers} applicable even when one site is traced out.
Define
\begin{align}
	h=-\frac14\sum_{i,j}T_{ij}\log_2T_{ij}.
	\label{eq:h-entropy}
\end{align}
The initial letter is uniform and every internal edge has
probability $T_{ij}/4$, so
\begin{align}
	S_L^{(N)} & =2+(L-1)h\qquad(1\leq L<N),\nonumber                  \\
	p_N(i_a=i,i_{a+1}=j)
	          & =T_{ij}(T^{N-1})_{ji}=T_{ij}/4\quad(N\geq3),\nonumber \\
	S_N^{(N)} & =Nh\qquad(N\geq3),\nonumber                           \\
	I_L^{(N)} & =4-2h\qquad(N\geq3,\ 1\leq L<N).
	\label{eq:SL-IL}
\end{align}
This proves finite-ring SAL; at $N=2$ there is only one
nonempty cut size.

\subsection{Normal form and the global canonical weight}

The coordinate matrix of the four slices, ordered as
$(11,12,21,22)$, satisfies
\begin{align}
	\det\left[\frac14,\frac{s_i}{100},\frac{t_i}{4},\frac{s_it_i}{100}\right]_{i=0}^3 & =-\frac3{2000}\ne0,\nonumber \\
	\operatorname{span}\{F_i\}                                                        & =M_2(\mathbb C).
	\label{eq:span-M2-det}
\end{align}
Hence $\mc E_F$ is strictly positive:
\begin{align}
	v^\dagger\mc E_F(H)v
	=\sum_i\|H^{1/2}F_i^\dagger v\|^2>0
	\quad(H\geq0,\ H\ne0,\ v\ne0).
	\label{eq:EF-strictly-positive}
\end{align}
Equality would force $H^{1/2}A^\dagger v=0$ for every
$A\in M_2(\mathbb C)$, whereas
$\{A^\dagger v:A\in M_2(\mathbb C)\}=\mathbb C^2$.
Perron--Frobenius theory therefore gives positive left and right
eigenmatrices, unique up to positive scalar multiples:
\begin{align}
	\mc E_F(\rho)=r\rho,\qquad
	\mc E_F^*(H_*)=rH_*,
	\qquad \rho,H_*>0,\quad r>0,
	\label{eq:perron-evals}
\end{align}
with no other peripheral eigenvalue; $\mc E_F^*$ is the trace adjoint.

For $H=\diag(20,1)$,
\begin{align}
	H-\mc E_F^*(H)
	                     & =\begin{pmatrix}85/8&-9/40\\-9/40&4697/5000\end{pmatrix}>0,\nonumber \\
	\det(H-\mc E_F^*(H)) & =\frac{19861}{2000}>0.
	\label{eq:H-contraction-val}
\end{align}
Pairing with $\rho$ shows
\begin{align}
	0<\tr[(H-\mc E_F^*(H))\rho]
	=(1-r)\tr(H\rho),\qquad 0<r<1.
	\label{eq:r-strictly-less-1}
\end{align}
Put $\mu=\sqrt r$, $G=H_*^{1/2}$, and
$\mc K_0^{ii}=\mu^{-1}GF_iG^{-1}$. Then
\begin{align}
	\sum_i(\mc K_0^{ii})^\dagger\mc K_0^{ii}
	=r^{-1}G^{-1}\mc E_F^*(H_*)G^{-1}=\Id_2,
	\label{eq:K0-normal}
\end{align}
so $\mc K_0$ is normal.

Embed its physical support in the first four coordinates of
$\mathbb C^5$, and take the bond-one normal tensor
$\mc K_1^{44}=1$ on the fifth coordinate, with other slices zero.
The required ambient tensor is
\begin{align}
	K=(\mu\mc K_0)\oplus\mc K_1,\qquad d_K=5,\quad D_K=3.
	\label{eq:ambient}
\end{align}
Its simultaneous canonical conditions are
\begin{align}
	 & \operatorname{span}_{i,j}
	\{\mc K_0^{ij}\oplus\mc K_1^{ij}\}
	=M_2(\mathbb C)\oplus\mathbb C,\nonumber                       \\
	 & (\mu_{0,1},\mu_{1,1})=(\mu,1),\qquad
	0<\mu<1=\max_{j,q}|\mu_{j,q}|,\nonumber                        \\
	 & B_{\mc K_0}=\mu^{-1}GPG^{-1},\qquad B_{\mc K_1}=1,\nonumber \\
	 & B_{\mc K_j}^{\,n}\ne0\qquad(j=0,1,\ n\geq1).
	\label{eq:ambient-canonical-conds}
\end{align}
Orthogonal physical supports make these representatives a BNT;
the span and nonnilpotence prove biCF and simplicity.

For the coordinate inclusion $V:\mathbb C^4\to\mathbb C^5$,
\begin{align}
	\sigma^{(N)}(K)
	                   & =V^{\otimes N}\sigma^{(N)}(F)(V^{\otimes N})^\dagger
	+\ketbra{4^N}{4^N}\geq0,\nonumber                                           \\
	\tr\sigma^{(N)}(K) & =2\qquad(N\geq1),\nonumber                             \\
	S_L^{(N)}(K)       & =1+\tfrac12S_L^{(N)}(F)\qquad(1\leq L\leq N),\nonumber \\
	I_L^{(N)}(K)       & =1+\tfrac12I_L^{(N)}(F)\qquad(1\leq L<N),\nonumber     \\
	B_K                & =(GPG^{-1})\oplus1,\qquad B_K^2=B_K.
	\label{eq:ambient-verified}
\end{align}
Thus positivity of generated states, global weight normalization,
biCF, simplicity, finite-ring SAL, and literal ZCL hold together.

\subsection{A boundary excluding trace-preserving completely positive maps}

Choose
\begin{align}
	X=\begin{pmatrix}-1&2/5\\-5&0\end{pmatrix}.
	\label{eq:boundary-X}
\end{align}
The endpoint coefficients in \cref{eq:rankone} are
\begin{align}
	D_{ij} & =r_jXl_i=-\frac14+\frac{t_i}{10}-\frac{s_j}{20},\nonumber \\
	D      & =\begin{pmatrix}
		          0    & -1/10 & -1/5  & -3/10  \\
		          1/10 & 0     & -1/10 & -1/5   \\
		          -4/5 & -9/10 & -1    & -11/10 \\
		          3/10 & 1/5   & 1/10  & 0
	          \end{pmatrix}.
	\label{eq:D-matrix}
\end{align}
Although $X$ need not be Hermitian, every closure is real diagonal:
\begin{align}
	F_N(X)
	             & =\sum_{\boldsymbol i}
	D_{i_1i_N}\prod_{a=1}^{N-1}T_{i_ai_{a+1}}
	\ketbra{\boldsymbol i}{\boldsymbol i}
	=F_N(X)^\dagger,\nonumber                         \\
	\|F_N(X)\|_1 & =\sum_{i,j}|D_{ij}|(T^{N-1})_{ij}.
	\label{eq:norm}
\end{align}
The displayed matrices give
\begin{align}
	\|F_2(X)\|_1                  & =\frac{337}{250}
	<\frac{27}{20}=\|F_3(X)\|_1=\|F_4(X)\|_1,\nonumber \\
	\frac{27}{20}-\frac{337}{250} & =\frac1{500}>0.
	\label{eq:norm-gap}
\end{align}
Use the same ambient boundary at every length:
\begin{align}
	Y      & =(GXG^{-1})\oplus0,\nonumber                 \\
	K_N(Y) & =V^{\otimes N}F_N(X)(V^{\otimes N})^\dagger.
	\label{eq:boundary}
\end{align}
Cyclicity cancels the gauges, the zero boundary removes the
fifth-letter sector, and the isometry preserves the trace norm.

Every positive trace-preserving map contracts that norm on
Hermitian inputs. For $Z=Z_+-Z_-$,
\begin{align}
	\|\Phi(Z)\|_1
	\leq\tr\Phi(Z_+)+\tr\Phi(Z_-)
	=\tr Z_++\tr Z_-=\|Z\|_1.
	\label{eq:contraction-prop}
\end{align}
Thus the failed conclusions are
\begin{align}
	\nexists\,\Phi\text{ tpCPM}:\quad
	\Phi[K_2(Y)]=K_m(Y),
	\qquad m=3,4.
	\label{eq:no-channel-m34}
\end{align}
The $m=3$ inequality refutes the all-boundary identity of
Proposition~\ref{cpsv-prop2to5}. If $M$ denotes the two-site block of $K$,
\begin{align}
	M_1(Y)=K_2(Y),\qquad M_2(Y)=K_4(Y),
	\qquad M\text{ is not an RFP}.
	\label{eq:M-block-id}
\end{align}
so $M$ is not an RFP. This also refutes Theorem~\ref{cpsv-thm:main-simple}(ii)$\Rightarrow$(v).
The obstruction excludes every trace-preserving completely positive map,
not only the proposed construction based on local factorizations. It makes no
claim about every larger blocking or about Theorem~\ref{cpsv-thm:IV.13}.

\subsection{Two points in the structural proof}

The matrix $T$ also exhibits the missing Jordan-block issue in
source Lemma~\ref{cpsv-SALZCL}:
\begin{align}
	T                      & =Q+N_0>0,\qquad N_0\ne0,\quad N_0^2=0,\nonumber     \\
	\operatorname{spec}(T) & =\{0,1\},\qquad \tr T^n=1\quad(n\geq1),\nonumber    \\
	T_{00}T_{11}-T_{01}T_{10}
	                       & =\frac{22\cdot23-19\cdot24}{10000}=\frac1{200}\ne0.
	\label{eq:T-jordan}
\end{align}
Here $T>0$ means entrywise positivity. Primitivity and traces of
powers do not imply rank one; the nilpotent part is invisible
in those traces.

Common-weight absorption after source Lemma~\ref{cpsv-lemmus}
has a separate normalization problem:
\begin{align}
	\mc E_{\mu_j\mc K_j}=|\mu_j|^2\mc E_{\mc K_j},
	\qquad
	r(\mc E_{\mu_j\mc K_j})=|\mu_j|^2.
	\label{eq:weight-absorption}
\end{align}
For a normal $\mc K_j$, absorption preserves normality only when
$|\mu_j|=1$. In \cref{eq:ambient},
$r(\mc E_{\mu\mc K_0})=r<1$.
These are defects in the structural argument. The refutation of
Proposition~\ref{cpsv-prop2to5} does not require a universal claim
about all possible physical-sector factorizations.

\section{Raw weights and peripheral phases}\label{sec:weights-phases}

\subsection{Physical ZCL does not determine raw weights}

For pure tensors, the source uses correlations independent of
distance (CID) and local orthogonality (LO):
\begin{align}
	 & \text{pure ZCL}\iff(\text{CID}\ \text{and}\ \text{LO}),\nonumber \\
	 & \text{LO}:\ \sum_iA_j^i\otimes\overline{A_{j'}^i}=0
	\quad(j\ne j').
	\label{eq:pure-zcl-lo}
\end{align}
This differs from mixed ZCL, $B_K^2=B_K$.
The pure transfer map and its canonical copy blocks are
\begin{align}
	\mc E_A(H) & =\sum_iA^iH(A^i)^\dagger,\qquad
	\mathbb E_A=\sum_iA^i\otimes\overline{A^i},\nonumber \\
	(\mathbb E_A)_{j,q;j',q'}
	           & =\mu_{j,q}\overline{\mu_{j',q'}}
	\sum_iA_j^i\otimes\overline{A_{j'}^i},
	\label{eq:pure-transfer-blocks}
\end{align}
in canonical coordinates. Literal idempotence retains the copy factors.

Take one physical letter:
\begin{align}
	A^0                   & =\diag(1,1/2),\qquad A_1^0=1,\nonumber \\
	(\mu_{1,1},\mu_{1,2}) & =(1,1/2),\nonumber                     \\
	\ket{V^{(N)}(A)}      & =(1+2^{-N})\ket{0^N}.
	\label{eq:single-letter-A}
\end{align}
This tensor is in CF; LO is vacuous. All physical observables
are scalar, so for observables $O_1,O_2$ on disjoint regions and
every translation $\tau_\Delta$ allowed by CID,
\begin{align}
	\langle V^{(N)}|O_1O_2|V^{(N)}\rangle
	=\langle V^{(N)}|O_1\tau_\Delta(O_2)|V^{(N)}\rangle.
	\label{eq:cid-scalar}
\end{align}
CID holds even for unnormalized expectations at fixed $N$, but
for the matrix unit $e_{12}$,
\begin{align}
	\mc E_A(e_{12})=\tfrac12e_{12},\qquad
	\mc E_A^2(e_{12})=\tfrac14e_{12}.
	\label{eq:transfer-non-idempotent}
\end{align}
Thus pure ZCL does not imply idempotence in source
Theorem~\ref{cpsv-TheoremZCLPure}, or the unrestricted equivalence
of Theorem~\ref{cpsv-thm:main-MPS}. Directly, the RFP equation gives
\begin{align}
	(A^0)^2=uA^0
	\quad\Longrightarrow\quad u=1,\quad \tfrac14=\tfrac12,
	\label{eq:rfp-pure-contradiction}
\end{align}
a contradiction.

As an MPDO tensor, $K^{00}=\diag(1,1/2)$ independently refutes
Theorem~\ref{cpsv-thm:main-simple}(iv)$\Rightarrow$(v):
\begin{align}
	 & \mc K_1^{00}=1,\qquad
	\operatorname{span}\{\mc K_1^{00}\}=\mathbb C,\qquad
	B_{\mc K_1}=1,\nonumber              \\
	 & \max_q|\mu_{1,q}|=1,\qquad
	\sigma^{(N)}(K)=1+2^{-N}>0,\nonumber \\
	 & \eta=1,\quad a=b=1.
	\label{eq:K00-props}
\end{align}
These are simplicity, biCF, global normalization, positivity,
and the complete scalar factorization in condition~(iv).
Trace preservation for its two-site block would require
\begin{align}
	(K^{00})^4=(K^{00})^2,\qquad
	\diag(1,1/16)=\diag(1,1/4),
	\label{eq:K00-four-two}
\end{align}
which is false. Condition~(iv) constrains normal representatives,
not raw copy weights. Here $B_K^2\ne B_K$; this is not the
counterexample to Proposition~\ref{cpsv-prop2to5}.

\subsection{Peripheral phases and similarity under squaring}

Let
\begin{align}
	\omega=e^{2\pi i/3},\qquad
	A^0=\diag(1,\omega),\qquad
	S=\begin{pmatrix}0&1\\1&0\end{pmatrix}.
	\label{eq:omega-A0}
\end{align}
Both scalar normal copies have unit-modulus weights, yet
\begin{align}
	\mc E_A^{2^n}(e_{21})
	=\omega^{2^n}e_{21}
	=\begin{cases}
		 \omega e_{21},  & n\text{ even}, \\
		 \omega^2e_{21}, & n\text{ odd}.
	 \end{cases}
	\label{eq:phase-cycle}
\end{align}
This refutes the universal CF convergence assertion following
source Theorem~\ref{cpsv-thm:renormalization-flow} and in Appendix~B.

It also refutes the pure-state equivalence asserted after source
Eq.~\eqref{cpsv-RFP-gauge}:
\begin{align}
	\omega S(A^0)^2S^{-1} & =\omega\diag(\omega^2,1)=A^0,\nonumber               \\
	M                     & =A^0\otimes\overline{A^0},\nonumber                  \\
	M                     & =(S\otimes\overline S)M^2(S\otimes\overline S)^{-1},
	\qquad M^2\ne M.
	\label{eq:gauge-blocking-tensor}
\end{align}
The doubled tensor is in CF, with scalar normal copies and
\begin{align}
	(\mu_q)_q=(1,\overline\omega,\omega,1),
	\qquad \rho^{(N)}(M)=|1+\omega^N|^2>0.
	\label{eq:doubled-weights}
\end{align}
Identity maps on the one-dimensional physical algebras give
blocking up to virtual gauge in both directions, but not an RFP
in the sense of Definition~\ref{cpsv-RFPMixedTS}. Similarity under
squaring allows root-of-unity cycles, not only spectrum in $\{0,1\}$.

Theorem~\ref{cpsv-thm:renormalization-flow} itself survives as a
limit characterization:
\begin{align}
	\mc E^{2^n}\longrightarrow F
	      & \quad\Longrightarrow\quad
	F^2=\lim_n\mc E^{2^{n+1}}=F,\nonumber        \\
	F^2=F & \quad\Longrightarrow\quad F^{2^n}=F.
	\label{eq:limit-characterization}
\end{align}
The correspondence between completely positive maps and physical
isometry classes of Kraus tensors gives the stated characterization.
It does not establish convergence for every CF initial tensor.

\section{Finite-chain entropy and thermodynamic limits}\label{sec:entropy}

\subsection{The inner thermodynamic limit}

Consider the CF tensor
\begin{align}
	M^{00} & =\diag(1,0,0),           &
	M^{11} & =\diag(0,1,-1),\nonumber                \\
	M^{01} & =0,                      & M^{10} & =0.
	\label{eq:parity-M}
\end{align}
Its scalar normal copies have globally normalized weights $1,1,-1$.
Mixed words vanish, giving
\begin{align}
	\rho^{(N)}
	              & =\ketbra{0^N}{0^N}+(1+(-1)^N)\ketbra{1^N}{1^N}\geq0,\nonumber \\
	\tr\rho^{(N)} & =2+(-1)^N.
	\label{eq:rho-parity}
\end{align}
At odd length, the normalized state and all nonempty marginals
are pure; at even length, their nonzero eigenvalues are $1/3,2/3$.
With $h_2(p)=-p\log_2p-(1-p)\log_2(1-p)$,
\begin{align}
	I_L^{(N)}
	=\begin{cases}
		 0,        & N\text{ odd}, \\
		 h_2(1/3), & N\text{ even}
	 \end{cases}
	\quad(N>L\geq1).
	\label{eq:IL-parity}
\end{align}
Therefore
\begin{align}
	\liminf_{N\to\infty}I_L^{(N)}=0
	<h_2(1/3)=\limsup_{N\to\infty}I_L^{(N)}
	\qquad(L\geq1),
	\label{eq:oscillation-limit}
\end{align}
so the unrestricted inner limit in source
Proposition~\ref{cpsv-PropILILp1} does not exist.

The finite-chain statements remain separate. Strong subadditivity
on consecutive lengths $1,L,N-2L-1$, with translation invariance,
gives
\begin{align}
	S_L^{(N)}+S_{N-L}^{(N)}
	          & \leq S_{L+1}^{(N)}+S_{N-L-1}^{(N)},\nonumber \\
	I_L^{(N)} & \leq I_{L+1}^{(N)}
	\qquad(1\leq L<\lfloor N/2\rfloor).
	\label{eq:finite-monotonicity}
\end{align}
The source proof also quotes $I_L^{(N)}\leq4\log_2D$.
These bounds do not imply convergence in $N$. If the fixed-$L$
limits exist, they do imply
\begin{align}
	J_L:=\lim_{N\to\infty}I_L^{(N)}
	                     & \quad\Longrightarrow\quad
	J_L\leq J_{L+1}\leq4\log_2D,\nonumber            \\
	\lim_{L\to\infty}J_L & \leq4\log_2D.
	\label{eq:JL-limits}
\end{align}

\subsection{The literal periodic state in Example~\ref{cpsv-ExSALnoZCL}}

In source Example~\ref{cpsv-ExSALnoZCL}, site $n$ is in
$\ket{k_n,k_n}$. The nonzero spectrum is the periodic distribution
\begin{align}
	p_N(k_1,\ldots,k_N) & =\frac1{Z_N}\prod_{n=1}^NW_{k_nk_{n+1}},
	\qquad k_{N+1}=k_1,\nonumber                                     \\
	W                   & =\begin{pmatrix}1&1/2\\1/2&1\end{pmatrix}.
	\label{eq:W-def}
\end{align}
With
$P_\pm=\frac12(\begin{smallmatrix}1&\pm1\\\pm1&1\end{smallmatrix})$,
\begin{align}
	W^m & =(3/2)^mP_++(1/2)^mP_-,\nonumber       \\
	Z_N & =\tr W^N=(3^N+1)/2^N,\nonumber         \\
	p_{N,L}(k_1,\ldots,k_L)
	    & =\frac{\prod_{n=1}^{L-1}W_{k_nk_{n+1}}
			       (W^{N-L+1})_{k_Lk_1}}{Z_N}\nonumber \\
	    & \hspace{3em}(1\leq L<N).
	\label{eq:W-marginals}
\end{align}
The complementary path is essential on a finite ring.

At $N=4$,
\begin{align}
	Z_4 & =\frac{41}{8},\qquad
	W^2=\begin{pmatrix}5/4&1\\1&5/4\end{pmatrix},\nonumber \\
	W^3 & =\begin{pmatrix}7/4&13/8\\13/8&7/4\end{pmatrix}.
	\label{eq:W-powers}
\end{align}
The exact marginal probabilities are
\begin{align}
	p_{4,1}(0)=p_{4,1}(1)   & =\tfrac12,\nonumber      \\
	p_{4,2}(00)=p_{4,2}(11) & =\frac{14}{41},\nonumber \\
	p_{4,2}(01)=p_{4,2}(10) & =\frac{13}{82},
	\label{eq:p41-p42}
\end{align}
\begin{align}
	 & p_{4,3}(k_1,k_2,k_3)\nonumber                               \\
	 & =\begin{cases}
		    10/41, & 0\text{ internal transitions}\ (2\text{ words}), \\
		    4/41,  & 1\text{ internal transition}\ (4\text{ words}),  \\
		    5/82,  & 2\text{ internal transitions}\ (2\text{ words}).
	    \end{cases}
	\label{eq:p43}
\end{align}
Their entropies give
\begin{align}
	I_2-I_1 & =2S_2-S_1-S_3\nonumber \\
	        & =\frac1{82}\log_2
	\frac{41^{82}5^{50}}{2^{48}13^{52}7^{112}}>0.
	\label{eq:I2-I1-defect}
\end{align}
Strictness has a short conditional-MI proof. Translation invariance
identifies the defect with $I(k_1:k_3\mid k_2)$, and
\begin{align}
	q_{ab} & :=\Pr(k_1=a,k_3=b\mid k_2=0)\nonumber                  \\
	       & =\frac1{41}\begin{pmatrix}20&8\\8&5\end{pmatrix}_{ab},
	\qquad \det q=\frac{100-64}{41^2}>0.
	\label{eq:q-conditional}
\end{align}
Thus $q$ is not the product $p$ of its marginals. For positive
probability distributions,
\begin{align}
	\sum_aq_a\ln(q_a/p_a)
	 & \geq\sum_a(q_a-p_a)=0,\nonumber \\
	 & \text{equality iff }q=p,
	\label{eq:kl-positivity}
\end{align}
by $-\ln x\geq1-x$. The conditional mutual information is
strictly positive, so the literal four-ring fails SAL.

The non-ZCL claim remains correct:
\begin{align}
	p_{4,2}(00)=\frac{14}{41}
	\ne\frac5{14}
	=\frac{(W^2)_{00}}{\tr W^3}=p_{3,2}(00),
	\quad 196\ne205.
	\label{eq:non-zcl-check}
\end{align}
Literal physical-trace idempotence would identify these reduced
closures and their normalizations. A common rescaling of $W$
changes neither probabilities nor entropies. This removes the
printed finite-periodic SAL witness, not the independence of SAL
and ZCL.

\section{A coefficient-separation argument for the fundamental theorem}\label{sec:matching}

The proof of source Theorem~\ref{cpsv-thm1} compares one $B$-sector
with the $A$-family, although the proportionality relation contains
all $B$-sectors. Full-family coefficient comparison supplies the
step and preserves that theorem and Corollary~\ref{cpsv-II_cor2}.
Retain only active, nonzero canonical weights; zero-weight blocks
carry no dimension information.

For distinct nonzero $z_1,\ldots,z_m$,
\begin{align}
	 & \sum_{a=1}^mc_az_a^N=0\quad(N\geq N_0),\nonumber        \\
	 & \det[z_a^{N_0+b}]_{\substack{0\leq b<m\\1\leq a\leq m}}
	=\left(\prod_{a=1}^mz_a^{N_0}\right)
	\prod_{a<b}(z_b-z_a)\ne0\nonumber                          \\
	 & \quad\Longrightarrow\quad c_a=0\quad\forall a.
	\label{eq:vandermonde}
\end{align}
After equal weights are grouped, a nonempty copy power sum has
positive integer multiplicities. It therefore cannot vanish on
an entire tail.

Form the union of both BNT families, quotienting first by
\begin{align}
	C\sim D
	 & \quad\Longleftrightarrow\quad
	\exists\,\phi\in\mathbb R,\ X\text{ invertible}:\nonumber \\
	 & D^i=e^{i\phi}XC^iX^{-1}\quad\forall i.
	\label{eq:bnt-equivalence}
\end{align}
Choose one representative $C_u$ per class. The source normal-tensor
overlap lemma gives
\begin{align}
	G_{uv}^{(N)}
	 & :=\langle V^{(N)}(C_u)|V^{(N)}(C_v)\rangle
	\longrightarrow\delta_{uv},\nonumber          \\
	 & \det G^{(N)}\ne0\quad(N\geq N_0).
	\label{eq:gram-matrix}
\end{align}
Hence the entire quotient family is jointly linearly independent
on a tail.

Absorb the matching phases into the coefficients and write
\begin{align}
	\ket{V^{(N)}(A)} & =\sum_u a_u(N)\ket{V^{(N)}(C_u)},\nonumber \\
	\ket{V^{(N)}(B)} & =\sum_u b_u(N)\ket{V^{(N)}(C_u)}.
	\label{eq:mpv-expansion}
\end{align}
Then the proportionality hypothesis gives the full-family comparison
\begin{align}
	 & \ket{V^{(N)}(A)}=c_N\ket{V^{(N)}(B)},\quad c_N\ne0\nonumber \\
	 & \Longrightarrow\quad
	a_u(N)=c_Nb_u(N)\quad\forall u,\ N\geq N_0.
	\label{eq:proportional-expansion}
\end{align}
Choosing an unmatched $A$-sector as its class representative would give
\begin{align}
	b_u(N)=0,\qquad
	a_u(N)=\sum_q\mu_{j,q}^N=0\quad(N\geq N_0),
	\label{eq:unmatched-sector}
\end{align}
contradicting the determinant argument. Interchanging the families
excludes unmatched $B$-sectors. Thus phase-gauge matching is
bijective, without a unit-weight assumption in each sector.

For equal MPVs, choose matched representatives
$B_j^i=e^{i\phi_j}Y_jA_j^iY_j^{-1}$. Coefficient comparison gives
\begin{align}
	\sum_q\mu_{j,q}^N
	                & =\sum_q(e^{i\phi_j}\nu_{j,q})^N\qquad(N\geq N_0),\nonumber \\
	\{\mu_{j,q}\}_q & =\{e^{i\phi_j}\nu_{j,q}\}_q
	\quad\text{as multisets}.
	\label{eq:multiset-matching}
\end{align}
The second line follows by grouping distinct nonzero weights and
applying the determinant argument to their multiplicity differences.
The sums then agree at every positive length; matching copies
gives the global similarity in Corollary~\ref{cpsv-II_cor2}.

For proportional families, $c_N$ remains in every sector identity.
The conclusion is sector matching, not copy-weight recovery.

\section{Local corrections}\label{sec:local}

\subsection{CID requires positive complementary gaps}

The forward implication from transfer idempotence to printed CID
has an independent adjacency obstruction. Take
\begin{align}
	A^{(a,b)}=\frac1{\sqrt2}\ketbra{a}{b},
	\qquad a,b\in\{0,1\}.
	\label{eq:bell-tensor}
\end{align}
Then
\begin{align}
	\operatorname{span}\{A^{(a,b)}\} & =M_2(\mathbb C),\nonumber   \\
	\mc E_A(H)                       & =\tfrac12\tr(H)\Id_2,\qquad
	\mc E_A^2=\mc E_A.
	\label{eq:bell-transfer}
\end{align}
This is a single normal RFP with no copy weights. Its periodic
MPS is a product of Bell pairs between the right qubit of each
site and the left qubit of the next.

On a four-ring, let $Z_{n,r}$ and $Z_{n,l}$ denote Pauli $Z$
on the respective qubits of site $n$. With
$\ket{\varphi}=(\ket{00}+\ket{11})/\sqrt2$,
\begin{align}
	\langle Z_{1,r}Z_{2,l}\rangle
	 & =\bra{\varphi}Z\otimes Z\ket{\varphi}=1,\nonumber \\
	\langle Z_{1,r}Z_{3,l}\rangle
	 & =\left[\tfrac12(1-1)\right]^2=0.
	\label{eq:bell-correlations}
\end{align}
The printed CID interval permits this translation. Thus
idempotence does not imply CID for arbitrary disjoint,
possibly adjacent regions.

Let $g_1,g_2$ and $g'_1,g'_2$ be the complementary gap lengths
before and after translation. The corrected forward statement
requires
\begin{align}
	g_1,g_2,g'_1,g'_2\geq1,\quad \mathbb E^2=\mathbb E\nonumber \\
	\Longrightarrow\quad
	\mathbb E^{g_1}=\mathbb E^{g_2}
	=\mathbb E^{g'_1}=\mathbb E^{g'_2}=\mathbb E.
	\label{eq:positive-gaps}
\end{align}
At adjacency, $\mathbb E^0=\Id$ instead. This is a positive-gap
correction, not a copy-weight normalization.

\subsection{Square-root normalization}

In source Theorem~\ref{cpsv-thm:charact-MPS} and
Lemma~\ref{cpsv-lem:charact-NT-pure-RFP}, the pair-index isometry
condition implies
\begin{align}
	\sum_iU^i_{ab}\overline{U^i_{a'b'}}
	=\delta_{aa'}\delta_{bb'}
	\quad\Longrightarrow\quad
	\sum_iU^iH(U^i)^\dagger=\Id\,\tr H.
	\label{eq:isometry-trace}
\end{align}
For a trace-normalized diagonal density fixed point,
\begin{align}
	 & \Lambda>0,\quad\tr\Lambda=1,\nonumber       \\
	 & \begin{cases}
		   A^i=\sqrt\Lambda U^i
		    & \Rightarrow\ \mc E_A(H)=\Lambda\tr H,   \\
		   A^i=\Lambda U^i
		    & \Rightarrow\ \mc E_A(H)=\Lambda^2\tr H.
	   \end{cases}
	\label{eq:sqrt-comparison}
\end{align}
The printed second form has spectral radius $\tr(\Lambda^2)$;
for bond dimension $D>1$, positivity and $\tr\Lambda=1$ give
$\tr(\Lambda^2)<1$. Including the gauge, the corrected forms are equivalently
\begin{align}
	A^i & =X\sqrt\Lambda U^iX^{-1},\quad\tr\Lambda=1,\nonumber \\
	\text{or}\qquad
	A^i & =X\Lambda U^iX^{-1},\quad\tr(\Lambda^2)=1.
	\label{eq:sqrt-corrected}
\end{align}
In the first convention, the bond state has amplitudes
$\sqrt{\Lambda_{aa}}$; in the second, $\Lambda$ is the
Schmidt-amplitude matrix. Neither convention removes the
independent copy-phase obstruction.

\subsection{Indices, exponents, and endpoints}

Source Eq.~\eqref{cpsv-AA=A} must leave $i_1,i_2$ free and sum
only over $j$:
\begin{align}
	A^{i_1}A^{i_2}=\sum_jU_{(i_1,i_2),j}A^j.
	\label{eq:corrected-AA=A}
\end{align}
In ``Commuting parent Hamiltonians'', the length-$L$ support
vectors are
\begin{align}
	\ket{v_{\alpha,\beta}}
	=\sum_{i_1,\ldots,i_L}
	 (A^{i_1}\cdots A^{i_L})_{\alpha,\beta}
	\ket{i_1,\ldots,i_L}.
	\label{eq:corrected-parent-support}
\end{align}

In the notation of Appendix~C's proof of
Theorem~\ref{cpsv-thm:IV.13},
\begin{align}
	A_\gamma      & =\frac{m_\gamma}{n_\gamma}
	U_\gamma M_\gamma U_\gamma^\dagger\nonumber \\
	\Longrightarrow\quad
	O_L(A_\gamma) & =
	\left(\frac{m_\gamma}{n_\gamma}\right)^LO_L(M_\gamma).
	\label{eq:corrected-OL-scaling}
\end{align}
Here $O_L$ is the vertically contracted length-$L$ operator,
and $m_\gamma=\tr\mu_\gamma$, $n_\gamma=\tr\nu_\gamma$ are
the traces of the one-site and blocked multiplicity matrices.
Thus the first simultaneous blocked-operator representation has
coefficient
\begin{align}
	\left(\frac{m_\gamma}{n_\gamma}\right)^L\tr(\nu_\gamma^L),
	\qquad\text{not}\qquad
	\frac{m_\gamma^L}{n_\gamma}\tr(\nu_\gamma^L),
	\label{eq:corrected-coefficient}
\end{align}
as in the later trace-power expression of that proof.

The entropy equality following SAL is
\begin{align}
	I_L=I_{L+1}
	\quad\Longleftrightarrow\quad
	S_L+S_{N-L}=S_{L+1}+S_{N-L-1}.
	\label{eq:corrected-sal-entropy}
\end{align}
The closing proof of Theorem~\ref{cpsv-thm:main-simple}
cyclically interchanges conditions~(iii) and~(iv) in its
proposition assignments; Proposition~\ref{cpsv-prop2to5}
starts from condition~(ii). Correct labels do not repair the
false implications above.

Finally, source Example~\ref{cpsv-ExZCLnoSAL} requires
\begin{align}
	U_n=\sigma_x^{(n,l)}\otimes\sigma_x^{(n,r)},\qquad
	\mc E_n(X)=pU_nXU_n^\dagger+(1-p)X.
	\label{eq:corrected-En-qubits}
\end{align}
The exception list for strict failure of SAL must include $p=1$:
\begin{align}
	p=1\quad\Longrightarrow\quad
	S_L=2\ (1\leq L<N),\qquad
	S_N=0,\qquad I_L=4.
	\label{eq:p1-exception}
\end{align}
Indeed, deterministic on-site unitary conjugation preserves the
Bell-pair ring entropies, with two maximally mixed boundary qubits
at every proper connected cut. The non-SAL witness at $p=1/4$
is unaffected.
