% General helper to import an external paper's aux file with a given prefix.
% Tries relative paths from docs/paper-gaps/, from repo root, and local directory.
\newcommand{\importpaperaux}[2]{%
  \IfFileExists{../../Papers/#2.aux}{%
    \externaldocument[#1]{../../Papers/#2}%
  }{%
    \IfFileExists{Papers/#2.aux}{%
      \externaldocument[#1]{Papers/#2}%
    }{%
      \IfFileExists{#2.aux}{%
        \externaldocument[#1]{#2}%
      }{}%
    }%
  }%
}
