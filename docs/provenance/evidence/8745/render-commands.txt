Portable focused-render commands (run from a checkout root with the recorded tools and pinned tenkz configured):

python3 docs/provenance/evidence/8745/prepare_render.py --root . --out build/8745
cd build/8745/blueprint/src
latexmk -xelatex -interaction=nonstopmode -halt-on-error print.tex
python3 - <<'PY'
from pathlib import Path
import re
numbers = iter(range(1, 100))
Path('web.bbl').write_text(re.sub(r'\\bibitem\{', lambda _: r'\bibitem[' + str(next(numbers)) + ']{', Path('print.bbl').read_text()))
PY
plastex -c plastex.cfg web.tex
pdftotext -layout print.pdf ../../print.txt
mkdir -p ../../pdf-pages
pdftoppm -r 120 -png print.pdf ../../pdf-pages/page
cd ../../../..
python3 docs/provenance/evidence/8745/verify_render.py --root . --out build/8745
